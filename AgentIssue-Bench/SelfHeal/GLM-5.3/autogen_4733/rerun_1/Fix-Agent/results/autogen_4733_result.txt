def with_requirements(
    python_packages: Sequence[str] = [], global_imports: Sequence[Import] = []
) -> Callable[[Callable[P, T]], FunctionWithRequirements[T, P]]:
    """Decorate a function with package and import requirements

    Args:
        python_packages (List[str], optional): Packages required to function. Can include version info.. Defaults to [].
        global_imports (List[Import], optional): Required imports. Defaults to [].

    Returns:
        Callable[[Callable[P, T]], FunctionWithRequirements[T, P]]: The decorated function
    """

    def wrapper(func: Callable[P, T]) -> FunctionWithRequirements[T, P]:
        func_with_reqs = FunctionWithRequirements(
            python_packages=python_packages, global_imports=global_imports, func=func
        )

        functools.update_wrapper(func_with_reqs, func)
        return func_with_reqs

    return wrapper

### 2. Root-Cause Correctness

The bug: `@dataclass` with default `eq=True, frozen=False` sets `__hash__ = None`, so `build_python_functions_file`'s `global_imports: Set[Import]` + `set.update(func.global_imports)` raises `TypeError: unhashable type: 'Alias'/'ImportFromModule'` — exactly what the reproduction script detects via `"unhashable type" in str(e)`.

The fix is the canonical remedy:
- **`Alias.__hash__`** → `hash((self.name, self.alias))` — both fields are strings; perfectly consistent with the dataclass-generated `__eq__`.
- **`ImportFromModule.__hash__`** → `hash((self.module, tuple(self.imports)))` — the `list→tuple` conversion is required (lists are unhashable), and each element is either a `str` or a now-hashable `Alias`. Hash/eq contract holds: equal lists compare element-wise equal, and equal `Alias` elements now produce equal hashes.

**Key semantic check**: defining `__hash__` in the dataclass *body* works — the dataclasses action table maps `(unsafe_hash=False, eq=True, frozen=False, has_explicit_hash=True)` → "leave alone", so the decorator does **not** overwrite it with `None`. (The docs confirm: "If the class already defines `__hash__()`, `unsafe_hash` is ignored.") No `frozen=True` or `unsafe_hash=True` needed.

**Deprecated-module copy**: `@deprecated` (typing_extensions / `warnings.deprecated` in 3.13+) wraps the already-`@dataclass`-decorated class (subclass or `__new__` patch); `__hash__`/`__eq__` are inherited/untouched, so the deprecated classes remain hashable too.

### 3. Structural Validation (`validate_fixed_code`)

Result: **ACCEPT** — "every file is present once with its full text". No structural issues.

    ) -> FunctionWithRequirements[T, P]:
        return cls(python_packages=python_packages, global_imports=global_imports, func=func)

    @staticmethod
    def from_str(
        func: str, python_packages: Sequence[str] = [], global_imports: Sequence[Import] = []
    ) -> FunctionWithRequirementsStr:
        return FunctionWithRequirementsStr(func=func, python_packages=python_packages, global_imports=global_imports)

    # Type this based on F
    def __call__(self, *args: P.args, **kwargs: P.kwargs) -> T:
        return self.func(*args, **kwargs)


def with_requirements(
    python_packages: Sequence[str] = [], global_imports: Sequence[Import] = []
) -> Callable[[Callable[P, T]], FunctionWithRequirements[T, P]]:
    """Decorate a function with package and import requirements

    Args:
        python_packages (List[str], optional): Packages required to function. Can include version info.. Defaults to [].
        global_imports (List[Import], optional): Required imports. Defaults to [].

    Returns:
        Callable[[Callable[P, T]], FunctionWithRequirements[T, P]]: The decorated function
    """

    def wrapper(func: Callable[P, T]) -> FunctionWithRequirements[T, P]:
        func_with_reqs = FunctionWithRequirements(
            python_packages=python_packages, global_imports=global_imports, func=func
        )

        functools.update_wrapper(func_with_reqs, func)
        return func_with_reqs

    return wrapper


def build_python_functions_file(
    funcs: Sequence[Union[FunctionWithRequirements[Any, P], Callable[..., Any], FunctionWithRequirementsStr]],
) -> str:
    """:meta private:"""
    # First collect all global imports
    global_imports: Set[Import] = set()
    for func in funcs:
        if isinstance(func, (FunctionWithRequirements, FunctionWithRequirementsStr)):
            global_imports.update(func.global_imports)

    content = "\n".join(map(_import_to_str, global_imports)) + "\n\n"

    for func in funcs:
        content += _to_code(func) + "\n\n"

    return content


def to_stub(func: Union[Callable[..., Any], FunctionWithRequirementsStr]) -> str:
    """Generate a stub for a function as a string

    Args:
        func (Callable[..., Any]): The function to generate a stub for

    Returns:
        str: The stub for the function
    """
    if isinstance(func, FunctionWithRequirementsStr):
        return to_stub(func.compiled_func)

    content = f"def {func.__name__}{inspect.signature(func)}:\n"
    docstring = func.__doc__

    if docstring:
        docstring = dedent(docstring)
        docstring = '"""' + docstring + '"""'
        docstring = indent(docstring, "    ")
        content += docstring + "\n"

    content += "    ..."
    return content

def test_alias_and_import_from_module_are_hashable() -> None:
    alias = Alias(name="polars", alias="pl")
    same_alias = Alias(name="polars", alias="pl")

    assert alias == same_alias
    assert hash(alias) == hash(same_alias)
    assert alias in {same_alias}

    import_from_module = ImportFromModule(module="polars", imports=["DataFrame", Alias(name="Series", alias="S")])
    same_import_from_module = ImportFromModule(
        module="polars", imports=["DataFrame", Alias(name="Series", alias="S")]
    )

    assert import_from_module == same_import_from_module
    assert hash(import_from_module) == hash(same_import_from_module)
    assert import_from_module in {same_import_from_module}

''',
        global_imports=[

def test_build_python_functions_file_deduplicates_imports() -> None:
    func_a = FunctionWithRequirements.from_str(
        '''
def a() -> None:
    pass
''',
        global_imports=[ImportFromModule(module="os", imports=["path"])],
    )
    func_b = FunctionWithRequirements.from_str(
        '''
def b() -> None:
    pass
''',
        global_imports=[ImportFromModule(module="os", imports=["path"])],
    )

    content = build_python_functions_file([func_a, func_b])
    assert content.count("from os import path") == 1
