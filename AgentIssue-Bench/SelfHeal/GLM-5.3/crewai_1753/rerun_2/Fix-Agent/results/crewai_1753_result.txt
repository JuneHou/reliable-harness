import json
from datetime import date, datetime
from decimal import Decimal
from enum import Enum
from typing import Any
from uuid import UUID

from pydantic import BaseModel


class CrewJSONEncoder(json.JSONEncoder):
    def default(self, obj):
        if isinstance(obj, BaseModel):
            return self._handle_pydantic_model(obj)
        elif isinstance(obj, UUID) or isinstance(obj, Decimal):
            return str(obj)
        elif isinstance(obj, Enum):
            return self._handle_enum(obj)

        elif isinstance(obj, datetime) or isinstance(obj, date):
            return obj.isoformat()

        return super().default(obj)

    def _handle_enum(self, obj: Enum) -> Any:
        """Convert an Enum member to a JSON serializable value.

        Prefers the enum's value when it is natively serializable, otherwise
        falls back to its string representation.
        """
        value = obj.value
        if isinstance(value, (str, int, float, bool)) or value is None:
            return value
        return str(value)

    def _handle_pydantic_model(self, obj):
        try:
            data = obj.model_dump()
            # Remove circular references
            for key, value in data.items():
                if isinstance(value, BaseModel):
                    data[key] = str(
                        value
                    )  # Convert nested models to string representation
                elif isinstance(value, Enum):
                    data[key] = self._handle_enum(
                        value
                    )  # Convert enum values to their underlying values
            return data
        except RecursionError:
            return str(
                obj
            )  # Fall back to string representation if circular reference is detected
