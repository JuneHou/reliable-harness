  splitArray,

import { DEFAULT_ADA_CONFIG } from "@evo-ninja/agents";
import { ERROR_SUBSIDY_MAX } from "./errors";

const MAX_RATE_LIMIT_RETRIES = 5;

export class ProxyEmbeddingApi implements EmbeddingApi {
  private _goalId: string | undefined = undefined;

  constructor(
    private _tokenizer: Tokenizer,
    private _setCapReached: () => void
  ) {}

  public setGoalId(goalId: string | undefined) {
    this._goalId = goalId;
  }

  public async createEmbeddings(
    input: string | string[],
    tries?: number
  ): Promise<EmbeddingCreationResult[]> {
    const config = DEFAULT_ADA_CONFIG;

    const inputs = Array.isArray(input) ? input : [input];
    for (const input of inputs) {
      if (this._tokenizer.encode(input).length > config.maxTokensPerInput) {
        throw new Error(
          `Input exceeds max request tokens: ${config.maxTokensPerInput}`
        );
      }
    }

    const batchedInputs = splitArray(inputs, config.maxInputsPerRequest);

    const goalId = this._goalId;
    if (!goalId) {
      throw Error("GoalID is not set");
    }

    let embeddings: EmbeddingCreationResult[] = [];

    // Send each batch in a separate request so that the request body
    // never exceeds the maximum size allowed by the embeddings proxy API
    for (const batch of batchedInputs) {
      const batchEmbeddings = await this.createEmbeddingsBatch(
        batch,
        goalId,
        config.model,
        tries
      );
      embeddings = embeddings.concat(batchEmbeddings);
    }

    return embeddings;
  }

  private async createEmbeddingsBatch(
    inputs: string[],
    goalId: string,
    model: string,
    tries?: number
  ): Promise<EmbeddingCreationResult[]> {
    const embeddingResponse = await fetch("/api/proxy/embeddings", {
      method: "POST",
      body: JSON.stringify({
        input: inputs,
        model,
        goalId,
      }),
      headers: {
        "Content-Type": "application/json",
      },
    });

    if (!embeddingResponse.ok) {
      let error: any;
      try {
        error = await embeddingResponse.json();
      } catch (e) {
        // The error response is not always JSON (e.g. "Body exceeds
        // maximum size" errors returned by the hosting platform)
        error = { error: embeddingResponse.statusText };
      }

      if (embeddingResponse.status === 400) {
        throw Error(
          "Error from OpenAI Embeddings: " +
            (error.error ?? embeddingResponse.statusText)
        );
      }
      if (embeddingResponse.status === 429) {
        await new Promise((resolve) => setTimeout(resolve, 15000));
        if (!tries || tries < MAX_RATE_LIMIT_RETRIES) {
          return this.createEmbeddingsBatch(
            inputs,
            goalId,
            model,
            tries == undefined ? 0 : ++tries
          );
        }
      }
      if (embeddingResponse.status === 403) {
        this._setCapReached();
        throw Error(ERROR_SUBSIDY_MAX);
      }
      throw Error("Error trying to get response from embeddings proxy.", error);
    }

    const { embeddings } = await embeddingResponse.json();
    return embeddings;
  }

import type { NextApiRequest, NextApiResponse } from "next";
import OpenAIApi from "openai";
import { authOptions } from "../auth/[...nextauth]";
import { getServerSession } from "next-auth";
import { isGoalValid } from "../../../api-utils/goal";
import { canUseSubsidy } from "../../../api-utils/subsidy";
import { createSupabaseClient } from "../../../api-utils/supabase";
import { createOpenAIApiClient } from "../../../api-utils/openai";

export default async function handler(
  req: NextApiRequest,
  res: NextApiResponse
): Promise<void> {
  if (!(req.method === "POST")) {
    return res.status(404).send({});
  }
  const goalId = req.body.goalId;

  // Ensure the user is logged in
  const session = await getServerSession(req, res, authOptions);
  const email = session?.user?.email;
  if (!session || !email) {
    return res.status(401).send({});
  }

  const supabase = createSupabaseClient();
  const openai = createOpenAIApiClient();

  // Ensure the goal is being tracked, and we're able to continue
  // subsidizing the goal's llm requests
  const isValid = await isGoalValid(goalId, supabase);
  if (!isValid) {
    return res.status(403).send({});
  }
  const canSubsidize = await canUseSubsidy("embedding", goalId, supabase);
  if (!canSubsidize) {
    return res.status(403).send({});
  }

  const input: string[] | string[][] = req.body.input;

  if (!Array.isArray(input) || input.length === 0) {
    return res.status(400).json({ error: "Invalid embeddings input." });
  }

  // The client sends a single batch of inputs per request so that the
  // request body stays below the maximum size allowed by the proxy API.
  // Batched input (string[][]) is still accepted for backwards
  // compatibility.
  const batches: string[][] = Array.isArray(input[0])
    ? (input as string[][])
    : [input as string[]];

  try {
    const embeddings: {
      embedding: number[];
      input: string;
    }[] = [];

    // Process the batches sequentially to avoid exceeding the
    // OpenAI rate limits
    for (const inputs of batches) {
      const { data } = await openai.embeddings.create({
        model: req.body.model,
        input: inputs,
      });

      embeddings.push(
        ...data.map((innerData) => {
          return {
            embedding: innerData.embedding,
            input: inputs[innerData.index],
          };
        })
      );
    }

    return res.status(200).json({ embeddings });
  } catch (e: any) {
    // Rate limit error
    if (e instanceof OpenAIApi.APIError && e.status === 429) {
      return res.status(429).send({});
    }
    // Bad request error (e.g. an input exceeds the model's limits)
    if (e instanceof OpenAIApi.APIError && e.status === 400) {
      return res.status(400).json({ error: e.message });
    }
    return res.status(500).send({});
  }
}

import OpenAIApi from "openai";
import { ILogger } from "@evo-ninja/agent-utils";

export const DEFAULT_ADA_CONFIG = {
  model: "text-embedding-ada-002",
  maxTokensPerInput: 8191,
  maxInputsPerRequest: 2048
}

export class OpenAIEmbeddingAPI implements EmbeddingApi {
  private api: OpenAIApi;

  constructor(
    private apiKey: string,
    private logger: ILogger,
    private tokenizer: Tokenizer,
    private modelConfig: {
      model: string,
      maxTokensPerInput: number,
      maxInputsPerRequest: number
    } = DEFAULT_ADA_CONFIG,
    private _maxRateLimitRetries: number = 5
  ) {
    this.api = new OpenAIApi({
      apiKey: this.apiKey,
      dangerouslyAllowBrowser: true
    });
  }

  async createEmbeddings(input: string | string[], tries?: number): Promise<EmbeddingCreationResult[]> {
    try {
      const inputs = Array.isArray(input) ? input : [input];
      this.validateInput(inputs);

      const batchedInputs = splitArray(inputs, this.modelConfig.maxInputsPerRequest);

      const results = await Promise.all(batchedInputs.map(async (inputs) => {
        const { data } = await this.api.embeddings.create({
          model: this.modelConfig.model,
          input: inputs,
        })

        return data.map((innerData) => {
          return {
            embedding: innerData.embedding,
            input: inputs[innerData.index]
          }
        });
      }))

      return results.flat();
    } catch (err) {
      const error = cleanOpenAIError(err);

      if (typeof error === "object") {
        const maybeOpenAiError = error as Partial<OpenAIError>;

        if (maybeOpenAiError.status === 429) {
          this.logger.warning("Warning: OpenAI rate limit exceeded, sleeping for 15 seconds.");

          await new Promise((resolve) => setTimeout(resolve, 15000));

          if (!tries || tries < this._maxRateLimitRetries) {
            return this.createEmbeddings(
              input,
              tries === undefined ? 0 : ++tries
            );
          }
        }
      }

      throw new Error(JSON.stringify(error, null, 2));
    }
  };

  validateInput(inputs: string[]) {
    for (const input of inputs) {
      if (this.tokenizer.encode(input).length > this.modelConfig.maxTokensPerInput) {
        throw new Error(`Input exceeds max request tokens: ${this.modelConfig.maxTokensPerInput}`);
      }
    }
  }
}

    const { chunkLength, overlap } = opts

    while (startIndex < text.length) {
      let endIndex = startIndex + chunkLength;
      let chunk = text.slice(startIndex, endIndex);

      chunks.push(chunk);

      startIndex = endIndex - overlap;
    }

import {
  EmbeddingApi,
  EmbeddingCreationResult,
  Tokenizer,
  splitArray,
} from "@evo-ninja/agents";
import { DEFAULT_ADA_CONFIG } from "@evo-ninja/agents";
import { ERROR_SUBSIDY_MAX } from "./errors";

const MAX_RATE_LIMIT_RETRIES = 5;

// The proxy is a Next.js API route which enforces a maximum request body
// size. OpenAI's embedding limits (maxInputsPerRequest = 2048 inputs,
// maxTokensPerInput = 8191 tokens) allow bodies that are far larger than
// what the proxy accepts, so the payload must additionally be bounded by
// the total amount of text sent in a single request.
// 300k characters of JSON-encoded text keeps us safely below the
// configured body size limit (see next.config.js).
const MAX_CHARACTERS_PER_REQUEST = 300_000;

interface ProxyEmbeddingApiOptions {
  maxRequestRetries?: number;
  maxCharsPerRequest?: number;
}

export class ProxyEmbeddingApi implements EmbeddingApi {
  private _goalId: string | undefined = undefined;
  private _maxRequestRetries: number;
  private _maxCharsPerRequest: number;

  constructor(
    private _tokenizer: Tokenizer,
    private _setCapReached: () => void,
    options?: ProxyEmbeddingApiOptions
  ) {
    this._maxRequestRetries = options?.maxRequestRetries ?? MAX_RATE_LIMIT_RETRIES;
    this._maxCharsPerRequest =
      options?.maxCharsPerRequest ?? MAX_CHARACTERS_PER_REQUEST;
  }

  public setGoalId(goalId: string | undefined) {
    this._goalId = goalId;
  }

  public async createEmbeddings(
    input: string | string[],
    tries?: number
  ): Promise<EmbeddingCreationResult[]> {
    const config = DEFAULT_ADA_CONFIG;

    const inputs = Array.isArray(input) ? input : [input];
    for (const input of inputs) {
      if (this._tokenizer.encode(input).length > config.maxTokensPerInput) {
        throw new Error(
          `Input exceeds max request tokens: ${config.maxTokensPerInput}`
        );
      }
    }

    // Batch by both the OpenAI limits and a maximum total payload size so
    // the request body never exceeds what the embeddings proxy accepts.
    const batchedInputs = this.batchInputs(
      inputs,
      config.maxInputsPerRequest,
      this._maxCharsPerRequest
    );

    const goalId = this._goalId;
    if (!goalId) {
      throw Error("GoalID is not set");
    }

    // Requests are sent sequentially so that, on rate limits, only the
    // batch(es) that still need to be processed are retried instead of the
    // whole payload (which would recreate an oversized request body).
    const embeddings: EmbeddingCreationResult[] = [];
    for (const batch of batchedInputs) {
      const batchEmbeddings = await this.requestEmbeddings(batch, goalId);
      embeddings.push(...batchEmbeddings);
    }
    return embeddings;
  }

  private async requestEmbeddings(
    inputs: string[],
    goalId: string,
    tries?: number
  ): Promise<EmbeddingCreationResult[]> {
    const config = DEFAULT_ADA_CONFIG;

    const embeddingResponse = await fetch("/api/proxy/embeddings", {
      method: "POST",
      body: JSON.stringify({
        input: [inputs],
        model: config.model,
        goalId
      }),
      headers: {
        "Content-Type": "application/json",
      },
    });

    if (!embeddingResponse.ok) {
      const error = await embeddingResponse.json();
      if (embeddingResponse.status === 400) {
        throw Error("Error from OpenAI Embeddings: " + error.error);
      }
      if (embeddingResponse.status === 429) {
        await new Promise((resolve) => setTimeout(resolve, 15000));
        if (tries === undefined || tries < this._maxRequestRetries) {
          return this.requestEmbeddings(
            inputs,
            goalId,
            tries === undefined ? 0 : ++tries
          );
        }
      }
      if (embeddingResponse.status === 403) {
        this._setCapReached();
        throw Error(ERROR_SUBSIDY_MAX);
      }
      throw Error("Error trying to get response from embeddings proxy.", error);
    }
    const { embeddings } = await embeddingResponse.json();
    return embeddings;
  }

  /**
   * Splits `inputs` into batches that respect both the OpenAI limits and a
   * maximum total character count per batch. Every individual input must
   * still be smaller than `maxCharsPerRequest` (guaranteed here by the
   * earlier token validation, since tokens never exceed characters).
   */
  private batchInputs(
    inputs: string[],
    maxInputsPerRequest: number,
    maxCharsPerRequest: number
  ): string[][] {
    const batches: string[][] = [];
    let currentBatch: string[] = [];
    let currentChars = 0;

    for (const input of inputs) {
      // An input that alone exceeds the character budget starts a new batch
      // on its own (it still fits within the OpenAI token limits, so it is
      // a valid request on its own).
      const inputChars = input.length;
      if (
        currentBatch.length > 0 &&
        (currentBatch.length >= maxInputsPerRequest ||
          currentChars + inputChars > maxCharsPerRequest)
      ) {
        batches.push(currentBatch);
        currentBatch = [];
        currentChars = 0;
      }

      currentBatch.push(input);
      currentChars += inputChars;
    }

    if (currentBatch.length > 0) {
      batches.push(currentBatch);
    }

    return batches;
  }
}

  static fixedCharacterLength(text: string, opts: { chunkLength: number, overlap: number }): string[] {
    const { chunkLength, overlap } = opts

    if (chunkLength <= overlap) {
      throw new Error("Chunk length must be greater than overlap length.");
    }

    let chunks: string[] = [];
    let startIndex = 0;

    while (startIndex < text.length) {
      let endIndex = startIndex + chunkLength;
      let chunk = text.slice(startIndex, endIndex);

      chunks.push(chunk);

      startIndex = endIndex - overlap;
    }

    return chunks;
  }

  static parentDocRetrieval(text: string, opts: {
    parentChunker: (text: string) => string[],
    childChunker: (parentText: string) => string[],
  }): {
    doc: string;
    metadata: {
      parent: string;
    }
  }[] {
    const { parentChunker, childChunker } = opts

    const parentChunks = parentChunker(text)
    const parentAndChildDocs = parentChunks.map(parentChunk => {
      return {
        parent: parentChunk,
        children: childChunker(parentChunk)
      }
    }).map(parentAndChildDoc => {
      return {
        metadatas: parentAndChildDoc.children.map(() => ({ parent: parentAndChildDoc.parent })),
        documents: parentAndChildDoc.children
      }
    })

    const docs = parentAndChildDocs.flatMap(parentAndChildDoc => parentAndChildDoc.documents)
    const metadatas = parentAndChildDocs.flatMap(parentAndChildDoc => parentAndChildDoc.metadatas)

    const chunks = docs.map((doc, index) => {
      return {
        doc,
        metadata: metadatas[index]
      }
    });

    return chunks;
  }

  static sentences(text: string): string[] {
    const alphabets = "([A-Za-z])";
    const prefixes = "(Mr|St|Mrs|Ms|Dr)[.]";
    const suffixes = "(Inc|Ltd|Jr|Sr|Co)";

    const digits = "([0-9])";
    const multipleDots = /\\.{2,}/g;

      text = " " + text + "  ";
      text = text.replaceAll("\n", " ");
      text = text.replaceAll(new RegExp(prefixes, 'g'), "$1<prd>");
      text = text.replaceAll(new RegExp(websites, 'g'), "<prd>$1");

const MAX_RATE_LIMIT_RETRIES = 5;

// Number of batches sent in parallel when the request has to be split
// across multiple HTTP requests to stay under the proxy's body size limit.
const MAX_CONCURRENT_REQUESTS = 4;

// Conservative safety margin (JSON escaping, base64/unicode expansion, headers).
const BYTE_SIZE_SAFETY_MARGIN = 0.8;

// Default Next.js pages/api body limit is 1MB. We keep every request comfortably
// below it so we never hit "Body Exceeds Maximum Size" from the Embedding Proxy API.
export const MAX_REQUEST_BODY_BYTES = 1024 * 1024;

/**
 * Returns the approximate byte length of a string (UTF-8 aware).
 */
function byteLength(value: string): number {
  if (typeof Buffer !== "undefined" && typeof Buffer.byteLength === "function") {
    return Buffer.byteLength(value, "utf8");
  }
  return new TextEncoder().encode(value).length;
}

export class ProxyEmbeddingApi implements EmbeddingApi {
  private _goalId: string | undefined = undefined;

  constructor(
    private _tokenizer: Tokenizer,
    private _setCapReached: () => void
  ) {}

  public setGoalId(goalId: string | undefined) {
    this._goalId = goalId;
  }

  /**
   * Splits the inputs into groups that, once serialized to JSON, stay below
   * `maxBytes`. Inputs are validated (tokens) beforehand by the caller.
   */
  private _batchByByteSize(inputs: string[], maxBytes: number): string[][] {
    const budget = Math.floor(maxBytes * BYTE_SIZE_SAFETY_MARGIN);
    // Overhead of the surrounding JSON structure + keys + goalId + model.
    const envelopeOverhead = 512;

    const batches: string[][] = [];
    let currentBatch: string[] = [];
    let currentSize = envelopeOverhead;

    for (const input of inputs) {
      // Approximate the JSON-encoded size of this element (quotes + escapes).
      const encodedSize = byteLength(JSON.stringify(input)) + 1;

      if (currentBatch.length > 0 && currentSize + encodedSize > budget) {
        batches.push(currentBatch);
        currentBatch = [];
        currentSize = envelopeOverhead;
      }

      currentBatch.push(input);
      currentSize += encodedSize;
    }

    if (currentBatch.length > 0) {
      batches.push(currentBatch);
    }

    return batches;
  }

  public async createEmbeddings(
    input: string | string[],
    tries?: number
  ): Promise<EmbeddingCreationResult[]> {
    const config = DEFAULT_ADA_CONFIG;

    const inputs = Array.isArray(input) ? input : [input];
    for (const input of inputs) {
      if (this._tokenizer.encode(input).length > config.maxTokensPerInput) {
        throw new Error(
          `Input exceeds max request tokens: ${config.maxTokensPerInput}`
        );
      }
    }

    // First split by the API's max inputs per request, then further split each
    // group so that no single HTTP request body exceeds the proxy limit.
    const tokenBatched = splitArray(inputs, config.maxInputsPerRequest);
    const batchedInputs = tokenBatched.flatMap((group) =>
      this._batchByByteSize(group, MAX_REQUEST_BODY_BYTES)
    );

    const goalId = this._goalId;
    if (!goalId) {
      throw Error("GoalID is not set");
    }

    const results = await this._embedBatchedInputs(
      batchedInputs,
      config.model,
      goalId,
      tries
    );

    return results.flat();
  }

  private async _embedBatchedInputs(
    batchedInputs: string[][],
    model: string,
    goalId: string,
    tries?: number
  ): Promise<EmbeddingCreationResult[][]> {
    // Send the batches with bounded concurrency so we never blow past the
    // body size limit (each request is small) and we don't hammer the proxy.
    const results: EmbeddingCreationResult[][] = [];
    let index = 0;

    const worker = async () => {
      while (index < batchedInputs.length) {
        const currentIndex = index++;
        results[currentIndex] = await this._requestBatch(
          batchedInputs[currentIndex],
          model,
          goalId,
          tries
        );
      }
    };

    const workers = Array.from(
      { length: Math.min(MAX_CONCURRENT_REQUESTS, batchedInputs.length) },
      () => worker()
    );

    await Promise.all(workers);
    return results;
  }

  private async _requestBatch(
    inputs: string[],
    model: string,
    goalId: string,
    tries?: number
  ): Promise<EmbeddingCreationResult[]> {
    const embeddingResponse = await fetch("/api/proxy/embeddings", {
      method: "POST",
      body: JSON.stringify({
        input: [inputs],
        model,
        goalId
      }),
      headers: {
        "Content-Type": "application/json",
      },
    });

    if (!embeddingResponse.ok) {
      const error = await embeddingResponse.json().catch(() => ({}));
      if (embeddingResponse.status === 400) {
        throw Error("Error from OpenAI Embeddings: " + error.error);
      }
      if (embeddingResponse.status === 413) {
        throw Error(
          "Embedding Proxy API: Body Exceeds Maximum Size. " +
          "The embedding request payload was too large for the proxy to handle."
        );
      }
      if (embeddingResponse.status === 429) {
        await new Promise((resolve) => setTimeout(resolve, 15000));
        if (!tries || tries < MAX_RATE_LIMIT_RETRIES) {
          return this._requestBatch(inputs, model, goalId, tries == undefined ? 0 : ++tries);
        }
      }
      if (embeddingResponse.status === 403) {
        this._setCapReached();
        throw Error(ERROR_SUBSIDY_MAX);
      }
      throw Error("Error trying to get response from embeddings proxy.", error);
    }
    const { embeddings } = await embeddingResponse.json();

  private _goalId: string | undefined = undefined;

  public async createEmbeddings(
    input: string | string[],
    tries?: number
  ): Promise<EmbeddingCreationResult[]> {
    const config = DEFAULT_ADA_CONFIG;

    const inputs = Array.isArray(input) ? input : [input];
    for (const input of inputs) {
      if (this._tokenizer.encode(input).length > config.maxTokensPerInput) {
        throw new Error(
          `Input exceeds max request tokens: ${config.maxTokensPerInput}`
        );
      }
    }

    const batchedInputs = splitArray(inputs, config.maxInputsPerRequest);

    const goalId = this._goalId;
    if (!goalId) {
      throw Error("GoalID is not set");
    }

    let embeddings: EmbeddingCreationResult[] = [];

    // Send each batch in a separate request so that the request body
    // never exceeds the maximum size allowed by the embeddings proxy API
    for (const batch of batchedInputs) {
      const batchEmbeddings = await this.createEmbeddingsBatch(
        batch,
        goalId,
        config.model,
        tries
      );
      embeddings = embeddings.concat(batchEmbeddings);
    }

    return embeddings;
  }

  private async createEmbeddingsBatch(
    inputs: string[],
    goalId: string,
    model: string,
    tries?: number
  ): Promise<EmbeddingCreationResult[]> {
    const embeddingResponse = await fetch("/api/proxy/embeddings", {
      method: "POST",
      body: JSON.stringify({
        input: inputs,
        model,
        goalId,
      }),
      headers: {
        "Content-Type": "application/json",
      },
    });

    if (!embeddingResponse.ok) {
      let error: any;
      try {
        error = await embeddingResponse.json();
      } catch (e) {
        // The error response is not always JSON (e.g. "Body exceeds
        // maximum size" errors returned by the hosting platform)
        error = { error: embeddingResponse.statusText };
      }

      if (embeddingResponse.status === 400) {
        throw Error(
          "Error from OpenAI Embeddings: " +
            (error.error ?? embeddingResponse.statusText)
        );
      }
      if (embeddingResponse.status === 429) {
        await new Promise((resolve) => setTimeout(resolve, 15000));
        if (!tries || tries < MAX_RATE_LIMIT_RETRIES) {
          return this.createEmbeddingsBatch(
            inputs,
            goalId,
            model,
            tries == undefined ? 0 : ++tries
          );
        }
      }
      if (embeddingResponse.status === 403) {
        this._setCapReached();
        throw Error(ERROR_SUBSIDY_MAX);
      }
      throw Error("Error trying to get response from embeddings proxy.", error);
    }

    const { embeddings } = await embeddingResponse.json();
    return embeddings;

import type { NextApiRequest, NextApiResponse } from "next";

import { createSupabaseClient } from "../../../api-utils/supabase";
import { createOpenAIApiClient } from "../../../api-utils/openai";

export default async function handler(
  req: NextApiRequest,
  res: NextApiResponse
): Promise<void> {
  if (!(req.method === "POST")) {
    return res.status(404).send({});
  }
  const goalId = req.body.goalId;

  // Ensure the user is logged in
  const session = await getServerSession(req, res, authOptions);
  const email = session?.user?.email;
  if (!session || !email) {
    return res.status(401).send({});
  }

  const supabase = createSupabaseClient();
  const openai = createOpenAIApiClient();

  // Ensure the goal is being tracked, and we're able to continue
  // subsidizing the goal's llm requests
  const isValid = await isGoalValid(goalId, supabase);
  if (!isValid) {
    return res.status(403).send({});
  }
  const canSubsidize = await canUseSubsidy("embedding", goalId, supabase);
  if (!canSubsidize) {
    return res.status(403).send({});
  }

  const input: string[] | string[][] = req.body.input;

  if (!Array.isArray(input) || input.length === 0) {
    return res.status(400).json({ error: "Invalid embeddings input." });
  }

  // The client sends a single batch of inputs per request so that the
  // request body stays below the maximum size allowed by the proxy API.
  // Batched input (string[][]) is still accepted for backwards
  // compatibility.
  const batches: string[][] = Array.isArray(input[0])
    ? (input as string[][])
    : [input as string[]];

  try {
    const embeddings: {
      embedding: number[];
      input: string;
    }[] = [];

    // Process the batches sequentially to avoid exceeding the
    // OpenAI rate limits
    for (const inputs of batches) {
      const { data } = await openai.embeddings.create({
        model: req.body.model,
        input: inputs,
      });

      embeddings.push(
        ...data.map((innerData) => {
          return {
            embedding: innerData.embedding,
            input: inputs[innerData.index],
          };
        })
      );
    }

    return res.status(200).json({ embeddings });
  } catch (e: any) {
    // Rate limit error
    if (e instanceof OpenAIApi.APIError && e.status === 429) {
      return res.status(429).send({});
    }
    // Bad request error (e.g. an input exceeds the model's limits)
    if (e instanceof OpenAIApi.APIError && e.status === 400) {
      return res.status(400).json({ error: e.message });
    }
    return res.status(500).send({});
