import { JSONSchema7Definition } from '@ai-sdk/provider';

/**
 * Converts JSON Schema 7 to OpenAPI Schema 3.0
 */
export function convertJSONSchemaToOpenAPISchema(
  jsonSchema: JSONSchema7Definition | undefined,
): unknown {
  // parameters need to be undefined if they are empty objects:
  if (jsonSchema == null || isEmptyObjectSchema(jsonSchema)) {
    return undefined;
  }

  if (typeof jsonSchema === 'boolean') {
    return { type: 'boolean', properties: {} };
  }

  const {
    type,
    description,
    required,
    properties,
    items,
    allOf,
    anyOf,
    oneOf,
    format,
    const: constValue,
    minLength,
    enum: enumValues,
    additionalProperties,
  } = jsonSchema;

  const result: Record<string, unknown> = {};

  if (description) result.description = description;
  if (required) result.required = required;
  if (format) result.format = format;

  if (constValue !== undefined) {
    result.enum = [constValue];
  }

  // Handle type
  if (type) {
    if (Array.isArray(type)) {
      if (type.includes('null')) {
        result.type = type.filter(t => t !== 'null')[0];
        result.nullable = true;
      } else {
        result.type = type;
      }
    } else if (type === 'null') {
      result.type = 'null';
    } else {
      result.type = type;
    }
  }

  // Handle enum
  if (enumValues !== undefined) {
    result.enum = enumValues;
  }

  if (properties != null) {
    result.properties = Object.entries(properties).reduce(
      (acc, [key, value]) => {
        acc[key] = convertJSONSchemaToOpenAPISchema(value);
        return acc;
      },
      {} as Record<string, unknown>,
    );
  }

  // Handle record types, e.g. schemas created with z.record(z.string()):
  // OpenAPI supports additionalProperties as a schema. Boolean values are not
  // part of the OpenAPI 3.0 subset accepted by the Google Vertex AI / Gemini
  // APIs, so they are removed here.
  if (additionalProperties != null && typeof additionalProperties === 'object') {
    result.additionalProperties = convertJSONSchemaToOpenAPISchema(
      additionalProperties,
    );
  }

  if (items) {
    result.items = Array.isArray(items)
      ? items.map(convertJSONSchemaToOpenAPISchema)
      : convertJSONSchemaToOpenAPISchema(items);
  }

  if (allOf) {
    result.allOf = allOf.map(convertJSONSchemaToOpenAPISchema);
  }
  if (anyOf) {
    // Handle cases where anyOf includes a null type
    if (
      anyOf.some(
        schema => typeof schema === 'object' && schema?.type === 'null',
      )
    ) {
      const nonNullSchemas = anyOf.filter(
        schema => !(typeof schema === 'object' && schema?.type === 'null'),
      );

      if (nonNullSchemas.length === 1) {
        // If there's only one non-null schema, convert it and make it nullable
        const converted = convertJSONSchemaToOpenAPISchema(nonNullSchemas[0]);
        if (typeof converted === 'object') {
          result.nullable = true;
          Object.assign(result, converted);
        }
      } else {
        // If there are multiple non-null schemas, keep them in anyOf
        result.anyOf = nonNullSchemas.map(convertJSONSchemaToOpenAPISchema);
        result.nullable = true;
      }
    } else {
      result.anyOf = anyOf.map(convertJSONSchemaToOpenAPISchema);
    }
  }
  if (oneOf) {
    result.oneOf = oneOf.map(convertJSONSchemaToOpenAPISchema);
  }

  if (minLength !== undefined) {
    result.minLength = minLength;
  }

  return result;
}

function isEmptyObjectSchema(jsonSchema: JSONSchema7Definition): boolean {
  return (
    jsonSchema != null &&
    typeof jsonSchema === 'object' &&
    jsonSchema.type === 'object' &&
    (jsonSchema.properties == null ||
      Object.keys(jsonSchema.properties).length === 0) &&
    // schemas that define additionalProperties (e.g. z.record(z.string()))
    // are not empty and must not be converted to undefined:
    !hasSchemaAdditionalProperties(jsonSchema)
  );
}

function hasSchemaAdditionalProperties(
  jsonSchema: Record<string, unknown>,
): boolean {
  return (
    jsonSchema.additionalProperties != null &&
    typeof jsonSchema.additionalProperties === 'object'
  );
}

## 1. Root-cause analysis

The reported issue: **Vertex AI requests fail when `z.record(z.string())` is used in tool schemas.**

`z.record(z.string())` converts to JSON Schema as `{ type: 'object', additionalProperties: { type: 'string' } }` (no `properties` key). In the buggy code, `isEmptyObjectSchema()` returns `true` for such a schema because `properties == null` satisfies the "empty" condition — the entire record schema is then collapsed to `undefined`, so the tool parameter is dropped/corrupted and the Vertex AI request fails.

1. **Destructures `additionalProperties`** from the input schema.
2. **Emits object-valued `additionalProperties`** as `result.additionalProperties`, recursively converted — this is the correct OpenAPI 3.0 representation of a map/record type.
3. **Excludes record schemas from the "empty object" check** via the new `hasSchemaAdditionalProperties()` guard, so they are no longer reduced to `undefined`.
4. Adds two regression unit tests matching the exact shape produced by `z.record(z.string())`.

## 5. Caveats noted (not blocking)

- **Typing nit:** `hasSchemaAdditionalProperties(jsonSchema: Record<string, unknown>)` receives a `JSONSchema7`-narrowed value. If `JSONSchema7` in `@ai-sdk/provider` is declared as an `interface` (likely, since it's vendored from `@types/json-schema`), `tsc` could raise TS2345 ("index signature for type 'string' is missing") because interfaces lack implicit index signatures. Typing the parameter as `JSONSchema7Definition` (or inlining the check) would remove any doubt. - `additionalProperties: true` still maps to `undefined` — consistent with the pre-existing "empty schema → undefined" contract and with Gemini's OpenAPI subset (booleans unsupported), and zod never emits `true` for records.

### packages/google/src/convert-json-schema-to-openapi-schema.ts
import { JSONSchema7Definition } from '@ai-sdk/provider';

/**
 * Converts JSON Schema 7 to OpenAPI Schema 3.0
 */
export function convertJSONSchemaToOpenAPISchema(
  jsonSchema: JSONSchema7Definition | undefined,
): unknown {
  // parameters need to be undefined if they are empty objects:
  if (jsonSchema == null || isEmptyObjectSchema(jsonSchema)) {
    return undefined;
  }

  if (typeof jsonSchema === 'boolean') {
    return { type: 'boolean', properties: {} };
  }

  const {
    type,
    description,
    required,
    properties,
    items,
    allOf,
    anyOf,
    oneOf,
    format,
    const: constValue,
    minLength,
    enum: enumValues,
    additionalProperties,
  } = jsonSchema;

  const result: Record<string, unknown> = {};

  if (description) result.description = description;
  if (required) result.required = required;
  if (format) result.format = format;

  if (constValue !== undefined) {
    result.enum = [constValue];
  }

  // Handle type
  if (type) {
    if (Array.isArray(type)) {
      if (type.includes('null')) {
        result.type = type.filter(t => t !== 'null')[0];
        result.nullable = true;
      } else {
        result.type = type;
      }
    } else if (type === 'null') {
      result.type = 'null';
    } else {
      result.type = type;
    }
  }

  // Handle enum
  if (enumValues !== undefined) {
    result.enum = enumValues;
  }

  if (properties != null) {
    result.properties = Object.entries(properties).reduce(
      (acc, [key, value]) => {
        acc[key] = convertJSONSchemaToOpenAPISchema(value);
        return acc;
      },
      {} as Record<string, unknown>,
    );
  }

  // Handle record types, e.g. schemas created with z.record(z.string()):
  // OpenAPI supports additionalProperties as a schema. Boolean values are not
  // part of the OpenAPI 3.0 subset accepted by the Google Vertex AI / Gemini
  // APIs, so they are removed here.
  if (additionalProperties != null && typeof additionalProperties === 'object') {
    result.additionalProperties = convertJSONSchemaToOpenAPISchema(
      additionalProperties,
    );
  }

  if (items) {
    result.items = Array.isArray(items)
      ? items.map(convertJSONSchemaToOpenAPISchema)
      : convertJSONSchemaToOpenAPISchema(items);
  }

  if (allOf) {
    result.allOf = allOf.map(convertJSONSchemaToOpenAPISchema);
  }
  if (anyOf) {
    // Handle cases where anyOf includes a null type
    if (
      anyOf.some(
        schema => typeof schema === 'object' && schema?.type === 'null',
      )
    ) {
      const nonNullSchemas = anyOf.filter(
        schema => !(typeof schema === 'object' && schema?.type === 'null'),
      );

      if (nonNullSchemas.length === 1) {
        // If there's only one non-null schema, convert it and make it nullable
        const converted = convertJSONSchemaToOpenAPISchema(nonNullSchemas[0]);
        if (typeof converted === 'object') {
          result.nullable = true;
          Object.assign(result, converted);
        }
      } else {
        // If there are multiple non-null schemas, keep them in anyOf
        result.anyOf = nonNullSchemas.map(convertJSONSchemaToOpenAPISchema);
        result.nullable = true;
      }
    } else {
      result.anyOf = anyOf.map(convertJSONSchemaToOpenAPISchema);
    }
  }
  if (oneOf) {
    result.oneOf = oneOf.map(convertJSONSchemaToOpenAPISchema);
  }

  if (minLength !== undefined) {
    result.minLength = minLength;
  }

  return result;
}

function isEmptyObjectSchema(jsonSchema: JSONSchema7Definition): boolean {
  return (
    jsonSchema != null &&
    typeof jsonSchema === 'object' &&
    jsonSchema.type === 'object' &&
    (jsonSchema.properties == null ||
      Object.keys(jsonSchema.properties).length === 0) &&
    // schemas that define additionalProperties (e.g. z.record(z.string()))
    // are not empty and must not be converted to undefined:
    !hasSchemaAdditionalProperties(jsonSchema)
  );
}

function hasSchemaAdditionalProperties(
  jsonSchema: Record<string, unknown>,
): boolean {
  return (
    jsonSchema.additionalProperties != null &&
    typeof jsonSchema.additionalProperties === 'object'
  );
}
