import { JSONSchema7 } from 'json-schema';
import { z } from 'zod';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * `zod-to-json-schema` drops the `description` of a schema that is wrapped in
 * `ZodOptional`/`ZodNullable` when the `openAi` target (strict mode) is used,
 * because the optional/nullable wrapper is replaced by an `anyOf` that only
 * contains the inner schema and `{ type: 'null' }`.
 *
 * This post-processing step re-attaches descriptions that were defined on
 * optional (or nullable) fields so that they are preserved in the generated
 * JSON schema regardless of whether a field is optional.
 */
function fixOptionalPropertyDescriptions(
  schema: JSONSchema7 | undefined,
): void {
  if (!schema || typeof schema !== 'object') {
    return;
  }

  const properties = schema.properties;

  if (properties && typeof properties === 'object') {
    for (const key of Object.keys(properties)) {
      const property = properties[key] as JSONSchema7;

      if (!property || typeof property !== 'object' || !('anyOf' in property)) {
        continue;
      }

      // ignore empty anyOf arrays
      if (
        !Array.isArray(property.anyOf) ||
        property.anyOf.length === 0
      ) {
        continue;
      }

      // ensure that the null variant is part of the anyOf (otherwise the anyOf
      // has a different origin)
      const containsNull = property.anyOf.some(
        entry =>
          typeof entry === 'object' &&
          entry !== null &&
          (entry as JSONSchema7).type === 'null',
      );

      if (!containsNull) {
        continue;
      }

      if (property.description !== undefined) {
        // the description is already set on the property - nothing to do
        continue;
      }

      // the description is kept on the first (non null) entry of the anyOf
      const firstEntry = property.anyOf.find(
        entry =>
          typeof entry === 'object' &&
          entry !== null &&
          (entry as JSONSchema7).type !== 'null',
      ) as JSONSchema7 | undefined;

      const description = firstEntry?.description;

      if (description !== undefined) {
        property.description = description;
      }
    }
  }

  // continue with nested schemas (the root schema is duplicated instead of
  // using references, so all nested schemas are embedded)
  for (const key of Object.keys(schema)) {
    const value = (schema as any)[key];

    if (value && typeof value === 'object' && !Array.isArray(value)) {
      fixOptionalPropertyDescriptions(value);
    }
  }
}

export function zodSchema<OBJECT>(
  zodSchema: z.Schema<OBJECT, z.ZodTypeDef, any>,
): Schema<OBJECT> {
  const jsonSchema7 = zodToJsonSchema(zodSchema, {
    $refStrategy: 'none', // no references (to support openapi conversion for google)
    target: 'openAi', // openai strict mode compatible
  }) as JSONSchema7;

  fixOptionalPropertyDescriptions(jsonSchema7);

  return jsonSchema(
    jsonSchema7,
    {
      validate: value => {
        const result = zodSchema.safeParse(value);
        return result.success
          ? { success: true, value: result.data }
          : { success: false, error: result.error };
      },
    },
  );
}

    const schema = zodSchema(
      z.object({
        name: z.string().describe('the name'),
      }),
    );

    expect(schema.jsonSchema).toMatchSnapshot();
  });

  it('should create a schema from a zod schema with an optional property description', () => {
    const schema = zodSchema(
      z.object({
        name: z.string().optional().describe('the name'),
      }),
    );

    expect(schema.jsonSchema).toMatchSnapshot();
  });

  it('should create a schema from a zod schema with a nullable property description', () => {
    const schema = zodSchema(
      z.object({
        name: z.string().nullable().describe('the name'),
      }),
    );

    expect(schema.jsonSchema).toMatchSnapshot();
  });

  it('should keep descriptions on optional properties in nested schemas', () => {
    const schema = zodSchema(
      z.object({
        nested: z.object({
          name: z.string().optional().describe('the name'),
        }),
      }),
    );

    expect(schema.jsonSchema).toMatchSnapshot();
  });

  it('should validate values', () => {
    const schema = zodSchema(z.object({ name: z.string() }));

    expect(schema.validate({ name: 'test' })).toStrictEqual({
      success: true,
      value: { name: 'test' },
    });
    expect(schema.validate({ name: 42 })).toStrictEqual({
      success: false,
      error: expect.any(Object),
    });

**The fix's approach is sound:**
- It walks the generated JSON schema recursively and, for any property whose `anyOf` contains a `null` variant, copies the description from the first non-null entry up to the property level.

- **Defensive guards**: Handles `undefined`/non-object schemas, boolean schemas, empty/non-array `anyOf`, and requires a `null` entry before treating an `anyOf` as an optional/nullable wrapper — so legitimate unions like `z.union([z.string(), z.number()])` are untouched.
- **Recursion** correctly reaches nested objects (via the `nested` test) and array `items` (object values), covering realistic tool parameter shapes.
- The `containsNull` check correctly distinguishes optional/nullable wrappers from ordinary unions.

    expect(schema.jsonSchema).toMatchSnapshot();
  });

  it('should create a schema from a zod schema with an optional property description', () => {
    const schema = zodSchema(
      z.object({
        name: z.string().optional().describe('the name'),
      }),
    );

    expect(schema.jsonSchema).toMatchSnapshot();
  });

  it('should create a schema from a zod schema with a nullable property description', () => {
    const schema = zodSchema(
      z.object({
        name: z.string().nullable().describe('the name'),
      }),
    );

    expect(schema.jsonSchema).toMatchSnapshot();
  });

  it('should keep descriptions on optional properties in nested schemas', () => {
    const schema = zodSchema(
      z.object({
        nested: z.object({
          name: z.string().optional().describe('the name'),
        }),
      }),
    );

    expect(schema.jsonSchema).toMatchSnapshot();
  });

  it('should validate values', () => {
    const schema = zodSchema(z.object({ name: z.string() }));

    expect(schema.validate({ name: 'test' })).toStrictEqual({
      success: true,
      value: { name: 'test' },
    });
    expect(schema.validate({ name: 42 })).toStrictEqual({
      success: false,
      error: expect.any(Object),
    });
  });
});
