                    return yaml.safe_load(file)

                    with open(config_path, "r", encoding=encoding, errors="replace") as file:
                        return yaml.safe_load(file)
                except Exception:
                    # Final fallback – latin-1 (maps bytes directly to the first 256 Unicode code points)
                    with open(config_path, "r", encoding="latin-1") as file:
                        return yaml.safe_load(file)

                print(f"File not found: {config_path}")
                raise
