// Define a reasonable max content length per request to avoid token limit errors.
// This is an approximation; the actual token count can differ, but keeping the
// content under a few thousand characters should stay safely within the model's
// context window.
const MAX_CONTENT_LENGTH = 3000; // characters per chunk

export async function POST(
  req: NextRequest,
) {
  const body = await req.json();
  const goalId = body.goalId;

  // Ensure the user is logged in
  const session = await getServerSession(getAuthOptions());
  const email = session?.user?.email;
  if (!session || !email) {
    return NextResponse.json({}, { status: 401 })
  }

  const supabase = createAdminSupabaseClient();
  const openai = createOpenAIApiClient();

  // Ensure the goal is being tracked, and we're able to continue
  // subsidizing the goal's completions requests
  const isValid = await isGoalValid(goalId, supabase);
  if (!isValid) {
    console.error("Goal is not valid: ", goalId);
    return NextResponse.json({}, { status: 403 });
  }
  const canSubsidize = await canUseSubsidy("completions", goalId, supabase);
  if (!canSubsidize) {
    console.error("Cannot subsidize goal: ", goalId);
    return NextResponse.json({}, { status: 403 });
  }

  const functions = body.functions ?? undefined;

  try {
    // Determine if any user‑message exceeds the safe content length.
    // We only split the last user message; earlier messages are assumed to be
    // short system or assistant prompts.
    const messages: any[] = body.messages;
    const lastMessage = messages[messages.length - 1];
    const needsSplit = lastMessage?.role === "user" && typeof lastMessage.content === "string" && lastMessage.content.length > MAX_CONTENT_LENGTH;

    if (!needsSplit) {
      // Simple case – forward the request as‑is.
      const completion = await openai.chat.completions.create({
        messages,
        model: body.options.model,
        functions,
        function_call: functions ? "auto" : undefined,
        temperature: body.options.temperature,
        max_tokens: body.options.maxTokens,
      });
      if (completion.choices.length < 1) {
        return NextResponse.json({}, { status: 400 })
      }
      const choice = completion.choices[0];
      if (!choice.message) {
        return NextResponse.json({}, { status: 400 })
      }
      return NextResponse.json({ message: choice.message }, { status: 200 });
    }

    // Split the large user content into manageable chunks.
    const content: string = lastMessage.content;
    const chunks = [];
    for (let i = 0; i < content.length; i += MAX_CONTENT_LENGTH) {
      chunks.push(content.substring(i, i + MAX_CONTENT_LENGTH));
    }

    const accumulatedResponses: string[] = [];
    // Preserve all preceding messages (system, assistant, etc.) for each chunk.
    const precedingMessages = messages.slice(0, -1);

    for (const chunk of chunks) {
      const chunkMessages = [...precedingMessages, { role: "user", content: chunk }];
      const chunkCompletion = await openai.chat.completions.create({
        messages: chunkMessages,
        model: body.options.model,
        functions,
        function_call: functions ? "auto" : undefined,
        temperature: body.options.temperature,
        max_tokens: body.options.maxTokens,
      });
      if (chunkCompletion.choices.length < 1) {
        // If a particular chunk fails, abort and surface an error.
        return NextResponse.json({}, { status: 400 })
      }
      const chunkChoice = chunkCompletion.choices[0];
      if (!chunkChoice.message) {
        return NextResponse.json({}, { status: 400 })
      }
      // If the model used function calling, we forward the function call response
      // directly; otherwise we concatenate the textual content.
      if (chunkChoice.message.function_call) {
        // For simplicity, we return the first function call encountered.
        return NextResponse.json({ message: chunkChoice.message }, { status: 200 });
      }
      accumulatedResponses.push(chunkChoice.message.content ?? "");
    }

    // Combine all partial responses into a single assistant message.
    const combinedMessage = {
      role: "assistant",
      content: accumulatedResponses.join("\n"),
    } as const;
    return NextResponse.json({ message: combinedMessage }, { status: 200 });
  } catch (e: any) {
    // Rate limit error
    if (e instanceof OpenAIApi.APIError && e.status === 429) {
      return NextResponse.json({}, { status: 429 })
    }

    console.error(e);
    return NextResponse.json({}, { status: 500 })
  }
}

// Helper function to split overly long message contents into smaller chunks
function splitLongMessages(
  messages: { role: string; content: string }[],
  maxChunkSize: number = 3000
): { role: string; content: string }[] {
  const result: { role: string; content: string }[] = [];
  for (const msg of messages) {
    if (!msg.content || typeof msg.content !== "string") {
      // Preserve the message as‑is if content is missing or not a string
      result.push(msg as any);
      continue;
    }
    const content = msg.content;
    if (content.length <= maxChunkSize) {
      result.push(msg as any);
    } else {
      // Split the content into chunks of maxChunkSize characters
      const chunks = [];
      for (let i = 0; i < content.length; i += maxChunkSize) {
        chunks.push(content.substring(i, i + maxChunkSize));
      }
      // Preserve the role for each chunk; optionally add an indicator
      for (let i = 0; i < chunks.length; i++) {
        const chunkContent = chunks[i];
        const newMsg = {
          role: msg.role,
          // Indicate continuation for readability (optional)
          content: i === 0 ? chunkContent : "[continued]\n" + chunkContent,
        } as any;
        result.push(newMsg);
      }
    }
  }
  return result;
}
