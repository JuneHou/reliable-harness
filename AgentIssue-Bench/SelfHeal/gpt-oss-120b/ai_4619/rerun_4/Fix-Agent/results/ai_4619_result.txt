import { ToolSet } from '../generate-text/tool-set';
import { CoreMessage, ToolCallPart, ToolResultPart } from '../prompt';
import { attachmentsToParts } from './attachments-to-parts';
import { MessageConversionError } from './message-conversion-error';
import { UIMessage } from './ui-message';

/**
 * Converts an array of messages from useChat into an array of CoreMessages that can be used
 * with the AI core functions (e.g. `streamText`).
 */
export function convertToCoreMessages<TOOLS extends ToolSet = never>(
  messages: Array<UIMessage>,
  options?: { tools?: TOOLS },
) {
  const tools = options?.tools ?? ({} as TOOLS);
  const coreMessages: CoreMessage[] = [];

  for (const message of messages) {
    const { role, content, toolInvocations, experimental_attachments } = message;

    switch (role) {
      case 'system': {
        coreMessages.push({ role: 'system', content });
        break;
      }

      case 'user': {
        coreMessages.push({
          role: 'user',
          content: experimental_attachments
            ? [{ type: 'text', text: content }, ...attachmentsToParts(experimental_attachments)]
            : content,
        });
        break;
      }

      case 'assistant': {
        // No tool invocations – just forward the content.
        if (!toolInvocations || toolInvocations.length === 0) {
          coreMessages.push({ role: 'assistant', content });
          break;
        }

        // If the assistant also supplied regular content, emit it **before** any tool related messages.
        // This ordering matches providers (e.g., Anthropic) that expect the assistant's textual response
        // to appear prior to the tool call/result sequence.
        if (content !== undefined && content !== null && content !== '') {
          coreMessages.push({ role: 'assistant', content });
        }

        const maxStep = toolInvocations.reduce((max, ti) => Math.max(max, ti.step ?? 0), 0);

        for (let i = 0; i <= maxStep; i++) {
          const stepInvocations = toolInvocations.filter(
            ti => (ti.step ?? 0) === i,
          );

          if (stepInvocations.length === 0) continue;

          // Assistant message containing the tool calls for this step.
          coreMessages.push({
            role: 'assistant',
            content: stepInvocations.map(
              ({ toolCallId, toolName, args }): ToolCallPart => ({
                type: 'tool-call' as const,
                toolCallId,
                toolName,
                args,
              }),
            ),
          });

          // Tool message containing the results of those calls.
          coreMessages.push({
            role: 'tool',
            content: stepInvocations.map((ti): ToolResultPart => {
              if (!('result' in ti)) {
                throw new MessageConversionError({
                  originalMessage: message,
                  message: 'ToolInvocation must have a result: ' + JSON.stringify(ti),
                });
              }
              const { toolCallId, toolName, result } = ti;
              const tool = tools[toolName];
              return tool?.experimental_toToolResultContent != null
                ? {
                    type: 'tool-result',
                    toolCallId,
                    toolName,
                    result: tool.experimental_toToolResultContent(result),
                    experimental_content: tool.experimental_toToolResultContent(result),
                  }
                : {
                    type: 'tool-result',
                    toolCallId,
                    toolName,
                    result,
                  };
            }),
          });
        }
        // No additional assistant content after tool sequences – it has already been emitted above if present.
        break;
      }

      case 'data': {
        // ignore
        break;
      }

      default: {
        const _exhaustiveCheck: never = role;
        throw new MessageConversionError({
          originalMessage: message,
          message: `Unsupported role: ${_exhaustiveCheck}`,
        });
      }
    }
  }

  return coreMessages;
}
