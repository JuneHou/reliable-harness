    """Lazily import symbols defined in ``__all__``.

    Python will call this function for attribute look‑ups that fail the normal
    lookup process.  We use it to import the required sub‑module only when the
    attribute is first accessed, avoiding the heavy imports that were previously
    executed at package import time.

        module = importlib.import_module(_name_to_module[name])
        attr = getattr(module, name)
        # Cache the attribute in the package namespace so subsequent look‑ups
        # are fast and do not trigger another import.
        globals()[name] = attr

def __getattr__(name: str):
    """Lazily import symbols defined in ``__all__``.

    Python will call this function for attribute look‑ups that fail the normal
    lookup process.  We use it to import the required sub‑module only when the
    attribute is first accessed, avoiding the heavy imports that were previously
    executed at package import time.
    """
    if name in _name_to_module:
        module = importlib.import_module(_name_to_module[name])
        attr = getattr(module, name)
        # Cache the attribute in the package namespace so subsequent look‑ups
        # are fast and do not trigger another import.
        globals()[name] = attr
        return attr
    raise AttributeError(f"module 'crewai' has no attribute {name!r}")

    """Return the list of available attributes for ``dir(crewai)``.

        self.trace_set = False

        try:
            from opentelemetry import trace  # noqa: F401
            from opentelemetry.exporter.otlp.proto.http.trace_exporter import (
                OTLPSpanExporter,
            )
            from opentelemetry.sdk.resources import SERVICE_NAME, Resource
            from opentelemetry.sdk.trace import TracerProvider
            from opentelemetry.sdk.trace.export import BatchSpanProcessor
            from opentelemetry.trace import Span, Status, StatusCode
        except Exception:
            # If any of the telemetry dependencies are missing, we simply disable telemetry.
            return

        # Store the imported symbols for later use.
        self._trace = trace
        self._Span = Span
        self._Status = Status
        self._StatusCode = StatusCode
        self._OTLPSpanExporter = OTLPSpanExporter
        self._TracerProvider = TracerProvider
        self._BatchSpanProcessor = BatchSpanProcessor
        self._Resource = Resource
        self._SERVICE_NAME = SERVICE_NAME

        try:
            telemetry_endpoint = "https://telemetry.crewai.com:4319"
            self.resource = self._Resource(
                attributes={self._SERVICE_NAME: "crewAI-telemetry"},
            )
            with suppress_warnings():
                self.provider = self._TracerProvider(resource=self.resource)

                self._OTLPSpanExporter(
                    endpoint=f"{telemetry_endpoint}/v1/traces",
                    timeout=30,
                )
            )

            self.provider.add_span_processor(processor)
            self.ready = True
        except BaseException as e:
            if isinstance(
                e,
                (SystemExit, KeyboardInterrupt, GeneratorExit, asyncio.CancelledError),
            ):
                raise
            self.ready = False

from datetime import datetime

from pydantic import BaseModel, Field, PrivateAttr

from crewai.utilities.printer import Printer


class Logger(BaseModel):
    verbose: bool = Field(default=False)
    _printer: Printer = PrivateAttr(default_factory=Printer)

    def log(self, level, message, color="bold_yellow"):
        if self.verbose:
            timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            self._printer.print(
                f"\n[{timestamp}][{level.upper()}]: {message}", color=color
            )
