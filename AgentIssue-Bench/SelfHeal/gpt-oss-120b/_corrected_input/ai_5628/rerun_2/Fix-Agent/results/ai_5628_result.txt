import { JSONSchema7 } from '@ai-sdk/provider';
import * as z3 from 'zod/v3';
import * as z4 from 'zod/v4';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Fixes JSON Schema generated from Zod `z.record` schemas.
 * Zod may emit a property named "undefined" to represent the
 * ``additionalProperties`` definition. Vertex AI does not accept this
 * and expects the schema to use the `additionalProperties` keyword.
 */
function fixRecordSchema(schema: JSONSchema7): JSONSchema7 {
  if (schema.type === 'object') {
    // If Zod generated a property called "undefined", move it to additionalProperties
    const props = (schema.properties ?? {}) as Record<string, unknown>;
    if (Object.prototype.hasOwnProperty.call(props, 'undefined')) {
      const undefProp = props['undefined'];
      // set additionalProperties if not already defined
      if (schema.additionalProperties === undefined) {
        schema.additionalProperties = undefProp as any;
      }
      // remove the bogus property
      delete props['undefined'];
      // clean up empty properties object
      if (Object.keys(props).length === 0) {
        delete schema.properties;
      } else {
        schema.properties = props as any;
      }
    }
  }
  return schema;
}

export function zod3Schema<OBJECT>(
  zodSchema: z3.Schema<OBJECT, z3.ZodTypeDef, any>,
  options?: {
    /**
     * Enables support for references in the schema.
     * This is required for recursive schemas, e.g. with `z.lazy`.
     * However, not all language models and providers support such references.
     * Defaults to `false`.
     */
    useReferences?: boolean;
  },
): Schema<OBJECT> {
  // default to no references (to support openapi conversion for google)
  const useReferences = options?.useReferences ?? false;

  const rawSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: useReferences ? 'root' : 'none',
    target: 'jsonSchema7', // note: openai mode breaks various gemini conversions
  }) as JSONSchema7;

  const fixedSchema = fixRecordSchema(rawSchema);

  return jsonSchema(
    fixedSchema,
    {
      validate: value => {
        const result = zodSchema.safeParse(value);
        return result.success
          ? { success: true, value: result.data }
          : { success: false, error: result.error };
      },
    },
  );
}

/**
 * Enrich a JSON Schema generated from a Zod schema so that Vertex AI correctly
 * interprets record types (`z.record(z.string())`). Vertex expects the
 * `additionalProperties` field to be explicitly defined for object schemas that
 * act as maps.
 */
function enrichRecordSchema<OBJECT>(
  schema: JSONSchema7,
  zodSchema: z3.Schema<OBJECT, z3.ZodTypeDef, any> | z4.ZodType<OBJECT, any>,
): JSONSchema7 {
  // Detect ZodRecord (both v3 and v4 expose the internal typeName).
  const typeName = (zodSchema as any)?._def?.typeName;
  if (typeName === 'ZodRecord') {
    // Determine the JSON Schema for the record's value type.
    const valueDef = (zodSchema as any)?._def?.valueType?._def?.typeName;
    const valueSchema: JSONSchema7 = {};
    if (valueDef === 'ZodString') {
      valueSchema.type = 'string';
    } else if (valueDef === 'ZodNumber') {
      valueSchema.type = 'number';
    } else if (valueDef === 'ZodBoolean') {
      valueSchema.type = 'boolean';
    } else {
      // For more complex value types let the generated schema stay as‑is.
    }

    // Ensure the object schema has no explicit properties (it works like a map).
    schema.properties = {};
    // Set additionalProperties to the determined value schema, or true if unknown.
    schema.additionalProperties = Object.keys(valueSchema).length ? valueSchema : true;
  }

  return schema;
}
