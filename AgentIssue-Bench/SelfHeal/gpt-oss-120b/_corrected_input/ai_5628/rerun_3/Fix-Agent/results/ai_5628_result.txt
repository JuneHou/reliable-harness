import { JSONSchema7 } from '@ai-sdk/provider';
import * as z3 from 'zod/v3';
import * as z4 from 'zod/v4';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Helper to ensure the generated JSON Schema is compatible with the expectations of the
 * OpenAPI conversion logic used by the Google Vertex provider. Certain Zod constructs –
 * most notably `z.record(z.string())` – result in a schema that only contains an
 * `additionalProperties` field without an explicit `type`. The downstream code interprets
 * the missing `type` as `undefined`, which leads to malformed OpenAPI definitions and
 * consequently failed Vertex AI requests.
 *
 * This function normalises the schema by explicitly adding `type: "object"` (and an
 * empty `properties` object) when a schema has `additionalProperties` but lacks a `type`.
 */
function normalizeRecordSchema(schema: JSONSchema7): JSONSchema7 {
  // If the schema already defines a type, we leave it untouched.
  if (schema.type) return schema;

  // Detect Zod record schemas: they are objects with `additionalProperties` defined
  // and no explicit `type` or `properties`.
  if (schema.additionalProperties && typeof schema.additionalProperties === 'object') {
    return {
      ...schema,
      type: 'object',
      // Ensure `properties` exists so that downstream code that iterates over it does not error.
      properties: schema.properties ?? {},
    } as JSONSchema7;
  }

  return schema;
}

export function zod3Schema<OBJECT>(
  zodSchema: z3.Schema<OBJECT, z3.ZodTypeDef, any>,
  options?: {
    /**
     * Enables support for references in the schema.
     * This is required for recursive schemas, e.g. with `z.lazy`.
     * However, not all language models and providers support such references.
     * Defaults to `false`.
     */
    useReferences?: boolean;
  },
): Schema<OBJECT> {
  // default to no references (to support openapi conversion for google)
  const useReferences = options?.useReferences ?? false;

  const rawSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: useReferences ? 'root' : 'none',
    target: 'jsonSchema7', // note: openai mode breaks various gemini conversions
  }) as JSONSchema7;

  const normalized = normalizeRecordSchema(rawSchema);

  return jsonSchema(
    normalized,
    {
      validate: value => {
        const result = zodSchema.safeParse(value);
        return result.success
          ? { success: true, value: result.data }
          : { success: false, error: result.error };
      },
    },
  );
}

export function zod4Schema<OBJECT>(
  zodSchema: z4.ZodType<OBJECT, any>,
  options?: {
    /**
     * Enables support for references in the schema.
     * This is required for recursive schemas, e.g. with `z.lazy`.
     * However, not all language models and providers support such references.
     * Defaults to `false`.
     */
    useReferences?: boolean;
  },
): Schema<OBJECT> {
  // default to no references (to support openapi conversion for google)
  const useReferences = options?.useReferences ?? false;

  const z4JSONSchema = z4.toJSONSchema(zodSchema, {
    target: 'draft-7',
    io: 'output',
    reused: useReferences ? 'ref' : 'inline',
  }) as JSONSchema7;

  const normalized = normalizeRecordSchema(z4JSONSchema);

  return jsonSchema(normalized, {
    validate: value => {
      const result = z4.safeParse(zodSchema, value);
      return result.success
        ? { success: true, value: result.data }
        : { success: false, error: result.error };
    },
  });
}

export function isZod4Schema(
  zodSchema: z4.ZodType<any, any> | z3.Schema<any, z3.ZodTypeDef, any>,
): zodSchema is z4.ZodType<any, any> {
  // https://zod.dev/library-authors?id=how-to-support-zod-3-and-zod-4-simultaneously
  return '_zod' in zodSchema;
}

  options?: {
    /**
