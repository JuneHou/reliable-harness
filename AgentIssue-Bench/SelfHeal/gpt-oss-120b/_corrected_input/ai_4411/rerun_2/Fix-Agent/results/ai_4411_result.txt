 * description. `zod-to-json-schema` emits `anyOf` arrays for optional fields
 * (e.g. `[ { type: 'string', description: 'my desc' }, { type: 'null' } ]`).
 * In OpenAI's function schema the description must be attached directly to the
 * property object, otherwise it is lost. This helper copies the description
 * from the non‑null branch of `anyOf` (or `oneOf`) back onto the property.
 */
function preserveOptionalDescriptions(schema: JSONSchema7): void {
  // Handle object properties
  if (schema.properties && typeof schema.properties === 'object') {
    for (const key of Object.keys(schema.properties)) {
      const prop = (schema.properties as Record<string, unknown>)[key];
      if (prop && typeof prop === 'object') {
        const propSchema = prop as JSONSchema7;
        // If the property lacks a description, look for anyOf/oneOf with a description
        if (!propSchema.description) {
          const candidates = (propSchema.anyOf ?? propSchema.oneOf) as
            | JSONSchema7[]
            | undefined;
          if (Array.isArray(candidates)) {
            const withDesc = candidates.find(
              (s) => s && typeof s === 'object' && (s as JSONSchema7).description,
            ) as JSONSchema7 | undefined;
            if (withDesc && withDesc.description) {
              propSchema.description = withDesc.description;
            }
          }
        }
        // Recurse into nested schemas (objects, arrays, etc.)
        preserveOptionalDescriptions(propSchema);
      }
    }
  }

  // Handle array items
  if (schema.items && typeof schema.items === 'object') {
    const itemsSchema = schema.items as JSONSchema7;
    preserveOptionalDescriptions(itemsSchema);
  }

  // Handle anyOf/oneOf at the current level for completeness
  const combinators = schema.anyOf ?? schema.oneOf;
  if (Array.isArray(combinators)) {
    for (const sub of combinators) {
      if (sub && typeof sub === 'object') {
        preserveOptionalDescriptions(sub as JSONSchema7);
      }
    }
  }
}
