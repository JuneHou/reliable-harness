import { JSONSchema7 } from 'json-schema';
import { z } from 'zod';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Helper to extract the description from a Zod schema, handling wrappers like ZodOptional.
 */
function extractDescription(zod: any): string | undefined {
  if (!zod || typeof zod !== 'object') return undefined;
  // Direct description on the schema
  if (zod._def?.description) return zod._def.description;
  // If this is a ZodOptional (or ZodNullable, ZodDefault, etc.) the description may be on the inner type
  if (zod._def?.innerType) return extractDescription(zod._def.innerType);
  return undefined;
}

export function zodSchema<OBJECT>(
  zodSchema: z.Schema<OBJECT, z.ZodTypeDef, any>,
): Schema<OBJECT> {
  // Convert the Zod schema to JSON Schema (OpenAI compatible)
  const rawJsonSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: 'none', // no references (to support openapi conversion for google)
    target: 'openAi', // openai strict mode compatible
  }) as JSONSchema7;

  // Preserve field descriptions for optional fields.
  // zod-to-json-schema can drop descriptions on optional properties.
  if (rawJsonSchema.type === 'object' && rawJsonSchema.properties) {
    // Access the underlying Zod object shape (if available)
    const shape = (zodSchema as any).shape;
    if (shape && typeof shape === 'object') {
      for (const [key, prop] of Object.entries(rawJsonSchema.properties)) {
        const zodField = shape[key];
        if (!zodField) continue;
        const desc = extractDescription(zodField);
        if (desc && !(prop as any).description) {
          (prop as any).description = desc;
        }
      }
    }
  }

  return jsonSchema(rawJsonSchema, {
    validate: value => {
      const result = zodSchema.safeParse(value);
      return result.success
        ? { success: true, value: result.data }
        : { success: false, error: result.error };
    },
  });
}
