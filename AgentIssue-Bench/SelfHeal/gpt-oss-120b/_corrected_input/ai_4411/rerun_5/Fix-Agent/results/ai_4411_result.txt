import { JSONSchema7 } from 'json-schema';
import { z } from 'zod';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Recursively lift description fields that are nested inside `anyOf` (which is how
 * `zod-to-json-schema` represents optional fields) up to the parent schema.
 * This ensures that tools like OpenAI retain the description for optional
 * properties.
 */
function liftDescription(schema: any): any {
  if (Array.isArray(schema)) {
    return schema.map(liftDescription);
  }
  if (schema && typeof schema === 'object') {
    // If the schema has an anyOf, propagate the description from the first
    // branch that defines it.
    if (Array.isArray(schema.anyOf)) {
      const withDesc = schema.anyOf.find((s: any) => typeof s.description === 'string');
      if (withDesc) {
        schema.description = withDesc.description;
      }
      schema.anyOf = schema.anyOf.map(liftDescription);
    }
    // Recurse into properties (object schemas)
    if (schema.properties && typeof schema.properties === 'object') {
      for (const key of Object.keys(schema.properties)) {
        schema.properties[key] = liftDescription(schema.properties[key]);
      }
    }
    // Recurse into items (array schemas)
    if (schema.items) {
      schema.items = liftDescription(schema.items);
    }
  }
  return schema;
}

export function zodSchema<OBJECT>(
  zodSchema: z.Schema<OBJECT, z.ZodTypeDef, any>,
): Schema<OBJECT> {
  // Convert Zod schema to JSON Schema using the openAi target for compatibility.
  const rawSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: 'none', // no references (to support openapi conversion for google)

    },
  });
