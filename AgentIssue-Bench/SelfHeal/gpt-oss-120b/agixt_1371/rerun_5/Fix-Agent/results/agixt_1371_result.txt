+        To make the resolver robust we normalise the ``prompt`` value to a

+        prompt as a dictionary, but also gracefully handles any legacy string
+        representation.
+        """

+
+        chain_manager = Chain(user=user)
+        chain_data = chain_manager.get_chain(chain_name=chain_name)

+        if not chain_data:
+            raise Exception(f"Chain {chain_name} not found")
+
+        # Normalise each step's prompt to a dict before unpacking.
+        def _normalise_prompt(prompt_raw):
+            # Prompt may already be a dict, a JSON string, or ``None``.
