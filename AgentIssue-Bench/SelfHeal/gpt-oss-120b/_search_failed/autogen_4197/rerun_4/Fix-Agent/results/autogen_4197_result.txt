tutorial notebooks. The real library provides richer functionality, but for the
purpose of the documentation examples a lightweight stub is sufficient.
"""

from __future__ import annotations

from dataclasses import dataclass
from typing import Any


class CancellationToken:
    """A simple cancellation token.

    The token can be shared across asynchronous tasks. Workers can periodically
    check the :pyattr:`is_cancelled` property to cooperatively stop execution.
    """

    def __init__(self) -> None:
        self._cancelled: bool = False

    def cancel(self) -> None:
        """Mark the token as cancelled.
        """
        self._cancelled = True

    @property
    def is_cancelled(self) -> bool:
        """Return ``True`` if ``cancel`` has been called.
        """
        return self._cancelled

    def __bool__(self) -> bool:
        """Allow using the token directly in boolean contexts.
        ``if token: ...`` evaluates to ``True`` when not cancelled.
        """
        return not self._cancelled

    def __repr__(self) -> str:
        status = "cancelled" if self._cancelled else "active"
        return f"<CancellationToken {status}>"

# Export the public names for `from autogen_core.base import *`
