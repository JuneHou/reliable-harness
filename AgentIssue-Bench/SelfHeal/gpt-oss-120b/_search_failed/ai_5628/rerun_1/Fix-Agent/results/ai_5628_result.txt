### packages/provider-utils/src/zod-schema.ts
import { JSONSchema7 } from '@ai-sdk/provider';
import * as z3 from 'zod/v3';
import * as z4 from 'zod/v4';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Normalizes JSON Schema generated from Zod records.
 *
 * Zod's `toJSONSchema` (especially for Zod v4) can emit `additionalProperties: true`
 * for `z.record(z.string())`. Vertex AI expects a concrete schema for the values,
 * i.e. `additionalProperties: { type: 'string' }`. The undefined value leads to
 * the property being rendered as `undefined` in the generated OpenAPI schema,
 * causing request failures.
 *
 * This helper rewrites such schemas to the expected form while leaving other
 * schemas untouched.
 */
function normalizeRecordSchema(schema: JSONSchema7): JSONSchema7 {
  // If the schema represents an object with "any" additional properties (`true`),
  // replace it with a schema that accepts string values.
  if (
    schema &&
    schema.type === 'object' &&
    (schema as any).additionalProperties === true
  ) {
    // Clone to avoid mutating the original reference.
    const newSchema = { ...schema, additionalProperties: { type: 'string' } } as JSONSchema7;
    return newSchema;
  }
  return schema;
}

export function zod3Schema<OBJECT>(
  zodSchema: z3.Schema<OBJECT, z3.ZodTypeDef, any>,
  options?: {
    /**
     * Enables support for references in the schema.
     * This is required for recursive schemas, e.g. with `z.lazy`.
     * However, not all language models and providers support such references.
     * Defaults to `false`.
     */
    useReferences?: boolean;
  },
): Schema<OBJECT> {
  // default to no references (to support openapi conversion for google)
  const useReferences = options?.useReferences ?? false;

  const rawJsonSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: useReferences ? 'root' : 'none',
    target: 'jsonSchema7', // note: openai mode breaks various gemini conversions
  }) as JSONSchema7;

  const jsonSchemaFixed = normalizeRecordSchema(rawJsonSchema);

  return jsonSchema(
    jsonSchemaFixed,
    {
      validate: value => {
        const result = zodSchema.safeParse(value);
        return result.success
          ? { success: true, value: result.data }
          : { success: false, error: result.error };
      },
    },
  );
}
