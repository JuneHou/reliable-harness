import { JSONSchema7 } from 'json-schema';
import { z } from 'zod';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Recursively traverses a JSON Schema and, for optional fields represented as an `anyOf`
 * array, lifts any `description` present on the inner schema objects to the parent
 * schema property. This works around a limitation of `zod-to-json-schema` where the
 * description is attached to the inner schema when a Zod field is marked as optional
 * (e.g. `z.string().optional()`). Without this fix the generated OpenAI tool schema
 * omits the description, leading to missing documentation for optional arguments.
 */
function liftOptionalDescriptions(schema: JSONSchema7): void {
  if (!schema || typeof schema !== 'object') return;

  // Process object properties
  if (schema.properties && typeof schema.properties === 'object') {
    for (const key of Object.keys(schema.properties)) {
      const prop: any = (schema.properties as any)[key];

      // If the property uses anyOf (typical for optional primitives), move description up
      if (Array.isArray(prop?.anyOf)) {
        const withDesc = prop.anyOf.find((p: any) => typeof p.description === 'string');
        if (withDesc && withDesc.description) {
          prop.description = withDesc.description;
        }
      }

      // Recurse into the property definition
      liftOptionalDescriptions(prop as JSONSchema7);
    }
  }

  // Handle array items
  if (schema.items) {
    liftOptionalDescriptions(schema.items as JSONSchema7);
  }

  // Recurse into combinators (allOf, anyOf, oneOf)
  for (const comb of ['allOf', 'anyOf', 'oneOf'] as const) {
    const arr = (schema as any)[comb];
    if (Array.isArray(arr)) {
      arr.forEach((s) => liftOptionalDescriptions(s as JSONSchema7));
    }
  }
}

export function zodSchema<OBJECT>(
  zodSchema: z.Schema<OBJECT, z.ZodTypeDef, any>,
): Schema<OBJECT> {
  // Convert the Zod schema to JSON Schema using the OpenAI‑compatible target.
  const rawSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: 'none', // no references (to support openapi conversion for google)
    target: 'openAi', // openai strict mode compatible
  }) as JSONSchema7;

  // Ensure optional field descriptions are retained.
  liftOptionalDescriptions(rawSchema);

  return jsonSchema(rawSchema, {
    validate: (value) => {
      const result = zodSchema.safeParse(value);
      return result.success
        ? { success: true, value: result.data }
        : { success: false, error: result.error };
    },
  });
}
