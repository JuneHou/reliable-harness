from typing import Any, Dict, List, Optional
from litellm import completion
import litellm


class LLM:
    def __init__(self, model: str, stop: Optional[List[str]] = None, callbacks: Optional[List[Any]] = None):
        """Create a wrapper around litellm's ``completion``.

        Parameters
        ----------
        model: str
            The model name to be used for the completion request.
        stop: Optional[List[str]]
            A list of stop words. Some models (e.g., o1‑preview/o1‑mini) do not
            support a ``stop`` parameter; in those cases the argument is omitted
            entirely. If ``None`` or an empty list is provided, ``stop`` is not
            sent to ``litellm.completion``.
        callbacks: Optional[List[Any]]
            Callback functions that will be invoked after a response is
            retrieved. Stored on the instance for later use.
        """
        self.stop = stop or []
        self.model = model
        # Store callbacks (avoid mutable default argument pitfalls)
        self.callbacks = callbacks or []
        litellm.callbacks = self.callbacks
        litellm.set_verbose = True

    def call(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        """Make a completion request.

        The ``stop`` argument is only included when a non‑empty list is set.
        This prevents errors with models that do not accept ``stop``.
        """
        if self.stop:
            response = completion(
                stop=self.stop,
                model=self.model,
                messages=messages,
                num_retries=5,
            )
        else:
            response = completion(
                model=self.model,
                messages=messages,
                num_retries=5,
            )
        # ``completion`` returns a dict; the content we care about is nested
        # under ``choices[0]['message']['content']``.
        return response["choices"][0]["message"]["content"]

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)
