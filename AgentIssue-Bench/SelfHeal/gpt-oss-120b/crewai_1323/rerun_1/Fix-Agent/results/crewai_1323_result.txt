from typing import Any, Dict, List, Optional

import litellm
from litellm import completion


class LLM:
    def __init__(
        self,
        model: str,
        stop: Optional[List[str]] = None,
        callbacks: Optional[List[Any]] = None,
    ):
        """Initialize the LLM wrapper.

        Parameters
        ----------
        model: str
            The model name to be used for completions.
        stop: Optional[List[str]]
            A list of stop words for the model. Some models (e.g., o1-preview/o1-mini)
            do not support a ``stop`` parameter; in those cases the value should be
            ``None`` or an empty list and the parameter will be omitted when calling
            ``litellm.completion``.
        callbacks: Optional[List[Any]]
            Callbacks that should be registered with ``litellm``. Stored on the
            instance for later use.
        """
        self.stop = stop
        self.model = model
        # Store callbacks safely; avoid mutable default arguments.
        self.callbacks: List[Any] = callbacks or []
        litellm.callbacks = self.callbacks
        litellm.set_verbose = True

    def call(self, messages: List[Dict[str, str]]) -> str:
        """Make a completion request.

        The ``stop`` argument is only included when it is provided and non‑empty,
        preventing errors with models that do not accept the parameter.
        """
        request_kwargs: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "num_retries": 5,
        }
        # Include ``stop`` only if it is a non‑empty list; ``None`` or ``[]`` means
        # the model does not support the parameter.
        if self.stop:
            request_kwargs["stop"] = self.stop

        response = completion(**request_kwargs)
        # ``completion`` returns a dict with a ``choices`` list; we forward the
        # generated content.
        return response["choices"][0]["message"]["content"]

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)
