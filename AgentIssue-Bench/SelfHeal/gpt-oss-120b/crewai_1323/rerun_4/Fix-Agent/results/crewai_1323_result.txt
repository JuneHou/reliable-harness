from typing import Any, Dict, List, Optional
from litellm import completion
import litellm


class LLM:
    def __init__(self, model: str, stop: Optional[List[str]] = None, callbacks: Optional[List[Any]] = None):
        """Initialize the LLM wrapper.

        Parameters
        ----------
        model: str
            The model name to be used for completion.
        stop: Optional[List[str]]
            A list of stop tokens. If ``None`` or empty, the ``stop`` argument will not be
            passed to the underlying ``litellm`` call. This is important for models that
            do not support a ``stop`` parameter (e.g., ``o1-preview``/``o1-mini``).
        callbacks: Optional[List[Any]]
            Optional callbacks to register with ``litellm``.
        """
        self.stop = stop or []
        self.model = model
        # Store callbacks locally for potential future use and register them with litellm.
        self.callbacks = callbacks or []
        litellm.callbacks = self.callbacks
        litellm.set_verbose = True

    def call(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        """Perform a completion request.

        The ``stop`` argument is only included if a non‑empty stop list is provided.
        This avoids errors with models that do not accept a ``stop`` parameter.
        """
        request_kwargs: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "num_retries": 5,
        }
        if self.stop:
            request_kwargs["stop"] = self.stop
        response = completion(**request_kwargs)
        return response["choices"][0]["message"]["content"]

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)
