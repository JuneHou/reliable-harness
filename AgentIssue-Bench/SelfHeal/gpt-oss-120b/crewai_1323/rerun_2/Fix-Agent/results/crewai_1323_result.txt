from typing import Any, Dict, List, Optional
from litellm import completion
import litellm


class LLM:
    def __init__(
        self,
        model: str,
        stop: Optional[List[str]] = None,
        callbacks: Optional[List[Any]] = None,
    ):
        """Create an LLM wrapper.

        Args:
            model: The model name to be used with litellm.
            stop: Optional list of stop sequences. If ``None`` or empty, the ``stop``
                argument will **not** be sent to ``litellm.completion`` – this is
                required for models like ``o1-preview`` and ``o1-mini`` which do
                not support a ``stop`` parameter.
            callbacks: Optional list of callbacks to be registered with litellm.
        """
        # ``stop`` is optional – keep ``None`` when not provided so we can omit it
        self.stop = stop
        self.model = model
        # Store callbacks for later use (e.g., in _call_callbacks)
        self.callbacks = callbacks or []
        litellm.callbacks = self.callbacks
        litellm.set_verbose = True

    def call(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        """Invoke the model via ``litellm.completion``.

        The ``stop`` argument is only passed when it is a non‑empty list. This
        prevents errors with models that do not accept the parameter.
        """
        # Build arguments dynamically to avoid sending unsupported parameters.
        completion_kwargs: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "num_retries": 5,
        }
        # Only include Stop if it is provided and not empty.
        if self.stop:
            completion_kwargs["stop"] = self.stop
        response = completion(**completion_kwargs)
        return response["choices"][0]["message"]["content"]

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)
