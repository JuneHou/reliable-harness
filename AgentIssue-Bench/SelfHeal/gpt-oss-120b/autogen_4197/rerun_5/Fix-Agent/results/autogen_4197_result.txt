"""Core base utilities.

Only a lightweight ``CancellationToken`` is provided for the tutorial notebooks.
The token can be queried for cancellation and can be set manually. In real
applications the token would be integrated with async task cancellation logic.
"""

from __future__ import annotations

import asyncio
from typing import Optional


class CancellationToken:
    """A simple cancellation token.

    The token starts in a *not cancelled* state. The :meth:`cancel` method marks
    it as cancelled. Consumers can call :meth:`throw_if_cancelled` to raise an
    ``asyncio.CancelledError`` when cancellation has been requested.
    """

    __slots__ = ("_cancelled",)

    def __init__(self) -> None:
        self._cancelled: bool = False

    def cancel(self) -> None:
        """Mark the token as cancelled."""

        self._cancelled = True

    @property
    def is_cancelled(self) -> bool:
        """Return ``True`` if cancellation was requested."""

        return self._cancelled

    def throw_if_cancelled(self) -> None:
        """Raise ``asyncio.CancelledError`` if the token is cancelled.

        This mirrors the behaviour of many async libraries where a cancellation
        token can be used to abort long‑running operations.
        """

        if self._cancelled:
            raise asyncio.CancelledError()

    def __repr__(self) -> str:
        return f"CancellationToken(cancelled={self._cancelled})"
