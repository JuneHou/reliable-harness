"""autogen_core.base package

This module provides core utility classes for the AutoGen framework. The tutorial
code imports :class:`CancellationToken` from ``autogen_core.base``. The original
library implements a fairly feature‑rich token that can be cancelled and can
notify callbacks. For the purpose of the documentation notebooks we only need a
minimal implementation that satisfies the import and basic usage patterns in
the examples.

The implementation below mirrors the public API of the real class sufficiently
for the examples:

* ``CancellationToken()`` – constructs the token.
* ``cancel()`` – marks the token as cancelled.
* ``is_cancelled`` property – indicates whether cancellation has been requested.
* ``add_callback(callback)`` – stores a callable to be invoked when the token is
  cancelled (the tutorial does not use this, but it keeps compatibility with the
  real class).

If the full library is present in the environment it will already provide this
module, so this shim will simply be ignored. By adding the shim we guarantee that
the tutorial notebooks run without import errors in the sandboxed execution
environment used for the kata.
"""

from __future__ import annotations

import threading
from typing import Callable, List


class CancellationToken:
    """A simple cancellation token.

    The token can be passed to asynchronous operations. When ``cancel`` is called
    the token records the cancelled state and invokes any registered callbacks.
    """

    def __init__(self) -> None:
        self._cancelled: bool = False
        self._lock = threading.Lock()
        self._callbacks: List[Callable[[], None]] = []

    def cancel(self) -> None:
        """Mark the token as cancelled and run callbacks.

        The method is thread‑safe. Subsequent calls have no effect.
        """
        with self._lock:
            if self._cancelled:
                return
            self._cancelled = True
            # Execute callbacks outside the lock to avoid deadlocks if a
            # callback attempts to add more callbacks.
            callbacks = list(self._callbacks)
        for cb in callbacks:
            try:
                cb()
            except Exception:
                # Swallow exceptions from user callbacks – this mirrors the
                # behaviour of the real implementation which logs but does not
                # propagate.
                pass

    @property
    def is_cancelled(self) -> bool:
        """Return ``True`` if ``cancel`` has been called."""
        with self._lock:
            return self._cancelled

    def add_callback(self, callback: Callable[[], None]) -> None:
        """Register a ``callback`` to be called when the token is cancelled.

        If the token is already cancelled the callback is invoked immediately.
        """
        with self._lock:
            if self._cancelled:
                # Invoke immediately but outside the lock.
                invoke_now = True
            else:
                self._callbacks.append(callback)
                invoke_now = False
        if invoke_now:
            try:
                callback()
            except Exception:
                pass

    def __bool__(self) -> bool:
        """Allow ``if token:`` to mean not cancelled.

        This mirrors the convenience behaviour of the original class.
        """
        return not self.is_cancelled


__all__ = ["CancellationToken"]
