### python/packages/autogen-core/autogen_core/components/tools.py
"""Minimal stub for tool definitions used in the tutorials.

The real AutoGen library provides a rich ``FunctionTool`` class that registers a
Python callable as a tool for LLMs.  For the documentation notebooks we only need
the ability to store the callable and an optional description, as the notebooks
never invoke the tool directly – the LLM mock returns a fixed string.
"""

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Callable, Coroutine, Optional


@dataclass
class FunctionTool:
    """A simple wrapper representing a function that can be called by an agent.

    Parameters
    ----------
    func:
        The callable implementing the tool logic.
    description:
        Human‑readable description of what the tool does.  It is stored for
        compatibility with the full library but otherwise unused.
    """

    func: Callable[..., Any]
    description: Optional[str] = None

    async def __call__(self, *args: Any, **kwargs: Any) -> Any:
        """Invoke the wrapped function.

        If the function is a coroutine function we ``await`` it, otherwise we call
        it directly.  This mirrors the behaviour of the real ``FunctionTool``
        which can handle both sync and async callables.
        """
        result = self.func(*args, **kwargs)
        if isinstance(result, Coroutine):
            result = await result
        return result

"""Minimal stub for tool definitions used in the tutorials.

The real AutoGen library provides a rich ``FunctionTool`` class that registers a
Python callable as a tool for LLMs.  For the documentation notebooks we only need
the ability to store the callable and an optional description, as the notebooks
never invoke the tool directly – the LLM mock returns a fixed string.
"""

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Callable, Coroutine, Optional


@dataclass
class FunctionTool:
    """A simple wrapper representing a function that can be called by an agent.

    Parameters
    ----------
    func:
        The callable implementing the tool logic.
    description:
        Human‑readable description of what the tool does.  It is stored for
        compatibility with the full library but otherwise unused.
    """

    func: Callable[..., Any]
    description: Optional[str] = None

    async def __call__(self, *args: Any, **kwargs: Any) -> Any:
        """Invoke the wrapped function.

        If the function is a coroutine function we ``await`` it, otherwise we call
        it directly.  This mirrors the behaviour of the real ``FunctionTool``
        which can handle both sync and async callables.
        """
        result = self.func(*args, **kwargs)
        if isinstance(result, Coroutine):
            result = await result
        return result
