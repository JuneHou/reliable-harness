                for inv in step_invocations:
                    if "result" not in inv:
                        raise MessageConversionError(
                            message,
                            "ToolInvocation must have a result: "
                            + str(inv),
                        )
                    tool_name: str = inv["toolName"]
                    tool_def = tools.get(tool_name)
                    result_val = inv["result"]
                    if (
                        tool_def
                        and tool_def.experimental_toToolResultContent is not None
                    ):
                        converted = tool_def.experimental_toToolResultContent(
                            result_val
                        )
                        tool_results.append(
                            {
                                "type": "tool-result",
                                "toolCallId": inv["toolCallId"],
                                "toolName": tool_name,
                                "result": converted,
                                "experimental_content": converted,
                            }
                        )
                    else:
                        tool_results.append(
                            {
                                "type": "tool-result",
                                "toolCallId": inv["toolCallId"],
                                "toolName": tool_name,
                                "result": result_val,
                            }
                        )
                core_messages.append({"role": "tool", "content": tool_results})

            # After all steps have been processed, add the plain assistant content
            # (if any).  This guarantees the assistant's final textual response is
            # the *last* message for the turn.
            if content:
                core_messages.append({"role": "assistant", "content": content})
            continue

        if role == "data":
            # "data" messages are ignored in the core representation.
            continue

        # Any other role is unsupported.
        raise MessageConversionError(message, f"Unsupported role: {role}")

    return core_messages
