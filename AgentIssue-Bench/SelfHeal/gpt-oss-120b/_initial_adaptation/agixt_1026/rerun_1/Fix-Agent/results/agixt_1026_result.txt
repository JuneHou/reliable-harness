from ApiClient import ApiClient
from Extensions import Extensions


def _is_true(value):
    """Return True if the given value represents a truthy setting.

    The configuration values may be provided as strings (e.g., "true"/"false")
    or as actual booleans. This helper normalises both representations.
    """
    if isinstance(value, str):
        return value.strip().lower() == "true"
    return bool(value)


class voice_chat(Extensions):
    def __init__(self, **kwargs):
        # Agent name handling
        self.agent_name = kwargs.get("agent_name", "gpt4free")

        # Default voice prompt – may be overridden by enabled_commands
        self.voice_prompt = "Voice Chat without Commands"
        if kwargs.get("enabled_commands"):
            if len(kwargs["enabled_commands"]) > 0:
                self.voice_prompt = "Voice Chat"

        # Default TTS command – will be overridden by specific settings below
        self.tts_command = "Speak with TTS with Streamlabs Text to Speech"

        # Streamlabs TTS setting – accept both bool and string values
        if "USE_STREAMLABS_TTS" in kwargs:
            if _is_true(kwargs["USE_STREAMLABS_TTS"]):
                self.tts_command = "Speak with TTS with Streamlabs Text to Speech"

        # GTTS setting – accept both bool and string values
        if "USE_GTTS" in kwargs:
            if _is_true(kwargs["USE_GTTS"]):
                self.tts_command = "Speak with GTTS"

        # HuggingFace TTS setting – requires both a truthy flag and an API key
        if "USE_HUGGINGFACE_TTS" in kwargs:
            if _is_true(kwargs["USE_HUGGINGFACE_TTS"]) and "HUGGINGFACE_API_KEY" in kwargs:
                if kwargs["HUGGINGFACE_API_KEY"] != "":
                    self.tts_command = "Read Audio with Huggingface"

        # ElevenLabs – if an API key is provided, prefer this engine
        if "ELEVENLABS_API_KEY" in kwargs:
            if kwargs["ELEVENLABS_API_KEY"] != "":
                self.tts_command = "Speak with TTS Using Elevenlabs"

        # Command mapping
        self.commands = {"Chat with Voice": self.chat_with_voice}

        # Conversation name handling
        self.conversation_name = f"Voice Chat with {self.agent_name}"
        if "conversation_name" in kwargs:
            self.conversation_name = kwargs["conversation_name"]

    def chat_with_voice(
        self,
        base64_audio,
        conversation_results=3,
        context_results=10,
    ):
        """Process an audio input, get a textual response, and return TTS audio.

        Steps:
        1. Transcribe the supplied base64‑encoded audio.
        2. Send the transcription to the configured agent using the appropriate
           voice prompt.
        3. Feed the agent's textual reply to the selected TTS command.
        4. Return the resulting audio payload.
        """
        # 1. Transcribe the audio to text.
        user_input = ApiClient.execute_command(
            agent_name=self.agent_name,
            command_name="Transcribe Base64 Audio",
            command_args={"base64_audio": base64_audio},
        )
        # 2. Send the transcribed text to the agent.
        text_response = ApiClient.prompt_agent(
            agent_name=self.agent_name,
            prompt_name=self.voice_prompt,
            prompt_args={
                "user_input": user_input,
                "conversation_name": self.conversation_name,
                "conversation_results": conversation_results,
                "context_results": context_results,
            },
        )
        # 3. Get the audio response from the selected TTS engine.
        audio_response = ApiClient.execute_command(
            agent_name=self.agent_name,
            command_name=self.tts_command,
            command_args={"text": text_response},
        )
        # 4. Return the audio payload.
        return audio_response
