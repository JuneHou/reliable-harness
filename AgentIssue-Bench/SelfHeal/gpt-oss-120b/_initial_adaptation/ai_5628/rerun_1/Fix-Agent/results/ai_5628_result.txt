"""Utility functions for fixing JSON Schema generated from Zod schemas.

}
```

record. It is safe to run on any JSON Schema – it only modifies objects that
match the pattern produced by Zod for ``z.record``.

The function is recursive, ensuring that nested records are also fixed.
"""

from __future__ import annotations

from copy import deepcopy
from typing import Any, Dict

def _is_zod_record(schema: Dict[str, Any]) -> bool:
    """Return ``True`` if *schema* appears to be a Zod ``record``.

    The JSON Schema generated by ``zod-to-json-schema`` for a record looks
    like:

    ```json
    {
      "type": "object",
      "zodType": "ZodRecord",
      "value": { ... }   # schema of the record values
    }
    ```

    In some cases the ``value`` key is omitted when the record is untyped –
    ``z.record(z.string())`` will still have a ``value`` entry describing the
    string type.
    """
    return schema.get("type") == "object" and schema.get("zodType") == "ZodRecord"

def _default_value_schema() -> Dict[str, Any]:
    """Fallback schema used when the record value type cannot be determined.

    ``z.record(z.string())`` should result in a string schema; however, other
    callers might use ``z.record(z.any())`` which translates to an unrestricted
    value.  The safest generic schema is ``{"type": "object"}`` but a string
    is more specific and matches the common case.
    """
    return {"type": "string"}

    """Recursively fix a JSON Schema so that Zod ``record`` definitions are
    compatible with OpenAPI providers such as Vertex AI.

    The function returns a **new** dictionary; the original input is left
    untouched.
    """

    schema = deepcopy(schema)
