// packages/ui-utils/src/zod-schema.ts
import { JSONSchema7 } from 'json-schema';
import { z } from 'zod';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Recursively copy Zod field descriptions onto the generated JSON Schema.
 *
 * Zod's `.optional()` creates a new schema instance that **does not** retain the
 * description that may have been attached to the original schema via
 * `.describe()`. When `zod-to-json-schema` converts such a schema, the resulting
 * property ends up without a `description` field, which breaks the expectation
 * for OpenAI tool schema generation.
 *
 * This helper walks the top‑level object schema (nested objects are handled as
 * well) and, for each property, injects the description from the original Zod
 * definition when it is missing.
 */
function injectZodDescriptions(
  json: JSONSchema7,
  zod: z.ZodTypeAny,
): JSONSchema7 {
  // Only handle object schemas – other kinds are returned unchanged.
  if (zod instanceof z.ZodObject && json.type === 'object' && json.properties) {
    const shape = (zod as z.ZodObject<any>).shape;
    for (const [key, zodProp] of Object.entries(shape)) {
      const propSchema = json.properties[key] as JSONSchema7 | undefined;
      if (!propSchema) continue;

      // Zod stores the description (added via `.describe`) on the internal
      // definition object under `_def.description`.
      const description = (zodProp as any)._def?.description as string | undefined;
      if (description && !propSchema.description) {
        (propSchema as JSONSchema7).description = description;
      }

      // Recurse for nested objects.
      if (
        propSchema.type === 'object' &&
        (zodProp as any).shape &&
        (propSchema as JSONSchema7).properties
      ) {
        injectZodDescriptions(propSchema as JSONSchema7, zodProp as z.ZodTypeAny);
      }
    }
  }
  return json;
}

export function zodSchema<OBJECT>(
  zodSchema: z.Schema<OBJECT, z.ZodTypeDef, any>,
): Schema<OBJECT> {
  // First, generate a JSON Schema from the Zod schema.
  const rawJson = zodToJsonSchema(zodSchema, {
    $refStrategy: 'none', // no references (to support openapi conversion for google)
    target: 'openAi', // openai strict mode compatible
  }) as JSONSchema7;

  // Then, ensure any missing descriptions caused by `.optional()` are added.
  const enrichedJson = injectZodDescriptions(rawJson, zodSchema as unknown as z.ZodTypeAny);

  return jsonSchema(enrichedJson, {
    validate: value => {
      const result = zodSchema.safeParse(value);
      return result.success
        ? { success: true, value: result.data }
        : { success: false, error: result.error };
    },
  });
}
