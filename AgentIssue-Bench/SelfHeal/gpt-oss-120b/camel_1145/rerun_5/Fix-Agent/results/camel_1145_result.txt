# not required for the core library. We import them lazily so that the package
# can be imported even when the optional dependencies are missing. If an import
# fails we expose a placeholder that raises an informative ImportError when
# accessed.

def _optional_import(module_path: str, names: tuple):
    """Attempt to import *names* from *module_path*.

    Returns a dictionary mapping each name to the imported object if successful;
    otherwise the name maps to a stub object that raises ImportError on use.
    """
    result = {}
    try:
        module = __import__(module_path, fromlist=names)
        for name in names:
            result[name] = getattr(module, name)
    except Exception as e:  # Catch ImportError and any runtime import errors.
        def _stub(*args, **kwargs):  # pragma: no cover
            raise ImportError(
                f"Optional toolkit '{module_path}' could not be imported: {e}"
            )

        for name in names:
            result[name] = _stub
    return result
