# Each optional toolkit is imported inside a try/except block.  If the
# import fails (e.g., because a third‑party dependency is missing), the
# toolkit is simply omitted from the public namespace.  This prevents the
# whole package from crashing on import when optional dependencies are not
# installed, while still exposing the toolkit when the environment satisfies
# its requirements.

_optional_exports = []  # Collect names that are successfully imported

def _try_import(module_path: str, names: list[str]):
    """Attempt to import *names* from *module_path*.

    On success, the imported objects are added to the module globals and the
    names are appended to ``_optional_exports``.  Any exception (ImportError or
    other import‑time errors) is silenced – the toolkit is treated as
    unavailable.
    """
    try:
        module = __import__(module_path, globals(), locals(), names, 1)
        for name in names:
            globals()[name] = getattr(module, name)
        _optional_exports.extend(names)
    except Exception:  # pragma: no cover – we deliberately swallow all errors
        # The optional toolkit cannot be used in the current environment.
        pass
