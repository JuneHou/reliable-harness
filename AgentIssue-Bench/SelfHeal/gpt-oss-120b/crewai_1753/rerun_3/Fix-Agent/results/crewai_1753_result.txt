import json
from datetime import date, datetime
from decimal import Decimal
from uuid import UUID
from enum import Enum

from pydantic import BaseModel


class CrewJSONEncoder(json.JSONEncoder):
    def default(self, obj):
        # Handle Pydantic BaseModel objects
        if isinstance(obj, BaseModel):
            return self._handle_pydantic_model(obj)
        # Handle Enum objects by serializing their value
        elif isinstance(obj, Enum):
            return obj.value
        # Handle UUID and Decimal by converting to string
        elif isinstance(obj, UUID) or isinstance(obj, Decimal):
            return str(obj)
        # Handle datetime/date objects by using ISO format
        elif isinstance(obj, datetime) or isinstance(obj, date):
            return obj.isoformat()
        # Fallback to the default JSONEncoder behavior
        return super().default(obj)

    def _handle_pydantic_model(self, obj):
        """Serialize a Pydantic model into a JSON‑serializable dict.

        This method uses ``model_dump`` (Pydantic v2) to obtain a dict
        representation of the model, then walks the resulting mapping to
        replace any values that are themselves ``BaseModel`` instances or
        ``Enum`` members. ``BaseModel`` instances are converted to their string
        representation to avoid nested objects that ``json.dumps`` cannot
        handle, while ``Enum`` members are replaced with their underlying
        value (typically a ``str`` or ``int``), ensuring they are JSON‑serializable.
        """
        try:
            data = obj.model_dump()
            # Remove circular references and convert Enums / nested BaseModels
            for key, value in data.items():
                if isinstance(value, BaseModel):
                    data[key] = str(value)  # Convert nested models to string
                elif isinstance(value, Enum):
                    data[key] = value.value  # Serialize enum as its value
            return data
        except RecursionError:
            # Fall back to string representation if circular reference is detected
            return str(obj)
