from uuid import UUID
from enum import Enum

from pydantic import BaseModel


class CrewJSONEncoder(json.JSONEncoder):
    def default(self, obj):
        # Handle Pydantic models
        if isinstance(obj, BaseModel):
            return self._handle_pydantic_model(obj)
        # Handle Enum types – serialize to their underlying value
        elif isinstance(obj, Enum):
            return obj.value
        # Handle UUID and Decimal by converting to string
        elif isinstance(obj, UUID) or isinstance(obj, Decimal):
            return str(obj)
        # Handle datetime and date objects
        elif isinstance(obj, datetime) or isinstance(obj, date):
            return obj.isoformat()
        # Fallback to the default JSONEncoder behaviour
        return super().default(obj)

    def _handle_pydantic_model(self, obj):

            data = obj.model_dump()
            # Remove circular references
            for key, value in data.items():
                if isinstance(value, BaseModel):
                    data[key] = str(
                        value
                    )  # Convert nested models to string representation
            return data
        except RecursionError:
            return str(
                obj
