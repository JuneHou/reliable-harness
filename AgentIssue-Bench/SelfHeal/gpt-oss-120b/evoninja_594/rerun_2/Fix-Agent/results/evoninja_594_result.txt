import { uploadedFilesAtom, userWorkspaceAtom } from "@/lib/store";
import { useUpdateUserFiles } from "@/lib/hooks/useUpdateUserFiles";
import { useAtom } from "jotai";
import { useEffect } from "react";

export default function WorkspaceFilesProvider({ children }: { children: React.ReactNode }) {
  const [uploadedFiles, setUploadedFiles] = useAtom(uploadedFilesAtom);
  const [userWorkspace] = useAtom(userWorkspaceAtom);
  const updateUserFiles = useUpdateUserFiles();

  useEffect(() => {
    // If there are no uploaded files, nothing to do.
    if (uploadedFiles.length === 0) {
      return;
    }

    // Ensure we have a workspace associated with a chat.
    // In the anonymous workspace case, the workspace may not have a valid chatId yet.
    // We skip processing until a proper workspace is available.
    if (!userWorkspace) {
      return;
    }

    // Some workspace implementations (e.g., SupabaseWorkspace) expose the chatId.
    // If the chatId is falsy, treat it as not ready.
    const maybeChatId = (userWorkspace as any).chatId;
    if (!maybeChatId) {
      return;
    }

    //eslint-disable-next-line @typescript-eslint/no-floating-promises
    (async () => {
      const decoder = new TextDecoder();
      const files = [...uploadedFiles];
      // Clear the uploaded files atom now that we are processing them.
      setUploadedFiles([]);

      await Promise.all(
        files.map((file) =>
          userWorkspace.writeFile(file.path, decoder.decode(file.content))
        )
      );
      await updateUserFiles(userWorkspace);
    })();
  }, [uploadedFiles, userWorkspace]);

  return <>{children}</>;
}
