import { useAtom } from "jotai";
import { useEffect } from "react";

export default function WorkspaceFilesProvider({ children }: { children: React.ReactNode }) {
  const [uploadedFiles, setUploadedFiles] = useAtom(uploadedFilesAtom);
  const [userWorkspace] = useAtom(userWorkspaceAtom);
  const updateUserFiles = useUpdateUserFiles();

  useEffect(() => {
    // eslint-disable-next-line @typescript-eslint/no-floating-promises
    (async () => {
      // If there are no uploaded files or the workspace is not yet associated with a chat, do nothing.
      if (uploadedFiles.length === 0) {
        return;
      }

      // The workspace may be an anonymous placeholder before a chat is created. In that case we defer
      // writing the files until the workspace is linked to a real chat (i.e., it has a chatId).
      // SupabaseWorkspace exposes a public readonly `chatId` property.
      if (!userWorkspace || !(userWorkspace as any).chatId) {
        // Workspace not ready for file persistence yet.
        return;
      }

      const decoder = new TextDecoder();
      const files = [...uploadedFiles];

      // Write all uploaded files to the current workspace (which should now be the chat workspace).
      await Promise.all(
        files.map((file) =>
          userWorkspace.writeFile(
            file.path,
            decoder.decode(file.content)
          )
        )
      );

      await updateUserFiles(userWorkspace);

      // Clear the uploaded files after they've been successfully written to the chat workspace.
      setUploadedFiles([]);
    })();
    // Re-run when the list of uploaded files changes or when the workspace changes (e.g., after a chat is created).
  }, [uploadedFiles, userWorkspace]);

  return <>{children}</>;
}
