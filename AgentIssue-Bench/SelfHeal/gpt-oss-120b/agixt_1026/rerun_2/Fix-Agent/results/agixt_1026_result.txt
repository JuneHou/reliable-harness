from ApiClient import ApiClient
from Extensions import Extensions


class voice_chat(Extensions):
    def __init__(self, **kwargs):
        # Determine agent name
        self.agent_name = kwargs.get("agent_name", "gpt4free")

        # Base prompt for voice chat
        self.voice_prompt = "Voice Chat without Commands"
        if kwargs.get("enabled_commands"):
            if len(kwargs["enabled_commands"]) > 0:
                self.voice_prompt = "Voice Chat"

        # Default TTS command (Streamlabs) – may be overridden by settings below
        self.tts_command = "Speak with TTS with Streamlabs Text to Speech"

        # Helper to interpret truthy values that may be either strings or booleans
        def _is_true(value):
            if isinstance(value, str):
                return value.lower() == "true"
            return bool(value)

        # Streamlabs TTS flag
        if "USE_STREAMLABS_TTS" in kwargs and _is_true(kwargs["USE_STREAMLABS_TTS"]):
            self.tts_command = "Speak with TTS with Streamlabs Text to Speech"

        # GTTS flag
        if "USE_GTTS" in kwargs and _is_true(kwargs["USE_GTTS"]):
            self.tts_command = "Speak with GTTS"

        # HuggingFace TTS flag – requires both flag and a non‑empty API key
        if (
            "USE_HUGGINGFACE_TTS" in kwargs
            and _is_true(kwargs["USE_HUGGINGFACE_TTS"])
            and kwargs.get("HUGGINGFACE_API_KEY")
        ):
            self.tts_command = "Read Audio with Huggingface"

        # ElevenLabs – overrides if a key is provided
        if kwargs.get("ELEVENLABS_API_KEY"):
            self.tts_command = "Speak with TTS Using Elevenlabs"

        # Register commands
        self.commands = {
            "Chat with Voice": self.chat_with_voice,
        }
        # Conversation name handling
        self.conversation_name = f"Voice Chat with {self.agent_name}"
        if "conversation_name" in kwargs:
            self.conversation_name = kwargs["conversation_name"]

    def chat_with_voice(
        self,
        base64_audio,
        conversation_results=3,
        context_results=10,
    ):
        """Transcribe the supplied audio, send the text to the LLM and return a TTS audio response.

        Args:
            base64_audio (str): Base64‑encoded audio data.
            conversation_results (int, optional): Number of recent conversation entries to include.
            context_results (int, optional): Number of context entries to include.

        Returns:
            Any: The result from the selected TTS command (typically base64 audio).
        """
        # Transcribe the audio to text.
        user_input = ApiClient.execute_command(
            agent_name=self.agent_name,
            command_name="Transcribe Base64 Audio",
            command_args={"base64_audio": base64_audio},
        )
        # Send the transcribed text to the agent.
        text_response = ApiClient.prompt_agent(
            agent_name=self.agent_name,
            prompt_name=self.voice_prompt,
            prompt_args={
                "user_input": user_input,
                "conversation_name": self.conversation_name,
                "conversation_results": conversation_results,
                "context_results": context_results,
            },
        )
        # Get the audio response from the TTS engine and return it.
        audio_response = ApiClient.execute_command(
            agent_name=self.agent_name,
            command_name=self.tts_command,
            command_args={"text": text_response},
        )
        return audio_response
