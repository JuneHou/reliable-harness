def listen(condition):
    """Decorator to register a method as a listener.

    The original implementation overwrote any previously set trigger methods,
    which broke the ability to stack multiple ``@listen`` decorators – for
    example ``@listen(and_)`` followed by ``@listen(step_a)`` and
    ``@listen(step_b)``.  The documentation expects that such stacking results in
    an ``AND`` condition that waits for *both* ``step_a`` and ``step_b`` before
    executing the listener.  To restore this behaviour we now merge conditions
    when the decorator is applied multiple times.
    """

    def decorator(func):
        # Initialise the trigger list if this is the first ``@listen`` applied.
        if not hasattr(func, "__trigger_methods__"):
            func.__trigger_methods__ = []
            func.__condition_type__ = "OR"  # default type

        # Helper to add a single method name to the existing trigger list.
        def _add_method(name: str):
            if name not in func.__trigger_methods__:
                func.__trigger_methods__.append(name)

        # Resolve the incoming ``condition`` and merge it with any existing ones.
        if isinstance(condition, str):
            _add_method(condition)
            # ``condition`` is a plain method name, keep the current type.
        elif (
            isinstance(condition, dict)
            and "type" in condition
            and "methods" in condition
        ):
            # When a dict is supplied we respect its ``type`` (AND/OR).
            # If this is the first condition we adopt its type, otherwise we
            # preserve the previously set type – stacking ``@listen(and_)``
            # followed by individual method names should keep the AND type.
            if not func.__trigger_methods__:
                func.__condition_type__ = condition["type"]
            else:
                # Preserve existing type; raise if there is a mismatch.
                if func.__condition_type__ != condition["type"]:
                    raise ValueError(
                        f"Conflicting condition types: existing {func.__condition_type__} vs new {condition['type']}"
                    )
            for m in condition["methods"]:
                _add_method(m)
        elif callable(condition) and hasattr(condition, "__name__"):
            _add_method(condition.__name__)
        else:
            raise ValueError(
                "Condition must be a method, string, or a result of or_() or and_()"
            )

        return func

    return decorator

def listen(condition):
    """Decorator to register a method as a listener.

    The original implementation overwrote any previously set trigger methods,
    which broke the ability to stack multiple ``@listen`` decorators – for
    example ``@listen(and_)`` followed by ``@listen(step_a)`` and
    ``@listen(step_b)``.  The documentation expects that such stacking results in
    an ``AND`` condition that waits for *both* ``step_a`` and ``step_b`` before
    executing the listener.  To restore this behaviour we now merge conditions
    when the decorator is applied multiple times.
    """

    def decorator(func):
        # Initialise the trigger list if this is the first ``@listen`` applied.
        if not hasattr(func, "__trigger_methods__"):
            func.__trigger_methods__ = []
            func.__condition_type__ = "OR"  # default type

        # Helper to add a single method name to the existing trigger list.
        def _add_method(name: str):
            if name not in func.__trigger_methods__:
                func.__trigger_methods__.append(name)

        # Resolve the incoming ``condition`` and merge it with any existing ones.
        if isinstance(condition, str):
            _add_method(condition)
            # ``condition`` is a plain method name, keep the current type.
        elif (
            isinstance(condition, dict)
            and "type" in condition
            and "methods" in condition
        ):
            # When a dict is supplied we respect its ``type`` (AND/OR).
            # If this is the first condition we adopt its type, otherwise we
            # preserve the previously set type – stacking ``@listen(and_)``
            # followed by individual method names should keep the AND type.
            if not func.__trigger_methods__:
                func.__condition_type__ = condition["type"]
            else:
                # Preserve existing type; raise if there is a mismatch.
                if func.__condition_type__ != condition["type"]:
                    raise ValueError(
                        f"Conflicting condition types: existing {func.__condition_type__} vs new {condition['type']}"
                    )
            for m in condition["methods"]:
                _add_method(m)
        elif callable(condition) and hasattr(condition, "__name__"):
            _add_method(condition.__name__)
        else:
            raise ValueError(
                "Condition must be a method, string, or a result of or_() or and_()"
            )

        return func

    return decorator
