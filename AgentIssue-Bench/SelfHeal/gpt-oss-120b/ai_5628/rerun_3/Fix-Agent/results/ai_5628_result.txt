import { JSONSchema7 } from '@ai-sdk/provider';
import * as z3 from 'zod/v3';
import * as z4 from 'zod/v4';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Recursively fix Zod record schemas that are converted to JSON Schema with
 * `type: undefined` but have `additionalProperties` defined. Vertex AI expects a
 * proper object schema, otherwise the generated OpenAPI specification contains
 * `undefined` for the property type which leads to request failures.
 */
function fixRecordSchema(schema: any): any {
  if (schema && typeof schema === 'object') {
    // If the schema represents a record (i.e., has additionalProperties) but no explicit type,
    // set the type to "object" so that OpenAPI generation works correctly.
    if (schema.type === undefined && schema.additionalProperties !== undefined) {
      schema.type = 'object';
    }

    // Recursively fix nested schemas (properties, items, additionalProperties, etc.).
    for (const key of Object.keys(schema)) {
      const value = (schema as any)[key];
      if (Array.isArray(value)) {
        (schema as any)[key] = value.map(fixRecordSchema);
      } else if (value && typeof value === 'object') {
        (schema as any)[key] = fixRecordSchema(value);
      }
    }
  }
  return schema;
}

export function zod3Schema<OBJECT>(
  zodSchema: z3.Schema<OBJECT, z3.ZodTypeDef, any>,
  options?: {
    /**
     * Enables support for references in the schema.
     * This is required for recursive schemas, e.g. with `z.lazy`.
     * However, not all language models and providers support such references.
     * Defaults to `false`.
     */
    useReferences?: boolean;
  },
): Schema<OBJECT> {
  // default to no references (to support openapi conversion for google)
  const useReferences = options?.useReferences ?? false;

  const rawJsonSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: useReferences ? 'root' : 'none',
    target: 'jsonSchema7', // note: openai mode breaks various gemini conversions
  }) as JSONSchema7;

  // Apply the fix for record schemas before creating the Schema wrapper.
  const fixedJsonSchema = fixRecordSchema(rawJsonSchema) as JSONSchema7;

  return jsonSchema(
    fixedJsonSchema,
    {
      validate: value => {
        const result = zodSchema.safeParse(value);
        return result.success
          ? { success: true, value: result.data }
          : { success: false, error: result.error };
      },
    },
  );
}

export function zod4Schema<OBJECT>(
  zodSchema: z4.ZodType<OBJECT, any>,
  options?: {
    /**
     * Enables support for references in the schema.
     * This is required for recursive schemas, e.g. with `z.lazy`.
     * However, not all language models and providers support such references.
