import { JSONSchema7 } from '@ai-sdk/provider';
import * as z3 from 'zod/v3';
import * as z4 from 'zod/v4';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Post-process a JSON Schema generated from Zod to ensure compatibility with
 * Vertex AI's OpenAPI conversion. Specifically, Zod's `z.record(z.string())`
 * (or any record) results in a schema that has `additionalProperties` but no
 * explicit `type`. Vertex AI expects the schema to have `type: "object"`
 * for such cases; otherwise the property appears as `undefined` in the
 * generated OpenAPI spec, causing request failures.
 */
function ensureObjectType(schema: JSONSchema7): JSONSchema7 {
  // If the schema already defines a type, keep it.
  if (schema.type) return schema;

  // When `additionalProperties` is present (record style) we infer an object.
  if (Object.prototype.hasOwnProperty.call(schema, 'additionalProperties')) {
    return { ...schema, type: 'object' } as JSONSchema7;
  }

  return schema;
}

export function zod3Schema<OBJECT>(
  zodSchema: z3.Schema<OBJECT, z3.ZodTypeDef, any>,
  options?: {
    /**
     * Enables support for references in the schema.
     * This is required for recursive schemas, e.g. with `z.lazy`.
     * However, not all language models and providers support such references.
     * Defaults to `false`.
     */
    useReferences?: boolean;
  },
): Schema<OBJECT> {
  // default to no references (to support openapi conversion for google)
  const useReferences = options?.useReferences ?? false;

  const rawSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: useReferences ? 'root' : 'none',
    target: 'jsonSchema7', // note: openai mode breaks various gemini conversions
  }) as JSONSchema7;

  const fixedSchema = ensureObjectType(rawSchema);

  return jsonSchema(fixedSchema, {
    validate: value => {
      const result = zodSchema.safeParse(value);
      return result.success

        : { success: false, error: result.error };
    },
  });
}
