import { JSONSchema7 } from '@ai-sdk/provider';
import * as z3 from 'zod/v3';
import * as z4 from 'zod/v4';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Ensure that object schemas always have a defined `properties` field.
 * This works around issues where certain schema generators (e.g., for Zod "record"
 * types) omit the `properties` key, resulting in `undefined` values that cause
 * downstream providers (such as Google Vertex AI) to generate malformed OpenAPI
 * specifications.
 */
function ensureObjectPropertiesDefined(schema: JSONSchema7): JSONSchema7 {
  if (schema.type === 'object' && schema.properties === undefined) {
    return { ...schema, properties: {} } as JSONSchema7;
  }
  return schema;
}

export function zod3Schema<OBJECT>(
  zodSchema: z3.Schema<OBJECT, z3.ZodTypeDef, any>,
  options?: {
    /**
     * Enables support for references in the schema.
     * This is required for recursive schemas, e.g. with `z.lazy`.
     * However, not all language models and providers support such references.
     * Defaults to `false`.
     */
    useReferences?: boolean;
  },
): Schema<OBJECT> {
  // default to no references (to support openapi conversion for google)
  const useReferences = options?.useReferences ?? false;

  const rawJsonSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: useReferences ? 'root' : 'none',
    target: 'jsonSchema7', // note: openai mode breaks various gemini conversions
  }) as JSONSchema7;

  const jsonSchemaWithProps = ensureObjectPropertiesDefined(rawJsonSchema);

  return jsonSchema(
    jsonSchemaWithProps,
    {
      validate: value => {
        const result = zodSchema.safeParse(value);
        return result.success
          ? { success: true, value: result.data }
          : { success: false, error: result.error };
      },
    },
  );
}

import * as z3 from 'zod/v3';
import * as z4 from 'zod/v4';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Patch JSON Schema generated from Zod to ensure compatibility with providers
 * that expect an explicit `type` property for object schemas. Zod records are
 * emitted with `additionalProperties` but without a `type: "object"`, which
 * results in an `undefined` property in the generated OpenAPI schema and causes
 * Vertex AI requests to fail.
 */
function patchRecordSchema(schema: JSONSchema7): JSONSchema7 {
  // If the schema already has a type, nothing to do.
  if (schema.type) return schema;

  // Zod's record type is represented with `additionalProperties`.
  // Vertex AI expects the schema to be an object with explicit type.
  if (schema.additionalProperties && typeof schema.additionalProperties === 'object') {
    // Clone to avoid mutating the original (just in case).
    const patched: JSONSchema7 = { ...schema, type: 'object' };
    return patched;
  }

  return schema;
}
