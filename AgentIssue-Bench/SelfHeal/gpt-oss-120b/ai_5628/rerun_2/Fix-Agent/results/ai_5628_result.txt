### packages/provider-utils/src/zod-schema.ts
import { JSONSchema7 } from '@ai-sdk/provider';
import * as z3 from 'zod/v3';
import * as z4 from 'zod/v4';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Post‑process a JSON Schema generated from a Zod schema to correctly represent
 * `z.record` definitions. The `zod-to-json-schema` library (used for Zod v3)
 * and the built‑in `toJSONSchema` (used for Zod v4) emit `{ additionalProperties:
 * true }` for a record without specifying the value type. This results in an
 * OpenAPI schema where the property is `undefined`, causing Vertex AI request
 * failures. We replace `additionalProperties: true` with a more precise schema
 * based on the Zod record's value type (string, number, boolean, etc.).
 */
function fixRecordSchema<OBJ>(zodSchema: any, jsonSchema: JSONSchema7): JSONSchema7 {
  // Detect ZodRecord (both v3 and v4 expose a `_def.typeName` of 'ZodRecord')
  const def = zodSchema?._def;
  if (!def || def.typeName !== 'ZodRecord') {
    return jsonSchema;
  }

  // Only modify if the generated schema looks like a generic record.
  if (jsonSchema.type === 'object' && jsonSchema.additionalProperties === true) {
    // Try to infer the value type from the Zod schema.
    const valueSchema = def.value; // Zod v3 stores the value schema here.
    let inferred: { type: string } | undefined;
    if (valueSchema && valueSchema._def && valueSchema._def.typeName) {
      const vt: string = valueSchema._def.typeName;
      if (vt === 'ZodString') inferred = { type: 'string' };
      else if (vt === 'ZodNumber') inferred = { type: 'number' };
      else if (vt === 'ZodBoolean') inferred = { type: 'boolean' };
      // For more complex types we fall back to `true` (any).
    }
    jsonSchema.additionalProperties = inferred ?? true;
  }
  return jsonSchema;
}

export function zod3Schema<OBJECT>(
  zodSchema: z3.Schema<OBJECT, z3.ZodTypeDef, any>,
  options?: {
    /**
     * Enables support for references in the schema.
     * This is required for recursive schemas, e.g. with `z.lazy`.
     * However, not all language models and providers support such references.
     * Defaults to `false`.
     */
    useReferences?: boolean;
  },
): Schema<OBJECT> {
  // default to no references (to support openapi conversion for google)
  const useReferences = options?.useReferences ?? false;

  const rawJsonSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: useReferences ? 'root' : 'none',
    target: 'jsonSchema7', // note: openai mode breaks various gemini conversions
  }) as JSONSchema7;

  const correctedSchema = fixRecordSchema(zodSchema, rawJsonSchema);

  return jsonSchema(
    correctedSchema,
    {
      validate: value => {
        const result = zodSchema.safeParse(value);
        return result.success
          ? { success: true, value: result.data }
          : { success: false, error: result.error };
      },
    },
  );
}

  zodSchema: z4.ZodType<OBJECT, any>,
  options?: {
    /**
     * Enables support for references in the schema.
     * This is required for recursive schemas, e.g. with `z.lazy`.
     * However, not all language models and providers support such references.
     * Defaults to `false`.

  }) as JSONSchema7;

  const correctedSchema = fixRecordSchema(zodSchema, z4JSONSchema);

  return jsonSchema(correctedSchema, {

    },
  });
}

export function isZod4Schema(
  zodSchema: z4.ZodType<any, any> | z3.Schema<any, z3.ZodTypeDef, any>,
): zodSchema is z4.ZodType<any, any> {
  // https://zod.dev/library-authors?id=how-to-support-zod-3-and-zod-4-simultaneously
  return '_zod' in zodSchema;
}

export function zodSchema<OBJECT>(
  zodSchema: z4.ZodType<OBJECT, any> | z3.Schema<OBJECT, z3.ZodTypeDef, any>,
  options?: {
    /**
     * Enables support for references in the schema.
     * This is required for recursive schemas, e.g. with `z.lazy`.
     * However, not all language models and providers support such references.
     * Defaults to `false`.

  },
): Schema<OBJECT> {
  if (isZod4Schema(zodSchema)) {
    return zod4Schema(zodSchema, options);
  } else {
    return zod3Schema(zodSchema, options);
  }
}
