 * Works for both Zod v3 and v4.
 */
function isZodRecord(schema: any): boolean {
  // Zod v4 uses _def.typeName === 'ZodRecord'
  // Zod v3 also has _def.typeName === 'ZodRecord'
  return (
    schema &&
    typeof schema === 'object' &&
    '_def' in schema &&
    (schema._def?.typeName === 'ZodRecord' ||
      // fallback for older versions where typeName may be undefined but constructor name matches
      (schema.constructor && schema.constructor.name === 'ZodRecord'))
  );
}

     * Defaults to `false`.

): Schema<OBJECT> {
