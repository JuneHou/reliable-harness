import { JSONSchema7, JSONSchema7Definition } from 'json-schema';
import { z } from 'zod';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Recursively walk a JSON Schema and ensure that optional fields (represented as
 * `anyOf` with a `null` type) retain the original description. The `zod-to-json-schema`
 * library drops the description when a field is wrapped with `.optional()`. OpenAI
 * expects the description to be present at the property level, so we copy the
 * description from the non‑null subschema to the parent schema.
 */
function propagateOptionalDescriptions(schema: JSONSchema7): JSONSchema7 {
  // Process properties if present
  if (schema.properties) {
    const newProps: Record<string, JSONSchema7Definition> = {};
    for (const [key, property] of Object.entries(schema.properties)) {
      // Preserve boolean schemas unchanged
      if (typeof property === 'boolean') {
        newProps[key] = property;
        continue;
      }

      // Work with a mutable copy of the property schema
      let prop = property as JSONSchema7;

      // If the property uses anyOf to represent an optional value, copy description
      if (Array.isArray(prop.anyOf)) {
        const nonNullSchema = prop.anyOf.find(
          (s) => typeof s !== 'boolean' && (s as JSONSchema7).type !== 'null',
        ) as JSONSchema7 | undefined;
        if (nonNullSchema && nonNullSchema.description && !prop.description) {
          prop.description = nonNullSchema.description;
        }
      }

      // Recurse into subschemas (objects, arrays, anyOf entries, etc.)
      if (prop.type === 'object' && prop.properties) {
        prop = propagateOptionalDescriptions(prop);
      } else if (prop.type === 'array' && prop.items && typeof prop.items !== 'boolean') {
        if (!Array.isArray(prop.items)) {
          prop.items = propagateOptionalDescriptions(prop.items as JSONSchema7);
        }
      } else if (Array.isArray(prop.anyOf)) {
        prop.anyOf = prop.anyOf.map((sub) => {
          if (typeof sub === 'boolean') return sub;
          return propagateOptionalDescriptions(sub as JSONSchema7);
        }) as JSONSchema7Definition[];
      }

      newProps[key] = prop as JSONSchema7Definition;
    }
    schema.properties = newProps;
  }

  // Handle anyOf at the root level similarly
  if (Array.isArray(schema.anyOf)) {
    schema.anyOf = schema.anyOf.map((sub) => {
      if (typeof sub === 'boolean') return sub;
      return propagateOptionalDescriptions(sub as JSONSchema7);
    }) as JSONSchema7Definition[];
  }

  return schema;
}

export function zodSchema<OBJECT>(
  zodSchema: z.Schema<OBJECT, z.ZodTypeDef, any>,
): Schema<OBJECT> {
  // Convert the Zod schema to JSON Schema targeting OpenAI's strict mode.
  const rawJsonSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: 'none', // no references (to support openapi conversion for google)
    // `zod-to-json-schema` can lose descriptions on optional fields; we will
    // fix that after conversion.
    target: 'openAi', // openai strict mode compatible
  }) as JSONSchema7;

  // Ensure optional field descriptions are preserved.
  const fixedJsonSchema = propagateOptionalDescriptions(rawJsonSchema);

  return jsonSchema(fixedJsonSchema, {
    validate: (value) => {
      const result = zodSchema.safeParse(value);
      return result.success
        ? { success: true, value: result.data }
        : { success: false, error: result.error };
    },
  });
}
