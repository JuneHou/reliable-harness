import { z } from 'zod';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Convert a Zod schema to our internal {@link Schema} while preserving field
 * descriptions for optional fields.
 *
 * The `zod-to-json-schema` library drops the `description` property when a
 * field is marked as optional (e.g., `z.string().optional()`). In the OpenAI
 * target format this results in a schema where the description is nested inside
 * an `anyOf` clause rather than being attached to the top‑level property.
 *
 * To work around this we post‑process the generated JSON schema and lift the
 * description from the non‑null member of an `anyOf` array onto the property
 * object itself.
 */
export function zodSchema<OBJECT>(
  zodSchema: z.Schema<OBJECT, z.ZodTypeDef, any>,
): Schema<OBJECT> {
  // Generate the base JSON schema using the library.
  const baseSchema = zodToJsonSchema(zodSchema, {
    $refStrategy: 'none', // no references (to support openapi conversion for google)
    // Preserve descriptions for optional properties. The library does not
    // expose a direct option for this, so we will manually adjust the result.
    target: 'openAi', // openai strict mode compatible
  }) as JSONSchema7;

  // Post‑process the schema to ensure optional fields retain their descriptions.
  const liftDescriptions = (schema: any): void => {
    if (!schema || typeof schema !== 'object') return;
    // Recurse into object properties.
    if (schema.properties && typeof schema.properties === 'object') {
      for (const key of Object.keys(schema.properties)) {
        const prop = schema.properties[key];
        // If the property lacks a description but has an anyOf (typical for
        // optional fields) we look for a non‑null schema inside anyOf that has a
        // description and promote it.
        if (prop && typeof prop === 'object' && !prop.description && Array.isArray(prop.anyOf)) {
          const nonNull = prop.anyOf.find((sub: any) => sub.type !== 'null');
          if (nonNull && nonNull.description) {
            prop.description = nonNull.description;
          }
        }
        // Recurse deeper for nested objects/arrays.
        liftDescriptions(prop);
      }
    }
    // Recurse for array items.
    if (schema.items) {
      liftDescriptions(schema.items);
    }
    // Recurse for combinators.
    if (Array.isArray(schema.anyOf)) {
      schema.anyOf.forEach(liftDescriptions);
    }
    if (Array.isArray(schema.oneOf)) {
      schema.oneOf.forEach(liftDescriptions);
    }
    if (Array.isArray(schema.allOf)) {
      schema.allOf.forEach(liftDescriptions);
    }
  };

  liftDescriptions(baseSchema);

  return jsonSchema(baseSchema, {
    validate: value => {
      const result = zodSchema.safeParse(value);
      return result.success
        ? { success: true, value: result.data }
        : { success: false, error: result.error };
    },
  });
}
