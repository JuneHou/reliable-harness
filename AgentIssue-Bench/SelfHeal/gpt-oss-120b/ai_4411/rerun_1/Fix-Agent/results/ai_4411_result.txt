import { JSONSchema7 } from 'json-schema';
import { z } from 'zod';
import zodToJsonSchema from 'zod-to-json-schema';
import { jsonSchema, Schema } from './schema';

/**
 * Convert a Zod schema to our internal {@link Schema} representation while preserving
 * field descriptions even when the field is marked as optional via `.optional()`.
 *
 * The `zod-to-json-schema` library drops the description on optional fields when
 * using the `openAi` target. We work around this by post‑processing the generated
 * JSON Schema and copying the description from the original Zod property definition
 * if it is missing.
 */
export function zodSchema<OBJECT>(
  zodSchema: z.Schema<OBJECT, z.ZodTypeDef, any>,
): Schema<OBJECT> {
  // First, generate the JSON Schema using the existing options.
  const rawJson = zodToJsonSchema(zodSchema, {
    $refStrategy: 'none', // no references (to support openapi conversion for google)
+    // The `target` option "openAi" produces a schema compatible with the OpenAI
+    // function calling specification, but it strips descriptions from optional
+    // fields. We keep the option for compatibility and later restore any missing
+    // descriptions.
    target: 'openAi', // openai strict mode compatible
  }) as JSONSchema7;
+
+  // Post‑process the generated schema to re‑attach missing descriptions for
+  // optional properties.
+  const enhancedJson = preserveOptionalDescriptions(rawJson, zodSchema);

  return jsonSchema(enhancedJson, {
    validate: value => {
      const result = zodSchema.safeParse(value);
      return result.success
        ? { success: true, value: result.data }
        : { success: false, error: result.error };
    },
  });
}

/**
 * Walk through an object schema and ensure that each property retains its Zod
 * description. This is needed because `zod-to-json-schema` drops descriptions on
 * optional fields when the `target` is set to "openAi".
 */
function preserveOptionalDescriptions(
  json: JSONSchema7,
  zod: z.Schema<any>,
): JSONSchema7 {
  // Only handle object schemas with a defined `properties` map.
  if (json.type !== 'object' || typeof json.properties !== 'object') {
    return json;
  }

  // Extract the shape definition from the Zod object schema. The internal API
  // provides a `shape` method on the ZodObject definition.
+  const shapeFn = (zod as any)._def?.shape;
+  if (typeof shapeFn !== 'function') {
+    return json;
+  }
+  const shape = shapeFn(); // shape: Record<string, ZodTypeAny>
+
+  for (const [key, propertySchema] of Object.entries(shape)) {
+    const jsonProp = (json.properties as Record<string, any>)[key];
+    if (!jsonProp) continue;
+
+    // If the JSON property already has a description, nothing to do.
+    if (jsonProp.description) continue;
+
+    // Zod stores the description on the inner schema's definition under
+    // `_def.description`.
+    const description = (propertySchema as any)._def?.description;
+    if (description) {
+      jsonProp.description = description;
+    }
+  }
+
+  return json;
+}
