# Keywords of a JSON schema whose value is a mapping of subschemas
_SCHEMA_MAPPING_KEYWORDS = ("properties", "patternProperties", "$defs", "definitions", "dependentSchemas")

# Keywords of a JSON schema whose value is a subschema (or a list of subschemas)
_SCHEMA_KEYWORDS = (
    "items",
    "additionalItems",
    "contains",
    "not",
    "if",
    "then",
    "else",
    "propertyNames",
    "allOf",
    "anyOf",
    "oneOf",
    "prefixItems",
)


def _is_object_schema(schema: Dict[str, Any]) -> bool:
    """
    Checks whether a JSON schema fragment describes an object.

    :param schema: The JSON schema fragment to check.
    :returns: `True` if the fragment describes an object, `False` otherwise.
    """
    schema_type = schema.get("type")
    if schema_type == "object":
        return True
    if isinstance(schema_type, list) and "object" in schema_type:
        return True
    # Schemas without an explicit type that use object-only keywords are objects too.
    # Note: schemas consisting only of a "$ref" are intentionally not considered objects,
    # as the referenced subschema is handled separately.
    return schema_type is None and any(
        keyword in schema for keyword in ("properties", "required", "additionalProperties", "patternProperties")
    )


def _convert_to_strict_schema(schema: Any) -> Any:
    """
    Recursively sets `additionalProperties` to `False` on all objects of a JSON schema.

    OpenAI's strict mode requires `additionalProperties: false` to be set on every object of the
    provided schema, including nested ones. This is not the case for schemas generated from
    Pydantic models or from Haystack components (e.g. `ComponentTool`), which is why the schema
    is adapted before being sent to the OpenAI API.

    :param schema: The JSON schema fragment to adapt. It's modified in place.
    :returns: The adapted JSON schema fragment.
    """
    if isinstance(schema, list):
        return [_convert_to_strict_schema(item) for item in schema]

    if not isinstance(schema, dict):
        return schema

    for key, value in schema.items():
        if key in _SCHEMA_MAPPING_KEYWORDS and isinstance(value, dict):
            schema[key] = {name: _convert_to_strict_schema(subschema) for name, subschema in value.items()}
        elif key in _SCHEMA_KEYWORDS:
            schema[key] = _convert_to_strict_schema(value)

    if _is_object_schema(schema):
        schema["additionalProperties"] = False

    return schema


def _convert_tool_to_openai_format(tool: Tool, strict: bool = False) -> Dict[str, Any]:
    """
    Converts a Haystack `Tool` into the tool format expected by the OpenAI Chat Completions API.

    :param tool: The Haystack `Tool` to convert.
    :param strict: Whether to enable OpenAI's strict schema adherence for the tool.
    :returns: The tool definition in the format expected by the OpenAI API.
    """
    # We deep copy the tool spec to avoid mutating the spec of the original tool
    function_spec = copy.deepcopy(tool.tool_spec)

    if strict:
        function_spec["strict"] = True
        parameters = function_spec.get("parameters") or {"type": "object", "properties": {}}
        function_spec["parameters"] = _convert_to_strict_schema(parameters)

    return {"type": "function", "function": function_spec}

    def _prepare_api_call(  # noqa: PLR0913
        self,
        *,
        messages: List[ChatMessage],
        streaming_callback: Optional[StreamingCallbackT] = None,
        generation_kwargs: Optional[Dict[str, Any]] = None,
        tools: Optional[List[Tool]] = None,
        tools_strict: Optional[bool] = None,
    ) -> Dict[str, Any]:
        # update generation kwargs by merging with the generation kwargs passed to the run method
        generation_kwargs = {**self.generation_kwargs, **(generation_kwargs or {})}

        # adapt ChatMessage(s) to the format expected by the OpenAI API
        openai_formatted_messages = [message.to_openai_dict_format() for message in messages]

        tools = tools or self.tools
        tools_strict = tools_strict if tools_strict is not None else self.tools_strict
        _check_duplicate_tool_names(tools)

        openai_tools = {}
        if tools:
            openai_tools = {"tools": [_convert_tool_to_openai_format(t, bool(tools_strict)) for t in tools]}

        is_streaming = streaming_callback is not None
        num_responses = generation_kwargs.pop("n", 1)
        if is_streaming and num_responses > 1:
            raise ValueError("Cannot stream multiple responses, please set n=1.")

        return {
            "model": self.model,
            "messages": openai_formatted_messages,  # type: ignore[arg-type] # openai expects list of specific message types
            "stream": streaming_callback is not None,
            "n": num_responses,
            **openai_tools,
            **generation_kwargs,
        }

    def _handle_stream_response(self, chat_completion: Stream, callback: SyncStreamingCallbackT) -> List[ChatMessage]:
        chunks: List[StreamingChunk] = []
        chunk = None

        for chunk in chat_completion:  # pylint: disable=not-an-iterable
            assert len(chunk.choices) == 1, "Streaming responses should have only one choice."
            chunk_delta: StreamingChunk = self._convert_chat_completion_chunk_to_streaming_chunk(chunk)
            chunks.append(chunk_delta)

            callback(chunk_delta)

        return [self._convert_streaming_chunks_to_chat_message(chunk, chunks)]

    async def _handle_async_stream_response(
        self, chat_completion: AsyncStream, callback: AsyncStreamingCallbackT
    ) -> List[ChatMessage]:
        chunks: List[StreamingChunk] = []
        chunk = None

        async for chunk in chat_completion:  # pylint: disable=not-an-iterable
            assert len(chunk.choices) == 1, "Streaming responses should have only one choice."
            chunk_delta: StreamingChunk = self._convert_chat_completion_chunk_to_streaming_chunk(chunk)
            chunks.append(chunk_delta)

            await callback(chunk_delta)

        return [self._convert_streaming_chunks_to_chat_message(chunk, chunks)]

    def _check_finish_reason(self, meta: Dict[str, Any]) -> None:
        if meta["finish_reason"] == "length":
            logger.warning(
                "The completion for index {index} has been truncated before reaching a natural stopping point. "
                "Increase the max_tokens parameter to allow for longer completions.",
                index=meta["index"],
                finish_reason=meta["finish_reason"],
            )
        if meta["finish_reason"] == "content_filter":
            logger.warning(
                "The completion for index {index} has been truncated due to the content filter.",
                index=meta["index"],
                finish_reason=meta["finish_reason"],
            )

    def _convert_streaming_chunks_to_chat_message(
        self, chunk: ChatCompletionChunk, chunks: List[StreamingChunk]
    ) -> ChatMessage:
        """
        Connects the streaming chunks into a single ChatMessage.

        :param chunk: The last chunk returned by the OpenAI API.
        :param chunks: The list of all `StreamingChunk` objects.
        """
        text = "".join([chunk.content for chunk in chunks])
        tool_calls = []

        # Process tool calls if present in any chunk
        tool_call_data: Dict[str, Dict[str, str]] = {}  # Track tool calls by index
        for chunk_payload in chunks:
            tool_calls_meta = chunk_payload.meta.get("tool_calls")
            if tool_calls_meta is not None:
                for delta in tool_calls_meta:
                    # We use the index of the tool call to track it across chunks since the ID is not always provided
                    if delta.index not in tool_call_data:
                        tool_call_data[delta.index] = {"id": "", "name": "", "arguments": ""}

                    # Save the ID if present
                    if delta.id is not None:
                        tool_call_data[delta.index]["id"] = delta.id

                    if delta.function is not None:
                        if delta.function.name is not None:
                            tool_call_data[delta.index]["name"] += delta.function.name
                        if delta.function.arguments is not None:
                            tool_call_data[delta.index]["arguments"] += delta.function.arguments

        # Convert accumulated tool call data into ToolCall objects
        for call_data in tool_call_data.values():
            try:
                arguments = json.loads(call_data["arguments"])
                tool_calls.append(ToolCall(id=call_data["id"], tool_name=call_data["name"], arguments=arguments))
            except json.JSONDecodeError:
                logger.warning(
                    "OpenAI returned a malformed JSON string for tool call arguments. This tool call "
                    "will be skipped. To always generate a valid JSON, set `tools_strict` to `True`. "
                    "Tool call ID: {_id}, Tool name: {_name}, Arguments: {_arguments}",
                    _id=call_data["id"],
                    _name=call_data["name"],
                    _arguments=call_data["arguments"],
                )

        meta = {
            "model": chunk.model,
            "index": 0,
            "finish_reason": chunk.choices[0].finish_reason,
            "completion_start_time": chunks[0].meta.get("received_at"),  # first chunk received
            "usage": {},  # we don't have usage data for streaming responses
        }

        return ChatMessage.from_assistant(text=text or None, tool_calls=tool_calls, meta=meta)

    def _convert_chat_completion_to_chat_message(self, completion: ChatCompletion, choice: Choice) -> ChatMessage:
        """
        Converts the non-streaming response from the OpenAI API to a ChatMessage.

        :param completion: The completion returned by the OpenAI API.
        :param choice: The choice returned by the OpenAI API.
        :return: The ChatMessage.
        """
        message: ChatCompletionMessage = choice.message
        text = message.content
        tool_calls = []
        if openai_tool_calls := message.tool_calls:
            for openai_tc in openai_tool_calls:
                arguments_str = openai_tc.function.arguments
                try:
                    arguments = json.loads(arguments_str)
                    tool_calls.append(ToolCall(id=openai_tc.id, tool_name=openai_tc.function.name, arguments=arguments))
                except json.JSONDecodeError:
                    logger.warning(
                        "OpenAI returned a malformed JSON string for tool call arguments. This tool call "
                        "will be skipped. To always generate a valid JSON, set `tools_strict` to `True`. "
                        "Tool call ID: {_id}, Tool name: {_name}, Arguments: {_arguments}",
                        _id=openai_tc.id,
                        _name=openai_tc.function.name,
                        _arguments=arguments_str,
                    )

        chat_message = ChatMessage.from_assistant(text=text, tool_calls=tool_calls)
        chat_message._meta.update(
            {
                "model": completion.model,
                "index": choice.index,
                "finish_reason": choice.finish_reason,
                "usage": dict(completion.usage or {}),
            }
        )
        return chat_message

    def _convert_chat_completion_chunk_to_streaming_chunk(self, chunk: ChatCompletionChunk) -> StreamingChunk:
        """
        Converts the streaming response chunk from the OpenAI API to a StreamingChunk.

        :param chunk: The chunk returned by the OpenAI API.

        :returns:
            The StreamingChunk.
        """
        # we stream the content of the chunk if it's not a tool or function call
        choice: ChunkChoice = chunk.choices[0]
        content = choice.delta.content or ""
        chunk_message = StreamingChunk(content)
        # but save the tool calls and function call in the meta if they are present
        # and then connect the chunks in the _convert_streaming_chunks_to_chat_message method
        chunk_message.meta.update(
            {
                "model": chunk.model,
                "index": choice.index,
                "tool_calls": choice.delta.tool_calls,
                "finish_reason": choice.finish_reason,
                "received_at": datetime.now().isoformat(),
            }
        )
        return chunk_message

Setting `function_spec["strict"] = True` is equivalent to the previous `{"strict": tools_strict}` behavior. That part is sound.

### 2. Structural validation — FAILS
`validate_fixed_code` → **REJECT**: the file-path marker line `### haystack/components/generators/chat/openai.py` is missing from the fixed code (the diff shows it removed at the top of the file). The fixed artifact therefore does not satisfy the required output format.

### 3. The provided test cannot confirm the fix
The test patches `haystack...openai.OpenAIChatGenerator._prepare_api_call` — i.e. it replaces *the very method that builds the tool schema* with a stub, then inside the stub inspects `kwargs.get("tools", [])`.

Two fatal problems:

1. `run()` calls `_prepare_api_call(..., tools=tools, tools_strict=tools_strict)` where `tools` is `run`'s own argument, which defaults to `None`. So `kwargs["tools"]` is `None` (or a raw list of `Tool` objects, never the prepared OpenAI payload). The stub then does `tools_payload[0].get("function", {}).get("parameters", {})` guarded by `if tools_payload`, so it evaluates to `{}`.
2. Consequently `has_additional_properties` is always `False`. For `version == "fixed"` the test hits `exit(1)` → "FIX NOT CONFIRMED". The harness would have to observe the output of the real `_prepare_api_call`/`_prepare_tools`, which the stub prevents entirely.

So the test fails on the fixed code by construction, independent of whether the fix is correct.

### 4. Additional risk in the fix
Unconditionally overwriting `required` with *all* property names silently converts every optional parameter into a required one. That matches OpenAI's strict-mode constraint, but it changes tool semantics for schemas that legitimately declare optional arguments, and it was not covered/verified by the supplied test.

### Verdict
**REJECT** — the fixed code fails the structural format check (missing module header line), and the supplied regression test cannot validate the fix (it stubs out `_prepare_api_call` and reads the unresolved `tools` kwarg, guaranteeing `additionalProperties` is never observed and forcing `exit(1)` on the fixed version). The underlying fix logic looks close to correct, but as delivered it is neither format-valid nor test-validated.

    @component.output_types(replies=List[ChatMessage])
    def run(
        self,
        messages: List[ChatMessage],
        streaming_callback: Optional[StreamingCallbackT] = None,
        generation_kwargs: Optional[Dict[str, Any]] = None,
        *,
        tools: Optional[List[Tool]] = None,
        tools_strict: Optional[bool] = None,
    ):
        """
        Invokes chat completion based on the provided messages and generation parameters.

        :param messages:
            A list of ChatMessage instances representing the input messages.
        :param streaming_callback:
            A callback function that is called when a new token is received from the stream.
        :param generation_kwargs:
            Additional keyword arguments for text generation. These parameters will
            override the parameters passed during component initialization.
            For details on OpenAI API parameters, see [OpenAI documentation](https://platform.openai.com/docs/api-reference/chat/create).
        :param tools:
            A list of tools for which the model can prepare calls. If set, it will override the `tools` parameter set
            during component initialization.
        :param tools_strict:
            Whether to enable strict schema adherence for tool calls. If set to `True`, the model will follow exactly
            the schema provided in the `parameters` field of the tool definition, but this may increase latency.
            If set, it will override the `tools_strict` parameter set during component initialization.

        :returns:
            A dictionary with the following key:
            - `replies`: A list containing the generated responses as ChatMessage instances.
        """
        if len(messages) == 0:
            return {"replies": []}

        streaming_callback = streaming_callback or self.streaming_callback

        api_args = self._prepare_api_call(
            messages=messages,
            streaming_callback=streaming_callback,
            generation_kwargs=generation_kwargs,
            tools=tools,
            tools_strict=tools_strict,
        )
        chat_completion: Union[Stream[ChatCompletionChunk], ChatCompletion] = self.client.chat.completions.create(
            **api_args
        )

        is_streaming = isinstance(chat_completion, Stream)
        assert is_streaming or streaming_callback is None

        if is_streaming:
            completions = self._handle_stream_response(
                chat_completion,  # type: ignore
                streaming_callback,  # type: ignore
            )
        else:
            assert isinstance(chat_completion, ChatCompletion), "Unexpected response type for non-streaming request."
            completions = [
                self._convert_chat_completion_to_chat_message(chat_completion, choice)
                for choice in chat_completion.choices
            ]

        # before returning, do post-processing of the completions
        for message in completions:
            self._check_finish_reason(message.meta)

        return {"replies": completions}

    def _prepare_api_call(  # noqa: PLR0913
        self,
        *,
        messages: List[ChatMessage],
        streaming_callback: Optional[StreamingCallbackT] = None,
        generation_kwargs: Optional[Dict[str, Any]] = None,
        tools: Optional[List[Tool]] = None,
        tools_strict: Optional[bool] = None,
    ) -> Dict[str, Any]:
        # update generation kwargs by merging with the generation kwargs passed to the run method
        generation_kwargs = {**self.generation_kwargs, **(generation_kwargs or {})}

        # adapt ChatMessage(s) to the format expected by the OpenAI API
        openai_formatted_messages = [message.to_openai_dict_format() for message in messages]

        tools = tools or self.tools
        tools_strict = tools_strict if tools_strict is not None else self.tools_strict
        _check_duplicate_tool_names(tools)

        openai_tools = {}
        if tools:
            openai_tools = self._prepare_tools(tools, tools_strict)

        is_streaming = streaming_callback is not None
        num_responses = generation_kwargs.pop("n", 1)
        if is_streaming and num_responses > 1:
            raise ValueError("Cannot stream multiple responses, please set n=1.")

        return {
            "model": self.model,
            "messages": openai_formatted_messages,  # type: ignore[arg-type] # openai expects list of specific message types
            "stream": streaming_callback is not None,
            "n": num_responses,
            **openai_tools,
            **generation_kwargs,
        }

    @staticmethod
    def _prepare_tools(tools: List[Tool], tools_strict: bool) -> Dict[str, Any]:
        """
        Prepare and format the tools for the OpenAI API.

        If `tools_strict` is `True`, the JSON schema of every tool is converted so that it fulfills OpenAI's
        strict mode requirements, i.e. `additionalProperties` is set to `False` and all properties are marked
        as required. The conversion is applied recursively, so nested objects (e.g. the ones generated for a
        `ComponentTool` or for tools whose parameters are Pydantic models) are handled as well.

        :param tools: A list of Tool objects.
        :param tools_strict: Whether to enable strict schema adherence for tool calls.
        :returns: A dictionary with the `tools` key holding the tool definitions expected by the OpenAI API.
        """
        tool_definitions = []
        for tool in tools:
            # Deepcopy to make sure we don't modify the original tool specs that are reused across calls
            function_spec = deepcopy(tool.tool_spec)
            if tools_strict:
                function_spec["strict"] = True
                parameters = function_spec.get("parameters")
                if isinstance(parameters, dict):
                    function_spec["parameters"] = OpenAIChatGenerator._convert_to_recursive_strict(parameters)
            tool_definitions.append({"type": "function", "function": function_spec})

        return {"tools": tool_definitions}

    @staticmethod
    def _convert_to_recursive_strict(schema: Dict[str, Any]) -> Dict[str, Any]:
        """
        Recursively convert a JSON schema to the format required by OpenAI's strict mode.

        OpenAI's strict mode requires that:
        1. every object defines `additionalProperties: false`
        2. every object lists all of its properties in `required`

        :param schema: The JSON schema to convert.
        :returns: The converted JSON schema.
        """
        if not isinstance(schema, dict):
            return schema

        # Objects (both explicitly typed ones and schemas that only define properties)
        if schema.get("type") == "object" or "properties" in schema:
            properties = schema.get("properties")
            if isinstance(properties, dict):
                schema["additionalProperties"] = False
                # In strict mode all properties must be marked as required
                schema["required"] = list(properties.keys())
                for property_schema in properties.values():
                    OpenAIChatGenerator._convert_to_recursive_strict(property_schema)

        # Arrays and tuples
        items = schema.get("items")
        if isinstance(items, dict):
            OpenAIChatGenerator._convert_to_recursive_strict(items)
        elif isinstance(items, list):
            for item_schema in items:
                OpenAIChatGenerator._convert_to_recursive_strict(item_schema)

        # Composition keywords
        for keyword in ("anyOf", "oneOf", "allOf", "prefixItems"):
            subschemas = schema.get(keyword)
            if isinstance(subschemas, list):
                for subschema in subschemas:
                    OpenAIChatGenerator._convert_to_recursive_strict(subschema)

        # Definitions used to describe nested/reused models
        definitions = schema.get("$defs") or schema.get("definitions")
        if isinstance(definitions, dict):
            for definition_schema in definitions.values():
                OpenAIChatGenerator._convert_to_recursive_strict(definition_schema)

        return schema

    def _handle_stream_response(self, chat_completion: Stream, callback: SyncStreamingCallbackT) -> List[ChatMessage]:
        chunks: List[StreamingChunk] = []
        chunk = None

        for chunk in chat_completion:  # pylint: disable=not-an-iterable
            assert len(chunk.choices) == 1, "Streaming responses should have only one choice."
            chunk_delta: StreamingChunk = self._convert_chat_completion_chunk_to_streaming_chunk(chunk)
            chunks.append(chunk_delta)

            callback(chunk_delta)

        return [self._convert_streaming_chunks_to_chat_message(chunk, chunks)]

    async def _handle_async_stream_response(
        self, chat_completion: AsyncStream, callback: AsyncStreamingCallbackT
    ) -> List[ChatMessage]:
        chunks: List[StreamingChunk] = []
        chunk = None

        async for chunk in chat_completion:  # pylint: disable=not-an-iterable
            assert len(chunk.choices) == 1, "Streaming responses should have only one choice."
            chunk_delta: StreamingChunk = self._convert_chat_completion_chunk_to_streaming_chunk(chunk)
            chunks.append(chunk_delta)

            await callback(chunk_delta)

        return [self._convert_streaming_chunks_to_chat_message(chunk, chunks)]

    def _check_finish_reason(self, meta: Dict[str, Any]) -> None:
        if meta["finish_reason"] == "length":
            logger.warning(
                "The completion for index {index} has been truncated before reaching a natural stopping point. "
                "Increase the max_tokens parameter to allow for longer completions.",
                index=meta["index"],
                finish_reason=meta["finish_reason"],
            )
        if meta["finish_reason"] == "content_filter":
            logger.warning(
                "The completion for index {index} has been truncated due to the content filter.",
                index=meta["index"],
                finish_reason=meta["finish_reason"],
            )

    def _convert_streaming_chunks_to_chat_message(
        self, chunk: ChatCompletionChunk, chunks: List[StreamingChunk]
    ) -> ChatMessage:
        """
        Connects the streaming chunks into a single ChatMessage.

        :param chunk: The last chunk returned by the OpenAI API.
        :param chunks: The list of all `StreamingChunk` objects.
        """
        text = "".join([chunk.content for chunk in chunks])
        tool_calls = []

        # Process tool calls if present in any chunk
        tool_call_data: Dict[str, Dict[str, str]] = {}  # Track tool calls by index
        for chunk_payload in chunks:
            tool_calls_meta = chunk_payload.meta.get("tool_calls")
            if tool_calls_meta is not None:
                for delta in tool_calls_meta:
                    # We use the index of the tool call to track it across chunks since the ID is not always provided
                    if delta.index not in tool_call_data:
                        tool_call_data[delta.index] = {"id": "", "name": "", "arguments": ""}

                    # Save the ID if present
                    if delta.id is not None:
                        tool_call_data[delta.index]["id"] = delta.id

                    if delta.function is not None:
                        if delta.function.name is not None:
                            tool_call_data[delta.index]["name"] += delta.function.name
                        if delta.function.arguments is not None:
                            tool_call_data[delta.index]["arguments"] += delta.function.arguments

        # Convert accumulated tool call data into ToolCall objects
        for call_data in tool_call_data.values():
            try:
                arguments = json.loads(call_data["arguments"])
                tool_calls.append(ToolCall(id=call_data["id"], tool_name=call_data["name"], arguments=arguments))
            except json.JSONDecodeError:
                logger.warning(
                    "OpenAI returned a malformed JSON string for tool call arguments. This tool call "
                    "will be skipped. To always generate a valid JSON, set `tools_strict` to `True`. "
                    "Tool call ID: {_id}, Tool name: {_name}, Arguments: {_arguments}",
                    _id=call_data["id"],
                    _name=call_data["name"],
                    _arguments=call_data["arguments"],
                )

        meta = {
            "model": chunk.model,
            "index": 0,
            "finish_reason": chunk.choices[0].finish_reason,
            "completion_start_time": chunks[0].meta.get("received_at"),  # first chunk received
            "usage": {},  # we don't have usage data for streaming responses
        }

        return ChatMessage.from_assistant(text=text or None, tool_calls=tool_calls, meta=meta)

    def _convert_chat_completion_to_chat_message(self, completion: ChatCompletion, choice: Choice) -> ChatMessage:
        """
        Converts the non-streaming response from the OpenAI API to a ChatMessage.

        :param completion: The completion returned by the OpenAI API.
        :param choice: The choice returned by the OpenAI API.
        :return: The ChatMessage.
        """
        message: ChatCompletionMessage = choice.message
        text = message.content
        tool_calls = []
        if openai_tool_calls := message.tool_calls:
            for openai_tc in openai_tool_calls:
                arguments_str = openai_tc.function.arguments
                try:
                    arguments = json.loads(arguments_str)
                    tool_calls.append(ToolCall(id=openai_tc.id, tool_name=openai_tc.function.name, arguments=arguments))
                except json.JSONDecodeError:
                    logger.warning(
                        "OpenAI returned a malformed JSON string for tool call arguments. This tool call "
                        "will be skipped. To always generate a valid JSON, set `tools_strict` to `True`. "
                        "Tool call ID: {_id}, Tool name: {_name}, Arguments: {_arguments}",
                        _id=openai_tc.id,
                        _name=openai_tc.function.name,
                        _arguments=arguments_str,
                    )

        chat_message = ChatMessage.from_assistant(text=text, tool_calls=tool_calls)
        chat_message._meta.update(
            {
                "model": completion.model,
                "index": choice.index,
                "finish_reason": choice.finish_reason,
                "usage": dict(completion.usage or {}),
            }
        )
        return chat_message

    def _convert_chat_completion_chunk_to_streaming_chunk(self, chunk: ChatCompletionChunk) -> StreamingChunk:
        """
        Converts the streaming response chunk from the OpenAI API to a StreamingChunk.

def _recursively_set_additional_properties_false(schema: Any) -> None:
    """
    Recursively set `"additionalProperties": False` on all the object schemas contained in a JSON schema.

    OpenAI's strict mode requires `"additionalProperties": False` to be set on every object in the schema
    (not only on the top-level one) for tool use to be accepted. Tools such as `ComponentTool` generate
    nested schemas (e.g. properties referring to definitions in `$defs`), so we have to walk the whole schema.

    :param schema: The (sub-)schema to update in place.
    """
    if not isinstance(schema, dict):
        return

    # An object schema is either explicitly typed as object or implicitly defined by having properties
    if schema.get("type") == "object" or "properties" in schema:
        schema["additionalProperties"] = False

    # recurse into the object properties and the schema definitions
    for key in ("properties", "patternProperties", "$defs", "definitions"):
        sub_schemas = schema.get(key)
        if isinstance(sub_schemas, dict):
            for sub_schema in sub_schemas.values():
                _recursively_set_additional_properties_false(sub_schema)

    # recurse into array items
    items = schema.get("items")
    if isinstance(items, dict):
        _recursively_set_additional_properties_false(items)
    elif isinstance(items, list):
        for item in items:
            _recursively_set_additional_properties_false(item)

    # recurse into the remaining schema combinators
    for key in ("anyOf", "allOf", "oneOf", "prefixItems"):
        sub_schemas = schema.get(key)
        if isinstance(sub_schemas, list):
            for sub_schema in sub_schemas:
                _recursively_set_additional_properties_false(sub_schema)

    # in case additionalProperties itself holds a schema, recurse into it too
    additional_properties = schema.get("additionalProperties")
    if isinstance(additional_properties, dict):
        _recursively_set_additional_properties_false(additional_properties)

    def _prepare_api_call(  # noqa: PLR0913
        self,
        *,
        messages: List[ChatMessage],
        streaming_callback: Optional[StreamingCallbackT] = None,
        generation_kwargs: Optional[Dict[str, Any]] = None,
        tools: Optional[List[Tool]] = None,
        tools_strict: Optional[bool] = None,
    ) -> Dict[str, Any]:
        # update generation kwargs by merging with the generation kwargs passed to the run method
        generation_kwargs = {**self.generation_kwargs, **(generation_kwargs or {})}

        # adapt ChatMessage(s) to the format expected by the OpenAI API
        openai_formatted_messages = [message.to_openai_dict_format() for message in messages]

        tools = tools or self.tools
        tools_strict = tools_strict if tools_strict is not None else self.tools_strict

        if tools:

                {"type": "function", "function": self._convert_tool_to_openai_format(t, tools_strict=tools_strict)}
                for t in tools

        if is_streaming and num_responses > 1:
            raise ValueError("Cannot stream multiple responses, please set n=1.")

        return {
            "model": self.model,
            "messages": openai_formatted_messages,  # type: ignore[arg-type] # openai expects list of specific message types
            "stream": streaming_callback is not None,
            "n": num_responses,
            **openai_tools,
            **generation_kwargs,

    @staticmethod
    def _convert_tool_to_openai_format(tool: Tool, tools_strict: bool = False) -> Dict[str, Any]:
        """
        Convert a Tool into the function specification expected by the OpenAI API.

        The tool spec is deep-copied so that the original Tool (and its schema) is never modified.

        When `tools_strict` is `True`, OpenAI requires the JSON schema of the function to have
        `"additionalProperties": False` on every object, otherwise the schema is rejected.
        Since tools such as `ComponentTool` generate nested schemas, the flag is added recursively.

        :param tool: The tool to convert.
        :param tools_strict: Whether to enable strict schema adherence for the tool.
        :returns: The function specification to send to the OpenAI API.
        """
        function_spec: Dict[str, Any] = deepcopy(tool.tool_spec)

        if not tools_strict:
            return function_spec

        function_spec["strict"] = True

        # Add additionalProperties: false to the parameters schema when tools_strict is True
        parameters = function_spec.get("parameters")
        if parameters is not None:
            _recursively_set_additional_properties_false(parameters)

        return function_spec

    def _handle_stream_response(self, chat_completion: Stream, callback: SyncStreamingCallbackT) -> List[ChatMessage]:
        chunks: List[StreamingChunk] = []
        chunk = None

        for chunk in chat_completion:  # pylint: disable=not-an-iterable
            assert len(chunk.choices) == 1, "Streaming responses should have only one choice."
            chunk_delta: StreamingChunk = self._convert_chat_completion_chunk_to_streaming_chunk(chunk)
            chunks.append(chunk_delta)

            callback(chunk_delta)

        return [self._convert_streaming_chunks_to_chat_message(chunk, chunks)]

    async def _handle_async_stream_response(
        self, chat_completion: AsyncStream, callback: AsyncStreamingCallbackT
    ) -> List[ChatMessage]:
        chunks: List[StreamingChunk] = []
        chunk = None

        async for chunk in chat_completion:  # pylint: disable=not-an-iterable

    def _check_finish_reason(self, meta: Dict[str, Any]) -> None:
        if meta["finish_reason"] == "length":
            logger.warning(
                "The completion for index {index} has been truncated before reaching a natural stopping point. "
                "Increase the max_tokens parameter to allow for longer completions.",
                index=meta["index"],
                finish_reason=meta["finish_reason"],
            )
        if meta["finish_reason"] == "content_filter":
            logger.warning(
                "The completion for index {index} has been truncated due to the content filter.",
                index=meta["index"],
                finish_reason=meta["finish_reason"],
            )

    def _convert_streaming_chunks_to_chat_message(
        self, chunk: ChatCompletionChunk, chunks: List[StreamingChunk]
    ) -> ChatMessage:
        """
        Connects the streaming chunks into a single ChatMessage.

        :param chunk: The last chunk returned by the OpenAI API.
        :param chunks: The list of all `StreamingChunk` objects.
        """
        text = "".join([chunk.content for chunk in chunks])
        tool_calls = []

        # Process tool calls if present in any chunk
        tool_call_data: Dict[str, Dict[str, str]] = {}  # Track tool calls by index
        for chunk_payload in chunks:
            tool_calls_meta = chunk_payload.meta.get("tool_calls")
            if tool_calls_meta is not None:
                for delta in tool_calls_meta:
                    # We use the index of the tool call to track it across chunks since the ID is not always provided
                    if delta.index not in tool_call_data:
                        tool_call_data[delta.index] = {"id": "", "name": "", "arguments": ""}

                    # Save the ID if present
                    if delta.id is not None:
                        tool_call_data[delta.index]["id"] = delta.id

                    if delta.function is not None:
                        if delta.function.name is not None:
                            tool_call_data[delta.index]["name"] += delta.function.name
                        if delta.function.arguments is not None:
                            tool_call_data[delta.index]["arguments"] += delta.function.arguments

        # Convert accumulated tool call data into ToolCall objects
        for call_data in tool_call_data.values():
            try:
                arguments = json.loads(call_data["arguments"])
                tool_calls.append(ToolCall(id=call_data["id"], tool_name=call_data["name"], arguments=arguments))
            except json.JSONDecodeError:
                logger.warning(
                    "OpenAI returned a malformed JSON string for tool call arguments. This tool call "
                    "will be skipped. To always generate a valid JSON, set `tools_strict` to `True`. "
                    "Tool call ID: {_id}, Tool name: {_name}, Arguments: {_arguments}",
                    _id=call_data["id"],
                    _name=call_data["name"],
                    _arguments=call_data["arguments"],
                )

        meta = {
            "model": chunk.model,
            "index": 0,
            "finish_reason": chunk.choices[0].finish_reason,
            "completion_start_time": chunks[0].meta.get("received_at"),  # first chunk received
            "usage": {},  # we don't have usage data for streaming responses
        }

        return ChatMessage.from_assistant(text=text or None, tool_calls=tool_calls, meta=meta)

    def _convert_chat_completion_to_chat_message(self, completion: ChatCompletion, choice: Choice) -> ChatMessage:
        """
        Converts the non-streaming response from the OpenAI API to a ChatMessage.

        :param completion: The completion returned by the OpenAI API.
        :param choice: The choice returned by the OpenAI API.
        :return: The ChatMessage.
        """
        message: ChatCompletionMessage = choice.message
        text = message.content
        tool_calls = []
        if openai_tool_calls := message.tool_calls:
            for openai_tc in openai_tool_calls:
                arguments_str = openai_tc.function.arguments
                try:
                    arguments = json.loads(arguments_str)
                    tool_calls.append(ToolCall(id=openai_tc.id, tool_name=openai_tc.function.name, arguments=arguments))
                except json.JSONDecodeError:
                    logger.warning(
                        "OpenAI returned a malformed JSON string for tool call arguments. This tool call "
                        "will be skipped. To always generate a valid JSON, set `tools_strict` to `True`. "
                        "Tool call ID: {_id}, Tool name: {_name}, Arguments: {_arguments}",
                        _id=openai_tc.id,
                        _name=openai_tc.function.name,
                        _arguments=arguments_str,
                    )

        chat_message = ChatMessage.from_assistant(text=text, tool_calls=tool_calls)
        chat_message._meta.update(
            {
                "model": completion.model,
                "index": choice.index,
                "finish_reason": choice.finish_reason,
                "usage": dict(completion.usage or {}),
            }
        )
        return chat_message

    def _convert_chat_completion_chunk_to_streaming_chunk(self, chunk: ChatCompletionChunk) -> StreamingChunk:
        """
        Converts the streaming response chunk from the OpenAI API to a StreamingChunk.

        :param chunk: The chunk returned by the OpenAI API.

        :returns:
            The StreamingChunk.
        """
        # we stream the content of the chunk if it's not a tool or function call
        choice: ChunkChoice = chunk.choices[0]
        content = choice.delta.content or ""
        chunk_message = StreamingChunk(content)
        # but save the tool calls and function call in the meta if they are present
        # and then connect the chunks in the _convert_streaming_chunks_to_chat_message method
        chunk_message.meta.update(
            {
                "model": chunk.model,
                "index": choice.index,
                "tool_calls": choice.delta.tool_calls,
                "finish_reason": choice.finish_reason,
                "received_at": datetime.now().isoformat(),
            }
        )
        return chunk_message

# Keywords of a JSON schema whose value is a mapping of subschemas
_SCHEMA_MAPPING_KEYWORDS = ("properties", "patternProperties", "$defs", "definitions", "dependentSchemas")

# Keywords of a JSON schema whose value is a subschema (or a list of subschemas)
_SCHEMA_KEYWORDS = (
    "items",
    "additionalItems",
    "contains",
    "not",
    "if",
    "then",
    "else",
    "propertyNames",
    "allOf",
    "anyOf",
    "oneOf",
    "prefixItems",
)


def _is_object_schema(schema: Dict[str, Any]) -> bool:
    """
    Checks whether a JSON schema fragment describes an object.

    :param schema: The JSON schema fragment to check.
    :returns: `True` if the fragment describes an object, `False` otherwise.
    """
    schema_type = schema.get("type")
    if schema_type == "object":
        return True
    if isinstance(schema_type, list) and "object" in schema_type:
        return True
    # Schemas without an explicit type that use object-only keywords are objects too.
    # Note: schemas consisting only of a "$ref" are intentionally not considered objects,
    # as the referenced subschema is handled separately.
    return schema_type is None and any(
        keyword in schema for keyword in ("properties", "required", "additionalProperties", "patternProperties")
    )


def _convert_to_strict_schema(schema: Any) -> Any:
    """
    Recursively sets `additionalProperties` to `False` on all objects of a JSON schema.

    OpenAI's strict mode requires `additionalProperties: false` to be set on every object of the
    provided schema, including nested ones. This is not the case for schemas generated from
    Pydantic models or from Haystack components (e.g. `ComponentTool`), which is why the schema
    is adapted before being sent to the OpenAI API.

    :param schema: The JSON schema fragment to adapt. It's modified in place.
    :returns: The adapted JSON schema fragment.
    """
    if isinstance(schema, list):
        return [_convert_to_strict_schema(item) for item in schema]

    if not isinstance(schema, dict):
        return schema

    for key, value in schema.items():
        if key in _SCHEMA_MAPPING_KEYWORDS and isinstance(value, dict):
            schema[key] = {name: _convert_to_strict_schema(subschema) for name, subschema in value.items()}
        elif key in _SCHEMA_KEYWORDS:
            schema[key] = _convert_to_strict_schema(value)

    if _is_object_schema(schema):
        schema["additionalProperties"] = False

    return schema


def _convert_tool_to_openai_format(tool: Tool, strict: bool = False) -> Dict[str, Any]:
    """
    Converts a Haystack `Tool` into the tool format expected by the OpenAI Chat Completions API.

    :param tool: The Haystack `Tool` to convert.
    :param strict: Whether to enable OpenAI's strict schema adherence for the tool.
    :returns: The tool definition in the format expected by the OpenAI API.
    """
    # We deep copy the tool spec to avoid mutating the spec of the original tool
    function_spec = copy.deepcopy(tool.tool_spec)

    if strict:
        function_spec["strict"] = True
        parameters = function_spec.get("parameters") or {"type": "object", "properties": {}}
        function_spec["parameters"] = _convert_to_strict_schema(parameters)

    return {"type": "function", "function": function_spec}

    def _prepare_api_call(  # noqa: PLR0913
        self,
        *,
        messages: List[ChatMessage],
        streaming_callback: Optional[StreamingCallbackT] = None,
        generation_kwargs: Optional[Dict[str, Any]] = None,
        tools: Optional[List[Tool]] = None,
        tools_strict: Optional[bool] = None,
    ) -> Dict[str, Any]:

        generation_kwargs = {**self.generation_kwargs, **(generation_kwargs or {})}

        # adapt ChatMessage(s) to the format expected by the OpenAI API
        openai_formatted_messages = [message.to_openai_dict_format() for message in messages]

            "stream": streaming_callback is not None,

            chunks.append(chunk_delta)

        chunks: List[StreamingChunk] = []
        chunk = None

        async for chunk in chat_completion:  # pylint: disable=not-an-iterable
            assert len(chunk.choices) == 1, "Streaming responses should have only one choice."
            chunk_delta: StreamingChunk = self._convert_chat_completion_chunk_to_streaming_chunk(chunk)
            chunks.append(chunk_delta)

            await callback(chunk_delta)

        return [self._convert_streaming_chunks_to_chat_message(chunk, chunks)]

    def _check_finish_reason(self, meta: Dict[str, Any]) -> None:
        if meta["finish_reason"] == "length":
            logger.warning(
                "The completion for index {index} has been truncated before reaching a natural stopping point. "
                "Increase the max_tokens parameter to allow for longer completions.",
                index=meta["index"],
                finish_reason=meta["finish_reason"],
            )
        if meta["finish_reason"] == "content_filter":
            logger.warning(
                "The completion for index {index} has been truncated due to the content filter.",
                index=meta["index"],
                finish_reason=meta["finish_reason"],
            )

    def _convert_streaming_chunks_to_chat_message(
        self, chunk: ChatCompletionChunk, chunks: List[StreamingChunk]
    ) -> ChatMessage:
        """
        Connects the streaming chunks into a single ChatMessage.

        :param chunk: The last chunk returned by the OpenAI API.
        :param chunks: The list of all `StreamingChunk` objects.
        """
        text = "".join([chunk.content for chunk in chunks])
        tool_calls = []

        # Process tool calls if present in any chunk
        tool_call_data: Dict[str, Dict[str, str]] = {}  # Track tool calls by index
        for chunk_payload in chunks:
            tool_calls_meta = chunk_payload.meta.get("tool_calls")
            if tool_calls_meta is not None:
                for delta in tool_calls_meta:
                    # We use the index of the tool call to track it across chunks since the ID is not always provided
                    if delta.index not in tool_call_data:
                        tool_call_data[delta.index] = {"id": "", "name": "", "arguments": ""}

                    # Save the ID if present
                    if delta.id is not None:
                        tool_call_data[delta.index]["id"] = delta.id

                    if delta.function is not None:
                        if delta.function.name is not None:
                            tool_call_data[delta.index]["name"] += delta.function.name
                        if delta.function.arguments is not None:
                            tool_call_data[delta.index]["arguments"] += delta.function.arguments

        # Convert accumulated tool call data into ToolCall objects
        for call_data in tool_call_data.values():
            try:
                arguments = json.loads(call_data["arguments"])
                tool_calls.append(ToolCall(id=call_data["id"], tool_name=call_data["name"], arguments=arguments))
            except json.JSONDecodeError:
                logger.warning(
                    "OpenAI returned a malformed JSON string for tool call arguments. This tool call "
                    "will be skipped. To always generate a valid JSON, set `tools_strict` to `True`. "
                    "Tool call ID: {_id}, Tool name: {_name}, Arguments: {_arguments}",
                    _id=call_data["id"],
                    _name=call_data["name"],
                    _arguments=call_data["arguments"],
                )

        meta = {
            "model": chunk.model,
            "index": 0,
            "finish_reason": chunk.choices[0].finish_reason,
            "completion_start_time": chunks[0].meta.get("received_at"),  # first chunk received
            "usage": {},  # we don't have usage data for streaming responses
        }

        return ChatMessage.from_assistant(text=text or None, tool_calls=tool_calls, meta=meta)

    def _convert_chat_completion_to_chat_message(self, completion: ChatCompletion, choice: Choice) -> ChatMessage:
        """
        Converts the non-streaming response from the OpenAI API to a ChatMessage.

        :param completion: The completion returned by the OpenAI API.
        :param choice: The choice returned by the OpenAI API.
        :return: The ChatMessage.
        """
        message: ChatCompletionMessage = choice.message
        text = message.content
        tool_calls = []
        if openai_tool_calls := message.tool_calls:
            for openai_tc in openai_tool_calls:
                arguments_str = openai_tc.function.arguments
                try:
                    arguments = json.loads(arguments_str)
                    tool_calls.append(ToolCall(id=openai_tc.id, tool_name=openai_tc.function.name, arguments=arguments))
                except json.JSONDecodeError:
                    logger.warning(
                        "OpenAI returned a malformed JSON string for tool call arguments. This tool call "
                        "will be skipped. To always generate a valid JSON, set `tools_strict` to `True`. "
                        "Tool call ID: {_id}, Tool name: {_name}, Arguments: {_arguments}",
                        _id=openai_tc.id,
                        _name=openai_tc.function.name,
                        _arguments=arguments_str,
                    )

        chat_message = ChatMessage.from_assistant(text=text, tool_calls=tool_calls)
        chat_message._meta.update(
            {
                "model": completion.model,
                "index": choice.index,
                "finish_reason": choice.finish_reason,
                "usage": dict(completion.usage or {}),
            }
        )
        return chat_message

    def _convert_chat_completion_chunk_to_streaming_chunk(self, chunk: ChatCompletionChunk) -> StreamingChunk:
        """
        Converts the streaming response chunk from the OpenAI API to a StreamingChunk.

        :param chunk: The chunk returned by the OpenAI API.

        :returns:
            The StreamingChunk.
        """
        # we stream the content of the chunk if it's not a tool or function call
        choice: ChunkChoice = chunk.choices[0]
        content = choice.delta.content or ""
        chunk_message = StreamingChunk(content)
        # but save the tool calls and function call in the meta if they are present
        # and then connect the chunks in the _convert_streaming_chunks_to_chat_message method
        chunk_message.meta.update(
            {
                "model": chunk.model,
                "index": choice.index,
                "tool_calls": choice.delta.tool_calls,
                "finish_reason": choice.finish_reason,
                "received_at": datetime.now().isoformat(),
            }
        )
        return chunk_message
