TELEMETRY_DISABLED_ENV_VAR = "OTEL_SDK_DISABLED"

# Values accepted as "disabled" for the OTEL_SDK_DISABLED env var. The
# OpenTelemetry specification defines "true", but users commonly set
# "True", "TRUE", "1", "yes" or "on", so all of them are honored.
_DISABLED_VALUES = frozenset({"true", "1", "yes", "on", "y", "t"})


def is_telemetry_disabled() -> bool:
    """Check whether telemetry has been disabled through the environment.

    The env var is evaluated on every call (instead of only once, when the
    ``Telemetry`` singleton is instantiated at import time) so that users can
    disable OpenTelemetry even after ``crewai`` has been imported, e.g. when
    loading a ``.env`` file or setting ``os.environ`` from their own code.

    Returns:
        bool: True if telemetry should be fully disabled, False otherwise.
    """
    value = os.getenv(TELEMETRY_DISABLED_ENV_VAR, "false")
    if value is None:
        return False
    return value.strip().lower() in _DISABLED_VALUES

#: Values treated (case-insensitively) as "telemetry is disabled".
_DISABLED_ENV_VALUES = frozenset({"true", "1", "yes", "on"})

#: Environment variables that turn CrewAI's telemetry exporter off.
_DISABLED_ENV_VARS = ("OTEL_SDK_DISABLED", "CREWAI_DISABLE_TELEMETRY")


def is_telemetry_disabled() -> bool:
    """Whether telemetry export has been explicitly disabled through env vars.

    ``OTEL_SDK_DISABLED=true`` disables CrewAI's telemetry exporter (and, as it
    is the standard OpenTelemetry switch, global OpenTelemetry instrumentation
    as well). ``CREWAI_DISABLE_TELEMETRY=true`` only disables CrewAI's exporter.

    Accepted truthy values are ``true``, ``1``, ``yes`` and ``on`` in any
    casing (surrounding whitespace is ignored). The environment is read on
    every call so that setting the variable after importing ``crewai`` (for
    example ``os.environ["OTEL_SDK_DISABLED"] = "true"`` at the top of a
    script) is still honoured.
    """
    for env_var in _DISABLED_ENV_VARS:
        value = os.getenv(env_var)
        if value is not None and value.strip().lower() in _DISABLED_ENV_VALUES:
            return True
    return False


class Telemetry:
    """A class to handle anonymous telemetry for the crewai package.

    The data being collected is for development purpose, all data is anonymous.

    There is NO data being collected on the prompts, tasks descriptions
    agents backstories or goals nor responses or any data that is being
    processed by the agents, nor any secrets and env vars.

    Users can opt-in to sharing more complete data using the `share_crew`
    attribute in the Crew class.

    Telemetry export can be fully disabled by setting the ``OTEL_SDK_DISABLED``
    (or ``CREWAI_DISABLE_TELEMETRY``) environment variable to ``true``. When it
    is disabled, no tracer provider / span processor is created, no tracer
    provider is installed globally and no data is ever queued for export.
    """

    def __init__(self):
        self.ready = False
        self.trace_set = False
        self.provider: Optional[TracerProvider] = None
        self.resource: Optional[Resource] = None

        # Honour the opt-out before touching the OpenTelemetry SDK at all, so
        # that nothing is ever collected or queued for export (which used to
        # hang and time out when the telemetry endpoint was unreachable).
        if is_telemetry_disabled():
            return

        try:
            telemetry_endpoint = "https://telemetry.crewai.com:4319"
            self.resource = Resource(
                attributes={SERVICE_NAME: "crewAI-telemetry"},
            )
            with suppress_warnings():
                self.provider = TracerProvider(resource=self.resource)

            processor = BatchSpanProcessor(
                OTLPSpanExporter(
                    endpoint=f"{telemetry_endpoint}/v1/traces",
                    timeout=30,
                )
            )

            self.provider.add_span_processor(processor)
            self.ready = True
        except Exception as e:
            if isinstance(
                e,
                (SystemExit, KeyboardInterrupt, GeneratorExit, asyncio.CancelledError),
            ):
                raise  # Re-raise the exception to not interfere with system signals
            self.ready = False

    def set_tracer(self):
        # Never install the telemetry tracer provider when telemetry is off.
        if is_telemetry_disabled():
            self.ready = False
            self.trace_set = False
            return

        if self.ready and not self.trace_set:
            try:
                with suppress_warnings():
                    trace.set_tracer_provider(self.provider)
                    self.trace_set = True
            except Exception:
                self.ready = False
                self.trace_set = False

    def _safe_telemetry_operation(self, operation):
        """Run a telemetry operation, never raising and never exporting.

        Returns whatever ``operation`` returns when telemetry is enabled, and
        ``None`` when telemetry is disabled or the operation failed.
        """
        if is_telemetry_disabled():
            self.ready = False
            self.trace_set = False
            return None

        if not self.ready:
            return None

        try:
            # Call the operation directly so that its return value (e.g. the
            # span created in `task_started`) is preserved.
            return operation()
        except Exception:
            self.ready = False
            self.trace_set = False
            return None

    def task_started(self, crew: Crew, task: Task) -> Span | None:
        """Records task started in a crew."""

        def operation():
            tracer = trace.get_tracer("crewai.telemetry")

            created_span = tracer.start_span("Task Created")

            self._add_attribute(created_span, "crew_key", crew.key)
            self._add_attribute(created_span, "crew_id", str(crew.id))
            self._add_attribute(created_span, "task_key", task.key)
            self._add_attribute(created_span, "task_id", str(task.id))

            if crew.share_crew:
                self._add_attribute(
                    created_span, "formatted_description", task.description
                )
                self._add_attribute(
                    created_span, "formatted_expected_output", task.expected_output
                )

            created_span.set_status(Status(StatusCode.OK))
            created_span.end()

            span = tracer.start_span("Task Execution")

            self._add_attribute(span, "crew_key", crew.key)
            self._add_attribute(span, "crew_id", str(crew.id))
            self._add_attribute(span, "task_key", task.key)
            self._add_attribute(span, "task_id", str(task.id))

            if crew.share_crew:
                self._add_attribute(span, "formatted_description", task.description)
                self._add_attribute(
                    span, "formatted_expected_output", task.expected_output
                )

            return span

        return self._safe_telemetry_operation(operation)

TELEMETRY_DISABLED_ENV_VAR = "OTEL_SDK_DISABLED"

# Values accepted as "disabled" for the OTEL_SDK_DISABLED env var. The
# OpenTelemetry specification defines "true", but users commonly set
# "True", "TRUE", "1", "yes" or "on", so all of them are honored.
_DISABLED_VALUES = frozenset({"true", "1", "yes", "on", "y", "t"})


def is_telemetry_disabled() -> bool:
    """Check whether telemetry has been disabled through the environment.

    The env var is evaluated on every call (instead of only once, when the
    ``Telemetry`` singleton is instantiated at import time) so that users can
    disable OpenTelemetry even after ``crewai`` has been imported, e.g. when
    loading a ``.env`` file or setting ``os.environ`` from their own code.

    Returns:
        bool: True if telemetry should be fully disabled, False otherwise.
    """
    value = os.getenv(TELEMETRY_DISABLED_ENV_VAR, "false")
    if value is None:
        return False
    return value.strip().lower() in _DISABLED_VALUES


class Telemetry:
    """A class to handle anonymous telemetry for the crewai package.

    The data being collected is for development purpose, all data is anonymous.

    There is NO data being collected on the prompts, tasks descriptions
    agents backstories or goals nor responses or any data that is being
    processed by the agents, nor any secrets and env vars.

    Users can opt-in to sharing more complete data using the `share_crew`
    attribute in the Crew class.
    """

    def __init__(self):
        self.ready = False
        self.trace_set = False
        self.provider: TracerProvider | None = None
        self.resource: Resource | None = None

        if is_telemetry_disabled():
            return

        try:
            telemetry_endpoint = "https://telemetry.crewai.com:4319"
            self.resource = Resource(
                attributes={SERVICE_NAME: "crewAI-telemetry"},
            )
            with suppress_warnings():
                self.provider = TracerProvider(resource=self.resource)

            processor = BatchSpanProcessor(
                OTLPSpanExporter(
                    endpoint=f"{telemetry_endpoint}/v1/traces",
                    timeout=30,
                )
            )

            self.provider.add_span_processor(processor)
            self.ready = True
        except Exception as e:
            if isinstance(
                e,
                (SystemExit, KeyboardInterrupt, GeneratorExit, asyncio.CancelledError),
            ):
                raise  # Re-raise the exception to not interfere with system signals
            self.ready = False

    def set_tracer(self):
        # Never register a tracer provider when telemetry is disabled, even if
        # the env var was set after this instance was created.
        if is_telemetry_disabled():
            self.ready = False
            self.trace_set = False
            return

        if self.ready and not self.trace_set and self.provider is not None:
            try:
                with suppress_warnings():
                    trace.set_tracer_provider(self.provider)
                    self.trace_set = True
            except Exception:
                self.ready = False
                self.trace_set = False

    def _safe_telemetry_operation(self, operation):
        # Bail out before doing any work (creating spans, calling the tracer,
        # exporting spans) when telemetry is disabled or not ready.
        if not self.ready or is_telemetry_disabled():
            return None
        try:
            return operation()
        except Exception:
            return None

    def task_started(self, crew: Crew, task: Task) -> Span | None:
        """Records task started in a crew."""

        def operation():
            tracer = trace.get_tracer("crewai.telemetry")

            created_span = tracer.start_span("Task Created")

            self._add_attribute(created_span, "crew_key", crew.key)
            self._add_attribute(created_span, "crew_id", str(crew.id))
            self._add_attribute(created_span, "task_key", task.key)
            self._add_attribute(created_span, "task_id", str(task.id))

            if crew.share_crew:
                self._add_attribute(
                    created_span, "formatted_description", task.description
                )
                self._add_attribute(
                    created_span, "formatted_expected_output", task.expected_output
                )

            created_span.set_status(Status(StatusCode.OK))
            created_span.end()

            span = tracer.start_span("Task Execution")

            self._add_attribute(span, "crew_key", crew.key)
            self._add_attribute(span, "crew_id", str(crew.id))
            self._add_attribute(span, "task_key", task.key)
            self._add_attribute(span, "task_id", str(task.id))

            if crew.share_crew:
                self._add_attribute(span, "formatted_description", task.description)
                self._add_attribute(
                    span, "formatted_expected_output", task.expected_output
                )

            return span

        return self._safe_telemetry_operation(operation)

    def task_ended(self, span: Span, task: Task, crew: Crew):
        """Records task execution in a crew."""

        def operation():
            if span is None:
                return

            if crew.share_crew:
                self._add_attribute(
                    span,
                    "task_output",
                    task.output.raw if task.output else "",
                )

            span.set_status(Status(StatusCode.OK))
            span.end()

        self._safe_telemetry_operation(operation)

    def end_crew(self, crew, final_string_output):
        def operation():
            if getattr(crew, "_execution_span", None) is None:
                return

            self._add_attribute(
                crew._execution_span,
                "crewai_version",
                pkg_resources.get_distribution("crewai").version,
            )
            self._add_attribute(
                crew._execution_span, "crew_output", final_string_output
            )
            self._add_attribute(
                crew._execution_span,
                "crew_tasks_output",
                json.dumps(
                    [
                        {
                            "id": str(task.id),
                            "description": task.description,
                            "output": task.output.raw_output,
                        }
                        for task in crew.tasks
                    ]
                ),
            )
            crew._execution_span.set_status(Status(StatusCode.OK))
            crew._execution_span.end()

        if crew.share_crew:
            self._safe_telemetry_operation(operation)

    def _add_attribute(self, span, key, value):
        """Add an attribute to a span."""

        if span is None:
            return

        def operation():
            return span.set_attribute(key, value)

        self._safe_telemetry_operation(operation)
