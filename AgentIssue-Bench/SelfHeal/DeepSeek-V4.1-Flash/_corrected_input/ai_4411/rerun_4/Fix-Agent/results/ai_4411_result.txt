// Minimal, duck-typed view on a Zod schema. It is only used to read the
// information that is needed to restore descriptions in JSON schemas, so that
// this module does not need a hard dependency on zod.
type ZodSchemaLike = {
  description?: string;
  _def?: {
    typeName?: string;
    innerType?: ZodSchemaLike;
    schema?: ZodSchemaLike;
    type?: ZodSchemaLike;
    in?: ZodSchemaLike;
    element?: ZodSchemaLike;
    valueType?: ZodSchemaLike;
    shape?: () => Record<string, ZodSchemaLike>;
    options?: Array<ZodSchemaLike> | Map<unknown, ZodSchemaLike>;
  };
};

function isZodSchemaLike(value: unknown): value is ZodSchemaLike {
  return (
    typeof value === 'object' &&
    value != null &&
    typeof (value as ZodSchemaLike)._def === 'object' &&
    (value as ZodSchemaLike)._def != null
  );
}

function isNullableZodSchema(zodSchema: ZodSchemaLike | undefined): boolean {
  if (zodSchema == null) {
    return false;
  }

  const def = zodSchema._def;

  switch (def?.typeName) {
    case 'ZodNullable':
      return true;
    case 'ZodOptional':
    case 'ZodDefault':
    case 'ZodReadonly':
    case 'ZodCatch':
      return isNullableZodSchema(def.innerType);
    case 'ZodUnion': {
      const options = def.options;
      const optionSchemas =
        options instanceof Map ? [...options.values()] : options;

      return (
        Array.isArray(optionSchemas) && optionSchemas.some(isNullableZodSchema)
      );
    }
    default:
      return false;
  }
}

/**
 * Returns a copy of the given JSON schema in which the wrappers that are
 * created for optional fields (`anyOf` with a single branch) are removed.
 */
function unwrapAnyOf(jsonSchema: JSONSchema7): JSONSchema7 {
  const anyOf = jsonSchema.anyOf;

  if (!Array.isArray(anyOf) || anyOf.length !== 1) {
    return jsonSchema;
  }

  const branch = anyOf[0];

  if (branch == null || typeof branch !== 'object') {
    return jsonSchema;
  }

  const result: JSONSchema7 = { ...(branch as JSONSchema7) };

  for (const [key, value] of Object.entries(jsonSchema)) {
    if (key === 'anyOf' || value === undefined) {
      continue;
    }

    (result as Record<string, unknown>)[key] = value;
  }

  return result;
}

/**
 * Returns a copy of the given JSON schema in which the artificial `null` type
 * that is created for optional fields is removed.
 */
function removeNullType(jsonSchema: JSONSchema7): JSONSchema7 {
  const types = jsonSchema.type;

  if (!Array.isArray(types) || !types.includes('null')) {
    return jsonSchema;
  }

  const nonNullTypes = types.filter(type => type !== 'null');

  if (nonNullTypes.length === 0) {
    return jsonSchema;
  }

  return {
    ...jsonSchema,
    type: nonNullTypes.length === 1 ? nonNullTypes[0] : nonNullTypes,
  };
}

/**
 * Copies the descriptions of the given Zod schema into the given JSON schema.
 *
 * When a Zod schema is converted into a JSON schema, the description of an
 * optional field ends up outside of the property schema: the field is either
 * wrapped in an `anyOf` branch or turned into a nullable type. LLM providers
 * however only pick up descriptions from the property schema itself, so the
 * descriptions need to be restored (and the optional wrappers removed).
 */
function addZodDescriptions(
  zodSchema: ZodSchemaLike,
  jsonSchema: JSONSchema7,
): JSONSchema7 {
  const def: NonNullable<ZodSchemaLike['_def']> = zodSchema._def ?? {};

  let result: JSONSchema7 = jsonSchema;

  switch (def.typeName) {
    // wrappers that contain the actual schema:
    case 'ZodOptional':
    case 'ZodNullable':
    case 'ZodDefault':
    case 'ZodReadonly':
    case 'ZodCatch': {
      result = unwrapAnyOf(result);

      // optional fields are not nullable:
      if (
        def.typeName === 'ZodOptional' &&
        !isNullableZodSchema(def.innerType)
      ) {
        result = removeNullType(result);
      }

      if (def.innerType != null) {
        result = addZodDescriptions(def.innerType, result);
      }
      break;
    }

    case 'ZodObject': {
      const shape = def.shape?.();

      if (shape != null && result.properties != null) {
        const properties: Record<string, JSONSchema7> = {};

        for (const [key, propertyJsonSchema] of Object.entries(
          result.properties,
        )) {
          const propertyZodSchema = shape[key];

          properties[key] =
            propertyZodSchema != null &&
            propertyJsonSchema != null &&
            typeof propertyJsonSchema === 'object'
              ? addZodDescriptions(
                  propertyZodSchema,
                  propertyJsonSchema as JSONSchema7,
                )
              : (propertyJsonSchema as JSONSchema7);
        }

        result = { ...result, properties };
      }
      break;
    }

    case 'ZodArray': {
      const items = result.items;

      if (
        def.element != null &&
        items != null &&
        !Array.isArray(items) &&
        typeof items === 'object'
      ) {
        result = { ...result, items: addZodDescriptions(def.element, items) };
      }
      break;
    }

    case 'ZodEffects': {
      if (def.schema != null) {
        result = addZodDescriptions(def.schema, result);
      }
      break;
    }

    case 'ZodBranded': {
      if (def.type != null) {
        result = addZodDescriptions(def.type, result);
      }
      break;
    }

    case 'ZodPipeline': {
      if (def.in != null) {
        result = addZodDescriptions(def.in, result);
      }
      break;
    }

    case 'ZodUnion':
    case 'ZodDiscriminatedUnion': {
      const options = def.options;
      const optionSchemas =
        options instanceof Map ? [...options.values()] : options;
      const branches = result.anyOf;

      if (Array.isArray(optionSchemas) && Array.isArray(branches)) {
        result = {
          ...result,
          anyOf: branches.map((branch, index) => {
            const branchZodSchema = optionSchemas[index];

            return branchZodSchema != null &&
              branch != null &&
              typeof branch === 'object'
              ? addZodDescriptions(branchZodSchema, branch as JSONSchema7)
              : branch;
          }),
        };
      }
      break;
    }

    case 'ZodRecord': {
      const additionalProperties = result.additionalProperties;

      if (
        def.valueType != null &&
        additionalProperties != null &&
        typeof additionalProperties === 'object'
      ) {
        result = {
          ...result,
          additionalProperties: addZodDescriptions(
            def.valueType,
            additionalProperties as JSONSchema7,
          ),
        };
      }
      break;
    }

    default:
      break;
  }

  // the description of the outermost zod schema takes precedence:
  if (zodSchema.description != null) {
    result = { ...result, description: zodSchema.description };
  } else if (result.description == null && Array.isArray(result.anyOf)) {
    // restore descriptions that ended up in the anyOf branches:
    const branchDescription = result.anyOf
      .map(branch =>
        branch != null && typeof branch === 'object'
          ? (branch as JSONSchema7).description
          : undefined,
      )
      .find(description => description != null);

    if (branchDescription != null) {
      result = { ...result, description: branchDescription };
    }
  }

  return result;
}

/**
 * Restores the descriptions of optional fields of the given Zod schema in the
 * generated JSON schema.
 */
function addZodDescriptionsToJsonSchema(
  jsonSchema: JSONSchema7,
  originalSchema?: unknown,
): JSONSchema7 {
  return isZodSchemaLike(originalSchema)
    ? addZodDescriptions(originalSchema, jsonSchema)
    : jsonSchema;
}

export function prepareToolsAndToolChoice<
  TOOLS extends Record<string, CoreTool>,
>({
  tools,
  toolChoice,
  activeTools,
}: {
  tools: TOOLS | undefined;
  toolChoice: CoreToolChoice<TOOLS> | undefined;
  activeTools: Array<keyof TOOLS> | undefined;
}): {
  tools:
    | Array<LanguageModelV1FunctionTool | LanguageModelV1ProviderDefinedTool>
    | undefined;
  toolChoice: LanguageModelV1ToolChoice | undefined;
} {
  if (!isNonEmptyObject(tools)) {
    return {
      tools: undefined,
      toolChoice: undefined,
    };
  }

  // when activeTools is provided, we only include the tools that are in the list:
  const filteredTools =
    activeTools != null
      ? Object.entries(tools).filter(([name]) =>
          activeTools.includes(name as keyof TOOLS),
        )
      : Object.entries(tools);

  return {
    tools: filteredTools.map(([name, tool]) => {
      const toolType = tool.type;
      switch (toolType) {
        case undefined:
        case 'function':
          return {
            type: 'function' as const,
            name,
            description: tool.description,
            parameters: addZodDescriptionsToJsonSchema(
              asSchema(tool.parameters).jsonSchema,
              tool.parameters,
            ),
          };
        case 'provider-defined':
          return {
            type: 'provider-defined' as const,
            name,
            id: tool.id,
            args: tool.args,
          };
        default: {
          const exhaustiveCheck: never = toolType;
          throw new Error(`Unsupported tool type: ${exhaustiveCheck}`);
        }
      }
    }),
    toolChoice:
      toolChoice == null
        ? { type: 'auto' }
        : typeof toolChoice === 'string'
        ? { type: toolChoice }
        : { type: 'tool' as const, toolName: toolChoice.toolName as string },
  };
}

const noSchemaOutputStrategy: OutputStrategy<JSONValue, JSONValue, never> = {
  type: 'no-schema',
  jsonSchema: undefined,

  validatePartialResult({ value, textDelta }) {
    return { success: true, value: { partial: value, textDelta } };
  },

  validateFinalResult(
    value: JSONValue | undefined,
    context: {
      text: string;

      : { success: true, value };

  createElementStream() {
    throw new UnsupportedFunctionalityError({
      functionality: 'element streams in no-schema mode',
    });
  },
};

const arrayOutputStrategy = <ELEMENT>(
  schema: Schema<ELEMENT>,
): OutputStrategy<ELEMENT[], ELEMENT[], AsyncIterableStream<ELEMENT>> => {
  // remove $schema from schema.jsonSchema:
  const { $schema, ...itemSchema } = schema.jsonSchema;

  return {
    type: 'array',

    // wrap in object that contains array of elements, since most LLMs will not
    // be able to generate an array directly:
    // possible future optimization: use arrays directly when model supports grammar-guided generation
    jsonSchema: {
      $schema: 'https://json-schema.org/draft/2019-09/schema#',
      type: 'object',
      properties: {
        elements: { type: 'array', items: itemSchema },
      },
      required: ['elements'],
      additionalProperties: false,
    },

    validatePartialResult({ value, latestObject, isFirstDelta, isFinalDelta }) {
      // check that the value is an object that contains an array of elements:
      if (!isJSONObject(value) || !isJSONArray(value.elements)) {
        return {
          success: false,
          error: new TypeValidationError({
            value,
            cause: 'value must be an object that contains an array of elements',
          }),
        };
      }

      const inputArray = value.elements as Array<JSONObject>;
      const resultArray: Array<ELEMENT> = [];

      for (let i = 0; i < inputArray.length; i++) {
        const element = inputArray[i];
        const result = safeValidateTypes({ value: element, schema });

        // special treatment for last processed element:
        // ignore parse or validation failures, since they indicate that the
        // last element is incomplete and should not be included in the result,
        // unless it is the final delta
        if (i === inputArray.length - 1 && !isFinalDelta) {
          continue;
        }

        if (!result.success) {
          return result;
        }

        resultArray.push(result.value);
      }

      // calculate delta:
      const publishedElementCount = latestObject?.length ?? 0;

      let textDelta = '';

      if (isFirstDelta) {
        textDelta += '[';
      }

      if (publishedElementCount > 0) {
        textDelta += ',';
      }

      textDelta += resultArray
        .slice(publishedElementCount) // only new elements
        .map(element => JSON.stringify(element))
        .join(',');

      if (isFinalDelta) {
        textDelta += ']';
      }

      return {
        success: true,
        value: {
          partial: resultArray,
          textDelta,
        },
      };
    },

    validateFinalResult(
      value: JSONValue | undefined,
    ): ValidationResult<Array<ELEMENT>> {
      // check that the value is an object that contains an array of elements:
      if (!isJSONObject(value) || !isJSONArray(value.elements)) {
        return {
          success: false,
          error: new TypeValidationError({
            value,
            cause: 'value must be an object that contains an array of elements',
          }),
        };
      }

      const inputArray = value.elements as Array<JSONObject>;

      // check that each element in the array is of the correct type:
      for (const element of inputArray) {
        const result = safeValidateTypes({ value: element, schema });
        if (!result.success) {
          return result;
        }
      }

      return { success: true, value: inputArray as Array<ELEMENT> };
    },

    createElementStream(
      originalStream: ReadableStream<ObjectStreamPart<ELEMENT[]>>,
    ) {
      let publishedElements = 0;

      return createAsyncIterableStream(
        originalStream.pipeThrough(
          new TransformStream<ObjectStreamPart<ELEMENT[]>, ELEMENT>({
            transform(chunk, controller) {
              switch (chunk.type) {
                case 'object': {
                  const array = chunk.object;

                  // publish new elements one by one:
                  for (
                    ;
                    publishedElements < array.length;
                    publishedElements++
                  ) {
                    controller.enqueue(array[publishedElements]);
                  }

                  break;
                }

                case 'text-delta':
                case 'finish':
                  break;

                case 'error':
                  controller.error(chunk.error);
                  break;

                default: {
                  const _exhaustiveCheck: never = chunk;
                  throw new Error(
                    `Unsupported chunk type: ${_exhaustiveCheck}`,
                  );
                }
              }
            },
          }),
        ),
      );
    },
  };
};

    },

    validateFinalResult(value: JSONValue | undefined): ValidationResult<ENUM> {
      // check that the value is an object that contains an array of elements:
      if (!isJSONObject(value) || typeof value.result !== 'string') {
        return {
          success: false,
          error: new TypeValidationError({
            value,
            cause:
              'value must be an object that contains a string in the "result" property.',
          }),
        };
      }

      const result = value.result as string;

      return enumValues.includes(result as ENUM)
        ? { success: true, value: result as ENUM }
        : {
            success: false,
            error: new TypeValidationError({
              value,
              cause: 'value must be a string in the enum',
            }),
          };
    },

    validatePartialResult() {
      // no streaming in enum mode
      throw new UnsupportedFunctionalityError({
        functionality: 'partial results in enum mode',
      });
    },

    createElementStream() {
      // no streaming in enum mode
      throw new UnsupportedFunctionalityError({
        functionality: 'element streams in enum mode',
      });
    },
  };
};

export function getOutputStrategy<SCHEMA>({
  output,
  schema,
  enumValues,
}: {
  output: 'object' | 'array' | 'enum' | 'no-schema';
  schema?: z.Schema<SCHEMA, z.ZodTypeDef, any> | Schema<SCHEMA>;
  enumValues?: Array<SCHEMA>;
}): OutputStrategy<any, any, any> {
  switch (output) {
    case 'object':
      return objectOutputStrategy(withDescriptions(asSchema(schema!), schema));
    case 'array':
      return arrayOutputStrategy(withDescriptions(asSchema(schema!), schema));
    case 'enum':
      return enumOutputStrategy(enumValues! as Array<string>);
    case 'no-schema':
      return noSchemaOutputStrategy;
    default: {
      const _exhaustiveCheck: never = output;
      throw new Error(`Unsupported output: ${_exhaustiveCheck}`);
    }
  }
}

// Minimal, duck-typed view on a Zod schema. It is only used to read the
// information that is needed to restore descriptions in JSON schemas, so that
// this module does not need a hard dependency on zod.
type ZodSchemaLike = {
  description?: string;
  _def?: {
    typeName?: string;
    innerType?: ZodSchemaLike;
    schema?: ZodSchemaLike;
    type?: ZodSchemaLike;
    in?: ZodSchemaLike;
    element?: ZodSchemaLike;
    valueType?: ZodSchemaLike;
    shape?: () => Record<string, ZodSchemaLike>;
    options?: Array<ZodSchemaLike> | Map<unknown, ZodSchemaLike>;
  };
};

function isZodSchemaLike(value: unknown): value is ZodSchemaLike {
  return (
    typeof value === 'object' &&
    value != null &&
    typeof (value as ZodSchemaLike)._def === 'object' &&
    (value as ZodSchemaLike)._def != null
  );
}

function isNullableZodSchema(zodSchema: ZodSchemaLike | undefined): boolean {
  if (zodSchema == null) {
    return false;
  }

  const def = zodSchema._def;

  switch (def?.typeName) {
    case 'ZodNullable':
      return true;
    case 'ZodOptional':
    case 'ZodDefault':
    case 'ZodReadonly':
    case 'ZodCatch':
      return isNullableZodSchema(def.innerType);
    case 'ZodUnion': {
      const options = def.options;
      const optionSchemas =
        options instanceof Map ? [...options.values()] : options;

      return (
        Array.isArray(optionSchemas) && optionSchemas.some(isNullableZodSchema)
      );
    }
    default:
      return false;
  }
}

/**
 * Returns a copy of the given JSON schema in which the wrappers that are
 * created for optional fields (`anyOf` with a single branch) are removed.
 */
function unwrapAnyOf(jsonSchema: JSONSchema7): JSONSchema7 {
  const anyOf = jsonSchema.anyOf;

  if (!Array.isArray(anyOf) || anyOf.length !== 1) {
    return jsonSchema;
  }

  const branch = anyOf[0];

  if (branch == null || typeof branch !== 'object') {
    return jsonSchema;
  }

  const result: JSONSchema7 = { ...(branch as JSONSchema7) };

  for (const [key, value] of Object.entries(jsonSchema)) {
    if (key === 'anyOf' || value === undefined) {
      continue;
    }

    (result as Record<string, unknown>)[key] = value;
  }

  return result;
}

/**
 * Returns a copy of the given JSON schema in which the artificial `null` type
 * that is created for optional fields is removed.
 */
function removeNullType(jsonSchema: JSONSchema7): JSONSchema7 {
  const types = jsonSchema.type;

  if (!Array.isArray(types) || !types.includes('null')) {
    return jsonSchema;
  }

  const nonNullTypes = types.filter(type => type !== 'null');

  if (nonNullTypes.length === 0) {
    return jsonSchema;
  }

  return {
    ...jsonSchema,
    type: nonNullTypes.length === 1 ? nonNullTypes[0] : nonNullTypes,
  };
}

/**
 * Copies the descriptions of the given Zod schema into the given JSON schema.
 *
 * When a Zod schema is converted into a JSON schema, the description of an
 * optional field ends up outside of the property schema: the field is either
 * wrapped in an `anyOf` branch or turned into a nullable type. LLM providers
 * however only pick up descriptions from the property schema itself, so the
 * descriptions need to be restored (and the optional wrappers removed).
 */
function addZodDescriptions(
  zodSchema: ZodSchemaLike,
  jsonSchema: JSONSchema7,
): JSONSchema7 {
  const def: NonNullable<ZodSchemaLike['_def']> = zodSchema._def ?? {};

  let result: JSONSchema7 = jsonSchema;

  switch (def.typeName) {
    // wrappers that contain the actual schema:
    case 'ZodOptional':
    case 'ZodNullable':
    case 'ZodDefault':
    case 'ZodReadonly':
    case 'ZodCatch': {
      result = unwrapAnyOf(result);

      // optional fields are not nullable:
      if (
        def.typeName === 'ZodOptional' &&
        !isNullableZodSchema(def.innerType)
      ) {
        result = removeNullType(result);
      }

      if (def.innerType != null) {
        result = addZodDescriptions(def.innerType, result);
      }
      break;
    }

    case 'ZodObject': {
      const shape = def.shape?.();

      if (shape != null && result.properties != null) {
        const properties: Record<string, JSONSchema7> = {};

        for (const [key, propertyJsonSchema] of Object.entries(
          result.properties,
        )) {
          const propertyZodSchema = shape[key];

          properties[key] =
            propertyZodSchema != null &&
            propertyJsonSchema != null &&
            typeof propertyJsonSchema === 'object'
              ? addZodDescriptions(
                  propertyZodSchema,
                  propertyJsonSchema as JSONSchema7,
                )
              : (propertyJsonSchema as JSONSchema7);
        }

        result = { ...result, properties };
      }
      break;
    }

    case 'ZodArray': {
      const items = result.items;

      if (
        def.element != null &&
        items != null &&
        !Array.isArray(items) &&
        typeof items === 'object'
      ) {
        result = { ...result, items: addZodDescriptions(def.element, items) };
      }
      break;
    }

    case 'ZodEffects': {
      if (def.schema != null) {
        result = addZodDescriptions(def.schema, result);
      }
      break;
    }

    case 'ZodBranded': {
      if (def.type != null) {
        result = addZodDescriptions(def.type, result);
      }
      break;
    }

    case 'ZodPipeline': {
      if (def.in != null) {
        result = addZodDescriptions(def.in, result);
      }
      break;
    }

    case 'ZodUnion':
    case 'ZodDiscriminatedUnion': {
      const options = def.options;
      const optionSchemas =
        options instanceof Map ? [...options.values()] : options;
      const branches = result.anyOf;

      if (Array.isArray(optionSchemas) && Array.isArray(branches)) {
        result = {
          ...result,
          anyOf: branches.map((branch, index) => {
            const branchZodSchema = optionSchemas[index];

            return branchZodSchema != null &&
              branch != null &&
              typeof branch === 'object'
              ? addZodDescriptions(branchZodSchema, branch as JSONSchema7)
              : branch;
          }),
        };
      }
      break;
    }

    case 'ZodRecord': {
      const additionalProperties = result.additionalProperties;

      if (
        def.valueType != null &&
        additionalProperties != null &&
        typeof additionalProperties === 'object'
      ) {
        result = {
          ...result,
          additionalProperties: addZodDescriptions(
            def.valueType,
            additionalProperties as JSONSchema7,
          ),
        };
      }
      break;
    }

    default:
      break;
  }

  // the description of the outermost zod schema takes precedence:
  if (zodSchema.description != null) {
    result = { ...result, description: zodSchema.description };
  } else if (result.description == null && Array.isArray(result.anyOf)) {
    // restore descriptions that ended up in the anyOf branches:
    const branchDescription = result.anyOf
      .map(branch =>
        branch != null && typeof branch === 'object'
          ? (branch as JSONSchema7).description
          : undefined,
      )
      .find(description => description != null);

    if (branchDescription != null) {
      result = { ...result, description: branchDescription };
    }
  }

  return result;
}

/**
 * Restores the descriptions of optional fields of the given Zod schema in the
 * generated JSON schema.
 */
function addZodDescriptionsToJsonSchema(
  jsonSchema: JSONSchema7,
  originalSchema?: unknown,
): JSONSchema7 {
  return isZodSchemaLike(originalSchema)
    ? addZodDescriptions(originalSchema, jsonSchema)
    : jsonSchema;
}

export function prepareToolsAndToolChoice<
  TOOLS extends Record<string, CoreTool>,
>({
  tools,
  toolChoice,
  activeTools,
}: {
  tools: TOOLS | undefined;
  toolChoice: CoreToolChoice<TOOLS> | undefined;
  activeTools: Array<keyof TOOLS> | undefined;
}): {
  tools:
    | Array<LanguageModelV1FunctionTool | LanguageModelV1ProviderDefinedTool>
    | undefined;
  toolChoice: LanguageModelV1ToolChoice | undefined;
} {
  if (!isNonEmptyObject(tools)) {
    return {
      tools: undefined,
      toolChoice: undefined,
    };
  }

  // when activeTools is provided, we only include the tools that are in the list:
  const filteredTools =
    activeTools != null
      ? Object.entries(tools).filter(([name]) =>
          activeTools.includes(name as keyof TOOLS),
        )
      : Object.entries(tools);

  return {
    tools: filteredTools.map(([name, tool]) => {
      const toolType = tool.type;
      switch (toolType) {
        case undefined:
        case 'function':
          return {
            type: 'function' as const,
            name,
            description: tool.description,
            parameters: addZodDescriptionsToJsonSchema(
              asSchema(tool.parameters).jsonSchema,
              tool.parameters,
            ),
          };
        case 'provider-defined':
          return {
            type: 'provider-defined' as const,
            name,
            id: tool.id,
            args: tool.args,
          };
        default: {
          const exhaustiveCheck: never = toolType;
          throw new Error(`Unsupported tool type: ${exhaustiveCheck}`);
        }
      }
    }),
    toolChoice:
      toolChoice == null
        ? { type: 'auto' }
        : typeof toolChoice === 'string'
        ? { type: toolChoice }
        : { type: 'tool' as const, toolName: toolChoice.toolName as string },
  };
}

  }

  const def = zodSchema._def;

  switch (def?.typeName) {
    case 'ZodNullable':
      return true;
    case 'ZodOptional':
    case 'ZodDefault':
    case 'ZodReadonly':
    case 'ZodCatch':
      return isNullableZodSchema(def.innerType);
    case 'ZodUnion': {
      const options = def.options;
      const optionSchemas =
        options instanceof Map ? [...options.values()] : options;

      return (
        Array.isArray(optionSchemas) && optionSchemas.some(isNullableZodSchema)
      );
    }
    default:
      return false;
  }
}

/**
 * Returns a copy of the given JSON schema in which the wrappers that are
 * created for optional fields (`anyOf` with a single branch) are removed.
 */
function unwrapAnyOf(jsonSchema: JSONSchema7): JSONSchema7 {
  const anyOf = jsonSchema.anyOf;

  if (!Array.isArray(anyOf) || anyOf.length !== 1) {
    return jsonSchema;
  }

  const branch = anyOf[0];

  if (branch == null || typeof branch !== 'object') {
    return jsonSchema;
  }

  const result: JSONSchema7 = { ...(branch as JSONSchema7) };

  for (const [key, value] of Object.entries(jsonSchema)) {
    if (key === 'anyOf' || value === undefined) {
      continue;
    }

    (result as Record<string, unknown>)[key] = value;
  }

  return result;
}

/**
 * Returns a copy of the given JSON schema in which the artificial `null` type
 * that is created for optional fields is removed.
 */
function removeNullType(jsonSchema: JSONSchema7): JSONSchema7 {
  const types = jsonSchema.type;

  if (!Array.isArray(types) || !types.includes('null')) {
    return jsonSchema;
  }

  const nonNullTypes = types.filter(type => type !== 'null');

  if (nonNullTypes.length === 0) {
    return jsonSchema;
  }

  return {
    ...jsonSchema,
    type: nonNullTypes.length === 1 ? nonNullTypes[0] : nonNullTypes,
  };
}

/**
 * Copies the descriptions of the given Zod schema into the given JSON schema.
 *
 * When a Zod schema is converted into a JSON schema, the description of an
 * optional field ends up outside of the property schema: the field is either
 * wrapped in an `anyOf` branch or turned into a nullable type. LLM providers
 * however only pick up descriptions from the property schema itself, so the
 * descriptions need to be restored (and the optional wrappers removed).
 */
function addZodDescriptions(
  zodSchema: ZodSchemaLike,
  jsonSchema: JSONSchema7,
): JSONSchema7 {
  const def: NonNullable<ZodSchemaLike['_def']> = zodSchema._def ?? {};

  let result: JSONSchema7 = jsonSchema;

  switch (def.typeName) {
    // wrappers that contain the actual schema:
    case 'ZodOptional':
    case 'ZodNullable':
    case 'ZodDefault':
    case 'ZodReadonly':
    case 'ZodCatch': {
      result = unwrapAnyOf(result);

      // optional fields are not nullable:
      if (
        def.typeName === 'ZodOptional' &&
        !isNullableZodSchema(def.innerType)
      ) {
        result = removeNullType(result);
      }

      if (def.innerType != null) {
        result = addZodDescriptions(def.innerType, result);
      }
      break;
    }

    case 'ZodObject': {
      const shape = def.shape?.();

      if (shape != null && result.properties != null) {
        const properties: Record<string, JSONSchema7> = {};

        for (const [key, propertyJsonSchema] of Object.entries(
          result.properties,
        )) {
          const propertyZodSchema = shape[key];

          properties[key] =
            propertyZodSchema != null &&
            propertyJsonSchema != null &&
            typeof propertyJsonSchema === 'object'
              ? addZodDescriptions(
                  propertyZodSchema,
                  propertyJsonSchema as JSONSchema7,
                )
              : (propertyJsonSchema as JSONSchema7);
        }

        result = { ...result, properties };
      }
      break;
    }

    case 'ZodArray': {
      const items = result.items;

      if (
        def.element != null &&
        items != null &&
        !Array.isArray(items) &&
        typeof items === 'object'
      ) {
        result = { ...result, items: addZodDescriptions(def.element, items) };
      }
      break;
    }

    case 'ZodEffects': {
      if (def.schema != null) {
        result = addZodDescriptions(def.schema, result);
      }
      break;
    }

    case 'ZodBranded': {
      if (def.type != null) {
        result = addZodDescriptions(def.type, result);
      }
      break;
    }

    case 'ZodPipeline': {
      if (def.in != null) {
        result = addZodDescriptions(def.in, result);
      }
      break;
    }

    case 'ZodUnion':
    case 'ZodDiscriminatedUnion': {
      const options = def.options;
      const optionSchemas =
        options instanceof Map ? [...options.values()] : options;
      const branches = result.anyOf;

      if (Array.isArray(optionSchemas) && Array.isArray(branches)) {
        result = {
          ...result,
          anyOf: branches.map((branch, index) => {
            const branchZodSchema = optionSchemas[index];

            return branchZodSchema != null &&
              branch != null &&
              typeof branch === 'object'
              ? addZodDescriptions(branchZodSchema, branch as JSONSchema7)
              : branch;
          }),
        };
      }
      break;
    }

    case 'ZodRecord': {
      const additionalProperties = result.additionalProperties;

      if (
        def.valueType != null &&
        additionalProperties != null &&
        typeof additionalProperties === 'object'
      ) {
        result = {
          ...result,
          additionalProperties: addZodDescriptions(
            def.valueType,
            additionalProperties as JSONSchema7,
          ),
        };
      }
      break;
    }

    default:
      break;
  }

  // the description of the outermost zod schema takes precedence:
  if (zodSchema.description != null) {
    result = { ...result, description: zodSchema.description };
  } else if (result.description == null && Array.isArray(result.anyOf)) {
    // restore descriptions that ended up in the anyOf branches:
    const branchDescription = result.anyOf
      .map(branch =>
        branch != null && typeof branch === 'object'
          ? (branch as JSONSchema7).description
          : undefined,
      )
      .find(description => description != null);

    if (branchDescription != null) {
      result = { ...result, description: branchDescription };
    }
  }

  return result;
}

/**
 * Returns a copy of the given schema whose JSON schema keeps the descriptions
 * of optional fields (see {@link addZodDescriptions}).
 */
function withDescriptions<OBJECT>(
  schema: Schema<OBJECT>,
  originalSchema?: unknown,
): Schema<OBJECT> {
  return isZodSchemaLike(originalSchema)
    ? {
        ...schema,
        jsonSchema: addZodDescriptions(originalSchema, schema.jsonSchema),
      }
    : schema;
}

const noSchemaOutputStrategy: OutputStrategy<JSONValue, JSONValue, never> = {
  type: 'no-schema',
  jsonSchema: undefined,

  validatePartialResult({ value, textDelta }) {
    return { success: true, value: { partial: value, textDelta } };
  },

  validateFinalResult(
    value: JSONValue | undefined,
    context: {
      text: string;
      response: LanguageModelResponseMetadata;
      usage: LanguageModelUsage;
    },
  ): ValidationResult<JSONValue> {
    return value === undefined
      ? {
          success: false,
          error: new NoObjectGeneratedError({
            message: 'No object generated: response did not match schema.',
            text: context.text,
            response: context.response,
            usage: context.usage,
          }),
        }
      : { success: true, value };
  },

  createElementStream() {
    throw new UnsupportedFunctionalityError({
      functionality: 'element streams in no-schema mode',
    });
  },
};

const objectOutputStrategy = <OBJECT>(
  schema: Schema<OBJECT>,
): OutputStrategy<DeepPartial<OBJECT>, OBJECT, never> => ({
  type: 'object',
  jsonSchema: schema.jsonSchema,

  validatePartialResult({ value, textDelta }) {
    return {
      success: true,
      value: {
        // Note: currently no validation of partial results:
        partial: value as DeepPartial<OBJECT>,
        textDelta,
      },
    };
  },

  validateFinalResult(value: JSONValue | undefined): ValidationResult<OBJECT> {
    return safeValidateTypes({ value, schema });
  },

  createElementStream() {
    throw new UnsupportedFunctionalityError({
      functionality: 'element streams in object mode',
    });
  },
});

const arrayOutputStrategy = <ELEMENT>(
  schema: Schema<ELEMENT>,
): OutputStrategy<ELEMENT[], ELEMENT[], AsyncIterableStream<ELEMENT>> => {
  // remove $schema from schema.jsonSchema:
  const { $schema, ...itemSchema } = schema.jsonSchema;

  return {
    type: 'array',

    // wrap in object that contains array of elements, since most LLMs will not
    // be able to generate an array directly:
    // possible future optimization: use arrays directly when model supports grammar-guided generation
    jsonSchema: {
      $schema: 'https://json-schema.org/draft/2019-09/schema#',
      type: 'object',
      properties: {
        elements: { type: 'array', items: itemSchema },
      },
      required: ['elements'],
      additionalProperties: false,
    },

    validatePartialResult({ value, latestObject, isFirstDelta, isFinalDelta }) {
      // check that the value is an object that contains an array of elements:
      if (!isJSONObject(value) || !isJSONArray(value.elements)) {
        return {
          success: false,
          error: new TypeValidationError({
            value,
            cause: 'value must be an object that contains an array of elements',
          }),
        };
      }

      const inputArray = value.elements as Array<JSONObject>;
      const resultArray: Array<ELEMENT> = [];

      for (let i = 0; i < inputArray.length; i++) {
        const element = inputArray[i];
        const result = safeValidateTypes({ value: element, schema });

        // special treatment for last processed element:
        // ignore parse or validation failures, since they indicate that the
        // last element is incomplete and should not be included in the result,
        // unless it is the final delta
        if (i === inputArray.length - 1 && !isFinalDelta) {
          continue;
        }

        if (!result.success) {
          return result;
        }

        resultArray.push(result.value);
      }

      // calculate delta:
      const publishedElementCount = latestObject?.length ?? 0;

      let textDelta = '';

      if (isFirstDelta) {
        textDelta += '[';
      }

      if (publishedElementCount > 0) {
        textDelta += ',';
      }

      textDelta += resultArray
        .slice(publishedElementCount) // only new elements
        .map(element => JSON.stringify(element))
        .join(',');

      if (isFinalDelta) {
        textDelta += ']';
      }

      return {
        success: true,
        value: {
          partial: resultArray,
          textDelta,
        },
      };
    },

    validateFinalResult(
      value: JSONValue | undefined,
    ): ValidationResult<Array<ELEMENT>> {
      // check that the value is an object that contains an array of elements:
      if (!isJSONObject(value) || !isJSONArray(value.elements)) {
        return {
          success: false,
          error: new TypeValidationError({
            value,
            cause: 'value must be an object that contains an array of elements',
          }),
        };
      }

      const inputArray = value.elements as Array<JSONObject>;

      // check that each element in the array is of the correct type:
      for (const element of inputArray) {
        const result = safeValidateTypes({ value: element, schema });
        if (!result.success) {
          return result;
        }
      }

      return { success: true, value: inputArray as Array<ELEMENT> };
    },

    createElementStream(
      originalStream: ReadableStream<ObjectStreamPart<ELEMENT[]>>,
    ) {
      let publishedElements = 0;

      return createAsyncIterableStream(
        originalStream.pipeThrough(
          new TransformStream<ObjectStreamPart<ELEMENT[]>, ELEMENT>({
            transform(chunk, controller) {
              switch (chunk.type) {
                case 'object': {
                  const array = chunk.object;

                  // publish new elements one by one:
                  for (
                    ;
                    publishedElements < array.length;
                    publishedElements++
                  ) {
                    controller.enqueue(array[publishedElements]);
                  }

                  break;
                }

                case 'text-delta':
                case 'finish':
                  break;

                case 'error':
                  controller.error(chunk.error);
                  break;

                default: {
                  const _exhaustiveCheck: never = chunk;
                  throw new Error(
                    `Unsupported chunk type: ${_exhaustiveCheck}`,
                  );
                }
              }
            },
          }),
        ),
      );
    },
  };
};

const enumOutputStrategy = <ENUM extends string>(
  enumValues: Array<ENUM>,
): OutputStrategy<ENUM, ENUM, never> => {
  return {
    type: 'enum',

    // wrap in object that contains result, since most LLMs will not
    // be able to generate an enum value directly:
    // possible future optimization: use enums directly when model supports top-level enums
    jsonSchema: {
      $schema: 'https://json-schema.org/draft/2019-09/schema#',
      type: 'object',
      properties: {
        result: { type: 'string', enum: enumValues },
      },
      required: ['result'],
      additionalProperties: false,
    },

    validateFinalResult(value: JSONValue | undefined): ValidationResult<ENUM> {
      // check that the value is an object that contains an array of elements:
      if (!isJSONObject(value) || typeof value.result !== 'string') {
        return {
          success: false,
          error: new TypeValidationError({
            value,
            cause:
              'value must be an object that contains a string in the "result" property.',
          }),
        };
      }

      const result = value.result as string;

      return enumValues.includes(result as ENUM)
        ? { success: true, value: result as ENUM }
        : {
            success: false,
            error: new TypeValidationError({
              value,
              cause: 'value must be a string in the enum',
            }),
          };
    },

    validatePartialResult() {
      // no streaming in enum mode
      throw new UnsupportedFunctionalityError({
        functionality: 'partial results in enum mode',
      });
    },

    createElementStream() {
      // no streaming in enum mode
      throw new UnsupportedFunctionalityError({
        functionality: 'element streams in enum mode',
      });
    },
  };
};

export function getOutputStrategy<SCHEMA>({
  output,
  schema,
  enumValues,
}: {
  output: 'object' | 'array' | 'enum' | 'no-schema';
  schema?: z.Schema<SCHEMA, z.ZodTypeDef, any> | Schema<SCHEMA>;
  enumValues?: Array<SCHEMA>;
}): OutputStrategy<any, any, any> {
  switch (output) {
    case 'object':
      return objectOutputStrategy(withDescriptions(asSchema(schema!), schema));
    case 'array':
      return arrayOutputStrategy(withDescriptions(asSchema(schema!), schema));
    case 'enum':
      return enumOutputStrategy(enumValues! as Array<string>);
    case 'no-schema':
      return noSchemaOutputStrategy;
    default: {
      const _exhaustiveCheck: never = output;
      throw new Error(`Unsupported output: ${_exhaustiveCheck}`);
    }
  }
}
