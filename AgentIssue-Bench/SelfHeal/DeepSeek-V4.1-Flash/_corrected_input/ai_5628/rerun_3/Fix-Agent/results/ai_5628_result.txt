### packages/google/src/convert-json-schema-to-openapi-schema.ts
import { JSONSchema7Definition } from '@ai-sdk/provider';

/**
 * Converts JSON Schema 7 to OpenAPI Schema 3.0
 */
export function convertJSONSchemaToOpenAPISchema(
  jsonSchema: JSONSchema7Definition | undefined,
  isRootSchema: boolean = true,
): unknown {
  // parameters need to be undefined if they are empty objects:
  // this is only valid for the root schema. Nested schemas (e.g. the ones
  // generated for `z.record(z.string())` or `z.object({})`) must be kept,
  // otherwise the generated OpenAPI schema contains `undefined` values
  // (e.g. `properties: { data: undefined }`) which are rejected by Google:
  if (jsonSchema == null || isEmptyObjectSchema(jsonSchema, isRootSchema)) {
    return undefined;
  }

  if (typeof jsonSchema === 'boolean') {
    return { type: 'boolean', properties: {} };
  }

  const {
    type,
    description,
    required,
    properties,
    items,
    allOf,
    anyOf,
    oneOf,
    format,
    const: constValue,
    minLength,
    enum: enumValues,
  } = jsonSchema;

  const result: Record<string, unknown> = {};

  if (description) result.description = description;
  if (required) result.required = required;
  if (format) result.format = format;

  if (constValue !== undefined) {
    result.enum = [constValue];
  }

  // Handle type
  if (type) {
    if (Array.isArray(type)) {
      if (type.includes('null')) {
        result.type = type.filter(t => t !== 'null')[0];
        result.nullable = true;
      } else {
        result.type = type;
      }
    } else if (type === 'null') {
      result.type = 'null';
    } else {
      result.type = type;
    }
  }

  // Handle enum
  if (enumValues !== undefined) {
    result.enum = enumValues;
  }

  if (properties != null) {
    result.properties = Object.entries(properties).reduce(
      (acc, [key, value]) => {
        acc[key] = convertJSONSchemaToOpenAPISchema(value, false);
        return acc;
      },
      {} as Record<string, unknown>,
    );
  }

  if (items) {
    result.items = Array.isArray(items)
      ? items.map(item => convertJSONSchemaToOpenAPISchema(item, false))
      : convertJSONSchemaToOpenAPISchema(items, false);
  }

  if (allOf) {
    result.allOf = allOf.map(schema =>
      convertJSONSchemaToOpenAPISchema(schema, false),
    );
  }
  if (anyOf) {
    // Handle cases where anyOf includes a null type
    if (
      anyOf.some(
        schema => typeof schema === 'object' && schema?.type === 'null',
      )
    ) {
      const nonNullSchemas = anyOf.filter(
        schema => !(typeof schema === 'object' && schema?.type === 'null'),
      );

      if (nonNullSchemas.length === 1) {
        // If there's only one non-null schema, convert it and make it nullable
        const converted = convertJSONSchemaToOpenAPISchema(
          nonNullSchemas[0],
          false,
        );
        if (typeof converted === 'object') {
          result.nullable = true;
          Object.assign(result, converted);
        }
      } else {
        // If there are multiple non-null schemas, keep them in anyOf
        result.anyOf = nonNullSchemas.map(schema =>
          convertJSONSchemaToOpenAPISchema(schema, false),
        );
        result.nullable = true;
      }
    } else {
      result.anyOf = anyOf.map(schema =>
        convertJSONSchemaToOpenAPISchema(schema, false),
      );
    }
  }
  if (oneOf) {
    result.oneOf = oneOf.map(schema =>
      convertJSONSchemaToOpenAPISchema(schema, false),
    );
  }

  if (minLength !== undefined) {
    result.minLength = minLength;
  }

  return result;
}

/**
 * Empty object schemas can only be dropped when they are the root schema
 * (= tools without parameters). Schemas that allow additional properties
 * (e.g. `z.record(z.string())`) are not empty either, since they describe
 * an object with arbitrary keys.
 */
function isEmptyObjectSchema(
  jsonSchema: JSONSchema7Definition,
  isRootSchema: boolean,
): boolean {
  return (
    isRootSchema &&
    jsonSchema != null &&
    typeof jsonSchema === 'object' &&
    jsonSchema.type === 'object' &&
    (jsonSchema.properties == null ||
      Object.keys(jsonSchema.properties).length === 0) &&
    !jsonSchema.additionalProperties
  );
}

### 1. What the diff changes

The fix touches `packages/google/src/convert-json-schema-to-openapi-schema.ts`:

### 2. Why this fixes the reported bug

`z.record(z.string())` produces `{ type: 'object', additionalProperties: { type: 'string' } }` with **no `properties`**. The old `isEmptyObjectSchema` classified this as an "empty object schema", so the recursive conversion returned `undefined`, yielding `properties: { metadata: undefined }`. `JSON.stringify` silently drops such keys, but the SDK hands the object directly to Google, which rejects the request. Two independent guards now remove that failure mode:

1. `additionalProperties` makes a schema non-empty → records are no longer collapsed.
2. The `isRootSchema` flag → even genuinely empty nested object schemas (e.g. `z.object({})` inside a parent, or inside `items`) are kept as `{ type: 'object' }` instead of `undefined`, so no `undefined` values can leak into nested `properties`/`items`/`anyOf`/`oneOf` output.

Dropping empty **root** schemas is preserved (that's the "tool has no parameters" case), which is exactly the only place the old behavior was desirable — matching upstream ai-sdk semantics (`isRootSchema` default `true`, `!jsonSchema.additionalProperties` guard).

- `should convert nullable string enum`: `anyOf: [{type:'string',enum:[...]}, {type:'null'}]` → still one non-null schema → `nullable: true` + merged fields. ✔

### 4. Type-safety / logic details

- `jsonSchema != null && typeof jsonSchema === 'object'` correctly narrows `JSONSchema7Definition` (`JSONSchema7 | boolean`) before touching `.type` / `.additionalProperties`; `additionalProperties?: boolean | JSONSchema7Definition` is legal in a `!` test.
- `typeof converted === 'object'` in the `anyOf` branch is now safer: a nested empty object no longer yields `undefined` (which would have skipped the `nullable` assignment).
- Boolean sub-schemas still map to `{ type: 'boolean', properties: {} }`, and with `isRootSchema === false` they can no longer be silently dropped.
- Added tests pin the intended behavior: nested empty objects kept, records kept (`{ type: 'object' }`), records inside arrays kept, root record not turned into `undefined`.

### 5. Caveats (non-blocking)

- The task's harness test only inspects `zodToJsonSchema(tool.parameters)`, i.e. it validates the Zod→JSON-Schema layer, not the converter itself, so it would pass even for a no-op change. It still passes for this fix (`metadata.additionalProperties` is defined), and the repo-side tests in the diff *do* exercise the converter directly — so the fix is properly covered where it matters.
- The converter still does not emit `additionalProperties` in output (pre-existing behavior, unchanged by this diff), so record value types are dropped at the provider level; this is a lossy-but-safe behavior consistent with OpenAPI/Gemini constraints and not introduced here.

### Verdict

The change is minimal, type-correct, internally consistent, preserves all root-level behavior (`undefined` for parameterless tools), and eliminates the `undefined`-in-output class of failures that caused the Vertex AI 400 errors for `z.record(z.string())` and nested `z.object({})`.

### packages/google/src/convert-json-schema-to-openapi-schema.ts
import { JSONSchema7Definition } from '@ai-sdk/provider';

/**
 * Converts JSON Schema 7 to OpenAPI Schema 3.0
 */
export function convertJSONSchemaToOpenAPISchema(
  jsonSchema: JSONSchema7Definition | undefined,
  isRootSchema: boolean = true,
): unknown {
  // parameters need to be undefined if they are empty objects:
  // this is only valid for the root schema. Nested schemas (e.g. the ones
  // generated for `z.record(z.string())` or `z.object({})`) must be kept,
  // otherwise the generated OpenAPI schema contains `undefined` values
  // (e.g. `properties: { data: undefined }`) which are rejected by Google:
  if (jsonSchema == null || isEmptyObjectSchema(jsonSchema, isRootSchema)) {
    return undefined;
  }

  if (typeof jsonSchema === 'boolean') {
    return { type: 'boolean', properties: {} };
  }

  const {
    type,
    description,
    required,
    properties,
    items,
    allOf,
    anyOf,
    oneOf,
    format,
    const: constValue,
    minLength,
    enum: enumValues,
  } = jsonSchema;

  const result: Record<string, unknown> = {};

  if (description) result.description = description;
  if (required) result.required = required;
  if (format) result.format = format;

  if (constValue !== undefined) {
    result.enum = [constValue];
  }

  // Handle type
  if (type) {
    if (Array.isArray(type)) {
      if (type.includes('null')) {
        result.type = type.filter(t => t !== 'null')[0];
        result.nullable = true;
      } else {
        result.type = type;
      }
    } else if (type === 'null') {
      result.type = 'null';
    } else {
      result.type = type;
    }
  }

  // Handle enum
  if (enumValues !== undefined) {
    result.enum = enumValues;
  }

  if (properties != null) {
    result.properties = Object.entries(properties).reduce(
      (acc, [key, value]) => {
        acc[key] = convertJSONSchemaToOpenAPISchema(value, false);
        return acc;
      },
      {} as Record<string, unknown>,
    );
  }

  if (items) {
    result.items = Array.isArray(items)
      ? items.map(item => convertJSONSchemaToOpenAPISchema(item, false))
      : convertJSONSchemaToOpenAPISchema(items, false);
  }

  if (allOf) {
    result.allOf = allOf.map(schema =>
      convertJSONSchemaToOpenAPISchema(schema, false),
    );
  }
  if (anyOf) {
    // Handle cases where anyOf includes a null type
    if (
      anyOf.some(
        schema => typeof schema === 'object' && schema?.type === 'null',
      )
    ) {
      const nonNullSchemas = anyOf.filter(
        schema => !(typeof schema === 'object' && schema?.type === 'null'),
      );

      if (nonNullSchemas.length === 1) {
        // If there's only one non-null schema, convert it and make it nullable
        const converted = convertJSONSchemaToOpenAPISchema(
          nonNullSchemas[0],
          false,
        );
        if (typeof converted === 'object') {
          result.nullable = true;
          Object.assign(result, converted);
        }
      } else {
        // If there are multiple non-null schemas, keep them in anyOf
        result.anyOf = nonNullSchemas.map(schema =>
          convertJSONSchemaToOpenAPISchema(schema, false),
        );
        result.nullable = true;
      }
    } else {
      result.anyOf = anyOf.map(schema =>
        convertJSONSchemaToOpenAPISchema(schema, false),
      );
    }
  }
  if (oneOf) {
    result.oneOf = oneOf.map(schema =>
      convertJSONSchemaToOpenAPISchema(schema, false),
    );
  }

  if (minLength !== undefined) {
    result.minLength = minLength;
  }

  return result;
}

/**
 * Empty object schemas can only be dropped when they are the root schema
 * (= tools without parameters). Schemas that allow additional properties
 * (e.g. `z.record(z.string())`) are not empty either, since they describe
 * an object with arbitrary keys.
 */
function isEmptyObjectSchema(
  jsonSchema: JSONSchema7Definition,
  isRootSchema: boolean,
): boolean {
  return (
    isRootSchema &&
    jsonSchema != null &&
    typeof jsonSchema === 'object' &&
    jsonSchema.type === 'object' &&
    (jsonSchema.properties == null ||
      Object.keys(jsonSchema.properties).length === 0) &&
    !jsonSchema.additionalProperties
  );
}
