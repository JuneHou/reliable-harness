def print_text_animated(text, delay: float = 0.02, end: str = ""):
    r"""Prints the given text with an animated effect.

    Args:
        text (str): The text to print.
        delay (float, optional): The delay between each character printed.
            (default: :obj:`0.02`)

            character of text. (default: :obj:`""`)

        time.sleep(delay)
    print('\n')


def get_prompt_template_key_words(template: str) -> Set[str]:
    r"""Given a string template containing curly braces {}, return a set of
    the words inside the braces.

    Args:
        template (str): A string containing curly braces.

    Returns:
        List[str]: A list of the words inside the curly braces.

    Example:
        >>> get_prompt_template_key_words('Hi, {name}! How are you {status}?')

    """
    return set(re.findall(r'{([^}]*)}', template))


def get_first_int(string: str) -> Optional[int]:
    r"""Returns the first integer number found in the given string.

    If no integer number is found, returns None.

    Args:
        string (str): The input string.

    Returns:
        int or None: The first integer number found in the string, or None if
            no integer number is found.
    """
    match = re.search(r'\d+', string)
    if match:
        return int(match.group())
    else:
        return None


def download_tasks(task: TaskType, folder_path: str) -> None:
    r"""Downloads task-related files from a specified URL and extracts them.

    This function downloads a zip file containing tasks based on the specified
    `task` type from a predefined URL, saves it to `folder_path`, and then
    extracts the contents of the zip file into the same folder. After
    extraction, the zip file is deleted.

    Args:
        task (TaskType): An enum representing the type of task to download.
        folder_path (str): The path of the folder where the zip file will be
                           downloaded and extracted.
    """
    # Define the path to save the zip file
    zip_file_path = os.path.join(folder_path, "tasks.zip")

    # Download the zip file from the Google Drive link
    response = requests.get(
        "https://huggingface.co/datasets/camel-ai/"
        f"metadata/resolve/main/{task.value}_tasks.zip"
    )

    # Save the zip file
    with open(zip_file_path, "wb") as f:
        f.write(response.content)

    with zipfile.ZipFile(zip_file_path, "r") as zip_ref:
        zip_ref.extractall(folder_path)

    # Delete the zip file
    os.remove(zip_file_path)


def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()
            if task_name.strip() and task_id.isnumeric():
                new_tasks_list.append(task_name)
    return new_tasks_list


def check_server_running(server_url: str) -> bool:
    r"""Check whether the port refered by the URL to the server
    is open.

    Args:
        server_url (str): The URL to the server running LLM inference
            service.

    Returns:
        bool: Whether the port is open for packets (server is running).
    """
    parsed_url = urlparse(server_url)
    url_tuple = (parsed_url.hostname, parsed_url.port)

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    result = sock.connect_ex(url_tuple)
    sock.close()

    # if the port is open, the result should be 0.
    return result == 0


def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required module dependencies.

    Raises:
        ImportError: If any of the required modules are not available.

    Example:
        ::

            @dependencies_required('numpy', 'pandas')
            def data_processing_function():
                # Function implementation...
    """

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            missing_modules = [
                m for m in required_modules if not is_module_available(m)
            ]
            if missing_modules:
                raise ImportError(
                    f"Missing required modules: {', '.join(missing_modules)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def api_keys_required(*required_keys: Any, **options: Any) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are presented in the
    environment variables, as an instance attribute or as an argument that is
    passed directly to the decorated function (e.g. to a decorated
    ``__init__`` method).

    Args:
        required_keys: The required API keys to be checked. Every key can be
            given either as:

            * a string, i.e. the name of the environment variable that stores
              the API key, or
            * a tuple ``(environment variable name, attribute name)``, where
              the attribute name is the name of the instance attribute (or of
              the argument of the decorated function) that stores the API key.

            A list or a tuple containing any combination of the above is also
            accepted, e.g.
            ``[('OPENAI_API_KEY', 'api_key'), ('ANTHROPIC_API_KEY',
            'api_key')]``.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables, the instance attributes and the arguments
            passed directly to the decorated function.

    Example:
        ::

            @api_keys_required([('OPENAI_API_KEY', 'api_key')])
            def some_api_function(api_key=None):
                # Function implementation...
    """

    # Normalize the several accepted ways of specifying the required keys
    # into a list of ``(environment variable, attribute name)`` tuples.
    normalized_keys: List[Tuple[Optional[str], Optional[str]]] = []

    def _flatten(item: Any) -> None:
        if item is None:
            return
        if isinstance(item, str):
            # Only an environment variable name was given.
            normalized_keys.append((item, None))
        elif (
            isinstance(item, tuple)
            and len(item) == 2
            and all(
                element is None or isinstance(element, str) for element in item
            )
        ):
            env_var, attr_name = item
            if env_var is not None or attr_name is not None:
                normalized_keys.append((env_var, attr_name))
        elif isinstance(item, (list, tuple, set, frozenset)):
            for element in item:
                _flatten(element)
        else:
            raise TypeError(
                "The required API keys should be given as strings or as "
                "(environment variable, attribute) tuples, "
                f"got: {item!r}."
            )

    # Also support being called as ``api_keys_required(required_keys=...)``.
    if 'required_keys' in options:
        required_keys = required_keys + (options.pop('required_keys'),)
    if options:
        raise TypeError(
            "api_keys_required() got unexpected keyword arguments: "
            f"{', '.join(options)}."
        )

    for key in required_keys:
        _flatten(key)

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            # Resolve the arguments that are passed to the decorated
            # function, so that API keys given directly as positional or
            # keyword arguments (e.g. ``SomeClass(api_key="...")``) are
            # recognized as well.
            call_arguments: Dict[str, Any] = {}
            try:
                signature = inspect.signature(func)
            except (TypeError, ValueError):  # pragma: no cover
                signature = None

            bound: Optional[inspect.BoundArguments] = None
            if signature is not None:
                try:
                    bound = signature.bind_partial(*args, **kwargs)
                except TypeError:  # pragma: no cover
                    bound = None

            if bound is not None and signature is not None:
                for name, value in bound.arguments.items():
                    parameter = signature.parameters[name]
                    if parameter.kind is inspect.Parameter.VAR_KEYWORD:
                        call_arguments.update(value)
                    elif parameter.kind is inspect.Parameter.VAR_POSITIONAL:
                        continue
                    else:
                        call_arguments[name] = value
            else:  # pragma: no cover
                call_arguments.update(kwargs)

            instance = args[0] if args else None

            missing_keys: List[str] = []
            for env_var, attr_name in normalized_keys:
                # 1. Check the environment variables.
                if env_var is not None and os.getenv(env_var):
                    continue

                # 2. Check the instance attributes and the arguments that
                # were directly passed to the decorated function.
                attribute_names: Tuple[str, ...] = (
                    (attr_name,)
                    if attr_name is not None
                    else ('_api_key', 'api_key')
                )

                if instance is not None and any(
                    getattr(instance, name, None) for name in attribute_names
                ):
                    continue

                if any(
                    call_arguments.get(name) for name in attribute_names
                ):
                    continue

                missing_keys.append(
                    env_var if env_var is not None else attribute_names[0]
                )

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )

            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper

def handle_http_error(response: requests.Response) -> str:
    r"""Handles the HTTP errors based on the status code of the response.

    Args:
        response (requests.Response): The HTTP response from the API call.

    Returns:
        str: The error type, based on the status code.
    """
    if response.status_code == HTTPStatus.UNAUTHORIZED:
        return "Unauthorized. Check your access token."
    elif response.status_code == HTTPStatus.FORBIDDEN:
        return "Forbidden. You do not have permission to perform this action."
    elif response.status_code == HTTPStatus.NOT_FOUND:
        return "Not Found. The resource could not be located."
    elif response.status_code == HTTPStatus.TOO_MANY_REQUESTS:
        return "Too Many Requests. You have hit the rate limit."
    else:
        return "HTTP Error"


def retry_request(
    func: Callable, retries: int = 3, delay: int = 1, *args: Any, **kwargs: Any
) -> Any:
    r"""Retries a function in case of any errors.

    Args:
        func (Callable): The function to be retried.
        retries (int): Number of retry attempts. (default: :obj:`3`)
        delay (int): Delay between retries in seconds. (default: :obj:`1`)
        *args: Arguments to pass to the function.
        **kwargs: Keyword arguments to pass to the function.

    Returns:
        Any: The result of the function call if successful.

    Raises:
        Exception: If all retry attempts fail.
    """
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            print(f"Attempt {attempt + 1}/{retries} failed: {e}")
            if attempt < retries - 1:
                time.sleep(delay)
            else:
                raise

def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:

def api_keys_required(*required_keys: Any) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are presented in the
    environment variables or passed directly to the decorated function (e.g.
    a decorated ``__init__`` method) or stored as an instance attribute.

    Each required key can be specified in one of the following ways:

    - A plain string, which is the name of the environment variable expected
      to hold the API key.
    - A tuple ``(env_var_name, arg_name)``, where ``env_var_name`` is the
      name of the environment variable expected to hold the API key and
      ``arg_name`` is the name of the argument (or of the instance attribute
      for methods) that holds the API key. Either element may be ``None``, in
      which case only the remaining one is used.

    A key is considered as provided as soon as one of the given identifiers is
    resolved: the environment variable is set to a non-empty value, or the
    corresponding argument (passed positionally or as a keyword argument) /
    instance attribute has a non-empty value. This makes it possible to pass
    API keys directly to a decorated ``__init__`` method instead of relying
    on environment variables, which is required when more than one API key is
    needed.

    Args:
        required_keys: The required API keys to be checked. Plain strings and
            ``(env_var_name, arg_name)`` tuples can be mixed, and a single
            list/tuple collecting several of them is also accepted.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables and are not passed to the decorated
            function or stored as an instance attribute.

    Example:
        ::

            @api_keys_required(
                [
                    (None, 'api_key'),
                    ('OPENAI_API_KEY', None),
                ]
            )
            def some_api_function(api_key: Optional[str] = None):
                # Function implementation...
    """

    def is_empty(value: Any) -> bool:
        r"""Whether the given value should be considered as missing."""
        if value is None:
            return True
        if isinstance(value, str):
            return not value.strip()
        return False

    # Normalize the accepted input formats, i.e. plain environment variable
    # names and/or (environment variable name, argument name) tuples.
    normalized_keys: List[Tuple[Optional[str], Optional[str]]] = []

    def add_key(item: Any) -> None:
        if isinstance(item, str):
            normalized_keys.append((item, None))
        elif isinstance(item, (tuple, list)):
            env_var_name = item[0] if len(item) > 0 else None
            arg_name = item[1] if len(item) > 1 else None
            if env_var_name is not None and not isinstance(env_var_name, str):
                raise TypeError(
                    "The environment variable name of an API key must be a "
                    f"string or None, got: {type(env_var_name)}"
                )
            if arg_name is not None and not isinstance(arg_name, str):
                raise TypeError(
                    "The argument name of an API key must be a string or "
                    f"None, got: {type(arg_name)}"
                )
            normalized_keys.append((env_var_name, arg_name))
        else:
            raise TypeError(
                "Each required API key must be a string or a tuple of "
                f"(env_var_name, arg_name), got: {type(item)}"
            )

    def is_key_spec(item: Any) -> bool:
        r"""Whether the item describes a single required API key."""
        if isinstance(item, str):
            return True
        return (
            isinstance(item, tuple)
            and len(item) == 2
            and all(e is None or isinstance(e, str) for e in item)
        )

    for key_spec in required_keys:
        if is_key_spec(key_spec):
            add_key(key_spec)
        elif isinstance(key_spec, (list, tuple)):
            # A container grouping several required API keys.
            for sub_spec in key_spec:
                add_key(sub_spec)
        else:
            add_key(key_spec)

    def decorator(func: F) -> F:
        # Resolve the parameter names of the decorated function once, so that
        # API keys passed positionally can be recognized as well.
        try:
            parameters = list(inspect.signature(func).parameters.values())
        except (TypeError, ValueError):  # pragma: no cover
            parameters = []
        positional_names = [
            parameter.name
            for parameter in parameters
            if parameter.kind
            in (
                inspect.Parameter.POSITIONAL_ONLY,
                inspect.Parameter.POSITIONAL_OR_KEYWORD,
            )
        ]

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            # Collect the values passed to the decorated function, mapping
            # positional arguments to their parameter names.
            call_arguments: Dict[str, Any] = dict(kwargs)
            for name, value in zip(positional_names, args):
                call_arguments.setdefault(name, value)

            missing_keys: List[str] = []
            for env_var_name, arg_name in normalized_keys:
                # 1. The key may be provided through an environment variable.
                if env_var_name and not is_empty(os.getenv(env_var_name)):
                    continue
                if arg_name:
                    # 2. The key may be passed directly to the decorated
                    #    function or method (e.g. a decorated `__init__`).
                    if not is_empty(call_arguments.get(arg_name)):
                        continue
                    # 3. The key may already be stored as an instance
                    #    attribute, e.g. when a method is decorated.
                    if args and not is_empty(
                        getattr(args[0], arg_name, None)
                    ):
                        continue
                # Backward compatibility: an instance may expose the key
                # through the generic `_api_key` attribute.
                elif args and not is_empty(getattr(args[0], '_api_key', None)):
                    continue
                missing_keys.append(env_var_name or arg_name or '')

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

The validator also notes no interface was checked (no issue/test name), so the file-format rule is the governing criterion here — and it fails.

### 5. Additional substantive concerns (secondary)

* The fix is **over-permissive**: with a bare string key, the attribute fallback widened from `_api_key` to `('_api_key', 'api_key')` *and* to any argument named `api_key`. Any decorated function/method that happens to receive a truthy `api_key`-named argument (or an object exposing `.api_key`) now silently bypasses the environment-variable check, which weakens the guard rather than just fixing the `__init__` case. A tighter fix would restrict argument-based satisfaction to the explicitly declared attribute name (`(env, attr)` form) only.
* The keyword-argument escape hatch (`if 'required_keys' in options`) plus the option-mutation logic is unnecessary surface area for this bug, and the set iteration order in `_flatten` makes error-message ordering non-deterministic for set inputs.
* Usage of truthiness (`os.getenv(...)`, `call_arguments.get(...)`) rather than explicit `is None` silently treats empty-string keys as missing — a behaviour change versus the original `k not in os.environ` check.

## Verdict

**REJECT**

The logic change does make the reproduction test pass (the `api_key` kwarg is now recognized, so no `ValueError`), and the `inspect` API usage is sound. However:

1. `validate_fixed_code` **fails the required output format** — the mandated `### camel/utils/commons.py` marker line is removed by the patch (and trailing newlines stripped), so the produced artifact is structurally invalid regardless of behaviour.
2. Beyond the format failure, the patch is needlessly broad: it loosens the key check for *all* keys (accepting any `api_key`-named argument or `.api_key` attribute) and adds unclear behaviour changes (truthiness vs. membership, set-order-dependent error messages), which risks silently disabling API-key validation in unrelated call sites.

The fix should preserve the original file header/format, and preferably narrow the change to the explicit `(env_var, attribute)` semantics plus `__init__`-argument handling.

def download_tasks(task: TaskType, folder_path: str) -> None:
    r"""Downloads task-related files from a specified URL and extracts them.

    This function downloads a zip file containing tasks based on the specified
    `task` type from a predefined URL, saves it to `folder_path`, and then
    extracts the contents of the zip file into the same folder. After
    extraction, the zip file is deleted.

    Args:
        task (TaskType): An enum representing the type of task to download.
        folder_path (str): The path of the folder where the zip file will be
                           downloaded and extracted.
    """
    # Define the path to save the zip file
    zip_file_path = os.path.join(folder_path, "tasks.zip")

    # Download the zip file from the Google Drive link
    response = requests.get(
        "https://huggingface.co/datasets/camel-ai/"
        f"metadata/resolve/main/{task.value}_tasks.zip"
    )

    # Save the zip file
    with open(zip_file_path, "wb") as f:
        f.write(response.content)

    with zipfile.ZipFile(zip_file_path, "r") as zip_ref:
        zip_ref.extractall(folder_path)

    # Delete the zip file
    os.remove(zip_file_path)


def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()
            if task_name.strip() and task_id.isnumeric():
                new_tasks_list.append(task_name)
    return new_tasks_list


def check_server_running(server_url: str) -> bool:
    r"""Check whether the port refered by the URL to the server
    is open.

    Args:
        server_url (str): The URL to the server running LLM inference
            service.

    Returns:
        bool: Whether the port is open for packets (server is running).
    """
    parsed_url = urlparse(server_url)
    url_tuple = (parsed_url.hostname, parsed_url.port)

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    result = sock.connect_ex(url_tuple)
    sock.close()

    # if the port is open, the result should be 0.
    return result == 0


def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required module dependencies.

    Raises:
        ImportError: If any of the required modules are not available.

    Example:
        ::

            @dependencies_required('numpy', 'pandas')
            def data_processing_function():
                # Function implementation...
    """

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            missing_modules = [
                m for m in required_modules if not is_module_available(m)
            ]
            if missing_modules:
                raise ImportError(
                    f"Missing required modules: {', '.join(missing_modules)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator


def is_module_available(module_name: str) -> bool:

    Returns:

def api_keys_required(*required_keys: Any, **options: Any) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are presented in the
    environment variables, as an instance attribute or as an argument that is
    passed directly to the decorated function (e.g. to a decorated
    ``__init__`` method).

    Args:
        required_keys: The required API keys to be checked. Every key can be
            given either as:

            * a string, i.e. the name of the environment variable that stores
              the API key, or
            * a tuple ``(environment variable name, attribute name)``, where
              the attribute name is the name of the instance attribute (or of
              the argument of the decorated function) that stores the API key.

            A list or a tuple containing any combination of the above is also
            accepted, e.g.
            ``[('OPENAI_API_KEY', 'api_key'), ('ANTHROPIC_API_KEY',
            'api_key')]``.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables, the instance attributes and the arguments
            passed directly to the decorated function.

    Example:
        ::

            @api_keys_required([('OPENAI_API_KEY', 'api_key')])
            def some_api_function(api_key=None):
                # Function implementation...
    """

    # Normalize the several accepted ways of specifying the required keys
    # into a list of ``(environment variable, attribute name)`` tuples.
    normalized_keys: List[Tuple[Optional[str], Optional[str]]] = []

    def _flatten(item: Any) -> None:
        if item is None:
            return
        if isinstance(item, str):
            # Only an environment variable name was given.
            normalized_keys.append((item, None))
        elif (
            isinstance(item, tuple)
            and len(item) == 2
            and all(
                element is None or isinstance(element, str) for element in item
            )
        ):
            env_var, attr_name = item
            if env_var is not None or attr_name is not None:
                normalized_keys.append((env_var, attr_name))
        elif isinstance(item, (list, tuple, set, frozenset)):
            for element in item:
                _flatten(element)
        else:
            raise TypeError(
                "The required API keys should be given as strings or as "
                "(environment variable, attribute) tuples, "
                f"got: {item!r}."
            )

    # Also support being called as ``api_keys_required(required_keys=...)``.
    if 'required_keys' in options:
        required_keys = required_keys + (options.pop('required_keys'),)
    if options:
        raise TypeError(
            "api_keys_required() got unexpected keyword arguments: "
            f"{', '.join(options)}."
        )

    for key in required_keys:
        _flatten(key)

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            # Resolve the arguments that are passed to the decorated
            # function, so that API keys given directly as positional or
            # keyword arguments (e.g. ``SomeClass(api_key="...")``) are
            # recognized as well.
            call_arguments: Dict[str, Any] = {}
            try:
                signature = inspect.signature(func)
            except (TypeError, ValueError):  # pragma: no cover
                signature = None

            bound: Optional[inspect.BoundArguments] = None
            if signature is not None:
                try:
                    bound = signature.bind_partial(*args, **kwargs)
                except TypeError:  # pragma: no cover
                    bound = None

            if bound is not None and signature is not None:
                for name, value in bound.arguments.items():
                    parameter = signature.parameters[name]
                    if parameter.kind is inspect.Parameter.VAR_KEYWORD:
                        call_arguments.update(value)
                    elif parameter.kind is inspect.Parameter.VAR_POSITIONAL:
                        continue
                    else:
                        call_arguments[name] = value
            else:  # pragma: no cover
                call_arguments.update(kwargs)

            instance = args[0] if args else None

            missing_keys: List[str] = []
            for env_var, attr_name in normalized_keys:
                # 1. Check the environment variables.
                if env_var is not None and os.getenv(env_var):
                    continue

                # 2. Check the instance attributes and the arguments that
                # were directly passed to the decorated function.
                attribute_names: Tuple[str, ...] = (
                    (attr_name,)
                    if attr_name is not None
                    else ('_api_key', 'api_key')
                )

                if instance is not None and any(
                    getattr(instance, name, None) for name in attribute_names
                ):
                    continue

                if any(
                    call_arguments.get(name) for name in attribute_names
                ):
                    continue

                missing_keys.append(
                    env_var if env_var is not None else attribute_names[0]
                )

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )

            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper


class AgentOpsMeta(type):
    r"""Metaclass that automatically decorates all callable attributes with
    the agentops_decorator,
    except for the 'get_tools' method.

    Methods:
    __new__(cls, name, bases, dct):
        Creates a new class with decorated methods.
    """

    def __new__(cls, name, bases, dct):
        if ToolEvent:
            for attr, value in dct.items():
                if callable(value) and attr != 'get_tools':
                    dct[attr] = agentops_decorator(value)
        return super().__new__(cls, name, bases, dct)

def handle_http_error(response: requests.Response) -> str:
    r"""Handles the HTTP errors based on the status code of the response.

    Args:
        response (requests.Response): The HTTP response from the API call.

    Returns:
        str: The error type, based on the status code.
    """
    if response.status_code == HTTPStatus.UNAUTHORIZED:
        return "Unauthorized. Check your access token."
    elif response.status_code == HTTPStatus.FORBIDDEN:
        return "Forbidden. You do not have permission to perform this action."
    elif response.status_code == HTTPStatus.NOT_FOUND:
        return "Not Found. The resource could not be located."
    elif response.status_code == HTTPStatus.TOO_MANY_REQUESTS:
        return "Too Many Requests. You have hit the rate limit."
    else:
        return "HTTP Error"


def retry_request(
    func: Callable, retries: int = 3, delay: int = 1, *args: Any, **kwargs: Any
) -> Any:
    r"""Retries a function in case of any errors.

    Args:
        func (Callable): The function to be retried.
        retries (int): Number of retry attempts. (default: :obj:`3`)
        delay (int): Delay between retries in seconds. (default: :obj:`1`)
        *args: Arguments to pass to the function.
        **kwargs: Keyword arguments to pass to the function.

    Returns:
        Any: The result of the function call if successful.

    Raises:
        Exception: If all retry attempts fail.
    """
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            print(f"Attempt {attempt + 1}/{retries} failed: {e}")
            if attempt < retries - 1:
                time.sleep(delay)
            else:
                raise
