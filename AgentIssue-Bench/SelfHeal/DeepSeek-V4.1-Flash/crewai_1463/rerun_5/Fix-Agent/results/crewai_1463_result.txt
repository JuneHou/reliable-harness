class FlowMeta(type):
    def __new__(mcs, name, bases, dct):
        cls = super().__new__(mcs, name, bases, dct)

        # Inherit orchestration metadata from any base flow class so that
        # listeners / routers / start methods declared on a parent flow keep
        # working in subclasses instead of silently disappearing.
        inherited_start_methods: List[str] = []
        inherited_listeners: Dict[str, tuple] = {}
        inherited_routers: Dict[str, str] = {}
        inherited_router_paths: Dict[str, List[str]] = {}

        for base in reversed(bases):
            for start_method in getattr(base, "_start_methods", []) or []:
                if start_method not in inherited_start_methods:
                    inherited_start_methods.append(start_method)
            inherited_listeners.update(getattr(base, "_listeners", {}) or {})
            inherited_routers.update(getattr(base, "_routers", {}) or {})
            inherited_router_paths.update(getattr(base, "_router_paths", {}) or {})

        start_methods: List[str] = list(inherited_start_methods)
        listeners: Dict[str, tuple] = dict(inherited_listeners)
        routers: Dict[str, str] = dict(inherited_routers)
        router_paths: Dict[str, List[str]] = dict(inherited_router_paths)

        for attr_name, attr_value in dct.items():
            if hasattr(attr_value, "__is_start_method__"):
                if attr_name not in start_methods:
                    start_methods.append(attr_name)

            if hasattr(attr_value, "__trigger_methods__"):
                # Copy the trigger list so shared decorator results (e.g. from
                # `and_(...)` / `or_(...)`) are never mutated across classes.
                methods = list(attr_value.__trigger_methods__)
                condition_type = getattr(attr_value, "__condition_type__", "OR")
                listeners[attr_name] = (condition_type, methods)
            elif hasattr(attr_value, "__is_router__"):
                trigger_method_name = attr_value.__router_for__
                routers[trigger_method_name] = attr_name
                possible_returns = get_possible_return_constants(attr_value)
                if possible_returns:
                    router_paths[attr_name] = possible_returns

                # Register the router as a listener of its triggering method so
                # the flow graph is complete. Routers are dispatched explicitly
                # by `_execute_listeners`, so they are never executed twice.
                listeners[attr_name] = ("OR", [trigger_method_name])

        setattr(cls, "_start_methods", start_methods)
        setattr(cls, "_listeners", listeners)
        setattr(cls, "_routers", routers)
        setattr(cls, "_router_paths", router_paths)

        OR listeners fire as soon as any of their trigger methods executes.
        AND listeners only fire once *all* of their trigger methods executed,
        which is verified against `_executed_methods` on every completion (not

            router_method = self._methods[router_name]

                path = await self._execute_method(router_name, router_method, result)
            else:
                path = await self._execute_method(router_name, router_method)
            if not isinstance(path, str):
                return
            # Use the path as the new trigger method
            trigger_method = path

        for listener_name, (condition_type, methods) in self._listeners.items():
            # Routers are already dispatched above; never run them twice.
            if router_name is not None and listener_name == router_name:
                continue

            if not self._is_available(listener_name):
                continue

            if condition_type == "OR":

                else:

                    listener_tasks.append(
                        self._schedule_listener(listener_name, result)
                    )

        # Run all listener tasks concurrently and wait for them to complete
        if listener_tasks:
            await asyncio.gather(*listener_tasks)

    async def _execute_single_listener(self, listener_name: str, result: Any) -> None:

            if self._accepts_arguments(method):

            else:
                # If listener does not expect parameters, call without arguments

            else method(*args, **kwargs)
        )
        self._method_outputs.append(result)  # Store the output

        self._executed_methods.add(method_name)

        return result

        listener_tasks = []

        # may be the trigger of a router, in which case the router is executed
        # here and the value it returns (the "path") becomes an additional
        # trigger. The original trigger is kept so that regular listeners of
        # the triggering method keep working even when it also has a router.
        triggers: Set[str] = {trigger_method}

        if trigger_method in self._routers:
            router_method_name = self._routers[trigger_method]
            router_method = self._methods.get(router_method_name)
            if router_method is not None:
                path = await self._execute_method(
                    router_method_name, router_method
                )
                if path is not None:
                    triggers.add(path if isinstance(path, str) else str(path))
                # The router itself is a valid trigger for its own listeners
                # and it has now been executed.
                triggers.add(router_method_name)
                self._scheduled_tasks.discard(router_method_name)

        for listener_name, (condition_type, methods) in list(
            self._listeners.items()
        ):
            # Routers are executed alongside their trigger method above, they
            # must not be executed again through the listener mechanism.
            if listener_name in self._routers.values():
                continue

            # A listener is executed at most once
            if (
                listener_name in self._executed_methods
                or listener_name in self._scheduled_tasks
            ):
                continue

            if condition_type == "OR":
                if any(trigger in methods for trigger in triggers):
                    self._scheduled_tasks.add(listener_name)
                    listener_tasks.append(
                        self._execute_single_listener(listener_name, result)
                    )
            elif condition_type == "AND":
                # Track, per listener, which of its triggering conditions are
                # still missing. Keeping the state per listener makes the AND
                # condition independent of the order in which the triggering
                # methods complete (and of both `and_(func)` / `and_("name")`).
                pending = self._pending_and_listeners.get(
                    listener_name, set(methods)
                )
                pending -= triggers
                pending -= self._executed_methods
                self._pending_and_listeners[listener_name] = pending

                if not pending:
                    self._scheduled_tasks.add(listener_name)
                    listener_tasks.append(
                        self._execute_single_listener(listener_name, result)
                    )

        # Run all listener tasks concurrently and wait for them to complete
        await asyncio.gather(*listener_tasks)

    async def _execute_single_listener(self, listener_name: str, result: Any) -> None:
        try:
            method = self._methods[listener_name]
            sig = inspect.signature(method)
            params = list(sig.parameters.values())

            # Exclude 'self' parameter
            method_params = [p for p in params if p.name != "self"]

            if method_params:
                # If listener expects parameters, pass the result
                listener_result = await self._execute_method(
                    listener_name, method, result
                )
            else:
                # If listener does not expect parameters, call without arguments
                listener_result = await self._execute_method(listener_name, method)

            # The listener already ran, its pending conditions are irrelevant
            self._pending_and_listeners.pop(listener_name, None)

            # Execute listeners of this listener
            await self._execute_listeners(listener_name, listener_result)
        except Exception as e:
            print(
                f"[Flow._execute_single_listener] Error in method {listener_name}: {e}"
            )
            import traceback

            traceback.print_exc()
        finally:
            # Remove from scheduled tasks after execution (even on errors), so
            # a failing listener doesn't block the rest of the flow forever.
            self._scheduled_tasks.discard(listener_name)

class FlowMeta(type):
    def __new__(mcs, name, bases, dct):
        cls = super().__new__(mcs, name, bases, dct)

        # Inherit orchestration metadata from any base flow class so that
        # listeners / routers / start methods declared on a parent flow keep
        # working in subclasses instead of silently disappearing.
        inherited_start_methods: List[str] = []
        inherited_listeners: Dict[str, tuple] = {}
        inherited_routers: Dict[str, str] = {}
        inherited_router_paths: Dict[str, List[str]] = {}

        for base in reversed(bases):
            for start_method in getattr(base, "_start_methods", []) or []:
                if start_method not in inherited_start_methods:
                    inherited_start_methods.append(start_method)
            inherited_listeners.update(getattr(base, "_listeners", {}) or {})
            inherited_routers.update(getattr(base, "_routers", {}) or {})
            inherited_router_paths.update(getattr(base, "_router_paths", {}) or {})

        start_methods: List[str] = list(inherited_start_methods)
        listeners: Dict[str, tuple] = dict(inherited_listeners)
        routers: Dict[str, str] = dict(inherited_routers)
        router_paths: Dict[str, List[str]] = dict(inherited_router_paths)

        for attr_name, attr_value in dct.items():
            if hasattr(attr_value, "__is_start_method__"):
                if attr_name not in start_methods:
                    start_methods.append(attr_name)

            if hasattr(attr_value, "__trigger_methods__"):
                # Copy the trigger list so shared decorator results (e.g. from
                # `and_(...)` / `or_(...)`) are never mutated across classes.
                methods = list(attr_value.__trigger_methods__)
                condition_type = getattr(attr_value, "__condition_type__", "OR")
                listeners[attr_name] = (condition_type, methods)
            elif hasattr(attr_value, "__is_router__"):
                trigger_method_name = attr_value.__router_for__
                routers[trigger_method_name] = attr_name
                possible_returns = get_possible_return_constants(attr_value)
                if possible_returns:
                    router_paths[attr_name] = possible_returns

                # Register the router as a listener of its triggering method so
                # the flow graph is complete. Routers are dispatched explicitly
                # by `_execute_listeners`, so they are never executed twice.
                listeners[attr_name] = ("OR", [trigger_method_name])

        setattr(cls, "_start_methods", start_methods)
        setattr(cls, "_listeners", listeners)
        setattr(cls, "_routers", routers)
        setattr(cls, "_router_paths", router_paths)

        return cls
