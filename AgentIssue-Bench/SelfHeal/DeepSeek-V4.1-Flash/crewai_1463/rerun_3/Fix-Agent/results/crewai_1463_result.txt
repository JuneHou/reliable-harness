def _trigger_name(condition: Any) -> str:
    """Return the name used to reference a trigger method.

    Both function references and plain strings are supported, which allows
    listeners to be declared as ``and_(step_1, "step_2")``.
    """
    if isinstance(condition, str):
        return condition

    name = getattr(condition, "__name__", None)
    if isinstance(name, str) and name:
        return name

    raise ValueError(
        f"Invalid condition: {condition!r}. "
        "A condition must be a method, a string or a result of or_() or and_()."
    )


def start(condition=None):
    def decorator(func):
        func.__is_start_method__ = True
        if condition is not None:
            if isinstance(condition, str):
                func.__trigger_methods__ = [condition]
                func.__condition_type__ = "OR"
            elif (
                isinstance(condition, dict)
                and "type" in condition
                and "methods" in condition
            ):
                func.__trigger_methods__ = list(condition["methods"])
                func.__condition_type__ = condition["type"]
            elif callable(condition):
                func.__trigger_methods__ = [_trigger_name(condition)]
                func.__condition_type__ = "OR"
            else:
                raise ValueError(
                    "Condition must be a method, string, or a result of or_() or and_()"
                )
        return func

    return decorator

def router(method):
    trigger = method if isinstance(method, str) else getattr(method, "__name__", None)
    if not trigger:
        raise ValueError("A router must be defined for a method or a method name")

    def decorator(func):
        func.__is_router__ = True
        func.__router_for__ = trigger
        return func

    return decorator


def _merge_conditions(condition_type: str, conditions: Tuple[Any, ...]) -> Dict[str, Any]:
    """Normalize a list of conditions into a single condition dictionary.

    Nested conditions using the same operator are flattened (their meaning does
    not change). Nested conditions using a different operator are preserved so
    that combinations such as ``and_(a, or_(b, c))`` keep their semantics
    instead of silently becoming ``AND(a, b, c)``, which could never be
    satisfied when only one of the branches runs.
    """
    methods: List[Any] = []

    for condition in conditions:
        if isinstance(condition, dict) and "methods" in condition:
            nested_type = str(condition.get("type", "OR")).upper()
            nested_methods = list(condition["methods"])
            if nested_type == condition_type:
                methods.extend(nested_methods)
            else:
                methods.append({"type": nested_type, "methods": nested_methods})
        elif isinstance(condition, str):
            methods.append(condition)
        elif callable(condition):
            methods.append(_trigger_name(condition))
        else:
            raise ValueError(f"Invalid condition in {condition_type.lower()}()")

    return {"type": condition_type, "methods": methods}


def or_(*conditions):
    return _merge_conditions("OR", conditions)


def and_(*conditions):
    return _merge_conditions("AND", conditions)


class FlowMeta(type):
    def __new__(mcs, name, bases, dct):
        cls = super().__new__(mcs, name, bases, dct)

        start_methods: List[str] = []
        listeners: Dict[str, Tuple[str, List[Any]]] = {}
        routers: Dict[str, str] = {}
        router_paths: Dict[str, List[str]] = {}

        for attr_name, attr_value in dct.items():
            if not callable(attr_value):
                continue

            if hasattr(attr_value, "__is_start_method__"):
                start_methods.append(attr_name)

            if hasattr(attr_value, "__trigger_methods__"):
                listeners[attr_name] = (
                    getattr(attr_value, "__condition_type__", "OR"),
                    list(attr_value.__trigger_methods__),
                )
            elif hasattr(attr_value, "__is_router__"):
                trigger_method_name = getattr(attr_value, "__router_for__", None)
                if trigger_method_name:
                    # The router is executed as soon as its trigger method
                    # finishes, it must not be registered as a regular listener
                    # otherwise it would be executed twice.
                    routers[trigger_method_name] = attr_name

                possible_returns = get_possible_return_constants(attr_value)
                if possible_returns:
                    router_paths[attr_name] = possible_returns

        setattr(cls, "_start_methods", start_methods)
        setattr(cls, "_listeners", listeners)
        setattr(cls, "_routers", routers)
        setattr(cls, "_router_paths", router_paths)

        return cls


class Flow(Generic[T], metaclass=FlowMeta):

    _listeners: Dict[str, Tuple[str, List[Any]]] = {}

    def __class_getitem__(cls: Type["Flow"], item: Type[T]) -> Type["Flow"]:
        class _FlowGeneric(cls):  # type: ignore
            _initial_state_T = item  # type: ignore

        _FlowGeneric.__name__ = f"{cls.__name__}[{item.__name__}]"
        return _FlowGeneric

    def __init__(self) -> None:

        self._state: T = self._create_initial_state()
        self._executed_methods: Set[str] = set()
        self._scheduled_tasks: Set[str] = set()
        self._pending_and_listeners: Dict[str, Set[str]] = {}
        self._method_outputs: List[Any] = []  # List to store all method outputs

        self._telemetry.flow_creation_span(self.__class__.__name__)

            if method_name.startswith("__"):
                continue
            method = getattr(self, method_name)
            if callable(method):

    def _create_initial_state(self) -> T:
        if self.initial_state is None and hasattr(self, "_initial_state_T"):
            return self._initial_state_T()  # type: ignore
        if self.initial_state is None:
            return {}  # type: ignore
        elif isinstance(self.initial_state, type):
            return self.initial_state()
        else:
            return self.initial_state

    @property
    def state(self) -> T:
        return self._state

    @property

        """Returns the list of all outputs from executed methods."""
        return self._method_outputs

    def kickoff(self) -> Any:
        return asyncio.run(self.kickoff_async())

    async def kickoff_async(self) -> Any:
        if not self._start_methods:
            raise ValueError("No start method defined")

        self._telemetry.flow_execution_span(
            self.__class__.__name__, list(self._methods.keys())
        )

        # Create tasks for all start methods
        tasks = [
            self._execute_start_method(start_method)
            for start_method in self._start_methods
        ]

        # Run all start methods concurrently
        await asyncio.gather(*tasks)

        # Return the final output (from the last executed method)
        if self._method_outputs:
            return self._method_outputs[-1]
        else:
            return None  # Or raise an exception if no methods were executed

    async def _execute_start_method(self, start_method_name: str) -> None:
        result = await self._execute_method(
            start_method_name, self._methods[start_method_name]
        )
        await self._execute_listeners(start_method_name, result)

    async def _execute_method(
        self, method_name: str, method: Callable, *args: Any, **kwargs: Any
    ) -> Any:
        result = (
            await method(*args, **kwargs)
            if asyncio.iscoroutinefunction(method)
            else method(*args, **kwargs)
        )
        self._method_outputs.append(result)  # Store the output

        self._executed_methods.add(method_name)

        return result

    def _is_trigger_relevant(self, condition: Any, trigger_method: str) -> bool:
        """Whether ``trigger_method`` takes part in the given condition."""
        if isinstance(condition, str):
            return condition == trigger_method
        if isinstance(condition, dict):
            return any(
                self._is_trigger_relevant(method, trigger_method)
                for method in condition.get("methods", []) or []
            )
        return False

    def _evaluate_condition(self, condition: Any, triggered: Set[str]) -> bool:
        """Evaluate a (possibly nested) condition against the triggered methods."""
        if isinstance(condition, str):
            return condition in triggered
        if isinstance(condition, dict):
            condition_type = str(condition.get("type", "OR")).upper()
            methods = condition.get("methods", []) or []
            if condition_type == "AND":
                return all(self._evaluate_condition(m, triggered) for m in methods)
            return any(self._evaluate_condition(m, triggered) for m in methods)

        raise ValueError(f"Invalid condition: {condition!r}")

    async def _execute_router(
        self, router_name: str, router_method: Callable, result: Any
    ) -> List[str]:
        """Execute a router and return the path(s) it selected."""
        signature = inspect.signature(router_method)
        accepts_arguments = any(
            param.name != "self" for param in signature.parameters.values()
        )

        path_result = (
            await self._execute_method(router_name, router_method, result)
            if accepts_arguments
            else await self._execute_method(router_name, router_method)
        )

        if path_result is None:
            return []

        paths = (
            path_result
            if isinstance(path_result, (list, tuple, set))
            else [path_result]
        )

        return [str(path) for path in paths if path is not None]

    async def _execute_listeners(self, trigger_method: str, result: Any) -> None:
        """Executes all listeners and routers triggered by a method."""
        listener_tasks: List[Any] = []

        # Every trigger that may fire listeners: the method itself, the router
        # attached to it and the path(s) returned by that router.
        triggers: List[str] = [trigger_method]

        if trigger_method in self._routers:
            router_name = self._routers[trigger_method]
            if router_name not in self._executed_methods:
                router_method = self._methods[router_name]
                paths = await self._execute_router(router_name, router_method, result)

                triggers.append(router_name)
                for path in paths:
                    if path not in triggers:
                        triggers.append(path)

        for trigger in triggers:
            for listener_name, (condition_type, methods) in self._listeners.items():
                if (
                    listener_name in self._executed_methods
                    or listener_name in self._scheduled_tasks
                ):
                    continue

                condition: Dict[str, Any] = {
                    "type": condition_type,
                    "methods": methods,
                }

                if condition_type == "AND":
                    # Remember every trigger seen so far: a router path, for
                    # instance, is never part of the executed methods, so the
                    # pending triggers are needed to satisfy the condition.
                    pending = self._pending_and_listeners.setdefault(
                        listener_name, set()
                    )
                    pending.add(trigger)

                    triggered = pending | self._executed_methods
                    if not self._evaluate_condition(condition, triggered):
                        continue
                else:
                    # "OR" listeners are only triggered by the method they
                    # listen to.
                    if not self._is_trigger_relevant(condition, trigger):
                        continue

                    triggered = self._pending_and_listeners.get(
                        listener_name, set()
                    ) | self._executed_methods
                    if not self._evaluate_condition(condition, triggered):
                        continue

                self._scheduled_tasks.add(listener_name)
                listener_tasks.append(
                    self._execute_single_listener(listener_name, result)
                )

        # Run all listener tasks concurrently and wait for them to complete
        await asyncio.gather(*listener_tasks)

    async def _execute_single_listener(self, listener_name: str, result: Any) -> None:
        try:
            method = self._methods[listener_name]
            sig = inspect.signature(method)
            params = list(sig.parameters.values())

            # Exclude 'self' parameter
            method_params = [p for p in params if p.name != "self"]

            if method_params:
                # If listener expects parameters, pass the result
                listener_result = await self._execute_method(
                    listener_name, method, result
                )
            else:
                # If listener does not expect parameters, call without arguments
                listener_result = await self._execute_method(listener_name, method)

            # Execute listeners of this listener
            await self._execute_listeners(listener_name, listener_result)
        except Exception as e:
            print(
                f"[Flow._execute_single_listener] Error in method {listener_name}: {e}"
            )
            import traceback

            traceback.print_exc()
        finally:
            # Remove from scheduled tasks after execution, even when it failed,
            # otherwise the listener would be blocked forever.
            self._scheduled_tasks.discard(listener_name)
            self._pending_and_listeners.pop(listener_name, None)

        routers: Dict[str, str] = {}
        router_paths: Dict[str, List[str]] = {}

        for attr_name, attr_value in dct.items():
            if not callable(attr_value):
                continue

            if hasattr(attr_value, "__is_start_method__"):
                start_methods.append(attr_name)

            if hasattr(attr_value, "__trigger_methods__"):
                listeners[attr_name] = (
                    getattr(attr_value, "__condition_type__", "OR"),
                    list(attr_value.__trigger_methods__),
                )
            elif hasattr(attr_value, "__is_router__"):
                trigger_method_name = getattr(attr_value, "__router_for__", None)
                if trigger_method_name:
                    # The router is executed as soon as its trigger method
                    # finishes, it must not be registered as a regular listener
                    # otherwise it would be executed twice.
                    routers[trigger_method_name] = attr_name

                possible_returns = get_possible_return_constants(attr_value)
                if possible_returns:
                    router_paths[attr_name] = possible_returns

        setattr(cls, "_start_methods", start_methods)
        setattr(cls, "_listeners", listeners)
        setattr(cls, "_routers", routers)
        setattr(cls, "_router_paths", router_paths)

        return cls


class Flow(Generic[T], metaclass=FlowMeta):
    _telemetry = Telemetry()

    _start_methods: List[str] = []
    _listeners: Dict[str, Tuple[str, List[Any]]] = {}
    _routers: Dict[str, str] = {}
    _router_paths: Dict[str, List[str]] = {}
    initial_state: Union[Type[T], T, None] = None

    def __class_getitem__(cls: Type["Flow"], item: Type[T]) -> Type["Flow"]:
        class _FlowGeneric(cls):  # type: ignore
            _initial_state_T = item  # type: ignore

        _FlowGeneric.__name__ = f"{cls.__name__}[{item.__name__}]"
        return _FlowGeneric

    def __init__(self) -> None:
        self._methods: Dict[str, Callable] = {}
        self._state: T = self._create_initial_state()
        self._executed_methods: Set[str] = set()
        self._scheduled_tasks: Set[str] = set()
        self._pending_and_listeners: Dict[str, Set[str]] = {}
        self._method_outputs: List[Any] = []  # List to store all method outputs

        self._telemetry.flow_creation_span(self.__class__.__name__)

        for method_name in dir(self):
            if method_name.startswith("__"):
                continue
            method = getattr(self, method_name)
            if callable(method):
                self._methods[method_name] = method

    def _create_initial_state(self) -> T:
        if self.initial_state is None and hasattr(self, "_initial_state_T"):
            return self._initial_state_T()  # type: ignore
        if self.initial_state is None:
            return {}  # type: ignore
        elif isinstance(self.initial_state, type):
            return self.initial_state()
        else:
            return self.initial_state

    @property
    def state(self) -> T:
        return self._state

    @property
    def method_outputs(self) -> List[Any]:
        """Returns the list of all outputs from executed methods."""
        return self._method_outputs

    def kickoff(self) -> Any:
        return asyncio.run(self.kickoff_async())

    async def kickoff_async(self) -> Any:
        if not self._start_methods:
            raise ValueError("No start method defined")

        self._telemetry.flow_execution_span(
            self.__class__.__name__, list(self._methods.keys())
        )

        # Create tasks for all start methods
        tasks = [
            self._execute_start_method(start_method)
            for start_method in self._start_methods
        ]

        # Run all start methods concurrently
        await asyncio.gather(*tasks)

        # Return the final output (from the last executed method)
        if self._method_outputs:
            return self._method_outputs[-1]
        else:
            return None  # Or raise an exception if no methods were executed

    async def _execute_start_method(self, start_method_name: str) -> None:
        result = await self._execute_method(
            start_method_name, self._methods[start_method_name]
        )
        await self._execute_listeners(start_method_name, result)

    async def _execute_method(
        self, method_name: str, method: Callable, *args: Any, **kwargs: Any
    ) -> Any:
        result = (
            await method(*args, **kwargs)
            if asyncio.iscoroutinefunction(method)
            else method(*args, **kwargs)
        )
        self._method_outputs.append(result)  # Store the output

        self._executed_methods.add(method_name)

        return result

    def _is_trigger_relevant(self, condition: Any, trigger_method: str) -> bool:
        """Whether ``trigger_method`` takes part in the given condition."""
        if isinstance(condition, str):
            return condition == trigger_method
        if isinstance(condition, dict):
            return any(
                self._is_trigger_relevant(method, trigger_method)
                for method in condition.get("methods", []) or []
            )
        return False

    def _evaluate_condition(self, condition: Any, triggered: Set[str]) -> bool:
        """Evaluate a (possibly nested) condition against the triggered methods."""
        if isinstance(condition, str):
            return condition in triggered
        if isinstance(condition, dict):
            condition_type = str(condition.get("type", "OR")).upper()
            methods = condition.get("methods", []) or []
            if condition_type == "AND":
                return all(self._evaluate_condition(m, triggered) for m in methods)
            return any(self._evaluate_condition(m, triggered) for m in methods)

        raise ValueError(f"Invalid condition: {condition!r}")

    async def _execute_router(
        self, router_name: str, router_method: Callable, result: Any
    ) -> List[str]:
        """Execute a router and return the path(s) it selected."""
        signature = inspect.signature(router_method)
        accepts_arguments = any(
            param.name != "self" for param in signature.parameters.values()
        )

        path_result = (
            await self._execute_method(router_name, router_method, result)
            if accepts_arguments
            else await self._execute_method(router_name, router_method)
        )

        if path_result is None:
            return []

        paths = (
            path_result
            if isinstance(path_result, (list, tuple, set))
            else [path_result]
        )

        return [str(path) for path in paths if path is not None]

    async def _execute_listeners(self, trigger_method: str, result: Any) -> None:
        """Executes all listeners and routers triggered by a method."""
        listener_tasks: List[Any] = []

        # Every trigger that may fire listeners: the method itself, the router
        # attached to it and the path(s) returned by that router.
        triggers: List[str] = [trigger_method]

        if trigger_method in self._routers:
            router_name = self._routers[trigger_method]
            if router_name not in self._executed_methods:
                router_method = self._methods[router_name]
                paths = await self._execute_router(router_name, router_method, result)

                triggers.append(router_name)
                for path in paths:
                    if path not in triggers:
                        triggers.append(path)

        for trigger in triggers:
            for listener_name, (condition_type, methods) in self._listeners.items():
                if (
                    listener_name in self._executed_methods
                    or listener_name in self._scheduled_tasks
                ):
                    continue

                condition: Dict[str, Any] = {
                    "type": condition_type,
                    "methods": methods,
                }

                if condition_type == "AND":
                    # Remember every trigger seen so far: a router path, for
                    # instance, is never part of the executed methods, so the
                    # pending triggers are needed to satisfy the condition.
                    pending = self._pending_and_listeners.setdefault(
                        listener_name, set()
                    )
                    pending.add(trigger)

                    triggered = pending | self._executed_methods
                    if not self._evaluate_condition(condition, triggered):
                        continue
                else:
                    # "OR" listeners are only triggered by the method they
                    # listen to.
                    if not self._is_trigger_relevant(condition, trigger):
                        continue

                    triggered = self._pending_and_listeners.get(
                        listener_name, set()
                    ) | self._executed_methods
                    if not self._evaluate_condition(condition, triggered):
                        continue

                self._scheduled_tasks.add(listener_name)
                listener_tasks.append(
                    self._execute_single_listener(listener_name, result)
                )

        # Run all listener tasks concurrently and wait for them to complete
        await asyncio.gather(*listener_tasks)

    async def _execute_single_listener(self, listener_name: str, result: Any) -> None:
        try:
            method = self._methods[listener_name]
            sig = inspect.signature(method)
            params = list(sig.parameters.values())

            # Exclude 'self' parameter
            method_params = [p for p in params if p.name != "self"]

            if method_params:
                # If listener expects parameters, pass the result
                listener_result = await self._execute_method(
                    listener_name, method, result
                )
            else:
                # If listener does not expect parameters, call without arguments
                listener_result = await self._execute_method(listener_name, method)

            # Execute listeners of this listener
            await self._execute_listeners(listener_name, listener_result)
        except Exception as e:
            print(
                f"[Flow._execute_single_listener] Error in method {listener_name}: {e}"
            )
            import traceback

            traceback.print_exc()
        finally:
            # Remove from scheduled tasks after execution, even when it failed,
            # otherwise the listener would be blocked forever.
            self._scheduled_tasks.discard(listener_name)
            self._pending_and_listeners.pop(listener_name, None)

    def plot(self, filename: str = "crewai_flow") -> None:
        self._telemetry.flow_plotting_span(
            self.__class__.__name__, list(self._methods.keys())
        )

        plot_flow(self, filename)
