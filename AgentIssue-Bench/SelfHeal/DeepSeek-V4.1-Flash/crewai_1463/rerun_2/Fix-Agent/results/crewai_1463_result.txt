def _condition_method_name(condition: Any) -> str:
    """Return the flow method name referenced by a single condition.

    Conditions can be provided as a string with the method name, as a
    (bound) callable / method reference or as a result of ``or_()`` / ``and_()``.
    """
    if isinstance(condition, str):
        return condition

    if isinstance(condition, dict) and "methods" in condition:
        # Nested conditions are flattened by _condition_methods.
        raise ValueError(_INVALID_CONDITION_MESSAGE)

    name = getattr(condition, "__name__", None)
    if isinstance(name, str) and name:
        return name

    # Bound methods (e.g. self.step) may only expose the underlying function.
    func = getattr(condition, "__func__", None)
    if func is not None:
        name = getattr(func, "__name__", None)
        if isinstance(name, str) and name:
            return name

    qualname = getattr(condition, "__qualname__", None)
    if isinstance(qualname, str) and qualname and "<" not in qualname:
        return qualname.split(".")[-1]

    raise ValueError(_INVALID_CONDITION_MESSAGE)


def _condition_methods(condition: Any) -> List[str]:
    """Flatten a condition (possibly built through or_()/and_()) into names."""
    if isinstance(condition, dict) and "methods" in condition:
        methods: List[str] = []
        for nested in condition["methods"]:
            if isinstance(nested, dict) and "methods" in nested:
                methods.extend(_condition_methods(nested))
            else:
                methods.append(_condition_method_name(nested))
        return methods
    return [_condition_method_name(condition)]


def _parse_condition(condition: Any) -> tuple[str, List[str]]:
    """Normalize any supported condition into (condition_type, method names)."""
    if isinstance(condition, dict) and "type" in condition and "methods" in condition:
        condition_type = str(condition["type"]).upper()
        if condition_type not in ("OR", "AND"):
            raise ValueError(_INVALID_CONDITION_MESSAGE)
        methods = _condition_methods(condition)
        if not methods:
            raise ValueError(_INVALID_CONDITION_MESSAGE)
        return condition_type, methods

    return "OR", _condition_methods(condition)

            return all(
                method in pending or method in self._executed_methods
                for method in methods

                    for listener_name in ready

                triggered_names.append(router_name)
            if isinstance(path, str) and path not in triggered_names:
                triggered_names.append(path)

            trigger_method = path

                        pending.add(triggered_name)

                # condition has completed, no matter which one finishes last
                # or in which order they were executed.
                if self._is_condition_met(listener_name, condition_type, methods):

                    listener_tasks.append(
                        self._execute_single_listener(listener_name, result)
                    )
            else:

                    triggered_name in methods for triggered_name in triggered_names

                    listener_tasks.append(
                        self._execute_single_listener(listener_name, result)
                    )

        # Run all listener tasks concurrently and wait for them to complete
        await asyncio.gather(*listener_tasks)

    async def _execute_single_listener(self, listener_name: str, result: Any) -> None:

import asyncio
import inspect
from typing import Any, Callable, Dict, Generic, List, Set, Type, TypeVar, Union

from pydantic import BaseModel

T = TypeVar("T", bound=Union[BaseModel, Dict[str, Any]])

_INVALID_CONDITION_MESSAGE = (
    "Condition must be a method, string, or a result of or_() or and_()"
)


def _condition_method_name(condition: Any) -> str:
    """Return the flow method name referenced by a single condition.

    Conditions can be provided as a string with the method name, as a
    (bound) callable / method reference or as a result of ``or_()`` / ``and_()``.
    """
    if isinstance(condition, str):
        return condition

    if isinstance(condition, dict) and "methods" in condition:
        # Nested conditions are flattened by _condition_methods.
        raise ValueError(_INVALID_CONDITION_MESSAGE)

    name = getattr(condition, "__name__", None)
    if isinstance(name, str) and name:
        return name

    # Bound methods (e.g. self.step) may only expose the underlying function.
    func = getattr(condition, "__func__", None)
    if func is not None:
        name = getattr(func, "__name__", None)
        if isinstance(name, str) and name:
            return name

    qualname = getattr(condition, "__qualname__", None)
    if isinstance(qualname, str) and qualname and "<" not in qualname:
        return qualname.split(".")[-1]

    raise ValueError(_INVALID_CONDITION_MESSAGE)


def _condition_methods(condition: Any) -> List[str]:
    """Flatten a condition (possibly built through or_()/and_()) into names."""
    if isinstance(condition, dict) and "methods" in condition:
        methods: List[str] = []
        for nested in condition["methods"]:
            if isinstance(nested, dict) and "methods" in nested:
                methods.extend(_condition_methods(nested))
            else:
                methods.append(_condition_method_name(nested))
        return methods
    return [_condition_method_name(condition)]


def _parse_condition(condition: Any) -> tuple[str, List[str]]:
    """Normalize any supported condition into (condition_type, method names)."""
    if isinstance(condition, dict) and "type" in condition and "methods" in condition:
        condition_type = str(condition["type"]).upper()
        if condition_type not in ("OR", "AND"):
            raise ValueError(_INVALID_CONDITION_MESSAGE)
        methods = _condition_methods(condition)
        if not methods:
            raise ValueError(_INVALID_CONDITION_MESSAGE)
        return condition_type, methods

    return "OR", _condition_methods(condition)


def start(condition=None):
    def decorator(func):
        func.__is_start_method__ = True
        if condition is not None:
            condition_type, methods = _parse_condition(condition)
            func.__trigger_methods__ = methods
            func.__condition_type__ = condition_type
        return func

    return decorator


def listen(condition):
    def decorator(func):
        condition_type, methods = _parse_condition(condition)
        func.__trigger_methods__ = methods
        func.__condition_type__ = condition_type
        return func

    return decorator


def router(method):
    def decorator(func):
        func.__is_router__ = True
        func.__router_for__ = _condition_method_name(method)
        return func

    return decorator


def or_(*conditions):
    methods: List[str] = []
    for condition in conditions:
        try:
            methods.extend(_condition_methods(condition))
        except ValueError:
            raise ValueError("Invalid condition in or_()")
    if not methods:
        raise ValueError("Invalid condition in or_()")
    return {"type": "OR", "methods": methods}


def and_(*conditions):
    methods: List[str] = []
    for condition in conditions:
        try:
            methods.extend(_condition_methods(condition))
        except ValueError:
            raise ValueError("Invalid condition in and_()")
    if not methods:
        raise ValueError("Invalid condition in and_()")
    return {"type": "AND", "methods": methods}


class FlowMeta(type):
    def __new__(mcs, name, bases, dct):
        cls = super().__new__(mcs, name, bases, dct)

        start_methods: List[str] = []
        listeners: Dict[str, tuple[str, List[str]]] = {}
        routers: Dict[str, str] = {}
        router_paths: Dict[str, List[str]] = {}

        for attr_name, attr_value in dct.items():
            if hasattr(attr_value, "__is_start_method__"):
                start_methods.append(attr_name)

            if hasattr(attr_value, "__trigger_methods__"):
                methods = list(attr_value.__trigger_methods__)
                condition_type = getattr(attr_value, "__condition_type__", "OR")
                condition_type = str(condition_type).upper()
                if condition_type not in ("OR", "AND"):
                    condition_type = "OR"
                listeners[attr_name] = (condition_type, methods)

            if hasattr(attr_value, "__is_router__"):
                trigger_method_name = attr_value.__router_for__
                routers[trigger_method_name] = attr_name
                possible_returns = get_possible_return_constants(attr_value)
                if possible_returns:
                    router_paths[attr_name] = possible_returns

                # Register the router as a listener of the method it routes on
                # (unless it explicitly declares its own trigger condition).
                if attr_name not in listeners:
                    listeners[attr_name] = ("OR", [trigger_method_name])

        setattr(cls, "_start_methods", start_methods)
        setattr(cls, "_listeners", listeners)
        setattr(cls, "_routers", routers)
        setattr(cls, "_router_paths", router_paths)

        return cls


class Flow(Generic[T], metaclass=FlowMeta):
    _telemetry = Telemetry()

    _start_methods: List[str] = []
    _listeners: Dict[str, tuple[str, List[str]]] = {}
    _routers: Dict[str, str] = {}
    _router_paths: Dict[str, List[str]] = {}
    initial_state: Union[Type[T], T, None] = None

    def __class_getitem__(cls: Type["Flow"], item: Type[T]) -> Type["Flow"]:
        class _FlowGeneric(cls):  # type: ignore
            _initial_state_T = item  # type: ignore

        _FlowGeneric.__name__ = f"{cls.__name__}[{item.__name__}]"
        return _FlowGeneric

    def __init__(self) -> None:
        self._methods: Dict[str, Callable] = {}
        self._state: T = self._create_initial_state()
        self._executed_methods: Set[str] = set()
        self._scheduled_tasks: Set[str] = set()
        self._pending_and_listeners: Dict[str, Set[str]] = {}
        self._method_outputs: List[Any] = []  # List to store all method outputs

        self._telemetry.flow_creation_span(self.__class__.__name__)

        for method_name in dir(self):
            if callable(getattr(self, method_name)) and not method_name.startswith(
                "__"
            ):
                self._methods[method_name] = getattr(self, method_name)

    def _create_initial_state(self) -> T:
        if self.initial_state is None and hasattr(self, "_initial_state_T"):
            return self._initial_state_T()  # type: ignore
        if self.initial_state is None:
            return {}  # type: ignore
        elif isinstance(self.initial_state, type):
            return self.initial_state()
        else:
            return self.initial_state

    @property
    def state(self) -> T:
        return self._state

    @property
    def method_outputs(self) -> List[Any]:
        """Returns the list of all outputs from executed methods."""
        return self._method_outputs

    def kickoff(self) -> Any:
        return asyncio.run(self.kickoff_async())

    async def kickoff_async(self) -> Any:
        if not self._start_methods:
            raise ValueError("No start method defined")

        self._telemetry.flow_execution_span(
            self.__class__.__name__, list(self._methods.keys())
        )

        # Create tasks for all start methods
        tasks = [
            self._execute_start_method(start_method)
            for start_method in self._start_methods
        ]

        # Run all start methods concurrently
        await asyncio.gather(*tasks)

        # Safety net: execute any listener whose condition is already satisfied
        # but that was not reached by one of the triggered execution paths.
        await self._flush_ready_listeners()

        # Return the final output (from the last executed method)
        if self._method_outputs:
            return self._method_outputs[-1]
        else:
            return None  # Or raise an exception if no methods were executed

    async def _execute_start_method(self, start_method_name: str) -> None:
        result = await self._execute_method(
            start_method_name, self._methods[start_method_name]
        )
        await self._execute_listeners(start_method_name, result)

    async def _execute_method(
        self, method_name: str, method: Callable, *args: Any, **kwargs: Any
    ) -> Any:
        result = (
            await method(*args, **kwargs)
            if asyncio.iscoroutinefunction(method)
            else method(*args, **kwargs)
        )
        self._method_outputs.append(result)  # Store the output

        self._executed_methods.add(method_name)

        return result

    def _is_condition_met(
        self, listener_name: str, condition_type: str, methods: List[str]
    ) -> bool:
        """Whether the condition of ``listener_name`` has been satisfied."""
        methods = list(methods)
        if not methods:
            return False

        if condition_type == "AND":
            pending = self._pending_and_listeners.get(listener_name, set())
            return all(
                method in pending or method in self._executed_methods
                for method in methods
            )

        return any(method in self._executed_methods for method in methods)

    async def _flush_ready_listeners(self) -> None:
        """Run listeners whose conditions are satisfied but never scheduled."""
        attempted: Set[str] = set()

        while True:
            ready: List[str] = []
            for listener_name, (condition_type, methods) in self._listeners.items():
                if listener_name in attempted:
                    continue
                if (
                    listener_name in self._executed_methods
                    or listener_name in self._scheduled_tasks
                ):
                    continue
                if self._is_condition_met(listener_name, condition_type, methods):
                    ready.append(listener_name)

            if not ready:
                return

            attempted.update(ready)
            for listener_name in ready:
                self._scheduled_tasks.add(listener_name)

            await asyncio.gather(
                *(
                    self._execute_single_listener(listener_name, None)
                    for listener_name in ready
                )
            )

    async def _execute_listeners(self, trigger_method: str, result: Any) -> None:
        listener_tasks: List[asyncio.Task] = []

        # Names that should be considered as "just completed" when matching
        # listener conditions (a router exposes both its own name and the path
        # it returned).
        triggered_names: List[str] = []
        if isinstance(trigger_method, str):
            triggered_names.append(trigger_method)

        # If the trigger method has a router attached, execute the router and
        # use the returned path as the new trigger method.
        if isinstance(trigger_method, str) and trigger_method in self._routers:
            router_name = self._routers[trigger_method]
            router_method = self._methods[router_name]
            path = await self._execute_method(router_name, router_method)

            if router_name not in triggered_names:
                triggered_names.append(router_name)
            if isinstance(path, str) and path not in triggered_names:
                triggered_names.append(path)

            trigger_method = path

        for listener_name, (condition_type, methods) in self._listeners.items():
            if listener_name in self._executed_methods:
                continue
            if listener_name in self._scheduled_tasks:
                continue

            if condition_type == "AND":
                pending = self._pending_and_listeners.setdefault(listener_name, set())
                for triggered_name in triggered_names:
                    if triggered_name in methods:
                        pending.add(triggered_name)

                # The listener only runs once every method of the `and_()`
                # condition has completed, no matter which one finishes last
                # or in which order they were executed.
                if self._is_condition_met(listener_name, condition_type, methods):
                    self._scheduled_tasks.add(listener_name)
                    listener_tasks.append(
                        self._execute_single_listener(listener_name, result)
                    )
            else:
                if any(
                    triggered_name in methods for triggered_name in triggered_names
                ):
                    self._scheduled_tasks.add(listener_name)
                    listener_tasks.append(
                        self._execute_single_listener(listener_name, result)
                    )

        # Run all listener tasks concurrently and wait for them to complete
        await asyncio.gather(*listener_tasks)

    async def _execute_single_listener(self, listener_name: str, result: Any) -> None:
        try:
            method = self._methods[listener_name]
            sig = inspect.signature(method)
            params = list(sig.parameters.values())

            # Exclude 'self' parameter
            method_params = [p for p in params if p.name != "self"]

            if method_params:
                # If listener expects parameters, pass the result
                listener_result = await self._execute_method(
                    listener_name, method, result
                )
            else:
                # If listener does not expect parameters, call without arguments
                listener_result = await self._execute_method(listener_name, method)

            # Remove from scheduled tasks after execution
            self._scheduled_tasks.discard(listener_name)
            self._pending_and_listeners.pop(listener_name, None)

            # Execute listeners of this listener
            await self._execute_listeners(listener_name, listener_result)
        except Exception as e:
            # Make sure a failing listener does not keep being re-scheduled
            self._scheduled_tasks.discard(listener_name)
            self._executed_methods.add(listener_name)
            self._pending_and_listeners.pop(listener_name, None)

            print(
                f"[Flow._execute_single_listener] Error in method {listener_name}: {e}"
            )
            import traceback

            traceback.print_exc()

    def plot(self, filename: str = "crewai_flow") -> None:
        self._telemetry.flow_plotting_span(
            self.__class__.__name__, list(self._methods.keys())
        )

        plot_flow(self, filename)
