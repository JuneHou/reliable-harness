        formatted_prompt = self._format_prompt(self.prompt, inputs)
        self.should_ask_for_human_input = inputs.get(
            "should_ask_for_human_input", False
        )
        self.messages = self._messages(formatted_prompt)

        formatted_answer = self._invoke_loop(formatted_answer)

        if self.should_ask_for_human_input:
            human_feedback = self._ask_human_input(formatted_answer.output)
            if self.crew and self.crew._train:
                self._handle_crew_training_output(formatted_answer, human_feedback)

            # Making sure we only ask for it once, so disabling for the next thought loop
            self.should_ask_for_human_input = False
            self.messages.append(
                {"role": "user", "content": f"Feedback: {human_feedback}"}

                        self.llm,

                        callbacks=self.callbacks,

                        formatted_answer.text += f"\nObservation: {action_result}"

from typing import Any, Dict, List, Optional

from litellm import completion
import litellm


def _is_unsupported_stop_param_error(error: Exception) -> bool:
    """Detect errors caused by models that do not accept the `stop` parameter.

    OpenAI reasoning models (o1-preview / o1-mini) reject parameters such as
    `stop`, and litellm surfaces this as `UnsupportedParamsError` (or as a
    provider `BadRequestError` mentioning the unsupported parameter).
    """
    error_type = type(error).__name__
    message = str(error).lower()

    if "stop" not in message:
        return False

    return error_type == "UnsupportedParamsError" or (
        "unsupported" in message or "not supported" in message or "unrecognized" in message
    )


class LLM:
    def __init__(
        self,
        model: str,
        stop: Optional[List[str]] = None,
        callbacks: Optional[List[Any]] = None,
    ):
        # `stop` is optional: reasoning models such as o1-preview and o1-mini do
        # not support it, so it must only be forwarded when actually provided.
        self.stop: List[str] = stop or []
        self.model = model
        self.callbacks: List[Any] = callbacks or []
        litellm.callbacks = self.callbacks
        litellm.set_verbose = True

    def call(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        params: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "num_retries": 5,
        }

        # Only send `stop` when the user actually provided stop words, otherwise
        # models that do not support it (o1-preview / o1-mini) would raise.
        if self.stop:
            params["stop"] = self.stop

        try:
            response = completion(**params)
        except Exception as e:
            if "stop" in params and _is_unsupported_stop_param_error(e):
                # Model does not support `stop`, retry without it.
                params.pop("stop", None)
                response = completion(**params)
            else:
                raise

        return response["choices"][0]["message"]["content"]

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)

        litellm.callbacks = self.callbacks
        litellm.set_verbose = True

    def call(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        params: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "num_retries": 5,
        }

        # Only send `stop` when the user actually provided stop words, otherwise
        # models that do not support it (o1-preview / o1-mini) would raise.
        if self.stop:
            params["stop"] = self.stop

        try:
            response = completion(**params)
        except Exception as e:
            if "stop" in params and _is_unsupported_stop_param_error(e):
                # Model does not support `stop`, retry without it.
                params.pop("stop", None)
                response = completion(**params)
            else:
                raise

        return response["choices"][0]["message"]["content"]

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)
