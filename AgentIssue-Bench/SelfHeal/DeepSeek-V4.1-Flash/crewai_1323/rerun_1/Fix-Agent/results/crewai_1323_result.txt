import re
from typing import Any, Dict, List, Optional

import litellm
from litellm import completion


class LLM:
    def __init__(
        self,
        model: str,
        stop: Optional[List[str]] = None,
        callbacks: Optional[List[Any]] = None,
    ):
        self.model = model
        self.callbacks = callbacks or []
        # The `stop` parameter is optional and must not be forwarded to models
        # that do not support it (e.g. o1-preview / o1-mini).
        self.stop = stop if self.supports_stop_words() else None
        litellm.callbacks = self.callbacks
        litellm.set_verbose = True

    @staticmethod
    def supports_stop_words(model: Any = None, default: Optional[Any] = None) -> bool:
        """Check whether the given model (or this instance's model) accepts `stop`.

        o1 family models (o1, o1-preview, o1-mini, ...) reject the `stop`
        parameter, so it has to be omitted for them.
        """
        target = model
        if isinstance(model, LLM):
            target = model.model
        elif model is None:
            target = default

        if target is None:
            return True

        name = str(target).strip().lower()
        # Remove provider/route prefixes such as "openai/" or "azure/o1-preview"
        name = name.split("/")[-1]
        return not bool(re.match(r"^o1([-_]|$)", name))

    def call(self, messages: List[Dict[str, str]], callbacks: List[Any] = []) -> str:
        if callbacks:
            litellm.success_callback = callbacks

        params: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "num_retries": 5,
        }

        # Only send `stop` when it is actually set and supported by the model.
        if self.stop and self.supports_stop_words(default=self.model):
            params["stop"] = self.stop

        response = completion(**params)
        return response["choices"][0]["message"]["content"]

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)

        # that do not support it (e.g. o1-preview / o1-mini).
        self.stop = stop if self.supports_stop_words() else None
        litellm.callbacks = self.callbacks
        litellm.set_verbose = True

    @staticmethod
    def supports_stop_words(model: Any = None, default: Optional[Any] = None) -> bool:
        """Check whether the given model (or this instance's model) accepts `stop`.

        o1 family models (o1, o1-preview, o1-mini, ...) reject the `stop`
        parameter, so it has to be omitted for them.
        """
        target = model
        if isinstance(model, LLM):
            target = model.model
        elif model is None:
            target = default

        if target is None:
            return True

        name = str(target).strip().lower()
        # Remove provider/route prefixes such as "openai/" or "azure/o1-preview"
        name = name.split("/")[-1]
        return not bool(re.match(r"^o1([-_]|$)", name))

    def call(self, messages: List[Dict[str, str]], callbacks: List[Any] = []) -> str:
        if callbacks:
            litellm.success_callback = callbacks

        params: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "num_retries": 5,
        }

        # Only send `stop` when it is actually set and supported by the model.
        if self.stop and self.supports_stop_words(default=self.model):
            params["stop"] = self.stop

        response = completion(**params)
        return response["choices"][0]["message"]["content"]

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)
