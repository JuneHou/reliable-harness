from typing import Any, Dict, List, Optional

import litellm
from litellm import completion


class LLM:
    """Thin wrapper around litellm's `completion` endpoint.

    The `stop` parameter is optional: models such as OpenAI's `o1-preview` and
    `o1-mini` do not support it and raise an "Unsupported parameter" error when
    it is sent. Therefore it is only forwarded to the provider when the
    configured model actually supports stop sequences.
    """

    def __init__(
        self,
        model: str,
        stop: Optional[List[str]] = None,
        callbacks: Optional[List[Any]] = None,
    ):
        self.model = model
        self.stop = self._normalize_stop(stop)
        self.callbacks = callbacks or []
        litellm.callbacks = self.callbacks
        # Silently drop any parameter that the target model does not support
        # instead of failing the request (e.g. `stop` for o1 models).
        litellm.drop_params = True
        litellm.set_verbose = True

    @staticmethod
    def _normalize_stop(stop: Optional[List[str]]) -> List[str]:
        if not stop:
            return []
        if isinstance(stop, str):
            return [stop]
        return list(stop)

    def is_o1_model(self) -> bool:
        """Whether the configured model belongs to the OpenAI o1 family."""
        model = (self.model or "").lower().strip()
        return (
            model.startswith("o1")
            or model.startswith("openai/o1")
            or "/o1" in model
        )

    def supports_stop_words(self) -> bool:
        """Stop sequences are not supported by reasoning (o1) models."""
        return not self.is_o1_model()

    def _completion_params(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        params: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "num_retries": 5,
        }
        if self.stop and self.supports_stop_words():
            params["stop"] = self.stop
        return params

    def call(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        response = completion(**self._completion_params(messages))
        return response["choices"][0]["message"]["content"]

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)

            return []
        if isinstance(stop, str):
            return [stop]
        return list(stop)

    def is_o1_model(self) -> bool:
        """Whether the configured model belongs to the OpenAI o1 family."""
        model = (self.model or "").lower().strip()
        return (
            model.startswith("o1")
            or model.startswith("openai/o1")
            or "/o1" in model
        )

    def supports_stop_words(self) -> bool:
        """Stop sequences are not supported by reasoning (o1) models."""
        return not self.is_o1_model()

    def _completion_params(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        params: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "num_retries": 5,
        }
        if self.stop and self.supports_stop_words():
            params["stop"] = self.stop
        return params

    def call(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        response = completion(**self._completion_params(messages))
        return response["choices"][0]["message"]["content"]

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)
