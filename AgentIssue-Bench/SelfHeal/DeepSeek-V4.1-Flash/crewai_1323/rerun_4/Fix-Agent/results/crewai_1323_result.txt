            if self.crew and self.crew._train:
                self._handle_crew_training_output(formatted_answer, human_feedback)

            self.should_ask_for_human_input = False
            self.messages.append(
                {"role": "user", "content": f"Feedback: {human_feedback}"}

import logging
import re
from typing import Any, Dict, List, Optional

import litellm
from litellm import completion


class LLM:
    def __init__(
        self,
        model: str,
        stop: Optional[List[str]] = None,
        callbacks: Optional[List[Any]] = None,
    ):
        self.model = model
        self.stop = stop or []
        self.callbacks = callbacks or []
        # Not every model/provider accepts the `stop` parameter (e.g. OpenAI's
        # o1-preview / o1-mini reasoning models), so it must be optional.
        self.supports_stop_words = self._supports_stop_words()

        litellm.callbacks = self.callbacks
        litellm.set_verbose = True

    def call(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        params: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "num_retries": 5,
        }

        # Only send `stop` when the model supports it and it was actually given,
        # otherwise providers like OpenAI raise an unsupported-parameter error.
        if self.supports_stop_words and self.stop:
            params["stop"] = self.stop

        response = completion(**params)
        return response["choices"][0]["message"]["content"]

    def _supports_stop_words(self) -> bool:
        """Check whether the underlying model supports the `stop` parameter."""
        model_name = (self.model or "").lower()

        # OpenAI reasoning models (o1-preview, o1-mini, o3-mini, ...) do not
        # support `stop`; short-circuit so we never forward it for them.
        if re.search(r"(^|[/:])o[13]([-_/]|$)", model_name):
            return False

        try:
            from litellm.utils import supports_stop_words

            return bool(supports_stop_words(self.model))
        except Exception as e:
            logging.error(
                f"Error checking if model {self.model} supports stop words: {e}"
            )
            # Default to the previous behaviour when the check can't be done.
            return True

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)

            "num_retries": 5,
        }

        # Only send `stop` when the model supports it and it was actually given,
        # otherwise providers like OpenAI raise an unsupported-parameter error.
        if self.supports_stop_words and self.stop:
            params["stop"] = self.stop

        response = completion(**params)
        return response["choices"][0]["message"]["content"]

    def _supports_stop_words(self) -> bool:
        """Check whether the underlying model supports the `stop` parameter."""
        model_name = (self.model or "").lower()

        # OpenAI reasoning models (o1-preview, o1-mini, o3-mini, ...) do not
        # support `stop`; short-circuit so we never forward it for them.
        if re.search(r"(^|[/:])o[13]([-_/]|$)", model_name):
            return False

        try:
            from litellm.utils import supports_stop_words

            return bool(supports_stop_words(self.model))
        except Exception as e:
            logging.error(
                f"Error checking if model {self.model} supports stop words: {e}"
            )
            # Default to the previous behaviour when the check can't be done.
            return True

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)
