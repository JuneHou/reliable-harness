from typing import Any, Dict, List, Optional

import litellm
from litellm import completion


class LLM:
    def __init__(
        self,
        model: str,
        stop: Optional[List[str]] = None,
        callbacks: Optional[List[Any]] = None,
    ):
        self.stop = stop or []
        self.model = model
        self.callbacks = callbacks or []
        litellm.callbacks = self.callbacks
        litellm.set_verbose = True

    def call(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        params: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "num_retries": 5,
        }

        # The `stop` parameter is optional: models like OpenAI's `o1-preview`
        # and `o1-mini` do not support stop sequences and raise an error when
        # it is passed. Only forward it when stop words are configured and the
        # target model supports them.
        if self.stop and not self._model_rejects_stop(self.model):
            params["stop"] = self.stop

        response = completion(**params)
        return response["choices"][0]["message"]["content"]

    @staticmethod
    def _model_rejects_stop(model: str) -> bool:
        """Return True when the model does not accept a `stop` parameter."""
        if not model:
            return False
        # Strip provider prefixes such as "openai/o1-preview" or "azure/o1-mini"
        normalized_model = model.split("/")[-1].strip().lower()
        return normalized_model.startswith(("o1", "o3"))

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)

        return normalized_model.startswith(("o1", "o3"))

    def _call_callbacks(self, formatted_answer):
        for callback in self.callbacks:
            callback(formatted_answer)
