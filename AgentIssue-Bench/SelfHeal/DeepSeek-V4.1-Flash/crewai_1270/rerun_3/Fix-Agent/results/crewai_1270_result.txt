# Encodings that are tried, in order, when reading YAML configuration files.
# UTF-8 (with or without a BOM) is the most common one, followed by the
# encodings typically used for Chinese (GBK / GB18030 / Big5) and a
# last-resort single byte encoding that can never fail to decode.
ENCODING_CANDIDATES = (
    "utf-8-sig",
    "utf-8",
    "gb18030",
    "gbk",
    "big5",
    "latin-1",
)


def detect_text_encoding(raw_content: bytes) -> str:
    """Best effort detection of the encoding used by ``raw_content``.

    Common encodings are tried first (UTF-8 first, then the Chinese variants),
    and ``chardet`` (or ``charset_normalizer``) is used as a fallback when it
    is available. ``latin-1`` is used as a final safety net because decoding
    arbitrary bytes with it never fails.
    """
    if not raw_content:
        return "utf-8"

    # Respect an explicit BOM when present.
    if raw_content.startswith(b"\xef\xbb\xbf"):
        return "utf-8-sig"
    if raw_content.startswith((b"\xff\xfe", b"\xfe\xff")):
        return "utf-16"

    for candidate in ENCODING_CANDIDATES:
        try:
            raw_content.decode(candidate)
        except (UnicodeDecodeError, LookupError):
            continue
        return candidate

    detected_encoding = None
    try:
        import chardet  # type: ignore

        detected_encoding = (chardet.detect(raw_content) or {}).get("encoding")
    except ImportError:
        try:
            from charset_normalizer import from_bytes  # type: ignore

            best_match = from_bytes(raw_content).best()
            detected_encoding = best_match.encoding if best_match else None
        except ImportError:
            detected_encoding = None

    if detected_encoding:
        try:
            raw_content.decode(detected_encoding)
        except (UnicodeDecodeError, LookupError):
            detected_encoding = None

    return detected_encoding or "utf-8"

            if context_list := task_info.get("context"):
