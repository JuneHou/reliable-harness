import { JSONSchema7Definition } from '@ai-sdk/provider';

/**
 * Converts JSON Schema 7 to OpenAPI Schema 3.0
 */
export function convertJSONSchemaToOpenAPISchema(
  jsonSchema: JSONSchema7Definition | undefined,
  isRoot: boolean = true,
): unknown {
  // parameters need to be undefined if they are empty objects:
  // (this is only valid for the root schema, because `undefined` is not a
  // valid value inside of a schema, e.g. for object properties)
  if (jsonSchema == null || (isRoot && isEmptyObjectSchema(jsonSchema))) {
    return undefined;
  }

  if (typeof jsonSchema === 'boolean') {
    return { type: 'boolean', properties: {} };
  }

  const {
    type,
    description,
    required,
    properties,
    items,
    allOf,
    anyOf,
    oneOf,
    format,
    const: constValue,
    minLength,
    enum: enumValues,
  } = jsonSchema;

  const result: Record<string, unknown> = {};

  if (description) result.description = description;
  if (required) result.required = required;
  if (format) result.format = format;

  if (constValue !== undefined) {
    result.enum = [constValue];
  }

  // Handle type
  if (type) {
    if (Array.isArray(type)) {
      if (type.includes('null')) {
        const nonNullTypes = type.filter(t => t !== 'null');
        // `type` cannot be undefined in the generated schema,
        // so we fall back to null if there is no other type:
        result.type = nonNullTypes.length > 0 ? nonNullTypes[0] : 'null';
        if (result.type !== 'null') {
          result.nullable = true;
        }
      } else {
        result.type = type;
      }
    } else if (type === 'null') {
      result.type = 'null';
    } else {
      result.type = type;
    }
  }

  // Handle enum
  if (enumValues !== undefined) {
    result.enum = enumValues;
  }

  if (properties != null) {
    result.properties = Object.entries(properties).reduce(
      (acc, [key, value]) => {
        const convertedValue = convertJSONSchemaToOpenAPISchema(value, false);

        // `undefined` is not a valid value inside of a schema. This can
        // happen for schemas that describe "empty" objects, such as
        // `z.record(...)` or `z.object({})`. Nested schemas are therefore
        // kept as (unconstrained) object schemas instead of being removed:
        if (convertedValue !== undefined) {
          acc[key] = convertedValue;
        }

        return acc;
      },
      {} as Record<string, unknown>,
    );
  }

  if (items) {
    result.items = Array.isArray(items)
      ? items.map(item => convertJSONSchemaToOpenAPISchema(item, false))
      : convertJSONSchemaToOpenAPISchema(items, false);
  }

  if (allOf) {
    result.allOf = allOf.map(schema =>
      convertJSONSchemaToOpenAPISchema(schema, false),
    );
  }
  if (anyOf) {
    // Handle cases where anyOf includes a null type
    if (
      anyOf.some(
        schema => typeof schema === 'object' && schema?.type === 'null',
      )
    ) {
      const nonNullSchemas = anyOf.filter(
        schema => !(typeof schema === 'object' && schema?.type === 'null'),
      );

      if (nonNullSchemas.length === 1) {
        // If there's only one non-null schema, convert it and make it nullable
        const converted = convertJSONSchemaToOpenAPISchema(
          nonNullSchemas[0],
          false,
        );
        if (typeof converted === 'object' && converted != null) {
          result.nullable = true;
          Object.assign(result, converted);
        }
      } else {
        // If there are multiple non-null schemas, keep them in anyOf
        result.anyOf = nonNullSchemas.map(schema =>
          convertJSONSchemaToOpenAPISchema(schema, false),
        );
        result.nullable = true;
      }
    } else {
      result.anyOf = anyOf.map(schema =>
        convertJSONSchemaToOpenAPISchema(schema, false),
      );
    }
  }
  if (oneOf) {
    result.oneOf = oneOf.map(schema =>
      convertJSONSchemaToOpenAPISchema(schema, false),
    );
  }

  if (minLength !== undefined) {
    result.minLength = minLength;
  }

  return result;
}

function isEmptyObjectSchema(jsonSchema: JSONSchema7Definition): boolean {
  return (
    jsonSchema != null &&
    typeof jsonSchema === 'object' &&
    jsonSchema.type === 'object' &&
    (jsonSchema.properties == null ||
      Object.keys(jsonSchema.properties).length === 0) &&
    // schemas with `additionalProperties` (such as `z.record(...)`) describe
    // objects with arbitrary keys and are therefore not empty:
    (jsonSchema.additionalProperties == null ||
      jsonSchema.additionalProperties === false)
  );
}

- `type: ['null']` no longer sets `result.type = undefined` (which produced an invalid `{type: undefined}`); it falls back to `'null'` and only sets `nullable` for a real non-null type. Consistent with the existing `explicitNullField` expectation (`{type:'null'}`).
- `converted != null` guard on the `anyOf` null-collapse `Object.assign` path prevents a `TypeError` from `Object.assign(result, null/undefined)`.

### 2. Compatibility with existing tests

I traced each existing test case: root-level empty-object tests still return `undefined` (`{type:'object'}`, `{type:'object', properties:{}}` — both `isRoot === true`); `{minLength: 5}`, `{type:'null'}`, `{type:'boolean'}` inside `allOf`/`oneOf`/`anyOf` are unaffected (they are not `type: 'object'`); the nullable-enum `anyOf` collapse still yields `{nullable: true, type:'string', enum:[...]}`; const→enum, format, required, description, and `additionalProperties: false`/`$schema` stripping all behave as before. Note existing tests use `toEqual`, which ignores `undefined` keys, so no regressions there.

### 3. New unit tests

The three added tests match the implementation exactly:
- `record` (`{type:'object', additionalProperties:{type:'string'}}`) → `{type:'object'}` (additionalProperties is dropped, consistent with this converter's existing behavior of stripping it, as asserted in tests 1–2); `emptyObject` → `{type:'object'}` instead of `undefined`.
- root-level `z.record` → `{type:'object'}` (not `undefined`, because `additionalProperties` now makes it non-empty).
- nested `items`/`anyOf` empty objects survive as `{type:'object'}`, i.e. no `undefined` in arrays/unions.

I also noted some limitations that are not correctness blockers: `additionalProperties` payload is still discarded (so record value types are lost — the same information loss as before this change), and the provided integration script inspects `zodToJsonSchema(zodSchema)` output directly rather than the converter's output, so it only weakly exercises the changed file.

### 4. API verification

`verify_api_usage` calls (`zodToJsonSchema`, `ai tool`, `JSONSchema7`, `additionalProperties`, `Object.assign`) returned unrelated/noisy web results, so no external contradiction was found; I validated usage against the source semantics instead. `additionalProperties` is a valid `JSONSchema7` optional field (`boolean | JSONSchema7Definition`), so `jsonSchema.additionalProperties == null || === false` type-checks, and the added optional parameter keeps the public signature backward compatible. `Object.assign` correctly requires a non-null source, which the new guard provides.

### 5. Structural validation

`validate_fixed_code` reports ACCEPT: both files present once with full text (no duplication/truncation), no interface conflict for this issue.

The fix addresses the root cause (recursive collapsing of empty object schemas to `undefined`), covers all recursion paths, adds defensive fixes for two adjacent invalid-output/throw cases (`type: ['null']`, `Object.assign` with null), is backward compatible with all existing tests, and is covered by new targeted tests.

### packages/google/src/convert-json-schema-to-openapi-schema.ts
import { JSONSchema7Definition } from '@ai-sdk/provider';

/**
 * Converts JSON Schema 7 to OpenAPI Schema 3.0
 */
export function convertJSONSchemaToOpenAPISchema(
  jsonSchema: JSONSchema7Definition | undefined,
  isRoot: boolean = true,
): unknown {
  // parameters need to be undefined if they are empty objects:
  // (this is only valid for the root schema, because `undefined` is not a
  // valid value inside of a schema, e.g. for object properties)
  if (jsonSchema == null || (isRoot && isEmptyObjectSchema(jsonSchema))) {
    return undefined;
  }

  if (typeof jsonSchema === 'boolean') {
    return { type: 'boolean', properties: {} };
  }

  const {
    type,
    description,
    required,
    properties,
    items,
    allOf,
    anyOf,
    oneOf,
    format,
    const: constValue,
    minLength,
    enum: enumValues,
  } = jsonSchema;

  const result: Record<string, unknown> = {};

  if (description) result.description = description;
  if (required) result.required = required;
  if (format) result.format = format;

  if (constValue !== undefined) {
    result.enum = [constValue];
  }

  // Handle type
  if (type) {
    if (Array.isArray(type)) {
      if (type.includes('null')) {
        const nonNullTypes = type.filter(t => t !== 'null');
        // `type` cannot be undefined in the generated schema,
        // so we fall back to null if there is no other type:
        result.type = nonNullTypes.length > 0 ? nonNullTypes[0] : 'null';
        if (result.type !== 'null') {
          result.nullable = true;
        }
      } else {
        result.type = type;
      }
    } else if (type === 'null') {
      result.type = 'null';
    } else {
      result.type = type;
    }
  }

  // Handle enum
  if (enumValues !== undefined) {
    result.enum = enumValues;
  }

  if (properties != null) {
    result.properties = Object.entries(properties).reduce(
      (acc, [key, value]) => {
        const convertedValue = convertJSONSchemaToOpenAPISchema(value, false);

        // `undefined` is not a valid value inside of a schema. This can
        // happen for schemas that describe "empty" objects, such as
        // `z.record(...)` or `z.object({})`. Nested schemas are therefore
        // kept as (unconstrained) object schemas instead of being removed:
        if (convertedValue !== undefined) {
          acc[key] = convertedValue;
        }

        return acc;
      },
      {} as Record<string, unknown>,
    );
  }

  if (items) {
    result.items = Array.isArray(items)
      ? items.map(item => convertJSONSchemaToOpenAPISchema(item, false))
      : convertJSONSchemaToOpenAPISchema(items, false);
  }

  if (allOf) {
    result.allOf = allOf.map(schema =>
      convertJSONSchemaToOpenAPISchema(schema, false),
    );
  }
  if (anyOf) {
    // Handle cases where anyOf includes a null type
    if (
      anyOf.some(
        schema => typeof schema === 'object' && schema?.type === 'null',
      )
    ) {
      const nonNullSchemas = anyOf.filter(
        schema => !(typeof schema === 'object' && schema?.type === 'null'),
      );

      if (nonNullSchemas.length === 1) {
        // If there's only one non-null schema, convert it and make it nullable
        const converted = convertJSONSchemaToOpenAPISchema(
          nonNullSchemas[0],
          false,
        );
        if (typeof converted === 'object' && converted != null) {
          result.nullable = true;
          Object.assign(result, converted);
        }
      } else {
        // If there are multiple non-null schemas, keep them in anyOf
        result.anyOf = nonNullSchemas.map(schema =>
          convertJSONSchemaToOpenAPISchema(schema, false),
        );
        result.nullable = true;
      }
    } else {
      result.anyOf = anyOf.map(schema =>
        convertJSONSchemaToOpenAPISchema(schema, false),
      );
    }
  }
  if (oneOf) {
    result.oneOf = oneOf.map(schema =>
      convertJSONSchemaToOpenAPISchema(schema, false),
    );
  }

  if (minLength !== undefined) {
    result.minLength = minLength;
  }

  return result;
}

function isEmptyObjectSchema(jsonSchema: JSONSchema7Definition): boolean {
  return (
    jsonSchema != null &&
    typeof jsonSchema === 'object' &&
    jsonSchema.type === 'object' &&
    (jsonSchema.properties == null ||
      Object.keys(jsonSchema.properties).length === 0) &&
    // schemas with `additionalProperties` (such as `z.record(...)`) describe
    // objects with arbitrary keys and are therefore not empty:
    (jsonSchema.additionalProperties == null ||
      jsonSchema.additionalProperties === false)
  );
}
