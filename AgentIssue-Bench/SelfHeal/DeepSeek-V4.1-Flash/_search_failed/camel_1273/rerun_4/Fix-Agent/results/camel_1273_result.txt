def print_text_animated(text, delay: float = 0.02, end: str = ""):
    r"""Prints the given text with an animated effect.

    Args:
        text (str): The text to print.
        delay (float, optional): The delay between each character printed.
            (default: :obj:`0.02`)

    print('\n')


def get_prompt_template_key_words(template: str) -> Set[str]:

    the words inside the braces.

    Returns:

def get_first_int(string: str) -> Optional[int]:
    r"""Returns the first integer number found in the given string.

    If no integer number is found, returns None.

    Args:
        string (str): The input string.

    Returns:
        int or None: The first integer number found in the string, or None if
            no integer number is found.
    """
    match = re.search(r'\d+', string)
    if match:
        return int(match.group())
    else:
        return None


def download_tasks(task: TaskType, folder_path: str) -> None:
    r"""Downloads task-related files from a specified URL and extracts them.

    This function downloads a zip file containing tasks based on the specified
    `task` type from a predefined URL, saves it to `folder_path`, and then
    extracts the contents of the zip file into the same folder. After
    extraction, the zip file is deleted.

    Args:
        task (TaskType): An enum representing the type of task to download.
        folder_path (str): The path of the folder where the zip file will be
                           downloaded and extracted.
    """
    # Define the path to save the zip file
    zip_file_path = os.path.join(folder_path, "tasks.zip")

    # Download the zip file from the Google Drive link
    response = requests.get(
        "https://huggingface.co/datasets/camel-ai/"
        f"metadata/resolve/main/{task.value}_tasks.zip"
    )

    # Save the zip file
    with open(zip_file_path, "wb") as f:
        f.write(response.content)

    with zipfile.ZipFile(zip_file_path, "r") as zip_ref:
        zip_ref.extractall(folder_path)

    # Delete the zip file
    os.remove(zip_file_path)


def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()
            if task_name.strip() and task_id.isnumeric():
                new_tasks_list.append(task_name)
    return new_tasks_list


def check_server_running(server_url: str) -> bool:
    r"""Check whether the port refered by the URL to the server
    is open.

    Args:
        server_url (str): The URL to the server running LLM inference
            service.

    Returns:
        bool: Whether the port is open for packets (server is running).
    """
    parsed_url = urlparse(server_url)
    url_tuple = (parsed_url.hostname, parsed_url.port)

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    result = sock.connect_ex(url_tuple)
    sock.close()

    # if the port is open, the result should be 0.
    return result == 0


def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required module dependencies.

    Raises:
        ImportError: If any of the required modules are not available.

    Example:
        ::

            @dependencies_required('numpy', 'pandas')
            def data_processing_function():
                # Function implementation...
    """

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            missing_modules = [
                m for m in required_modules if not is_module_available(m)
            ]
            if missing_modules:
                raise ImportError(
                    f"Missing required modules: {', '.join(missing_modules)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

# Names that are commonly used for an API key argument/attribute, no matter
# which service the key belongs to.
_GENERIC_API_KEY_NAMES: Set[str] = {
    'apikey',
    'apikeys',
    'key',
    'keys',
    'token',
    'accesstoken',
}


def _normalize_api_key_name(name: str) -> str:
    r"""Normalize the name of an API key so that differently formatted names
    of the same key can be compared.

    Args:
        name (str): The name of the API key, e.g. :obj:`'OPENAI_API_KEY'`.

    Returns:
        str: The normalized name, e.g. :obj:`'openaiapikey'`.
    """
    return re.sub(r'[^a-z0-9]', '', str(name).lower())


def _api_key_aliases(key: str) -> Set[str]:
    r"""Return the normalized names under which an API key may be provided.

    For example :obj:`'OPENAI_API_KEY'` produces
    :obj:`{'openaiapikey', 'openai'}`, so that values passed as
    :obj:`openai_api_key` or :obj:`openai` are recognized.

    Args:
        key (str): The name of the API key.

    Returns:
        Set[str]: The normalized aliases of the key name.
    """
    normalized = _normalize_api_key_name(key)
    aliases: Set[str] = set()
    if normalized:
        aliases.add(normalized)
    for suffix in ('apikeys', 'apikey', 'key', 'token'):
        if normalized.endswith(suffix) and len(normalized) > len(suffix):
            aliases.add(normalized[: -len(suffix)])
    return aliases


def _api_key_attribute_candidates(key: str) -> Set[str]:
    r"""Return the instance attribute names that may hold an API key.

    Args:
        key (str): The name of the API key.

    Returns:
        Set[str]: The candidate attribute names.
    """
    candidates: Set[str] = {
        'api_key',
        '_api_key',
        'api_keys',
        '_api_keys',
        key,
        key.lower(),
        f'_{key.lower()}',
    }
    for alias in _api_key_aliases(key):
        candidates.update(
            {
                alias,
                f'_{alias}',
                f'{alias}_api_key',
                f'_{alias}_api_key',
                f'{alias}_key',
                f'_{alias}_key',
            }
        )
    return {c for c in candidates if c}


def _safe_getattr(obj: Any, name: str) -> Any:
    r"""Get an attribute of an object without raising any error.

    Args:
        obj (Any): The object to read the attribute from.
        name (str): The name of the attribute.

    Returns:
        Any: The value of the attribute, or :obj:`None` if it is not
            available.
    """
    try:
        return getattr(obj, name, None)
    except Exception:
        return None


def _is_api_key_provided(
    key_name: str,
    env_var: Optional[str],
    arguments: Mapping[str, Any],
    instance: Any,
    key_mappings: List[Mapping],
    single_key_required: bool,
) -> bool:
    r"""Check whether a required API key has been provided.

    The key is looked up, in order, in the arguments passed to the decorated
    function (this is what makes keys passed directly to a decorated
    ``__init__`` recognized), in the collected API key mappings, in the
    attributes of the instance and finally in the environment variables.

    Args:
        key_name (str): The name of the required API key.
        env_var (Optional[str]): The name of the environment variable holding
            the key, :obj:`None` if it should not be checked.
        arguments (Mapping[str, Any]): The arguments bound to the decorated
            function.
        instance (Any): The instance the decorated method is called on, if
            any.
        key_mappings (List[Mapping]): Mappings of API key names to values.
        single_key_required (bool): Whether the decorated function only
            requires a single API key.

    Returns:
        bool: Whether the API key has been provided.
    """
    aliases = _api_key_aliases(key_name)

    def _matches(name: str) -> bool:
        normalized = _normalize_api_key_name(name)
        if not normalized:
            return False
        if normalized in aliases:
            return True
        # A single required key may be passed through a generic argument or
        # attribute, e.g. ``__init__(self, api_key=...)``.
        return single_key_required and normalized in _GENERIC_API_KEY_NAMES

    # 1. Values passed directly to the decorated function. For a decorated
    #    ``__init__`` method, no instance attribute exists yet, hence the
    #    arguments have to be inspected.
    for arg_name, arg_value in arguments.items():
        if str(arg_name) in ('self', 'cls'):
            continue
        if not arg_value or not isinstance(arg_value, (str, bytes)):
            continue
        if _matches(str(arg_name)):
            return True

    # 2. Values collected in mappings such as
    #    ``api_keys={'OPENAI_API_KEY': '...'}``.
    for mapping in key_mappings:
        for map_key, map_value in mapping.items():
            if map_value and _matches(str(map_key)):
                return True

    # 3. Values stored as attributes of the instance.
    if instance is not None:
        candidates = _api_key_attribute_candidates(key_name)
        try:
            for attr_name in vars(instance):
                candidates.add(str(attr_name))
                candidates.add(str(attr_name).lstrip('_'))
        except TypeError:
            pass
        for attr_name in candidates:
            if not _matches(attr_name):
                continue
            value = _safe_getattr(instance, attr_name)
            if value and isinstance(value, (str, bytes)):
                return True

    # 4. Values provided through the environment variables.
    if env_var is not None:
        if env_var in os.environ or os.getenv(env_var.upper()):
            return True

    return False


def api_keys_required(*required_keys: Any) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are provided.

    Each key is looked up in the arguments passed to the decorated function
    (so that API keys passed directly to a decorated :obj:`__init__` method
    are recognized), in the attributes of the instance, and in the
    environment variables.

    Args:
        required_keys (str): The required API keys to be checked. A tuple of
            :obj:`(name, environment variable)` can also be provided when the
            name of the key and the name of the environment variable differ.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            arguments of the function, the instance attributes and the
            environment variables.

    Example:
        ::

            @api_keys_required('API_KEY_1', 'API_KEY_2')
            def some_api_function():
                # Function implementation...
    """

    def _parse_required_keys() -> List[Tuple[str, Optional[str]]]:
        r"""Normalize the required keys into (name, env variable) pairs."""
        parsed: List[Tuple[str, Optional[str]]] = []

        def _add(item: Any) -> None:
            if isinstance(item, str):
                parsed.append((item, item))
            elif isinstance(item, (list, tuple)):
                if item and isinstance(item[0], str):
                    env_var = item[1] if len(item) > 1 else item[0]
                    parsed.append(
                        (
                            item[0],
                            None if env_var is None else str(env_var),
                        )
                    )
                else:
                    for sub_item in item:
                        _add(sub_item)
            else:
                parsed.append((str(item), str(item)))

        for required_key in required_keys:
            _add(required_key)
        return parsed

    keys_to_check = _parse_required_keys()

    def decorator(func: F) -> F:
        try:
            signature: Optional[inspect.Signature] = inspect.signature(func)
        except (TypeError, ValueError):
            signature = None

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            # Bind the arguments so that keys passed positionally as well as
            # by keyword are detected, which is required for decorated
            # ``__init__`` methods.
            arguments: Dict[str, Any] = dict(kwargs)
            if signature is not None:
                try:
                    bound_args = signature.bind(*args, **kwargs)
                    bound_args.apply_defaults()
                    arguments = dict(bound_args.arguments)
                except TypeError:
                    pass

            instance = args[0] if args else arguments.get('self')

            # API keys may also be supplied as a mapping, e.g. through an
            # ``api_keys`` argument or an ``api_keys`` instance attribute.
            key_mappings: List[Mapping] = []
            for source in (
                kwargs.get('api_keys'),
                arguments.get('api_keys'),
                _safe_getattr(instance, 'api_keys'),
                _safe_getattr(instance, '_api_keys'),
            ):
                if isinstance(source, Mapping):
                    key_mappings.append(source)

            single_key_required = len(keys_to_check) == 1
            missing_keys: List[str] = []

            for key_name, env_var in keys_to_check:
                if _is_api_key_provided(
                    key_name,
                    env_var,
                    arguments,
                    instance,
                    key_mappings,
                    single_key_required,
                ):
                    continue
                if env_var in (None, key_name):
                    missing_keys.append(key_name)
                else:
                    missing_keys.append(
                        f"{key_name} (or environment variable: {env_var})"
                    )

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper

def handle_http_error(response: requests.Response) -> str:
    r"""Handles the HTTP errors based on the status code of the response.

    Args:
        response (requests.Response): The HTTP response from the API call.

    Returns:
        str: The error type, based on the status code.
    """
    if response.status_code == HTTPStatus.UNAUTHORIZED:
        return "Unauthorized. Check your access token."
    elif response.status_code == HTTPStatus.FORBIDDEN:
        return "Forbidden. You do not have permission to perform this action."
    elif response.status_code == HTTPStatus.NOT_FOUND:

    elif response.status_code == HTTPStatus.TOO_MANY_REQUESTS:
        return "Too Many Requests. You have hit the rate limit."
    else:
        return "HTTP Error"


def retry_request(
    func: Callable, retries: int = 3, delay: int = 1, *args: Any, **kwargs: Any
) -> Any:
    r"""Retries a function in case of any errors.

    Args:
        func (Callable): The function to be retried.
        retries (int): Number of retry attempts. (default: :obj:`3`)
        delay (int): Delay between retries in seconds. (default: :obj:`1`)
        *args: Arguments to pass to the function.
        **kwargs: Keyword arguments to pass to the function.

    Returns:
        Any: The result of the function call if successful.

    Raises:
        Exception: If all retry attempts fail.
    """
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            print(f"Attempt {attempt + 1}/{retries} failed: {e}")
            if attempt < retries - 1:
                time.sleep(delay)
            else:
                raise

def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()
            if task_name.strip() and task_id.isnumeric():
                new_tasks_list.append(task_name)
    return new_tasks_list

def api_keys_required(
    *required_keys: Any,
) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are provided either as
    environment variables or as (keyword) arguments of the decorated function,
    e.g. the API key passed directly to a decorated ``__init__`` method.

    Args:
        required_keys: The required API keys to be checked. Each item can be:

            - a :obj:`tuple` :obj:`(param_name, env_var_name)`, where
              :obj:`param_name` is the name of the function parameter that may
              hold the API key (or :obj:`None` if only the environment
              variable should be checked) and :obj:`env_var_name` is the name
              of the corresponding environment variable;
            - a :obj:`list` of such tuples, e.g.
              ``@api_keys_required([('api_key', 'API_KEY')])``;
            - a plain :obj:`str` environment variable name, for backward
              compatibility, e.g. ``@api_keys_required('API_KEY')``.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            function arguments and the environment variables.

    Example:
        ::

            @api_keys_required([('api_key', 'API_KEY_1'),
                                ('other_api_key', 'API_KEY_2')])
            def some_api_function(api_key: Optional[str] = None,
                                  other_api_key: Optional[str] = None):
                # Function implementation...
    """
    # Normalize the accepted forms of ``required_keys`` into a list of
    # ``(param_name, env_var_name)`` tuples.
    normalized_keys: List[Tuple[Optional[str], str]] = []

    def _add_key(key: Any) -> None:
        if isinstance(key, str):
            normalized_keys.append((None, key))
        elif isinstance(key, (tuple, list)):
            if len(key) == 1:
                normalized_keys.append((None, key[0]))
            elif len(key) == 2:
                normalized_keys.append((key[0], key[1]))
            else:
                raise ValueError(
                    "Each required API key must be provided as "
                    "(param_name, env_var_name) or as an environment "
                    f"variable name, got: {key!r}"
                )
        else:
            raise ValueError(
                "Each required API key must be provided as "
                "(param_name, env_var_name) or as an environment "
                f"variable name, got: {key!r}"
            )

    for key in required_keys:
        # Support ``@api_keys_required([(...), (...)])`` usage.
        if isinstance(key, (list, tuple)) and key and all(
            isinstance(item, (list, tuple)) for item in key
        ):
            for item in key:
                _add_key(item)
        else:
            _add_key(key)

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            # Bind the arguments to the signature of the wrapped function so
            # that API keys passed positionally or as keyword arguments (e.g.
            # directly to an ``__init__`` method) are recognized as well.
            signature = inspect.signature(func)
            bound_arguments = signature.bind(*args, **kwargs)
            bound_arguments.apply_defaults()

            missing_keys: List[str] = []
            for param_name, env_var_name in normalized_keys:
                if param_name is not None:
                    param_value = bound_arguments.arguments.get(param_name)
                    if param_value is not None and param_value != "":
                        # The API key is passed directly to the function.
                        continue
                if env_var_name not in os.environ:
                    missing_keys.append(env_var_name)

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

- Generic single-key matching (`key`, `token`, `apikey`, `accesstoken`) is deliberately permissive: a value passed to an unrelated `token=` parameter could satisfy a differently-named single-key requirement. This loosens validation slightly but matches the issue's intent (keys passed directly to `__init__` must be honored) and doesn't affect the test.
- `api_keys_required(["A", "B"])` (list of two plain strings) would be interpreted as `("A", env="B")` rather than two keys; the previous code crashed on that form, so it's not a regression.
- The test harness itself is lenient (any non-`ValueError` exception counts as a pass for `fixed`); however, tracing shows the fixed code raises nothing at all here.

## Verdict

**ACCEPT** — the fix correctly removes the spurious `ValueError` for API keys passed to a decorated `__init__` (the exact case in the test), preserves the "still raise when the key is genuinely absent" behavior, uses `inspect` APIs correctly with proper exception guards, and introduces no interface incompatibility with existing `api_keys_required` call sites.

def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()
            if task_name.strip() and task_id.isnumeric():
                new_tasks_list.append(task_name)
    return new_tasks_list


def check_server_running(server_url: str) -> bool:
    r"""Check whether the port refered by the URL to the server
    is open.

    Args:
        server_url (str): The URL to the server running LLM inference
            service.

    Returns:
        bool: Whether the port is open for packets (server is running).
    """
    parsed_url = urlparse(server_url)
    url_tuple = (parsed_url.hostname, parsed_url.port)

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required module dependencies.

    Raises:
        ImportError: If any of the required modules are not available.

    Example:
        ::

            def data_processing_function():
                # Function implementation...

        @wraps(func)

            missing_modules = [
                m for m in required_modules if not is_module_available(m)
            ]
            if missing_modules:
                raise ImportError(
                    f"Missing required modules: {', '.join(missing_modules)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

# Names that are commonly used for an API key argument/attribute, no matter
# which service the key belongs to.
_GENERIC_API_KEY_NAMES: Set[str] = {
    'apikey',
    'apikeys',
    'key',
    'keys',
    'token',
    'accesstoken',
}


def _normalize_api_key_name(name: str) -> str:
    r"""Normalize the name of an API key so that differently formatted names
    of the same key can be compared.

    Args:
        name (str): The name of the API key, e.g. :obj:`'OPENAI_API_KEY'`.

    Returns:
        str: The normalized name, e.g. :obj:`'openaiapikey'`.
    """
    return re.sub(r'[^a-z0-9]', '', str(name).lower())


def _api_key_aliases(key: str) -> Set[str]:
    r"""Return the normalized names under which an API key may be provided.

    For example :obj:`'OPENAI_API_KEY'` produces
    :obj:`{'openaiapikey', 'openai'}`, so that values passed as
    :obj:`openai_api_key` or :obj:`openai` are recognized.

    Args:
        key (str): The name of the API key.

    Returns:
        Set[str]: The normalized aliases of the key name.
    """
    normalized = _normalize_api_key_name(key)
    aliases: Set[str] = set()
    if normalized:
        aliases.add(normalized)
    for suffix in ('apikeys', 'apikey', 'key', 'token'):
        if normalized.endswith(suffix) and len(normalized) > len(suffix):
            aliases.add(normalized[: -len(suffix)])
    return aliases


def _api_key_attribute_candidates(key: str) -> Set[str]:
    r"""Return the instance attribute names that may hold an API key.

    Args:
        key (str): The name of the API key.

    Returns:
        Set[str]: The candidate attribute names.
    """
    candidates: Set[str] = {
        'api_key',
        '_api_key',
        'api_keys',
        '_api_keys',
        key,
        key.lower(),
        f'_{key.lower()}',
    }
    for alias in _api_key_aliases(key):
        candidates.update(
            {
                alias,
                f'_{alias}',
                f'{alias}_api_key',
                f'_{alias}_api_key',
                f'{alias}_key',
                f'_{alias}_key',
            }
        )
    return {c for c in candidates if c}


def _safe_getattr(obj: Any, name: str) -> Any:
    r"""Get an attribute of an object without raising any error.

    Args:
        obj (Any): The object to read the attribute from.
        name (str): The name of the attribute.

    Returns:
        Any: The value of the attribute, or :obj:`None` if it is not
            available.
    """
    try:
        return getattr(obj, name, None)
    except Exception:
        return None


def _is_api_key_provided(
    key_name: str,
    env_var: Optional[str],
    arguments: Mapping[str, Any],
    instance: Any,
    key_mappings: List[Mapping],
    single_key_required: bool,
) -> bool:
    r"""Check whether a required API key has been provided.

    The key is looked up, in order, in the arguments passed to the decorated
    function (this is what makes keys passed directly to a decorated
    ``__init__`` recognized), in the collected API key mappings, in the
    attributes of the instance and finally in the environment variables.

    Args:
        key_name (str): The name of the required API key.
        env_var (Optional[str]): The name of the environment variable holding
            the key, :obj:`None` if it should not be checked.
        arguments (Mapping[str, Any]): The arguments bound to the decorated
            function.
        instance (Any): The instance the decorated method is called on, if
            any.
        key_mappings (List[Mapping]): Mappings of API key names to values.
        single_key_required (bool): Whether the decorated function only
            requires a single API key.

    Returns:
        bool: Whether the API key has been provided.
    """
    aliases = _api_key_aliases(key_name)

    def _matches(name: str) -> bool:
        normalized = _normalize_api_key_name(name)
        if not normalized:
            return False
        if normalized in aliases:
            return True
        # A single required key may be passed through a generic argument or
        # attribute, e.g. ``__init__(self, api_key=...)``.
        return single_key_required and normalized in _GENERIC_API_KEY_NAMES

    # 1. Values passed directly to the decorated function. For a decorated
    #    ``__init__`` method, no instance attribute exists yet, hence the
    #    arguments have to be inspected.
    for arg_name, arg_value in arguments.items():
        if str(arg_name) in ('self', 'cls'):
            continue
        if not arg_value or not isinstance(arg_value, (str, bytes)):
            continue
        if _matches(str(arg_name)):
            return True

    # 2. Values collected in mappings such as
    #    ``api_keys={'OPENAI_API_KEY': '...'}``.
    for mapping in key_mappings:
        for map_key, map_value in mapping.items():
            if map_value and _matches(str(map_key)):
                return True

    # 3. Values stored as attributes of the instance.
    if instance is not None:
        candidates = _api_key_attribute_candidates(key_name)
        try:
            for attr_name in vars(instance):
                candidates.add(str(attr_name))
                candidates.add(str(attr_name).lstrip('_'))
        except TypeError:
            pass
        for attr_name in candidates:
            if not _matches(attr_name):
                continue
            value = _safe_getattr(instance, attr_name)
            if value and isinstance(value, (str, bytes)):
                return True

    # 4. Values provided through the environment variables.
    if env_var is not None:
        if env_var in os.environ or os.getenv(env_var.upper()):
            return True

    return False


def api_keys_required(*required_keys: Any) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are provided.

    Each key is looked up in the arguments passed to the decorated function
    (so that API keys passed directly to a decorated :obj:`__init__` method
    are recognized), in the attributes of the instance, and in the
    environment variables.

    Args:
        required_keys (str): The required API keys to be checked. A tuple of
            :obj:`(name, environment variable)` can also be provided when the
            name of the key and the name of the environment variable differ.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            arguments of the function, the instance attributes and the
            environment variables.

    Example:
        ::

            @api_keys_required('API_KEY_1', 'API_KEY_2')
            def some_api_function():
                # Function implementation...
    """

    def _parse_required_keys() -> List[Tuple[str, Optional[str]]]:
        r"""Normalize the required keys into (name, env variable) pairs."""
        parsed: List[Tuple[str, Optional[str]]] = []

        def _add(item: Any) -> None:
            if isinstance(item, str):
                parsed.append((item, item))
            elif isinstance(item, (list, tuple)):
                if item and isinstance(item[0], str):
                    env_var = item[1] if len(item) > 1 else item[0]
                    parsed.append(
                        (
                            item[0],
                            None if env_var is None else str(env_var),
                        )
                    )
                else:
                    for sub_item in item:
                        _add(sub_item)
            else:
                parsed.append((str(item), str(item)))

        for required_key in required_keys:
            _add(required_key)
        return parsed

    keys_to_check = _parse_required_keys()

    def decorator(func: F) -> F:
        try:
            signature: Optional[inspect.Signature] = inspect.signature(func)
        except (TypeError, ValueError):
            signature = None

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            # Bind the arguments so that keys passed positionally as well as
            # by keyword are detected, which is required for decorated
            # ``__init__`` methods.
            arguments: Dict[str, Any] = dict(kwargs)
            if signature is not None:
                try:
                    bound_args = signature.bind(*args, **kwargs)
                    bound_args.apply_defaults()
                    arguments = dict(bound_args.arguments)
                except TypeError:
                    pass

            instance = args[0] if args else arguments.get('self')

            # API keys may also be supplied as a mapping, e.g. through an
            # ``api_keys`` argument or an ``api_keys`` instance attribute.
            key_mappings: List[Mapping] = []
            for source in (
                kwargs.get('api_keys'),
                arguments.get('api_keys'),
                _safe_getattr(instance, 'api_keys'),
                _safe_getattr(instance, '_api_keys'),
            ):
                if isinstance(source, Mapping):
                    key_mappings.append(source)

            single_key_required = len(keys_to_check) == 1
            missing_keys: List[str] = []

            for key_name, env_var in keys_to_check:
                if _is_api_key_provided(
                    key_name,
                    env_var,
                    arguments,
                    instance,
                    key_mappings,
                    single_key_required,
                ):
                    continue
                if env_var in (None, key_name):
                    missing_keys.append(key_name)
                else:
                    missing_keys.append(
                        f"{key_name} (or environment variable: {env_var})"
                    )

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper


class AgentOpsMeta(type):
    r"""Metaclass that automatically decorates all callable attributes with
    the agentops_decorator,
    except for the 'get_tools' method.

    Methods:

        Creates a new class with decorated methods.
    """

    def __new__(cls, name, bases, dct):
        if ToolEvent:
            for attr, value in dct.items():
                if callable(value) and attr != 'get_tools':
                    dct[attr] = agentops_decorator(value)
        return super().__new__(cls, name, bases, dct)

def handle_http_error(response: requests.Response) -> str:
    r"""Handles the HTTP errors based on the status code of the response.

    Args:
        response (requests.Response): The HTTP response from the API call.

    Returns:
        str: The error type, based on the status code.
    """
    if response.status_code == HTTPStatus.UNAUTHORIZED:
        return "Unauthorized. Check your access token."
    elif response.status_code == HTTPStatus.FORBIDDEN:
        return "Forbidden. You do not have permission to perform this action."
    elif response.status_code == HTTPStatus.NOT_FOUND:
        return "Not Found. The resource could not be located."
    elif response.status_code == HTTPStatus.TOO_MANY_REQUESTS:
        return "Too Many Requests. You have hit the rate limit."
    else:
        return "HTTP Error"


def retry_request(
    func: Callable, retries: int = 3, delay: int = 1, *args: Any, **kwargs: Any
) -> Any:
    r"""Retries a function in case of any errors.

    Args:
        func (Callable): The function to be retried.
        retries (int): Number of retry attempts. (default: :obj:`3`)
        delay (int): Delay between retries in seconds. (default: :obj:`1`)
        *args: Arguments to pass to the function.
        **kwargs: Keyword arguments to pass to the function.

    Returns:
        Any: The result of the function call if successful.

    Raises:
        Exception: If all retry attempts fail.
    """
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            print(f"Attempt {attempt + 1}/{retries} failed: {e}")
            if attempt < retries - 1:
                time.sleep(delay)
            else:
                raise
