def print_text_animated(text, delay: float = 0.02, end: str = ""):
    r"""Prints the given text with an animated effect.

    Args:
        text (str): The text to print.
        delay (float, optional): The delay between each character printed.
            (default: :obj:`0.02`)
        end (str, optional): The end character to print after each
            character of text. (default: :obj:`""`)

        time.sleep(delay)
    print('\n')


def get_prompt_template_key_words(template: str) -> Set[str]:
    r"""Given a string template containing curly braces {}, return a set of
    the words inside the braces.

    Args:
        template (str): A string containing curly braces.

    Returns:
        List[str]: A list of the words inside the curly braces.

    Example:
        >>> get_prompt_template_key_words('Hi, {name}! How are you {status}?')

def get_first_int(string: str) -> Optional[int]:
    r"""Returns the first integer number found in the given string.

    If no integer number is found, returns None.

    Args:
        string (str): The input string.

    Returns:
        int or None: The first integer number found in the string, or None if
            no integer number is found.
    """
    match = re.search(r'\d+', string)
    if match:
        return int(match.group())
    else:
        return None


def download_tasks(task: TaskType, folder_path: str) -> None:
    r"""Downloads task-related files from a specified URL and extracts them.

    This function downloads a zip file containing tasks based on the specified
    `task` type from a predefined URL, saves it to `folder_path`, and then
    extracts the contents of the zip file into the same folder. After
    extraction, the zip file is deleted.

    Args:
        task (TaskType): An enum representing the type of task to download.
        folder_path (str): The path of the folder where the zip file will be
                           downloaded and extracted.
    """
    # Define the path to save the zip file
    zip_file_path = os.path.join(folder_path, "tasks.zip")

    # Download the zip file from the Google Drive link
    response = requests.get(
        "https://huggingface.co/datasets/camel-ai/"
        f"metadata/resolve/main/{task.value}_tasks.zip"
    )

    # Save the zip file
    with open(zip_file_path, "wb") as f:
        f.write(response.content)

    with zipfile.ZipFile(zip_file_path, "r") as zip_ref:
        zip_ref.extractall(folder_path)

    # Delete the zip file
    os.remove(zip_file_path)


def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()
            if task_name.strip() and task_id.isnumeric():
                new_tasks_list.append(task_name)
    return new_tasks_list


def check_server_running(server_url: str) -> bool:
    r"""Check whether the port refered by the URL to the server
    is open.

    Args:
        server_url (str): The URL to the server running LLM inference
            service.

    Returns:
        bool: Whether the port is open for packets (server is running).
    """
    parsed_url = urlparse(server_url)
    url_tuple = (parsed_url.hostname, parsed_url.port)

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    result = sock.connect_ex(url_tuple)
    sock.close()

    # if the port is open, the result should be 0.
    return result == 0


def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required module dependencies.

    Raises:
        ImportError: If any of the required modules are not available.

    Example:
        ::

            @dependencies_required('numpy', 'pandas')
            def data_processing_function():
                # Function implementation...
    """

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            missing_modules = [
                m for m in required_modules if not is_module_available(m)
            ]
            if missing_modules:
                raise ImportError(
                    f"Missing required modules: {', '.join(missing_modules)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def api_keys_required(
    *required_keys: Union[
        str,
        Tuple[Optional[str], Optional[str]],
        List[Union[str, Tuple[Optional[str], Optional[str]]]],
    ],
) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are presented in the
    environment variables, are passed directly as function arguments, or are
    set as attributes of the decorated object.

    Args:
        required_keys (Union[str, Tuple[Optional[str], Optional[str]]]): The
            required API keys to be checked. Each element can be one of:

            - A string, i.e. the name of the environment variable which may
              hold the API key (legacy usage).
            - A tuple ``(env_key, arg_name)``, where ``env_key`` is the name of
              the environment variable and ``arg_name`` is the name of the
              argument (or of the attribute of the decorated object) through
              which the very same API key can be provided. Either element of
              the tuple can be :obj:`None`, in which case that source is
              ignored.
            - A list of any of the above, which is convenient when several API
              keys are required.

            Each key is checked independently, so requiring multiple API keys
            enforces that every one of them is provided by *some* supported
            source.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables, the function arguments, and the instance
            attributes.

    Example:
        ::

            @api_keys_required(
                [
                    ('API_KEY_1', None),
                    (None, 'api_key_2'),
                ]
            )
            def some_api_function(api_key_2: Optional[str] = None):
                # Function implementation...

            # The legacy style is still supported:
            @api_keys_required('API_KEY_1', 'API_KEY_2')
            def another_api_function():
                # Function implementation...
    """
    # Normalize the given keys into a list of `(env_key, arg_name)` tuples so
    # that both the legacy style, e.g. `('KEY_1', 'KEY_2')`, and the extended
    # style, e.g. `([('KEY_1', 'arg_1'), (None, 'arg_2')])`, are supported.
    key_specs: List[Tuple[Optional[str], Optional[str]]] = []

    def _add_key_spec(spec: Any) -> None:
        if spec is None:
            return
        if isinstance(spec, str):
            # Legacy style: only the environment variable is named, but the
            # key may still be passed through the `api_key` argument or the
            # corresponding instance attribute.
            key_specs.append((spec, 'api_key'))
        elif isinstance(spec, tuple):
            if len(spec) == 0:
                return
            if len(spec) == 2 and all(
                item is None or isinstance(item, str) for item in spec
            ):
                env_key, arg_name = spec
                if env_key is None and arg_name is None:
                    return
                key_specs.append((env_key, arg_name))
            else:
                for item in spec:
                    _add_key_spec(item)
        elif isinstance(spec, list):
            for item in spec:
                _add_key_spec(item)
        else:
            raise TypeError(
                "The required API keys must be given as strings, "
                f"`(env_key, arg_name)` tuples or lists of them, got: {spec!r}"
            )

    for required_key in required_keys:
        _add_key_spec(required_key)

    def decorator(func: F) -> F:
        try:
            func_signature: Optional[inspect.Signature] = inspect.signature(
                func
            )
        except (TypeError, ValueError):  # pragma: no cover
            func_signature = None

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            bound_arguments: Mapping[str, Any] = dict(kwargs)
            if func_signature is not None:
                try:
                    bound_arguments = func_signature.bind_partial(
                        *args, **kwargs
                    ).arguments
                except TypeError:
                    bound_arguments = dict(kwargs)

            instance = args[0] if args else None

            missing_keys: List[str] = []
            for env_key, arg_name in key_specs:
                if env_key and os.environ.get(env_key):
                    continue
                if arg_name:
                    # The key may be passed directly as an argument of the
                    # decorated function (either by keyword or by position).
                    if bound_arguments.get(arg_name):
                        continue
                    # Or it may have been stored on the instance already, e.g.
                    # by a previously run `__init__` method.
                    if instance is not None:
                        value = getattr(instance, arg_name, None)
                        if not value:
                            value = getattr(
                                instance, f'_{arg_name}', None
                            )
                        if value:
                            continue
                missing_keys.append(env_key or cast(str, arg_name))

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code


def text_extract_from_web(url: str) -> str:

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper


class AgentOpsMeta(type):
    r"""Metaclass that automatically decorates all callable attributes with
    the agentops_decorator,
    except for the 'get_tools' method.

    Methods:
    __new__(cls, name, bases, dct):
        Creates a new class with decorated methods.
    """

    def __new__(cls, name, bases, dct):
        if ToolEvent:
            for attr, value in dct.items():
                if callable(value) and attr != 'get_tools':
                    dct[attr] = agentops_decorator(value)
        return super().__new__(cls, name, bases, dct)

def handle_http_error(response: requests.Response) -> str:
    r"""Handles the HTTP errors based on the status code of the response.

    Args:
        response (requests.Response): The HTTP response from the API call.

    Returns:
        str: The error type, based on the status code.
    """
    if response.status_code == HTTPStatus.UNAUTHORIZED:
        return "Unauthorized. Check your access token."
    elif response.status_code == HTTPStatus.FORBIDDEN:
        return "Forbidden. You do not have permission to perform this action."
    elif response.status_code == HTTPStatus.NOT_FOUND:
        return "Not Found. The resource could not be located."
    elif response.status_code == HTTPStatus.TOO_MANY_REQUESTS:
        return "Too Many Requests. You have hit the rate limit."
    else:
        return "HTTP Error"


def retry_request(
    func: Callable, retries: int = 3, delay: int = 1, *args: Any, **kwargs: Any
) -> Any:
    r"""Retries a function in case of any errors.

    Args:
        func (Callable): The function to be retried.
        retries (int): Number of retry attempts. (default: :obj:`3`)
        delay (int): Delay between retries in seconds. (default: :obj:`1`)
        *args: Arguments to pass to the function.
        **kwargs: Keyword arguments to pass to the function.

    Returns:
        Any: The result of the function call if successful.

    Raises:
        Exception: If all retry attempts fail.
    """
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            print(f"Attempt {attempt + 1}/{retries} failed: {e}")
            if attempt < retries - 1:
                time.sleep(delay)
            else:
                raise

def print_text_animated(text, delay: float = 0.02, end: str = ""):
    r"""Prints the given text with an animated effect.

    Args:
        text (str): The text to print.
        delay (float, optional): The delay between each character printed.
            (default: :obj:`0.02`)
        end (str, optional): The end character to print after each
            character of text. (default: :obj:`""`)
    """
    for char in text:
        print(char, end=end, flush=True)
        time.sleep(delay)
    print('\n')


def get_prompt_template_key_words(template: str) -> Set[str]:
    r"""Given a string template containing curly braces {}, return a set of
    the words inside the braces.

    Args:
        template (str): A string containing curly braces.

    Returns:
        List[str]: A list of the words inside the curly braces.

    Example:
        >>> get_prompt_template_key_words('Hi, {name}! How are you {status}?')
        {'name', 'status'}
    """
    return set(re.findall(r'{([^}]*)}', template))


def get_first_int(string: str) -> Optional[int]:
    r"""Returns the first integer number found in the given string.

    If no integer number is found, returns None.

    Args:
        string (str): The input string.

    Returns:
        int or None: The first integer number found in the string, or None if
            no integer number is found.
    """
    match = re.search(r'\d+', string)
    if match:
        return int(match.group())
    else:
        return None


def download_tasks(task: TaskType, folder_path: str) -> None:
    r"""Downloads task-related files from a specified URL and extracts them.

    This function downloads a zip file containing tasks based on the specified
    `task` type from a predefined URL, saves it to `folder_path`, and then
    extracts the contents of the zip file into the same folder. After
    extraction, the zip file is deleted.

    Args:
        task (TaskType): An enum representing the type of task to download.
        folder_path (str): The path of the folder where the zip file will be
                           downloaded and extracted.
    """
    # Define the path to save the zip file
    zip_file_path = os.path.join(folder_path, "tasks.zip")

    # Download the zip file from the Google Drive link
    response = requests.get(
        "https://huggingface.co/datasets/camel-ai/"
        f"metadata/resolve/main/{task.value}_tasks.zip"
    )

    # Save the zip file
    with open(zip_file_path, "wb") as f:
        f.write(response.content)

    with zipfile.ZipFile(zip_file_path, "r") as zip_ref:
        zip_ref.extractall(folder_path)

    # Delete the zip file
    os.remove(zip_file_path)


def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()
            if task_name.strip() and task_id.isnumeric():
                new_tasks_list.append(task_name)
    return new_tasks_list


def check_server_running(server_url: str) -> bool:
    r"""Check whether the port refered by the URL to the server
    is open.

    Args:
        server_url (str): The URL to the server running LLM inference
            service.

    Returns:
        bool: Whether the port is open for packets (server is running).
    """
    parsed_url = urlparse(server_url)
    url_tuple = (parsed_url.hostname, parsed_url.port)

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    result = sock.connect_ex(url_tuple)
    sock.close()

    # if the port is open, the result should be 0.
    return result == 0


def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required module dependencies.

    Raises:
        ImportError: If any of the required modules are not available.

    Example:
        ::

            @dependencies_required('numpy', 'pandas')
            def data_processing_function():
                # Function implementation...
    """

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            missing_modules = [
                m for m in required_modules if not is_module_available(m)
            ]
            if missing_modules:
                raise ImportError(
                    f"Missing required modules: {', '.join(missing_modules)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def _is_env_var_name(name: Any) -> bool:
    r"""Check whether the given name looks like an environment variable name,
    i.e. whether it is an upper-cased string such as :obj:`"OPENAI_API_KEY"`.

    Args:
        name (Any): The name to be checked.

    Returns:
        bool: Whether the name looks like an environment variable name.
    """
    if not isinstance(name, str):
        return False
    stripped = name.strip()
    if not stripped:
        return False
    return stripped.upper() == stripped and stripped.lower() != stripped


def _parse_api_key_requirements(
    required_keys: Any,
) -> List[Tuple[Optional[str], Optional[str]]]:
    r"""Normalize the arguments given to :obj:`api_keys_required` into a list
    of :obj:`(environment variable name, instance attribute name)` tuples.

    Both the old style (plain environment variable names) and the new style
    (tuples of environment variable name and instance attribute name) are
    supported, e.g. all the following are valid::

        @api_keys_required('OPENAI_API_KEY', 'ANTHROPIC_API_KEY')
        @api_keys_required(['OPENAI_API_KEY', 'ANTHROPIC_API_KEY'])
        @api_keys_required([('OPENAI_API_KEY', 'api_key')])
        @api_keys_required(('OPENAI_API_KEY', 'api_key'))

    Args:
        required_keys (Any): The raw arguments given to the decorator.

    Returns:
        List[Tuple[Optional[str], Optional[str]]]: The normalized
            requirements.
    """
    requirements: List[Tuple[Optional[str], Optional[str]]] = []

    def add(entry: Any) -> None:
        if entry is None:
            return
        if isinstance(entry, str):
            requirements.append((entry, None))
            return
        if isinstance(entry, (list, tuple)):
            entries = list(entry)
            if not entries:
                return
            # A single ``(environment variable, attribute)`` pair, e.g.
            # ``('OPENAI_API_KEY', 'api_key')`` or
            # ``('OPENAI_API_KEY', None)``.
            if (
                len(entries) == 2
                and isinstance(entries[0], str)
                and _is_env_var_name(entries[0])
                and (entries[1] is None or not _is_env_var_name(entries[1]))
            ):
                requirements.append((entries[0], entries[1]))
                return
            # A sequence of plain environment variable names, e.g.
            # ``['OPENAI_API_KEY', 'ANTHROPIC_API_KEY']``.
            if all(isinstance(e, str) for e in entries):
                for e in entries:
                    requirements.append((e, None))
                return
            # A nested sequence of requirements.
            for e in entries:
                add(e)
            return
        requirements.append((None, str(entry)))

    add(required_keys)
    return requirements


def _api_key_candidate_names(
    env_var: Optional[str], attr_name: Optional[str]
) -> Set[str]:
    r"""Build the set of names under which an API key may be found, either as
    an instance attribute or as a keyword argument.

    Args:
        env_var (Optional[str]): The name of the environment variable holding
            the API key.
        attr_name (Optional[str]): The name of the instance attribute holding
            the API key.

    Returns:
        Set[str]: All the candidate names for the API key.
    """
    candidates: Set[str] = set()
    for name in (attr_name, env_var):
        if not isinstance(name, str):
            continue
        stripped = name.strip().lstrip('_')
        if not stripped:
            continue
        lowered = stripped.lower()
        candidates.add(name)
        candidates.add(stripped)
        candidates.add(lowered)
        candidates.add('_' + stripped)
        candidates.add('_' + lowered)
        parts = [part for part in lowered.split('_') if part]
        # e.g. "OPENAI_API_KEY" -> "api_key" / "_api_key"
        if len(parts) >= 2:
            candidates.add('_'.join(parts[-2:]))
            candidates.add('_' + '_'.join(parts[-2:]))
    return candidates


def _api_key_is_provided(
    env_var: Optional[str],
    attr_name: Optional[str],
    instance: Any,
    provided_values: Mapping[str, Any],
) -> bool:
    r"""Check whether a single required API key is provided either as an
    instance attribute or as an argument passed to the decorated callable.

    Args:
        env_var (Optional[str]): The name of the environment variable holding
            the API key.
        attr_name (Optional[str]): The name of the instance attribute holding
            the API key.
        instance (Any): The instance the decorated callable is bound to, if
            any.
        provided_values (Mapping[str, Any]): The arguments (values keyed by
            their parameter names) passed to the decorated callable.

    Returns:
        bool: Whether the API key is provided.
    """
    candidates = _api_key_candidate_names(env_var, attr_name)
    if instance is not None:
        for name in candidates:
            try:
                value = getattr(instance, name, None)
            except Exception:
                value = None
            if value:
                return True
    for name, value in provided_values.items():
        if name in candidates and value:
            return True
    return False


def api_keys_required(*required_keys: Any, **decorator_kwargs: Any):
    r"""A decorator to check if the required API keys are presented in the
    environment variables, as an instance attribute, or as an argument
    passed directly to the decorated callable.

    This makes it possible to decorate an ``__init__`` method and pass the
    API key(s) directly to the constructor, e.g.
    ``MyClass(api_key="...")``, instead of relying on environment variables.

    Args:
        required_keys: The required API keys to be checked. Each entry can be
            either the name of an environment variable, e.g.
            :obj:`'OPENAI_API_KEY'`, or a tuple of the environment variable
            name and the corresponding instance attribute (or argument)
            name, e.g. :obj:`('OPENAI_API_KEY', 'api_key')`. The instance
            attribute name can be :obj:`None` if only the environment
            variable should be checked. They can be given either as separate
            positional arguments or as a single list/tuple of entries.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables, the instance attributes and the arguments
            passed to the decorated callable.

    Example:
        ::

            @api_keys_required(
                'API_KEY_1',
                ('API_KEY_2', 'api_key'),
            )
            def some_api_function():
                # Function implementation...
    """
    keyword_keys = decorator_kwargs.pop('required_keys', None)
    if decorator_kwargs:
        raise TypeError(
            "api_keys_required() got an unexpected keyword argument "
            f"'{next(iter(decorator_kwargs))}'"
        )

    raw_keys: List[Any] = list(required_keys)
    if keyword_keys is not None:
        if isinstance(keyword_keys, list):
            raw_keys.extend(keyword_keys)
        else:
            raw_keys.append(keyword_keys)

    requirements = _parse_api_key_requirements(raw_keys)

    def decorator(func: F) -> F:
        try:
            signature: Optional[inspect.Signature] = inspect.signature(func)
        except (TypeError, ValueError):
            signature = None

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            provided_values: Dict[str, Any] = dict(kwargs)
            if signature is not None:
                try:
                    bound = signature.bind_partial(*args, **kwargs)
                except TypeError:
                    bound = None
                if bound is not None:
                    provided_values = dict(bound.arguments)

            instance = args[0] if args else None

            missing_keys: List[str] = []
            for env_var, attr_name in requirements:
                if isinstance(env_var, str) and env_var in os.environ:
                    continue
                if _api_key_is_provided(
                    env_var, attr_name, instance, provided_values
                ):
                    continue
                if env_var is not None:
                    missing_keys.append(env_var)
                elif attr_name is not None:
                    missing_keys.append(attr_name)

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper


class AgentOpsMeta(type):

    Methods:
    __new__(cls, name, bases, dct):
        Creates a new class with decorated methods.
    """

    def __new__(cls, name, bases, dct):
        if ToolEvent:
            for attr, value in dct.items():
                if callable(value) and attr != 'get_tools':
                    dct[attr] = agentops_decorator(value)
        return super().__new__(cls, name, bases, dct)

def handle_http_error(response: requests.Response) -> str:
    r"""Handles the HTTP errors based on the status code of the response.

    Args:
        response (requests.Response): The HTTP response from the API call.

    Returns:
        str: The error type, based on the status code.
    """
    if response.status_code == HTTPStatus.UNAUTHORIZED:
        return "Unauthorized. Check your access token."
    elif response.status_code == HTTPStatus.FORBIDDEN:
        return "Forbidden. You do not have permission to perform this action."
    elif response.status_code == HTTPStatus.NOT_FOUND:
        return "Not Found. The resource could not be located."
    elif response.status_code == HTTPStatus.TOO_MANY_REQUESTS:
        return "Too Many Requests. You have hit the rate limit."
    else:
        return "HTTP Error"


def retry_request(
    func: Callable, retries: int = 3, delay: int = 1, *args: Any, **kwargs: Any
) -> Any:
    r"""Retries a function in case of any errors.

    Args:
        func (Callable): The function to be retried.
        retries (int): Number of retry attempts. (default: :obj:`3`)
        delay (int): Delay between retries in seconds. (default: :obj:`1`)
        *args: Arguments to pass to the function.
        **kwargs: Keyword arguments to pass to the function.

    Returns:
        Any: The result of the function call if successful.

    Raises:
        Exception: If all retry attempts fail.
    """
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            print(f"Attempt {attempt + 1}/{retries} failed: {e}")
            if attempt < retries - 1:
                time.sleep(delay)
            else:
                raise

def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()

def check_server_running(server_url: str) -> bool:

def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required module dependencies.

    Raises:
        ImportError: If any of the required modules are not available.

    Example:
        ::

            def data_processing_function():
                # Function implementation...
    """

                    f"Missing required modules: {', '.join(missing_modules)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def api_keys_required(required_keys) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are presented in the
    environment variables or as an argument passed to the decorated
    function.

    Both sources are supported at the same time, which is important for
    decorators applied on `__init__` methods: the instance attribute that
    stores the API key (for example `self._api_key`) is only created inside
    the body of the constructor, therefore the value has to be read from the
    arguments passed to the constructor (or from the environment variables)
    instead of from the (not yet existing) instance attribute.

    Args:
        required_keys (List[Optional[List[str]]]): A list containing lists of
            tuples. Each tuple represents a required API key, where the first
            element is the name of the argument that might contain the API key
            and the second element is the name of the environment variable
            that might contain the API key. If the name of the argument or the
            name of the environment variable is :obj:`None`, the corresponding
            check will be skipped. A plain string is also accepted and is
            interpreted as the name of an environment variable, which keeps
            the former behavior working.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables and the arguments.

    Example:
        ::

            @api_keys_required([
                ('api_key', 'OPENAI_API_KEY'),
                (None, 'ANTHROPIC_API_KEY'),
            ])
            def some_api_function(api_key: Optional[str] = None):
                # Function implementation...
    """

    def _normalize_required_keys(keys: Any) -> List[Any]:
        r"""Normalize the input into a list of
        `(argument name, environment variable name)` tuples.
        """
        if isinstance(keys, str):
            candidates: List[Any] = [keys]
        elif isinstance(keys, tuple) and len(keys) == 2 and all(
            item is None or isinstance(item, str) for item in keys
        ):
            # A single `(argument name, environment variable name)` pair.
            candidates = [keys]
        elif isinstance(keys, (list, tuple)):
            candidates = list(keys)
        else:
            raise ValueError(
                "The required API keys should be provided as a list of "
                f"tuples of (argument name, environment variable name), "
                f"got: {keys!r}."
            )

        normalized: List[Any] = []
        for candidate in candidates:
            if candidate is None:
                continue
            if isinstance(candidate, str):
                # Backward compatibility: only an environment variable.
                normalized.append((None, candidate))
            elif isinstance(candidate, (list, tuple)):
                if len(candidate) != 2:
                    raise ValueError(
                        "Each required API key should be provided as a tuple "
                        "of (argument name, environment variable name), "
                        f"got: {candidate!r}."
                    )
                arg_name, env_name = candidate
                normalized.append((arg_name, env_name))
            else:
                raise ValueError(
                    "Each required API key should be provided as a tuple of "
                    "(argument name, environment variable name), "
                    f"got: {candidate!r}."
                )
        return normalized

    api_key_specs = _normalize_required_keys(required_keys)

    def decorator(func: F) -> F:
        try:
            signature: Optional[inspect.Signature] = inspect.signature(func)
        except (TypeError, ValueError):  # pragma: no cover
            signature = None

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            arguments: Dict[str, Any] = {}
            if signature is not None:
                try:
                    bound_args = signature.bind(*args, **kwargs)
                    bound_args.apply_defaults()
                    arguments = dict(bound_args.arguments)
                except TypeError:
                    arguments = dict(kwargs)
            else:  # pragma: no cover
                arguments = dict(kwargs)

            missing_keys: List[str] = []
            for arg_name, env_name in api_key_specs:
                found = False
                if arg_name is not None:
                    # The key can be passed explicitly to the decorated
                    # function, e.g. `Model(..., api_key="...")`.
                    if arguments.get(arg_name):
                        found = True
                    elif args and getattr(args[0], arg_name, None):
                        # The key is already stored on the instance, e.g. when
                        # the decorator is applied on a regular method.
                        found = True
                    elif args and getattr(
                        args[0], f"_{arg_name}", None
                    ):
                        # Backward compatibility with instances storing the
                        # key in a private attribute, e.g. `self._api_key`.
                        found = True
                if not found and env_name is not None:
                    if os.environ.get(env_name):
                        found = True
                if not found:
                    if arg_name is not None and env_name is not None:
                        missing_keys.append(
                            f"'{arg_name}' (argument) or '{env_name}' "
                            "(environment variable)"
                        )
                    elif arg_name is not None:
                        missing_keys.append(f"'{arg_name}' (argument)")
                    else:
                        missing_keys.append(f"'{env_name}' (environment variable)")

            if missing_keys:
                raise ValueError(f"Missing API keys: {', '.join(missing_keys)}")

            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def retry_request(

    r"""Retries a function in case of any errors.

    Args:

                raise

def print_text_animated(text, delay: float = 0.02, end: str = ""):
    r"""Prints the given text with an animated effect.

    Args:
        text (str): The text to print.
        delay (float, optional): The delay between each character printed.
            (default: :obj:`0.02`)
        end (str, optional): The end character to print after each
            character of text. (default: :obj:`""`)
    """
    for char in text:
        print(char, end=end, flush=True)
        time.sleep(delay)
    print('\n')


def get_prompt_template_key_words(template: str) -> Set[str]:
    r"""Given a string template containing curly braces {}, return a set of
    the words inside the braces.

    Args:
        template (str): A string containing curly braces.

    Returns:
        List[str]: A list of the words inside the curly braces.

    Example:
        >>> get_prompt_template_key_words('Hi, {name}! How are you {status}?')
        {'name', 'status'}
    """
    return set(re.findall(r'{([^}]*)}', template))


def get_first_int(string: str) -> Optional[int]:
    r"""Returns the first integer number found in the given string.

    If no integer number is found, returns None.

    Args:
        string (str): The input string.

    Returns:
        int or None: The first integer number found in the string, or None if
            no integer number is found.
    """
    match = re.search(r'\d+', string)
    if match:
        return int(match.group())
    else:
        return None


def download_tasks(task: TaskType, folder_path: str) -> None:
    r"""Downloads task-related files from a specified URL and extracts them.

    This function downloads a zip file containing tasks based on the specified
    `task` type from a predefined URL, saves it to `folder_path`, and then
    extracts the contents of the zip file into the same folder. After
    extraction, the zip file is deleted.

    Args:
        task (TaskType): An enum representing the type of task to download.
        folder_path (str): The path of the folder where the zip file will be
                           downloaded and extracted.
    """
    # Define the path to save the zip file
    zip_file_path = os.path.join(folder_path, "tasks.zip")

    # Download the zip file from the Google Drive link
    response = requests.get(
        "https://huggingface.co/datasets/camel-ai/"
        f"metadata/resolve/main/{task.value}_tasks.zip"
    )

    # Save the zip file
    with open(zip_file_path, "wb") as f:
        f.write(response.content)

    with zipfile.ZipFile(zip_file_path, "r") as zip_ref:
        zip_ref.extractall(folder_path)

    # Delete the zip file
    os.remove(zip_file_path)


def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()
            if task_name.strip() and task_id.isnumeric():
                new_tasks_list.append(task_name)
    return new_tasks_list


def check_server_running(server_url: str) -> bool:
    r"""Check whether the port refered by the URL to the server
    is open.

    Args:
        server_url (str): The URL to the server running LLM inference
            service.

    Returns:
        bool: Whether the port is open for packets (server is running).
    """
    parsed_url = urlparse(server_url)
    url_tuple = (parsed_url.hostname, parsed_url.port)

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    result = sock.connect_ex(url_tuple)
    sock.close()

    # if the port is open, the result should be 0.
    return result == 0


def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required module dependencies.

    Raises:
        ImportError: If any of the required modules are not available.

    Example:
        ::

            @dependencies_required('numpy', 'pandas')
            def data_processing_function():
                # Function implementation...
    """

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            missing_modules = [
                m for m in required_modules if not is_module_available(m)
            ]
            if missing_modules:
                raise ImportError(
                    f"Missing required modules: {', '.join(missing_modules)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator


def is_module_available(module_name: str) -> bool:
    r"""Check if a module is available for import.

    Args:
        module_name (str): The name of the module to check for availability.

    Returns:
        bool: True if the module can be imported, False otherwise.
    """
    try:
        importlib.import_module(module_name)
        return True
    except ImportError:
        return False


def api_keys_required(
    *required_keys: Union[
        str,
        Tuple[Optional[str], Optional[str]],
        List[Union[str, Tuple[Optional[str], Optional[str]]]],
    ],
) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are presented in the
    environment variables, are passed directly as function arguments, or are
    set as attributes of the decorated object.

    Args:
        required_keys (Union[str, Tuple[Optional[str], Optional[str]]]): The
            required API keys to be checked. Each element can be one of:

            - A string, i.e. the name of the environment variable which may
              hold the API key (legacy usage).
            - A tuple ``(env_key, arg_name)``, where ``env_key`` is the name of
              the environment variable and ``arg_name`` is the name of the
              argument (or of the attribute of the decorated object) through
              which the very same API key can be provided. Either element of
              the tuple can be :obj:`None`, in which case that source is
              ignored.
            - A list of any of the above, which is convenient when several API
              keys are required.

            Each key is checked independently, so requiring multiple API keys
            enforces that every one of them is provided by *some* supported
            source.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables, the function arguments, and the instance
            attributes.

    Example:
        ::

            @api_keys_required(
                [
                    ('API_KEY_1', None),
                    (None, 'api_key_2'),
                ]
            )
            def some_api_function(api_key_2: Optional[str] = None):
                # Function implementation...

            # The legacy style is still supported:
            @api_keys_required('API_KEY_1', 'API_KEY_2')
            def another_api_function():
                # Function implementation...
    """
    # Normalize the given keys into a list of `(env_key, arg_name)` tuples so
    # that both the legacy style, e.g. `('KEY_1', 'KEY_2')`, and the extended
    # style, e.g. `([('KEY_1', 'arg_1'), (None, 'arg_2')])`, are supported.
    key_specs: List[Tuple[Optional[str], Optional[str]]] = []

    def _add_key_spec(spec: Any) -> None:
        if spec is None:
            return
        if isinstance(spec, str):
            # Legacy style: only the environment variable is named, but the
            # key may still be passed through the `api_key` argument or the
            # corresponding instance attribute.
            key_specs.append((spec, 'api_key'))
        elif isinstance(spec, tuple):
            if len(spec) == 0:
                return
            if len(spec) == 2 and all(
                item is None or isinstance(item, str) for item in spec
            ):
                env_key, arg_name = spec
                if env_key is None and arg_name is None:
                    return
                key_specs.append((env_key, arg_name))
            else:
                for item in spec:
                    _add_key_spec(item)
        elif isinstance(spec, list):
            for item in spec:
                _add_key_spec(item)
        else:
            raise TypeError(
                "The required API keys must be given as strings, "
                f"`(env_key, arg_name)` tuples or lists of them, got: {spec!r}"
            )

    for required_key in required_keys:
        _add_key_spec(required_key)

    def decorator(func: F) -> F:
        try:
            func_signature: Optional[inspect.Signature] = inspect.signature(
                func
            )
        except (TypeError, ValueError):  # pragma: no cover
            func_signature = None

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            bound_arguments: Mapping[str, Any] = dict(kwargs)
            if func_signature is not None:
                try:
                    bound_arguments = func_signature.bind_partial(
                        *args, **kwargs
                    ).arguments
                except TypeError:
                    bound_arguments = dict(kwargs)

            instance = args[0] if args else None

            missing_keys: List[str] = []
            for env_key, arg_name in key_specs:
                if env_key and os.environ.get(env_key):
                    continue
                if arg_name:
                    # The key may be passed directly as an argument of the
                    # decorated function (either by keyword or by position).
                    if bound_arguments.get(arg_name):
                        continue
                    # Or it may have been stored on the instance already, e.g.
                    # by a previously run `__init__` method.
                    if instance is not None:
                        value = getattr(instance, arg_name, None)
                        if not value:
                            value = getattr(
                                instance, f'_{arg_name}', None
                            )
                        if value:
                            continue
                missing_keys.append(env_key or cast(str, arg_name))

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator


def get_system_information():
    r"""Gathers information about the operating system.

    Returns:
        dict: A dictionary containing various pieces of OS information.

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def func_string_to_callable(code: str):
    r"""Convert a function code string to a callable function object.

    Args:
        code (str): The function code as a string.

    Returns:
        Callable[..., Any]: The callable function object extracted from the
            code string.
    """
    local_vars: Mapping[str, object] = {}
    exec(code, globals(), local_vars)
    func = local_vars.get(Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT)
    return func


def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code


def text_extract_from_web(url: str) -> str:
    r"""Get the text information from given url.

    Args:
        url (str): The website you want to search.

    Returns:
        str: All texts extract from the web.
    """
    try:
        import requests
        from newspaper import Article

        # Request the target page
        article = Article(url)
        article.download()
        article.parse()
        text = article.text

    except requests.RequestException as e:
        text = f"Can't access {url}, error: {e}"

    except Exception as e:
        text = f"Can't extract text from {url}, error: {e}"

    return text


def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks


def is_docker_running() -> bool:
    r"""Check if the Docker daemon is running.

    Returns:
        bool: True if the Docker daemon is running, False otherwise.
    """
    try:
        result = subprocess.run(
            ["docker", "info"],
            check=True,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
        )
        return result.returncode == 0
    except (subprocess.CalledProcessError, FileNotFoundError):
        return False

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper


class AgentOpsMeta(type):
    r"""Metaclass that automatically decorates all callable attributes with
    the agentops_decorator,
    except for the 'get_tools' method.

    Methods:
    __new__(cls, name, bases, dct):
        Creates a new class with decorated methods.
    """

    def __new__(cls, name, bases, dct):
        if ToolEvent:
            for attr, value in dct.items():
                if callable(value) and attr != 'get_tools':
                    dct[attr] = agentops_decorator(value)
        return super().__new__(cls, name, bases, dct)

def handle_http_error(response: requests.Response) -> str:
    r"""Handles the HTTP errors based on the status code of the response.

    Args:
        response (requests.Response): The HTTP response from the API call.

    Returns:
        str: The error type, based on the status code.
    """
    if response.status_code == HTTPStatus.UNAUTHORIZED:
        return "Unauthorized. Check your access token."
    elif response.status_code == HTTPStatus.FORBIDDEN:
        return "Forbidden. You do not have permission to perform this action."
    elif response.status_code == HTTPStatus.NOT_FOUND:
        return "Not Found. The resource could not be located."
    elif response.status_code == HTTPStatus.TOO_MANY_REQUESTS:
        return "Too Many Requests. You have hit the rate limit."
    else:
        return "HTTP Error"


def retry_request(
    func: Callable, retries: int = 3, delay: int = 1, *args: Any, **kwargs: Any
) -> Any:
    r"""Retries a function in case of any errors.

    Args:
        func (Callable): The function to be retried.
        retries (int): Number of retry attempts. (default: :obj:`3`)
        delay (int): Delay between retries in seconds. (default: :obj:`1`)
        *args: Arguments to pass to the function.
        **kwargs: Keyword arguments to pass to the function.

    Returns:
        Any: The result of the function call if successful.

    Raises:
        Exception: If all retry attempts fail.
    """
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            print(f"Attempt {attempt + 1}/{retries} failed: {e}")
            if attempt < retries - 1:
                time.sleep(delay)
            else:
                raise
