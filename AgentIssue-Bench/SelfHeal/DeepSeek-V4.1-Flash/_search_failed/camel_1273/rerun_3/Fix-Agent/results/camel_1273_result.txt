def download_tasks(task: TaskType, folder_path: str) -> None:

def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()
            if task_name.strip() and task_id.isnumeric():
                new_tasks_list.append(task_name)
    return new_tasks_list


def check_server_running(server_url: str) -> bool:
    r"""Check whether the port refered by the URL to the server
    is open.

            service.

    url_tuple = (parsed_url.hostname, parsed_url.port)

def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required module dependencies.

    Raises:
        ImportError: If any of the required modules are not available.

    Example:
        ::

            @dependencies_required('numpy', 'pandas')
            def data_processing_function():
                # Function implementation...
    """

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            missing_modules = [
                m for m in required_modules if not is_module_available(m)
            ]
            if missing_modules:
                raise ImportError(
                    f"Missing required modules: {', '.join(missing_modules)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def _is_api_key_pair(key: Any) -> bool:
    r"""Check whether ``key`` is a ``(argument_name, env_var_name)`` pair.

    Args:
        key (Any): The object to be checked.

    Returns:
        bool: Whether the object describes an API key with both the name of the
            argument (which may be :obj:`None`) and the name of the environment
            variable that may hold the key.
    """
    if not isinstance(key, (list, tuple)) or len(key) != 2:
        return False

    argument_name, env_var_name = key
    if not isinstance(env_var_name, str) or not (
        _ENVIRONMENT_VARIABLE_PATTERN.match(env_var_name)
    ):
        return False

    if argument_name is None:
        return True

    if not isinstance(argument_name, str):
        return False

    # An all-caps string is an environment variable name (e.g. when a list of
    # environment variable names is passed), not an argument name.
    return bool(_ARGUMENT_NAME_PATTERN.match(argument_name)) and not (
        _ENVIRONMENT_VARIABLE_PATTERN.match(argument_name)
    )


def _normalize_required_api_keys(
    required_keys: Tuple[Any, ...],
) -> List[Tuple[Optional[str], str]]:
    r"""Normalize the arguments of :func:`api_keys_required`.

    Args:
        required_keys (Tuple[Any, ...]): The raw arguments given to
            :func:`api_keys_required`. Each element can be the name of an
            environment variable, a ``(argument_name, env_var_name)`` pair, or
            a (nested) list/tuple of such elements.

    Returns:
        List[Tuple[Optional[str], str]]: A list of ``(argument_name,
            env_var_name)`` pairs. An ``argument_name`` of :obj:`None` means
            that the key was only described by the name of its environment
            variable.

    Raises:
        ValueError: If a required key can not be interpreted.
    """
    normalized_keys: List[Tuple[Optional[str], str]] = []

    def _add_key(key: Any) -> None:
        if isinstance(key, str):
            normalized_keys.append((None, key))
            return

        if isinstance(key, (list, tuple)):
            if _is_api_key_pair(key):
                argument_name, env_var_name = key
                normalized_keys.append((argument_name, env_var_name))
                return
            for sub_key in key:
                _add_key(sub_key)
            return

        raise ValueError(
            f"Invalid API key specification: {key!r}. Please provide the name "
            "of an environment variable or a (argument_name, env_var_name) "
            "tuple."
        )

    for key in required_keys:
        _add_key(key)

    return normalized_keys


def _get_api_key_candidate_names(
    argument_name: Optional[str], env_var_name: str
) -> List[str]:
    r"""Get the names that may hold the required API key.

    Args:
        argument_name (Optional[str]): The name of the argument that is
            expected to hold the API key, if it was explicitly specified.
        env_var_name (str): The name of the environment variable that is
            expected to hold the API key.

    Returns:
        List[str]: The names to look for among the arguments of the decorated
            function and among the attributes of the decorated instance.
    """
    if argument_name is not None:
        names = [argument_name, f"_{argument_name}"]
        return list(dict.fromkeys(names))

    # The key was only described by the name of its environment variable, so
    # all plausible argument/attribute names are checked, e.g.
    # 'OPENAI_API_KEY' -> ['openai_api_key', 'api_key', 'openai'].
    names = [env_var_name.lower(), *_API_KEY_ARGUMENT_NAMES]
    suffix = "_api_key"
    if env_var_name.lower().endswith(suffix):
        prefix = env_var_name.lower()[: -len(suffix)]
        if prefix:
            names.append(prefix)

    names = list(dict.fromkeys(names))
    names.extend(
        f"_{name}" for name in list(names) if not name.startswith("_")
    )
    return list(dict.fromkeys(names))


def _is_api_key_provided(
    argument_name: Optional[str],
    env_var_name: str,
    bound_arguments: Mapping[str, Any],
    instance: Any,
) -> bool:
    r"""Check whether a single required API key is provided.

    The key is looked up, in order, in the environment variables, in the
    arguments that were passed to the decorated function (which also covers
    decorated ``__init__`` methods, e.g. ``SomeModel(api_key='...')``), and in
    the attributes of the instance the decorated method is called on (e.g.
    ``self._api_key``).

    Args:
        argument_name (Optional[str]): The name of the argument that is
            expected to hold the API key, if it was explicitly specified.
        env_var_name (str): The name of the environment variable that is
            expected to hold the API key.
        bound_arguments (Mapping[str, Any]): The arguments that were passed to
            the decorated function, bound to its signature.
        instance (Any): The instance the decorated function is called on, if
            the decorated function is a method.

    Returns:
        bool: Whether the API key is provided.
    """
    # 1. The key can be provided by an environment variable.
    if os.environ.get(env_var_name):
        return True

    candidate_names = _get_api_key_candidate_names(argument_name, env_var_name)

    # 2. The key can be passed directly to the decorated function. This is the
    #    case for decorated `__init__` methods, where the key is given as an
    #    argument and is not yet available as an instance attribute.
    for name in candidate_names:
        if bound_arguments.get(name):
            return True

    # 3. The key can be stored as an attribute of the instance the decorated
    #    method is called on, e.g. `self._api_key`.
    if instance is not None:
        for name in candidate_names:
            if getattr(instance, name, None):
                return True

    return False


def api_keys_required(*required_keys: Any) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are presented as
    arguments of the decorated function, as attributes of the decorated
    instance, or in the environment variables.

    Each required key can be specified either as the name of an environment
    variable (e.g. ``'OPENAI_API_KEY'``), in which case the decorator looks for
    the key in the environment variables, in a matching argument of the
    decorated function (e.g. ``api_key``) and in a matching attribute of the
    decorated instance (e.g. ``self._api_key``), or as an explicit
    ``(argument_name, env_var_name)`` pair, in which case only those two
    locations are checked. Each key is checked individually, so that several
    keys can be required at the same time.

    Args:
        required_keys: The required API keys to be checked. Each of them can be
            given as a string (the name of the environment variable) or as a
            ``(argument_name, env_var_name)`` pair. A list of keys can also be
            passed directly, e.g.
            :obj:`@api_keys_required([('api_key', 'OPENAI_API_KEY')])`.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables, the function arguments and the instance
            attributes.

    Example:
        ::

            @api_keys_required([('api_key', 'OPENAI_API_KEY')])
            def some_api_function(api_key: str):
                # Function implementation...

            @api_keys_required('API_KEY_1', 'API_KEY_2')
            def another_api_function():
                # Function implementation...
    """
    normalized_keys = _normalize_required_api_keys(required_keys)

    def decorator(func: F) -> F:
        try:
            signature: Optional[inspect.Signature] = inspect.signature(func)
        except (TypeError, ValueError):  # pragma: no cover - builtins, etc.
            signature = None

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            bound_arguments: Mapping[str, Any] = {}
            if signature is not None:
                try:
                    bound = signature.bind(*args, **kwargs)
                    bound.apply_defaults()
                    bound_arguments = dict(bound.arguments)
                except TypeError:
                    # Let the decorated function itself report the error.
                    bound_arguments = {}

            instance = args[0] if args else None

            missing_keys: List[str] = []
            for argument_name, env_var_name in normalized_keys:
                if not _is_api_key_provided(
                    argument_name,
                    env_var_name,
                    bound_arguments,
                    instance,
                ):
                    missing_keys.append(env_var_name)

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )

            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper

def retry_request(
    func: Callable, retries: int = 3, delay: int = 1, *args: Any, **kwargs: Any
) -> Any:
    r"""Retries a function in case of any errors.

    Args:
        func (Callable): The function to be retried.
        retries (int): Number of retry attempts. (default: :obj:`3`)
        delay (int): Delay between retries in seconds. (default: :obj:`1`)
        *args: Arguments to pass to the function.
        **kwargs: Keyword arguments to pass to the function.

    Returns:
        Any: The result of the function call if successful.

    Raises:
        Exception: If all retry attempts fail.
    """
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            print(f"Attempt {attempt + 1}/{retries} failed: {e}")
            if attempt < retries - 1:
                time.sleep(delay)
            else:
                raise

**Fix:** it now binds the call arguments with `inspect.signature(func).bind(...)`, normalizes each required key into `(argument_name|None, env_var_name)`, and checks (per key) in three places: env var → bound arguments → instance attributes. It raises only for the keys that are genuinely missing, listing them.

### 2. Test trace (fixed version)

- `normalized_keys = [(None, "DUMMY_TOKEN")]`
- `signature` of `__init__` = `(self, api_key: str)`; `bind(self, api_key="xxxxx")` → `{'self': <instance>, 'api_key': 'xxxxx'}`
- `_is_api_key_provided(None, "DUMMY_TOKEN", ...)`:
  1. env var absent → continue
  2. candidates = `['dummy_token', 'api_key', '_api_key', '_dummy_token']` → `bound_arguments['api_key'] == 'xxxxx'` → **True**
- No missing keys → no `ValueError` → test returns `0` (fix succeeded). Buggy version still raises with `"DUMMY_TOKEN"` → returns `0` for buggy (reproduced). Both branches behave as intended.

### 3. Structural / API checks

- `validate_fixed_code` → **ACCEPT** (file present, single full text, no required interface violated).
- `inspect.signature` / `Signature.bind` / `BoundArguments.apply_defaults()` / `.arguments` are used in the documented way (`bind` raises `TypeError` on mismatch, which the code catches and degrades gracefully to the attribute-based check). The web lookup returned irrelevant noise, but this is standard-library usage I can confirm directly.
- New imports (`inspect`, `Tuple`) and the two module-level regexes are all defined before use; `callable(value)` metaclass and all other helpers untouched.
- Backward compatibility: plain-string keys (`@api_keys_required('API_KEY_1', 'API_KEY_2')`) still normalize to env-var checks; lists of keys and `(arg, ENV)` tuples are handled; `_is_api_key_pair` correctly rejects all-caps two-tuples (so `("A_KEY","B_KEY")` is treated as two env vars, not a pair). Methods where `self._api_key` is already set keep working through the instance-attribute path, so no regression versus the old behavior for normal (non-`__init__`) methods.
- No new exception types leak into the tested path; `signature.bind` failures are swallowed so the original function still reports its own argument errors.

### 4. Minor, non-blocking observations

- If a decorated callable forwarded the key only through `**kwargs`, the candidate lookup wouldn't inspect inside `kwargs`. That's a narrow edge case, does not affect the reported bug or the test, and does not regress any previously-supported usage (the old code could never satisfy that path either).
- The `(argument_name, env_var_name)` pair support is broader than strictly needed but is internally consistent and covered by the normalization logic.

def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()
            if task_name.strip() and task_id.isnumeric():
                new_tasks_list.append(task_name)
    return new_tasks_list


def check_server_running(server_url: str) -> bool:
    r"""Check whether the port refered by the URL to the server
    is open.

    Args:
        server_url (str): The URL to the server running LLM inference
            service.

    Returns:
        bool: Whether the port is open for packets (server is running).
    """
    parsed_url = urlparse(server_url)
    url_tuple = (parsed_url.hostname, parsed_url.port)

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

    # if the port is open, the result should be 0.
    return result == 0


def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required module dependencies.

    Raises:
        ImportError: If any of the required modules are not available.

    Example:
        ::

            @dependencies_required('numpy', 'pandas')
            def data_processing_function():
                # Function implementation...
    """

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            missing_modules = [
                m for m in required_modules if not is_module_available(m)
            ]
            if missing_modules:
                raise ImportError(
                    f"Missing required modules: {', '.join(missing_modules)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def _is_api_key_pair(key: Any) -> bool:
    r"""Check whether ``key`` is a ``(argument_name, env_var_name)`` pair.

    Args:
        key (Any): The object to be checked.

    Returns:
        bool: Whether the object describes an API key with both the name of the
            argument (which may be :obj:`None`) and the name of the environment
            variable that may hold the key.
    """
    if not isinstance(key, (list, tuple)) or len(key) != 2:
        return False

    argument_name, env_var_name = key
    if not isinstance(env_var_name, str) or not (
        _ENVIRONMENT_VARIABLE_PATTERN.match(env_var_name)
    ):
        return False

    if argument_name is None:
        return True

    if not isinstance(argument_name, str):
        return False

    # An all-caps string is an environment variable name (e.g. when a list of
    # environment variable names is passed), not an argument name.
    return bool(_ARGUMENT_NAME_PATTERN.match(argument_name)) and not (
        _ENVIRONMENT_VARIABLE_PATTERN.match(argument_name)
    )


def _normalize_required_api_keys(
    required_keys: Tuple[Any, ...],
) -> List[Tuple[Optional[str], str]]:
    r"""Normalize the arguments of :func:`api_keys_required`.

    Args:
        required_keys (Tuple[Any, ...]): The raw arguments given to
            :func:`api_keys_required`. Each element can be the name of an
            environment variable, a ``(argument_name, env_var_name)`` pair, or
            a (nested) list/tuple of such elements.

    Returns:
        List[Tuple[Optional[str], str]]: A list of ``(argument_name,
            env_var_name)`` pairs. An ``argument_name`` of :obj:`None` means
            that the key was only described by the name of its environment
            variable.

    Raises:
        ValueError: If a required key can not be interpreted.
    """
    normalized_keys: List[Tuple[Optional[str], str]] = []

    def _add_key(key: Any) -> None:
        if isinstance(key, str):
            normalized_keys.append((None, key))
            return

        if isinstance(key, (list, tuple)):
            if _is_api_key_pair(key):
                argument_name, env_var_name = key
                normalized_keys.append((argument_name, env_var_name))
                return
            for sub_key in key:
                _add_key(sub_key)
            return

        raise ValueError(
            f"Invalid API key specification: {key!r}. Please provide the name "
            "of an environment variable or a (argument_name, env_var_name) "
            "tuple."
        )

    for key in required_keys:
        _add_key(key)

    return normalized_keys


def _get_api_key_candidate_names(
    argument_name: Optional[str], env_var_name: str
) -> List[str]:
    r"""Get the names that may hold the required API key.

    Args:
        argument_name (Optional[str]): The name of the argument that is
            expected to hold the API key, if it was explicitly specified.
        env_var_name (str): The name of the environment variable that is
            expected to hold the API key.

    Returns:
        List[str]: The names to look for among the arguments of the decorated
            function and among the attributes of the decorated instance.
    """
    if argument_name is not None:
        names = [argument_name, f"_{argument_name}"]
        return list(dict.fromkeys(names))

    # The key was only described by the name of its environment variable, so
    # all plausible argument/attribute names are checked, e.g.
    # 'OPENAI_API_KEY' -> ['openai_api_key', 'api_key', 'openai'].
    names = [env_var_name.lower(), *_API_KEY_ARGUMENT_NAMES]
    suffix = "_api_key"
    if env_var_name.lower().endswith(suffix):
        prefix = env_var_name.lower()[: -len(suffix)]
        if prefix:
            names.append(prefix)

    names = list(dict.fromkeys(names))
    names.extend(
        f"_{name}" for name in list(names) if not name.startswith("_")
    )
    return list(dict.fromkeys(names))


def _is_api_key_provided(
    argument_name: Optional[str],
    env_var_name: str,
    bound_arguments: Mapping[str, Any],
    instance: Any,
) -> bool:
    r"""Check whether a single required API key is provided.

    The key is looked up, in order, in the environment variables, in the
    arguments that were passed to the decorated function (which also covers
    decorated ``__init__`` methods, e.g. ``SomeModel(api_key='...')``), and in
    the attributes of the instance the decorated method is called on (e.g.
    ``self._api_key``).

    Args:
        argument_name (Optional[str]): The name of the argument that is
            expected to hold the API key, if it was explicitly specified.
        env_var_name (str): The name of the environment variable that is
            expected to hold the API key.
        bound_arguments (Mapping[str, Any]): The arguments that were passed to
            the decorated function, bound to its signature.
        instance (Any): The instance the decorated function is called on, if
            the decorated function is a method.

    Returns:
        bool: Whether the API key is provided.
    """
    # 1. The key can be provided by an environment variable.
    if os.environ.get(env_var_name):
        return True

    candidate_names = _get_api_key_candidate_names(argument_name, env_var_name)

    # 2. The key can be passed directly to the decorated function. This is the
    #    case for decorated `__init__` methods, where the key is given as an
    #    argument and is not yet available as an instance attribute.
    for name in candidate_names:
        if bound_arguments.get(name):
            return True

    # 3. The key can be stored as an attribute of the instance the decorated
    #    method is called on, e.g. `self._api_key`.
    if instance is not None:
        for name in candidate_names:
            if getattr(instance, name, None):
                return True

    return False


def api_keys_required(*required_keys: Any) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are presented as
    arguments of the decorated function, as attributes of the decorated
    instance, or in the environment variables.

    Each required key can be specified either as the name of an environment
    variable (e.g. ``'OPENAI_API_KEY'``), in which case the decorator looks for
    the key in the environment variables, in a matching argument of the
    decorated function (e.g. ``api_key``) and in a matching attribute of the
    decorated instance (e.g. ``self._api_key``), or as an explicit
    ``(argument_name, env_var_name)`` pair, in which case only those two
    locations are checked. Each key is checked individually, so that several
    keys can be required at the same time.

    Args:
        required_keys: The required API keys to be checked. Each of them can be
            given as a string (the name of the environment variable) or as a
            ``(argument_name, env_var_name)`` pair. A list of keys can also be
            passed directly, e.g.
            :obj:`@api_keys_required([('api_key', 'OPENAI_API_KEY')])`.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables, the function arguments and the instance
            attributes.

    Example:
        ::

            @api_keys_required([('api_key', 'OPENAI_API_KEY')])
            def some_api_function(api_key: str):
                # Function implementation...

            @api_keys_required('API_KEY_1', 'API_KEY_2')
            def another_api_function():
                # Function implementation...
    """
    normalized_keys = _normalize_required_api_keys(required_keys)

    def decorator(func: F) -> F:
        try:
            signature: Optional[inspect.Signature] = inspect.signature(func)
        except (TypeError, ValueError):  # pragma: no cover - builtins, etc.
            signature = None

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            bound_arguments: Mapping[str, Any] = {}
            if signature is not None:
                try:
                    bound = signature.bind(*args, **kwargs)
                    bound.apply_defaults()
                    bound_arguments = dict(bound.arguments)
                except TypeError:
                    # Let the decorated function itself report the error.
                    bound_arguments = {}

            instance = args[0] if args else None

            missing_keys: List[str] = []
            for argument_name, env_var_name in normalized_keys:
                if not _is_api_key_provided(
                    argument_name,
                    env_var_name,
                    bound_arguments,
                    instance,
                ):
                    missing_keys.append(env_var_name)

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )

            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper


class AgentOpsMeta(type):
    r"""Metaclass that automatically decorates all callable attributes with
    the agentops_decorator,
    except for the 'get_tools' method.

    Methods:
    __new__(cls, name, bases, dct):
        Creates a new class with decorated methods.
    """

    def __new__(cls, name, bases, dct):
        if ToolEvent:
            for attr, value in dct.items():
                if callable(value) and attr != 'get_tools':
                    dct[attr] = agentops_decorator(value)
        return super().__new__(cls, name, bases, dct)

def handle_http_error(response: requests.Response) -> str:
    r"""Handles the HTTP errors based on the status code of the response.

    Args:
        response (requests.Response): The HTTP response from the API call.

    Returns:
        str: The error type, based on the status code.
    """
    if response.status_code == HTTPStatus.UNAUTHORIZED:
        return "Unauthorized. Check your access token."
    elif response.status_code == HTTPStatus.FORBIDDEN:
        return "Forbidden. You do not have permission to perform this action."
    elif response.status_code == HTTPStatus.NOT_FOUND:
        return "Not Found. The resource could not be located."
    elif response.status_code == HTTPStatus.TOO_MANY_REQUESTS:
        return "Too Many Requests. You have hit the rate limit."
    else:
        return "HTTP Error"


def retry_request(
    func: Callable, retries: int = 3, delay: int = 1, *args: Any, **kwargs: Any
) -> Any:
    r"""Retries a function in case of any errors.

    Args:
        func (Callable): The function to be retried.
        retries (int): Number of retry attempts. (default: :obj:`3`)
        delay (int): Delay between retries in seconds. (default: :obj:`1`)
        *args: Arguments to pass to the function.
        **kwargs: Keyword arguments to pass to the function.

    Returns:
        Any: The result of the function call if successful.

    Raises:
        Exception: If all retry attempts fail.
    """
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            print(f"Attempt {attempt + 1}/{retries} failed: {e}")
            if attempt < retries - 1:
                time.sleep(delay)
            else:
                raise
