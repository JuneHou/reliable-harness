def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()

def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:

    Raises:

            def data_processing_function():
                # Function implementation...
    """

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            missing_modules = [

        return cast(F, wrapper)

    return decorator

def _is_value_provided(value: Any) -> bool:
    r"""Check whether a value passed by the user can be considered as
    provided, i.e. it is not :obj:`None`, not an empty string, and not a
    callable (such as a method or a class).

    Args:
        value (Any): The value to be checked.

    Returns:
        bool: Whether the value is considered as provided.
    """
    if value is None or callable(value):
        return False
    if isinstance(value, str):
        return bool(value.strip())
    return True


def _get_candidate_names(key: str) -> List[str]:
    r"""Infer the possible argument or attribute names an API key
    environment variable could be passed with.

    For instance, the environment variable :obj:`"OPENAI_API_KEY"` may be
    passed through the arguments/attributes :obj:`"openai_api_key"`,
    :obj:`"api_key"` or :obj:`"key"`.

    Args:
        key (str): The name of the environment variable.

    Returns:
        List[str]: The possible names, ordered from the most specific to the
            most generic one.
    """
    names: List[str] = []

    def add(name: str) -> None:
        if name and name not in names:
            names.append(name)

    add(key)
    add(key.lower())

    parts = [part for part in key.lower().split('_') if part]
    for index in range(len(parts)):
        name = '_'.join(parts[index:])
        add(name)
        add(f'_{name}')

    return names


def _bind_arguments(
    func: Callable[..., Any], args: Tuple[Any, ...], kwargs: Dict[str, Any]
) -> Dict[str, Any]:
    r"""Bind the given positional and keyword arguments to the parameter
    names of the decorated function.

    Args:
        func (Callable[..., Any]): The decorated function.
        args (Tuple[Any, ...]): The positional arguments of the call.
        kwargs (Dict[str, Any]): The keyword arguments of the call.

    Returns:
        Dict[str, Any]: A mapping from parameter names to the values they
            were given in the call.
    """
    try:
        signature = inspect.signature(func)
        bound = signature.bind_partial(*args, **kwargs)
        return dict(bound.arguments)
    except (TypeError, ValueError):  # pragma: no cover
        return dict(kwargs)


def _parse_required_key(entry: Any) -> Tuple[Optional[str], Optional[str]]:
    r"""Normalize a required key declaration into an
    :obj:`(argument_name, environment_variable_name)` tuple.

    Args:
        entry (Any): The declaration of a required key. It can be the name of
            the environment variable, a :obj:`(argument_name,
            environment_variable_name)` tuple, or :obj:`None` to explicitly
            state that the key is optional.

    Returns:
        Tuple[Optional[str], Optional[str]]: The argument name (if it is
            explicitly given) and the name of the environment variable.
    """
    if isinstance(entry, str):
        return None, (entry or None)
    if isinstance(entry, (tuple, list)):
        if len(entry) == 1:
            return None, _parse_required_key(entry[0])[1]
        if len(entry) >= 2:
            argument_name, key = entry[0], entry[1]
            if not isinstance(argument_name, str):
                argument_name = None
            if not isinstance(key, str) or not key:
                key = None
            return argument_name, key
    return None, None


def _instance_attribute_provided(instance: Any, name: str) -> bool:
    r"""Check whether the given attribute of the instance holds a value
    that has been provided by the user.

    Args:
        instance (Any): The instance the decorated function is bound to.
        name (str): The name of the attribute to be checked.

    Returns:
        bool: Whether the attribute exists and holds a provided value.
    """
    if instance is None or name.startswith('__'):
        return False
    try:
        if not hasattr(instance, name):
            return False
        value = getattr(instance, name)
    except Exception:  # pragma: no cover
        return False
    return _is_value_provided(value)


def _key_passed_directly(
    key: str,
    argument_name: Optional[str],
    bound_arguments: Mapping[str, Any],
    instance: Any,
    used_names: Set[str],
) -> bool:
    r"""Check whether the API key identified by an environment variable name
    has been passed directly to the decorated function, e.g. as the
    :obj:`api_key` argument of a decorated :obj:`__init__` method, or stored
    as an instance attribute.

    Every provided value is consumed by at most one requirement, so that
    having multiple required API keys can not be satisfied by a single
    value.

    Args:
        key (str): The name of the environment variable holding the key.
        argument_name (Optional[str]): The name of the argument that
            explicitly carries the key, if declared by the user.
        bound_arguments (Mapping[str, Any]): The arguments of the call bound
            to the parameter names of the decorated function.
        instance (Any): The instance the decorated function is bound to, if
            any.
        used_names (Set[str]): The names already used to satisfy another
            required key. Updated in place.

    Returns:
        bool: Whether the key has been provided directly.
    """
    candidate_names: List[str] = []
    if argument_name:
        candidate_names.append(argument_name)
    for name in _get_candidate_names(key):
        if name not in candidate_names:
            candidate_names.append(name)

    for name in candidate_names:
        if name in used_names or name in ('self', 'cls'):
            continue
        if name in bound_arguments and _is_value_provided(
            bound_arguments[name]
        ):
            used_names.add(name)
            return True
        if _instance_attribute_provided(instance, name):
            used_names.add(name)
            return True
    return False


def api_keys_required(*required_keys: str) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are
    presented in the environment variables, passed directly to the decorated
    function (e.g. through the ``api_key`` argument of an ``__init__``
    method), or stored as an instance attribute.

    Each required key can be given either as the name of the environment
    variable, or as a :obj:`(argument_name, environment_variable_name)`
    tuple, where :obj:`argument_name` is the name of the argument carrying
    the key directly (use :obj:`None` to let it be inferred from
    :obj:`environment_variable_name`). Passing :obj:`None` as a key means
    that the key is optional.

    Args:
        required_keys (str): The required API keys to be checked.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables and are not passed to the decorated
            function nor stored as an instance attribute.

    Example:
        ::

            @api_keys_required('API_KEY_1', 'API_KEY_2')
            def some_api_function():
                # Function implementation...

            class SomeClient:

                @api_keys_required('OPENAI_API_KEY')
                def __init__(self, api_key: Optional[str] = None):
                    self._api_key = api_key

                @api_keys_required([(None, 'OPENAI_API_KEY')])
                def some_method(self):
                    # Function implementation...
    """

    keys: List[Any] = list(required_keys)
    # The required keys may also be given as a single list/tuple, e.g.
    # ``@api_keys_required(['API_KEY_1', 'API_KEY_2'])``.
    if len(keys) == 1 and isinstance(keys[0], (list, tuple, set, frozenset)):
        keys = list(keys[0])

    def decorator(func: F) -> F:
        try:
            parameters = list(inspect.signature(func).parameters)
        except (TypeError, ValueError):  # pragma: no cover
            parameters = []

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            bound_arguments = _bind_arguments(func, args, kwargs)

            # Resolve the instance the decorated function is bound to, if
            # any, so that API keys stored as instance attributes (e.g.
            # ``self._api_key``) are taken into account.
            instance: Any = None
            if parameters:
                first_parameter = parameters[0]
                if first_parameter in bound_arguments:
                    instance = bound_arguments[first_parameter]
                elif args:
                    instance = args[0]
                if isinstance(
                    instance,
                    (
                        str,
                        bytes,
                        bytearray,
                        int,
                        float,
                        complex,
                        bool,
                        dict,
                        list,
                        tuple,
                        set,
                        frozenset,
                    ),
                ):
                    instance = None

            used_names: Set[str] = set()
            missing_keys: List[str] = []

            for entry in keys:
                argument_name, key = _parse_required_key(entry)
                if key is None:
                    # The key is explicitly marked as optional.
                    continue
                if key in os.environ:
                    # The key is presented in the environment variables.
                    continue
                if _key_passed_directly(
                    key,
                    argument_name,
                    bound_arguments,
                    instance,
                    used_names,
                ):
                    # The key has been passed directly to the function.
                    continue
                missing_keys.append(key)

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper

def retry_request(
    func: Callable, retries: int = 3, delay: int = 1, *args: Any, **kwargs: Any
) -> Any:
    r"""Retries a function in case of any errors.

    Args:
        func (Callable): The function to be retried.
        retries (int): Number of retry attempts. (default: :obj:`3`)
        delay (int): Delay between retries in seconds. (default: :obj:`1`)
        *args: Arguments to pass to the function.
        **kwargs: Keyword arguments to pass to the function.

    Returns:
        Any: The result of the function call if successful.

    Raises:
        Exception: If all retry attempts fail.
    """
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            print(f"Attempt {attempt + 1}/{retries} failed: {e}")
            if attempt < retries - 1:
                time.sleep(delay)
            else:
                raise

The fix works because:

- `_bind_arguments` uses `inspect.signature(func).bind_partial(*args, **kwargs)` and reads `.arguments` — correct stdlib usage (returns a `BoundArguments` whose `.arguments` is a mapping of parameter name → value). For `__init__(self, api_key)` called with `("xxxxx",)`, it yields `{'self': <instance>, 'api_key': 'xxxxx'}`.
- `_key_passed_directly` then finds `api_key` (via `_get_candidate_names("DUMMY_TOKEN")` → `dummy_token`, `token`, `_token` … plus the explicit/inferred name) and `_is_value_provided("xxxxx")` → `True`, so no `ValueError` is raised → the test's `fixed/patched` branch returns `0` ("fix succeeded").
- Buggy version still raises (env var absent, `self._api_key` not yet set) → `0` ("bug reproduced"). So the intended behavioral transition `buggy:0 → fixed:0/-no error` holds.

**Structural validation — failed:**

`validate_fixed_code` returned **REJECT**: `"the line ### camel/utils/commons.py appears 0 times, expected once"`. The diff confirms this is a real, unintended edit — line 1 of the file (`-### camel/utils/commons.py`) was deleted in the fixed version. That is a violation of the mandated output/format rules (the file-path marker must appear exactly once), independent of whether the logic is right. The tool also reported no interface was verifiable against the test name, so there is no counter-evidence redeeming the format failure.

**Secondary quality concerns (even ignoring the format gate):**

1. **Over-permissive inference:** `_get_candidate_names("OPENAI_API_KEY")` emits `openai_api_key`, `api_key`, `_api_key`, `key`, `_key`. Any object carrying a generic `api_key`/`key` attribute would satisfy *every* `*_API_KEY` requirement, which can silently swallow genuinely missing keys (the exact failure mode the decorator exists to prevent). With multiple required keys, `used_names` dedup helps, but only per distinct name string.
2. **`_is_value_provided` rejects callables** — a key legitimately supplied as a lazy/callable secret (or a `str` subclass/property object) is now treated as missing, a behavior change not required by the bug.
3. **Heuristic `self` detection:** any bound first argument that isn't a primitive container is treated as the instance (`instance = bound_arguments[parameters[0]]`). For a plain function `some_api_function(obj)` decorated with `api_keys_required`, arbitrary attributes on `obj` get scanned and can spuriously satisfy requirements.
4. `_parse_required_key` silently drops malformed entries (`return None, None`), turning a mistyped key declaration into "no requirement" rather than an error.

**Conclusion:**

The core logic fix is sound and would pass the supplied `reproduce.py` for both `buggy` and `fixed`, and the `inspect`/`bind_partial` usage is correct. However, `validate_fixed_code` rejects the submission because the required file-identifier line was dropped, and the diff shows that as an unrelated, unintended modification to the file — a structural/format rule failure. Combined with the over-broad key-matching heuristic that weakens the decorator's guarantees, the submission cannot be accepted as-is.

**REJECT** — fix the missing `### camel/utils/commons.py` header line so the diff contains no spurious edits and passes structural validation, and narrow the candidate-name matching (e.g., prefer the explicit `(argument_name, ENV_VAR)` form and only fall back to `*_API_KEY` suffixes, not bare `key`) so missing keys are still reported.

def get_task_list(task_response: str) -> List[str]:
    r"""Parse the response of the Agent and return task list.

    Args:
        task_response (str): The string response of the Agent.

    Returns:
        List[str]: A list of the string tasks.
    """

    new_tasks_list = []
    task_string_list = task_response.strip().split('\n')
    # each task starts with #.
    for task_string in task_string_list:
        task_parts = task_string.strip().split(".", 1)
        if len(task_parts) == 2:
            task_id = ''.join(s for s in task_parts[0] if s.isnumeric())
            task_name = re.sub(r'[^\w\s_]+', '', task_parts[1]).strip()
            if task_name.strip() and task_id.isnumeric():
                new_tasks_list.append(task_name)
    return new_tasks_list


def check_server_running(server_url: str) -> bool:
    r"""Check whether the port refered by the URL to the server
    is open.

    Args:
        server_url (str): The URL to the server running LLM inference
            service.

    Returns:
        bool: Whether the port is open for packets (server is running).
    """
    parsed_url = urlparse(server_url)
    url_tuple = (parsed_url.hostname, parsed_url.port)

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    result = sock.connect_ex(url_tuple)
    sock.close()

    # if the port is open, the result should be 0.
    return result == 0


def dependencies_required(*required_modules: str) -> Callable[[F], F]:
    r"""A decorator to ensure that specified Python modules
    are available before a function executes.

    Args:
        required_modules (str): The required modules to be checked for
            availability.

    Returns:
        Callable[[F], F]: The original function with the added check for
            required module dependencies.

    Raises:
        ImportError: If any of the required modules are not available.

    Example:
        ::

            @dependencies_required('numpy', 'pandas')
            def data_processing_function():
                # Function implementation...
    """

    def decorator(func: F) -> F:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            missing_modules = [
                m for m in required_modules if not is_module_available(m)
            ]
            if missing_modules:
                raise ImportError(
                    f"Missing required modules: {', '.join(missing_modules)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def _is_value_provided(value: Any) -> bool:
    r"""Check whether a value passed by the user can be considered as
    provided, i.e. it is not :obj:`None`, not an empty string, and not a
    callable (such as a method or a class).

    Args:
        value (Any): The value to be checked.

    Returns:
        bool: Whether the value is considered as provided.
    """
    if value is None or callable(value):
        return False
    if isinstance(value, str):
        return bool(value.strip())
    return True


def _get_candidate_names(key: str) -> List[str]:
    r"""Infer the possible argument or attribute names an API key
    environment variable could be passed with.

    For instance, the environment variable :obj:`"OPENAI_API_KEY"` may be
    passed through the arguments/attributes :obj:`"openai_api_key"`,
    :obj:`"api_key"` or :obj:`"key"`.

    Args:
        key (str): The name of the environment variable.

    Returns:
        List[str]: The possible names, ordered from the most specific to the
            most generic one.
    """
    names: List[str] = []

    def add(name: str) -> None:
        if name and name not in names:
            names.append(name)

    add(key)
    add(key.lower())

    parts = [part for part in key.lower().split('_') if part]
    for index in range(len(parts)):
        name = '_'.join(parts[index:])
        add(name)
        add(f'_{name}')

    return names


def _bind_arguments(
    func: Callable[..., Any], args: Tuple[Any, ...], kwargs: Dict[str, Any]
) -> Dict[str, Any]:
    r"""Bind the given positional and keyword arguments to the parameter
    names of the decorated function.

    Args:
        func (Callable[..., Any]): The decorated function.
        args (Tuple[Any, ...]): The positional arguments of the call.
        kwargs (Dict[str, Any]): The keyword arguments of the call.

    Returns:
        Dict[str, Any]: A mapping from parameter names to the values they
            were given in the call.
    """
    try:
        signature = inspect.signature(func)
        bound = signature.bind_partial(*args, **kwargs)
        return dict(bound.arguments)
    except (TypeError, ValueError):  # pragma: no cover
        return dict(kwargs)


def _parse_required_key(entry: Any) -> Tuple[Optional[str], Optional[str]]:
    r"""Normalize a required key declaration into an
    :obj:`(argument_name, environment_variable_name)` tuple.

    Args:
        entry (Any): The declaration of a required key. It can be the name of
            the environment variable, a :obj:`(argument_name,
            environment_variable_name)` tuple, or :obj:`None` to explicitly
            state that the key is optional.

    Returns:
        Tuple[Optional[str], Optional[str]]: The argument name (if it is
            explicitly given) and the name of the environment variable.
    """
    if isinstance(entry, str):
        return None, (entry or None)
    if isinstance(entry, (tuple, list)):
        if len(entry) == 1:
            return None, _parse_required_key(entry[0])[1]
        if len(entry) >= 2:
            argument_name, key = entry[0], entry[1]
            if not isinstance(argument_name, str):
                argument_name = None
            if not isinstance(key, str) or not key:
                key = None
            return argument_name, key
    return None, None


def _instance_attribute_provided(instance: Any, name: str) -> bool:
    r"""Check whether the given attribute of the instance holds a value
    that has been provided by the user.

    Args:
        instance (Any): The instance the decorated function is bound to.
        name (str): The name of the attribute to be checked.

    Returns:
        bool: Whether the attribute exists and holds a provided value.
    """
    if instance is None or name.startswith('__'):
        return False
    try:
        if not hasattr(instance, name):
            return False
        value = getattr(instance, name)
    except Exception:  # pragma: no cover
        return False
    return _is_value_provided(value)


def _key_passed_directly(
    key: str,
    argument_name: Optional[str],
    bound_arguments: Mapping[str, Any],
    instance: Any,
    used_names: Set[str],
) -> bool:
    r"""Check whether the API key identified by an environment variable name
    has been passed directly to the decorated function, e.g. as the
    :obj:`api_key` argument of a decorated :obj:`__init__` method, or stored
    as an instance attribute.

    Every provided value is consumed by at most one requirement, so that
    having multiple required API keys can not be satisfied by a single
    value.

    Args:
        key (str): The name of the environment variable holding the key.
        argument_name (Optional[str]): The name of the argument that
            explicitly carries the key, if declared by the user.
        bound_arguments (Mapping[str, Any]): The arguments of the call bound
            to the parameter names of the decorated function.
        instance (Any): The instance the decorated function is bound to, if
            any.
        used_names (Set[str]): The names already used to satisfy another
            required key. Updated in place.

    Returns:
        bool: Whether the key has been provided directly.
    """
    candidate_names: List[str] = []
    if argument_name:
        candidate_names.append(argument_name)
    for name in _get_candidate_names(key):
        if name not in candidate_names:
            candidate_names.append(name)

    for name in candidate_names:
        if name in used_names or name in ('self', 'cls'):
            continue
        if name in bound_arguments and _is_value_provided(
            bound_arguments[name]
        ):
            used_names.add(name)
            return True
        if _instance_attribute_provided(instance, name):
            used_names.add(name)
            return True
    return False


def api_keys_required(*required_keys: str) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are
    presented in the environment variables, passed directly to the decorated
    function (e.g. through the ``api_key`` argument of an ``__init__``
    method), or stored as an instance attribute.

    Each required key can be given either as the name of the environment
    variable, or as a :obj:`(argument_name, environment_variable_name)`
    tuple, where :obj:`argument_name` is the name of the argument carrying
    the key directly (use :obj:`None` to let it be inferred from
    :obj:`environment_variable_name`). Passing :obj:`None` as a key means
    that the key is optional.

    Args:
        required_keys (str): The required API keys to be checked.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables and are not passed to the decorated
            function nor stored as an instance attribute.

    Example:
        ::

            @api_keys_required('API_KEY_1', 'API_KEY_2')
            def some_api_function():
                # Function implementation...

            class SomeClient:

                @api_keys_required('OPENAI_API_KEY')
                def __init__(self, api_key: Optional[str] = None):
                    self._api_key = api_key

                @api_keys_required([(None, 'OPENAI_API_KEY')])
                def some_method(self):
                    # Function implementation...
    """

    keys: List[Any] = list(required_keys)
    # The required keys may also be given as a single list/tuple, e.g.
    # ``@api_keys_required(['API_KEY_1', 'API_KEY_2'])``.
    if len(keys) == 1 and isinstance(keys[0], (list, tuple, set, frozenset)):
        keys = list(keys[0])

    def decorator(func: F) -> F:
        try:
            parameters = list(inspect.signature(func).parameters)
        except (TypeError, ValueError):  # pragma: no cover
            parameters = []

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            bound_arguments = _bind_arguments(func, args, kwargs)

            # Resolve the instance the decorated function is bound to, if
            # any, so that API keys stored as instance attributes (e.g.
            # ``self._api_key``) are taken into account.
            instance: Any = None
            if parameters:
                first_parameter = parameters[0]
                if first_parameter in bound_arguments:
                    instance = bound_arguments[first_parameter]
                elif args:
                    instance = args[0]
                if isinstance(
                    instance,
                    (
                        str,
                        bytes,
                        bytearray,
                        int,
                        float,
                        complex,
                        bool,
                        dict,
                        list,
                        tuple,
                        set,
                        frozenset,
                    ),
                ):
                    instance = None

            used_names: Set[str] = set()
            missing_keys: List[str] = []

            for entry in keys:
                argument_name, key = _parse_required_key(entry)
                if key is None:
                    # The key is explicitly marked as optional.
                    continue
                if key in os.environ:
                    # The key is presented in the environment variables.
                    continue
                if _key_passed_directly(
                    key,
                    argument_name,
                    bound_arguments,
                    instance,
                    used_names,
                ):
                    # The key has been passed directly to the function.
                    continue
                missing_keys.append(key)

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper


class AgentOpsMeta(type):
    r"""Metaclass that automatically decorates all callable attributes with
    the agentops_decorator,
    except for the 'get_tools' method.

    Methods:
    __new__(cls, name, bases, dct):
        Creates a new class with decorated methods.
    """

    def __new__(cls, name, bases, dct):
        if ToolEvent:

def _is_value_provided(value: Any) -> bool:
    r"""Check whether a value passed by the user can be considered as
    provided, i.e. it is not :obj:`None`, not an empty string, and not a
    callable (such as a method or a class).

    Args:
        value (Any): The value to be checked.

    Returns:
        bool: Whether the value is considered as provided.
    """
    if value is None or callable(value):
        return False

def _get_candidate_names(key: str) -> List[str]:
    r"""Infer the possible argument or attribute names an API key
    environment variable could be passed with.

    For instance, the environment variable :obj:`"OPENAI_API_KEY"` may be
    passed through the arguments/attributes :obj:`"openai_api_key"`,
    :obj:`"api_key"` or :obj:`"key"`.

    Args:
        key (str): The name of the environment variable.

    Returns:
        List[str]: The possible names, ordered from the most specific to the
            most generic one.
    """
    names: List[str] = []

    def add(name: str) -> None:
        if name and name not in names:
            names.append(name)

    add(key)
    add(key.lower())

    parts = [part for part in key.lower().split('_') if part]
    for index in range(len(parts)):
        name = '_'.join(parts[index:])
        add(name)
        add(f'_{name}')

    return names


def _bind_arguments(
    func: Callable[..., Any], args: Tuple[Any, ...], kwargs: Dict[str, Any]
) -> Dict[str, Any]:
    r"""Bind the given positional and keyword arguments to the parameter
    names of the decorated function.

    Args:
        func (Callable[..., Any]): The decorated function.
        args (Tuple[Any, ...]): The positional arguments of the call.
        kwargs (Dict[str, Any]): The keyword arguments of the call.

    Returns:
        Dict[str, Any]: A mapping from parameter names to the values they
            were given in the call.
    """
    try:
        signature = inspect.signature(func)
        bound = signature.bind_partial(*args, **kwargs)
        return dict(bound.arguments)
    except (TypeError, ValueError):  # pragma: no cover
        return dict(kwargs)


def _parse_required_key(entry: Any) -> Tuple[Optional[str], Optional[str]]:
    r"""Normalize a required key declaration into an
    :obj:`(argument_name, environment_variable_name)` tuple.

    Args:
        entry (Any): The declaration of a required key. It can be the name of
            the environment variable, a :obj:`(argument_name,
            environment_variable_name)` tuple, or :obj:`None` to explicitly
            state that the key is optional.

    Returns:
        Tuple[Optional[str], Optional[str]]: The argument name (if it is
            explicitly given) and the name of the environment variable.
    """
    if isinstance(entry, str):
        return None, (entry or None)
    if isinstance(entry, (tuple, list)):
        if len(entry) == 1:
            return None, _parse_required_key(entry[0])[1]
        if len(entry) >= 2:
            argument_name, key = entry[0], entry[1]
            if not isinstance(argument_name, str):
                argument_name = None
            if not isinstance(key, str) or not key:
                key = None
            return argument_name, key
    return None, None


def _instance_attribute_provided(instance: Any, name: str) -> bool:
    r"""Check whether the given attribute of the instance holds a value
    that has been provided by the user.

    Args:
        instance (Any): The instance the decorated function is bound to.
        name (str): The name of the attribute to be checked.

    Returns:
        bool: Whether the attribute exists and holds a provided value.
    """
    if instance is None or name.startswith('__'):
        return False
    try:
        if not hasattr(instance, name):
            return False
        value = getattr(instance, name)
    except Exception:  # pragma: no cover
        return False
    return _is_value_provided(value)


def _key_passed_directly(
    key: str,
    argument_name: Optional[str],
    bound_arguments: Mapping[str, Any],
    instance: Any,
    used_names: Set[str],
) -> bool:
    r"""Check whether the API key identified by an environment variable name
    has been passed directly to the decorated function, e.g. as the
    :obj:`api_key` argument of a decorated :obj:`__init__` method, or stored
    as an instance attribute.

    Every provided value is consumed by at most one requirement, so that
    having multiple required API keys can not be satisfied by a single
    value.

    Args:
        key (str): The name of the environment variable holding the key.
        argument_name (Optional[str]): The name of the argument that
            explicitly carries the key, if declared by the user.
        bound_arguments (Mapping[str, Any]): The arguments of the call bound
            to the parameter names of the decorated function.
        instance (Any): The instance the decorated function is bound to, if
            any.
        used_names (Set[str]): The names already used to satisfy another
            required key. Updated in place.

    Returns:
        bool: Whether the key has been provided directly.
    """
    candidate_names: List[str] = []
    if argument_name:
        candidate_names.append(argument_name)
    for name in _get_candidate_names(key):
        if name not in candidate_names:
            candidate_names.append(name)

    for name in candidate_names:
        if name in used_names or name in ('self', 'cls'):
            continue
        if name in bound_arguments and _is_value_provided(
            bound_arguments[name]
        ):
            used_names.add(name)
            return True
        if _instance_attribute_provided(instance, name):
            used_names.add(name)
            return True
    return False


def api_keys_required(*required_keys: str) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are
    presented in the environment variables, passed directly to the decorated
    function (e.g. through the ``api_key`` argument of an ``__init__``
    method), or stored as an instance attribute.

    Each required key can be given either as the name of the environment
    variable, or as a :obj:`(argument_name, environment_variable_name)`
    tuple, where :obj:`argument_name` is the name of the argument carrying
    the key directly (use :obj:`None` to let it be inferred from
    :obj:`environment_variable_name`). Passing :obj:`None` as a key means
    that the key is optional.

    Args:
        required_keys (str): The required API keys to be checked.

            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables and are not passed to the decorated
            function nor stored as an instance attribute.

    Example:
        ::

            @api_keys_required('API_KEY_1', 'API_KEY_2')
            def some_api_function():
                # Function implementation...

            class SomeClient:

                @api_keys_required('OPENAI_API_KEY')
                def __init__(self, api_key: Optional[str] = None):
                    self._api_key = api_key

                @api_keys_required([(None, 'OPENAI_API_KEY')])
                def some_method(self):
                    # Function implementation...
    """

    # The required keys may also be given as a single list/tuple, e.g.
    # ``@api_keys_required(['API_KEY_1', 'API_KEY_2'])``.
    if len(keys) == 1 and isinstance(keys[0], (list, tuple, set, frozenset)):
        keys = list(keys[0])

    def decorator(func: F) -> F:
        try:
            parameters = list(inspect.signature(func).parameters)
        except (TypeError, ValueError):  # pragma: no cover
            parameters = []

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            bound_arguments = _bind_arguments(func, args, kwargs)

            # Resolve the instance the decorated function is bound to, if
            # any, so that API keys stored as instance attributes (e.g.
            # ``self._api_key``) are taken into account.
            instance: Any = None
            if parameters:
                first_parameter = parameters[0]
                if first_parameter in bound_arguments:
                    instance = bound_arguments[first_parameter]
                elif args:
                    instance = args[0]
                if isinstance(
                    instance,
                    (
                        str,
                        bytes,
                        bytearray,
                        int,
                        float,
                        complex,
                        bool,
                        dict,
                        list,
                        tuple,
                        set,
                        frozenset,
                    ),
                ):
                    instance = None

            used_names: Set[str] = set()
            missing_keys: List[str] = []

            for entry in keys:
                argument_name, key = _parse_required_key(entry)
                if key is None:
                    # The key is explicitly marked as optional.
                    continue
                if key in os.environ:
                    # The key is presented in the environment variables.
                    continue
                if _key_passed_directly(
                    key,
                    argument_name,
                    bound_arguments,
                    instance,
                    used_names,
                ):
                    # The key has been passed directly to the function.
                    continue
                missing_keys.append(key)

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code

def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks

def _get_candidate_names(key: str) -> List[str]:
    r"""Infer the possible argument or attribute names an API key
    environment variable could be passed with.

    For instance, the environment variable :obj:`"OPENAI_API_KEY"` may be
    passed through the arguments/attributes :obj:`"openai_api_key"`,
    :obj:`"api_key"` or :obj:`"key"`.

    Args:
        key (str): The name of the environment variable.

    Returns:
        List[str]: The possible names, ordered from the most specific to the
            most generic one.
    """
    names: List[str] = []

    def add(name: str) -> None:
        if name and name not in names:
            names.append(name)

    add(key)
    add(key.lower())

    parts = [part for part in key.lower().split('_') if part]
    for index in range(len(parts)):
        name = '_'.join(parts[index:])
        add(name)
        add(f'_{name}')

    return names


def _bind_arguments(
    func: Callable[..., Any], args: Tuple[Any, ...], kwargs: Dict[str, Any]
) -> Dict[str, Any]:
    r"""Bind the given positional and keyword arguments to the parameter
    names of the decorated function.

    Args:
        func (Callable[..., Any]): The decorated function.
        args (Tuple[Any, ...]): The positional arguments of the call.
        kwargs (Dict[str, Any]): The keyword arguments of the call.

    Returns:
        Dict[str, Any]: A mapping from parameter names to the values they
            were given in the call.
    """
    try:
        signature = inspect.signature(func)
        bound = signature.bind_partial(*args, **kwargs)
        return dict(bound.arguments)
    except (TypeError, ValueError):  # pragma: no cover
        return dict(kwargs)


def _parse_required_key(entry: Any) -> Tuple[Optional[str], Optional[str]]:
    r"""Normalize a required key declaration into an
    :obj:`(argument_name, environment_variable_name)` tuple.

    Args:
        entry (Any): The declaration of a required key. It can be the name of
            the environment variable, a :obj:`(argument_name,
            environment_variable_name)` tuple, or :obj:`None` to explicitly
            state that the key is optional.

    Returns:
        Tuple[Optional[str], Optional[str]]: The argument name (if it is
            explicitly given) and the name of the environment variable.
    """
    if isinstance(entry, str):
        return None, (entry or None)
    if isinstance(entry, (tuple, list)):
        if len(entry) == 1:
            return None, _parse_required_key(entry[0])[1]
        if len(entry) >= 2:
            argument_name, key = entry[0], entry[1]
            if not isinstance(argument_name, str):
                argument_name = None
            if not isinstance(key, str) or not key:
                key = None
            return argument_name, key
    return None, None


def _instance_attribute_provided(instance: Any, name: str) -> bool:
    r"""Check whether the given attribute of the instance holds a value
    that has been provided by the user.

    Args:
        instance (Any): The instance the decorated function is bound to.
        name (str): The name of the attribute to be checked.

    Returns:
        bool: Whether the attribute exists and holds a provided value.
    """
    if instance is None or name.startswith('__'):
        return False
    try:
        if not hasattr(instance, name):
            return False
        value = getattr(instance, name)
    except Exception:  # pragma: no cover
        return False
    return _is_value_provided(value)


def _key_passed_directly(
    key: str,
    argument_name: Optional[str],
    bound_arguments: Mapping[str, Any],
    instance: Any,
    used_names: Set[str],
) -> bool:
    r"""Check whether the API key identified by an environment variable name
    has been passed directly to the decorated function, e.g. as the
    :obj:`api_key` argument of a decorated :obj:`__init__` method, or stored
    as an instance attribute.

    Every provided value is consumed by at most one requirement, so that
    having multiple required API keys can not be satisfied by a single
    value.

    Args:
        key (str): The name of the environment variable holding the key.
        argument_name (Optional[str]): The name of the argument that
            explicitly carries the key, if declared by the user.
        bound_arguments (Mapping[str, Any]): The arguments of the call bound
            to the parameter names of the decorated function.
        instance (Any): The instance the decorated function is bound to, if
            any.
        used_names (Set[str]): The names already used to satisfy another
            required key. Updated in place.

    Returns:
        bool: Whether the key has been provided directly.
    """
    candidate_names: List[str] = []
    if argument_name:
        candidate_names.append(argument_name)
    for name in _get_candidate_names(key):
        if name not in candidate_names:
            candidate_names.append(name)

    for name in candidate_names:
        if name in used_names or name in ('self', 'cls'):
            continue
        if name in bound_arguments and _is_value_provided(
            bound_arguments[name]
        ):
            used_names.add(name)
            return True
        if _instance_attribute_provided(instance, name):
            used_names.add(name)
            return True
    return False


def api_keys_required(*required_keys: str) -> Callable[[F], F]:
    r"""A decorator to check if the required API keys are
    presented in the environment variables, passed directly to the decorated
    function (e.g. through the ``api_key`` argument of an ``__init__``
    method), or stored as an instance attribute.

    Each required key can be given either as the name of the environment
    variable, or as a :obj:`(argument_name, environment_variable_name)`
    tuple, where :obj:`argument_name` is the name of the argument carrying

    Args:
        required_keys (str): The required API keys to be checked.

    Returns:
        Callable[[F], F]: The original function with the added check
            for required API keys.

    Raises:
        ValueError: If any of the required API keys are missing in the
            environment variables and are not passed to the decorated
            function nor stored as an instance attribute.

    Example:
        ::

            @api_keys_required('API_KEY_1', 'API_KEY_2')
            def some_api_function():
                # Function implementation...

            class SomeClient:

                @api_keys_required('OPENAI_API_KEY')
                def __init__(self, api_key: Optional[str] = None):
                    self._api_key = api_key

                @api_keys_required([(None, 'OPENAI_API_KEY')])
                def some_method(self):
                    # Function implementation...
    """

    keys: List[Any] = list(required_keys)
    # The required keys may also be given as a single list/tuple, e.g.
    # ``@api_keys_required(['API_KEY_1', 'API_KEY_2'])``.
    if len(keys) == 1 and isinstance(keys[0], (list, tuple, set, frozenset)):
        keys = list(keys[0])

    def decorator(func: F) -> F:
        try:
            parameters = list(inspect.signature(func).parameters)
        except (TypeError, ValueError):  # pragma: no cover
            parameters = []

        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            bound_arguments = _bind_arguments(func, args, kwargs)

            # Resolve the instance the decorated function is bound to, if
            # any, so that API keys stored as instance attributes (e.g.
            # ``self._api_key``) are taken into account.
            instance: Any = None
            if parameters:
                first_parameter = parameters[0]
                if first_parameter in bound_arguments:
                    instance = bound_arguments[first_parameter]
                elif args:
                    instance = args[0]
                if isinstance(
                    instance,
                    (
                        str,
                        bytes,
                        bytearray,
                        int,
                        float,
                        complex,
                        bool,
                        dict,
                        list,
                        tuple,
                        set,
                        frozenset,
                    ),
                ):
                    instance = None

            used_names: Set[str] = set()
            missing_keys: List[str] = []

            for entry in keys:
                argument_name, key = _parse_required_key(entry)
                if key is None:
                    # The key is explicitly marked as optional.
                    continue
                if key in os.environ:
                    # The key is presented in the environment variables.
                    continue
                if _key_passed_directly(
                    key,
                    argument_name,
                    bound_arguments,
                    instance,
                    used_names,
                ):
                    # The key has been passed directly to the function.
                    continue
                missing_keys.append(key)

            if missing_keys:
                raise ValueError(
                    f"Missing API keys: {', '.join(missing_keys)}"
                )
            return func(*args, **kwargs)

        return cast(F, wrapper)

    return decorator

def to_pascal(snake: str) -> str:
    """Convert a snake_case string to PascalCase.

    Args:
        snake (str): The snake_case string to be converted.

    Returns:
        str: The converted PascalCase string.
    """
    # Check if the string is already in PascalCase
    if re.match(r'^[A-Z][a-zA-Z0-9]*([A-Z][a-zA-Z0-9]*)*$', snake):
        return snake
    # Remove leading and trailing underscores
    snake = snake.strip('_')
    # Replace multiple underscores with a single one
    snake = re.sub('_+', '_', snake)
    # Convert to PascalCase
    return re.sub(
        '_([0-9A-Za-z])',
        lambda m: m.group(1).upper(),
        snake.title(),
    )

def func_string_to_callable(code: str):
    r"""Convert a function code string to a callable function object.

    Args:
        code (str): The function code as a string.

    Returns:
        Callable[..., Any]: The callable function object extracted from the
            code string.
    """
    local_vars: Mapping[str, object] = {}
    exec(code, globals(), local_vars)
    func = local_vars.get(Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT)
    return func


def json_to_function_code(json_obj: Dict) -> str:
    r"""Generate a Python function code from a JSON schema.

    Args:
        json_obj (dict): The JSON schema object containing properties and
            required fields, and json format is follow openai tools schema

    Returns:
        str: The generated Python function code as a string.
    """
    properties = json_obj.get('properties', {})
    required = json_obj.get('required', [])

    if not properties or not required:
        raise ValueError(
            "JSON schema must contain 'properties' and 'required' fields"
        )

    args = []
    docstring_args = []
    return_keys = []

    prop_to_python = {
        'string': 'str',
        'number': 'float',
        'integer': 'int',
        'boolean': 'bool',
    }

    for prop in required:
        description = properties[prop]['description']
        prop_type = properties[prop]['type']
        python_type = prop_to_python.get(prop_type, prop_type)
        args.append(f"{prop}: {python_type}")
        docstring_args.append(
            f"        {prop} ({python_type}): {description}."
        )
        return_keys.append(prop)

    # extract entity of schema
    args_str = ", ".join(args)
    docstring_args_str = "\n".join(docstring_args)
    return_keys_str = ", ".join(return_keys)

    # function template
    function_code = f'''
def {Constants.FUNC_NAME_FOR_STRUCTURED_OUTPUT}({args_str}):
    r"""Return response with a specified json format.
    Args:
{docstring_args_str}
    Returns:
        Dict: A dictionary containing {return_keys_str}.
    """
    return {{{", ".join([f'"{prop}": {prop}' for prop in required])}}}
    '''

    return function_code


def text_extract_from_web(url: str) -> str:
    r"""Get the text information from given url.

    Args:
        url (str): The website you want to search.

    Returns:
        str: All texts extract from the web.
    """
    try:
        import requests
        from newspaper import Article

        # Request the target page
        article = Article(url)
        article.download()
        article.parse()
        text = article.text

    except requests.RequestException as e:
        text = f"Can't access {url}, error: {e}"

    except Exception as e:
        text = f"Can't extract text from {url}, error: {e}"

    return text


def create_chunks(text: str, n: int) -> List[str]:
    r"""Returns successive n-sized chunks from provided text. Split a text
    into smaller chunks of size n".

    Args:
        text (str): The text to be split.
        n (int): The max length of a single chunk.

    Returns:
        List[str]: A list of split texts.
    """

    chunks = []
    i = 0
    while i < len(text):
        # Find the nearest end of sentence within a range of 0.5 * n
        # and 1.5 * n tokens
        j = min(i + int(1.2 * n), len(text))
        while j > i + int(0.8 * n):
            # Decode the tokens and check for full stop or newline
            chunk = text[i:j]
            if chunk.endswith(".") or chunk.endswith("\n"):
                break
            j -= 1
        # If no end of sentence found, use n tokens as the chunk size
        if j == i + int(0.8 * n):
            j = min(i + n, len(text))
        chunks.append(text[i:j])
        i = j
    return chunks


def is_docker_running() -> bool:
    r"""Check if the Docker daemon is running.

    Returns:
        bool: True if the Docker daemon is running, False otherwise.
    """
    try:
        result = subprocess.run(
            ["docker", "info"],
            check=True,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
        )
        return result.returncode == 0
    except (subprocess.CalledProcessError, FileNotFoundError):
        return False

def agentops_decorator(func):
    r"""Decorator that records the execution of a function if ToolEvent is
    available.

    Parameters:
        func (callable): The function to be decorated.

    Returns:
        callable: The wrapped function which records its execution details.
    """

    @wraps(func)
    def wrapper(*args, **kwargs):
        if ToolEvent:
            tool_event = ToolEvent(name=func.__name__, params=kwargs)
            result = func(*args, **kwargs)
            tool_event.returns = result
            record(tool_event)
            return result
        return func(*args, **kwargs)

    return wrapper


class AgentOpsMeta(type):
    r"""Metaclass that automatically decorates all callable attributes with
    the agentops_decorator,
    except for the 'get_tools' method.

    Methods:
    __new__(cls, name, bases, dct):
        Creates a new class with decorated methods.
    """

    def __new__(cls, name, bases, dct):
        if ToolEvent:
            for attr, value in dct.items():
                if callable(value) and attr != 'get_tools':
                    dct[attr] = agentops_decorator(value)
        return super().__new__(cls, name, bases, dct)

def handle_http_error(response: requests.Response) -> str:
    r"""Handles the HTTP errors based on the status code of the response.

    Args:
        response (requests.Response): The HTTP response from the API call.

    Returns:
        str: The error type, based on the status code.
    """
    if response.status_code == HTTPStatus.UNAUTHORIZED:
        return "Unauthorized. Check your access token."
    elif response.status_code == HTTPStatus.FORBIDDEN:
        return "Forbidden. You do not have permission to perform this action."
    elif response.status_code == HTTPStatus.NOT_FOUND:
        return "Not Found. The resource could not be located."
    elif response.status_code == HTTPStatus.TOO_MANY_REQUESTS:
        return "Too Many Requests. You have hit the rate limit."
    else:
        return "HTTP Error"


def retry_request(
    func: Callable, retries: int = 3, delay: int = 1, *args: Any, **kwargs: Any
) -> Any:
    r"""Retries a function in case of any errors.

    Args:
        func (Callable): The function to be retried.
        retries (int): Number of retry attempts. (default: :obj:`3`)
        delay (int): Delay between retries in seconds. (default: :obj:`1`)
        *args: Arguments to pass to the function.
        **kwargs: Keyword arguments to pass to the function.

    Returns:
        Any: The result of the function call if successful.

    Raises:
        Exception: If all retry attempts fail.
    """
    for attempt in range(retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            print(f"Attempt {attempt + 1}/{retries} failed: {e}")
            if attempt < retries - 1:
                time.sleep(delay)
            else:
                raise
