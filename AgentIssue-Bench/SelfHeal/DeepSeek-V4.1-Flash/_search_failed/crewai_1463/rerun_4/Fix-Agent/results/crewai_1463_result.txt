        # prerequisite finished (concurrent branches / routed trigger). Resolve
        # every AND listener whose requirements are now satisfied.
        await self._resolve_ready_and_listeners()

        # Return the final output (from the last executed method)
        if self._method_outputs:
            return self._method_outputs[-1]
        else:
            return None  # Or raise an exception if no methods were executed

    async def _execute_start_method(self, start_method_name: str) -> None:
        result = await self._execute_method(
            start_method_name, self._methods[start_method_name]
        )
        await self._execute_listeners(start_method_name, result)

    async def _execute_method(
        self, method_name: str, method: Callable, *args: Any, **kwargs: Any
    ) -> Any:
        result = (
            await method(*args, **kwargs)
            if asyncio.iscoroutinefunction(method)
            else method(*args, **kwargs)
        )
        self._method_outputs.append(result)  # Store the output

        self._executed_methods.add(method_name)

        return result

    def _is_settled(self, listener_name: str) -> bool:
        """Whether a listener already ran or is already queued to run."""
        return (
            listener_name in self._executed_methods
            or listener_name in self._scheduled_tasks
        )

    def _schedule_listener(
        self, listener_name: str, result: Any, listener_tasks: List[Any]
    ) -> None:
        """Queue a listener for execution, ensuring it only runs once."""
        if self._is_settled(listener_name):
            return

        self._scheduled_tasks.add(listener_name)
        listener_tasks.append(self._execute_single_listener(listener_name, result))

    def _and_condition_met(self, listener_name: str, methods: List[str]) -> bool:
        """An AND listener fires once every one of its triggers has fired.

        Triggers are remembered in ``_pending_and_listeners`` (so router paths and
        other non-method triggers count as well) and cross-checked against the
        set of already executed methods.
        """
        if not methods:
            return False

        pending = self._pending_and_listeners.get(listener_name, set())
        return all(
            method in pending or method in self._executed_methods
            for method in methods
        )

    async def _execute_router(self, trigger_method: str, result: Any) -> Any:
        """Execute the router attached to a trigger and return its path."""
        router_method_name = self._routers[trigger_method]
        router_method = self._methods[router_method_name]

        sig = inspect.signature(router_method)
        method_params = [p for p in sig.parameters.values() if p.name != "self"]

        if method_params:
            # The router expects the triggering result.
            path = await self._execute_method(
                router_method_name, router_method, result
            )
        else:
            path = await self._execute_method(router_method_name, router_method)

        if isinstance(path, str):
            return path

        # Routers may also return an object exposing the chosen route.
        route = getattr(path, "route", None)
        return route if isinstance(route, str) else None

    async def _execute_listeners(self, trigger_method: str, result: Any) -> None:
        """Dispatch every listener unlocked by ``trigger_method`` finishing."""
        listener_tasks: List[Any] = []

        # The trigger methods considered by this dispatch pass. A router hands
        # the flow over to the path it returns.
        triggers: List[str] = [trigger_method]

        if trigger_method in self._routers:
            path = await self._execute_router(trigger_method, result)
            if isinstance(path, str) and path:
                triggers = [path]

        for listener_name, (condition_type, methods) in self._listeners.items():
            # Routers are executed exactly once, by _execute_router() above.
            if listener_name in self._router_method_names:
                continue

            if self._is_settled(listener_name):
                continue

            if condition_type == "OR":
                if any(trigger in methods for trigger in triggers):
                    self._schedule_listener(listener_name, result, listener_tasks)

            elif condition_type == "AND":
                matched = [trigger for trigger in triggers if trigger in methods]
                if matched:
                    self._pending_and_listeners.setdefault(
                        listener_name, set()
                    ).update(matched)

                if self._and_condition_met(listener_name, methods):
                    # All requirements are satisfied: fire and reset the tracker
                    # so the listener can be triggered again by a later cycle.
                    self._pending_and_listeners.pop(listener_name, None)
                    self._schedule_listener(listener_name, result, listener_tasks)

        # Run all listener tasks concurrently and wait for them to complete
        await asyncio.gather(*listener_tasks)

    async def _resolve_ready_and_listeners(self) -> None:
        """Fire any AND listener whose requirements finished but never dispatched."""
        result = self._method_outputs[-1] if self._method_outputs else None
        attempted: Set[str] = set()

        while True:
            ready: List[str] = []
            for listener_name, (condition_type, methods) in self._listeners.items():
                if condition_type != "AND":
                    continue
                if listener_name in self._router_method_names:
                    continue
                if listener_name in attempted or self._is_settled(listener_name):
                    continue
                if self._and_condition_met(listener_name, methods):
                    self._pending_and_listeners.pop(listener_name, None)
                    self._scheduled_tasks.add(listener_name)
                    attempted.add(listener_name)
                    ready.append(listener_name)

            if not ready:
                return

            await asyncio.gather(
                *(
                    self._execute_single_listener(listener_name, result)
                    for listener_name in ready
                )
            )

    async def _execute_single_listener(self, listener_name: str, result: Any) -> None:
        try:
            method = self._methods[listener_name]
            sig = inspect.signature(method)
            params = list(sig.parameters.values())

            # Exclude 'self' parameter
            method_params = [p for p in params if p.name != "self"]

            if method_params:
                # If listener expects parameters, pass the result
                listener_result = await self._execute_method(
                    listener_name, method, result
                )
            else:
                # If listener does not expect parameters, call without arguments
                listener_result = await self._execute_method(listener_name, method)

            # Execute listeners of this listener
            await self._execute_listeners(listener_name, listener_result)
        except Exception as e:
            print(
                f"[Flow._execute_single_listener] Error in method {listener_name}: {e}"
            )
            import traceback

            traceback.print_exc()
        finally:
            # Always release the scheduling slot so a failing listener can be
            # retried and never blocks the dispatch of the remaining steps.
            self._scheduled_tasks.discard(listener_name)

    def plot(self, filename: str = "crewai_flow") -> None:
        self._telemetry.flow_plotting_span(
            self.__class__.__name__, list(self._methods.keys())
        )

        plot_flow(self, filename)

    def decorator(func):
        func.__is_start_method__ = True
        if condition is not None:
            if isinstance(condition, str):

                func.__condition_type__ = "OR"
            elif (

                and "methods" in condition

class FlowMeta(type):
    def __new__(mcs, name, bases, dct):
        cls = super().__new__(mcs, name, bases, dct)

        start_methods = []
        listeners = {}
        routers = {}
        router_paths = {}

        for attr_name, attr_value in dct.items():
            if hasattr(attr_value, "__is_start_method__"):
                start_methods.append(attr_name)
                if hasattr(attr_value, "__trigger_methods__"):
                    methods = attr_value.__trigger_methods__
                    condition_type = getattr(attr_value, "__condition_type__", "OR")
                    listeners[attr_name] = (condition_type, methods)
            elif hasattr(attr_value, "__trigger_methods__"):
                methods = attr_value.__trigger_methods__
                condition_type = getattr(attr_value, "__condition_type__", "OR")
                listeners[attr_name] = (condition_type, methods)
            elif hasattr(attr_value, "__is_router__"):
                routers[attr_value.__router_for__] = attr_name
                possible_returns = get_possible_return_constants(attr_value)
                if possible_returns:
                    router_paths[attr_name] = possible_returns

        setattr(cls, "_start_methods", start_methods)
        setattr(cls, "_listeners", listeners)
        setattr(cls, "_routers", routers)
        setattr(cls, "_router_paths", router_paths)

        return cls

            else method(*args, **kwargs)

    def _can_schedule(self, listener_name: str) -> bool:
        """Whether a listener can still be scheduled for execution."""
        return (
            listener_name not in self._executed_methods
            and listener_name not in self._scheduled_tasks
        )

        """All methods listed in an `and_()` condition must have been executed."""
        if not methods:

        return all(method in self._executed_methods for method in methods)

        return self._routers.get(trigger_method, "")

    async def _execute_listeners(self, trigger_method: str, result: Any) -> None:
        """Execute every listener/step that is triggered by ``trigger_method``."""
        listener_tasks: List[Any] = []

        # If a router is attached to this method, execute it (once) and use the
        # returned path as the new trigger method. Routers are executed under
        # their own name so they are properly registered as executed.
        router_name = self._router_trigger(trigger_method)
        if router_name:
            if router_name not in self._executed_methods:
                router_method = self._methods[router_name]
                router_result = await self._execute_method(router_name, router_method)
                result = router_result
                if isinstance(router_result, str) and router_result:
                    trigger_method = router_result
                else:
                    trigger_method = router_name
            elif router_name not in self._scheduled_tasks:
                trigger_method = router_name

        for listener_name, (condition_type, methods) in self._listeners.items():
            if condition_type == "OR":
                if trigger_method in methods and self._can_schedule(listener_name):
                    self._scheduled_tasks.add(listener_name)
                    listener_tasks.append(
                        self._execute_single_listener(listener_name, result)
                    )
            elif condition_type == "AND":
                # Track that this branch has been fulfilled, so `and_()`
                # listeners fire only once *all* of their methods completed,
                # regardless of which one finished last.
                pending = self._pending_and_listeners.setdefault(listener_name, set())
                pending.add(trigger_method)

                if self._and_condition_met(methods) and self._can_schedule(
                    listener_name
                ):
                    self._scheduled_tasks.add(listener_name)
                    listener_tasks.append(
                        self._execute_single_listener(listener_name, result)
                    )

        # Run all listener tasks concurrently and wait for them to complete
        if listener_tasks:
            await asyncio.gather(*listener_tasks)

    async def _drain_pending_and_listeners(self) -> None:
        """Execute any pending `and_()` listener whose conditions are satisfied."""
        attempted: Set[str] = set()

        while True:
            ready = [
                listener_name
                for listener_name, (condition_type, methods) in self._listeners.items()
                if condition_type == "AND"
                and listener_name not in attempted
                and self._and_condition_met(methods)
                and self._can_schedule(listener_name)
            ]

            if not ready:
                return

            attempted.update(ready)
            tasks = []
            for listener_name in ready:
                self._scheduled_tasks.add(listener_name)
                tasks.append(
                    self._execute_single_listener(listener_name, self._last_result())
                )
            await asyncio.gather(*tasks)

    async def _execute_single_listener(self, listener_name: str, result: Any) -> None:

            method = self._methods[listener_name]
            sig = inspect.signature(method)
            params = list(sig.parameters.values())

                # If listener expects parameters, pass the result
                listener_result = await self._execute_method(
                    listener_name, method, result
                )
            else:
                # If listener does not expect parameters, call without arguments
                listener_result = await self._execute_method(listener_name, method)

            # Execute listeners of this listener
            await self._execute_listeners(listener_name, listener_result)
        except Exception as e:

            )
            import traceback

            traceback.print_exc()
        finally:

            # failed listener never blocks the rest of the flow).
            self._scheduled_tasks.discard(listener_name)
