def _resolve_condition(condition: Any) -> Tuple[List[str], str]:
    """Normalizes a condition into (method names, condition type).

    A condition can be expressed as a method reference, as a method name
    (string) or as the result of ``or_()``/``and_()``.
    """
    if isinstance(condition, str):
        return [condition], "OR"
    if isinstance(condition, dict) and "type" in condition and "methods" in condition:
        return list(condition["methods"]), condition["type"]
    if callable(condition) and hasattr(condition, "__name__"):
        return [condition.__name__], "OR"
    raise ValueError(_CONDITION_ERROR)


def _flatten_conditions(conditions: Tuple[Any, ...], operator: str) -> List[str]:
    """Extracts the method names of the conditions given to or_()/and_()."""
    methods: List[str] = []
    for condition in conditions:
        if isinstance(condition, dict) and "methods" in condition:
            methods.extend(condition["methods"])
        elif isinstance(condition, str):
            methods.append(condition)
        elif callable(condition) and hasattr(condition, "__name__"):
            methods.append(condition.__name__)
        else:
            raise ValueError(f"Invalid condition in {operator}()")
    return methods


def start(condition=None):
    """Decorator that marks a method as a start method of a flow."""

    def decorator(func):
        func.__is_start_method__ = True
        if condition is not None:

            func.__trigger_methods__ = methods
            func.__condition_type__ = condition_type

def listen(condition):

    def decorator(func):
        methods, condition_type = _resolve_condition(condition)
        func.__trigger_methods__ = methods
        func.__condition_type__ = condition_type
        return func

    return decorator


def router(method):
    """Decorator that marks a method as the router of ``method``."""

    def decorator(func):
        func.__is_router__ = True
        func.__router_for__ = getattr(method, "__name__", method)
        return func

    return decorator


def or_(*conditions):
    """Builds a condition that triggers when **any** of the given ones happen."""
    return {"type": "OR", "methods": _flatten_conditions(conditions, "or_")}


def and_(*conditions):
    """Builds a condition that triggers only when **all** the given ones happen."""
    return {"type": "AND", "methods": _flatten_conditions(conditions, "and_")}
