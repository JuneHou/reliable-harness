def _parse_condition(condition: Any) -> Tuple[List[str], str]:
    """Normalizes a decorator condition into (methods, condition_type)."""
    if isinstance(condition, str):
        return [condition], "OR"

    if (
        isinstance(condition, dict)
        and "type" in condition
        and "methods" in condition
    ):
        return list(condition["methods"]), condition["type"]

    if callable(condition) and hasattr(condition, "__name__"):
        return [condition.__name__], "OR"

    raise ValueError(_CONDITION_ERROR)


def start(condition=None):
    def decorator(func):
        func.__is_start_method__ = True
        if condition is not None:
            methods, condition_type = _parse_condition(condition)
            func.__trigger_methods__ = methods
            func.__condition_type__ = condition_type
        return func

    return decorator


def listen(condition):
    def decorator(func):
        methods, condition_type = _parse_condition(condition)
        func.__trigger_methods__ = methods
        func.__condition_type__ = condition_type
        return func

    return decorator


def router(method):
    def decorator(func):
        func.__is_router__ = True
        func.__router_for__ = (
            method.__name__ if callable(method) else method
        )
        return func

    return decorator


def _collect_condition_methods(
    conditions: Tuple[Any, ...], decorator_name: str
) -> List[str]:
    """Flattens the given conditions into a list of method names."""
    methods: List[str] = []
    for condition in conditions:
        if isinstance(condition, dict) and "methods" in condition:
            methods.extend(condition["methods"])
        elif isinstance(condition, str):
            methods.append(condition)
        elif callable(condition):
            methods.append(getattr(condition, "__name__", repr(condition)))
        else:
            raise ValueError(f"Invalid condition in {decorator_name}()")
    return methods


def or_(*conditions):
    return {"type": "OR", "methods": _collect_condition_methods(conditions, "or_")}


def and_(*conditions):
    return {"type": "AND", "methods": _collect_condition_methods(conditions, "and_")}

        AND conditions require every one of their methods to have been triggered

    """Flatten the conditions passed to or_()/and_() into method names."""
    methods: List[str] = []

        if isinstance(condition, dict) and "methods" in condition:
            methods.extend(condition["methods"])
            continue

    # De-duplicate while preserving order so AND conditions are evaluated
    # deterministically even if a method is passed more than once.
    unique_methods: List[str] = []
    for method in methods:
        if method not in unique_methods:

    return unique_methods

        self._scheduled_tasks: Set[str] = set()
        # Tracks, per listener, which of its required triggers already fired.
        self._pending_and_listeners: Dict[str, Set[str]] = {}
        self._method_outputs: List[Any] = []  # List to store all method outputs

        self._telemetry.flow_creation_span(self.__class__.__name__)

        for method_name in dir(self):
            if method_name.startswith("__"):
                continue
            method = getattr(self, method_name, None)
            if callable(method):
                self._methods[method_name] = method

    def _create_initial_state(self) -> T:
        if self.initial_state is None and hasattr(self, "_initial_state_T"):
            return self._initial_state_T()  # type: ignore
        if self.initial_state is None:
            return {}  # type: ignore
        elif isinstance(self.initial_state, type):
            return self.initial_state()
        else:
            return self.initial_state

    @property
    def state(self) -> T:
        return self._state

    @property
    def method_outputs(self) -> List[Any]:
        """Returns the list of all outputs from executed methods."""
        return self._method_outputs

    def kickoff(self) -> Any:
        return asyncio.run(self.kickoff_async())

    async def kickoff_async(self) -> Any:
        if not self._start_methods:
            raise ValueError("No start method defined")

        self._telemetry.flow_execution_span(
            self.__class__.__name__, list(self._methods.keys())
        )

        # Reset the execution bookkeeping so a flow can be kicked off again.
        self._executed_methods = set()
        self._scheduled_tasks = set()
        self._pending_and_listeners = {}
        self._method_outputs = []

        # Create tasks for all start methods
        tasks = [
            self._execute_start_method(start_method)
            for start_method in self._start_methods
        ]

        # Run all start methods concurrently
        await asyncio.gather(*tasks)

        # Return the final output (from the last executed method)
        if self._method_outputs:
            return self._method_outputs[-1]
        else:
            return None  # Or raise an exception if no methods were executed

    async def _execute_start_method(self, start_method_name: str) -> None:
        result = await self._execute_method(
            start_method_name, self._methods[start_method_name]
        )
        await self._execute_listeners(start_method_name, result)

    async def _execute_method(
        self, method_name: str, method: Callable, *args: Any, **kwargs: Any
    ) -> Any:
        result = (
            await method(*args, **kwargs)
            if asyncio.iscoroutinefunction(method)
            else method(*args, **kwargs)
        )
        self._method_outputs.append(result)  # Store the output

        self._executed_methods.add(method_name)

        return result

    def _is_condition_met(
        self,
        listener_name: str,
        condition_type: str,
        methods: List[str],
        triggered_methods: Set[str],
    ) -> bool:
        """Checks whether a listener condition is satisfied.

        OR conditions are met as soon as one of their methods is triggered, while
        AND conditions require every one of their methods to have been triggered
        (or executed) at some point.
        """
        if not methods:
            return False

        if condition_type == "AND":
            pending = self._pending_and_listeners.setdefault(listener_name, set())
            # Keep track of every trigger observed for this listener so that a
            # prerequisite that already ran is never lost, even when the
            # remaining prerequisites finish later (or concurrently).
            pending |= set(methods) & triggered_methods
            pending |= set(methods) & self._executed_methods
            return set(methods).issubset(pending)

        # Default behavior is OR: any matching trigger fires the listener.
        return bool(set(methods) & triggered_methods)

    async def _execute_listeners(self, trigger_method: str, result: Any) -> None:
        listener_tasks = []
        triggered_methods = {trigger_method}

        # Handle routers first: a router is triggered by the method it routes from
        if trigger_method in self._routers:
            router_method_name = self._routers[trigger_method]
            router_method = self._methods[router_method_name]
            path = await self._execute_method(router_method_name, router_method)
            # Use the returned path as an additional trigger method
            if isinstance(path, str):
                triggered_methods.add(path)

        for listener_name, (condition_type, methods) in self._listeners.items():
            if (
                listener_name in self._executed_methods
                or listener_name in self._scheduled_tasks
            ):
                continue

            if self._is_condition_met(
                listener_name, condition_type, list(methods), triggered_methods
            ):
                self._scheduled_tasks.add(listener_name)
                self._pending_and_listeners.pop(listener_name, None)
                listener_tasks.append(
                    self._execute_single_listener(listener_name, result)
                )

        # Run all listener tasks concurrently and wait for them to complete
        await asyncio.gather(*listener_tasks)

    async def _execute_single_listener(self, listener_name: str, result: Any) -> None:
        try:
            method = self._methods[listener_name]
            sig = inspect.signature(method)
            params = list(sig.parameters.values())

            # Exclude 'self' parameter
            method_params = [p for p in params if p.name != "self"]

            if method_params:
                # If listener expects parameters, pass the result
                listener_result = await self._execute_method(
                    listener_name, method, result
                )
            else:
                # If listener does not expect parameters, call without arguments
                listener_result = await self._execute_method(listener_name, method)

            # Execute listeners of this listener
            await self._execute_listeners(listener_name, listener_result)
        except Exception as e:
            print(
                f"[Flow._execute_single_listener] Error in method {listener_name}: {e}"
            )
            import traceback

            traceback.print_exc()
        finally:
            # Remove from scheduled tasks after execution
            self._scheduled_tasks.discard(listener_name)

    def plot(self, filename: str = "crewai_flow") -> None:
        self._telemetry.flow_plotting_span(
            self.__class__.__name__, list(self._methods.keys())
        )

        plot_flow(self, filename)
