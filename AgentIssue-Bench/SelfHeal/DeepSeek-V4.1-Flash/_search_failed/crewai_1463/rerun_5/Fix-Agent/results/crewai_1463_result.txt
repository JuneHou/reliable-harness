def _resolve_condition(condition: Any) -> Tuple[str, List[str]]:
    """Normalize a decorator condition into a (condition_type, methods) tuple."""
    if isinstance(condition, str):
        return "OR", [condition]

    if isinstance(condition, dict) and "type" in condition and "methods" in condition:
        return condition["type"], list(condition["methods"])

    if callable(condition) and hasattr(condition, "__name__"):
        return "OR", [condition.__name__]

    raise ValueError(
        "Condition must be a method, string, or a result of or_() or and_()"
    )


def _collect_method_names(conditions: Tuple[Any, ...], function_name: str) -> List[str]:
    """Collect the method names referenced by the arguments of or_() / and_()."""
    methods: List[str] = []
    for condition in conditions:
        if isinstance(condition, dict) and "methods" in condition:
            methods.extend(condition["methods"])
        elif isinstance(condition, str):
            methods.append(condition)
        elif callable(condition):
            methods.append(getattr(condition, "__name__", repr(condition)))
        else:
            raise ValueError(f"Invalid condition in {function_name}()")
    return methods


def start(condition=None):
    def decorator(func):
        func.__is_start_method__ = True
        if condition is not None:
            condition_type, methods = _resolve_condition(condition)
            func.__trigger_methods__ = methods
            func.__condition_type__ = condition_type
        return func

    return decorator

        if not methods:
            return False

        triggered_conditions = self._pending_and_listeners.get(listener_name, set())

        await asyncio.gather(*listener_tasks)

        # Create tasks for all start methods
        tasks = [
            self._execute_start_method(start_method)
            for start_method in self._start_methods
        ]

        # Run all start methods concurrently
        await asyncio.gather(*tasks)

        # An AND condition might only be satisfied once every start method and
        # its listeners finished executing, so make sure any listener that is
        # ready but was never triggered gets executed before returning.
        await self._execute_pending_and_listeners()

        # Return the final output (from the last executed method)
        if self._method_outputs:
            return self._method_outputs[-1]
        else:
            return None  # Or raise an exception if no methods were executed

    async def _execute_start_method(self, start_method_name: str) -> None:
        result = await self._execute_method(
            start_method_name, self._methods[start_method_name]
        )
        await self._execute_listeners(start_method_name, result)

    async def _execute_method(
        self, method_name: str, method: Callable, *args: Any, **kwargs: Any
    ) -> Any:
        if asyncio.iscoroutinefunction(method):
            result = await method(*args, **kwargs)
        else:
            result = method(*args, **kwargs)
            # Support callables that return an awaitable without being
            # coroutine functions themselves.
            if inspect.isawaitable(result):
                result = await result

        self._method_outputs.append(result)  # Store the output

        self._executed_methods.add(method_name)

        return result

    def _is_listener_done(self, listener_name: str) -> bool:
        """Whether a listener already executed or is already scheduled."""
        return (
            listener_name in self._executed_methods
            or listener_name in self._scheduled_tasks
        )

    def _is_and_condition_met(self, listener_name: str, methods: List[str]) -> bool:
        """Check whether every method of an AND condition has been triggered.

        A condition can be a regular method (tracked in ``_executed_methods``)
        or a router path, which is just a string returned by a router and thus
        only tracked in ``_pending_and_listeners``.
        """
        if not methods:
            return False

        triggered_conditions = self._pending_and_listeners.get(listener_name, set())
        return all(
            method in self._executed_methods or method in triggered_conditions
            for method in methods
        )

    async def _execute_listeners(self, trigger_method: str, result: Any) -> None:
        """Execute all listeners that depend on the provided method."""
        # If the trigger method has a router, execute it and use the returned
        # path as the new trigger method.
        if trigger_method in self._routers:
            router_method_name = self._routers[trigger_method]
            router_method = self._methods[router_method_name]
            path = await self._execute_method(router_method_name, router_method)

            # Use the path as the new trigger method
            if isinstance(path, str):
                trigger_method = path

        router_methods = set(self._routers.values())
        listener_tasks = []

        for listener_name, (condition_type, methods) in self._listeners.items():
            # Routers are resolved above and never run as regular listeners
            if listener_name in router_methods:
                continue

            # A listener must never run twice
            if self._is_listener_done(listener_name):
                continue

            if condition_type == "AND":
                # Keep track of every condition triggered so far for this
                # listener, so the AND condition can be evaluated no matter the
                # order in which the triggering methods were executed.
                pending_conditions = self._pending_and_listeners.setdefault(
                    listener_name, set()
                )
                if trigger_method in methods:
                    pending_conditions.add(trigger_method)

                if not self._is_and_condition_met(listener_name, methods):
                    continue
            else:
                if trigger_method not in methods:
                    continue

            self._scheduled_tasks.add(listener_name)
            listener_tasks.append(
                self._execute_single_listener(listener_name, result)
            )

        # Run all listener tasks concurrently and wait for them to complete
        await asyncio.gather(*listener_tasks)

    async def _execute_pending_and_listeners(self) -> None:
        """Execute AND listeners whose conditions are met but never triggered."""
        while True:
            ready_listeners = [
                listener_name
                for listener_name, (condition_type, methods) in self._listeners.items()
                if condition_type == "AND"
                and listener_name not in set(self._routers.values())
                and not self._is_listener_done(listener_name)
                and self._is_and_condition_met(listener_name, methods)
            ]

            if not ready_listeners:
                return

            last_result = self._method_outputs[-1] if self._method_outputs else None
            listener_tasks = []

            for listener_name in ready_listeners:
                if self._is_listener_done(listener_name):
                    continue

                self._scheduled_tasks.add(listener_name)
                listener_tasks.append(
                    self._execute_single_listener(listener_name, last_result)
                )

            if not listener_tasks:
                return

            await asyncio.gather(*listener_tasks)

    async def _execute_single_listener(self, listener_name: str, result: Any) -> None:
        try:
            method = self._methods[listener_name]
            sig = inspect.signature(method)
            params = list(sig.parameters.values())

            # Exclude 'self' parameter
            method_params = [p for p in params if p.name != "self"]

            if method_params:
                # If listener expects parameters, pass the result
                listener_result = await self._execute_method(
                    listener_name, method, result
                )
            else:
                # If listener does not expect parameters, call without arguments
                listener_result = await self._execute_method(listener_name, method)

            # Execute listeners of this listener
            await self._execute_listeners(listener_name, listener_result)
        except Exception as e:
            print(
                f"[Flow._execute_single_listener] Error in method {listener_name}: {e}"
            )
            import traceback

            traceback.print_exc()
        finally:
            # Remove from scheduled tasks after execution
            self._scheduled_tasks.discard(listener_name)
            self._pending_and_listeners.pop(listener_name, None)

    def plot(self, filename: str = "crewai_flow") -> None:
        self._telemetry.flow_plotting_span(
            self.__class__.__name__, list(self._methods.keys())
        )

        plot_flow(self, filename)
