import { createSupabaseClient } from "../supabase/createSupabaseClient"
import { Chat } from "@/lib/queries/useChats"
import { useMutation, useQueryClient } from "@tanstack/react-query"
import { useSession } from "next-auth/react"
import { useAtom, useAtomValue } from "jotai"
import { uploadedFilesAtom, userWorkspaceAtom } from "@/lib/store"
import { SupabaseWorkspace } from "@/lib/supabase/SupabaseWorkspace"
import { useUpdateUserFiles } from "@/lib/hooks/useUpdateUserFiles"
import { Workspace } from "@evo-ninja/agent-utils"

interface WorkspaceFile {
  path: string;
  content: string;
}

// Reads every file that currently lives in the given workspace. Entries that
// can't be read (e.g. directories) are ignored.
const readWorkspaceFiles = async (workspace: Workspace): Promise<WorkspaceFile[]> => {
  const entries = await workspace.readdir("/")

  const files = await Promise.all(
    entries.map(async (entry) => {
      try {
        return {
          path: entry.name,
          content: await workspace.readFile(entry.name)
        }
      } catch {
        return null
      }
    })
  )

  return files.filter((file): file is WorkspaceFile => file !== null)
}

export const useCreateChat = () => {
  const { data: session } = useSession()
  const queryClient = useQueryClient()
  const userWorkspace = useAtomValue(userWorkspaceAtom)
  const [uploadedFiles, setUploadedFiles] = useAtom(uploadedFilesAtom)
  const updateUserFiles = useUpdateUserFiles()

  return useMutation({
    mutationFn: async (chatId: string) => {
      const supabase = createSupabaseClient(session?.supabaseAccessToken as string)
      const { data, error } = await supabase
        .from("chats")
        .insert({
          id: chatId
        })
        .select("id")

      if (error) {
        throw new Error(error.message);
      }

      // The new chat gets its own workspace, so every file that was uploaded
      // before the chat existed (in the anonymous workspace) has to be carried
      // over, otherwise it would be lost.
      const chatWorkspace = new SupabaseWorkspace(chatId, supabase.storage)
      const decoder = new TextDecoder()
      const files = new Map<string, string>()

      if (userWorkspace) {
        for (const file of await readWorkspaceFiles(userWorkspace)) {
          files.set(file.path, file.content)
        }
      }

      // Files that were uploaded but not written to the anonymous workspace yet.
      for (const file of uploadedFiles) {
        files.set(file.path, decoder.decode(file.content))
      }

      await Promise.all(
        [...files].map(([path, content]) => chatWorkspace.writeFile(path, content))
      )

      if (files.size > 0) {
        await updateUserFiles(chatWorkspace)
      }

      setUploadedFiles([])

      return data[0]
    },
    onMutate: async (chatId: string) => {
      await queryClient.cancelQueries({ queryKey: ['chats'] })
      queryClient.setQueryData<Chat[]>(['chats'], (old) => [...(old ?? []), {
        id: chatId,
        messages: [],
        logs: [],
        title: null,
        variables: new Map(),
        created_at: new Date().toISOString()
      }])
    },
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: ['chats'] })
    },
  })
}

const BUCKET_NAME = "workspaces";

    public readonly chatId: string,
    private readonly supabaseStorage: StorageClient
  ) {}

    const path = this.toWorkspacePath(subpath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)

    if (error) {
      throw error;
    }

    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    if (error) {
      throw error;
    }

    if (!data) {
      throw new Error("File not found");
    }

    return await data.text();
  }

  async exists(subpath: string): Promise<boolean> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    if (error) {
      throw error;
    }

    return data !== null;
  }

  async rename(oldPath: string, newPath: string): Promise<void> {
    const absOldPath = this.toWorkspacePath(oldPath);
    const absNewPath = this.toWorkspacePath(newPath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .move(absOldPath, absNewPath);

    if (error) {
      throw error;
    }
  }

  async mkdir(_subpath: string): Promise<void> {}

  async rmdir(subpath: string, opts?: { recursive: boolean }): Promise<void> {
    if (!opts?.recursive) {
      throw new Error("Non-recursive rmdir is not supported");
    }

    const { data: list, error: listError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .list(path);

    if (listError) {
      throw listError;
    }
    const filesToRemove = list.map((x) => `${path}/${x.name}`);

    if (filesToRemove.length === 0) {
      return;
    }

    const { error: removeError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .remove(filesToRemove);

    if (removeError) {
      throw removeError;
    }
  }

  async readdir(subpath: string): Promise<DirectoryEntry[]> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .list(path);

    if (error) {
      throw error;
    }

      throw new Error("Directory not found");

        type: "file",
      }));

    if (writeError) {
      throw writeError;
    }
  }

  async rm(subpath: string): Promise<void> {
    const path = this.toWorkspacePath(subpath);

    const { error } = await this.supabaseStorage

      .remove([path]);

    if (error) {
