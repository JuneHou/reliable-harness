import { createSupabaseClient } from "../supabase/createSupabaseClient"
import { Chat } from "@/lib/queries/useChats"
import { useMutation, useQueryClient } from "@tanstack/react-query"
import { useSession } from "next-auth/react"
import { useAtom } from "jotai"
import { userWorkspaceAtom } from "@/lib/store"
import { SupabaseWorkspace } from "../supabase/SupabaseWorkspace"

export const useCreateChat = () => {
  const { data: session } = useSession()
  const queryClient = useQueryClient()
  const [userWorkspace] = useAtom(userWorkspaceAtom)

  return useMutation({
    mutationFn: async (chatId: string) => {
      const supabase = createSupabaseClient(session?.supabaseAccessToken as string)
      const { data, error } = await supabase
        .from("chats")
        .insert({
          id: chatId
        })
        .select("id")

      if (error) {
        throw new Error(error.message);
      }

      // Creating a chat also creates a brand new workspace for it (scoped by
      // the chat id). Everything the user uploaded while working in the
      // previous workspace (for example the anonymous, in memory workspace)
      // lives there, so it has to be copied over to the workspace of the chat
      // that was just created. Otherwise the uploaded files would be lost as
      // soon as the chat is opened.
      const chatWorkspace = new SupabaseWorkspace(chatId, supabase.storage)
      const isOtherWorkspace =
        !(userWorkspace instanceof SupabaseWorkspace) || userWorkspace.chatId !== chatId

      if (isOtherWorkspace) {
        try {
          await chatWorkspace.copyFrom(userWorkspace)
        } catch (e) {
          // Failing to carry the files over should not prevent the chat from
          // being created
          console.error("Failed to carry over the uploaded files to the new chat", e)
        }
      }

      return data[0]
    },
    onMutate: async (chatId: string) => {
      await queryClient.cancelQueries({ queryKey: ['chats'] })
      queryClient.setQueryData<Chat[]>(['chats'], (old) => [...(old ?? []), {
        id: chatId,
        messages: [],
        logs: [],
        title: null,
        variables: new Map(),
        created_at: new Date().toISOString()
      }])
    },
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: ['chats'] })
    },
  })
}

import { uploadedFilesAtom, userWorkspaceAtom } from "@/lib/store";
import { useUpdateUserFiles } from "@/lib/hooks/useUpdateUserFiles";
import { useAtom } from "jotai";
import { useEffect } from "react";

export default function WorkspaceFilesProvider({ children }: { children: React.ReactNode }) {
  const [uploadedFiles, setUploadedFiles] = useAtom(uploadedFilesAtom);
  const [userWorkspace] = useAtom(userWorkspaceAtom);
  const updateUserFiles = useUpdateUserFiles();

  useEffect(() => {
    //eslint-disable-next-line @typescript-eslint/no-floating-promises
    (async () => {
      if (uploadedFiles.length === 0) {
        return;
      }

      const decoder = new TextDecoder();
      const files = [...uploadedFiles];

      // Take the files out of the queue before writing them, so that a
      // re-render (or a workspace change happening at the same time) cannot
      // write the very same file twice
      setUploadedFiles((current) => current.filter((x) => !files.includes(x)));

      // Write all uploaded files. A single failing file must not prevent the
      // remaining ones from being written to the workspace
      const results = await Promise.all(
        files.map(async (file) => {
          try {
            await userWorkspace.writeFile(
              file.path,
              decoder.decode(file.content)
            );
            return true;
          } catch (e) {
            console.error(`Failed to write ${file.path} to the workspace`, e);
            return false;
          }
        })
      );

      if (!results.some((x) => x)) {
        return;
      }

      try {
        await updateUserFiles(userWorkspace);
      } catch (e) {
        console.error("Failed to update the user files", e);
      }
    })();
  }, [uploadedFiles, userWorkspace]);

  return (
    <>
      {children}
    </>
  )
}


### apps/browser/lib/supabase/SupabaseWorkspace.ts
import { DirectoryEntry, Workspace } from "@evo-ninja/agent-utils";
import { StorageClient } from "@supabase/storage-js";
import * as path from "path-browserify";

const BUCKET_NAME = "workspaces";

const isNotFoundError = (
  error: { message?: string; statusCode?: string | number } | null
): boolean => {
  if (!error) {
    return false;
  }

  const message = (error.message ?? "").toLowerCase();

  return message.includes("not found") || `${error.statusCode ?? ""}` === "404";
};

export class SupabaseWorkspace implements Workspace {
  constructor(
    public readonly chatId: string,
    private readonly supabaseStorage: StorageClient
  ) {}

  async writeFile(subpath: string, data: string): Promise<void> {
    const path = this.toWorkspacePath(subpath);

    // Files are copied over from the workspace the user was previously working
    // in and can be re-uploaded with the same name, so the content is always
    // overwritten instead of failing when the file already exists
    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .upload(path, data, { upsert: true });

    if (error) {
      throw error;
    }

    return;
  }

  async readFile(subpath: string): Promise<string> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    if (error) {
      throw error;
    }

    if (!data) {
      throw new Error("File not found");
    }

    return await data.text();
  }

  async exists(subpath: string): Promise<boolean> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    // A missing file is not an error, it simply means that it does not exist
    if (error && !isNotFoundError(error)) {
      throw error;
    }

    return !!data;
  }

  async rename(oldPath: string, newPath: string): Promise<void> {
    const absOldPath = this.toWorkspacePath(oldPath);
    const absNewPath = this.toWorkspacePath(newPath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .move(absOldPath, absNewPath);

    if (error) {
      throw error;
    }
  }

  async mkdir(_subpath: string): Promise<void> {}

  async rmdir(subpath: string, opts?: { recursive: boolean }): Promise<void> {
    if (!opts?.recursive) {
      throw new Error("Non-recursive rmdir is not supported");
    }

    const path = this.toWorkspacePath(subpath);

    const { data: list, error: listError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .list(path);

    if (listError) {
      throw listError;
    }

    const filesToRemove = list.map((x) => `${path}/${x.name}`);

    if (filesToRemove.length === 0) {
      return;
    }

    const { error: removeError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .remove(filesToRemove);

    if (removeError) {
      throw removeError;
    }
  }

  async readdir(subpath: string): Promise<DirectoryEntry[]> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .list(path);

    if (error) {
      throw error;
    }

    if (!data) {
      throw new Error("Directory not found");
    }

    return data
      .filter((x) => !x.name.startsWith("."))
      .map<DirectoryEntry>((x) => ({
        name: x.name,
        type: x.id ? "file" : "directory",
      }));
  }

  async appendFile(subpath: string, data: string): Promise<void> {
    let existingData = "";

    try {
      existingData = await this.readFile(subpath);
    } catch (e) {
      // The file does not exist yet, so it is created with the given data
      if (!isNotFoundError(e as { message?: string; statusCode?: string | number })) {
        console.error(`Failed to read ${subpath} before appending to it`, e);
      }
    }

    // The previous content has to be awaited, otherwise its promise is
    // concatenated with the data (resulting in a corrupted file)
    await this.writeFile(subpath, existingData + data);
  }

  async rm(subpath: string): Promise<void> {
    const path = this.toWorkspacePath(subpath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .remove([path]);

    if (error) {
      throw error;
    }
  }

  /**
   * Copies every file of the given workspace into this workspace, preserving
   * the (relative) paths of the files.
   *
   * This is used to carry over the files a user uploaded before the chat (and
   * therefore the chat workspace) existed, e.g. files uploaded in the
   * anonymous workspace.
   */
  async copyFrom(source: Workspace, subpath: string = "/"): Promise<void> {
    let entries: DirectoryEntry[] = [];

    try {
      entries = await source.readdir(subpath);
    } catch {
      // The directory does not exist (yet), there is nothing to copy
      return;
    }

    await Promise.all(
      entries.map(async (entry) => {
        const entryPath = path.join(subpath, entry.name);

        if (entry.type === "directory") {
          await this.copyFrom(source, entryPath);
          return;
        }

        const content = await source.readFile(entryPath);
        await this.writeFile(entryPath, content);
      })
    );
  }

  async exec(
    _command: string,
    _args?: string[],
    _timeout?: number
  ): Promise<{ exitCode: number; stdout: string; stderr: string }> {
    throw new Error(
      "Executing commands is not supported in this application environment."
    );
  }

  private toWorkspacePath(subpath: string): string {
    return path.resolve("/", this.chatId, subpath).slice(1);
  }
}

import { useAtom } from "jotai";
import { uploadedFilesAtom } from "../store";


export function useUploadFiles() {
  const [, setUploadedFiles] = useAtom(uploadedFilesAtom);

  const { acceptedFiles, getRootProps, getInputProps, isDragAccept, open } = useDropzone({ noClick: true });

  useEffect(() => {
    (async () => {
      if (acceptedFiles && acceptedFiles.length) {
        const result = await Promise.all(
          acceptedFiles.map(async (x) => {
            return await readFile(x);
          })
        );

        // Newly dropped files are added to the files that are still waiting to
        // be written to the workspace (instead of replacing them), so that
        // files dropped in several batches are all kept
        setUploadedFiles((current: InMemoryFile[]) => [
          ...current.filter((x) => !result.some((y) => y.path === x.path)),
          ...result,
        ]);
      }
    })();
  }, [acceptedFiles, setUploadedFiles]);

  return {
    getRootProps, getInputProps, isDragAccept, open
  }
}

import { createSupabaseClient } from "../supabase/createSupabaseClient"
import { Chat } from "@/lib/queries/useChats"
import { useMutation, useQueryClient } from "@tanstack/react-query"
import { useSession } from "next-auth/react"
import { useAtom } from "jotai"
import { userWorkspaceAtom } from "@/lib/store"
import { SupabaseWorkspace } from "../supabase/SupabaseWorkspace"

export const useCreateChat = () => {
  const { data: session } = useSession()
  const queryClient = useQueryClient()
  const [userWorkspace] = useAtom(userWorkspaceAtom)

  return useMutation({
    mutationFn: async (chatId: string) => {
      const supabase = createSupabaseClient(session?.supabaseAccessToken as string)
      const { data, error } = await supabase
        .from("chats")
        .insert({
          id: chatId
        })
        .select("id")

      if (error) {
        throw new Error(error.message);
      }

      // Creating a chat also creates a brand new workspace for it (scoped by
      // the chat id). Everything the user uploaded while working in the
      // previous workspace (for example the anonymous, in memory workspace)
      // lives there, so it has to be copied over to the workspace of the chat
      // that was just created. Otherwise the uploaded files would be lost as
      // soon as the chat is opened.
      const chatWorkspace = new SupabaseWorkspace(chatId, supabase.storage)
      const isOtherWorkspace =
        !(userWorkspace instanceof SupabaseWorkspace) || userWorkspace.chatId !== chatId

      if (isOtherWorkspace) {
        try {
          await chatWorkspace.copyFrom(userWorkspace)
        } catch (e) {
          // Failing to carry the files over should not prevent the chat from
          // being created
          console.error("Failed to carry over the uploaded files to the new chat", e)
        }
      }

      return data[0]
    },
    onMutate: async (chatId: string) => {
      await queryClient.cancelQueries({ queryKey: ['chats'] })
      queryClient.setQueryData<Chat[]>(['chats'], (old) => [...(old ?? []), {
        id: chatId,
        messages: [],
        logs: [],
        title: null,
        variables: new Map(),
        created_at: new Date().toISOString()
      }])
    },
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: ['chats'] })
    },
  })
}

### apps/browser/components/providers/WorkspaceFilesProvider.tsx
import { uploadedFilesAtom, userWorkspaceAtom } from "@/lib/store";
import { useUpdateUserFiles } from "@/lib/hooks/useUpdateUserFiles";
import { useAtom } from "jotai";
import { useEffect } from "react";

export default function WorkspaceFilesProvider({ children }: { children: React.ReactNode }) {
  const [uploadedFiles, setUploadedFiles] = useAtom(uploadedFilesAtom);
  const [userWorkspace] = useAtom(userWorkspaceAtom);
  const updateUserFiles = useUpdateUserFiles();

  useEffect(() => {
    //eslint-disable-next-line @typescript-eslint/no-floating-promises
    (async () => {
      if (uploadedFiles.length === 0) {
        return;
      }

      const decoder = new TextDecoder();
      const files = [...uploadedFiles];

      // Take the files out of the queue before writing them, so that a
      // re-render (or a workspace change happening at the same time) cannot
      // write the very same file twice
      setUploadedFiles((current) => current.filter((x) => !files.includes(x)));

      // Write all uploaded files. A single failing file must not prevent the
      // remaining ones from being written to the workspace
      const results = await Promise.all(
        files.map(async (file) => {
          try {
            await userWorkspace.writeFile(
              file.path,
              decoder.decode(file.content)
            );
            return true;
          } catch (e) {
            console.error(`Failed to write ${file.path} to the workspace`, e);
            return false;
          }
        })
      );

      if (!results.some((x) => x)) {
        return;
      }

      try {
        await updateUserFiles(userWorkspace);
      } catch (e) {
        console.error("Failed to update the user files", e);
      }
    })();
  }, [uploadedFiles, userWorkspace]);

  return (
    <>
      {children}
    </>
  )
}

import { DirectoryEntry, Workspace } from "@evo-ninja/agent-utils";
import { StorageClient } from "@supabase/storage-js";
import * as path from "path-browserify";

const BUCKET_NAME = "workspaces";

const isNotFoundError = (
  error: { message?: string; statusCode?: string | number } | null
): boolean => {
  if (!error) {
    return false;
  }

  const message = (error.message ?? "").toLowerCase();

  return message.includes("not found") || `${error.statusCode ?? ""}` === "404";
};

export class SupabaseWorkspace implements Workspace {
  constructor(
    public readonly chatId: string,
    private readonly supabaseStorage: StorageClient
  ) {}

  async writeFile(subpath: string, data: string): Promise<void> {
    const path = this.toWorkspacePath(subpath);

    // Files are copied over from the workspace the user was previously working
    // in and can be re-uploaded with the same name, so the content is always
    // overwritten instead of failing when the file already exists
    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .upload(path, data, { upsert: true });

    if (error) {
      throw error;
    }

    return;
  }

  async readFile(subpath: string): Promise<string> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    if (error) {
      throw error;
    }

    if (!data) {
      throw new Error("File not found");
    }

    return await data.text();
  }

  async exists(subpath: string): Promise<boolean> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    // A missing file is not an error, it simply means that it does not exist
    if (error && !isNotFoundError(error)) {
      throw error;
    }

    return !!data;
  }

  async rename(oldPath: string, newPath: string): Promise<void> {
    const absOldPath = this.toWorkspacePath(oldPath);
    const absNewPath = this.toWorkspacePath(newPath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .move(absOldPath, absNewPath);

    if (error) {
      throw error;
    }
  }

  async mkdir(_subpath: string): Promise<void> {}

  async rmdir(subpath: string, opts?: { recursive: boolean }): Promise<void> {
    if (!opts?.recursive) {
      throw new Error("Non-recursive rmdir is not supported");
    }

    const path = this.toWorkspacePath(subpath);

    const { data: list, error: listError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .list(path);

    if (listError) {
      throw listError;
    }

    const filesToRemove = list.map((x) => `${path}/${x.name}`);

    if (filesToRemove.length === 0) {
      return;
    }

    const { error: removeError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .remove(filesToRemove);

    if (removeError) {
      throw removeError;
    }
  }

  async readdir(subpath: string): Promise<DirectoryEntry[]> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .list(path);

    if (error) {
      throw error;
    }

    if (!data) {
      throw new Error("Directory not found");
    }

    return data
      .filter((x) => !x.name.startsWith("."))
      .map<DirectoryEntry>((x) => ({
        name: x.name,
        type: x.id ? "file" : "directory",
      }));
  }

  async appendFile(subpath: string, data: string): Promise<void> {
    let existingData = "";

    try {
      existingData = await this.readFile(subpath);
    } catch (e) {
      // The file does not exist yet, so it is created with the given data
      if (!isNotFoundError(e as { message?: string; statusCode?: string | number })) {
        console.error(`Failed to read ${subpath} before appending to it`, e);
      }
    }

    // The previous content has to be awaited, otherwise its promise is
    // concatenated with the data (resulting in a corrupted file)
    await this.writeFile(subpath, existingData + data);
  }

  async rm(subpath: string): Promise<void> {
    const path = this.toWorkspacePath(subpath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .remove([path]);

    if (error) {
      throw error;
    }
  }

  /**
   * Copies every file of the given workspace into this workspace, preserving
   * the (relative) paths of the files.
   *
   * This is used to carry over the files a user uploaded before the chat (and
   * therefore the chat workspace) existed, e.g. files uploaded in the
   * anonymous workspace.
   */
  async copyFrom(source: Workspace, subpath: string = "/"): Promise<void> {
    let entries: DirectoryEntry[] = [];

    try {
      entries = await source.readdir(subpath);
    } catch {
      // The directory does not exist (yet), there is nothing to copy
      return;
    }

    await Promise.all(
      entries.map(async (entry) => {
        const entryPath = path.join(subpath, entry.name);

        if (entry.type === "directory") {
          await this.copyFrom(source, entryPath);
          return;
        }

        const content = await source.readFile(entryPath);
        await this.writeFile(entryPath, content);
      })
    );
  }

  async exec(
    _command: string,
    _args?: string[],
    _timeout?: number
  ): Promise<{ exitCode: number; stdout: string; stderr: string }> {
    throw new Error(
      "Executing commands is not supported in this application environment."
    );
  }

  private toWorkspacePath(subpath: string): string {
    return path.resolve("/", this.chatId, subpath).slice(1);
  }
