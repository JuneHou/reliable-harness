import { useAtom } from "jotai"

const readDirEntries = async (
  workspace: Workspace,
  subpath?: string
): Promise<DirectoryEntry[]> => {
  if (subpath) {
    return workspace.readdir(subpath);
  }

  // Different workspace implementations refer to the root differently
  for (const root of ["/", ""]) {
    try {
      return await workspace.readdir(root);
    } catch {
      // try the next root notation
    }
  }

  return [];
}

/**
 * Recursively copies the content of `source` into `target`.
 * This is used to carry over the files that were uploaded while the user
 * was still in the anonymous workspace into the workspace of a new chat.
 */
const copyWorkspace = async (
  source: Workspace,
  target: Workspace,
  subpath?: string
): Promise<void> => {
  const entries = await readDirEntries(source, subpath);
  const parent = (subpath ?? "").replace(/^\/+/, "").replace(/\/+$/, "");

  for (const entry of entries) {
    const entryPath = parent ? `${parent}/${entry.name}` : entry.name;

    if (entry.type === "directory") {
      await target.mkdir(entryPath);
      await copyWorkspace(source, target, entryPath);
      continue;
    }

    const content = await source.readFile(entryPath);
    await target.writeFile(entryPath, content);
  }
}

export const useCreateChat = () => {
  const { data: session } = useSession()
  const queryClient = useQueryClient()
  const [userWorkspace, setUserWorkspace] = useAtom(userWorkspaceAtom)
  const [uploadedFiles, setUploadedFiles] = useAtom(uploadedFilesAtom)

  return useMutation({
    mutationFn: async (chatId: string) => {
      const supabase = createSupabaseClient(session?.supabaseAccessToken as string)
      const { data, error } = await supabase
        .from("chats")
        .insert({
          id: chatId
        })
        .select("id")

      if (error) {
        throw new Error(error.message);
      }

      const chatWorkspace = new SupabaseWorkspace(chatId, supabase.storage)

      // Files the user uploaded while using the anonymous (in memory)
      // workspace have to be carried over into the workspace of the new
      // chat, otherwise they are lost as soon as the chat is created.
      const currentChatId = "chatId" in userWorkspace
        ? (userWorkspace as SupabaseWorkspace).chatId
        : undefined;

      if (currentChatId !== chatId) {
        try {
          await copyWorkspace(userWorkspace, chatWorkspace);
        } catch (copyError) {
          // Creating the chat must not fail because a workspace file could
          // not be copied, so we log the error and continue.
          console.error("Failed to carry over workspace files", copyError);
        }
      }

      // Files that are still staged (uploaded but not yet flushed into a
      // workspace) must also end up in the workspace of the new chat.
      if (uploadedFiles.length > 0) {
        setUploadedFiles([]);

        const decoder = new TextDecoder();
        await Promise.all(
          uploadedFiles.map((file) =>
            chatWorkspace.writeFile(
              file.path,
              decoder.decode(file.content)
            )
          )
        );
      }

      setUserWorkspace(chatWorkspace);

      return data[0]
    },
    onMutate: async (chatId: string) => {
      await queryClient.cancelQueries({ queryKey: ['chats'] })
      queryClient.setQueryData<Chat[]>(['chats'], (old) => [...(old ?? []), {
        id: chatId,
        messages: [],
        logs: [],
        title: null,
        variables: new Map(),
        created_at: new Date().toISOString()
      }])
    },
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: ['chats'] })
    },
  })
}

export default function WorkspaceFilesProvider({ children }: { children: React.ReactNode }) {

  const [userWorkspace] = useAtom(userWorkspaceAtom);
  const updateUserFiles = useUpdateUserFiles();

  useEffect(() => {

        return;
      }

      const files = [...uploadedFiles];
      // Take the files out of the queue right away so a file is never
      // written twice
      setUploadedFiles([]);

        // Write all uploaded files
        await Promise.all(
          files.map((file) =>
            userWorkspace.writeFile(
              file.path,
              decoder.decode(file.content)

        console.error("Failed to write uploaded files to the workspace", error);

    // staged get flushed into the workspace as soon as it changes

      {children}
    </>

### apps/browser/lib/supabase/SupabaseWorkspace.ts
import { DirectoryEntry, Workspace } from "@evo-ninja/agent-utils";
import { StorageClient } from "@supabase/storage-js";
import * as path from "path-browserify";

const BUCKET_NAME = "workspaces";

export class SupabaseWorkspace implements Workspace {
  constructor(
    public readonly chatId: string,
    private readonly supabaseStorage: StorageClient
  ) {}

  async writeFile(subpath: string, data: string): Promise<void> {
    const path = this.toWorkspacePath(subpath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      // Upsert, files are copied over from the anonymous workspace when a
      // chat is created and writing them must not fail because the object
      // already exists
      .upload(path, data, { upsert: true });

    if (error) {
      throw error;
    }

    return;
  }

  async readFile(subpath: string): Promise<string> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    if (error) {
      throw error;
    }

    if (!data) {
      throw new Error("File not found");
    }

    return await data.text();
  }

  async exists(subpath: string): Promise<boolean> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    if (error) {
      // A missing file is a valid answer for "does this file exist?" and
      // must not be reported as an error
      if (!data || isNotFoundError(error)) {
        return false;
      }

      throw error;
    }

    return data !== null;
  }

  async rename(oldPath: string, newPath: string): Promise<void> {
    const absOldPath = this.toWorkspacePath(oldPath);
    const absNewPath = this.toWorkspacePath(newPath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .move(absOldPath, absNewPath);

    if (error) {
      throw error;
    }
  }

  async mkdir(_subpath: string): Promise<void> {}

  async rmdir(subpath: string, opts?: { recursive: boolean }): Promise<void> {
    if (!opts?.recursive) {
      throw new Error("Non-recursive rmdir is not supported");
    }

    const path = this.toWorkspacePath(subpath);

    const { data: list, error: listError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .list(path);

    if (listError) {
      throw listError;
    }
    const filesToRemove = list.map((x) => `${path}/${x.name}`);

    if (filesToRemove.length === 0) {
      return;
    }

    const { error: removeError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .remove(filesToRemove);

    if (removeError) {
      throw removeError;
    }
  }

  async readdir(subpath: string): Promise<DirectoryEntry[]> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .list(path);

    if (error) {
      throw error;
    }

    if (!data) {
      throw new Error("Directory not found");
    }

    return data
      .filter((x) => !x.name.startsWith("."))
      .map((x) => ({
        name: x.name,
        // Supabase only returns an id for objects, folders have none
        type: x.id ? ("file" as const) : ("directory" as const),
      }));
  }

  async appendFile(subpath: string, data: string): Promise<void> {
    const path = this.toWorkspacePath(subpath);

    const { data: existingData, error: readError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    // The file does not need to exist yet in order to append to it
    let newData = data;

    if (!readError && existingData) {
      newData = (await existingData.text()) + data;
    }

    const { error: writeError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .upload(path, newData, { upsert: true });

    if (writeError) {
      throw writeError;
    }
  }

  async rm(subpath: string): Promise<void> {
    const path = this.toWorkspacePath(subpath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .remove([path]);

    if (error) {
      throw error;
    }
  }

  async exec(
    _command: string,
    _args?: string[],
    _timeout?: number
  ): Promise<{ exitCode: number; stdout: string; stderr: string }> {
    throw new Error(
      "Executing commands is not supported in this application environment."
    );
  }

  private toWorkspacePath(subpath: string): string {
    return path.resolve("/", this.chatId, subpath).slice(1);
  }
}

const isNotFoundError = (error: { message?: string; statusCode?: string | number }): boolean => {
  return (
    error?.statusCode === "404" ||
    error?.statusCode === 404 ||
    (error?.message ?? "").toLowerCase().includes("not found")
  );
}

import { useEffect } from "react";
import { useDropzone } from "react-dropzone";
import { InMemoryFile } from "@nerfzael/memory-fs";
import { readFile } from "../sys/file";
import { useAtom } from "jotai";
import { uploadedFilesAtom } from "../store";


export function useUploadFiles() {
  const [, setUploadedFiles] = useAtom(uploadedFilesAtom);

  const { acceptedFiles, getRootProps, getInputProps, isDragAccept, open } = useDropzone({ noClick: true });

  useEffect(() => {
    (async () => {
      if (acceptedFiles && acceptedFiles.length) {
        const result = await Promise.all(
          acceptedFiles.map(async (x) => {
            return await readFile(x);
          })
        );

        // Append the newly uploaded files to the ones that are already
        // staged, otherwise uploading files one by one (or in several
        // batches) silently discards the previously uploaded ones
        setUploadedFiles((files: InMemoryFile[]) => [...files, ...result]);
      }
    })();
  }, [acceptedFiles, setUploadedFiles]);

  return {
    getRootProps, getInputProps, isDragAccept, open
  }
}

### apps/browser/lib/supabase/SupabaseWorkspace.ts
import { DirectoryEntry, Workspace } from "@evo-ninja/agent-utils";
import { StorageClient } from "@supabase/storage-js";
import * as path from "path-browserify";

const BUCKET_NAME = "workspaces";

export class SupabaseWorkspace implements Workspace {
  constructor(
    public readonly chatId: string,
    private readonly supabaseStorage: StorageClient
  ) {}

  async writeFile(subpath: string, data: string): Promise<void> {
    const path = this.toWorkspacePath(subpath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      // Upsert, files are copied over from the anonymous workspace when a
      // chat is created and writing them must not fail because the object
      // already exists
      .upload(path, data, { upsert: true });

    if (error) {
      throw error;
    }

    return;
  }

  async readFile(subpath: string): Promise<string> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    if (error) {
      throw error;
    }

    if (!data) {
      throw new Error("File not found");
    }

    return await data.text();
  }

  async exists(subpath: string): Promise<boolean> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    if (error) {
      // A missing file is a valid answer for "does this file exist?" and
      // must not be reported as an error
      if (!data || isNotFoundError(error)) {
        return false;
      }

      throw error;
    }

    return data !== null;
  }

  async rename(oldPath: string, newPath: string): Promise<void> {
    const absOldPath = this.toWorkspacePath(oldPath);
    const absNewPath = this.toWorkspacePath(newPath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .move(absOldPath, absNewPath);

    if (error) {
      throw error;
    }
  }

  async mkdir(_subpath: string): Promise<void> {}

  async rmdir(subpath: string, opts?: { recursive: boolean }): Promise<void> {
    if (!opts?.recursive) {
      throw new Error("Non-recursive rmdir is not supported");
    }

    const path = this.toWorkspacePath(subpath);

    const { data: list, error: listError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .list(path);

    if (listError) {
      throw listError;
    }
    const filesToRemove = list.map((x) => `${path}/${x.name}`);

    if (filesToRemove.length === 0) {
      return;
    }

    const { error: removeError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .remove(filesToRemove);

    if (removeError) {
      throw removeError;
    }
  }

  async readdir(subpath: string): Promise<DirectoryEntry[]> {
    const path = this.toWorkspacePath(subpath);

    const { data, error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .list(path);

    if (error) {
      throw error;
    }

    if (!data) {
      throw new Error("Directory not found");
    }

    return data
      .filter((x) => !x.name.startsWith("."))
      .map((x) => ({
        name: x.name,
        // Supabase only returns an id for objects, folders have none
        type: x.id ? ("file" as const) : ("directory" as const),
      }));
  }

  async appendFile(subpath: string, data: string): Promise<void> {
    const path = this.toWorkspacePath(subpath);

    const { data: existingData, error: readError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .download(path);

    // The file does not need to exist yet in order to append to it
    let newData = data;

    if (!readError && existingData) {
      newData = (await existingData.text()) + data;
    }

    const { error: writeError } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .upload(path, newData, { upsert: true });

    if (writeError) {
      throw writeError;
    }
  }

  async rm(subpath: string): Promise<void> {
    const path = this.toWorkspacePath(subpath);

    const { error } = await this.supabaseStorage
      .from(BUCKET_NAME)
      .remove([path]);

    if (error) {
      throw error;
    }
  }

  async exec(
    _command: string,
    _args?: string[],
    _timeout?: number
  ): Promise<{ exitCode: number; stdout: string; stderr: string }> {
    throw new Error(
      "Executing commands is not supported in this application environment."
    );
  }

  private toWorkspacePath(subpath: string): string {
    return path.resolve("/", this.chatId, subpath).slice(1);
  }
}

const isNotFoundError = (error: { message?: string; statusCode?: string | number }): boolean => {
  return (
    error?.statusCode === "404" ||
    error?.statusCode === 404 ||
    (error?.message ?? "").toLowerCase().includes("not found")
  );
}

### apps/browser/lib/hooks/useUploadFile.ts
import { useEffect } from "react";
import { useDropzone } from "react-dropzone";
import { InMemoryFile } from "@nerfzael/memory-fs";
import { readFile } from "../sys/file";
import { useAtom } from "jotai";
import { uploadedFilesAtom } from "../store";


export function useUploadFiles() {
  const [, setUploadedFiles] = useAtom(uploadedFilesAtom);

  const { acceptedFiles, getRootProps, getInputProps, isDragAccept, open } = useDropzone({ noClick: true });

  useEffect(() => {
    (async () => {
      if (acceptedFiles && acceptedFiles.length) {
        const result = await Promise.all(
          acceptedFiles.map(async (x) => {
            return await readFile(x);
          })
        );

        // Append the newly uploaded files to the ones that are already
        // staged, otherwise uploading files one by one (or in several
        // batches) silently discards the previously uploaded ones
        setUploadedFiles((files: InMemoryFile[]) => [...files, ...result]);
      }
    })();
  }, [acceptedFiles, setUploadedFiles]);

  return {
    getRootProps, getInputProps, isDragAccept, open
  }
}
