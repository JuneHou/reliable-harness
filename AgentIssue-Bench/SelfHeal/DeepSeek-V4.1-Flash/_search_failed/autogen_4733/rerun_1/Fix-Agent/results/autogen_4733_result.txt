class FunctionWithRequirements(Generic[T, P]):
    func: Callable[P, T]
    python_packages: Sequence[str] = field(default_factory=list)
    global_imports: Sequence[Import] = field(default_factory=list)

    @classmethod

        return cls(python_packages=python_packages, global_imports=global_imports, func=func)

    @staticmethod
    def from_str(
        func: str, python_packages: Sequence[str] = [], global_imports: Sequence[Import] = []
    ) -> FunctionWithRequirementsStr:
        return FunctionWithRequirementsStr(func=func, python_packages=python_packages, global_imports=global_imports)

    def __call__(self, *args: P.args, **kwargs: P.kwargs) -> T:
        return self.func(*args, **kwargs)


def with_requirements(
    python_packages: Sequence[str] = [], global_imports: Sequence[Import] = []
) -> Callable[[Callable[P, T]], FunctionWithRequirements[T, P]]:
    """Decorate a function with package and import requirements

    Args:
        python_packages (List[str], optional): Packages required to function. Can include version info.. Defaults to [].
        global_imports (List[Import], optional): Required imports. Defaults to [].

    Returns:
        Callable[[Callable[P, T]], FunctionWithRequirements[T, P]]: The decorated function
    """

    def wrapper(func: Callable[P, T]) -> FunctionWithRequirements[T, P]:
        func_with_reqs = FunctionWithRequirements(
            python_packages=python_packages, global_imports=global_imports, func=func
        )

        functools.update_wrapper(func_with_reqs, func)
        return func_with_reqs

    return wrapper


def build_python_functions_file(
    funcs: Sequence[Union[FunctionWithRequirements[Any, P], Callable[..., Any], FunctionWithRequirementsStr]],
) -> str:
    """:meta private:"""
    # First collect all global imports. Note that `Alias` and `ImportFromModule`
    # are dataclasses and are collected into an ordered list (deduplicated by
    # their rendered import string) so that they do not need to be hashable.
    global_imports: List[Import] = []
    seen_imports: Set[str] = set()
    for func in funcs:
        if isinstance(func, (FunctionWithRequirements, FunctionWithRequirementsStr)):
            for imp in func.global_imports:
                imp_str = _import_to_str(imp)
                if imp_str not in seen_imports:
                    seen_imports.add(imp_str)
                    global_imports.append(imp)

    content = "\n".join(map(_import_to_str, global_imports)) + "\n\n"

    for func in funcs:
        content += _to_code(func) + "\n\n"

    return content


def to_stub(func: Union[Callable[..., Any], FunctionWithRequirementsStr]) -> str:
    """Generate a stub for a function as a string

    Args:
        func (Callable[..., Any]): The function to generate a stub for

        docstring = '"""' + docstring + '"""'
        docstring = indent(docstring, "    ")
        content += docstring + "\n"

    content += "    ..."
    return content

## Analysis

### The Bug
The root cause is in `build_python_functions_file`:
```python

for func in funcs:
    if isinstance(func, (FunctionWithRequirements, FunctionWithRequirementsStr)):
        global_imports.update(func.global_imports)
```

`Alias` and `ImportFromModule` are `@dataclass` types with default `eq=True, frozen=False`. Per dataclass semantics, defining `__eq__` implicitly sets `__hash__ = None`, making instances **unhashable**. Inserting them into a `set()` raises `TypeError: unhashable type: 'Alias'` — exactly what the test detects via `"unhashable type" in str(e)`.

### The Fix
Two complementary changes are applied to both the new module (`autogen_core/code_executor/_func_with_reqs.py`) and the deprecated shim (`autogen_core/components/code_executor/_func_with_reqs.py`):

1. **Explicit `__hash__` on `Alias` and `ImportFromModule`:**
   - `Alias.__hash__` → `hash((self.name, self.alias))`
   - `ImportFromModule.__hash__` → `hash((self.module, tuple(self.imports)))`
   - Critically, when a class body defines `__hash__` explicitly, `has_explicit_hash` is `True` in dataclass processing, so the `_hash_set_none` action is skipped and the explicit hash is preserved. `ImportFromModule`'s hash correctly depends on `Alias` being hashable too (nested case handled).

2. **List-based deduplication instead of a set:**
   - `global_imports` becomes a `List[Import]` with a `seen_imports: Set[str]` keyed on `_import_to_str(imp)`. This removes the hashing requirement entirely for the collection step and also makes output ordering deterministic (an incidental improvement over set iteration).
