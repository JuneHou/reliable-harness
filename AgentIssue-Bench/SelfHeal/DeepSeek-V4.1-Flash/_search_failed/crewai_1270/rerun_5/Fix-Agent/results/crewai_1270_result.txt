# Encodings tried (in order) when loading a configuration file.
# "utf-8-sig" transparently handles both plain UTF-8 and UTF-8 files with a BOM,
# while "gb18030"/"gbk"/"big5" cover the most common non UTF-8 Chinese encodings.
_FALLBACK_ENCODINGS = (
    "utf-8-sig",
    "utf-8",
    "gb18030",
    "gbk",
    "big5",
    "cp1252",
    "latin-1",
)


def _detect_file_encoding(config_path: Path) -> Optional[str]:
    """Best-effort detection of a file's encoding using charset_normalizer."""
    if _detect_encoding_from_path is None:
        return None

    try:
        best_match = _detect_encoding_from_path(str(config_path)).best()
    except Exception:  # noqa: BLE001 - detection must never break loading
        return None

    if best_match is None:
        return None

    return best_match.encoding or None


def _candidate_encodings(config_path: Path) -> List[str]:
    """Build the ordered list of encodings to try for a given file."""
    candidates: List[str] = []

    detected_encoding = _detect_file_encoding(config_path)
    if detected_encoding:
        candidates.append(detected_encoding)

    for encoding in _FALLBACK_ENCODINGS:
        if encoding not in candidates:
            candidates.append(encoding)

    return candidates


def _read_text(config_path: Path) -> str:
    """Read a text file, automatically detecting its encoding.

    This makes loading resilient to configuration files saved with a wide
    variety of encodings (UTF-8, UTF-8 with BOM, GBK/GB18030, ...), so users
    writing non-ASCII (e.g. Chinese) content no longer get a UnicodeDecodeError.
    """
    config_path = Path(config_path)
    raw_content = config_path.read_bytes()

    for encoding in _candidate_encodings(config_path):
        try:
            return raw_content.decode(encoding)
        except (UnicodeDecodeError, LookupError):
            continue

    # Last resort: never crash because of a decoding issue.
    return raw_content.decode("utf-8", errors="replace")

`simulate_yaml_loading_bug` hard-codes the *old* pattern and writes `b'\xff\xfe' + content`; on a normal UTF-8 Linux/Docker locale test 1 succeeds and test 2 raises `UnicodeDecodeError`, so `bug_reproduced = True`. With `found_fix_in_code = False` and `bug_reproduced = True`, the fixed branch ends in **"INCONCLUSIVE: Could not determine if the bug was fixed" → `return 1`**.

So the canonical minimal fix (`with open(config_path, "r", encoding="utf-8") as file:`) is what satisfies both the functional issue and the test's code-inspection contract; this implementation sidesteps `open()` entirely and therefore never trips the "fix found" branch.

### 4. Additional concerns (beyond the test)
- **Silent corruption instead of failing loudly:** `latin-1` never raises `UnicodeDecodeError`, and the final `decode("utf-8", errors="replace")` guarantees *some* string is produced. A mis-encoded/corrupt `agents.yaml` will now silently yield mojibake agent roles/backstories instead of a clear error — a behavior regression in error-reporting quality.
- **Nondeterministic results:** heuristic detection (`charset_normalizer`) plus a wide fallback list can decode the *same* file differently across machines/dependency versions, so configs may parse differently between environments.
- **Unnecessary dependency surface:** adds an optional third-party import and reads the file twice (detection + `read_bytes`), broadening scope well beyond the reported encoding bug.
- The YAML BOM case (bytes `\xff\xfe` UTF-16 BOM) is also not handled: `utf-8-sig`/`gb18030` etc. will not decode that content correctly, so even the "robustness" story is incomplete.

## Verdict

<reasoning>
The change is functionally plausible and the APIs are used correctly (`Path.read_bytes`, `charset_normalizer.from_path(...).best().encoding`, optional import guard, `yaml.safe_load(str)`). However, it does not satisfy this case's expectations:

1. The test's `find_yaml_loader_in_crewai` only recognizes a fix when a source line contains both `open(` and `encoding="utf-8"`. The candidate removed the `open()` call from `load_yaml` entirely, so `found_fix_in_code` stays `False`.
2. The runtime leg `simulate_yaml_loading_bug` uses the legacy pattern with an intentionally invalid `\xff\xfe` file, which raises `UnicodeDecodeError` on a normal UTF-8 locale, so `bug_reproduced = True`.
3. Consequently the "fixed" branch falls through to `else: return 1` ("INCONCLUSIVE"), i.e. the fixed version fails the test rather than reporting "FIX CONFIRMED".
4. Additionally, the implementation introduces undesirable behavior changes: `latin-1` plus `errors="replace"` can silently produce corrupted config data instead of surfacing an encoding error, and encoding auto-detection with a broad fallback chain makes config parsing nondeterministic and adds an optional dependency and a second file read.

A minimal, deterministic fix (`open(config_path, "r", encoding="utf-8")`) would resolve the reported UnicodeDecodeError and match the test's expected fix signature. As written, this change over-engineers the solution and fails the case's validation path.
</reasoning>

**REJECT**

The fix addresses the underlying UnicodeDecodeError in spirit but (a) never produces the `open(..., encoding="utf-8")` pattern the test looks for, so the fixed version is reported as INCONCLUSIVE and exits non-zero, and (b) replaces a deterministic read with encoding guessing plus silent `latin-1`/`errors="replace"` fallbacks that can mask mis-encoded configs. Prefer `with open(config_path, "r", encoding="utf-8") as file:` (optionally `utf-8-sig`) in `load_yaml`.

import inspect
from pathlib import Path
from typing import Any, Callable, Dict, List, Optional

import yaml
from dotenv import load_dotenv

try:  # Optional dependency: used to auto-detect a YAML file's encoding
    from charset_normalizer import from_path as _detect_encoding_from_path
except ImportError:  # pragma: no cover - charset_normalizer is optional
    _detect_encoding_from_path = None

load_dotenv()

# Encodings tried (in order) when loading a configuration file.
# "utf-8-sig" transparently handles both plain UTF-8 and UTF-8 files with a BOM,
# while "gb18030"/"gbk"/"big5" cover the most common non UTF-8 Chinese encodings.
_FALLBACK_ENCODINGS = (
    "utf-8-sig",
    "utf-8",
    "gb18030",
    "gbk",
    "big5",
    "cp1252",
    "latin-1",
)


def _detect_file_encoding(config_path: Path) -> Optional[str]:
    """Best-effort detection of a file's encoding using charset_normalizer."""
    if _detect_encoding_from_path is None:
        return None

    try:
        best_match = _detect_encoding_from_path(str(config_path)).best()
    except Exception:  # noqa: BLE001 - detection must never break loading
        return None

    if best_match is None:
        return None

    return best_match.encoding or None


def _candidate_encodings(config_path: Path) -> List[str]:
    """Build the ordered list of encodings to try for a given file."""
    candidates: List[str] = []

    detected_encoding = _detect_file_encoding(config_path)
    if detected_encoding:
        candidates.append(detected_encoding)

    for encoding in _FALLBACK_ENCODINGS:
        if encoding not in candidates:
            candidates.append(encoding)

    return candidates


def _read_text(config_path: Path) -> str:
    """Read a text file, automatically detecting its encoding.

    This makes loading resilient to configuration files saved with a wide
    variety of encodings (UTF-8, UTF-8 with BOM, GBK/GB18030, ...), so users
    writing non-ASCII (e.g. Chinese) content no longer get a UnicodeDecodeError.
    """
    config_path = Path(config_path)
    raw_content = config_path.read_bytes()

    for encoding in _candidate_encodings(config_path):
        try:
            return raw_content.decode(encoding)
        except (UnicodeDecodeError, LookupError):
            continue

    # Last resort: never crash because of a decoding issue.
    return raw_content.decode("utf-8", errors="replace")


def CrewBase(cls):
    class WrappedClass(cls):
        is_crew_class: bool = True  # type: ignore

        # Get the directory of the class being decorated
        base_directory = Path(inspect.getfile(cls)).parent

        original_agents_config_path = getattr(
            cls, "agents_config", "config/agents.yaml"
        )
        original_tasks_config_path = getattr(cls, "tasks_config", "config/tasks.yaml")

        def __init__(self, *args, **kwargs):
            super().__init__(*args, **kwargs)

            agents_config_path = self.base_directory / self.original_agents_config_path
            tasks_config_path = self.base_directory / self.original_tasks_config_path

            self.agents_config = self.load_yaml(agents_config_path)
            self.tasks_config = self.load_yaml(tasks_config_path)

            self.map_all_agent_variables()
            self.map_all_task_variables()

        @staticmethod
        def load_yaml(config_path: Path):
            try:
                content = _read_text(config_path)
                return yaml.safe_load(content)
            except FileNotFoundError:
                print(f"File not found: {config_path}")
                raise

        def _get_all_functions(self):
            return {
                name: getattr(self, name)
                for name in dir(self)
                if callable(getattr(self, name))
            }

        def _filter_functions(
            self, functions: Dict[str, Callable], attribute: str
        ) -> Dict[str, Callable]:
            return {
                name: func
                for name, func in functions.items()
                if hasattr(func, attribute)
            }

        def map_all_agent_variables(self) -> None:
            all_functions = self._get_all_functions()
            llms = self._filter_functions(all_functions, "is_llm")
            tool_functions = self._filter_functions(all_functions, "is_tool")
            cache_handler_functions = self._filter_functions(
                all_functions, "is_cache_handler"
            )
            callbacks = self._filter_functions(all_functions, "is_callback")
            agents = self._filter_functions(all_functions, "is_agent")

            for agent_name, agent_info in self.agents_config.items():
                self._map_agent_variables(
                    agent_name,
                    agent_info,
                    agents,
                    llms,
                    tool_functions,
                    cache_handler_functions,
                    callbacks,
                )

        def _map_agent_variables(
            self,
            agent_name: str,
            agent_info: Dict[str, Any],
            agents: Dict[str, Callable],
            llms: Dict[str, Callable],
            tool_functions: Dict[str, Callable],
            cache_handler_functions: Dict[str, Callable],
            callbacks: Dict[str, Callable],
        ) -> None:
            if llm := agent_info.get("llm"):
                try:
                    self.agents_config[agent_name]["llm"] = llms[llm]()
                except KeyError:
                    self.agents_config[agent_name]["llm"] = llm

            if tools := agent_info.get("tools"):
                self.agents_config[agent_name]["tools"] = [
                    tool_functions[tool]() for tool in tools
                ]

            if function_calling_llm := agent_info.get("function_calling_llm"):
                self.agents_config[agent_name]["function_calling_llm"] = agents[
                    function_calling_llm
                ]()

            if step_callback := agent_info.get("step_callback"):
                self.agents_config[agent_name]["step_callback"] = callbacks[
                    step_callback
                ]()

            if cache_handler := agent_info.get("cache_handler"):
                self.agents_config[agent_name]["cache_handler"] = (
                    cache_handler_functions[cache_handler]()
                )

        def map_all_task_variables(self) -> None:
            all_functions = self._get_all_functions()
            agents = self._filter_functions(all_functions, "is_agent")
            tasks = self._filter_functions(all_functions, "is_task")
            output_json_functions = self._filter_functions(
                all_functions, "is_output_json"
            )
            tool_functions = self._filter_functions(all_functions, "is_tool")
            callback_functions = self._filter_functions(all_functions, "is_callback")
            output_pydantic_functions = self._filter_functions(
                all_functions, "is_output_pydantic"
            )

            for task_name, task_info in self.tasks_config.items():
                self._map_task_variables(
                    task_name,
                    task_info,
                    agents,
                    tasks,
                    output_json_functions,
                    tool_functions,
                    callback_functions,
                    output_pydantic_functions,
                )

        def _map_task_variables(
            self,
            task_name: str,
            task_info: Dict[str, Any],
            agents: Dict[str, Callable],
            tasks: Dict[str, Callable],
            output_json_functions: Dict[str, Callable],
            tool_functions: Dict[str, Callable],
            callback_functions: Dict[str, Callable],
            output_pydantic_functions: Dict[str, Callable],
        ) -> None:
            if context_list := task_info.get("context"):
                self.tasks_config[task_name]["context"] = [
                    tasks[context_task_name]() for context_task_name in context_list
                ]

            if tools := task_info.get("tools"):
                self.tasks_config[task_name]["tools"] = [
                    tool_functions[tool]() for tool in tools
                ]

            if agent_name := task_info.get("agent"):
                self.tasks_config[task_name]["agent"] = agents[agent_name]()

            if output_json := task_info.get("output_json"):
                self.tasks_config[task_name]["output_json"] = output_json_functions[
                    output_json
                ]

            if output_pydantic := task_info.get("output_pydantic"):
                self.tasks_config[task_name]["output_pydantic"] = (
                    output_pydantic_functions[output_pydantic]
                )

            if callbacks := task_info.get("callbacks"):
                self.tasks_config[task_name]["callbacks"] = [
                    callback_functions[callback]() for callback in callbacks
                ]

    return WrappedClass
