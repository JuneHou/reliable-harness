# Encodings that are attempted, in order, when the encoding of a configuration
# file cannot be determined any other way. GB18030 is a superset of GBK/GB2312
# (the encodings most commonly used for Chinese text), "latin-1" never fails.
_FALLBACK_ENCODINGS: tuple[str, ...] = ("utf-8", "gb18030", "big5", "shift_jis", "latin-1")


def _encoding_from_bom(raw: bytes) -> Optional[str]:
    """Return the encoding announced by a byte order mark, if any."""
    if raw.startswith(codecs.BOM_UTF8):
        return "utf-8-sig"
    if raw.startswith((codecs.BOM_UTF32_LE, codecs.BOM_UTF32_BE)):
        return "utf-32"
    if raw.startswith((codecs.BOM_UTF16_LE, codecs.BOM_UTF16_BE)):
        return "utf-16"
    return None


def _encoding_from_detectors(raw: bytes) -> Optional[str]:
    """Ask an optional charset detector (charset-normalizer / chardet)."""
    try:  # pragma: no cover - depends on the installed environment
        import charset_normalizer

        best_match = charset_normalizer.from_bytes(raw).best()
        if best_match is not None and best_match.encoding:
            return str(best_match.encoding)
    except ImportError:  # pragma: no cover
        pass
    except Exception:  # pragma: no cover - a detector must never break loading
        pass

    try:  # pragma: no cover - depends on the installed environment
        import chardet

        detected = chardet.detect(raw) or {}
        if detected.get("encoding"):
            return str(detected["encoding"])
    except ImportError:  # pragma: no cover
        pass
    except Exception:  # pragma: no cover - a detector must never break loading
        pass

    return None


def detect_encoding(raw: bytes) -> str:
    """Detect the encoding of the given raw bytes.

    The detection is best effort: BOM markers are honoured first, then the
    content is validated against UTF-8 and against the encodings reported by
    optional charset detectors, and finally a list of common encodings
    (including GBK/GB2312 via GB18030) is tried. UTF-8 is used as the last
    resort so that the function always returns something usable.
    """
    if not raw:
        return "utf-8"

    bom_encoding = _encoding_from_bom(raw)
    if bom_encoding is not None:
        return bom_encoding

    # UTF-8 is by far the most common encoding, check it first.
    try:
        raw.decode("utf-8")
        return "utf-8"
    except UnicodeDecodeError:
        pass

    # Trust a detector only if its answer actually decodes the whole buffer.
    detected_encoding = _encoding_from_detectors(raw)
    if detected_encoding:
        try:
            raw.decode(detected_encoding)
            return detected_encoding
        except (UnicodeDecodeError, LookupError):
            pass

    for encoding in _FALLBACK_ENCODINGS:
        try:
            raw.decode(encoding)
        except (UnicodeDecodeError, LookupError):
            continue
        return encoding

    return "utf-8"


def _read_text_with_detected_encoding(config_path: Path) -> str:
    """Read a file as text, detecting its encoding instead of assuming UTF-8."""
    raw_content = Path(config_path).read_bytes()
    encoding = detect_encoding(raw_content)
    try:
        return raw_content.decode(encoding)
    except (UnicodeDecodeError, LookupError):
        # Never crash on undecodable bytes: keep whatever can be decoded.
        return raw_content.decode("utf-8", errors="replace")

def _decode_config_content(raw_data: bytes) -> str:
    """Decode the raw YAML content using its detected encoding.

    The encoding of the file is detected automatically (e.g. UTF-8, GBK, ...)
    so that YAML files containing non-ASCII characters, such as Chinese text,
    do not raise a ``UnicodeDecodeError`` while being loaded.
    """
    # Fast path: most configuration files are UTF-8 (optionally with a BOM).
    for encoding in PREFERRED_ENCODINGS:
        try:
            return raw_data.decode(encoding)
        except UnicodeDecodeError:
            continue

    # Detect the encoding of the file (e.g. GBK / GB2312 / Big5 / ...).
    candidates = []
    try:
        detected_encoding = (chardet.detect(raw_data) or {}).get("encoding")

    for encoding in candidates:
        try:
            return raw_data.decode(encoding)
        except (UnicodeDecodeError, LookupError):
            continue

def _encoding_from_bom(raw: bytes) -> Optional[str]:
    """Return the encoding announced by a byte order mark, if any."""
    if raw.startswith(codecs.BOM_UTF8):
        return "utf-8-sig"
    if raw.startswith((codecs.BOM_UTF32_LE, codecs.BOM_UTF32_BE)):
        return "utf-32"
    if raw.startswith((codecs.BOM_UTF16_LE, codecs.BOM_UTF16_BE)):
        return "utf-16"
    return None


def _encoding_from_detectors(raw: bytes) -> Optional[str]:
    """Ask an optional charset detector (charset-normalizer / chardet)."""
    try:  # pragma: no cover - depends on the installed environment
        import charset_normalizer

        best_match = charset_normalizer.from_bytes(raw).best()
        if best_match is not None and best_match.encoding:
            return str(best_match.encoding)
    except ImportError:  # pragma: no cover
        pass
    except Exception:  # pragma: no cover - a detector must never break loading
        pass

    try:  # pragma: no cover - depends on the installed environment
        import chardet

        detected = chardet.detect(raw) or {}
        if detected.get("encoding"):
            return str(detected["encoding"])
    except ImportError:  # pragma: no cover
        pass
    except Exception:  # pragma: no cover - a detector must never break loading
        pass

    return None


def detect_encoding(raw: bytes) -> str:
    """Detect the encoding of the given raw bytes.

    The detection is best effort: BOM markers are honoured first, then the
    content is validated against UTF-8 and against the encodings reported by
    optional charset detectors, and finally a list of common encodings
    (including GBK/GB2312 via GB18030) is tried. UTF-8 is used as the last
    resort so that the function always returns something usable.
    """
    if not raw:
        return "utf-8"

    bom_encoding = _encoding_from_bom(raw)
    if bom_encoding is not None:
        return bom_encoding

    # UTF-8 is by far the most common encoding, check it first.
    try:
        raw.decode("utf-8")
        return "utf-8"
    except UnicodeDecodeError:
        pass

    # Trust a detector only if its answer actually decodes the whole buffer.
    detected_encoding = _encoding_from_detectors(raw)
    if detected_encoding:
        try:
            raw.decode(detected_encoding)
            return detected_encoding
        except (UnicodeDecodeError, LookupError):
            pass

    for encoding in _FALLBACK_ENCODINGS:
        try:
            raw.decode(encoding)
        except (UnicodeDecodeError, LookupError):
            continue
        return encoding

    return "utf-8"


def _read_text_with_detected_encoding(config_path: Path) -> str:
    """Read a file as text, detecting its encoding instead of assuming UTF-8."""
    raw_content = Path(config_path).read_bytes()
    encoding = detect_encoding(raw_content)
    try:
        return raw_content.decode(encoding)
    except (UnicodeDecodeError, LookupError):
        # Never crash on undecodable bytes: keep whatever can be decoded.
        return raw_content.decode("utf-8", errors="replace")
