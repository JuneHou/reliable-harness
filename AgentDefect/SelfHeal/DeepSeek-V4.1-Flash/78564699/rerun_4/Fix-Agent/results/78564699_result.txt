import json
import os
import time

from llama_index.core.llms import ChatMessage
from llama_index.llms.llama_cpp import LlamaCPP


MODEL_URL = (
    "https://huggingface.co/TheBloke/Llama-2-13B-chat-GGUF/resolve/main/"
    "llama-2-13b-chat.Q4_0.gguf"
)


def usable_cpu_threads() -> int:
    """Number of CPU threads llama.cpp should really use.

    llama-cpp-python defaults `n_threads` to *all* logical CPUs.  On machines
    with hyper-threading (or in containers/WSL where the reported CPU count is
    much larger than the CPU quota actually granted) the OpenMP pool gets
    over-subscribed: the threads fight over the same caches / cores and the
    generation step makes no visible progress, which looks exactly like a
    permanent hang after model initialisation.  Using only the physical cores
    (and never more than the CPUs this process is allowed to run on) fixes it.
    """
    try:
        # Respect cgroup / taskset / CPU affinity limits where available.
        n_cpus = len(os.sched_getaffinity(0))  # type: ignore[attr-defined]
    except AttributeError:  # Windows, macOS
        n_cpus = os.cpu_count() or 4

    if n_cpus > 2:
        # 2 logical cores per physical core => use the physical count.
        return max(1, n_cpus // 2)
    return max(1, n_cpus)


def llama2_messages_to_prompt(messages) -> str:
    """Format messages with the Llama-2-chat template.

    llama-2-13b-chat is a *chat* model: it was fine-tuned on
    "<s>[INST] <<SYS>> ... <</SYS>> ... [/INST] answer </s>".  Feeding a bare
    question (what `complete()` does) makes it continue the text instead of
    answering, so it never emits a stop token and generates garbage until the
    token budget is exhausted - which on CPU takes minutes and looks like a
    hang.  This formatter makes the model answer and stop properly.
    """
    prompt_parts = []
    system_prompt = ""
    role_value = lambda role: getattr(role, "value", role)

    for message in messages:
        role = role_value(message.role)
        if role == "system":
            system_prompt = message.content.strip()
            continue
        if role == "user":
            if system_prompt:
                prompt_parts.append(
                    f"<s>[INST] <<SYS>>\n{system_prompt}\n<</SYS>>\n\n"
                    f"{message.content.strip()} [/INST]"
                )
                # The system block only belongs to the first turn.
                system_prompt = ""
            else:
                prompt_parts.append(f"<s>[INST] {message.content.strip()} [/INST]")
        else:  # assistant / model turn
            prompt_parts.append(f" {message.content.strip()} </s>")

    return "".join(prompt_parts)


def resolve_model_source():
    """Return (model_url, model_path) for LlamaCPP - only one may be set."""
    paths_file = "./paths.json"
    model_url, model_path = MODEL_URL, None

    if os.path.exists(paths_file):
        with open(paths_file, "r", encoding="utf-8") as f:
            paths = json.load(f)

        local_model = paths.get("llama-2-13b-chat")
        if local_model:
            local_model = os.path.abspath(os.path.expanduser(local_model))
            if not os.path.isfile(local_model):
                raise FileNotFoundError(
                    f"paths.json points to '{local_model}', which does not exist. "
                    "Fix the path or remove the entry so the GGUF can be downloaded."
                )
            model_url, model_path = None, local_model

    if not model_url and not model_path:
        raise ValueError("Neither a model_url nor a model_path is available.")

    return model_url, model_path


def run_agent(query: str) -> str:
    model_url, model_path = resolve_model_source()
    n_threads = usable_cpu_threads()

    llm = LlamaCPP(
        model_url=model_url,
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        generate_kwargs={},
        model_kwargs={
            "n_gpu_layers": 0,      # pure CPU inference
            "n_threads": n_threads,  # <-- do NOT over-subscribe the CPU
            "n_batch": 512,
            "use_mmap": True,
            "use_mlock": False,
        },
        messages_to_prompt=llama2_messages_to_prompt,
        verbose=True,
    )

    print(f"[info] generating with {n_threads} CPU thread(s) ...", flush=True)
    start = time.time()
    response = llm.chat([ChatMessage(role="user", content=query)])
    elapsed = time.time() - start
    print(f"[info] finished in {elapsed:.1f}s", flush=True)

    return str(response)


if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
