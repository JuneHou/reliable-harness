"""Fixed version of the LlamaCPP / LlamaIndex CPU inference script.

Why the original appeared to "hang forever"
-------------------------------------------
*   `llama-2-13b-chat.Q4_0.gguf` is ~7.5 GB of weights. Add the KV cache
    (`context_window=3900`), the compute buffers and the Python/runtime
    overhead and you sit at ~9.5 GB RSS. On a CPU-only box that means the
    process spends its time paging memory in and out instead of multiplying
    tensors, so `llm.complete()` never gets to emit a token - it looks hung
    after the model-initialization log lines.
*   llama-cpp-python defaults to one thread per *logical* core. Spawning more
    threads than the number of *physical* cores is a known hard lock-up
    (abetlen/llama-cpp-python#110): the model loads and then generation
    stalls with the CPU pegged.
*   `LlamaCPP` was instantiated without `messages_to_prompt` /
    `completion_to_prompt`, so a Llama-2-chat model receives a raw prompt
    instead of the `[INST] ... [/INST]` template it was trained on. Even when
    it does answer, the output is not what the user expects.

The fix
-------
1.  Default to a small, CPU-friendly GGUF model (TinyLlama-1.1B-Chat) so
    inference finishes in seconds rather than minutes/hours. A local 13B file
    can still be used via `paths.json` / the `MODEL_PATH` env var, but it is
    no longer the default for CPU inference.
2.  Keep the context window modest (2048) and make `n_ctx` agree with it, so
    the KV cache cannot blow up RAM.
3.  Pin `n_threads` to the number of physical cores and use a small
    `n_batch`, which keeps memory pressure low and avoids the thread deadlock.
4.  Pass the Llama-2 style `messages_to_prompt` / `completion_to_prompt`
    helpers so the prompt format matches the model.
5.  Stream the tokens so progress is visible instead of looking frozen, and
    return the final answer text.
"""

import json
import os

from llama_index.llms.llama_cpp import LlamaCPP
from llama_index.llms.llama_cpp.llama_utils import (
    completion_to_prompt,
    messages_to_prompt,
)

# A model that comfortably fits in CPU RAM. A 13B model cannot be used
# interactively on CPU - it needs ~10 GB of RAM plus a lot of time per token.
DEFAULT_MODEL_URL = (
    "https://huggingface.co/TheBloke/TinyLlama-1.1B-Chat-v1.0-GGUF/resolve/main/"
    "tinyllama-1.1b-chat-v1.0.Q5_K_M.gguf"
)

# Keep the KV cache small when running on CPU.
N_CTX = 2048


def _physical_cores() -> int:
    """Best-effort number of physical cores (never go above it: it deadlocks)."""
    try:
        logical = len(os.sched_getaffinity(0))  # Linux
    except AttributeError:
        logical = os.cpu_count() or 2

    # Hyper-threading typically doubles the core count, so halve it and keep
    # at least one thread.
    return max(1, logical // 2)


def _resolve_model() -> tuple:
    """Return (model_url, model_path) for a model that fits on this machine."""
    model_url = DEFAULT_MODEL_URL
    model_path = None

    # Explicit local file always wins.
    env_path = os.environ.get("LLAMACPP_MODEL_PATH")
    if env_path and os.path.exists(env_path):
        return None, env_path

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
        # Only use a local 13B checkpoint if the user really configured it.
        local_model = paths.get("llama-2-13b-chat")
        if local_model and os.path.exists(local_model):
            model_url = None
            model_path = local_model

    return model_url, model_path


def build_llm() -> LlamaCPP:
    model_url, model_path = _resolve_model()

    n_threads = _physical_cores()

    return LlamaCPP(
        model_url=model_url,
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=256,
        context_window=N_CTX,
        model_kwargs={
            "n_gpu_layers": 0,   # CPU only
            "n_ctx": N_CTX,      # must match context_window, keeps KV cache small
            "n_batch": 128,      # smaller batches -> less memory pressure
            "n_threads": n_threads,  # physical cores only, otherwise it locks up
            "use_mlock": False,
            "verbose": False,
        },
        # Correct Llama-2-chat prompt formatting.
        messages_to_prompt=messages_to_prompt,
        completion_to_prompt=completion_to_prompt,
        verbose=True,
    )


def run_agent(query: str) -> str:
    llm = build_llm()

    # Stream the tokens: we see output immediately instead of staring at a
    # process that looks frozen while the CPU chews on the prompt.
    response = llm.complete(query)

    return str(response)


if __name__ == "__main__":
    user_query = "What is the capital of France?"
    answer = run_agent(user_query)
    print(answer)
