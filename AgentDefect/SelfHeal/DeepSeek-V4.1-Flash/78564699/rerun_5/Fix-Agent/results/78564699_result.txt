import json
import os

from llama_index.core.llms import ChatMessage
from llama_index.llms.llama_cpp import LlamaCPP

# ---------------------------------------------------------------------------
# Why the original code appears to "hang" forever
# ---------------------------------------------------------------------------
# * llama-cpp-python's default `n_threads` is roughly cpu_count() // 2 (and it
#   counts hyper-threads), so a 13B model only ever uses ~half of the CPU
#   (that's exactly the "CPU usage is about 50%" symptom). Token generation
#   then takes seconds per token, so a 256-token answer looks like a freeze.
# * A 13B Q4_0 GGUF needs ~9.5 GB of RAM just for the weights, plus the KV
#   cache for `context_window=3900`. On a machine with less headroom the
#   process thrashes swap after the (cheap, mmap'ed) model load, which looks
#   like "initialization finishes, then nothing happens forever".
# * Nothing is actually deadlocked: it is just running far slower than the
#   user's expectation. Fix = use a model/quantization that fits comfortably
#   (7B Q4_K_M instead of 13B Q4_0), give it a realistic context window and
#   explicitly hand all physical cores to llama.cpp.
# ---------------------------------------------------------------------------


def _physical_cores() -> int:
    """Best-effort count of *physical* cores (hyper-threads hurt llama.cpp)."""
    try:
        import psutil  # optional dependency

        cores = psutil.cpu_count(logical=False)
        if cores:
            return max(1, cores)
    except Exception:
        pass
    logical = os.cpu_count() or 2
    return max(1, logical // 2)


N_THREADS = _physical_cores()

# 7B + Q4_K_M (~4.1 GB) fits in RAM on a CPU-only box; 13B Q4_0 (~7.4 GB +
# KV cache) does not and makes the process swap forever.
DEFAULT_MODEL_FILENAME = "llama-2-7b-chat.Q4_K_M.gguf"
DEFAULT_MODEL_URL = (
    "https://huggingface.co/TheBloke/Llama-2-7B-Chat-GGUF/resolve/main/"
    + DEFAULT_MODEL_FILENAME
)


def _resolve_model() -> tuple:
    """Return (model_url, model_path) for a model that fits on CPU."""
    model_url = DEFAULT_MODEL_URL
    model_path = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)

        # Accept either the new (7B) key or the legacy 13B key.
        for key in ("llama-2-7b-chat", "llama-2-13b-chat"):
            candidate = paths.get(key)
            if candidate and os.path.exists(candidate):
                model_url = None
                model_path = candidate
                break

    return model_url, model_path


def build_llm() -> LlamaCPP:
    model_url, model_path = _resolve_model()

    return LlamaCPP(
        model_url=model_url,
        model_path=model_path,
        # A local file is already there; don't re-download it.
        model_filename=None if model_path else DEFAULT_MODEL_FILENAME,
        temperature=0.1,
        max_new_tokens=128,
        # 3900 was far more context than this one-shot question needs; every
        # extra slot costs RAM (KV cache) that a CPU box simply does not have.
        context_window=2048,
        model_kwargs={
            "n_gpu_layers": 0,          # CPU-only machine
            "n_threads": N_THREADS,     # <- the actual fix for the "hang"
            "n_batch": 64,              # small batches keep RAM flat
            "use_mlock": False,         # never lock a model bigger than RAM
            "verbose": False,
        },
        # llama-cpp-python's own chat template handling for Llama-2
        chat_format="llama-2",
        verbose=True,
    )


def run_agent(query: str) -> str:
    llm = build_llm()

    # The model is a *chat* model: feeding the bare question to `complete()`
    # gives a continuation instead of an answer. Use the chat interface (or an
    # explicit [INST] wrapper) so the response matches expectations.
    try:
        response = llm.chat([ChatMessage(role="user", content=query)])
        return response.message.content or ""
    except AttributeError:
        # Older llama-index versions: fall back to complete() with Llama-2
        # chat formatting so the prompt is still correct.
        response = llm.complete(f"[INST] {query} [/INST]")
        return str(response)


if __name__ == "__main__":
    question = "What is the capital of France?"
    print(f"Running inference on {N_THREADS} physical CPU cores...", flush=True)
    answer = run_agent(question)
    print(answer)
