import json
import os

from llama_index.llms.llama_cpp import LlamaCPP
from llama_index.llms.llama_utils import (
    completion_to_prompt,
    messages_to_prompt,
)

MODEL_URL = (
    "https://huggingface.co/TheBloke/Llama-2-13B-chat-GGUF/resolve/main/"
    "llama-2-13b-chat.Q4_0.gguf"
)
PATHS_FILE = "./paths.json"

# ---------------------------------------------------------------------------
# llama.cpp on CPU tuning
#
#  * n_threads MUST be set explicitly.  llama.cpp busy-waits (spin-lock
#    barriers) inside its generation loop, so when more threads are started
#    than the CPU can actually run (hyper-threading, cpu.max/cpuset limited
#    containers, restricted affinity) the threads convoy against each other:
#    the process pins the CPU at ~100% and never emits a single token, which
#    looks exactly like "hangs forever after model initialization".
#    llama-cpp-python's default (cpu_count() // 2) is already too high there,
#    hence we always pass a value derived from the CPUs this process can use.
#  * n_ctx / context_window=3900 allocated a very large KV cache on top of the
#    ~7.4 GB Q4_0 13B weights; on a CPU-only machine that normally means swap
#    thrashing (again: "hangs").  2048 is plenty for a single Q/A turn.
#  * 13B Q4_0 needs ~10 GB free RAM.  If the box has less, use a 7B quant
#    (e.g. llama-2-7b-chat.Q4_K_M.gguf) - it is the same code path, only the
#    GGUF file changes.
#  * messages_to_prompt / completion_to_prompt apply the Llama-2 chat template,
#    so the model emits its EOS token and generation actually terminates.
# ---------------------------------------------------------------------------
N_CTX = 2048
N_BATCH = 256
MAX_NEW_TOKENS = 256


def _usable_cpu_count() -> int:
    """Number of CPUs this process may really run on (honours affinity/cgroups)."""
    try:
        # Linux: respects taskset / cgroup cpuset restrictions.
        return len(os.sched_getaffinity(0))
    except AttributeError:
        # macOS / Windows: fall back to the logical CPU count.
        return os.cpu_count() or 1


def _n_threads() -> int:
    """Thread count that cannot oversubscribe the CPU (never < 1)."""
    return max(1, _usable_cpu_count() // 2)


def _resolve_model():
    """
    Return (model_url, model_path) with exactly one of them non-None.

    Guessing the local path from paths.json is fine, but a stale/incorrect
    entry used to be handed to llama.cpp anyway and the model then either
    re-downloaded or failed to load - both of which present themselves as a
    silent freeze.  Validate before constructing the llm.
    """
    if os.path.exists(PATHS_FILE):
        with open(PATHS_FILE, "r") as f:
            paths = json.load(f)

        local_path = paths.get("llama-2-13b-chat")
        if local_path:
            if not os.path.isfile(local_path):
                raise FileNotFoundError(
                    f"'llama-2-13b-chat' in {PATHS_FILE} points to "
                    f"{local_path!r}, which does not exist."
                )
            return None, local_path

    return MODEL_URL, None


def build_llm() -> LlamaCPP:
    model_url, model_path = _resolve_model()

    return LlamaCPP(
        model_url=model_url,
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=MAX_NEW_TOKENS,
        context_window=N_CTX,
        messages_to_prompt=messages_to_prompt,
        completion_to_prompt=completion_to_prompt,
        model_kwargs={
            # pure CPU inference
            "n_gpu_layers": 0,
            # keep the KV cache in sync with context_window
            "n_ctx": N_CTX,
            # smaller prompt-eval batches -> lower peak memory on CPU
            "n_batch": N_BATCH,
            # the actual fix for the apparent infinite hang
            "n_threads": _n_threads(),
            "use_mmap": True,
            "use_mlock": False,
        },
        # verbose=True lets you see llama.cpp's per-token output on stderr,
        # i.e. you can tell "slow but progressing" from "really stuck".
        verbose=True,
    )


def run_agent(query: str) -> str:
    llm = build_llm()
    response = llm.complete(query)
    return str(response)


if __name__ == "__main__":
    print(f"usable CPUs: {_usable_cpu_count()} -> n_threads={_n_threads()}")
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
