from tqdm import tqdm
from langchain.docstore.document import Document
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from dotenv import load_dotenv
load_dotenv()


def run_agent():

    embeddings = OpenAIEmbeddings()
    results = {}

    for db_collection_name in tqdm(["class1-sub2-chap3", "class2-sub3-chap4"]):
        documents = []
        doc_ids = []

        for doc_index in range(3):
            cl, sub, chap = db_collection_name.split("-")
            content = f"This is {db_collection_name}-doc{doc_index}"
            doc = Document(
                page_content=content,
                metadata={"chunk_num": doc_index, "chapter": chap, "class": cl, "subject": sub},
            )
            documents.append(doc)
            doc_ids.append(str(doc_index))

        db = Chroma.from_documents(
            collection_name=db_collection_name,
            documents=documents,
            ids=doc_ids,
            embedding=embeddings,
            persist_directory="./data",
        )

        db.persist()

    initial_content = "This is an initial document content"
    document_id = "doc1"

    original_doc = Document(page_content=initial_content, metadata={"page": "0"})

    new_db = Chroma.from_documents(
        collection_name="test_collection",
        documents=[original_doc],
        embedding=embeddings,
        ids=[document_id],
        persist_directory="./data",
    )
    new_db.persist()

    # `get()` defaults to include=["metadatas", "documents"], so the embeddings
    # field comes back as None.  Ask for the embeddings explicitly.
    include = ["embeddings", "documents", "metadatas"]

    for db_collection_name in ["class1-sub2-chap3", "class2-sub3-chap4"]:
        stored_db = Chroma(
            collection_name=db_collection_name,
            embedding_function=embeddings,
            persist_directory="./data",
        )
        results[db_collection_name] = stored_db.get(include=include)

    results["test_collection"] = new_db.get(include=include)

    return results


if __name__ == "__main__":
    result = run_agent()
    for collection_name, data in result.items():
        print(f"=== {collection_name} ===")
        for doc_id, content, metadata, embedding in zip(
            data["ids"],
            data["documents"],
            data["metadatas"],
            data["embeddings"],
        ):
            print(f"id={doc_id!r}")
            print(f"  content  : {content}")
            print(f"  metadata : {metadata}")
            print(f"  embedding: {embedding}")
