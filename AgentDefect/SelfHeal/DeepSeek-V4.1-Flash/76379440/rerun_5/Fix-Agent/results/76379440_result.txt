from tqdm import tqdm
from langchain.docstore.document import Document
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from dotenv import load_dotenv

load_dotenv()


def run_agent():

    embeddings = OpenAIEmbeddings()

    last_db = None

    for db_collection_name in tqdm(["class1-sub2-chap3", "class2-sub3-chap4"]):
        documents = []
        doc_ids = []

        for doc_index in range(3):
            cl, sub, chap = db_collection_name.split("-")
            content = f"This is {db_collection_name}-doc{doc_index}"
            doc = Document(
                page_content=content,
                metadata={
                    "chunk_num": doc_index,
                    "chapter": chap,
                    "class": cl,
                    "subject": sub,
                },
            )
            documents.append(doc)
            doc_ids.append(str(doc_index))

        db = Chroma.from_documents(
            collection_name=db_collection_name,
            documents=documents,
            ids=doc_ids,
            embedding=embeddings,
            persist_directory="./data",
        )

        db.persist()

        # keep an explicit reference to the collection we want to inspect later,
        # instead of relying on the loop variable leaking out of the for-loop
        last_db = db

    initial_content = "This is an initial document content"
    document_id = "doc1"

    original_doc = Document(page_content=initial_content, metadata={"page": "0"})

    new_db = Chroma.from_documents(
        collection_name="test_collection",
        documents=[original_doc],
        embedding=embeddings,
        ids=[document_id],
        persist_directory="./data",
    )
    new_db.persist()

    # The stored embeddings are NOT returned by default: Chroma only includes
    # "metadatas" and "documents" unless you explicitly ask for embeddings.
    # So request them via include=[...] and read them from the returned dict.
    stored = last_db.get(include=["embeddings", "documents", "metadatas"])

    return stored


if __name__ == "__main__":
    result = run_agent()

    print("ids:        ", result["ids"])
    print("documents:  ", result["documents"])
    print("metadatas:  ", result["metadatas"])
    print("embeddings: ")
    for doc_id, embedding in zip(result["ids"], result["embeddings"]):
        print(f"  {doc_id}: dim={len(embedding)} -> {embedding[:5]}...")
