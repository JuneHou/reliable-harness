from tqdm import tqdm
from dotenv import load_dotenv

try:  # langchain >= 0.1 layout, with fallback for older installs
    from langchain_core.documents import Document
except ImportError:  # pragma: no cover
    from langchain.docstore.document import Document

from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings

load_dotenv()

PERSIST_DIRECTORY = "./data"
COLLECTION_NAMES = ["class1-sub2-chap3", "class2-sub3-chap4"]
TEST_COLLECTION = "test_collection"


def _persist(db):
    """chromadb >= 0.4.0 persists automatically; older versions need .persist()."""
    try:
        db.persist()
    except Exception:
        pass


def get_stored_embeddings(store, include=("documents", "metadatas", "embeddings")):
    """Return ids/documents/metadatas/embeddings for a Chroma collection.

    IMPORTANT: Chroma's ``get()`` only returns ``ids``, ``documents`` and
    ``metadatas`` by default -- ``embeddings`` is ``None`` unless you
    explicitly ask for it via ``include=[...]``.  That is why the original
    ``db._collection.get()`` appeared to return no embeddings.
    """
    include = list(include)

    try:  # LangChain wrapper (returns python lists)
        raw = store.get(include=include)
    except TypeError:  # very old LangChain: go straight to the raw collection
        raw = store._collection.get(include=include)

    if raw.get("embeddings") is None:  # be defensive, retry on the raw collection
        raw = store._collection.get(include=include)

    embeddings = raw.get("embeddings")
    if embeddings is not None:
        embeddings = [
            emb.tolist() if hasattr(emb, "tolist") else list(emb) for emb in embeddings
        ]

    return {
        "ids": raw.get("ids"),
        "documents": raw.get("documents"),
        "metadatas": raw.get("metadatas"),
        "embeddings": embeddings,
    }


def load_collection(collection_name, embeddings):
    """Re-open an already persisted collection (read-only use case)."""
    return Chroma(
        collection_name=collection_name,
        embedding_function=embeddings,
        persist_directory=PERSIST_DIRECTORY,
    )


def run_agent():
    embeddings = OpenAIEmbeddings()

    # ---------------------------------------------------------------- ingest
    for db_collection_name in tqdm(COLLECTION_NAMES):
        documents = []
        doc_ids = []

        for doc_index in range(3):
            cl, sub, chap = db_collection_name.split("-")
            content = f"This is {db_collection_name}-doc{doc_index}"
            doc = Document(
                page_content=content,
                metadata={
                    "chunk_num": doc_index,
                    "chapter": chap,
                    "class": cl,
                    "subject": sub,
                },
            )
            documents.append(doc)
            # ids must be unique *within* a collection -> make them descriptive
            doc_ids.append(f"{db_collection_name}-doc{doc_index}")

        db = Chroma.from_documents(
            collection_name=db_collection_name,
            documents=documents,
            ids=doc_ids,
            embedding=embeddings,
            persist_directory=PERSIST_DIRECTORY,
        )
        _persist(db)

    initial_content = "This is an initial document content"
    document_id = "doc1"

    original_doc = Document(page_content=initial_content, metadata={"page": "0"})

    new_db = Chroma.from_documents(
        collection_name=TEST_COLLECTION,
        documents=[original_doc],
        embedding=embeddings,
        ids=[document_id],
        persist_directory=PERSIST_DIRECTORY,
    )
    _persist(new_db)

    # ------------------------------------------------------- read back data
    # Re-open every collection from the persist directory and request the
    # embeddings explicitly so that they are actually returned.
    results = {}
    for collection_name in COLLECTION_NAMES + [TEST_COLLECTION]:
        store = load_collection(collection_name, embeddings)
        results[collection_name] = get_stored_embeddings(
            store, include=["documents", "metadatas", "embeddings"]
        )

    return results


if __name__ == "__main__":
    result = run_agent()

    for collection_name, payload in result.items():
        embeddings = payload["embeddings"] or []
        dims = len(embeddings[0]) if embeddings else 0

        print(f"\n=== collection: {collection_name} ===")
        print(f"stored documents : {len(payload['ids'] or [])}")
        print(f"embedding shape  : ({len(embeddings)}, {dims})")
        for doc_id, doc, meta, emb in zip(
            payload["ids"] or [],
            payload["documents"] or [],
            payload["metadatas"] or [],
            embeddings,

            print(f"    metadata  = {meta}")
            print(f"    embedding = [{emb[0]:.6f}, {emb[1]:.6f}, ..., {emb[-1]:.6f}]")

def fetch_stored_records(vector_store, include=None):
    """Return ids/documents/metadatas/embeddings for every stored record."""
    include = list(include or INCLUDE)
    try:
        # LangChain wrapper: get(ids=None, where=None, ..., include=[...])
        return vector_store.get(include=include)
    except (TypeError, ValueError):
        # Fall back to the raw chromadb collection API (older/newer versions)
        collection = getattr(vector_store, "_collection", None)
        if collection is None:
            raise
        return collection.get(include=include)
