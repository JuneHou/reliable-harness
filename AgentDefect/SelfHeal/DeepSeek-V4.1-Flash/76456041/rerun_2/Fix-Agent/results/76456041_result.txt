import os
import warnings

from dotenv import load_dotenv, find_dotenv

from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.prompts import PromptTemplate

warnings.filterwarnings('ignore')

load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


# ---------------------------------------------------------------------------
# FIX 1 (why the answer did not match expectations)
# ---------------------------------------------------------------------------
# `RetrievalQA(chain_type="map_reduce")` does NOT reuse the "stuff" prompt.
# It uses the MapReduce summarisation prompts:
#   * the MAP prompt is:
#       "Use the following pieces of context to answer the question at the end.
#        If you don't know the answer, just say that you don't know ..."
#     -> each single CSV row/chunk does not by itself contain the full list,
#        so almost every map call answers "I don't know".
#   * the REDUCE/COMBINE prompt is:
#       "The following is a set of summaries: ... Take these and distill it
#        into a final, consolidated summary of the main themes."
#     -> it asks for a *thematic summary*, not for the names, so the final text
#        is a vague paragraph instead of the list of German drivers.
# The fix is to supply explicit extraction prompts. Their input variables must
# match exactly what LangChain expects:
#   question_prompt -> ["context", "question"]
#   combine_prompt  -> ["summaries", "question"]
QUESTION_PROMPT = PromptTemplate(
    template="""You are given one chunk of a CSV file. Each line of the chunk is one
complete record (row) of the file, with its fields.

{context}

Question: {question}

Read EVERY line of the chunk above, then:
- keep only the records that satisfy the question,
- output the requested field value(s) verbatim, one value per line,
- no numbering, no explanations, no additional text,
- if no record in this chunk satisfies the question, output exactly: NONE

Answer:""",
    input_variables=["context", "question"],
)

COMBINE_PROMPT = PromptTemplate(
    template="""Several chunks of the same CSV file were searched independently for the
question: "{question}"

Below are the per-chunk results. Each line is a matching value, or the word
NONE when that chunk contained no matching record.

{summaries}

Build the final, consolidated answer:
- keep every distinct matching value coming from any chunk (drop none of them),
- remove duplicates and ignore NONE entries,
- never invent a value that is not present above,
- answer with a plain comma separated list on a single line,
- if nothing matched anywhere, answer exactly:
  No matching record was found in the file.

Final answer:""",
    input_variables=["summaries", "question"],
)

CHAIN_TYPE_KWARGS = {
    "question_prompt": QUESTION_PROMPT,   # MAP step prompt
    "combine_prompt": COMBINE_PROMPT,     # REDUCE step prompt
    "return_intermediate_steps": False,
    "token_max": 4000,                    # fewer collapse/merge round-trips
}

# ---------------------------------------------------------------------------
# FIX 2 (the unexpectedly high API cost)
# ---------------------------------------------------------------------------
# `search_kwargs={"k": 1000}` returned ~1000 documents per query and the
# map_reduce strategy issues ONE LLM call per document, i.e. ~1000 map calls
# plus the reduce calls for every single query. Keep k small and bound it by
# the number of rows actually loaded.
DEFAULT_K = 25


def build_qa_chain(file_path, k=DEFAULT_K):
    """Build the CSV index once and return a reusable map_reduce QA chain."""
    loader = CSVLoader(file_path=file_path, encoding='utf-8')
    docs = loader.load()

    embeddings = OpenAIEmbeddings()
    db = Chroma.from_documents(docs, embeddings)

    n_docs = len(docs)
    retriever = db.as_retriever(
        search_type="mmr",  # diverse rows instead of near-duplicate ones
        search_kwargs={
            "k": min(k, n_docs),
            "fetch_k": min(n_docs, max(k * 4, k)),
        },
    )

    llm = OpenAI(
        temperature=0,
        max_tokens=1024,      # long enough to hold a real list of names
        batch_size=20,
    )

    qa_chain = RetrievalQA.from_chain_type(
        llm=llm,
        chain_type="map_reduce",
        retriever=retriever,
        chain_type_kwargs=CHAIN_TYPE_KWARGS,
        verbose=True,
    )
    return qa_chain, n_docs


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")

        # build the vectorstore / chain ONCE per file, not once per query
        qa_chain, n_docs = build_qa_chain(file)

        file_cost = 0.0
        file_tokens = 0
        for query in queries:
            print(query)

            with get_openai_callback() as cb:
                result = qa_chain.run(query)

            file_cost += cb.total_cost
            file_tokens += cb.total_tokens

            results.append(result)
            print(result)
            print(cb)

        print(f"[{file}] rows={n_docs} "
              f"tokens={file_tokens} cost=${file_cost:.6f}")

    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German"
    ]
    res = run_agent(queries)
