import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv

dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# The PythonREPL (and its namespace) must be created ONCE at module scope
# (shared scope) instead of inside run_agent(), so that:
#   * every tool call sees the same interpreter state
#   * the main script can read the objects the agent created
# ---------------------------------------------------------------------------
REPL_NAMESPACE: dict = {}
try:
    python_repl = PythonREPL(_globals=REPL_NAMESPACE)
except TypeError:  # older/newer signature fallback
    python_repl = PythonREPL()
    REPL_NAMESPACE = python_repl.globals

# Plain, always-valid registry holding the *real* objects (never their repr).
# Tool outputs are stringified before reaching the LLM, so the actual
# DataFrame has to be kept out-of-band and handed back to the caller.
AGENT_VARIABLES: dict = {}


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file, transform it, and store the resulting DataFrame
    in the shared `python_repl` variable named `df`.

    Returns a short textual summary (tool results are always converted to
    strings before they are shown to the LLM, so returning the DataFrame
    itself would lose the object).
    """
    df = pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    df = df.select_dtypes(include=["object"])

    # 1) keep a real reference the main script can access
    AGENT_VARIABLES["df"] = df
    # 2) expose it to the agent's python_repl sandbox
    REPL_NAMESPACE["df"] = df
    try:
        python_repl.locals["df"] = df
    except Exception:
        pass

    return (
        f"DataFrame loaded from '{filepath}' with shape {df.shape} and "
        f"columns {list(df.columns)}. The object is bound to the "
        f"python_repl variable `df`."
    )


def build_tools():
    shell_tool = ShellTool()
    repl_tool = Tool(
        name="python_repl",
        description=(
            "Run Python commands in a REPL environment. The variable `df` holds "
            "the DataFrame produced by load_excel_response."
        ),
        func=python_repl.run,
    )
    return [load_excel_response, repl_tool, shell_tool]


def run_agent(user_prompt, file_name="data.json"):
    """Run the ReAct agent and return BOTH the agent response and the
    DataFrame that the agent produced (if any)."""
    llm = ChatOpenAI(model="gpt-4o-mini")
    tools_list = build_tools()

    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(
        agent=shell_agent,
        tools=tools_list,
        verbose=True,
        handle_parsing_errors=True,
    )

    # Start each run from a clean registry
    AGENT_VARIABLES.pop("df", None)
    REPL_NAMESPACE.pop("df", None)

    response = agent_executor.invoke({"input": user_prompt})

    # Pick the DataFrame up from the shared state, not from the LLM answer
    df = AGENT_VARIABLES.get("df", REPL_NAMESPACE.get("df"))
    return response, df


if __name__ == "__main__":
    path = "excel_file.xlsx"
    # Explicit prompt: tell the agent exactly which tool to use, which variable
    # name to bind the result to, and that it must not fabricate data.
    user_prompt = (
        f"Load the Excel file located at {path} by calling the "
        f"load_excel_response tool with filepath='{path}'. "
        "That tool stores the resulting DataFrame in the python_repl variable "
        "named `df`. Afterwards use the python_repl tool to run "
        "`print(df.shape)` and `print(df.head())` and report exactly that "
        "output. Never invent or guess any data; only describe what the tools "
        "actually returned."
    )

    result, df = run_agent(user_prompt)

    # 1) the agent's textual answer
    print(result["output"] if isinstance(result, dict) else result)

    if df is None:

        )

    print(type(df))
    print(df.shape)
    print(df.head())
