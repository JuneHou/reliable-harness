import pandas as pd
import dotenv
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub

dotenv.load_dotenv()

# Shared namespace used by BOTH the tools and the main script.
# Anything a tool puts in here (e.g. the DataFrame) stays accessible
# as a normal local variable in the main script after the agent finishes.
SHARED_NAMESPACE = {"pd": pd, "df": None}


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file, store the resulting DataFrame in the shared
    namespace under the name `df`, and return a short text summary."""
    df = (
        pd.read_excel(filepath, skiprows=2, names=["responses"])
        .reset_index(drop=True)
        .select_dtypes(include=["object"])
    )
    # Keep the real object locally instead of only returning a string
    SHARED_NAMESPACE["df"] = df
    return (
        f"Loaded {len(df)} rows from {filepath}. "
        "The DataFrame is stored in the shared variable `df`."
    )


def run_agent(user_prompt, file_name="data.json"):
    llm = ChatOpenAI(model="gpt-4o-mini")

    # This is the key fix: the REPL writes to the same dict the main
    # script can read, so variables created by the agent survive the run.
    python_repl = PythonREPL(_globals=SHARED_NAMESPACE)
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description=(
            "Run Python commands in a REPL environment. The loaded DataFrame is "
            "already available as the variable `df` in this environment."
        ),
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(
        agent=shell_agent,
        tools=tools_list,
        verbose=True,
        handle_parsing_errors=True,
        return_intermediate_steps=True,
    )

    response = agent_executor.invoke({"input": user_prompt})
    return response


if __name__ == "__main__":
    path = "excel_file.xlsx"
    user_prompt = (
        f"Load the Excel file located at {path} using the load_excel_response tool. "
        "That tool stores the resulting DataFrame in the shared variable `df`. "
        "Then use the python_repl tool to print df.head()."
    )

    result = run_agent(user_prompt)
    print(result["output"])

    # The DataFrame is now a real local variable in the main script
    df = SHARED_NAMESPACE.get("df")
    if isinstance(df, pd.DataFrame):
        print(df.head())
        df.to_json("data.json", orient="records")
    else:
        print("DataFrame was not captured:", repr(df))
