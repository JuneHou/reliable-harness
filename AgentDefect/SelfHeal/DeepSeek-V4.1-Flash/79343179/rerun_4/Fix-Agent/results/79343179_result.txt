import pandas as pd

from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv

dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# The problem: everything the agent produces lives *inside* the agent run.
# `df` created by the python_repl tool (or by the load_excel_response tool)
# is stored in the REPL's private namespace, which disappears when
# `agent_executor.invoke(...)` returns.  The main script therefore only ever
# sees the agent's final text summary -> "output does not match expectation".
#
# Fix: give the tools and the REPL one *shared, module level* namespace and
# hand that namespace back to the caller, so the resulting DataFrame can be
# retrieved as a normal local variable in the main script.
# ---------------------------------------------------------------------------
AGENT_NAMESPACE = {"pd": pd}


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a transformed DataFrame.

    The DataFrame is stored in the shared namespace under the name ``df`` so
    the calling script can access it after the agent has finished.
    """
    df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
    df = df.select_dtypes(include=['object'])

    AGENT_NAMESPACE["df"] = df          # <-- persists beyond the agent run
    AGENT_NAMESPACE["dataframe"] = df   # small alias, handy for the LLM

    # A tool must return something the LLM can read; a DataFrame object itself
    # is not a valid tool output, so serialise it.
    return df.to_json(orient="records")


def run_agent(user_prompt, file_name="data.json", namespace=None):
    """Run the ReAct agent and return (agent_response, shared_namespace)."""
    namespace = AGENT_NAMESPACE if namespace is None else namespace
    namespace.setdefault("pd", pd)

    llm = ChatOpenAI(model='gpt-4o-mini')

    # Use the SAME dict as the REPL globals so variables created inside the
    # REPL ('df = ...') survive the agent invocation as well.
    python_repl = PythonREPL(_globals=namespace)
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description=(
            "Run Python commands in a REPL environment. The pandas module is "
            "already imported as `pd` and the namespace persists between calls. "
            "Any DataFrame you build must be assigned to the variable named `df`."
        ),
        func=python_repl.run,
    )

    # tool names must be unique and match what the prompt/tool schema exposes
    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(
        agent=shell_agent,
        tools=tools_list,
        verbose=True,
        handle_parsing_errors=True,
        return_intermediate_steps=True,  # gives you the tool outputs too
    )

    response = agent_executor.invoke({"input": user_prompt})
    return response, namespace


def get_dataframe(namespace, name="df"):
    """Fetch the DataFrame produced during the agent run as a local variable."""
    obj = namespace.get(name)
    if obj is None:
        # fall back to any alias / any DataFrame living in the namespace
        for value in namespace.values():
            if isinstance(value, pd.DataFrame):
                return value
        raise ValueError(
            f"No DataFrame named '{name}' was found in the agent namespace. "
            "Make sure the agent actually loaded the data."
        )
    return obj


if __name__ == "__main__":
    path = 'excel_file.xlsx'

    # Be explicit: tell the agent to call the loading tool and to keep the
    # result in `df` (the tool does that automatically now).
    user_prompt = (
        f"Load the Excel file located at {path} by calling the "
        f"`load_excel_response` tool with filepath='{path}'. "
        "The tool stores the resulting pandas DataFrame in the variable `df`. "
        "Do not modify or print the whole DataFrame; finish with a short summary."
    )

    result, agent_namespace = run_agent(user_prompt)

    # The agent's textual answer (what you used to get)
    print(result["output"])

    # ... and now the actual DataFrame, as a plain local variable
    df = get_dataframe(agent_namespace, "df")
    print(df)
    print(df.head())
