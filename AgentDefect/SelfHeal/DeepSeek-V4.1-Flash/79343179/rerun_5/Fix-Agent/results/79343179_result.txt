"""
Fix: after the ReAct agent finishes, the main script wants the ACTUAL pandas
DataFrame (a local variable), not only the agent's text summary.

Why the original code "lost" the DataFrame:
  * `load_excel_response` returned the DataFrame object to the agent, which
    serialises tool output to a string -> the object never reaches the caller.
  * `PythonREPL()` was created with its own private globals/locals, so a
    `df` created inside `python_repl` lived in a throw-away scope that the
    main script could not see.
  * The AgentExecutor's return value is only a dict of strings
    ({"input": ..., "output": ...}).

Fix (move the objects to a shared scope):
  * Create one shared namespace dict in the main script.
  * Make every tool write the DataFrame into that namespace.
  * Build PythonREPL with that same dict (_globals/_locals), so any variable
    the agent creates persists after the run.
  * Return/extract `df` from the namespace in `__main__`.
"""

import pandas as pd

from langchain import hub
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_experimental.utilities import PythonREPL
from langchain_openai import ChatOpenAI

import dotenv

dotenv.load_dotenv()


# ---------------------------------------------------------------------------
# Shared execution scope. This dict *is* the "local scope" of the main script:
# every variable the agent creates (e.g. `df`) is stored here and stays
# available after `agent_executor.invoke(...)` returns.
# ---------------------------------------------------------------------------
agent_locals: dict = {}


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file, store the transformed DataFrame in the shared
    namespace under the name `df`, and return a short description of it.

    A tool can only hand strings back to the LLM, so the DataFrame itself is
    put into the shared namespace instead of being returned.
    """
    raw_df = pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    # pandas >= 3 may store text as StringDtype, so include it explicitly.
    df = raw_df.select_dtypes(include=["object", "string"])

    agent_locals["df"] = df

    return (
        "Stored the transformed DataFrame in the shared variable 'df' "
        f"(shape={df.shape}, columns={list(df.columns)})."
    )


def _get_namespace(namespace=None) -> dict:
    """Always operate on the *same* dict instance."""
    return agent_locals if namespace is None else namespace


def run_agent(user_prompt, namespace=None, file_name="data.json"):
    """Run the ReAct agent and return (agent_response, shared_namespace)."""
    namespace = _get_namespace(namespace)

    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

    # The REPL writes into the same dict the caller reads from, so variables
    # created by the agent survive the agent run.
    python_repl = PythonREPL(_globals=namespace, _locals=namespace)
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description=(
            "Run Python commands in a REPL environment. The REPL shares its "
            "namespace with the calling script, so variables you create "
            "(for example `df`) remain available to the user afterwards."
        ),
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(
        agent=shell_agent,
        tools=tools_list,
        verbose=True,
        return_intermediate_steps=True,
        handle_parsing_errors=True,
        max_iterations=10,
    )

    response = agent_executor.invoke({"input": user_prompt})

    # Guarantee the DataFrame exists even if the LLM skipped the tool.
    if "df" not in namespace:
        load_excel_response.func(namespace.get("__path__", "excel_file.xlsx"))

    return response, namespace


def get_dataframe(path, namespace=None) -> pd.DataFrame:
    """Convenience accessor: return the DataFrame produced during the run."""
    namespace = _get_namespace(namespace)
    if "df" not in namespace:
        load_excel_response.func(path)
    return namespace["df"]


if __name__ == "__main__":
    path = "excel_file.xlsx"
    agent_locals["__path__"] = path

    user_prompt = (
        f"Use the load_excel_response tool on the file at {path} so the "
        "resulting DataFrame is stored in the shared variable named df, "
        "then report its shape and column names."
    )

    result, local_vars = run_agent(user_prompt, namespace=agent_locals)

    print("--- agent summary ---")
    print(result["output"])

    # The DataFrame now exists as a real local variable in the main script.
    df: pd.DataFrame = local_vars.get("df", get_dataframe(path, local_vars))
    print("--- DataFrame retrieved locally ---")
    print(df.head())
    print(df.shape)
