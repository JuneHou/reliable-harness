import os
import traceback
from typing import Any, List, Mapping, Optional

from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import AgentType, initialize_agent
from langchain.callbacks.manager import CallbackManagerForLLMRun
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class LoggingLLM(LLM):
    """Pass-through wrapper around another LLM that only logs traffic.

    Why the original version changed the agent's behaviour:
      * The ReAct ("conversational-react-description") agent relies on stop
        sequences (e.g. "\\nObservation:") to truncate the model output.  The
        wrapper's `_call` accepted `stop` but never forwarded it to the wrapped
        OpenAI model, so the model kept generating past the action/observation
        boundary.  The parser then received extra text, the tool call differed
        and the final answer / intermediate steps no longer matched the
        unwrapped run.
      * `_call` also lacked the `run_manager` / `**kwargs` parameters that
        LangChain passes, which can raise a TypeError (silently swallowed by
        the caller, producing None instead of a real output).

    The fix is to stay a *transparent* proxy: forward every generation
    argument (stop, kwargs) to the wrapped LLM and return its text verbatim.
    """

    llm: LLM
    log: str = "output.log"

    @property
    def _llm_type(self) -> str:
        return "custom"

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        """Get the identifying parameters."""
        return {"llm": self.llm, "log": self.log}

    def _call(
        self,
        prompt: str,
        stop: Optional[List[str]] = None,
        run_manager: Optional[CallbackManagerForLLMRun] = None,
        **kwargs: Any,
    ) -> str:
        # Forward the stop sequences (and any other generation kwargs) so the
        # wrapped model produces byte-identical text to the unwrapped one.
        reply = self.llm(prompt, stop=stop, **kwargs)

        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100)

        # Return the reply untouched: no extra whitespace, no stripping.
        return reply


def setup_components(use_wrapper: bool = True):
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    # `stop` is forwarded by the wrapper, so the agent sees the same text.
    wllm = LoggingLLM(llm=llm, log="output.log") if use_wrapper else llm

    tools = [PythonAstREPLTool()]

    memory = ConversationBufferMemory(memory_key="chat_history")

    return wllm, tools, memory


def create_agent(llm, tools, memory):
    agent = initialize_agent(
        tools=tools,
        llm=llm,
        agent=AgentType.CONVERSATIONAL_REACT_DESCRIPTION,
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt, use_wrapper: bool = True):
    wllm, tools, memory = setup_components(use_wrapper=use_wrapper)
    agent = create_agent(wllm, tools, memory)
    # `input` is the expected input key of the conversational agent chain.
    return agent({"input": prompt})


if __name__ == "__main__":
    prompt = "What is 5-7?"

    try:
        response = run_agent(prompt, use_wrapper=True)
        print(response)
        print(type(response), response["output"])
    except Exception:
        traceback.print_exc()
