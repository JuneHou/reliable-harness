import os
from typing import Any, List, Mapping, Optional

from dotenv import load_dotenv

from langchain.agents import initialize_agent
from langchain.callbacks.manager import CallbackManagerForLLMRun
from langchain.llms.base import LLM
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

try:  # langchain >= 0.2 style import
    from langchain_openai import OpenAI
except ImportError:  # fall back to the legacy module
    from langchain.llms.openai import OpenAI

load_dotenv()

LOG_PATH = "output.log"


class LoggingLLM(LLM):
    """A transparent logging wrapper around another LLM.

    The important part is that *every* argument the caller (the agent) passes in
    must be forwarded to the wrapped LLM.

    Agents such as ``conversational-react-description`` run the LLM through an
    ``LLMChain`` with ``stop=["\\nObservation:"]``. The raw OpenAI LLM forwards
    that ``stop`` list to the API, so generation ends right after the
    ``Action Input``. The original wrapper called ``self.llm(prompt)`` and threw
    ``stop`` away, so the model kept writing its own ``Observation:`` /
    ``Thought:`` lines. That extra text changes the agent's intermediate steps
    (and therefore the final answer), which is why wrapped and unwrapped runs
    looked different.

    ``run_manager`` / ``**kwargs`` are also accepted because newer LangChain
    versions inject them into ``_call``; without them the wrapper raises a
    TypeError that got swallowed by the caller's ``try/except``.
    """

    llm: Any
    log: str = LOG_PATH

    @property
    def _llm_type(self) -> str:
        return "custom"

    def _call(
        self,
        prompt: str,
        stop: Optional[List[str]] = None,
        run_manager: Optional[CallbackManagerForLLMRun] = None,
        **kwargs: Any,
    ) -> str:
        # Forward ``stop`` (and any extra kwargs) so the wrapped LLM behaves
        # exactly like the unwrapped one.
        reply = self.llm(prompt, stop=stop, **kwargs)

        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100)

        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        """Get the identifying parameters."""
        return {"llm": self.llm, "log": self.log}


def setup_components(use_wrapper: bool = True, log_path: str = LOG_PATH):
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    model = LoggingLLM(llm=llm, log=log_path) if use_wrapper else llm

    tools = [PythonAstREPLTool()]

    memory = ConversationBufferMemory(memory_key="chat_history")

    return model, tools, memory


def create_agent(model, tools, memory):
    agent = initialize_agent(
        tools,
        model,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
        return_intermediate_steps=True,
    )
    return agent


def run_agent(prompt: str, use_wrapper: bool = True) -> Optional[dict]:
    try:
        model, tools, memory = setup_components(use_wrapper=use_wrapper)
        agent = create_agent(model, tools, memory)
        response = agent(prompt)
        # ``AgentExecutor`` returns a plain string for single-output agents on
        # some versions, so normalise it to the dict shape the caller expects.
        if isinstance(response, str):
            return {"input": prompt, "output": response, "intermediate_steps": []}
        return response
    except Exception as e:  # surface the error instead of hiding it
        print(f"run_agent failed: {type(e).__name__}: {e}")
        return None


if __name__ == "__main__":
    prompt = "What is 5-7?"

    wrapped = run_agent(prompt, use_wrapper=True)
    raw = run_agent(prompt, use_wrapper=False)

    if wrapped is None or raw is None:
        raise SystemExit("agent run failed, see message above")

    print(type(wrapped), wrapped["output"])
    print(type(raw), raw["output"])

    # The wrapper is now purely cosmetic: prompts, stop handling and results
    # are identical to the unwrapped LLM.
    assert wrapped["output"] == raw["output"], (wrapped["output"], raw["output"])
    print("Wrapped and unwrapped LLM produce identical output.")
