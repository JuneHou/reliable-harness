import os
from typing import Any, List, Mapping, Optional

from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class LoggingWrapper(LLM):
    """Transparent wrapper around an LLM that only logs traffic.

    The important part is that EVERY generation argument the agent relies on is
    forwarded to the wrapped model. In particular the agent passes ``stop``
    sequences such as ``"\nObservation:"`` so that the model stops right after
    producing "Thought / Action / Action Input". The original wrapper called
    ``self.llm(prompt)`` and silently dropped ``stop``, so the wrapped model kept
    generating past the action line (fabricating its own observations/answers).
    That extra text changed the intermediate scratchpad, which is why the wrapped
    and unwrapped LLM produced different agent outputs (and why the parser saw a
    different prompt on the next step).
    """

    llm: OpenAI
    log: str = "output.log"

    @property
    def _llm_type(self) -> str:
        return "custom"

    def _call(
        self,
        prompt: str,
        stop: Optional[List[str]] = None,
        run_manager: Optional[Any] = None,
        **kwargs: Any,
    ) -> str:
        # Forward stop (and any other generation kwargs) so behaviour is identical
        # to calling the underlying LLM directly.
        if stop is None:
            stop = self.stop
        reply = self.llm(prompt, stop=stop, **kwargs)

        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + str(reply) + "\n}\n" + "." * 100)
        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        """Get the identifying parameters."""
        return {"llm": self.llm}


def setup_components():
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    wllm = LoggingWrapper(llm=llm, log="output.log")

    tools = [PythonAstREPLTool()]

    memory = ConversationBufferMemory(memory_key="chat_history")

    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent=AgentType.CONVERSATIONAL_REACT_DESCRIPTION,
        verbose=True,
        memory=memory,
        return_only_outputs=False,
    )
    return agent


def run_agent(prompt, use_wrapper: bool = True):
    try:
        wllm, tools, memory = setup_components()
    except Exception as e:  # keep the failure visible instead of returning None
        print(f"[setup_components] {type(e).__name__}: {e}")
        return None

    llm = wllm if use_wrapper else wllm.llm

    try:
        agent = create_agent(llm, tools, memory)
    except Exception as e:
        print(f"[create_agent] {type(e).__name__}: {e}")
        return None

    try:
        response = agent(prompt)
        return response
    except Exception as e:
        print(f"[run_agent] {type(e).__name__}: {e}")
        return None


if __name__ == "__main__":
    prompt = "What is 5-7?"

    response = run_agent(prompt)
    print(response)
    if isinstance(response, dict):
        print(type(response), response["output"])
