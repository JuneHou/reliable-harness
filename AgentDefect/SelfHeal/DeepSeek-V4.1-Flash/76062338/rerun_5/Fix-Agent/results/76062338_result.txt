import os
import traceback
from typing import Any, List, Mapping, Optional

from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class wrapper(LLM):
    """Logging proxy around another LLM.

    IMPORTANT: a custom `LLM` wrapper must behave *exactly* like the LLM it
    wraps.  LangChain's agents do not only pass a prompt to the model, they
    also pass a list of `stop` sequences (for the
    `conversational-react-description` agent this is
    `["\\nObservation:"]`).  The stop sequence is what tells the model to
    stop writing after its `Action Input:` line, so the agent can execute
    the tool and inject the *real* observation.

    The original implementation accepted `stop` but never forwarded it:

        def _call(self, prompt, stop=None):
            reply = self.llm(prompt)      # <-- stop is dropped!

    As a result the wrapped LLM kept on generating a *hallucinated*
    `Observation: ...` (and sometimes a whole `Thought:/Final Answer:`
    block) after the action.  The raw LLM would have been cut off by the
    stop sequence, so the wrapped model produced different intermediate
    steps / final answer than the un-wrapped model.

    Fix: forward `stop` (and any other kwargs) to the wrapped model so the
    wrapped and unwrapped LLMs return identical text.
    """

    llm: OpenAI
    log: str

    @property
    def _llm_type(self) -> str:
        return "custom"

    def _call(
        self,
        prompt: str,
        stop: Optional[List[str]] = None,
        run_manager: Optional[Any] = None,
        **kwargs: Any,
    ) -> str:
        # Forward `stop` and every extra kwarg to the wrapped LLM, so the
        # wrapper is transparent: identical prompt + identical stop =>
        # identical completion.
        if hasattr(self.llm, "invoke"):
            reply = self.llm.invoke(prompt, stop=stop, **kwargs)
        else:  # very old langchain versions
            reply = self.llm(prompt, stop=stop, **kwargs)

        # Make sure the reply is a plain string (Runnable/legacy call
        # variants may hand back a message-like object).
        if not isinstance(reply, str):
            reply = getattr(reply, "content", str(reply))

        with open(self.log, "a") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100)

        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        """Get the identifying parameters."""
        return {"llm": self.llm}


def setup_components():
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    wllm = wrapper(llm=llm, log="output.log")

    tools = [PythonAstREPLTool()]

    memory = ConversationBufferMemory(memory_key="chat_history")

    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt):
    try:
        wllm, tools, memory = setup_components()
    except Exception:
        traceback.print_exc()
        return None

    try:
        agent = create_agent(wllm, tools, memory)
    except Exception:
        traceback.print_exc()
        return None

    try:
        response = agent(prompt)
        return response
    except Exception:
        traceback.print_exc()
        return None


if __name__ == "__main__":
    prompt = "What is 5-7?"

    response = run_agent(prompt)

    print(response)
    if isinstance(response, dict) and "output" in response:
        print(type(response), response["output"])
    else:
        print(type(response), response)

The change is targeted and addresses a real root-cause bug rather than symptoms:

**The core bug:** `wrapper._call(self, prompt, stop=None)` accepted `stop` but silently dropped it — it called `self.llm(prompt)` with no `stop`. For the `conversational-react-description` agent, the underlying `LLMChain` is built with `stop=["\nObservation:"]`. Without forwarding that stop sequence, the wrapped model keeps generating a *hallucinated* `Observation: …` (and often a whole `Thought:/Final Answer:` block) after the `Action Input:` line.

Consequence: the agent's output parser regex (`Action\s*:...Action\s*Input\s*:(.*)` with `DOTALL`, greedy) then swallows the hallucinated tail into `tool_input`, so `PythonAstREPLTool` receives something like `"5-7\nObservation: -2\nThought: ..."` → `SyntaxError`/wrong tool result → wrong or failed final answer. That precisely explains why `run_agent("What is 5-7?")` didn't reliably yield `-2`. Forwarding `stop` makes the wrapper transparent (same prompt + same stop ⇒ same completion as the un-wrapped LLM), which is the correct fix.
