"""Reproduce the exact token count that BAAI/bge-m3 uses for a text.

Key points:
  * The embedding model's *own* tokenizer is used (not a freshly loaded one and
    not the TokenCountingHandler's default tiktoken tokenizer).
  * The same preprocessing the model applies is reproduced: special tokens
    (<s> ... </s> for XLM-RoBERTa based BGE-M3) and truncation to max_length.
  * Therefore no embedding (forward pass) is required to know the token count,
    and the number reported by the callback handler matches the tokenizer count.
"""

import os
from typing import Callable, List, Tuple

from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from transformers import AutoTokenizer

MODEL_NAME = "BAAI/bge-m3"
CACHE_FOLDER = "./embeddings"
CHUNK_SIZE = 512  # == embedding model max_length


def create_embedding_model(_chunk_size=None, callback_manager=None):
    print('loading embeddings...')
    os.makedirs(CACHE_FOLDER, exist_ok=True)
    _embed_model = HuggingFaceEmbedding(
        model_name=MODEL_NAME,
        max_length=_chunk_size,
        cache_folder=CACHE_FOLDER,
        callback_manager=callback_manager,
    )
    print('embeddings loaded')
    return _embed_model


def get_embed_model_tokenizer(embed_model):
    """Return the exact tokenizer instance the embedding model encodes with."""
    tokenizer = getattr(embed_model, "_tokenizer", None)
    if tokenizer is None:
        tokenizer = getattr(embed_model, "tokenizer", None)

    # sentence-transformers may wrap the HF tokenizer
    if tokenizer is not None and not hasattr(tokenizer, "__call__"):
        inner = getattr(tokenizer, "tokenizer", None)
        if inner is not None:
            tokenizer = inner

    if tokenizer is None:
        # last resort: same checkpoint + same cache folder as the model
        tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME, cache_dir=CACHE_FOLDER)
    return tokenizer


def resolve_max_length(embed_model, tokenizer) -> int:
    """Max number of tokens the model will actually keep (its truncation limit)."""
    max_length = getattr(embed_model, "max_length", None)
    if max_length is None or max_length <= 0 or max_length > 1_000_000:
        max_length = getattr(tokenizer, "model_max_length", None)
    if max_length is None or max_length > 1_000_000:
        # bge-m3 supports 8192 tokens; fall back to that instead of int(1e30)
        max_length = 8192
    return int(max_length)


def encode_like_model(embed_model, text: str) -> List[int]:
    """Token ids exactly as BGE-M3 sees them: special tokens + truncation."""
    tokenizer = get_embed_model_tokenizer(embed_model)
    max_length = resolve_max_length(embed_model, tokenizer)
    encoding = tokenizer(
        text,
        add_special_tokens=True,   # BGE-M3 (XLM-R) adds <s> ... </s>
        truncation=True,           # the model truncates to max_length
        max_length=max_length,
        padding=False,             # padding is not part of the content
        return_attention_mask=False,
        return_token_type_ids=False,
    )
    return encoding["input_ids"]


def count_embedding_tokens(embed_model, text: str) -> int:
    """Exact number of tokens BGE-M3 embeds for `text` - no embedding needed."""
    return len(encode_like_model(embed_model, text))


def build_token_counter(embed_model) -> TokenCountingHandler:
    """A TokenCountingHandler that counts with BGE-M3's own tokenizer.

    Without this, TokenCountingHandler falls back to the default (tiktoken)
    tokenizer, which can never reproduce the BGE-M3 token count.
    """
    tokenizer = get_embed_model_tokenizer(embed_model)
    max_length = resolve_max_length(embed_model, tokenizer)

    def bge_tokenizer(text: str) -> List[int]:
        if not text:
            return []
        return tokenizer(
            text,
            add_special_tokens=True,
            truncation=True,
            max_length=max_length,
            padding=False,
            return_attention_mask=False,
            return_token_type_ids=False,
        )["input_ids"]

    return TokenCountingHandler(tokenizer=bge_tokenizer)


def run_agent(text: str) -> Tuple[int, int]:
    chunk_size = CHUNK_SIZE

    # 1. embedding model (callback manager wired up below)
    embed_model = create_embedding_model(_chunk_size=chunk_size)

    # 2. token counter sharing the embedding model's tokenizer + preprocessing
    token_counter = build_token_counter(embed_model)
    callback_manager = CallbackManager([token_counter])

    # 3. register *both* on Settings; the Settings setter propagates the
    #    callback manager to the already-created embed model, and we set it
    #    explicitly as well so the embedding event is never missed.
    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager
    embed_model.callback_manager = callback_manager

    # 4. embed (this is only needed to validate the counter; the token count
    #    below is computed without running the model at all)
    Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # 5. independent reproduction of the model's tokenization
    token_count = count_embedding_tokens(embed_model, text)

    # 6. they must agree now
    if embedding_tokens != token_count:
        print(
            f"WARNING: handler reported {embedding_tokens} tokens, "
            f"model tokenizer reports {token_count}"
        )
    return embedding_tokens, token_count


if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
    print(f"The tokenizer broke the text into {token_count} tokens")

    assert embedding_tokens == token_count, "token counts still disagree"

    # Token-level detail, so the preprocessing is fully transparent:
    embed_model = Settings.embed_model
    hf_tokenizer = get_embed_model_tokenizer(embed_model)
    ids = encode_like_model(embed_model, text)
    print(f"token ids: {ids}")
    print(f"tokens:    {hf_tokenizer.convert_ids_to_tokens(ids)}")
