"""Reproduce the *exact* token count that BGE-M3 (BAAI/bge-m3) consumes,
including its preprocessing, without running the embedding itself.

Why the two numbers did not match in the original snippet
---------------------------------------------------------
1) ``TokenCountingHandler`` does not use the embedding model's tokenizer. Its
   default tokenizer is a tiktoken encoder (GPT-3.5 / cl100k_base). BGE-M3 uses
   an XLM-RoBERTa *sentencepiece* tokenizer which segments text differently, so
   the "embedding tokens" reported by the handler belonged to another vocab.
2) ``Settings.callback_manager = callback_manager`` was assigned *after* the
   embed model had been constructed. An instantiated embed model keeps its own
   ``callback_manager`` attribute, so the EMBEDDING event could be dropped
   (count stays 0) or handled by a handler with the wrong tokenizer.

Fix
---
* Build the model's tokenizer once and pass it to ``TokenCountingHandler``
  (``tokenizer=...``), so the handler counts with the very same vocabulary.
* Attach the ``CallbackManager`` directly to the embed model instance.
* Apply the model's preprocessing when counting: special tokens ([CLS]/[SEP])
  plus truncation to ``max_length`` -- BGE-M3 adds no instruction prefixes for
  plain text. This is exactly the number of ids fed to the transformer, and it
  is obtained *without* running the model.
"""

import os
from typing import Callable, List, Optional, Tuple

from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from transformers import AutoTokenizer

MODEL_NAME = "BAAI/bge-m3"
CACHE_FOLDER = "./embeddings"
DEFAULT_MAX_LENGTH = 512


def create_embedding_model(_chunk_size: Optional[int] = None) -> HuggingFaceEmbedding:
    """Load BGE-M3 from the local cache (downloading it once if missing)."""
    print("loading embeddings...")
    os.makedirs(CACHE_FOLDER, exist_ok=True)
    _embed_model = HuggingFaceEmbedding(
        model_name=MODEL_NAME,
        max_length=_chunk_size,  # truncation length actually used by the model
        cache_folder=CACHE_FOLDER,
    )
    print("embeddings loaded")
    return _embed_model


def _resolve_tokenizer(embed_model: HuggingFaceEmbedding):
    """Return the exact tokenizer object used by the loaded embedding model."""
    for candidate in (
        getattr(embed_model, "tokenizer", None),
        getattr(embed_model, "_tokenizer", None),
        getattr(getattr(embed_model, "_model", None), "tokenizer", None),
    ):
        if candidate is not None and hasattr(candidate, "encode"):
            return candidate
    # Fallback: load the same tokenizer from the model name / cache folder.
    return AutoTokenizer.from_pretrained(MODEL_NAME, cache_dir=CACHE_FOLDER)


def make_counting_tokenizer(
    tokenizer, max_length: Optional[int]
) -> Callable[[str], List[int]]:
    """Wrap the HF tokenizer so it returns the ids the model really sees.

    Mirrors HuggingFaceEmbedding's preprocessing:
      * add special tokens ([CLS] ... [SEP]),
      * truncate to ``max_length`` (identical to the model's own truncation),
      * no query/text instruction prefix for BGE-M3.
    """

    def _encode(text: str) -> List[int]:
        return tokenizer(
            text,
            add_special_tokens=True,
            truncation=max_length is not None,
            max_length=max_length,
        )["input_ids"]

    return _encode


def count_model_tokens(text: str, tokenizer, max_length: Optional[int] = None) -> int:
    """Exact token count the model uses -- no embedding is computed."""
    return len(make_counting_tokenizer(tokenizer, max_length)(text))


def run_agent(text: str) -> Tuple[int, int]:
    chunk_size = DEFAULT_MAX_LENGTH

    # 1) Load the model, then take *its* tokenizer (same vocab / normalizer).
    embed_model = create_embedding_model(_chunk_size=chunk_size)
    tokenizer = _resolve_tokenizer(embed_model)
    counting_tokenizer = make_counting_tokenizer(tokenizer, chunk_size)

    # 2) Make the token counter use that tokenizer instead of the tiktoken default.
    token_counter = TokenCountingHandler(tokenizer=counting_tokenizer)
    callback_manager = CallbackManager([token_counter])

    # 3) Wire the callback manager into the already-created embed model, otherwise
    #    the EMBEDDING event never reaches the handler.
    embed_model.callback_manager = callback_manager

    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager
    # Re-assert on the instance: assigning Settings.callback_manager may rebind it.
    Settings.embed_model.callback_manager = callback_manager

    # 4) Exact count, no model run -- this is the answer to the question.
    token_count = count_model_tokens(text, tokenizer, max_length=chunk_size)

    # 5) Verification only: the handler must now agree with the tokenizer.
    Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    return embedding_tokens, token_count


if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
    print(f"The tokenizer broke the text into {token_count} tokens")
    assert embedding_tokens == token_count, (embedding_tokens, token_count)
