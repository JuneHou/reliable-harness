import os
import uuid
from datetime import datetime

from dotenv import load_dotenv
import chromadb
from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import CharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings


load_dotenv()


config = {
    "OPENAI_API_KEY": os.getenv("OPENAI_API_KEY")
}


class configs:
    EMBEDDINGS_MODEL = "text-embedding-3-small"


PERSIST_DIRECTORY = "./trained_db"
COLLECTION_NAME = "PDF_Embeddings"


class MockPDFFile:
    def __init__(self, filepath, filename):
        self.filepath = filepath
        self.filename = filename


def get_embeddings():
    return OpenAIEmbeddings(
        openai_api_key=config["OPENAI_API_KEY"],
        model=configs.EMBEDDINGS_MODEL,
    )


def read_docs(pdf_file):
    pdf_loader = PyPDFLoader(pdf_file.filepath)
    pdf_documents = pdf_loader.load()

    text_splitter = CharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
    documents = text_splitter.split_documents(pdf_documents)

    return documents


def generate_and_store_embeddings(documents, pdf_file, user_id):
    client = chromadb.PersistentClient(path=PERSIST_DIRECTORY)
    now = datetime.now()

    # Metadata that has to live on EVERY chunk (not on the collection itself),
    # otherwise it can never be used in a `where` filter.
    metadata = {
        "source": pdf_file.filename,
        "file_path": pdf_file.filepath,
        "user": str(user_id),
        "created_at": now.strftime("%d/%m/%Y %H:%M:%S"),
    }

    # PyPDFLoader only gives: {'source': '<full path>/file.pdf', 'page': n}
    # -> overwrite/merge with our own metadata so the chunks can be looked up
    #    (and deleted) by file name.
    for document in documents:
        document.metadata.update(metadata)

    ids = [str(uuid.uuid4()) for _ in range(len(documents))]

    try:
        vectordb = Chroma.from_documents(
            documents,
            embedding=get_embeddings(),
            persist_directory=PERSIST_DIRECTORY,
            collection_name=COLLECTION_NAME,
            client=client,
            ids=ids,
        )
        vectordb.persist()

    except Exception as err:
        print(f"An error occured: {err=}, {type(err)=}")
        return {"answer": "An error occured while generating embeddings. Please check terminal for more details."}

    return vectordb


def get_pdf_embeddings(vectordb, pdf_file):
    """Return the ids + metadatas of the embeddings that belong to one PDF."""
    result = vectordb.get(where={"source": pdf_file.filename}, include=["metadatas"])

    if not result.get("ids"):
        # Fallback for collections that were ingested *before* the per-chunk
        # metadata fix, where Chroma only holds PyPDFLoader's metadata
        # (source == the full file path).
        result = vectordb.get(where={"source": pdf_file.filepath}, include=["metadatas"])

    return {
        "ids": result.get("ids") or [],
        "metadatas": result.get("metadatas") or [],
    }


def delete_pdf_embeddings(vectordb, pdf_file):
    """Delete only the embeddings that came from this PDF."""
    result = get_pdf_embeddings(vectordb, pdf_file)
    if result["ids"]:
        vectordb.delete(ids=result["ids"])
    return result["ids"]


def run_agent(pdf_path, user_id):

    pdf_file = MockPDFFile(pdf_path, os.path.basename(pdf_path))

    documents = read_docs(pdf_file)
    print(f"    ✓ Loaded {len(documents)} document chunks")

    vectordb = generate_and_store_embeddings(documents, pdf_file, user_id)

    if isinstance(vectordb, dict):

        return None, pdf_file.filename

    return vectordb, pdf_file.filename


if __name__ == "__main__":

    pdf_path = None
    user_id = "test_user_123"

    if os.path.exists("docs"):
        for file in sorted(os.listdir("docs")):
            if file.endswith(".pdf"):
                pdf_path = os.path.join("docs", file)
                break

    if not pdf_path:
        print("No PDF file found in docs folder")
        exit(1)

    vectordb, filename = run_agent(pdf_path, user_id)

    if vectordb is None:
        print("Agent failed to create embeddings")
        exit(1)

    pdf_file = MockPDFFile(pdf_path, filename)

    # `vectordb.get(...)` returns a dict of columns (ids, metadatas, documents),
    # so `len(result)` was always 2/3 - it is `len(result["ids"])` that counts
    # the embeddings of this PDF.
    result = get_pdf_embeddings(vectordb, pdf_file)

    print(f"Embeddings stored for '{filename}': {len(result['ids'])}")
    print(f"IDs: {result['ids']}")
    print(f"Metadatas: {result['metadatas']}")

    # To remove this PDF's embeddings only:
