import os
import uuid
from datetime import datetime

from dotenv import load_dotenv
import chromadb
from chromadb.utils import embedding_functions
from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import CharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings


load_dotenv()


config = {
    "OPENAI_API_KEY": os.getenv("OPENAI_API_KEY")
}

COLLECTION_NAME = "PDF_Embeddings"
PERSIST_DIR = "./trained_db"


class configs:
    EMBEDDINGS_MODEL = "text-embedding-3-small"


class MockPDFFile:
    def __init__(self, filepath, filename):
        self.filepath = filepath
        self.filename = filename


def read_docs(pdf_file):
    pdf_loader = PyPDFLoader(pdf_file.filepath)
    pdf_documents = pdf_loader.load()

    text_splitter = CharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
    documents = text_splitter.split_documents(pdf_documents)

    # Drop empty chunks so every stored row has text and metadata.
    documents = [doc for doc in documents if doc.page_content and doc.page_content.strip()]

    return documents


def tag_documents(documents, pdf_file, user_id, created_at):
    """Replace the loader metadata with the exact metadata we want to query/delete by.

    PyPDFLoader sets `source` to the FULL path (e.g. 'docs/report.pdf'), which is why a
    filter on just the file name ('report.pdf') matched nothing. We normalise it here so
    that `where={"source": <filename>}` is a valid, exact-match filter.
    """
    for index, doc in enumerate(documents):
        page = doc.metadata.get("page", 0)
        doc.metadata = {
            "source": pdf_file.filename,          # file name only -> used for filtering/deleting
            "file_path": pdf_file.filepath,       # remember the original path
            "user": str(user_id),
            "created_at": created_at,
            "page": page,
            "chunk_index": index,
        }
    return documents


def generate_and_store_embeddings(documents, pdf_file, user_id):
    client = chromadb.PersistentClient(path=PERSIST_DIR)

    # Per-document metadata (queryable / deletable) must live on the documents,
    # NOT in `collection_metadata` (which is collection-level only, e.g. hnsw:space).
    now = datetime.now()
    created_at = now.strftime("%d/%m/%Y %H:%M:%S")
    documents = tag_documents(documents, pdf_file, user_id, created_at)

    if not documents:
        print("No usable text chunks were produced for this PDF.")
        return {"answer": "No text could be extracted from the PDF."}

    ids = [str(uuid.uuid4()) for _ in range(len(documents))]

    try:
        vectordb = Chroma.from_documents(
            documents,
            embedding=OpenAIEmbeddings(
                openai_api_key=config["OPENAI_API_KEY"],
                model=configs.EMBEDDINGS_MODEL,
            ),
            persist_directory=PERSIST_DIR,
            collection_name=COLLECTION_NAME,
            client=client,
            ids=ids,
        )
        try:
            vectordb.persist()  # no-op on recent Chroma/LangChain versions
        except Exception:
            pass

    except Exception as err:
        print(f"An error occured: {err=}, {type(err)=}")
        return {"answer": "An error occured while generating embeddings. Please check terminal for more details."}

    return vectordb


def get_pdf_embeddings(vectordb, filename, user_id=None):
    """Return the ids + metadata of the embeddings that belong to a single PDF."""
    where = {"source": filename}
    if user_id is not None:
        where = {"$and": [{"source": filename}, {"user": str(user_id)}]}

    result = vectordb.get(where=where, include=["metadatas", "documents"])

    ids = result.get("ids") or []
    metadatas = result.get("metadatas") or []
    documents = result.get("documents") or []
    return {"ids": ids, "metadatas": metadatas, "documents": documents}


def delete_pdf_embeddings(vectordb, filename, user_id=None):
    """Delete every embedding that came from `filename` (optionally scoped to a user)."""
    found = get_pdf_embeddings(vectordb, filename, user_id)
    ids = found["ids"]
    if ids:
        vectordb.delete(ids=ids)
    return ids


def run_agent(pdf_path, user_id):

    pdf_file = MockPDFFile(pdf_path, os.path.basename(pdf_path))

    documents = read_docs(pdf_file)
    print(f"    ✓ Loaded {len(documents)} document chunks")

    vectordb = generate_and_store_embeddings(documents, pdf_file, user_id)

    if isinstance(vectordb, dict):
        return None, pdf_file.filename

    return vectordb, pdf_file.filename


if __name__ == "__main__":

    pdf_path = None
    user_id = "test_user_123"

    if os.path.exists("docs"):
        for file in os.listdir("docs"):
            if file.endswith(".pdf"):
                pdf_path = os.path.join("docs", file)
                break

    if not pdf_path:
        print("No PDF file found in docs folder")
        exit(1)

    vectordb, filename = run_agent(pdf_path, user_id)

    if vectordb is None:
        print("Agent failed to create embeddings")
        exit(1)

    # Only the embeddings/chunks belonging to THIS pdf are fetched.
    found = get_pdf_embeddings(vectordb, filename, user_id)

    print(f"Embeddings stored for '{filename}': {len(found['ids'])}")
    print("IDs:", found["ids"])
    print("Metadata:", found["metadatas"])

That argument sets **collection-level** metadata (a single dict describing the whole collection — meant for things like `hnsw:space`), not metadata on individual embeddings. Consequently, documents were stored with PyPDFLoader's default metadata only, where `source` is the **full path** (e.g. `./docs/test.pdf`), never the bare filename.
