DUPLICATE_THRESHOLD = float(os.getenv("DUPLICATE_THRESHOLD", "0.95"))
TOP_K = int(os.getenv("TOP_K", "5"))

_DESC_KEYS = ("description", "desc", "control_description", "statement", "text", "details")

# --------------------------------------------------------------------------- #
def _pick(field_map, candidates):
    for candidate in candidates:
        if candidate in field_map:
            return field_map[candidate]
    return None


def load_controls(csv_path=CSV_PATH):
    """Read control.csv and build one record/document per row (row == control)."""
    controls = []
    with open(csv_path, newline="", encoding="utf-8-sig") as handle:
        reader = csv.DictReader(handle)
        field_map = {(f or "").strip().lower(): f for f in (reader.fieldnames or [])}
        name_key = _pick(field_map, _NAME_KEYS)
        desc_key = _pick(field_map, _DESC_KEYS)
        id_key = _pick(field_map, _ID_KEYS)

        for idx, row in enumerate(reader, start=1):
            if not any((value or "").strip() for value in row.values()):
                continue
            name = (row.get(name_key) or "").strip() if name_key else ""
            description = (row.get(desc_key) or "").strip() if desc_key else ""
            control_id = (row.get(id_key) or "").strip() if id_key else ""
            control_id = control_id or f"control-{idx}"
            name = name or control_id

            # Name and Description must stay together: no lossy chunking here.
            text = (
                f"Control ID: {control_id}\n"
                f"Name: {name}\n"
                f"Description: {description}"
            ).strip()

            controls.append(
                {
                    "id": control_id,
                    "name": name,
                    "description": description,
                    "text": text,
                }
            )

    if not controls:
        raise ValueError(f"No controls could be loaded from {csv_path!r}")
    return controls


def controls_to_documents(controls):
    return [
        Document(page_content=c["text"], metadata={"id": c["id"], "name": c["name"]})
        for c in controls
    ]


# --------------------------------------------------------------------------- #
# 2. Pairwise similarity (this is what actually yields the expected list)
# --------------------------------------------------------------------------- #
def embed_controls(controls, embeddings):
    """Embed every control once and return L2-normalized vectors."""
    vectors = embeddings.embed_documents([c["text"] for c in controls])
    matrix = np.asarray(vectors, dtype=np.float32)
    norms = np.linalg.norm(matrix, axis=1, keepdims=True)
    norms[norms == 0] = 1.0
    return matrix / norms


def classify(score):
    if score >= DUPLICATE_THRESHOLD:
        return "duplicate", DUPLICATE_THRESHOLD
    if score >= SIMILAR_THRESHOLD:
        return "similar", SIMILAR_THRESHOLD
    return None, None


def find_candidate_pairs(
    controls,
    unit_vectors,
    similar_threshold=SIMILAR_THRESHOLD,
    duplicate_threshold=DUPLICATE_THRESHOLD,
):
    """All unordered control pairs whose cosine similarity clears a threshold."""
    similarity_matrix = unit_vectors @ unit_vectors.T
    pairs = []
    for i in range(len(controls)):
        for j in range(i + 1, len(controls)):
            score = round(float(similarity_matrix[i, j]), 4)
            if score >= duplicate_threshold:
                relationship, threshold_used = "duplicate", duplicate_threshold
            elif score >= similar_threshold:
                relationship, threshold_used = "similar", similar_threshold
            else:
                continue
            pairs.append(
                {
                    "control_1_id": controls[i]["id"],
                    "control_1": controls[i]["name"],
                    "control_2_id": controls[j]["id"],
                    "control_2": controls[j]["name"],
                    "similarity_score": score,
                    "threshold": threshold_used,
                    "relationship": relationship,
                }
            )
    pairs.sort(key=lambda p: p["similarity_score"], reverse=True)
    return pairs

PROMPT_TEMPLATE = """You are a security/compliance control analyst. You compare control statements to decide which ones are similar and which ones are duplicates.

The candidate pairs have ALREADY been computed for you with cosine similarity on the control embeddings. Each candidate carries its numeric similarity_score and its threshold bucket:
- relationship "duplicate" means similarity_score >= {duplicate_threshold}
- relationship "similar"  means similarity_score >= {similar_threshold} and < {duplicate_threshold}

Your task: keep the candidate pairs that are genuinely about the same requirement, and for each of them explain in one short sentence why.

Strict rules:
1. Only report pairs that appear in the "Candidate pairs" section. Never invent a control, a pair or a score.
2. Copy similarity_score and threshold exactly as given (do not re-estimate them).
3. Sort the result by similarity_score, highest first.
4. If no candidate pairs are relevant, return an empty JSON array: []
5. Reply with a JSON array ONLY - no prose, no markdown code fences.

Every element of the array must have exactly these keys:
"control_1" (string), "control_2" (string), "similarity_score" (number), "threshold" (number), "relationship" ("duplicate" or "similar"), "reason" (string).

Question: {question}

Candidate pairs:
{context}

JSON:
"""


def _message_text(message):
    if isinstance(message, str):
        return message
    content = getattr(message, "content", message)
    if isinstance(content, list):
        return "".join(
            part.get("text", "") if isinstance(part, dict) else str(part)
            for part in content
        )
    return str(content)


def extract_json_array(raw_text):
    text = _message_text(raw_text).strip()
    fenced = re.search(r"```(?:json)?\s*(.*?)```", text, flags=re.S)
    if fenced:
        text = fenced.group(1).strip()
    start, end = text.find("["), text.rfind("]")
    if start == -1 or end == -1 or end < start:
        return []
    try:
        parsed = json.loads(text[start : end + 1])
    except json.JSONDecodeError:
        return []
    return parsed if isinstance(parsed, list) else []


def reconcile_with_computed(llm_pairs, computed_pairs):
    """Trust the computed numbers, not the model's: match names back to the pairs."""
    index = {}
    for pair in computed_pairs:
        index[(pair["control_1"], pair["control_2"])] = pair
        index[(pair["control_2"], pair["control_1"])] = pair

    reconciled = []
    for item in llm_pairs:
        if not isinstance(item, dict):
            continue
        pair = index.get((item.get("control_1"), item.get("control_2")))
        if pair is None:
            continue
        reconciled.append({**pair, "reason": str(item.get("reason", "")).strip()})

    if not reconciled:  # model ignored the contract -> fall back to computed pairs
        return [{**pair, "reason": ""} for pair in computed_pairs]
    return reconciled


def review_pairs(question, pairs, llm):
    """LLM pass over the pre-computed pairs (uses an explicit output contract)."""
    context = json.dumps(pairs[:MAX_PAIRS_FOR_LLM], indent=1)
    prompt = ChatPromptTemplate.from_template(PROMPT_TEMPLATE)
    prompt = prompt.partial(
        similar_threshold=SIMILAR_THRESHOLD,
        duplicate_threshold=DUPLICATE_THRESHOLD,
    )
    chain = (
        {"context": lambda _question: context, "question": RunnablePassthrough()}
        | prompt
        | llm
        | StrOutputParser()
    )
    raw_answer = chain.invoke(question)
    return raw_answer, reconcile_with_computed(extract_json_array(raw_answer), pairs)


# --------------------------------------------------------------------------- #
# 4. Per-control retrieval with real similarity scores (cosine space)
# --------------------------------------------------------------------------- #
def build_vectorstore(controls, embeddings):
    store = Chroma(
        collection_name="controls",
        embedding_function=embeddings,
        collection_metadata={"hnsw:space": "cosine"},  # similarity, not L2 distance
    )
    store.add_documents(controls_to_documents(controls))
    return store


def similar_controls_for(control, store, k=TOP_K):
    hits = store.similarity_search_with_relevance_scores(control["text"], k=k + 1)
    results = []
    for doc, score in hits:
        if doc.metadata.get("id") == control["id"]:
            continue
        score = round(float(score), 4)
        if score < SIMILAR_THRESHOLD:
            continue
        relationship, threshold_used = classify(score)
        results.append(
            {
                "control": doc.metadata.get("name"),
                "similarity_score": score,
                "threshold": threshold_used,
                "relationship": relationship,
            }
        )
    return results[:k]


# --------------------------------------------------------------------------- #
# 5. Agent entry point
# --------------------------------------------------------------------------- #
def run_agent(
    query="List the control pairs that are similar or duplicates, using each control's Name and Description.",
    control_name=None,
):
    embeddings = OpenAIEmbeddings(model=EMBEDDING_MODEL)
    llm = ChatOpenAI(temperature=0, model=LLM_MODEL)

    controls = load_controls()
    unit_vectors = embed_controls(controls, embeddings)
    computed_pairs = find_candidate_pairs(controls, unit_vectors)

    raw_answer, pairs = review_pairs(query, computed_pairs, llm)

    result = {
        "question": query,
        "thresholds": {
            "similar": SIMILAR_THRESHOLD,
            "duplicate": DUPLICATE_THRESHOLD,
        },
        "pair_count": len(pairs),
        "pairs": pairs,
    }

    # Optional: same information from the retrieval side for one control.
    if control_name:
        store = build_vectorstore(controls, embeddings)
        match = next(
            (c for c in controls if c["name"].lower() == control_name.lower()),
            None,
        )
        if match is None:
            result["retrieval"] = {"error": f"{control_name!r} not found in {CSV_PATH}"}
        else:
            result["retrieval"] = {
                "control": match["name"],
                "matches": similar_controls_for(match, store),
            }

    if not pairs:
        result["note"] = (
            "No pair reached the similarity threshold "
            f"({SIMILAR_THRESHOLD}); lower SIMILAR_THRESHOLD or check that the CSV "
            "really contains Name/Description columns."
        )

    return result


if __name__ == "__main__":

    )

def extract_json_array(raw_text):
    text = _message_text(raw_text).strip()
    fenced = re.search(r"```(?:json)?\s*(.*?)```", text, flags=re.S)
    if fenced:
        text = fenced.group(1).strip()
    start, end = text.find("["), text.rfind("]")
    if start == -1 or end == -1 or end < start:
        return []
    try:
        parsed = json.loads(text[start : end + 1])
    except json.JSONDecodeError:
        return []
    return parsed if isinstance(parsed, list) else []


def reconcile_with_computed(llm_pairs, computed_pairs):
    """Trust the computed numbers, not the model's: match names back to the pairs."""
    index = {}
    for pair in computed_pairs:
        index[(pair["control_1"], pair["control_2"])] = pair
        index[(pair["control_2"], pair["control_1"])] = pair

    reconciled = []
    for item in llm_pairs:
        if not isinstance(item, dict):
            continue
        pair = index.get((item.get("control_1"), item.get("control_2")))
        if pair is None:
            continue
        reconciled.append({**pair, "reason": str(item.get("reason", "")).strip()})

    if not reconciled:  # model ignored the contract -> fall back to computed pairs
        return [{**pair, "reason": ""} for pair in computed_pairs]
    return reconciled


def review_pairs(question, pairs, llm):
    """LLM pass over the pre-computed pairs (uses an explicit output contract)."""
    context = json.dumps(pairs[:MAX_PAIRS_FOR_LLM], indent=1)
    prompt = ChatPromptTemplate.from_template(PROMPT_TEMPLATE)
    prompt = prompt.partial(
        similar_threshold=SIMILAR_THRESHOLD,
        duplicate_threshold=DUPLICATE_THRESHOLD,
    )
    chain = (
        {"context": lambda _question: context, "question": RunnablePassthrough()}
        | prompt
        | llm
        | StrOutputParser()
    )
    raw_answer = chain.invoke(question)
    return raw_answer, reconcile_with_computed(extract_json_array(raw_answer), pairs)


# --------------------------------------------------------------------------- #
# 4. Per-control retrieval with real similarity scores (cosine space)
# --------------------------------------------------------------------------- #
def build_vectorstore(controls, embeddings):
    store = Chroma(
        collection_name="controls",
        embedding_function=embeddings,
        collection_metadata={"hnsw:space": "cosine"},  # similarity, not L2 distance
    )
    store.add_documents(controls_to_documents(controls))
    return store


def similar_controls_for(control, store, k=TOP_K):
    hits = store.similarity_search_with_relevance_scores(control["text"], k=k + 1)
    results = []
    for doc, score in hits:
        if doc.metadata.get("id") == control["id"]:
            continue
        score = round(float(score), 4)
        if score < SIMILAR_THRESHOLD:
            continue
        relationship, threshold_used = classify(score)
        results.append(
            {
                "control": doc.metadata.get("name"),
                "similarity_score": score,
                "threshold": threshold_used,
                "relationship": relationship,
            }
        )
    return results[:k]


# --------------------------------------------------------------------------- #
# 5. Agent entry point
# --------------------------------------------------------------------------- #
def run_agent(
    query="List the control pairs that are similar or duplicates, using each control's Name and Description.",
    control_name=None,
):
    embeddings = OpenAIEmbeddings(model=EMBEDDING_MODEL)
    llm = ChatOpenAI(temperature=0, model=LLM_MODEL)

    controls = load_controls()
    unit_vectors = embed_controls(controls, embeddings)
    computed_pairs = find_candidate_pairs(controls, unit_vectors)

    raw_answer, pairs = review_pairs(query, computed_pairs, llm)

    result = {
        "question": query,
        "thresholds": {
            "similar": SIMILAR_THRESHOLD,
            "duplicate": DUPLICATE_THRESHOLD,
        },
        "pair_count": len(pairs),
        "pairs": pairs,
    }

    # Optional: same information from the retrieval side for one control.
    if control_name:
        store = build_vectorstore(controls, embeddings)
        match = next(
            (c for c in controls if c["name"].lower() == control_name.lower()),
            None,
        )
        if match is None:
            result["retrieval"] = {"error": f"{control_name!r} not found in {CSV_PATH}"}
        else:
            result["retrieval"] = {
                "control": match["name"],
                "matches": similar_controls_for(match, store),
            }

    if not pairs:
        result["note"] = (
            "No pair reached the similarity threshold "
            f"({SIMILAR_THRESHOLD}); lower SIMILAR_THRESHOLD or check that the CSV "
            "really contains Name/Description columns."
        )

    return result


if __name__ == "__main__":
    query = (
        "Find controls that are similar or duplicates, combining their Name and "
        "Description, and report each pair with its similarity score and threshold."
    )
    result = run_agent(query)
    print(json.dumps(result, indent=2))
