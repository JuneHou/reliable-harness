"""

Why the original code failed to produce the expected output:
1. A RAG chain never computes similarities. `retriever` returns the top-k chunks for ONE

* and finally asks the LLM to explain that *already computed* list (the prompt is

COLLECTION_NAME = "control_pairs"
PERSIST_DIRECTORY = os.getenv("CHROMA_DIR", "chroma_control_store")

NAME_CANDIDATES = ("Name", "name", "Control", "control", "Title", "title")
DESCRIPTION_CANDIDATES = (
    "Description",
    "description",
    "Desc",
    "desc",
    "Details",
    "details",
)

TOP_K = 10                 # candidates retrieved per control
SIMILARITY_THRESHOLD = 0.70  # below this the pair is discarded
DUPLICATE_THRESHOLD = 0.90   # at/above this the pair is a duplicate


# --------------------------------------------------------------------------- #
# Data loading
# --------------------------------------------------------------------------- #
def _first_present(row: Dict[str, Any], candidates) -> str:
    for key in candidates:
        value = row.get(key)
        if value:
            return str(value).strip()
    return ""


def load_control_documents(csv_path: str = CSV_PATH) -> List[Document]:
    """One Document per control: Name + Description (the text that is embedded)."""
    if not os.path.exists(csv_path):
        raise FileNotFoundError(f"CSV file not found: {csv_path}")

    documents: List[Document] = []
    with open(csv_path, newline="", encoding="utf-8-sig") as handle:
        reader = csv.DictReader(handle)
        if reader.fieldnames is None:
            raise ValueError(f"{csv_path} is empty or has no header row")

        for row_index, row in enumerate(reader):
            name = _first_present(row, NAME_CANDIDATES)
            description = _first_present(row, DESCRIPTION_CANDIDATES)

            # Fall back to the whole row if the column names are unknown.
            if not name and not description:
                name = " | ".join(str(v).strip() for v in row.values() if v)

            if not name and not description:
                continue

            content = f"Name: {name}\nDescription: {description}".strip()
            documents.append(
                Document(
                    page_content=content,
                    metadata={
                        "id": str(row_index),
                        "name": name,
                        "description": description,
                    },
                )
            )

    if not documents:
        raise ValueError(f"No usable rows found in {csv_path}")

    return documents


# --------------------------------------------------------------------------- #
# Vector store
# --------------------------------------------------------------------------- #
def build_vectorstore(documents: List[Document]) -> Chroma:
    """Cosine space: Chroma then returns `1 - cosine_similarity` as the distance."""
    return Chroma.from_documents(
        documents=documents,
        embedding=OpenAIEmbeddings(),
        collection_name=COLLECTION_NAME,
        persist_directory=PERSIST_DIRECTORY,
        collection_metadata={"hnsw:space": "cosine"},
    )


def distance_to_similarity(distance: float) -> float:
    """Cosine distance (0..2) -> similarity (1..-1), clamped to [0, 1]."""
    return max(0.0, min(1.0, 1.0 - float(distance)))


# --------------------------------------------------------------------------- #
# Core: deterministic pair extraction (this is what the user actually wanted)
# --------------------------------------------------------------------------- #
def find_control_pairs(
    documents: List[Document],
    vectorstore: Chroma,
    top_k: int = TOP_K,
    similarity_threshold: float = SIMILARITY_THRESHOLD,
    duplicate_threshold: float = DUPLICATE_THRESHOLD,
) -> List[Dict[str, Any]]:
    """Return every similar/duplicate pair, its score and its classification."""
    if duplicate_threshold < similarity_threshold:
        raise ValueError("duplicate_threshold must be >= similarity_threshold")

    k = max(1, min(top_k + 1, len(documents)))
    collected: Dict[tuple, Dict[str, Any]] = {}
    names = {doc.metadata["id"]: doc.metadata.get("name", "") for doc in documents}

    for document in documents:
        doc_id = document.metadata["id"]
        results = vectorstore.similarity_search_with_score(document.page_content, k=k)

        for other, distance in results:
            other_id = other.metadata.get("id")
            if other_id is None or other_id == doc_id:
                continue  # never pair a control with itself

            similarity = round(distance_to_similarity(distance), 4)
            if similarity < similarity_threshold:
                continue

            key = tuple(sorted((doc_id, other_id)))
            existing = collected.get(key)
            if existing is not None and existing["similarity"] >= similarity:
                continue  # symmetric hit with a lower score -> keep the best one

            collected[key] = {
                "control_a_id": key[0],
                "control_a": names.get(key[0], ""),
                "control_b_id": key[1],
                "control_b": names.get(key[1], ""),
                "similarity": similarity,
                "duplicate": similarity >= duplicate_threshold,
                "similarity_threshold": similarity_threshold,
                "duplicate_threshold": duplicate_threshold,
            }

    return sorted(collected.values(), key=lambda pair: pair["similarity"], reverse=True)

PROMPT_TEMPLATE = """You are an audit assistant that reports on control comparisons.

You are given a PRE-COMPUTED list of control pairs. Do NOT invent pairs and do NOT
recompute anything: use only the pairs, similarity values and thresholds below.

Definitions:
- similar  : similarity >= {similarity_threshold} (all pairs below already satisfy this)
- duplicate: similarity >= {duplicate_threshold}

Rules:
- List EVERY pair that is supplied, ordered by similarity (highest first).
- For each pair output exactly: "A" <-> "B" | similarity=<score> | duplicate=yes/no
- After the list, state both thresholds and how many pairs are duplicates.
- Do not answer with prose about the task, and do not add pairs that are not supplied.
- If the list is empty, reply exactly: "No pairs met the similarity threshold of
  {similarity_threshold}."

Task/question: {question}

Pre-computed pairs ({pair_count} total):
{context}

Answer:"""


def build_llm_chain():
    prompt = ChatPromptTemplate.from_template(PROMPT_TEMPLATE)
    llm = ChatOpenAI(temperature=0, model="gpt-3.5-turbo")
    return prompt | llm | StrOutputParser()


def format_pairs_for_prompt(pairs: List[Dict[str, Any]]) -> str:
    if not pairs:
        return "(empty)"
    return "\n".join(
        (
            f'- "{pair["control_a"]}" <-> "{pair["control_b"]}" | '
            f'similarity={pair["similarity"]:.4f} | '
            f'duplicate={"yes" if pair["duplicate"] else "no"}'
        )
        for pair in pairs
    )

# Entry point
# --------------------------------------------------------------------------- #
def run_agent(
    query: str,
    csv_path: str = CSV_PATH,
    top_k: int = TOP_K,
    similarity_threshold: float = SIMILARITY_THRESHOLD,
    duplicate_threshold: float = DUPLICATE_THRESHOLD,
    with_llm_summary: bool = True,
) -> Dict[str, Any]:

    vectorstore = build_vectorstore(documents)

        documents=documents,
        vectorstore=vectorstore,
        top_k=top_k,
        similarity_threshold=similarity_threshold,

    result: Dict[str, Any] = {
        "pairs": pairs,
        "similar_pairs": [p for p in pairs if not p["duplicate"]],
        "duplicate_pairs": [p for p in pairs if p["duplicate"]],
        "similarity_threshold": similarity_threshold,
        "duplicate_threshold": duplicate_threshold,
        "controls_compared": len(documents),
        "answer": None,

        chain = build_llm_chain()
        result["answer"] = chain.invoke(

                "question": query,
                "context": format_pairs_for_prompt(pairs),
                "pair_count": len(pairs),
                "similarity_threshold": similarity_threshold,
                "duplicate_threshold": duplicate_threshold,
            }
        )

    return result
