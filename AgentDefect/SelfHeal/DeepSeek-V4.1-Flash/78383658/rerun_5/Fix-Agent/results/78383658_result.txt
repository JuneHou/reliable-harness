#!/usr/bin/env python3
"""
Fixed LangChain pipeline: find SIMILAR / DUPLICATE control pairs and report the
similarity scores and thresholds that produced them.

Why the original script never produced the expected list of pairs
---------------------------------------------------------------
1. Wrong task framing / prompt: the template was a generic QA prompt
   ("answer in three sentences maximum, keep it concise"). Such a prompt asks
   the model for prose, never for an *exhaustive list of pairs* and never for a
   score or threshold per pair. The prompt must state the task, the scoring
   rules and the exact JSON output shape (and it must be fed the scores).
2. The scores were thrown away: `vectorstore.as_retriever()` returns only
   `Document` objects. The similarity value is lost, so the chain had no way to
   apply a threshold. Use `similarity_search_with_relevance_scores(...)` (or a
   `similarity_score_threshold` retriever) instead.
3. `CharacterTextSplitter(chunk_size=500, ...)` chopped the CSV rows into
   arbitrary fragments, so a control's "Name + Description" was split across
   chunks and no longer belonged to one comparable unit. One row = one document.
4. The vector store was rebuilt on every `run_agent()` call and never persisted,
   so results were neither cheap nor reproducible. Build/persist it once.
5. Deprecated import paths (`langchain.vectorstores`, `langchain.embeddings`,
   `langchain.schema.runnable`, `langchain.prompts`, `ChatOpenAI(verbose=True)`).
6. Comparing a document with the store it lives in returns the document itself
   first: self-hits must be filtered out and unordered pairs de-duplicated
   (A->B and B->A are the same pair).

import json
import os
from typing import Any

from dotenv import load_dotenv

from langchain_community.vectorstores import Chroma
from langchain_core.output_parsers import JsonOutputParser
from langchain_core.prompts import ChatPromptTemplate

EMBEDDING_MODEL = os.getenv("EMBEDDING_MODEL", "text-embedding-3-small")
LLM_MODEL = os.getenv("LLM_MODEL", "gpt-4o-mini")

# text-embedding-3-* gives ~0.70+ for paraphrases and ~0.95+ for near duplicates.
SIMILARITY_THRESHOLD = float(os.getenv("SIMILARITY_THRESHOLD", "0.75"))
DUPLICATE_THRESHOLD = float(os.getenv("DUPLICATE_THRESHOLD", "0.90"))
TOP_K = int(os.getenv("TOP_K", "5"))          # neighbours per control
REBUILD_INDEX = os.getenv("REBUILD_INDEX", "0") == "1"

Request: {question}

You are given pre-scored candidate pairs that were retrieved from a vector store.
"similarity_score" is the cosine similarity of the embeddings of
"Name + Description" and is already in [0, 1]. Apply these rules exactly:
- similarity_score >= {duplicate_threshold}  -> label "duplicate"
- {similarity_threshold} <= similarity_score < {duplicate_threshold} -> label "similar"

{candidates}

Return ONLY a JSON object with no prose and no markdown fences, in this shape:

Rules for the answer:

- Order pairs from most to least similar.

- If no candidate reaches the threshold, return {{"pairs": []}}.

)

def load_controls(path: str) -> list[dict[str, Any]]:
    with open(path, newline="", encoding="utf-8") as fh:
        rows = list(csv.DictReader(fh))
    if not rows:
        raise ValueError(f"{path} is empty or has no header row")

    fields = [f for f in rows[0].keys() if f]
    lower = {f.strip().lower(): f for f in fields}
    name_field = next(
        (lower[k] for k in ("name", "movie", "title", "control", "control name") if k in lower),
        fields[0],
    )
    desc_field = next((f for f in fields if "desc" in f.lower()), fields[-1])

    controls: list[dict[str, Any]] = []
    for idx, row in enumerate(rows):
        name = (row.get(name_field) or "").strip()
        description = (row.get(desc_field) or "").strip()
        if not (name or description):
            continue
        controls.append(
            {
                "id": str(idx),
                "name": name,
                "description": description,
                # the comparable unit: Name and Description combined
                "text": f"Name: {name}\nDescription: {description}",
            }
        )

    # Identical texts would otherwise match each other with score ~1.0 forever.
    seen: set[str] = set()
    unique: list[dict[str, Any]] = []
    for control in controls:
        if control["text"] not in seen:
            seen.add(control["text"])
            unique.append(control)
    return unique

def get_controls() -> list[dict[str, Any]]:

def get_vectorstore() -> Chroma:
    """Build the index once and reuse/persist it (cosine space => sane scores)."""
    if "vectorstore" in _CACHE:
        return _CACHE["vectorstore"]

    existing = os.path.isdir(PERSIST_DIR) and os.listdir(PERSIST_DIR)
    if existing and not REBUILD_INDEX:
        store = Chroma(
            collection_name=COLLECTION,
            embedding_function=embeddings,
            persist_directory=PERSIST_DIR,
            collection_metadata={"hnsw:space": "cosine"},
        )
        if store._collection.count() == len(get_controls()):
            _CACHE["vectorstore"] = store
            return store

    controls = get_controls()
    store = Chroma.from_texts(
        texts=[c["text"] for c in controls],
        embedding=embeddings,
        metadatas=[
            {"id": c["id"], "name": c["name"], "description": c["description"]}
            for c in controls
        ],
        collection_name=COLLECTION,
        persist_directory=PERSIST_DIR,
        collection_metadata={"hnsw:space": "cosine"},
    )
    if hasattr(store, "persist"):
        store.persist()
    _CACHE["vectorstore"] = store
    return store

# Pair generation with real scores + thresholds
# --------------------------------------------------------------------------- #
def _clamp(value: float) -> float:
    return max(0.0, min(1.0, float(value)))


def find_candidate_pairs(
    question: str | None = None,
    similarity_threshold: float = SIMILARITY_THRESHOLD,
    k: int = TOP_K,
) -> list[dict[str, Any]]:
    controls = get_controls()
    store = get_vectorstore()
    by_id = {c["id"]: c for c in controls}
    k = max(2, min(k, len(controls)))

    best: dict[tuple[str, str], dict[str, Any]] = {}
    for control in controls:
        hits = store.similarity_search_with_relevance_scores(control["text"], k=k)
        for doc, score in hits:
            other_id = doc.metadata.get("id")
            if other_id is None or other_id == control["id"]:
                continue  # never pair a control with itself
            score = _clamp(score)
            key = tuple(sorted((control["id"], other_id)))
            if key in best and best[key]["similarity_score"] >= score:
                continue
            first, second = by_id[key[0]], by_id[key[1]]
            best[key] = {
                "id_1": first["id"],
                "name_1": first["name"],
                "text_1": first["text"][:400],
                "id_2": second["id"],
                "name_2": second["name"],
                "text_2": second["text"][:400],
                "similarity_score": round(score, 4),
            }

    pairs = [p for p in best.values() if p["similarity_score"] >= similarity_threshold]
    pairs.sort(key=lambda p: p["similarity_score"], reverse=True)
    return pairs


def _apply_labels(pairs: list[dict[str, Any]]) -> list[dict[str, Any]]:
    for pair in pairs:
        pair["label"] = (
            "duplicate" if pair["similarity_score"] >= DUPLICATE_THRESHOLD else "similar"
        )
        pair.setdefault("reason", "vector similarity above threshold")
    pairs.sort(key=lambda p: p["similarity_score"], reverse=True)
    return pairs


def verify_pairs_with_llm(
    question: str,
    pairs: list[dict[str, Any]],
) -> list[dict[str, Any]]:
    """Ask the LLM to explain/label the pre-scored candidates (never to invent)."""
    if not pairs:
        return []

    chain = PROMPT | llm | JsonOutputParser()
    payload = {
        "question": question,
        "similarity_threshold": SIMILARITY_THRESHOLD,
        "duplicate_threshold": DUPLICATE_THRESHOLD,
        "candidates": json.dumps(pairs, ensure_ascii=False, indent=2),
    }

    try:
        parsed = chain.invoke(payload)
    except Exception as exc:  # noqa: BLE001 - keep the deterministic result instead
        print(f"[warn] LLM verification failed ({exc}); returning scored pairs only")
        return _apply_labels([dict(p) for p in pairs])

    llm_pairs = parsed.get("pairs", []) if isinstance(parsed, dict) else []

    known: dict[tuple[str, str], dict[str, Any]] = {}
    for pair in pairs:
        known[(pair["name_1"], pair["name_2"])] = pair
        known[(pair["name_2"], pair["name_1"])] = pair

    result: list[dict[str, Any]] = []
    seen_keys: set[tuple[str, str]] = set()
    for item in llm_pairs:
        if not isinstance(item, dict):
            continue
        key = (item.get("name_1"), item.get("name_2"))
        source = known.get(key)
        if source is None:
            continue  # hallucinated / unknown pair -> drop it
        canonical = tuple(sorted((source["id_1"], source["id_2"])))
        if canonical in seen_keys:
            continue
        seen_keys.add(canonical)

        out = dict(source)
        out["similarity_score"] = source["similarity_score"]  # trust the vector store
        out["label"] = (
            "duplicate"
            if source["similarity_score"] >= DUPLICATE_THRESHOLD
            else "similar"
        )
        reason = (item.get("reason") or "").strip()
        out["reason"] = reason or "vector similarity above threshold"
        out.pop("text_1", None)
        out.pop("text_2", None)
        result.append(out)

    if not result:  # model returned nothing usable -> deterministic fallback
        result = _apply_labels([dict(p) for p in pairs])

    result.sort(key=lambda p: p["similarity_score"], reverse=True)
    return result


def run_agent(query: str) -> dict[str, Any]:
    """Return the similar/duplicate pairs for `query`, with scores and thresholds."""
    candidates = find_candidate_pairs(query)
    pairs = verify_pairs_with_llm(query, candidates)
    return {
        "question": query,
        "similarity_threshold": SIMILARITY_THRESHOLD,
        "duplicate_threshold": DUPLICATE_THRESHOLD,
        "pair_count": len(pairs),
        "pairs": pairs,
    }


if __name__ == "__main__":

    result = run_agent(query)

    seen_keys: set[tuple[str, str]] = set()
    for item in llm_pairs:
        if not isinstance(item, dict):
            continue
        key = (item.get("name_1"), item.get("name_2"))
        source = known.get(key)
        if source is None:
            continue  # hallucinated / unknown pair -> drop it
        canonical = tuple(sorted((source["id_1"], source["id_2"])))
        if canonical in seen_keys:
            continue
        seen_keys.add(canonical)

        out = dict(source)
        out["similarity_score"] = source["similarity_score"]  # trust the vector store
        out["label"] = (
            "duplicate"
            if source["similarity_score"] >= DUPLICATE_THRESHOLD
            else "similar"
        )
        reason = (item.get("reason") or "").strip()
        out["reason"] = reason or "vector similarity above threshold"
        out.pop("text_1", None)
        out.pop("text_2", None)
        result.append(out)

    if not result:  # model returned nothing usable -> deterministic fallback
        result = _apply_labels([dict(p) for p in pairs])

    result.sort(key=lambda p: p["similarity_score"], reverse=True)
    return result


def run_agent(query: str) -> dict[str, Any]:
    """Return the similar/duplicate pairs for `query`, with scores and thresholds."""
    candidates = find_candidate_pairs(query)
    pairs = verify_pairs_with_llm(query, candidates)
    return {
        "question": query,
        "similarity_threshold": SIMILARITY_THRESHOLD,
        "duplicate_threshold": DUPLICATE_THRESHOLD,
        "pair_count": len(pairs),
        "pairs": pairs,
    }


if __name__ == "__main__":
    query = "Find movies for family, combining their Name and Description."
    result = run_agent(query)
    print(json.dumps(result, indent=2, ensure_ascii=False))
