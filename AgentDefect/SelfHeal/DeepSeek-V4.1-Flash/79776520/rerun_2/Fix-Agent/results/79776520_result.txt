import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    simple_search = TavilySearch(max_results=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key,
    )

    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory,
    )

    # Same thread_id => the checkpointer persists the conversation across calls.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, memory, config, user_input):
    # Do NOT read the state and re-append the history by hand.
    # With a checkpointer attached, the graph already loads the thread state and
    # the `messages` channel uses the `add_messages` reducer, so simply passing the
    # new human message appends it to the stored history. Manually concatenating
    # history + new message re-injected the whole conversation on every call,
    # duplicating turns (and producing broken/incorrect model replies).
    final_message = None
    for step in agent_executor.stream(
        {"messages": [HumanMessage(content=user_input)]},
        config,
        stream_mode="values",
    ):
        final_message = step["messages"][-1]

    if final_message is None:
        return ""

    content = final_message.content
    # Gemini may return content as a list of content blocks.
    if isinstance(content, list):
        parts = []
        for block in content:
            if isinstance(block, dict):
                parts.append(block.get("text", ""))
            else:
                parts.append(str(block))
        content = "".join(parts)
    return content


def main():
    agent_executor, memory, config = initialize_agent()
    response1 = run_agent(agent_executor, memory, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, memory, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
