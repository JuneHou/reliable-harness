import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def _content_to_text(content) -> str:
    """Gemini may return a list of content blocks instead of a plain string."""
    if isinstance(content, str):
        return content
    if isinstance(content, list):
        parts = []
        for block in content:
            if isinstance(block, str):
                parts.append(block)
            elif isinstance(block, dict):
                text = block.get("text")
                if text:
                    parts.append(text)
        return "\n".join(parts).strip()
    return str(content)


def initialize_agent():
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    simple_search = TavilySearch(max_results=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key
    )

    # The checkpointer is what gives the agent cross-call memory.
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory
    )

    # All calls sharing this thread_id share the same persisted state.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, memory, config, user_input):
    # IMPORTANT: only send the *new* user message.
    # When a checkpointer is configured, LangGraph automatically loads the
    # persisted state for `thread_id` and applies the input through the
    # `add_messages` reducer. Manually re-injecting the stored history here
    # (memory.get(...) + history + [HumanMessage(...)]) duplicates every
    # message in the thread, which corrupts the conversation and makes the
    # agent lose track of earlier turns such as the user's name.
    result = agent_executor.invoke(
        {"messages": [HumanMessage(content=user_input)]},
        config,
    )

    final_message = result["messages"][-1]
    return _content_to_text(final_message.content)


def main():
    agent_executor, memory, config = initialize_agent()

    response1 = run_agent(agent_executor, memory, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, memory, config, "What's my name?")

    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
