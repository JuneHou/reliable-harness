import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool


load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    name = "calculator"
    description = "Performs basic arithmetic"
    # Short-circuit the agent loop: once this tool runs, its raw output is
    # returned to the caller immediately instead of being fed back to the LLM
    # for reformatting.
    return_direct = True

    def _run(self, query: str) -> str:
        return str(eval(query))

    async def _arun(self, query: str) -> str:
        raise NotImplementedError()


def create_agent():
    llm_gpt4 = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    return initialize_agent(
        agent="chat-conversational-react-description",
        tools=[CalculatorTool()],
        llm=llm_gpt4,
        verbose=True,
        max_iterations=2,
        early_stopping_method="generate",
        memory=memory,
        # agent_kwargs=dict(output_parser=output_parser),
    )


def run_agent(prompt: str) -> str:
    agent = create_agent()
    return agent.run(prompt)

### 2. Root-cause analysis — does this fix the failing tests?
The tests assert an **exact string equality**:

```python
self.assertEqual(run_agent("10*10"), "100")
```

With the buggy code, the ReAct loop works like this:
1. LLM emits `Action: calculator / Action Input: 10*10`
2. Tool returns `"100"` as the **observation**
3. Because the tool is a normal (non-`return_direct`) tool, the observation is fed back to the LLM and the LLM produces a natural-language `Final Answer`, e.g. *"The result of 10*10 is 100."*
4. `agent.run()` returns that prose, not `"100"` → `assertEqual` fails.

Setting `return_direct = True` makes `AgentExecutor` short-circuit: after the tool runs, the raw observation becomes an `AgentFinish` whose `return_values["output"]` is the observation itself, i.e. exactly `"100"` / `"12"`. This is precisely the mechanism LangChain provides for "tool output *is* the answer." So the change is semantically correct and targets the real defect (final-answer reformatting), not a symptom.
