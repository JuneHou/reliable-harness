import os

from dotenv import load_dotenv
from langchain.agents import AgentType, initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain.tools import BaseTool
from langchain_openai import ChatOpenAI


load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    name: str = "calculator"
    description: str = "Performs basic arithmetic"

    # KEY FIX: mark this tool as "return direct". As soon as the agent calls it,
    # AgentExecutor stops the loop and returns the tool's raw observation as the
    # final output -- the observation is never fed back to the LLM, so there is
    # no extra LLM call that reformats/rewraps the result for the client.
    return_direct: bool = True

    def _run(self, query: str) -> str:
        # Return a plain string so the value handed back to the caller is the
        # raw tool output (not a dict/serialized action input).
        return str(eval(query))

    async def _arun(self, query: str) -> str:
        raise NotImplementedError()


def create_agent():
    llm_gpt4 = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    return initialize_agent(
        agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
        tools=[CalculatorTool()],
        llm=llm_gpt4,
        verbose=True,
        max_iterations=2,
        # "force" stops with the last observation instead of "generate", which
        # would call the LLM one more time and reformat the output.
        early_stopping_method="force",
        memory=memory,
    )


def run_agent(prompt: str) -> str:
    agent = create_agent()
    # With return_direct=True the raw tool output is the final answer.
    return agent.run(prompt)


if __name__ == "__main__":
    print(run_agent("What is 5 * 6?"))
