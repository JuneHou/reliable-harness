import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool


load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    name = "calculator"
    description = "Performs basic arithmetic"

    # Key fix: when a tool has return_direct=True, the AgentExecutor stops
    # right after the tool runs and returns the tool's raw output as the
    # final agent answer, without asking the LLM to reformat/paraphrase it.
    return_direct = True

    def _run(self, query: str) -> str:
        return str(eval(query))

    async def _arun(self, query: str) -> str:
        raise NotImplementedError()


def create_agent():
    llm_gpt4 = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    return initialize_agent(
        agent="chat-conversational-react-description",
        tools=[CalculatorTool()],
        llm=llm_gpt4,
        verbose=True,
        max_iterations=2,
        # "force" simply returns a generic iteration-limit message, so it is
        # not needed anymore: return_direct already terminates the run after
        # the calculator tool executes and passes the raw result straight
        # through to the caller.
        early_stopping_method="force",
        memory=memory,
    )


def run_agent(prompt: str) -> str:
    agent = create_agent()
    # With return_direct=True on the calculator tool, this is the tool's raw
    # output (e.g. "4") when the calculator is used, with no extra LLM pass.
    return agent.run(prompt)


if __name__ == "__main__":
    print(run_agent("What is 2 + 2?"))
