from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# For a *chat* agent the memory must be injected into the prompt as a list of
# messages (a MessagesPlaceholder("chat_history")), so it must return messages
# instead of a plain string. Without return_messages=True the string buffer is
# not usable by chat prompts.
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True,
    output_key="output",
)

agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    # CHAT_ZERO_SHOT_REACT_DESCRIPTION's default prompt only contains
    # [system, human("{input}\n\n{agent_scratchpad}")] - it has no
    # "chat_history" placeholder, so the memory was written but never read
    # back into the model input (hence "what is the last question I asked?"
    # could not be answered).
    # CHAT_CONVERSATIONAL_REACT_DESCRIPTION ships a prompt that already
    # includes MessagesPlaceholder("chat_history"), so memory is actually
    # visible to the chat model.
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    agent_kwargs={
        "ai_prefix": "AI",
        "human_prefix": "Human",
    },
    verbose=False,
)


def run_agent(prompt: str) -> str:
    return agent_test.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    result2 = run_agent("what is the last question I asked?")
    print(result2)

    # Notes:
    # * If you want to keep AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION, you must
    #   supply a custom prompt that contains a MessagesPlaceholder("chat_history")
    #   (e.g. build the agent manually with
    #    ChatPromptTemplate.from_messages([... MessagesPlaceholder("chat_history") ...])
    #    plus a MessagesPlaceholder("agent_scratchpad") and pass it to an
    #    AgentExecutor together with the memory). Otherwise the agent simply
    #    cannot see previous turns.
