"""Fixed example: making a LangChain chat agent actually use conversation memory.

Root cause of the wrong output
------------------------------
`AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION` builds its prompt from
`langchain/agents/chat/prompt.py`, and that prompt declares only these input
variables:  ``{tools}``, ``{tool_names}``, ``{input}``, ``{agent_scratchpad}``.

There is **no** ``{chat_history}`` placeholder in that prompt.  AgentExecutor
will still load the memory variables and store every turn in
ConversationBufferMemory, but ``Agent.plan()`` renders the prompt using only
the variables the prompt itself declares, so ``chat_history`` is silently
dropped.  The model therefore never sees the previous question and the second
call ("what is the last question I asked?") answers as if nothing had been
asked before.

Two valid fixes
---------------
1) (used below) Switch to the *conversational* chat agent,
   ``AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION``, whose prompt contains a
   ``MessagesPlaceholder("chat_history")``.  Because that placeholder expects a
   list of messages, the memory must be built with
   ``ConversationBufferMemory(memory_key="chat_history", return_messages=True)``.
   The memory key must match the placeholder name exactly.

2) Alternatively keep ``CHAT_ZERO_SHOT_REACT_DESCRIPTION`` and supply your own
   prompt that contains ``MessagesPlaceholder("chat_history")`` (plus the usual
   ``tools``/``tool_names``/``input``/``agent_scratchpad`` variables) through
   ``agent_kwargs={"prompt": my_custom_prompt}``.  Without that extra
   configuration the plain zero-shot chat agent simply cannot use memory.
"""

from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# The conversational chat agent's prompt has a `chat_history` MessagesPlaceholder,
# so memory must (a) use the same key and (b) return chat *messages*, not a string.
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True,
)

agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    # CHAT_ZERO_SHOT_REACT_DESCRIPTION has no chat_history slot in its prompt,
    # so use the conversational variant which does.
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False,
)


def run_agent(prompt: str) -> str:
    # dict input is required so the memory variables are merged by the chain
    return agent_test.invoke({"input": prompt})["output"]


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)  # -> "The capital of France is Paris."
    result2 = run_agent("what is the last question I asked?")
    print(result2)  # -> mentions "the capital of France"
