"""Chat agent (CHAT_ZERO_SHOT_REACT_DESCRIPTION) that can actually SEE its memory.

Why the original code "forgot" the previous turn
------------------------------------------------
`AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION` does NOT build its prompt from your
memory. It instantiates `ChatAgent` with its own hard-coded
`ChatPromptTemplate` that only has the variables:

    SystemMessage(...)            # tool descriptions + ReAct/JSON format
    HumanMessage("{input}\n\n{agent_scratchpad}")

There is no `{chat_history}` placeholder (nor a `MessagesPlaceholder`) anywhere
in that prompt, so `ConversationBufferMemory` happily *stores* the turn but the
history is never rendered into the messages sent to the LLM. Also, the memory
was created without `return_messages=True`, which a chat prompt needs when it
uses a `MessagesPlaceholder`.

The fix (two parts, both required):
  1. Build your own `ChatPromptTemplate` with a `MessagesPlaceholder`
     (variable_name="chat_history") between the system message and the human
     message, and inject it via `agent_kwargs={"prompt": prompt}`.
  2. Create the memory with `memory_key="chat_history"` AND
     `return_messages=True` so that the memory returns a list of
     HumanMessage/AIMessage objects (exactly what a MessagesPlaceholder wants),
     and `output_key="output"` so the agent's final answer is stored back.
"""

import os

from dotenv import load_dotenv

from langchain_openai import ChatOpenAI
from langchain.agents import AgentType, initialize_agent, load_tools
from langchain.memory import ConversationBufferMemory

try:  # langchain >= 0.1
    from langchain_core.prompts import (
        ChatPromptTemplate,
        HumanMessagePromptTemplate,
        MessagesPlaceholder,
        SystemMessagePromptTemplate,
    )
except ImportError:  # older langchain
    from langchain.prompts import (
        ChatPromptTemplate,
        HumanMessagePromptTemplate,
        MessagesPlaceholder,
        SystemMessagePromptTemplate,
    )

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# ---------------------------------------------------------------------------
# Memory: chat-style memory must return messages, not a single string
# ---------------------------------------------------------------------------
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True,          # <-- required by MessagesPlaceholder
    output_key="output",           # AgentExecutor stores its answer under "output"
    human_prefix="Human",
    ai_prefix="AI",
)

# ---------------------------------------------------------------------------
# Custom prompt that actually contains the conversation history
# ---------------------------------------------------------------------------
SYSTEM_TEMPLATE = """Respond to the human as helpfully and accurately as possible. You have access to the following tools:

{tools}

Use a json blob to specify a tool by providing an action key (tool name) and an action_input key (tool input).

Valid "action" values: "Final Answer" or {tool_names}

Provide only ONE action per $JSON_BLOB, as shown:

```
{{
  "action": $TOOL_NAME,
  "action_input": $INPUT
}}
```

Follow this format:

Question: the input question you must answer
Thought: you should always think about what to do
Action:
```
$JSON_BLOB
```
Observation: the result of the action
... (this Thought/Action/Observation can repeat N times)
Thought: I now know the final answer
Action:
```
{{
  "action": "Final Answer",
  "action_input": "Final response to human"
}}
```

The conversation history with the human so far is provided above this message.
Use it: if the human refers to something said earlier (e.g. "the last question I
asked", "that city", "my previous request"), answer from the history instead of
asking the human to repeat themselves.

Begin! Reminder to ALWAYS respond with a valid json blob of a single action. Use
tools if necessary. Respond directly if appropriate."""

HUMAN_TEMPLATE = """{input}

{agent_scratchpad}"""

tool_strings = "\n".join(f"> {tool.name}: {tool.description}" for tool in tools)
tool_names = ", ".join(tool.name for tool in tools)

prompt = ChatPromptTemplate.from_messages(
    [
        SystemMessagePromptTemplate.from_template(
            SYSTEM_TEMPLATE,
            partial_variables={
                "tools": tool_strings,
                "tool_names": tool_names,
            },
        ),
        # This is the missing piece: the history is injected as real chat messages.
        MessagesPlaceholder(variable_name="chat_history"),
        HumanMessagePromptTemplate.from_template(HUMAN_TEMPLATE),
    ]
)

agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,                       # memory now flows into the prompt
    agent_kwargs={"prompt": prompt},     # <-- give the agent a prompt with chat_history
    verbose=False,
)


def run_agent(prompt_text: str) -> str:
    return agent_test.run(input=prompt_text)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)

    result2 = run_agent("what is the last question I asked?")
    print(result2)  # -> now correctly answers: the capital of France

    # Sanity check: the history really is in memory.
    print("\n--- chat_history in memory ---")
    for message in memory.load_memory_variables({})["chat_history"]:
        print(f"{message.type}: {message.content}")
