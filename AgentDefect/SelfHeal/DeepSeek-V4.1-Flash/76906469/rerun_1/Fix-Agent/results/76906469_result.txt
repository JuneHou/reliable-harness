"""
FIXED CODE

Root cause of the output mismatch
---------------------------------
`ConversationBufferMemory(memory_key="chat_history")` defaults to
`return_messages=False`, i.e. it returns ONE string of concatenated history.

Chat-based agent prompt(s) -- including
`CHAT_ZERO_SHOT_REACT_DESCRIPTION` and `CHAT_CONVERSATIONAL_REACT_DESCRIPTION` --
declare `chat_history` as a `MessagesPlaceholder`, so they expect a *list of
BaseMessage objects*. When memory hands back a plain string, the history is
either silently dropped/mis-formatted, so the agent behaves as if it has no
memory at all: the second question ("what is the last question I asked?")
cannot be answered from the conversation, and the reply does not match what
the user expects.

Fix (required configuration for a chat agent to read its memory)
---------------------------------------------------------------
1. Create the memory with `return_messages=True` (and be explicit about
   `memory_key`/`input_key`/`output_key` so it lines up with the agent prompt
   and the AgentExecutor keys).
2. Use a chat/conversational agent so the memory is actually wired into the
   prompt as messages -- `CHAT_CONVERSATIONAL_REACT_DESCRIPTION` is the agent
   explicitly built for "chat + tools + memory" use.
3. Call the agent with an `input` dict (`invoke({"input": ...})`) so the memory
   can store/retrieve the turn under the expected `input` key.
"""

from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# KEY FIX: return_messages=True -> chat_history is a list of messages,
# which is what the chat agent's MessagesPlaceholder requires.
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True,
    input_key="input",
    output_key="output",
    human_prefix="Human",
    ai_prefix="AI",
)

agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    # Chat-oriented agent that supports both tools and conversation memory.
    # (CHAT_ZERO_SHOT_REACT_DESCRIPTION also works once return_messages=True,
    #  but this one is designed for conversational use.)
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False,
    return_intermediate_steps=False,
)


def run_agent(prompt: str) -> str:
    # Use the dict form so memory saves/loads the turn under `input`.
    return agent_test.invoke({"input": prompt})["output"]


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)

    # Now answered from the stored conversation history in memory.
    result2 = run_agent("what is the last question I asked?")
    print(result2)

    # Optional: inspect what has actually been stored in memory.
    print(memory.load_memory_variables({})["chat_history"])
