from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# FIX: CHAT_ZERO_SHOT_REACT_DESCRIPTION is a *chat* agent, so its prompt is built as
#   SystemMessage + MessagesPlaceholder("chat_history") + HumanMessage + agent_scratchpad.
# A MessagesPlaceholder only accepts a list of BaseMessage objects. With the default
# `return_messages=False`, ConversationBufferMemory renders the history into a plain
# string, so the chat prompt either rejects it or silently ignores it and the agent
# never actually receives the previous turn (hence "what is the last question I asked?"
# cannot be answered). Setting `return_messages=True` makes the memory return the
# message list the chat prompt expects, and the memory_key must match the prompt's
# placeholder name ("chat_history").
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True,
    input_key="input",
    output_key="output",
)

agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False
)


def run_agent(prompt: str) -> str:
    # `AgentExecutor.run` is deprecated in newer langchain versions; prefer invoke.
    if hasattr(agent_test, "invoke"):
        return agent_test.invoke({"input": prompt})["output"]
    return agent_test.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    result2 = run_agent("what is the last question I asked?")
    print(result2)
    # The second answer now correctly references the first question
    # ("what is the capital of France?"), because the chat history is passed
    # back into the chat prompt on every run.
