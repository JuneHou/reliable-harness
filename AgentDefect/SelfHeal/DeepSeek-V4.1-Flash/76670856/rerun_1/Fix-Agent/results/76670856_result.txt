Why the original snippet produced answers that did not match expectations
-------------------------------------------------------------------------
1. JSONLoader setup (the main culprit)
   `JSONLoader("data/review.json", jq_schema=".reviews[]", content_key="text")`
   was called with the *default* `text_content=True`.  In that mode `content_key`
   is effectively ignored and every Document's `page_content` becomes the JSON
   serialisation of the whole record, e.g.

       {"text": "Great hotel, excellent service and comfortable rooms."}

   instead of just the review text.  Those JSON strings are what got embedded and
   what got stuffed into the prompt, so the model answered about JSON blobs (and
   copied the quoting/brace formatting) instead of answering about the reviews.
   `text_content=False` must be passed *together with* `content_key` so the review
   text itself becomes `page_content`.

2. Retriever coverage
   `VectorstoreIndexCreator` builds the retriever with the default `k=4`, so only
   4 of the 10 reviews were ever handed to the LLM.  Counting / summarising /
   "are there any negative reviews?" style questions were therefore answered from
   an incomplete context.  The retriever below returns *all* review documents.

3. Prompt
   The generic default stuff prompt encourages the model to improvise.  An
   explicit prompt that says the context is a flat list of hotel reviews, forbids
   outside knowledge and requires "not available" when the reviews do not contain
   the answer keeps the output grounded and accurate.

The chain/retriever wiring is otherwise the same as before: a JSONLoader feeding a
Chroma vector store, a ConversationalRetrievalChain with an explicit chat history,
and one answer appended per query.
"""

from __future__ import annotations

import json
import os
import sys
from pathlib import Path
from typing import Iterable, List, Union

from dotenv import load_dotenv

from langchain.chains import ConversationalRetrievalChain
from langchain.docstore.document import Document
from langchain.document_loaders import JSONLoader
from langchain.prompts import PromptTemplate
from langchain.vectorstores import Chroma

try:  # langchain >= 0.1
    from langchain_openai import ChatOpenAI, OpenAIEmbeddings
except ImportError:  # older langchain releases
    from langchain.chat_models import ChatOpenAI
    from langchain.embeddings.openai import OpenAIEmbeddings

load_dotenv()

REVIEW_PAYLOAD = {
    "reviews": [
        {"text": "Great hotel, excellent service and comfortable rooms."},
        {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
        {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
        {"text": "Average hotel. The room was okay, but nothing special."},
        {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
        {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
        {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
        {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
        {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
        {"text": "Not a bad hotel, but there are better options available in the area."},
    ]
}

# Prompt used to answer the (already condensed) question from the retrieved reviews.
QA_PROMPT = PromptTemplate(
    template=(
        "You are a precise assistant answering questions about a dataset of hotel "
        "reviews.\n"
        "The context below contains retrieved review excerpts; each line/excerpt is "
        "ONE separate review.\n"
        "Use ONLY the information in the context. Do not use outside knowledge and do "
        "not invent reviews.\n"
        "If the context does not contain the answer, reply that the information is not "
        "available in the reviews.\n"
        "When the question asks for a count or a summary, reason over every review in "
        "the context and answer concisely in plain English (never echo raw JSON).\n\n"
        "Reviews:\n"
        "{context}\n\n"
        "Question: {question}\n"
        "Accurate answer:"
    ),
    input_variables=["context", "question"],
)


def _ensure_review_file(review_file: Path) -> None:
    """Create data/review.json with the sample reviews if it is missing."""
    review_file.parent.mkdir(parents=True, exist_ok=True)
    if not review_file.exists():
        review_file.write_text(json.dumps(REVIEW_PAYLOAD, indent=2))


def load_review_documents(review_file: Path) -> List[Document]:
    """Load one Document per review, with the review text as page_content.

    `jq_schema=".reviews[]"` yields every object inside the ``reviews`` array and
    ``content_key="text"`` + ``text_content=False`` tells JSONLoader to use the
    value of the ``text`` field as page_content (instead of the JSON dump of the
    whole object, which is what the default ``text_content=True`` does).
    """
    loader = JSONLoader(
        file_path=str(review_file),
        jq_schema=".reviews[]",
        content_key="text",
        text_content=False,
    )

    try:
        raw_documents = loader.load()
    except Exception:
        # Defensive fallback for langchain versions where content_key/text_content
        # behave differently: build the documents straight from the JSON.
        payload = json.loads(review_file.read_text())
        raw_documents = [
            Document(page_content=str(review.get("text", "")), metadata={})
            for review in payload.get("reviews", [])
        ]

    documents: List[Document] = []
    for index, document in enumerate(raw_documents):
        content = (document.page_content or "").strip()
        # Normalise anything that still looks like a serialised record.
        if content.startswith("{") and content.endswith("}"):
            try:
                parsed = json.loads(content)
                if isinstance(parsed, dict) and "text" in parsed:
                    content = str(parsed["text"]).strip()
            except (ValueError, TypeError):
                pass
        metadata = dict(document.metadata or {})
        metadata.setdefault("source", str(review_file))
        metadata["review_index"] = index
        documents.append(Document(page_content=content, metadata=metadata))
    return documents


def build_vectorstore(documents: List[Document], persist: bool) -> Chroma:
    """Embed the review documents (optionally persisting them to ./persist)."""
    if persist and os.path.exists("persist"):
        print("Reusing index...\n")
        return Chroma(
            persist_directory="persist",
            embedding_function=OpenAIEmbeddings(),
        )
    if persist:
        return Chroma.from_documents(
            documents=documents,
            embedding=OpenAIEmbeddings(),
            persist_directory="persist",
        )
    return Chroma.from_documents(documents=documents, embedding=OpenAIEmbeddings())


def build_chain(vectorstore: Chroma, document_count: int) -> ConversationalRetrievalChain:
    """Retrieve *all* reviews (k = number of documents) and answer with QA_PROMPT."""
    retriever = vectorstore.as_retriever(
        search_type="similarity",
        search_kwargs={"k": max(document_count, 1)},
    )
    llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)
    return ConversationalRetrievalChain.from_llm(
        llm=llm,
        retriever=retriever,
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
        return_source_documents=True,
    )


def _normalise_queries(queries: Union[str, Iterable[str], None]) -> List[str]:
    if isinstance(queries, str):
        return [queries]
    normalised = [str(query) for query in (queries or [])]
    if not normalised and len(sys.argv) > 1:
        normalised = sys.argv[1:]
    return normalised


def _ask(chain: ConversationalRetrievalChain, question: str, chat_history):
    payload = {"question": question, "chat_history": chat_history}
    invoke = getattr(chain, "invoke", None)
    if callable(invoke):
        return invoke(payload)
    return chain(payload)


def run_agent(queries):
    data_dir = Path("data")
    data_dir.mkdir(parents=True, exist_ok=True)

    review_file = data_dir / "review.json"
    _ensure_review_file(review_file)

    # Enable to save to disk & reuse the index (for repeated queries on the same data).
    # NOTE: if you change the loader/jq_schema, delete ./persist so stale documents
    # (with the old, JSON-dumped page_content) are not reused.
    PERSIST = False

    documents = load_review_documents(review_file)
    vectorstore = build_vectorstore(documents, PERSIST)
    chain = build_chain(vectorstore, len(documents))

    chat_history = []
    results = []
    for query in _normalise_queries(queries):
        query = query.strip()
        if not query:
            continue
        if query.lower() in ("quit", "q", "exit"):
            break
        result = _ask(chain, query, chat_history)
        answer = result["answer"]
        print(answer)
        results.append(answer)
        chat_history.append((query, answer))
    return results


if __name__ == "__main__":
    cli_queries = sys.argv[1:]
    if not cli_queries:
        print("Enter a question about the reviews (blank line to finish):")
        while True:
            try:
                line = input("> ").strip()
            except EOFError:
                break
            if not line:
                break
            cli_queries.append(line)
    if cli_queries:
        run_agent(cli_queries)
