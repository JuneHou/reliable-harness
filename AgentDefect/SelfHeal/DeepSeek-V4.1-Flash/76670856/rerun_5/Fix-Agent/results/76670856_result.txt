def ensure_review_file(data_dir: Path = Path("data")) -> Path:
    """Create data/review.json from the sample reviews if it does not exist."""

    review_file = data_dir / "review.json"

def _documents_from_json(review_file: Path) -> List[Document]:
    """Pure-python fallback: one Document per review, guaranteed text content."""
    raw = json.loads(review_file.read_text(encoding="utf-8"))
    reviews = raw.get("reviews", []) if isinstance(raw, dict) else raw
    documents: List[Document] = []
    for index, record in enumerate(reviews):
        if isinstance(record, dict):
            text = record.get("text") or record.get("content") or ""
        else:
            text = str(record)
        text = (text or "").strip()
        if text:
            documents.append(
                Document(
                    page_content=text,
                    metadata={"source": str(review_file), "review_index": index},
                )
            )
    return documents


def load_review_documents(review_file: Path) -> List[Document]:
    """Load reviews with JSONLoader/jq_schema, validating the extracted text.

    The canonical setup for the payload ``{"reviews": [{"text": ...}, ...]}`` is
    ``jq_schema=".reviews[]"`` (one object per Document) plus
    ``content_key="text"`` (which field becomes ``page_content``).  Because that
    combination is ignored/empty on some LangChain versions, two safe variants
    are tried and a plain-python parse is used as the last resort.
    """
    candidates = [
        # (jq_schema, content_key) pairs, most idiomatic first
        (".reviews[]", "text"),      # select each object, pick its "text" field
        (".reviews[].text", None),   # select the strings directly, no content_key
    ]

    for jq_schema, content_key in candidates:
        kwargs = {"jq_schema": jq_schema}
        if content_key is not None:
            kwargs["content_key"] = content_key
        try:
            loader = JSONLoader(str(review_file), **kwargs)
        except TypeError:
            # very old signature: JSONLoader(file_path, jq_schema=..., content_key=...)
            try:
                loader = JSONLoader(
                    file_path=str(review_file), jq_schema=jq_schema,
                    content_key=content_key,
                )
            except Exception:
                continue
        except Exception:
            continue

        try:
            documents = loader.load()
        except Exception:
            continue

        documents = [doc for doc in documents if (doc.page_content or "").strip()]
        # Reject the "whole record dumped as text" case as well.
        documents = [
            doc for doc in documents
            if not doc.page_content.strip().startswith("{")
        ]
        if documents:
            return documents

    return _documents_from_json(review_file)

def build_or_reuse_vectorstore(embedding_function):
    """Return (vectorstore, number_of_indexed_documents)."""
    if PERSIST and os.path.exists(PERSIST_DIR):
        print("Reusing index...\n")
        vectorstore = Chroma(
            persist_directory=PERSIST_DIR, embedding_function=embedding_function
        )
        try:
            total = vectorstore._collection.count()
        except Exception:
            total = 10
        return vectorstore, total

    review_file = ensure_review_file()
    documents = load_review_documents(review_file)
    if not documents:
        raise RuntimeError(

            "Check the JSON structure and the jq_schema."

    print(

        file=sys.stderr,

    vectorstore = Chroma.from_documents(
        documents=documents,
        embedding=embedding_function,
        persist_directory=persist_dir,

    if PERSIST:

def _as_query_list(queries: Iterable[str] | str) -> Sequence[str]:
    if isinstance(queries, str):
        return [queries]
    return list(queries)


def run_agent(queries):
    queries = _as_query_list(queries)

    embedding_function = OpenAIEmbeddings()
    vectorstore, indexed_docs = build_or_reuse_vectorstore(embedding_function)

    # Retrieve *every* review of this small corpus so counting/summarising
    # questions are answered from the full data instead of 4 random chunks.
    retriever_k = max(1, min(indexed_docs, 50))
    retriever = vectorstore.as_retriever(search_kwargs={"k": retriever_k})

    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model="gpt-3.5-turbo", temperature=0),
        retriever=retriever,
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
        condense_question_prompt=CONDENSE_QUESTION_PROMPT,
        return_source_documents=True,
        verbose=False,
    )

    chat_history: List[tuple] = []
    results: List[str] = []

    for query in queries:
        if not query or not query.strip():
            continue
        if query.strip().lower() in {"quit", "q", "exit"}:
            break

        payload = {"question": query, "chat_history": chat_history}
        try:  # LangChain >= 0.1
            result = chain.invoke(payload)
        except AttributeError:  # older LangChain
            result = chain(payload)

        answer = result["answer"]
        print(answer)
        results.append(answer)

        if os.getenv("DEBUG"):
            print("--- retrieved ---", file=sys.stderr)
            for doc in result.get("source_documents", []):
                print(f"  * {doc.page_content[:120]}", file=sys.stderr)

        # keep the (question, answer) pair so follow-up questions stay grounded
        chat_history.append((query, answer))

    return results
