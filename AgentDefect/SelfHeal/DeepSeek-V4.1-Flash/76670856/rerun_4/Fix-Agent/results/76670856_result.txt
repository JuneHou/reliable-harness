"""QA over JSON review data.

Diagnosis of the original script
--------------------------------
1. ``JSONLoader("data/review.json", jq_schema=".reviews[]", content_key="text")``
   is *almost* right, but it is fragile: depending on the LangChain version the
   ``content_key`` is only applied to the *first* element of the jq result, so
   every ``Document`` can end up carrying the same / wrong text (or the whole
   review dict as a JSON string).  The jq filter must be applied element-wise
   (``.reviews[]``) and the resulting documents must be validated/back-filled
   before they are embedded.  If the loader silently returns unusable content
   the vector store never contains the real reviews, which is exactly why the
   answers look "hallucinated".

2. Even with a correct loader, ``index.vectorstore.as_retriever()`` uses the
   LangChain default ``k=4``.  Only 4 of the 10 reviews are ever sent to the
   LLM, so any question that needs the whole corpus (counts, overall sentiment,
   "which review is the worst"...) is answered from an incomplete context and
   is therefore wrong.

3. ``ConversationalRetrievalChain``'s default prompt does not tell the model
   that the context is a set of customer reviews, nor that it must aggregate
   over all of them, so answers drift and get invented.

Fixes below: element-wise jq loading + validation/fallback, retrieval over the
whole corpus (k = number of reviews), temperature 0, an explicit QA prompt that
forces the model to use the supplied reviews, and source documents returned for
verification.
"""

from dotenv import load_dotenv
from pathlib import Path
from typing import Dict, List

import json
import os
import sys

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.document_loaders import JSONLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.prompts import PromptTemplate
from langchain.schema import Document
from langchain.vectorstores import Chroma

load_dotenv()

PERSIST = False
PERSIST_DIR = "persist"

# Prompt that makes the expected behaviour explicit: the context contains the
# customer reviews, all of them must be considered, and nothing may be invented.
QA_PROMPT = PromptTemplate(
    template="""You are an assistant that answers questions about a dataset of hotel
customer reviews.

The context below contains the customer reviews retrieved for this question
(usually every review in the dataset).
Read ALL of the reviews before answering and reason over them explicitly when the
question is about the dataset as a whole (counts, overall sentiment, best/worst
review, recurring complaints, ...).
Answer only with information that is present in the reviews. If the reviews do
not contain the answer, reply exactly: "I don't know".

Context:
{context}

Question: {question}

Accurate answer:""",
    input_variables=["context", "question"],
)


def _metadata_func(record: Dict, metadata: Dict) -> Dict:
    """Keep the review fields as metadata (values must be Chroma-friendly)."""
    metadata = dict(metadata or {})
    for key, value in (record or {}).items():
        if key == "text":
            continue
        if isinstance(value, (str, int, float, bool)) or value is None:
            metadata[key] = value
        else:
            metadata[key] = json.dumps(value, ensure_ascii=False)
    metadata.setdefault("source", "review.json")
    return metadata


def _load_review_documents(review_file: Path) -> List[Document]:
    """Load one Document per review, with the review text as page_content."""
    documents: List[Document] = []

    # Preferred path: jq filter over the reviews array, one document per entry.
    try:
        loader = JSONLoader(
            file_path=str(review_file),
            jq_schema=".reviews[]",
            content_key="text",
            metadata_func=_metadata_func,
        )
        documents = [
            doc for doc in loader.load()
            if doc.page_content and doc.page_content.strip()
        ]
    except Exception as exc:  # content_key unsupported / jq mismatch
        print(f"JSONLoader could not parse {review_file} ({exc}); "
              "falling back to direct JSON parsing.")
        documents = []

    if documents:
        for index, doc in enumerate(documents):
            doc.metadata.setdefault("source", str(review_file))
            doc.metadata.setdefault("review_index", index)
        return documents

    # Fallback: build the documents straight from the JSON payload so that the
    # reviews are never silently dropped from the index.
    payload = json.loads(review_file.read_text(encoding="utf-8"))
    records = payload.get("reviews", []) if isinstance(payload, dict) else payload
    if isinstance(records, dict):
        records = [records]

    for index, record in enumerate(records):
        if isinstance(record, dict):
            text = record.get("text") or json.dumps(record, ensure_ascii=False)
            metadata = {
                key: (
                    value
                    if isinstance(value, (str, int, float, bool)) or value is None
                    else json.dumps(value, ensure_ascii=False)
                )
                for key, value in record.items()
                if key != "text"
            }
        else:
            text = str(record)
            metadata = {}
        metadata["source"] = str(review_file)
        metadata["review_index"] = index
        documents.append(Document(page_content=text.strip(), metadata=metadata))

    return documents


def _create_payload(review_file: Path) -> None:
    payload = {
        "reviews": [
            {"text": "Great hotel, excellent service and comfortable rooms."},
            {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
            {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
            {"text": "Average hotel. The room was okay, but nothing special."},
            {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
            {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
            {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
            {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
            {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
            {"text": "Not a bad hotel, but there are better options available in the area."}
        ]
    }
    review_file.write_text(json.dumps(payload, indent=2), encoding="utf-8")


def run_agent(queries):
    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)

    review_file = data_dir / "review.json"
    if not review_file.exists():
        _create_payload(review_file)

    documents = _load_review_documents(review_file)
    if not documents:
        raise ValueError(f"No reviews could be loaded from {review_file}")

    embeddings = OpenAIEmbeddings(model="text-embedding-ada-002")

    if PERSIST and os.path.exists(PERSIST_DIR):
        print("Reusing index...\n")
        vectorstore = Chroma(
            persist_directory=PERSIST_DIR,
            embedding_function=embeddings,
        )
        try:
            num_documents = vectorstore._collection.count()
        except Exception:
            num_documents = len(documents)
    else:
        vectorstore = Chroma.from_documents(
            documents=documents,
            embedding=embeddings,
            persist_directory=PERSIST_DIR if PERSIST else None,
        )
        if PERSIST:
            vectorstore.persist()
        num_documents = len(documents)

    # KEY FIX for accuracy: send *all* reviews to the LLM instead of the
    # default k=4, otherwise whole-corpus questions are answered from a
    # random subset of the data.
    retriever = vectorstore.as_retriever(
        search_type="similarity",
        search_kwargs={"k": max(num_documents, 4)},
    )

    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model="gpt-3.5-turbo", temperature=0),
        retriever=retriever,
        return_source_documents=True,
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
    )

    chat_history = []
    results = []
    for query in queries:
        if query in ["quit", "q", "exit"]:
            break
        result = chain({"question": query, "chat_history": chat_history})
        answer = result["answer"]
        print(answer)
        results.append(answer)
        chat_history.append((query, answer))
    return results


if __name__ == "__main__":
    cli_queries = sys.argv[1:] or ["What do the reviews say about this hotel?"]
    run_agent(cli_queries)
