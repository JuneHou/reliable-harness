def _api_key():

def configure_settings(api_key, model=MODEL, temperature=TEMPERATURE):
    """Chunking / LLM settings (PromptHelper is no longer needed)."""

    Settings.chunk_overlap = CHUNK_OVERLAP


def create_vector_index(path, api_key, persist_dir=PERSIST_DIR):
    """Load the persisted index if present, otherwise build it from docs/."""
    cache_key = (os.path.abspath(path), persist_dir)
    if cache_key in _INDEX_CACHE:
        return _INDEX_CACHE[cache_key]

    if not os.path.isdir(path) or not os.listdir(path):
        raise RuntimeError(
            "No reference documents found in '{}'. Put the example NDA files there "
            "(.pdf/.docx/.txt/.md) before running.".format(path)
        )

    if os.path.isdir(persist_dir) and os.listdir(persist_dir):
        storage_context = StorageContext.from_defaults(persist_dir=persist_dir)
        vector_index = load_index_from_storage(storage_context)
    else:
        docs = SimpleDirectoryReader(path).load_data()
        vector_index = VectorStoreIndex.from_documents(docs)
        vector_index.storage_context.persist(persist_dir=persist_dir)

    _INDEX_CACHE[cache_key] = vector_index
    return vector_index


def infer_mode(user_input):
    """Route drafting requests to the drafting prompt, questions to the QA prompt."""
    text = (user_input or "").strip().lower()
    drafting_markers = (
        "draft", "write", "generate", "create", "prepare", "compose",
        "produce a", "new nda", "new document", "template", "based on",
    )
    if any(marker in text for marker in drafting_markers):
        return "draft"
    question_markers = (
        "?", "what ", "who ", "whom ", "when ", "where ", "which ", "how ",
        "summar", "explain", "list ", "is there", "does the",
    )
    if any(marker in text for marker in question_markers):
        return "qa"
    return "draft"  # this app is a drafter first


def _looks_like_refusal(text):
    if not text or not text.strip():
        return True
    head = text.strip().lower()[:400]
    return any(marker in head for marker in REFUSAL_MARKERS)


def _build_query_engine(vector_index, mode, temperature=None, model=None):
    system, user_template = (
        (DRAFT_SYSTEM_PROMPT, DRAFT_USER_TEMPLATE)
        if mode == "draft"
        else (QA_SYSTEM_PROMPT, QA_USER_TEMPLATE)
    )
    flat_template = DRAFT_FLAT_TEMPLATE if mode == "draft" else QA_FLAT_TEMPLATE

    kwargs = {
        "similarity_top_k": SIMILARITY_TOP_K,
        "response_mode": "compact",
    }
    if temperature is not None or model is not None:
        kwargs["llm"] = OpenAI(
            api_key=os.getenv("OPENAI_API_KEY"),
            temperature=TEMPERATURE if temperature is None else temperature,
            model=MODEL if model is None else model,
            max_tokens=MAX_TOKENS,
        )

    if _HAS_CHAT_PROMPT:
        try:
            prompt = ChatPromptTemplate(
                system_prompt=system,
                message_templates=[
                    ChatMessage(role=MessageRole.USER, content=user_template)
                ],
            )
            kwargs["text_qa_template"] = prompt
            return vector_index.as_query_engine(**kwargs)
        except Exception:
            pass  # prompt/model combination not supported -> flat fallback

    kwargs["text_qa_template"] = flat_template
    return vector_index.as_query_engine(**kwargs)

def run_agent(user_input, docs_path="docs", mode=None):
    """Answer `user_input` using the example NDAs as context.

    Drafting requests produce a new document draft instead of a refusal.
    """
    api_key = _api_key()
    configure_settings(api_key)
    vector_index = create_vector_index(docs_path, api_key)

    if not user_input or not user_input.strip():
        return "Please describe the document you would like me to draft."

    mode = mode or infer_mode(user_input)

    # 1st attempt: role-correct drafting / QA prompt.
    engine = _build_query_engine(vector_index, mode)
    answer = engine.query(user_input).response

    # 2nd attempt: flat prompt with more freedom (different sampling).
    if _looks_like_refusal(answer):
        engine = _build_query_engine(vector_index, mode, temperature=0.7)
        answer = engine.query(user_input).response

    # 3rd attempt: flat prompt + stronger model if available.
    if _looks_like_refusal(answer):
        try:
            engine = _build_query_engine(
                vector_index, mode, temperature=0.4, model=FALLBACK_MODEL
            )
            answer = engine.query(user_input).response
        except Exception:
            pass

    answer = (answer or "").strip()
    if not answer:
        return (
            "The model returned an empty response. Please rephrase the request, e.g. "
            "'Draft a mutual NDA between Acme Corp (disclosing) and Beta LLC "
            "(receiving), based on the example documents.'"
        )
    return answer


def build_ui(docs_path="docs"):
    """Optional Gradio chat front-end."""
    import gradio as gr

    def chat(message, history):
        return run_agent(message, docs_path=docs_path)

    return gr.ChatInterface(
        fn=chat,
        title="NDA Drafter",
        description=(
            "Ask a question about the example NDAs, or ask for a brand-new draft "
            "(e.g. 'Draft a mutual NDA for Acme Corp and Beta LLC')."
        ),
    )

def create_vector_index(path, api_key, persist_dir=PERSIST_DIR):
    """Load the persisted index if present, otherwise build it from docs/."""
    cache_key = (os.path.abspath(path), persist_dir)
    if cache_key in _INDEX_CACHE:
        return _INDEX_CACHE[cache_key]

    if not os.path.isdir(path) or not os.listdir(path):
        raise RuntimeError(
            "No reference documents found in '{}'. Put the example NDA files there "
            "(.pdf/.docx/.txt/.md) before running.".format(path)
        )

    if os.path.isdir(persist_dir) and os.listdir(persist_dir):
        storage_context = StorageContext.from_defaults(persist_dir=persist_dir)
        vector_index = load_index_from_storage(storage_context)
    else:
        docs = SimpleDirectoryReader(path).load_data()
        vector_index = VectorStoreIndex.from_documents(docs)
        vector_index.storage_context.persist(persist_dir=persist_dir)

    _INDEX_CACHE[cache_key] = vector_index
    return vector_index


def infer_mode(user_input):
    """Route drafting requests to the drafting prompt, questions to the QA prompt."""
    text = (user_input or "").strip().lower()
    drafting_markers = (
        "draft", "write", "generate", "create", "prepare", "compose",
        "produce a", "new nda", "new document", "template", "based on",
    )
    if any(marker in text for marker in drafting_markers):
        return "draft"
    question_markers = (
        "?", "what ", "who ", "whom ", "when ", "where ", "which ", "how ",
        "summar", "explain", "list ", "is there", "does the",
    )
    if any(marker in text for marker in question_markers):
        return "qa"
    return "draft"  # this app is a drafter first


def _looks_like_refusal(text):
    if not text or not text.strip():
        return True
    head = text.strip().lower()[:400]
    return any(marker in head for marker in REFUSAL_MARKERS)


def _build_query_engine(vector_index, mode, temperature=None, model=None):
    system, user_template = (
        (DRAFT_SYSTEM_PROMPT, DRAFT_USER_TEMPLATE)
        if mode == "draft"
        else (QA_SYSTEM_PROMPT, QA_USER_TEMPLATE)
    )
    flat_template = DRAFT_FLAT_TEMPLATE if mode == "draft" else QA_FLAT_TEMPLATE

    kwargs = {
        "similarity_top_k": SIMILARITY_TOP_K,
        "response_mode": "compact",
    }
    if temperature is not None or model is not None:
        kwargs["llm"] = OpenAI(
            api_key=os.getenv("OPENAI_API_KEY"),
            temperature=TEMPERATURE if temperature is None else temperature,
            model=MODEL if model is None else model,
            max_tokens=MAX_TOKENS,
        )

    if _HAS_CHAT_PROMPT:
        try:
            prompt = ChatPromptTemplate(
                system_prompt=system,
                message_templates=[
                    ChatMessage(role=MessageRole.USER, content=user_template)
                ],
            )
            kwargs["text_qa_template"] = prompt
            return vector_index.as_query_engine(**kwargs)
        except Exception:
            pass  # prompt/model combination not supported -> flat fallback

    kwargs["text_qa_template"] = flat_template
    return vector_index.as_query_engine(**kwargs)

def run_agent(user_input, docs_path="docs", mode=None):
    """Answer `user_input` using the example NDAs as context.

    Drafting requests produce a new document draft instead of a refusal.
    """
    api_key = _api_key()
    configure_settings(api_key)
    vector_index = create_vector_index(docs_path, api_key)

    if not user_input or not user_input.strip():
        return "Please describe the document you would like me to draft."

    mode = mode or infer_mode(user_input)

    # 1st attempt: role-correct drafting / QA prompt.
    engine = _build_query_engine(vector_index, mode)
    answer = engine.query(user_input).response

    # 2nd attempt: flat prompt with more freedom (different sampling).
    if _looks_like_refusal(answer):
        engine = _build_query_engine(vector_index, mode, temperature=0.7)
        answer = engine.query(user_input).response

    # 3rd attempt: flat prompt + stronger model if available.
    if _looks_like_refusal(answer):
        try:
            engine = _build_query_engine(
                vector_index, mode, temperature=0.4, model=FALLBACK_MODEL
            )
            answer = engine.query(user_input).response
        except Exception:
            pass

    answer = (answer or "").strip()
    if not answer:
        return (
            "The model returned an empty response. Please rephrase the request, e.g. "
            "'Draft a mutual NDA between Acme Corp (disclosing) and Beta LLC "
            "(receiving), based on the example documents.'"
        )
    return answer


def build_ui(docs_path="docs"):
    """Optional Gradio chat front-end."""
    import gradio as gr

    def chat(message, history):
        return run_agent(message, docs_path=docs_path)

    return gr.ChatInterface(
        fn=chat,
        title="NDA Drafter",
        description=(
            "Ask a question about the example NDAs, or ask for a brand-new draft "
            "(e.g. 'Draft a mutual NDA for Acme Corp and Beta LLC')."
        ),
    )
