import os

from dotenv import load_dotenv, find_dotenv

from llama_index.core import (
    SimpleDirectoryReader,
    VectorStoreIndex,
    StorageContext,
    load_index_from_storage,
    Settings,
)
from llama_index.core.prompts import PromptTemplate
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())

# Keep the (empty) CA bundle override from the original script if that is what
# the local environment needs.  It must be set *before* any HTTPS request is
# made, i.e. before the index / LLM is created.
os.environ.setdefault("REQUESTS_CA_BUNDLE", "")


# ---------------------------------------------------------------------------
# Prompt configuration -- this is the actual bug fix.
#
# The default LlamaIndex question-answering template tells the model:
#   "You are an expert Q&A system ... Do not try to fabricate an answer ...
#    If you don't know, say you don't know."
# and it also forwards the model's own safety layer, which sees an NDA
# ("legal document") request and answers "I'm sorry, I can't provide legal
# advice / I cannot draft a binding agreement".
#
# Because the task is *document drafting from examples* and not *question
# answering*, we must supply explicit, correctly formatted draft prompts that
#   * use the required placeholders ({context_str} / {query_str} for the QA
#     template and {context_msg} / {existing_answer} / {query_str} for the
#     refine template), and
#   * instruct the model to write a complete draft in the style of the
#     examples, and explicitly forbid refusals / clarifying questions.
# ---------------------------------------------------------------------------

DRAFT_QA_TEMPLATE = (
    "We have provided several example NDA documents and a drafting request below.\n"
    "---------------------\n"
    "{context_str}\n"
    "---------------------\n"
    "You are a legal document drafting assistant. Using only the structure, clause "
    "language, defined terms and tone of the example documents above, write a "
    "complete Non-Disclosure Agreement that fulfils the drafting request.\n"
    "Rules:\n"
    "1. Never refuse, never ask the user for more information, never say that you "
    "are not a lawyer and never mention that you were given example documents.\n"
    "2. Where the request is silent, follow the examples' default wording and mark "
    "unspecified placeholders such as [PARTY NAME] or [DATE] in square brackets.\n"
    "3. Output only the drafted agreement as plain text, then one short closing "
    "line stating that it is a draft for review by a qualified attorney.\n"
    "Drafting request: {query_str}\n"
    "Draft:\n"
)

DRAFT_REFINE_TEMPLATE = (
    "You are a legal document drafting assistant improving an existing draft NDA "
    "with additional example material.\n"
    "---------------------\n"
    "{context_msg}\n"
    "---------------------\n"
    "Existing draft:\n"
    "{existing_answer}\n"
    "Drafting request: {query_str}\n"
    "If the additional material contains clauses, definitions or structure that the "
    "existing draft is missing, rewrite the draft so that it includes them and "
    "keeps the wording style of the examples. Otherwise return the existing draft "
    "unchanged. Never refuse and never ask questions. Return only the "
    "(possibly improved) draft text:\n"
)


def _build_index(path="docs", persist_dir="vectorIndex.json"):
    """Build the vector index from the example documents (or reuse a cached one)."""

    Settings.llm = OpenAI(temperature=0.1, model="gpt-3.5-turbo", max_tokens=512)
    Settings.chunk_size = 600
    Settings.chunk_overlap = int(600 * 0.1)

    if os.path.isdir(persist_dir) and os.listdir(persist_dir):
        storage_context = StorageContext.from_defaults(persist_dir=persist_dir)
        return load_index_from_storage(storage_context)

    docs = SimpleDirectoryReader(path).load_data()
    index = VectorStoreIndex.from_documents(docs)
    index.storage_context.persist(persist_dir=persist_dir)
    return index


def run_agent(user_input, docs_path="docs", persist_dir="vectorIndex.json"):
    """Draft a new document from the example documents in ``docs_path``."""

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise RuntimeError(
            "OPENAI_API_KEY is not set. Add it to your environment or .env file."
        )
    os.environ["OPENAI_API_KEY"] = openai_api_key

    vector_index = _build_index(docs_path, persist_dir)

    query_engine = vector_index.as_query_engine(
        similarity_top_k=3,
        text_qa_template=DRAFT_QA_TEMPLATE,
        refine_template=DRAFT_REFINE_TEMPLATE,
    )

    response = query_engine.query(user_input)
    return str(response)


if __name__ == "__main__":
    # A drafting request matches the intended use case: the examples are
    # context for *producing* a document, not material to be summarised.
    user_input = (
        "Draft a mutual non-disclosure agreement between Acme Corp. and Beta LLC "
        "for a three year term covering the evaluation of a jointly developed "
        "software product. Follow the structure, clause order and tone of the "
        "example NDAs."
    )
    result = run_agent(user_input)
    print("\n=== Draft Result ===")
    print(f"Request: {user_input}")
    print(f"Draft:\n{result}")
