import os

from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr

from llama_index.core import (
    SimpleDirectoryReader,
    VectorStoreIndex,
    Settings,
    PromptTemplate,
)
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())

os.environ['REQUESTS_CA_BUNDLE'] = ''

# ---------------------------------------------------------------------------
# The default llama_index QA prompt is a *question answering* prompt:
#   "Context information is below ... answer the question ... if you don't
#    know, say you don't know ... you are not a lawyer ..."
# When the user asks the bot to *draft a new NDA* from the example documents,
# that prompt makes the model (a) treat the request as unanswerable / out of
# scope and (b) fall back to a refusal ("I'm sorry, but I can't ...").
#
# Fix: replace the prompt template with an explicit *drafting* prompt that
#  - tells the model the context is a template/style reference,
#  - states exactly what output is expected (the full document text),
#  - and forbids the refusal / meta-commentary behaviour.
# ---------------------------------------------------------------------------
DRAFT_QA_TEMPLATE = (
    "You are a legal drafting assistant for the business team.\n"
    "The excerpts below come from the company's own example documents. They are "
    "provided as a template and style reference so that you can imitate their "
    "structure, clause ordering, wording and tone.\n"
    "---------------------\n"
    "{context_str}\n"
    "---------------------\n"
    "Write a complete NEW document that fulfils the request below, using the format "
    "and language of the examples above. Output the actual document text only: "
    "title, parties, numbered clauses and signature block.\n"
    "Rules:\n"
    "1. Do not summarise the examples and do not explain your reasoning.\n"
    "2. Do not refuse, do not ask clarifying questions first, do not add "
    "disclaimers (no 'I am not a lawyer', no 'consult an attorney').\n"
    "3. If a detail is missing, insert a clearly marked placeholder such as "
    "[PARTY B ADDRESS] instead of refusing.\n"
    "Request: {query_str}\n"
    "Draft:\n"
)

DRAFT_REFINE_TEMPLATE = (
    "You are writing a new document based on example documents.\n"
    "Existing draft:\n"
    "---------------------\n"
    "{existing_answer}\n"
    "---------------------\n"
    "Additional example excerpts:\n"
    "---------------------\n"
    "{context_msg}\n"
    "---------------------\n"
    "Refine the draft so that it fully satisfies the request while keeping the "
    "structure and style of the examples. Output only the complete refined "
    "document, with no commentary. If the new excerpts add nothing useful, return "
    "the existing draft unchanged.\n"
    "Request: {query_str}\n"
    "Refined draft:\n"
)


def run_agent(user_input, docs_path='docs'):
    """Build (or load) an index over the example NDAs and draft a document."""

    openai_api_key = os.getenv('OPENAI_API_KEY')
    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key

    def createVectorIndex(path):
        tokens = 512
        chunk_size = 600
        max_chunk_overlap = 0.1

        Settings.llm = OpenAI(temperature=0.3, model='gpt-3.5-turbo', max_tokens=tokens)
        Settings.chunk_size = chunk_size
        Settings.chunk_overlap = int(chunk_size * max_chunk_overlap)

        # load data (the example NDA documents used as context)
        docs = SimpleDirectoryReader(path).load_data()
        # create vector index
        vectorIndex = VectorStoreIndex.from_documents(docs)
        vectorIndex.storage_context.persist(persist_dir='vector_index')
        return vectorIndex

    vectorIndex = createVectorIndex(docs_path)

    draft_prompt = PromptTemplate(DRAFT_QA_TEMPLATE)
    refine_prompt = PromptTemplate(DRAFT_REFINE_TEMPLATE)

    def chatbot(input_index):
        query_engine = vectorIndex.as_query_engine(
            similarity_top_k=3,
            text_qa_template=draft_prompt,
            refine_template=refine_prompt,
        )
        response = query_engine.query(input_index)
        return str(response.response)

    result = chatbot(user_input)

    return result


if __name__ == "__main__":
    user_input = (
        "Draft a mutual non-disclosure agreement between our company and "
        "'Acme Robotics Inc.' for a potential joint product evaluation. "
        "Effective on the date of signature, term of 3 years, governing law "
        "Delaware, and include the standard confidentiality, exclusions, "
        "return-of-materials and no-license clauses from the examples."
    )
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result}")

    if os.getenv("LAUNCH_UI") == "1":
        demo = gr.Interface(
            fn=lambda text: run_agent(text),
            inputs=gr.Textbox(lines=5, label="Describe the NDA you need"),
            outputs=gr.Textbox(lines=25, label="Draft"),
            title="NDA Draft Generator",
        )
        demo.launch()
