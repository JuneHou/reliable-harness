from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()


def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    # The old template only contained a single, vague instruction
    # ("Answer the question based on the following context"), which made
    # gpt-3.5-turbo reply with short, fragmentary snippets. ChatOpenAI also
    # received no system message, so there was nothing telling it to behave
    # like a conversational chatbot. Using explicit chat messages (a system
    # role + a human role) with clear instructions produces the full-sentence,
    # conversational answers llama2 was giving.
    system_template = (
        "You are a helpful, friendly chatbot assistant answering questions "
        "about movies and their budgets.\n"
        "Use ONLY the retrieved context below to answer the user's question.\n"
        "Always answer in complete, well-formed sentences in a natural, "
        "conversational tone (like a human chatting with the user), not as "
        "isolated fragments or bullet points.\n"
        "If the context does not contain the answer, say so politely in a "
        "full sentence instead of guessing.\n\n"
        "Context:\n{context}"
    )

    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", system_template),
            ("human", "{question}"),
        ]
    )

    chain = (
        {
            "context": retriever,
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)

    return result


if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
