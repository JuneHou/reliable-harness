from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    # ChatOpenAI is a CHAT model: without an explicit system instruction it behaves like a
    # "question answering over documents" extractor and replies with terse fragments
    # (e.g. "Inception" or "The highest budget movie is Inception.").
    # llama2 is a plain completion model and therefore tends to answer conversationally.
    # To get the llama2 style output we must (a) use proper chat roles (system + human) and
    # (b) explicitly instruct the model to reply in full, conversational sentences.
    SYSTEM_TEMPLATE = (
        "You are a friendly, conversational AI assistant in a chat application. "
        "Answer the user's question using only the information provided in the context. "
        "Always respond the way a helpful chatbot would in a normal conversation: write "
        "complete, grammatically correct, full-sentence answers that restate the question "
        "and explain the answer naturally. Never reply with only a bare value, a single "
        "word, a fragment, or a bullet list of keywords. If the context does not contain "
        "the answer, politely say in a full sentence that you do not know."
    )

    ANSWER_TEMPLATE = """Context:
{context}

Question: {question}"""

    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", SYSTEM_TEMPLATE),
            ("human", ANSWER_TEMPLATE),
        ]
    )

    chain = (
        {
            "context": retriever,
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)

    return result

if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
