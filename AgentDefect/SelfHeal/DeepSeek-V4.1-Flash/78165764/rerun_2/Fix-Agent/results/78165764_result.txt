from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    # The prompt is the only thing that controls *how* the model answers.
    # A bare "{context} / Question: {question}" template gives the chat model
    # no role and no formatting instructions, so gpt-3.5-turbo tends to reply
    # with a bare fragment (e.g. just the movie title) instead of a
    # conversational, full-sentence answer. Adding an explicit system message
    # plus answer-formatting instructions makes ChatOpenAI behave like llama2
    # (chatty, complete sentences) without touching the rest of the pipeline.
    SYSTEM_TEMPLATE = (
        "You are a helpful, friendly conversational assistant in a RAG chatbot. "
        "You behave exactly like llama2 would: you always reply in complete, "
        "natural-sounding sentences and in a warm, conversational tone. "
        "Never answer with a bare fragment, a single value, a keyword, or a "
        "raw data dump. Ground your answer in the provided context, but explain "
        "it in your own words, as a chatbot talking to a person would."
    )

    ANSWER_TEMPLATE = """Answer the question using only the following context.

Context:
{context}

Question: {question}

Write the answer as a short conversational paragraph of complete sentences.
Mention the relevant detail(s) from the context explicitly (for example, name
the item and the value you found) and explain it in a sentence or two.
If the context does not contain the answer, say so politely in a full
sentence instead of guessing."""

    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", SYSTEM_TEMPLATE),
            ("human", ANSWER_TEMPLATE),
        ]
    )

    chain = (
        {
            "context": retriever,
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)

    return result

if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
