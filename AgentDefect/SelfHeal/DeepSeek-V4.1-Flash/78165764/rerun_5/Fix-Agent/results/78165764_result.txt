from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    # ChatOpenAI is a *chat* model, so it expects a conversation made of messages
    # with explicit roles. A single terse "Answer the question based on the
    # following context" template nudges the model to emit short, fragmentary
    # extractive answers. Giving it a proper system message that defines the
    # assistant persona and asks for complete, conversational sentences makes
    # it behave like a chatty llama2-style model.
    SYSTEM_TEMPLATE = """You are a helpful, friendly chatbot assistant.

Use the following pieces of context to answer the user's question. If the answer
cannot be found in the context, say that you don't know instead of making
something up.

Always respond the way a helpful chatbot would: with complete, well-formed,
natural sentences that directly answer the question. Never reply with bare
keywords or sentence fragments.

Context:
{context}"""

    HUMAN_TEMPLATE = "{question}"

    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", SYSTEM_TEMPLATE),
            ("human", HUMAN_TEMPLATE),
        ]
    )

    chain = (
        {
            "context": retriever,
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)

    return result

if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
