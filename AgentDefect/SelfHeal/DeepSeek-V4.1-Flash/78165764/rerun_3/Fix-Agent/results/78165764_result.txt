from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()


def format_docs(docs):
    """Turn the retriever's Document objects into clean plain text.

    Passing raw Document objects into the prompt makes the model see
    `Document(page_content='...', metadata={...})` noise, which is one of the
    reasons ChatOpenAI replies with short, fragmented snippets instead of a
    normal conversational answer.
    """
    return "\n\n".join(doc.page_content for doc in docs)


# A chat model expects role-based messages and explicit style instructions.
# The previous prompt was a bare text blob ("Answer the question based on the
# following context"), which the chat model satisfied as literally as possible:
# a terse, extracted fragment. Adding a system message that asks for a full,
# conversational answer makes ChatOpenAI behave like llama2 does.
SYSTEM_TEMPLATE = """You are a helpful, friendly conversational assistant.
Answer the user's question using ONLY the provided context.
Write a complete, natural, well-formed answer in full sentences (a short
paragraph, not bullet fragments), and speak directly to the user in a
conversational tone. If the context does not contain the answer, say so
politely instead of guessing."""

ANSWER_TEMPLATE = """Context:
{context}

Question: {question}

Answer the question in a complete, conversational response."""


def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", SYSTEM_TEMPLATE),
            ("human", ANSWER_TEMPLATE),
        ]
    )

    chain = (
        {
            "context": retriever | format_docs,
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)

    return result


if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
