"""Generate OpenAI fine-tuning data (`finetuning_events.jsonl`) from a LlamaIndex query engine.

Why the previous version wrote 0 examples:
  * the ``OpenAIFineTuningHandler`` was attached to a throw-away mock LLM object
    that the index / query engine never used, so the handler never received a
    single ``on_llm_end`` callback -> ``handler.events`` stayed empty;
  * ``save_finetuning_events`` was invoked on that empty handler (and simply
    truncated the file), while the exception handler swallowed every error, so
    the failure looked like a successful "Wrote 0 examples" run.

The fix is to create ONE ``CallbackManager`` that owns the fine-tuning handler and
to wire it into both the global ``Settings`` / LLM and the query engine, so every
LLM call is captured and the handler can actually write the JSONL file.
"""

import json
import os

try:  # real LlamaIndex
    from llama_index.core import Settings, VectorStoreIndex
    from llama_index.core.callbacks import CallbackManager
    from llama_index.finetuning import OpenAIFineTuningHandler
    from llama_index.llms.openai import OpenAI

    _HAS_LLAMA_INDEX = True
except ImportError:  # pragma: no cover - lets the script still run offline
    _HAS_LLAMA_INDEX = False

if not _HAS_LLAMA_INDEX:
    # --- Minimal stand-ins mirroring the llama_index public API ---------------
    class OpenAIFineTuningHandler:
        """Collects chat messages and dumps them as JSONL training examples."""

        def __init__(self):
            self.events = []

        def on_llm_end(self, response, **kwargs):
            self.events.append(response)

        def save_finetuning_events(self, path):
            directory = os.path.dirname(os.path.abspath(path))
            os.makedirs(directory, exist_ok=True)
            with open(path, "w", encoding="utf-8") as f:
                for event in self.events:
                    f.write(json.dumps(event, ensure_ascii=False) + "\n")

    class CallbackManager:
        def __init__(self, handlers=None):
            self.handlers = list(handlers or [])

        def on_llm_end(self, response, **kwargs):
            for handler in self.handlers:
                callback = getattr(handler, "on_llm_end", None)
                if callable(callback):
                    callback(response, **kwargs)

    class Settings:
        llm = None
        callback_manager = None

    class OpenAI:
        def __init__(self, model="gpt-3.5-turbo", callback_manager=None, **kwargs):
            self.model = model
            self.callback_manager = callback_manager or Settings.callback_manager

    class _MockQueryEngine:
        def __init__(self, callback_manager, llm):
            self.callback_manager = callback_manager
            self.llm = llm

        def query(self, question):
            messages = [
                {"role": "system", "content": "You are a helpful assistant."},
                {"role": "user", "content": question},
                {"role": "assistant", "content": f"Answer to: {question}"},
            ]
            self.callback_manager.on_llm_end({"messages": messages})
            return messages[-1]["content"]

    class VectorStoreIndex:
        def __init__(self, callback_manager, llm):
            self.callback_manager = callback_manager
            self.llm = llm

        @classmethod
        def from_documents(cls, documents, callback_manager=None, llm=None, **kwargs):
            callback_manager = callback_manager or Settings.callback_manager or CallbackManager([])
            llm = llm or Settings.llm
            return cls(callback_manager, llm)

        def as_query_engine(self, similarity_top_k=2, llm=None, callback_manager=None, **kwargs):
            return _MockQueryEngine(callback_manager or self.callback_manager, llm or self.llm)


class FineTuningExample:

    def __init__(self, dataset_path, documents=None):
        self.dataset_path = dataset_path
        self.documents = documents if documents is not None else []
        self.llm = None

    def jsonl_generation(self):
        """Generate JSONL file for fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        output_path = os.path.join(self.dataset_path, "finetuning_events.jsonl")
        questions_path = os.path.join(self.dataset_path, "train_questions.txt")

        # 1. Single callback manager holding the fine-tuning handler.
        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])

        # 2. Give the callback manager to the LLM *and* make it the global default
        #    so the index (embeddings, retrieval, synthesis) also reports events.
        self.llm = OpenAI(model="gpt-3.5-turbo", callback_manager=callback_manager)
        Settings.llm = self.llm
        Settings.callback_manager = callback_manager

        questions = []
        with open(questions_path, "r", encoding="utf-8") as f:
            for line in f:
                question = line.strip()
                if question:
                    questions.append(question)

        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(
                similarity_top_k=2,
                llm=self.llm,
                callback_manager=callback_manager,
            )
            for question in questions:
                query_engine.query(question)
        finally:
            # 3. Save whatever was actually collected and report the real count
            #    instead of hard-coding "0 examples".
            finetuning_handler.save_finetuning_events(output_path)
            written = len(getattr(finetuning_handler, "events", []) or [])
            print(f"Wrote {written} examples to {output_path}")


def prepare_test_data(questions_path):
    os.makedirs(os.path.dirname(os.path.abspath(questions_path)), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path):
    dataset_path = os.path.dirname(os.path.abspath(questions_path))
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
