class OpenAIFineTuningHandler(BaseCallbackHandler):
    """Records every LLM input/output pair as an OpenAI chat fine-tuning example.

    Examples accumulate in ``self.events`` while the pipeline runs and are only
    written to disk by :meth:`save_finetuning_events`.
    """

    def __init__(self, callback_manager: Optional["CallbackManager"] = None) -> None:
        self.events: List[Dict[str, Any]] = []
        self._current_messages: List[ChatMessage] = []
        self._event_ids: List[str] = []
        if callback_manager is not None:
            callback_manager.add_handler(self)

    # -- callback API -------------------------------------------------------
    def on_event_start(
        self,
        event_type: CBEventType,
        payload: Optional[Dict[Any, Any]] = None,
        event_id: str = "",
        parent_id: str = "",
        **kwargs: Any,
    ) -> str:
        if event_type == CBEventType.LLM:
            payload = payload or {}
            self._current_messages = [
                _as_chat_message(m)
                for m in (payload.get(EventPayload.MESSAGES) or [])
            ]
            if not self._current_messages and payload.get(EventPayload.PROMPT):
                self._current_messages = [
                    ChatMessage(role="user", content=str(payload[EventPayload.PROMPT]))
                ]
            self._event_ids.append(event_id)
        return event_id

    def on_event_end(
        self,
        event_type: CBEventType,
        payload: Optional[Dict[Any, Any]] = None,
        event_id: str = "",
        **kwargs: Any,
    ) -> None:
        if event_type != CBEventType.LLM:
            return

        payload = payload or {}
        messages = list(self._current_messages)
        if not messages:
            messages = [
                _as_chat_message(m)
                for m in (payload.get(EventPayload.MESSAGES) or [])
            ]

        response = payload.get(EventPayload.RESPONSE)
        if isinstance(response, ChatMessage):
            messages.append(response)
        elif response is not None:
            messages.append(ChatMessage(role="assistant", content=str(response)))

        if len(messages) >= 2:  # need at least an input and an output
            self.events.append(
                {"messages": [_to_openai_message(m) for m in messages]}
            )

        self._current_messages = []

    # -- persistence --------------------------------------------------------
    def save_finetuning_events(self, path: str) -> None:
        """Write the collected chat examples to ``path`` as JSONL."""
        directory = os.path.dirname(os.path.abspath(path))
        if directory:
            os.makedirs(directory, exist_ok=True)

        with open(path, "w", encoding="utf-8") as f:
            for event in self.events:
                f.write(json.dumps(event, ensure_ascii=False) + "\n")
            f.flush()

        print(f"Wrote {len(self.events)} examples to {path}")


class CallbackManager:
    """Fan-out of callback events to every registered handler."""

    def __init__(self, handlers: Optional[List[BaseCallbackHandler]] = None) -> None:
        self.handlers: List[BaseCallbackHandler] = list(handlers or [])

    def add_handler(self, handler: BaseCallbackHandler) -> None:
        if handler not in self.handlers:
            self.handlers.append(handler)

    def on_event_start(
        self,
        event_type: CBEventType,
        payload: Optional[Dict[Any, Any]] = None,
        event_id: str = "",
        parent_id: str = "",
    ) -> str:
        for handler in self.handlers:
            handler.on_event_start(
                event_type=event_type,
                payload=payload,
                event_id=event_id,
                parent_id=parent_id,
            )
        return event_id

    def on_event_end(
        self,
        event_type: CBEventType,
        payload: Optional[Dict[Any, Any]] = None,
        event_id: str = "",
        parent_id: str = "",
    ) -> None:
        for handler in self.handlers:
            handler.on_event_end(
                event_type=event_type,
                payload=payload,
                event_id=event_id,
                parent_id=parent_id,
            )

    def start_trace(self, trace_id: Optional[str] = None) -> None:
        for handler in self.handlers:
            handler.start_trace(trace_id=trace_id)

    def end_trace(
        self,
        trace_id: Optional[str] = None,
        trace_map: Optional[Dict[str, List[str]]] = None,
    ) -> None:
        for handler in self.handlers:
            handler.end_trace(trace_id=trace_id, trace_map=trace_map)

class LLM:
    """Small LLM stand-in that emits ``CBEventType.LLM`` events."""

    def __init__(
        self,
        callback_manager: Optional[CallbackManager] = None,
        model: str = "gpt-3.5-turbo",
    ) -> None:
        self.callback_manager = callback_manager or CallbackManager()
        self.model = model
        self._counter = 0

    def _generate(self, messages: List[ChatMessage]) -> str:
        prompt = messages[-1].content if messages else ""
        question = prompt
        for marker in ("Question:", "question:"):
            if marker in question:
                question = question.split(marker, 1)[1]
                break
        question = question.strip().rstrip("?.! ")
        return f"Answer for '{question}'."

    def chat(self, messages: List[Any]) -> ChatMessage:
        chat_messages = [_as_chat_message(m) for m in messages]
        self._counter += 1
        event_id = f"llm-{self.model}-{self._counter}"

        self.callback_manager.on_event_start(
            CBEventType.LLM,
            payload={
                EventPayload.MESSAGES: list(chat_messages),
                EventPayload.PROMPT: str(chat_messages[-1]) if chat_messages else "",
            },
            event_id=event_id,
        )

        response: Optional[ChatMessage] = None
        try:
            response = ChatMessage(
                role="assistant", content=self._generate(chat_messages)
            )
            return response
        finally:
            self.callback_manager.on_event_end(
                CBEventType.LLM,
                payload={
                    EventPayload.MESSAGES: list(chat_messages),
                    EventPayload.RESPONSE: response,
                },
                event_id=event_id,
            )

    def complete(self, prompt: str) -> ChatMessage:
        return self.chat([ChatMessage(role="user", content=prompt)])

class Response:
    def __init__(self, response: Any, source_nodes: Optional[List[str]] = None) -> None:

    def __str__(self) -> str:

def _document_text(document: Any) -> str:
    if isinstance(document, str):
        return document
    for attribute in ("text", "content", "page_content"):
        value = getattr(document, attribute, None)
        if isinstance(value, str):
            return value
    return str(document)


class QueryEngine:
    def __init__(
        self,
        index: "VectorStoreIndex",
        llm: LLM,
        similarity_top_k: int = 2,
        callback_manager: Optional[CallbackManager] = None,
    ) -> None:
        self.index = index
        self.llm = llm
        self.similarity_top_k = similarity_top_k
        self.callback_manager = callback_manager or index.callback_manager
        # The LLM performing the call MUST share the pipeline's callback
        # manager, otherwise the fine-tuning handler never sees the event.
        self.llm.callback_manager = self.callback_manager

    def query(self, question: str) -> Response:
        nodes = self.index.retrieve(question, self.similarity_top_k)
        context = "\n\n".join(nodes) if nodes else "(no context available)"
        messages = [
            ChatMessage(role="system", content=SYSTEM_PROMPT),
            ChatMessage(
                role="user",
                content=f"Context:\n{context}\n\nQuestion: {question}",
            ),
        ]
        answer = self.llm.chat(messages)
        return Response(response=answer, source_nodes=nodes)


class VectorStoreIndex:
    def __init__(
        self,
        documents: Optional[List[Any]] = None,
        callback_manager: Optional[CallbackManager] = None,
    ) -> None:
        self.documents: List[str] = [_document_text(doc) for doc in (documents or [])]
        if callback_manager is None:
            callback_manager = Settings.callback_manager
        self.callback_manager = callback_manager or CallbackManager()

    @staticmethod
    def from_documents(
        docs: List[Any],
        callback_manager: Optional[CallbackManager] = None,
        **kwargs: Any,
    ) -> "VectorStoreIndex":
        return VectorStoreIndex(documents=docs, callback_manager=callback_manager)

    def retrieve(self, question: str, similarity_top_k: int = 2) -> List[str]:
        if not self.documents:
            return []
        terms = {t for t in question.lower().split() if t}
        scored = []
        for document in self.documents:
            score = sum(1 for term in terms if term in document.lower())
            scored.append((score, document))
        scored.sort(key=lambda item: item[0], reverse=True)
        return [doc for score, doc in scored[:similarity_top_k] if score > 0]

    def as_query_engine(
        self,
        similarity_top_k: int = 2,
        llm: Optional[LLM] = None,
        callback_manager: Optional[CallbackManager] = None,
        **kwargs: Any,
    ) -> QueryEngine:
        llm = llm or Settings.llm
        if llm is None:
            raise ValueError(
                "No LLM provided: pass `llm=` or set `Settings.llm` before "
                "building the query engine."
            )
        manager = (
            callback_manager
            or self.callback_manager
            or Settings.callback_manager
            or llm.callback_manager
        )
        self.callback_manager = manager
        # Make sure the LLM used by the engine emits into the same manager.
        llm.callback_manager = manager
        return QueryEngine(
            index=self,
            llm=llm,
            similarity_top_k=similarity_top_k,
            callback_manager=manager,
        )

class FineTuningExample:
    def __init__(self, dataset_path: str, documents: Optional[List[Any]]) -> None:
        self.dataset_path = dataset_path
        self.documents = documents or []
        self.llm: Optional[LLM] = None

    def _read_questions(self) -> List[str]:
        questions_path = os.path.join(self.dataset_path, "train_questions.txt")
        if not os.path.exists(questions_path):
            raise FileNotFoundError(f"Questions file not found: {questions_path}")
        questions: List[str] = []
        with open(questions_path, "r", encoding="utf-8") as f:
            for line in f:
                question = line.strip()
                if question:
                    questions.append(question)
        return questions

    def jsonl_generation(self) -> None:
        """Generate the JSONL file for fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        # 1) One handler + ONE callback manager shared by everything.
        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        Settings.callback_manager = callback_manager

        # 2) An LLM attached to that callback manager, and the LLM actually
        #    used by the query engine.
        self.llm = LLM(callback_manager=callback_manager)
        Settings.llm = self.llm

        questions = self._read_questions()
        if not questions:

                callback_manager=callback_manager,
            )

        except Exception as e:
            print(f"An error occurred: {e}")
        finally:

            os.makedirs(directory, exist_ok=True)

        with open(path, "w", encoding="utf-8") as f:
            for event in self.events:
                f.write(json.dumps(event, ensure_ascii=False) + "\n")
            f.flush()

        print(f"Wrote {len(self.events)} examples to {path}")


class CallbackManager:
    """Fan-out of callback events to every registered handler."""

    def __init__(self, handlers: Optional[List[BaseCallbackHandler]] = None) -> None:
        self.handlers: List[BaseCallbackHandler] = list(handlers or [])

    def add_handler(self, handler: BaseCallbackHandler) -> None:
        if handler not in self.handlers:
            self.handlers.append(handler)

    def on_event_start(
        self,
        event_type: CBEventType,
        payload: Optional[Dict[Any, Any]] = None,
        event_id: str = "",
        parent_id: str = "",
    ) -> str:
        for handler in self.handlers:
            handler.on_event_start(
                event_type=event_type,
                payload=payload,
                event_id=event_id,
                parent_id=parent_id,
            )
        return event_id

    def on_event_end(
        self,
        event_type: CBEventType,
        payload: Optional[Dict[Any, Any]] = None,
        event_id: str = "",
        parent_id: str = "",
    ) -> None:
        for handler in self.handlers:
            handler.on_event_end(
                event_type=event_type,
                payload=payload,
                event_id=event_id,
                parent_id=parent_id,
            )

    def start_trace(self, trace_id: Optional[str] = None) -> None:
        for handler in self.handlers:
            handler.start_trace(trace_id=trace_id)

    def end_trace(
        self,
        trace_id: Optional[str] = None,
        trace_map: Optional[Dict[str, List[str]]] = None,
    ) -> None:
        for handler in self.handlers:
            handler.end_trace(trace_id=trace_id, trace_map=trace_map)


# ---------------------------------------------------------------------------
# Global settings (stand-in for llama_index.core.Settings)
# ---------------------------------------------------------------------------
class Settings:
    """Single source of truth for the LLM and callback manager in use."""

    llm: Optional["LLM"] = None
    callback_manager: Optional[CallbackManager] = None

    @classmethod
    def reset(cls) -> None:
        cls.llm = None
        cls.callback_manager = None


# ---------------------------------------------------------------------------
# LLM
# ---------------------------------------------------------------------------
class LLM:
    """Small LLM stand-in that emits ``CBEventType.LLM`` events."""

    def __init__(
        self,
        callback_manager: Optional[CallbackManager] = None,
        model: str = "gpt-3.5-turbo",
    ) -> None:
        self.callback_manager = callback_manager or CallbackManager()
        self.model = model
        self._counter = 0

    def _generate(self, messages: List[ChatMessage]) -> str:
        prompt = messages[-1].content if messages else ""
        question = prompt
        for marker in ("Question:", "question:"):
            if marker in question:
                question = question.split(marker, 1)[1]
                break
        question = question.strip().rstrip("?.! ")
        return f"Answer for '{question}'."

    def chat(self, messages: List[Any]) -> ChatMessage:
        chat_messages = [_as_chat_message(m) for m in messages]
        self._counter += 1
        event_id = f"llm-{self.model}-{self._counter}"

        self.callback_manager.on_event_start(
            CBEventType.LLM,
            payload={
                EventPayload.MESSAGES: list(chat_messages),
                EventPayload.PROMPT: str(chat_messages[-1]) if chat_messages else "",
            },
            event_id=event_id,
        )

        response: Optional[ChatMessage] = None
        try:
            response = ChatMessage(
                role="assistant", content=self._generate(chat_messages)
            )
            return response
        finally:
            self.callback_manager.on_event_end(
                CBEventType.LLM,
                payload={
                    EventPayload.MESSAGES: list(chat_messages),
                    EventPayload.RESPONSE: response,
                },
                event_id=event_id,
            )

    def complete(self, prompt: str) -> ChatMessage:
        return self.chat([ChatMessage(role="user", content=prompt)])


# ---------------------------------------------------------------------------
# Index / query engine
# ---------------------------------------------------------------------------
SYSTEM_PROMPT = (
    "You are a helpful assistant answering questions using the provided context."
)


class Response:
    def __init__(self, response: Any, source_nodes: Optional[List[str]] = None) -> None:
        self.response = response
        self.source_nodes = source_nodes or []

    def __str__(self) -> str:
        return str(self.response)


def _document_text(document: Any) -> str:
    if isinstance(document, str):
        return document
    for attribute in ("text", "content", "page_content"):
        value = getattr(document, attribute, None)
        if isinstance(value, str):
            return value
    return str(document)


class QueryEngine:
    def __init__(
        self,
        index: "VectorStoreIndex",
        llm: LLM,
        similarity_top_k: int = 2,
        callback_manager: Optional[CallbackManager] = None,
    ) -> None:
        self.index = index
        self.llm = llm
        self.similarity_top_k = similarity_top_k
        self.callback_manager = callback_manager or index.callback_manager
        # The LLM performing the call MUST share the pipeline's callback
        # manager, otherwise the fine-tuning handler never sees the event.
        self.llm.callback_manager = self.callback_manager

    def query(self, question: str) -> Response:
        nodes = self.index.retrieve(question, self.similarity_top_k)
        context = "\n\n".join(nodes) if nodes else "(no context available)"
        messages = [
            ChatMessage(role="system", content=SYSTEM_PROMPT),
            ChatMessage(
                role="user",
                content=f"Context:\n{context}\n\nQuestion: {question}",
            ),
        ]
        answer = self.llm.chat(messages)
        return Response(response=answer, source_nodes=nodes)


class VectorStoreIndex:
    def __init__(
        self,
        documents: Optional[List[Any]] = None,
        callback_manager: Optional[CallbackManager] = None,
    ) -> None:
        self.documents: List[str] = [_document_text(doc) for doc in (documents or [])]
        if callback_manager is None:
            callback_manager = Settings.callback_manager
        self.callback_manager = callback_manager or CallbackManager()

    @staticmethod
    def from_documents(
        docs: List[Any],
        callback_manager: Optional[CallbackManager] = None,
        **kwargs: Any,
    ) -> "VectorStoreIndex":
        return VectorStoreIndex(documents=docs, callback_manager=callback_manager)

    def retrieve(self, question: str, similarity_top_k: int = 2) -> List[str]:
        if not self.documents:
            return []
        terms = {t for t in question.lower().split() if t}
        scored = []
        for document in self.documents:
            score = sum(1 for term in terms if term in document.lower())
            scored.append((score, document))
        scored.sort(key=lambda item: item[0], reverse=True)
        return [doc for score, doc in scored[:similarity_top_k] if score > 0]

    def as_query_engine(
        self,
        similarity_top_k: int = 2,
        llm: Optional[LLM] = None,
        callback_manager: Optional[CallbackManager] = None,
        **kwargs: Any,
    ) -> QueryEngine:
        llm = llm or Settings.llm
        if llm is None:
            raise ValueError(
                "No LLM provided: pass `llm=` or set `Settings.llm` before "
                "building the query engine."
            )
        manager = (
            callback_manager
            or self.callback_manager
            or Settings.callback_manager
            or llm.callback_manager
        )
        self.callback_manager = manager
        # Make sure the LLM used by the engine emits into the same manager.
        llm.callback_manager = manager
        return QueryEngine(
            index=self,
            llm=llm,
            similarity_top_k=similarity_top_k,
            callback_manager=manager,
        )


# ---------------------------------------------------------------------------
# Fine-tuning dataset generation
# ---------------------------------------------------------------------------
class FineTuningExample:
    def __init__(self, dataset_path: str, documents: Optional[List[Any]]) -> None:
        self.dataset_path = dataset_path
        self.documents = documents or []
        self.llm: Optional[LLM] = None

    def _read_questions(self) -> List[str]:
        questions_path = os.path.join(self.dataset_path, "train_questions.txt")
        if not os.path.exists(questions_path):
            raise FileNotFoundError(f"Questions file not found: {questions_path}")
        questions: List[str] = []
        with open(questions_path, "r", encoding="utf-8") as f:
            for line in f:
                question = line.strip()
                if question:
                    questions.append(question)
        return questions

    def jsonl_generation(self) -> None:
        """Generate the JSONL file for fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        # 1) One handler + ONE callback manager shared by everything.
        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        Settings.callback_manager = callback_manager

        # 2) An LLM attached to that callback manager, and the LLM actually
        #    used by the query engine.
        self.llm = LLM(callback_manager=callback_manager)
        Settings.llm = self.llm

        questions = self._read_questions()
        if not questions:
            print(f"No questions found in {self.dataset_path}/train_questions.txt")

        try:
            index = VectorStoreIndex.from_documents(
                self.documents, callback_manager=callback_manager
            )
            query_engine = index.as_query_engine(
                similarity_top_k=2,
                llm=self.llm,
                callback_manager=callback_manager,
            )
            for question in questions:
                query_engine.query(question)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                f'{self.dataset_path}/finetuning_events.jsonl'
            )


def prepare_test_data(questions_path: str) -> None:
    directory = os.path.dirname(questions_path)
    if directory:
        os.makedirs(directory, exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path: str) -> None:
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
