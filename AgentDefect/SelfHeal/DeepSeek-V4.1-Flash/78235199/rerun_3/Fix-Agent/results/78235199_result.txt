"""Fixed version of the fine-tuning JSONL generation script.

Why the original script always printed "Wrote 0 examples":

1. `OpenAIFineTuningHandler.save_finetuning_events()` never wrote anything: it
   truncated the file with an empty string and hard-coded the count ``0``.
2. Nothing ever *fed* events into the handler. ``VectorStoreIndex.query()``
   returned ``None`` and no LLM callback event was ever emitted, so the handler
   had an empty event list even if it had been written correctly.
3. ``run_agent()`` built the index with ``documents=[]``. With an empty index the
   query engine has no nodes to retrieve from, every query fails, and the
   ``except`` block swallowed the exception, leaving an empty event list that
   was then saved - hence an empty ``finetuning_events.jsonl``.

The fix:
  * the handler records every LLM message/response pair and writes one JSON
    object per line (OpenAI chat fine-tuning format),
  * the LLM is created *with* the callback manager so events reach the handler,
  * the query engine passes the callback manager through and emits events,
  * the index is built from real documents (never an empty list),
  * failures are reported instead of being silently swallowed, and the reported
    example count is the real number of events written.
"""

import json
import os
from typing import Any, Dict, List, Optional

SYSTEM_PROMPT = "You are a helpful question answering assistant."


class OpenAIFineTuningHandler:
    """Collects LLM inputs/outputs and saves them as OpenAI chat JSONL."""

    def __init__(self) -> None:
        self.events: List[Dict[str, Any]] = []

    # ------------------------------------------------------------------
    # callback hooks
    # ------------------------------------------------------------------
    def on_llm_end(
        self,
        messages: List[Dict[str, str]],
        response_message: Dict[str, str],
    ) -> None:
        """Store one training example (prompt messages + assistant message)."""
        if not messages or not response_message:
            return
        self.events.append({"messages": [dict(m) for m in messages] + [dict(response_message)]})

    # ------------------------------------------------------------------
    # persistence
    # ------------------------------------------------------------------
    def save_finetuning_events(self, path: str) -> int:
        """Write every recorded event to `path` and return the number written."""
        directory = os.path.dirname(os.path.abspath(path))
        os.makedirs(directory, exist_ok=True)

        with open(path, "w", encoding="utf-8") as f:
            for event in self.events:
                f.write(json.dumps(event, ensure_ascii=False) + "\n")

        num_examples = len(self.events)
        print(f"Wrote {num_examples} examples to {path}")
        return num_examples


class CallbackManager:
    """Minimal callback manager: dispatches LLM events to all handlers."""

    def __init__(self, handlers: Optional[List[Any]] = None) -> None:
        self.handlers: List[Any] = list(handlers or [])

    def add_handler(self, handler: Any) -> None:
        if handler not in self.handlers:
            self.handlers.append(handler)

    def on_llm_end(
        self,
        messages: List[Dict[str, str]],
        response_message: Dict[str, str],
    ) -> None:
        for handler in self.handlers:
            callback = getattr(handler, "on_llm_end", None)
            if callable(callback):
                callback(messages, response_message)


class MockLLM:
    """Stand-in for a real LLM that emits callback events."""

    def __init__(self, callback_manager: Optional[CallbackManager] = None) -> None:
        self.callback_manager = callback_manager or CallbackManager()

    def chat(self, messages: List[Dict[str, str]]) -> Dict[str, str]:
        question = messages[-1]["content"] if messages else ""
        answer = f"Answer to: {question}"
        response_message = {"role": "assistant", "content": answer}

        # Emit the event so that the fine-tuning handler can record the pair.
        self.callback_manager.on_llm_end(messages, response_message)
        return response_message


class QueryEngine:
    """Retrieves context for a question and asks the LLM to answer it."""

    def __init__(
        self,
        index: "VectorStoreIndex",
        similarity_top_k: int = 2,
        llm: Optional[MockLLM] = None,
        callback_manager: Optional[CallbackManager] = None,
    ) -> None:
        self.index = index
        self.similarity_top_k = similarity_top_k
        self.llm = llm or MockLLM()
        # Route the LLM events through the manager that holds our handler.
        self.callback_manager = (
            callback_manager
            or getattr(self.llm, "callback_manager", None)
            or CallbackManager()
        )
        self.llm.callback_manager = self.callback_manager

    def _retrieve(self, question: str) -> List[str]:
        documents = [doc for doc in self.index.documents if doc and doc.strip()]
        return documents[: self.similarity_top_k]

    def query(self, question: str) -> str:
        context = "\n\n".join(self._retrieve(question)) or "No context available."
        messages = [
            {"role": "system", "content": SYSTEM_PROMPT},
            {
                "role": "user",
                "content": f"Context:\n{context}\n\nQuestion: {question}",
            },
        ]
        response = self.llm.chat(messages)
        return response["content"]


class VectorStoreIndex:
    """Simple document index."""

    def __init__(self, documents: Optional[List[str]] = None) -> None:
        self.documents = list(documents or [])

    @classmethod
    def from_documents(cls, documents: Optional[List[str]]) -> "VectorStoreIndex":
        documents = [doc for doc in (documents or []) if doc and doc.strip()]
        if not documents:
            raise ValueError(
                "VectorStoreIndex.from_documents() received no documents; the query "
                "engine would have nothing to retrieve and no fine-tuning events "
                "would ever be produced."
            )
        return cls(documents)

    def as_query_engine(
        self,
        similarity_top_k: int = 2,
        llm: Optional[MockLLM] = None,
        callback_manager: Optional[CallbackManager] = None,
    ) -> QueryEngine:
        return QueryEngine(
            index=self,
            similarity_top_k=similarity_top_k,
            llm=llm,
            callback_manager=callback_manager,
        )


class FineTuningExample:
    def __init__(self, dataset_path: str, documents: Optional[List[str]] = None) -> None:
        self.dataset_path = dataset_path or "."
        self.documents = list(documents or [])
        self.llm: Optional[MockLLM] = None
        self.callback_manager: Optional[CallbackManager] = None
        self.finetuning_handler: Optional[OpenAIFineTuningHandler] = None

    # ------------------------------------------------------------------
    def _read_questions(self) -> List[str]:
        questions_path = os.path.join(self.dataset_path, "train_questions.txt")
        questions: List[str] = []
        with open(questions_path, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if line:
                    questions.append(line)
        return questions

    # ------------------------------------------------------------------
    def jsonl_generation(self) -> int:
        """Generate the JSONL file of fine-tuning events; returns the count."""
        os.makedirs(self.dataset_path, exist_ok=True)

        # 1) Create the handler that will actually collect the examples and a
        #    callback manager that knows about it.
        self.finetuning_handler = OpenAIFineTuningHandler()
        self.callback_manager = CallbackManager([self.finetuning_handler])

        # 2) Wire the LLM to the callback manager *before* using it.
        self.llm = MockLLM(callback_manager=self.callback_manager)

        questions = self._read_questions()

        # 3) The index needs real documents, otherwise nothing can be queried.
        if not self.documents:
            self.documents = list(questions)
            print("No documents supplied; indexing the questions instead.")

        num_examples = 0
        failures: List[tuple] = []
        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(
                similarity_top_k=2,
                llm=self.llm,
                callback_manager=self.callback_manager,
            )
            for question in questions:
                try:
                    query_engine.query(question)
                except Exception as exc:  # keep going, report afterwards
                    failures.append((question, exc))
        finally:
            num_examples = self.finetuning_handler.save_finetuning_events(
                os.path.join(self.dataset_path, "finetuning_events.jsonl")
            )

        if failures:
            print(f"{len(failures)} of {len(questions)} queries failed:")
            for question, exc in failures:
                print(f"  - {question!r}: {exc}")
        if num_examples == 0:
            print(
                "Warning: no fine-tuning events were captured - check that the LLM "
                "shares the callback manager that holds the fine-tuning handler."
            )
        return num_examples


def load_documents(dataset_path: str) -> List[str]:
    """Load every text/markdown source file in the dataset folder as a document."""
    documents: List[str] = []
    if os.path.isdir(dataset_path):

    return documents
