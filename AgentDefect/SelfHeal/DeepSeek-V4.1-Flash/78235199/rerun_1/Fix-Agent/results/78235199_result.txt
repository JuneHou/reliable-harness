import json
import os
import re

SYSTEM_PROMPT = (
    "You are a helpful assistant that answers questions using the provided context."
)


class Settings:
    """Module level configuration (mirrors ``llama_index.core.Settings``)."""

    callback_manager = None
    llm = None


class CallbackManager:
    """Dispatches the events emitted by LlamaIndex components to every handler."""

    def __init__(self, handlers=None):
        self.handlers = list(handlers or [])

    def add_handler(self, handler):
        self.handlers.append(handler)
        return self

    def on_event(self, event):
        for handler in self.handlers:
            callback = getattr(handler, "on_event", None)
            if callable(callback):
                callback(dict(event))


class OpenAIFineTuningHandler:
    """Records LLM input/output pairs and saves them as an OpenAI JSONL dataset."""

    def __init__(self):
        # The handler must actually keep the events it is given; the previous
        # version never stored anything, so nothing could ever be written.
        self.events = []

    def on_event(self, event):
        messages = event.get("messages") or []
        if not messages:
            return
        # Only keep complete question/answer pairs.
        if not messages[-1].get("content"):
            return
        self.events.append({"messages": messages})

    def save_finetuning_events(self, path):
        directory = os.path.dirname(os.path.abspath(path))
        if directory:
            os.makedirs(directory, exist_ok=True)

        with open(path, "w", encoding="utf-8") as f:
            # One JSON object per line is what OpenAI expects, writing an empty
            # string here is what produced an empty dataset before.
            for event in self.events:
                f.write(json.dumps(event, ensure_ascii=False) + "\n")

        print(f"Wrote {len(self.events)} examples to {path}")
        return len(self.events)


class Document:
    """Minimal stand-in for ``llama_index.core.Document``."""

    def __init__(self, text, metadata=None):
        self.text = text
        self.metadata = dict(metadata or {})

    def get_content(self):
        return self.text


class MockLLM:
    """Minimal stand-in for ``llama_index.llms.openai.OpenAI``.

    It keeps a ``callback_manager`` like a real LLM does and produces a
    deterministic, context grounded answer so the example runs offline.
    """

    def __init__(self, model="gpt-3.5-turbo", temperature=0.0, callback_manager=None):
        self.model = model
        self.temperature = temperature
        self.callback_manager = callback_manager or Settings.callback_manager

    @staticmethod
    def _between(text, start, end):
        pattern = re.escape(start) + r"(.*?)" + re.escape(end)
        match = re.search(pattern, text, re.DOTALL)
        return match.group(1).strip() if match else ""

    def complete(self, prompt):
        question = self._between(prompt, "Question:", "Answer:")
        context = self._between(prompt, "Context:", "Question:")

        if context and context != "(no context available)":
            sentences = [s.strip() for s in re.split(r"(?<=[.!?])\s+", context) if s.strip()]
            snippet = sentences[0] if sentences else context.strip()
            return f"{snippet}"

        return f"I don't have enough context to answer: {question}"


class QueryEngine:
    """Retrieves nodes from the index and records the LLM interaction."""

    def __init__(self, index, similarity_top_k=2, llm=None, callback_manager=None):
        self.index = index
        self.similarity_top_k = similarity_top_k
        self.llm = llm or Settings.llm or MockLLM()
        # Fall back to the index / global callback manager so the handlers are
        # never lost between the index and the query engine.
        self.callback_manager = (
            callback_manager
            or getattr(index, "callback_manager", None)
            or getattr(self.llm, "callback_manager", None)
            or Settings.callback_manager
        )

    def _retrieve(self, question):
        # Toy retriever: keep the first ``similarity_top_k`` documents.
        nodes = []
        for document in self.index.documents[: self.similarity_top_k]:
            nodes.append(document)
        return nodes

    def _build_prompt(self, question, nodes):
        context = "\n\n".join(node.get_content() for node in nodes)
        if not context.strip():
            context = "(no context available)"
        return (
            f"{SYSTEM_PROMPT}\n\n"
            f"Context:\n{context}\n\n"
            f"Question: {question}\n"
            f"Answer:"
        )

    def query(self, question):
        if not question or not question.strip():
            raise ValueError("The query text is empty.")

        nodes = self._retrieve(question)
        prompt = self._build_prompt(question, nodes)
        response = self.llm.complete(prompt)

        if not response:
            raise RuntimeError(f"The LLM returned no response for query: {question}")

        # Emitting the event is what feeds the fine-tuning handler.
        event = {
            "messages": [
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": question},
                {"role": "assistant", "content": response},
            ]
        }
        if self.callback_manager is not None:
            self.callback_manager.on_event(event)

        return response


class VectorStoreIndex:
    def __init__(self, documents=None, callback_manager=None):
        self.documents = list(documents or [])
        self.callback_manager = callback_manager or Settings.callback_manager

    @classmethod
    def from_documents(cls, docs, callback_manager=None):
        documents = [
            doc if isinstance(doc, Document) else Document(text=str(doc))
            for doc in list(docs or [])
        ]
        if not documents:
            print(
                "Warning: the index was built from an empty document list; "
                "queries will run without retrieved context."
            )

        # The callback manager has to live on the index (and on the query engine
        # created from it). Attaching it only to the LLM is the reason the
        # handler never saw a single event and 0 examples were written.
        return cls(documents, callback_manager=callback_manager or Settings.callback_manager)

    def as_query_engine(self, similarity_top_k=2, llm=None, callback_manager=None):
        return QueryEngine(
            self,
            similarity_top_k=similarity_top_k,
            llm=llm,
            callback_manager=callback_manager or self.callback_manager or Settings.callback_manager,
        )

    def query(self, question, **kwargs):
        return self.as_query_engine(**kwargs).query(question)


class FineTuningExample:
    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path or "."
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate JSONL file for fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])

        # Share ONE callback manager between the LLM, the index and the query
        # engine, and expose it through Settings as llama_index does.
        Settings.callback_manager = callback_manager
        self.llm = MockLLM(callback_manager=callback_manager)

        questions = []
        questions_path = f"{self.dataset_path}/train_questions.txt"
        with open(questions_path, "r", encoding="utf-8") as f:
            for line in f:
                question = line.strip()
                if question:
                    questions.append(question)

        if not questions:
            print(f"Warning: no questions found in {questions_path}")

        count = 0
        try:
            index = VectorStoreIndex.from_documents(
                self.documents, callback_manager=callback_manager
            )
            query_engine = index.as_query_engine(
                similarity_top_k=2, llm=self.llm, callback_manager=callback_manager
            )
            for question in questions:
                try:
                    query_engine.query(question)
                except Exception as e:
                    # Failing one question must not throw away the events that
                    # were already recorded for the other questions.
                    print(f"An error occurred while answering '{question}': {e}")
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            count = finetuning_handler.save_finetuning_events(
                f"{self.dataset_path}/finetuning_events.jsonl"
            )

        return count


def prepare_test_data(questions_path):
    directory = os.path.dirname(questions_path)
    if directory:
        os.makedirs(directory, exist_ok=True)

    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")

    # The index needs source documents, otherwise every query is answered
    # without context (and, with real llama_index, building the index fails).
    context_path = os.path.join(directory or ".", "context.txt")
    if not os.path.exists(context_path):
        with open(context_path, "w", encoding="utf-8") as f:
            f.write(
                "LlamaIndex is a framework that connects private data with large language models. "
                "The OpenAIFineTuningHandler records every LLM input and output pair. "
                "A callback manager dispatches events to all of its callback handlers.\n"
            )

    return questions_path


def load_documents(dataset_path):
    """Load the source documents that will be indexed."""
    documents = []
    if not os.path.isdir(dataset_path):
        return documents

    for file_name in sorted(os.listdir(dataset_path)):
        if file_name == "train_questions.txt" or file_name == "finetuning_events.jsonl":
            continue
        if not file_name.endswith((".txt", ".md")):
            continue
        file_path = os.path.join(dataset_path, file_name)
        with open(file_path, "r", encoding="utf-8") as f:
            documents.append(Document(text=f.read(), metadata={"file_name": file_name}))

    return documents


def run_agent(questions_path):
    dataset_path = os.path.dirname(questions_path) or "."
    documents = load_documents(dataset_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=documents)
    return ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
