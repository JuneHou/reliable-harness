import os

from dotenv import load_dotenv
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
from langchain_openai import ChatOpenAI

load_dotenv()


def run_agent(prompt: str) -> str:
    """Send a prompt to the chat model and return only the text answer."""
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise RuntimeError(
            "OPENAI_API_KEY is not set. Add it to your .env file or environment."
        )

    llm = ChatOpenAI(model="gpt-4o-mini", openai_api_key=api_key)

    # invoke() returns an AIMessage, not a plain string. Extract .content
    # (or pipe through StrOutputParser) so you get the raw text back.
    response = llm.invoke(prompt)
    return response.content


def run_agent_chain(prompt: str) -> str:
    """Same thing, but built as an explicit prompt -> model -> parser chain."""
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise RuntimeError(
            "OPENAI_API_KEY is not set. Add it to your .env file or environment."
        )

    llm = ChatOpenAI(model="gpt-4o-mini", openai_api_key=api_key)

    chain = (
        ChatPromptTemplate.from_messages([("human", "{question}")])
        | llm
        | StrOutputParser()
    )
    return chain.invoke({"question": prompt})


if __name__ == "__main__":
