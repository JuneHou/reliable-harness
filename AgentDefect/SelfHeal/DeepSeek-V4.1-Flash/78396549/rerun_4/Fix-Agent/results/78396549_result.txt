import os

from dotenv import load_dotenv
from langchain_core.messages import AIMessage
from langchain_openai import ChatOpenAI

load_dotenv()


def run_agent(prompt: str) -> str:
    """Send a prompt to the chat model and return the *text* of the answer.

    ``ChatOpenAI.invoke`` returns an ``AIMessage`` object (a structured
    message), not a plain string. Printing that object only shows its repr
    (``content='...' additional_kwargs={...} response_metadata={...}``), which
    is why the call looked like it "did nothing" / printed the wrong thing.
    The actual model text lives in ``response.content``.
    """
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise RuntimeError(
            "OPENAI_API_KEY is not set. Put it in a .env file or export it first."
        )

    llm = ChatOpenAI(model="gpt-4o-mini", openai_api_key=api_key)

    response: AIMessage = llm.invoke(prompt)

    # ``content`` can be a string or a list of content blocks (e.g. for
    # multimodal / tool models); normalise it to a single string.
    content = response.content
    if isinstance(content, list):
        text = "".join(
            block if isinstance(block, str) else block.get("text", "")
            for block in content
        )
    else:
        text = content

    return text.strip()


if __name__ == "__main__":
    user_prompt = "How can LangSmith help with testing?"
    answer = run_agent(user_prompt)
    print("LLM response:", answer)
