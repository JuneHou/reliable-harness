import os

from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

load_dotenv()


def run_agent(prompt: str) -> str:
    """Send a prompt to the chat model and return the plain-text answer.

    ``ChatOpenAI.invoke`` returns an ``AIMessage`` object, not a string.
    Printing that object directly shows its repr (or appears to print
    nothing useful), which is why the output looked "empty". We extract
    the actual text from ``response.content`` and return that.
    """
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise RuntimeError("OPENAI_API_KEY is not set (check your .env file)")

    llm = ChatOpenAI(
        model="gpt-4o-mini",
        openai_api_key=api_key,
        temperature=0,
    )

    response = llm.invoke(prompt)  # -> AIMessage
    content = response.content  # -> str (may be a list of blocks for some models)
    if isinstance(content, list):
        # Normalize multimodal / block-style content into a single string
        content = "".join(
            block.get("text", "") if isinstance(block, dict) else str(block)
            for block in content
        )
    return content


if __name__ == "__main__":
    user_prompt = "How can LangSmith help with testing?"
    answer = run_agent(user_prompt)
    print("LLM response:", answer)
