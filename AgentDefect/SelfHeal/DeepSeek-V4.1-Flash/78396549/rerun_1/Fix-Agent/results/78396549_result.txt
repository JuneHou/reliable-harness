import os

from dotenv import load_dotenv
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
from langchain_openai import ChatOpenAI

load_dotenv()


def run_agent(prompt: str) -> str:
    """Run a single prompt through the chat model and return plain text.

    `ChatOpenAI.invoke()` returns an `AIMessage` object, not a string.
    Printing the message directly makes the output look "empty"/wrong, so the
    textual answer must be read from `AIMessage.content` (or via
    `StrOutputParser`).
    """
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise RuntimeError(
            "OPENAI_API_KEY is not set. Add it to your environment or .env file."
        )

    llm = ChatOpenAI(model="gpt-4o-mini", openai_api_key=api_key.strip())

    # Option A (explicit): get the AIMessage and pull out the text content.
    response = llm.invoke(prompt)
    return response.content
