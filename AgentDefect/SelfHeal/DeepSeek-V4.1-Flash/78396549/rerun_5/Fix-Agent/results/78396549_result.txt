"""Minimal, working LangChain getting-started example.

The original snippet did call OpenAI, but `ChatOpenAI.invoke()` returns an
`AIMessage` object, not a plain string. Printing it shows the whole object
representation, so it looks like "nothing happened". The fix is to read the
generated text from `response.content` (and to fail loudly when the key is
missing or the model returns an empty answer).
"""

import os

from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

load_dotenv()


def run_agent(prompt: str) -> str:
    """Send `prompt` to the chat model and return only the generated text."""
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise RuntimeError(
            "OPENAI_API_KEY is not set. Put it in a .env file next to this "
            "script (OPENAI_API_KEY=sk-...) or export it before running."
        )

    # Use a currently supported chat model (gpt-3.5-turbo is retired).
    llm = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=api_key,
    )

    # invoke() returns an AIMessage, not a str.
    response = llm.invoke(prompt)

    # Extract the text payload through the documented attribute.
    content = response.content
    if isinstance(content, list):
        # Some models return a list of content blocks instead of a string.
        content = "".join(
            block if isinstance(block, str) else block.get("text", "")
            for block in content
        )

    if not content or not content.strip():
        raise RuntimeError(f"Model returned no text. Raw response: {response!r}")

    return content


if __name__ == "__main__":
    user_prompt = "How can LangSmith help with testing?"
    answer = run_agent(user_prompt)
    print("LLM response:", answer)
