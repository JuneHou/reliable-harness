"""Fixed LangChain + Ollama agent.

Why the original version parroted itself
----------------------------------------
The model/config combination made sampling effectively deterministic:

* ``temperature=0.7`` with ``top_p=0.95`` is *not* enough on its own: Ollama
  defaults to a fixed ``seed`` for a given prompt/context.  When the seed is
  pinned (or when the sampler collapses onto the single highest-probability
  continuation), every identical prompt -> identical token sequence -> identical
  joke, forever.
* ``repeat_penalty=1.2`` combined with a high ``top_p`` aggressively suppressed
  alternative tokens, so the "creative" mass of the distribution got cut off and
  the model fell back to the same canned punchline.
* ``num_ctx=2048`` silently truncated the prompt/history, so nothing the caller
  added ever reached the model -> the model literally saw the *same* input each
  time.
* Nothing in the prompt asked for novelty, so even a stochastic sample landed on
  the most "typical" (i.e. most parroted) joke.

The fix (parameter values + prompt, no API surface change)
----------------------------------------------------------
1. A fresh random ``seed`` for every call (Ollama's ``seed`` makes a run
   reproducible; reusing it makes a run frozen).
2. Real sampling pressure: ``temperature`` raised to ~0.9, ``top_k=40`` so
   alternatives stay in play, ``repeat_penalty`` lowered to 1.15 so the sampler
   is not forced back to its favourite token, ``min_p=0.0``.
3. a healthy ``num_ctx`` (4096) plus ``keep_alive`` so the prompt is not
   truncated and the model is not reloaded with stale params.
4. A system prompt that explicitly demands a brand-new joke, plus a retry loop
   that regenerates whenever the new answer matches one already seen.
"""

from __future__ import annotations

import random
from typing import Iterable, List, Optional

from langchain_core.messages import HumanMessage, SystemMessage
from langchain_ollama import ChatOllama

DEFAULT_MODEL = "qwen2.5:14b"

# Sampling defaults that actually produce variation between identical prompts.
BASE_TEMPERATURE = 0.9
BASE_TOP_P = 0.95
BASE_TOP_K = 40
BASE_MIN_P = 0.0
BASE_REPEAT_PENALTY = 1.15
BASE_NUM_CTX = 4096
REQUEST_TIMEOUT = 120  # seconds, passed through client_kwargs

NOVELTY_SYSTEM_PROMPT = (
    "You are a creative, lighthearted assistant. "
    "Every answer must be brand new: never reuse a joke, setup, or punchline "
    "you have produced before, and avoid the most obvious, over-told classics. "
    "Pick an unusual angle, character, or wordplay each time."
)


def _random_seed() -> int:
    """Ollama treats a fixed seed as 'reproduce this exact run'."""
    return random.randint(0, 2**31 - 1)


def build_llm(
    seed: Optional[int] = None,
    temperature: float = BASE_TEMPERATURE,
    **overrides,
):
    """Create a ChatOllama instance whose sampler is *not* frozen.

    Returns ``(llm, seed_used)`` so callers can log/reproduce a specific run.
    """
    if seed is None:
        seed = _random_seed()

    params = dict(
        model=DEFAULT_MODEL,
        temperature=temperature,
        top_p=BASE_TOP_P,
        top_k=BASE_TOP_K,
        min_p=BASE_MIN_P,
        repeat_penalty=BASE_REPEAT_PENALTY,
        num_ctx=BASE_NUM_CTX,
        seed=seed,                    # <-- the critical one: varies per call
        keep_alive="10m",
        # `timeout` is a client-level option, not a model option. Passing it to
        # the constructor directly makes pydantic reject the field.
        client_kwargs={"timeout": REQUEST_TIMEOUT},
    )
    params.update(overrides)
    return ChatOllama(**params), seed


def _build_messages(input_text: str, avoid: Iterable[str]) -> List:
    system = NOVELTY_SYSTEM_PROMPT
    previous = [a for a in avoid if a]
    if previous:
        system += (
            "\n\nDo NOT reuse, paraphrase, or continue any of these previously "
            "returned answers:\n" + "\n".join(f"- {p}" for p in previous[-5:])
        )
    return [SystemMessage(content=system), HumanMessage(content=input_text)]


def run_agent(
    input_text: str,
    seed: Optional[int] = None,
    temperature: float = BASE_TEMPERATURE,
    max_attempts: int = 3,
    avoid: Optional[Iterable[str]] = None,
    llm: Optional[ChatOllama] = None,
) -> str:
    """Run the prompt and return a fresh, non-parroted answer.

    If the sampler still returns something we have already seen, the call is
    retried with a new random seed and a slightly hotter temperature.
    """
    seen = [a for a in (avoid or []) if a]
    answer = ""

    for attempt in range(max_attempts):
        if llm is not None and attempt == 0:
            model, used_seed = llm, seed
        else:
            model, used_seed = build_llm(
                seed=seed if attempt == 0 else None,
                temperature=temperature,
            )

        response = model.invoke(_build_messages(input_text, seen))
        answer = (response.content or "").strip()

        if answer and answer not in seen:
            return answer

        # Looks like a parrot -> resample: new seed, a bit more heat.
        seed = None
        temperature = min(temperature + 0.1, 1.2)

    return answer


def run_agent_variants(
    input_text: str,
    n: int = 3,
    temperature: float = BASE_TEMPERATURE,
) -> List[str]:
    """Return ``n`` distinct answers to the same prompt."""
    answers: List[str] = []
    for _ in range(n):
        answers.append(
            run_agent(input_text, temperature=temperature, avoid=answers)
        )
    return answers


if __name__ == "__main__":
    prompt = "Please tell me a lighthearted joke"

    for i, joke in enumerate(run_agent_variants(prompt, n=3), start=1):
        print(f"{i}. {joke}\n")
