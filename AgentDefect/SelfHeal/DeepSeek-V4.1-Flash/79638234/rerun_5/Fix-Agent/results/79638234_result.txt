"""Ask a local Ollama model for a *new* joke on every call.

Root cause of the reported behaviour:
    * ``ChatOllama`` was never given a ``seed``.  Ollama then falls back to its
      default (fixed) sampling seed, so the same prompt + same options is
      decoded deterministically -> the very same joke every time, no matter
      what ``temperature`` is set to.
    * The prompt was a bare string with no system instructions, so the model
      has no incentive to produce "new material" and tends to parrot the
      request / recycle the same canned answer.
    * ``temperature=0.7`` with a large ``repeat_penalty`` further flattens the
      distribution, making repeats even more likely.

Fix:
    * pass a fresh (random) ``seed`` per invocation (``-1`` also works in
      Ollama to mean "randomise"), so sampling actually varies;
    * raise ``temperature`` and widen ``top_k``/``top_p`` a bit;
    * send a proper system message + explicit instruction that discourages
      parroting and repetition of previous punchlines;
    * keep everything else (context window, timeout) unchanged.
"""

import random

from langchain_core.messages import HumanMessage, SystemMessage
from langchain_ollama import ChatOllama

MODEL_NAME = "qwen2.5:14b"

SYSTEM_PROMPT = (
    "You are a witty stand-up comedian who is constantly writing brand-new material. "
    "Reply with exactly one short, original, lighthearted joke. "
    "Never restate, quote or paraphrase the user's request, and never reuse a "
    "punchline or setup you have produced before."
)

MAX_SEED = 2**31 - 1


def build_llm(seed=None):
    """Create a ChatOllama client.

    ``seed`` defaults to a fresh random value so that two identical prompts do
    not decode to the same tokens (Ollama's default seed is fixed, which is why
    the original code always returned the same joke).
    """
    if seed is None:
        seed = random.randint(0, MAX_SEED)

    return ChatOllama(
        model=MODEL_NAME,
        temperature=0.9,          # more variety than 0.7
        top_p=0.95,
        top_k=40,                 # widen the sampling pool
        repeat_penalty=1.05,      # was 1.2 -> too aggressive, kills variety
        repeat_last_n=256,
        num_ctx=2048,
        num_predict=256,          # don't let the model ramble/parrot
        seed=seed,                # <-- key fix: randomised per call
        timeout=120,
    )


def run_agent(input_text, seed=None, llm=None):
    """Send ``input_text`` to the model and return the generated text."""
    llm = llm or build_llm(seed=seed)

    messages = [
        SystemMessage(content=SYSTEM_PROMPT),
        HumanMessage(
            content=(
                f"{input_text}\n\n"
                "Write a joke I have never heard before. "
                "Do not repeat my question and do not explain the joke."
            )
        ),
    ]

    response = llm.invoke(messages)
    return response.content.strip()


if __name__ == "__main__":
    # A single client re-used across calls is fine: the seed is randomised per
    # request.  Building one per call (as below) works just as well.
    for attempt in range(3):
        print(f"--- attempt {attempt + 1} ---")
        print(run_agent("Please tell me a lighthearted joke"))
