"""Varied, non-parroted answers from a local Ollama model via LangChain.

Why the original snippet always answered the same thing
-------------------------------------------------------
1. Sampling was effectively too "peaked".  A 14B instruct-tuned model has a
   very dominant highest-probability completion for a bare prompt like
   "Please tell me a lighthearted joke".  With temperature=0.7 *and*
   repeat_penalty=1.2 (which flattens/reorders the distribution after the
   first tokens) the sampler keeps landing on exactly the same joke.
   -> temperature raised to 1.0, repeat_penalty lowered to 1.1, and top_k is
      set explicitly (Ollama's default is the rather narrow 40).
2. The random-number state is what makes an LLM non-deterministic.  Ollama
   reuses a seed for the loaded model; if a fixed/implicit seed is used the
   output is byte-identical for the same prompt no matter what temperature
   says.  -> a freshly generated `seed` is sent with every request.
3. The prompt itself never changed, so the model replayed its one memorised
   joke (and any prompt-level / response-level cache in front of Ollama would
   also return the same body).  -> a system message forbids repetition and the
   human message carries a per-call nonce, so every request is a genuinely
   new prompt.
"""

import random
import string
import sys

from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
from langchain_ollama import ChatOllama

MODEL = "qwen2.5:14b"

# --- 1. prompt: make "be different every time" an explicit instruction -------
SYSTEM_PROMPT = (
    "You are a stand-up comedian who never repeats material. "
    "Every time you are asked for a joke you must invent a brand new one: "
    "a new subject, a new setup and a new punchline. "
    "Never reuse a joke, a character or a punchline you have already used. "
    "Answer with the joke only - no preamble, no explanation."
)

HUMAN_PROMPT = (
    "{input_text}\n\n"
    "Requirements for this answer:\n"
    "- Pick one random, ordinary topic (animals, food, offices, weather, "
    "traffic, technology, siblings, ...) and build the joke around it.\n"
    "- Do not reuse anything from previous answers.\n"
    "- Variation token (meaningless, it only makes this request unique): {nonce}"
)


def _fresh_nonce(length: int = 8) -> str:
    """A short random token so that no two prompts are ever identical."""
    alphabet = string.ascii_lowercase + string.digits
    return "".join(random.choices(alphabet, k=length))


def _build_llm(seed):
    """ChatOllama with sampling options that actually produce variation."""
    kwargs = dict(
        model=MODEL,
        # 2. sampling: keep the distribution wide enough to vary
        temperature=1.0,
        top_p=0.95,
        top_k=64,
        repeat_penalty=1.1,
        num_ctx=2048,
        num_predict=256,
        # 3. fresh RNG state per call -> different sample each time
        seed=seed,
        keep_alive="10m",
    )
    try:
        # modern langchain-ollama: transport options go through client_kwargs
        return ChatOllama(client_kwargs={"timeout": 120}, **kwargs)
    except TypeError:  # older langchain-ollama / plain httpx client class
        return ChatOllama(**kwargs)


def run_agent(input_text, seed=None):
    """Ask the model `input_text` and return one freshly generated answer."""
    # `seed=None` -> brand new random seed for every call, never a fixed one
    if seed is None:
        seed = random.randint(0, 2**31 - 1)

    llm = _build_llm(seed)

    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", SYSTEM_PROMPT),
            ("human", HUMAN_PROMPT),
        ]
    )
    chain = prompt | llm | StrOutputParser()
    return chain.invoke({"input_text": input_text, "nonce": _fresh_nonce()}).strip()


if __name__ == "__main__":
    for i in range(1, 4):
        joke = run_agent("Please tell me a lighthearted joke")
        print(f"--- answer {i} ---")
        print(joke or "<empty answer: check that the Ollama server is running>")
        if not joke:
            sys.exit(1)
