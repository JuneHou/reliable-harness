import random
from typing import List, Optional, Union

from langchain_core.messages import BaseMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama

MODEL_NAME = "qwen2.5:14b"
REQUEST_TIMEOUT = 120

# A bare string prompt gives the model no role or format guidance, so a small
# local model tends to just "continue" the text it was handed and can end up
# parroting the question back. Send an explicit system message instead.
SYSTEM_PROMPT = (
    "You are a witty stand-up comedian chatting with a friend. "
    "When the user asks for a joke, reply with exactly ONE short, original, "
    "lighthearted joke and nothing else. Every reply must be a joke you have not "
    "told before in this conversation. Never repeat, quote or paraphrase the "
    "user's message back to them, never add commentary or explanations unless "
    "explicitly asked, and never output the bracketed variation tag."
)


def _new_seed() -> int:
    """Ollama's sampler is deterministic unless a seed is supplied.

    With identical prompt + identical options it walks the exact same sampling
    path, which is why the old code returned byte-identical answers even with
    temperature=0.7. A new seed per call guarantees a different decoding path.
    """
    return random.randint(0, 2**31 - 1)


def build_llm(seed: Optional[int] = None) -> ChatOllama:
    """Create a ChatOllama client whose sampling options Ollama actually honours."""
    return ChatOllama(
        model=MODEL_NAME,
        # --- variety / sampling knobs ---
        temperature=0.9,            # 0.7 was too "safe"; 0.9 yields real variety
        top_k=50,
        top_p=0.95,
        repeat_penalty=1.1,         # 1.2 over-penalised -> flat, repetitive text
        repeat_last_n=256,
        seed=_new_seed() if seed is None else seed,  # fresh RNG path per call
        # --- generation limits ---
        num_ctx=4096,               # 2048 could truncate the system prompt
        num_predict=256,
        keep_alive="30m",           # keep the model loaded between calls
        # NOTE: "timeout" is not a ChatOllama field (it used to be silently
        # dropped), so hand it to the underlying ollama client instead.
        client_kwargs={"timeout": REQUEST_TIMEOUT},
    )


def run_agent(
    input_text: str,
    llm: Optional[ChatOllama] = None,
    system_prompt: str = SYSTEM_PROMPT,
    max_attempts: int = 3,
) -> str:
    """Ask the model for a fresh answer, retrying if it just echoes the request."""
    messages_head: List[BaseMessage] = [SystemMessage(content=system_prompt)]
    lowered_input = input_text.lower().strip()

    last_text = ""
    for _ in range(max_attempts):
        # A new client each attempt carries a new random seed, so the model
        # explores a different decoding path instead of replaying itself.
        client = llm if (llm is not None and _ == 0) else build_llm()

        messages = messages_head + [
            HumanMessage(
                content=(
                    f"{input_text}\n\n"
                    f"[variation tag: {random.randrange(10**9)} — respond with a "
                    "brand new answer and do not echo this tag or the request]"
                )
            )
        ]

        response = client.invoke(messages)
        text = (response.content or "").strip()

        # Drop a leading echo of the user's request if the model parroted it.
        if lowered_input and text.lower().startswith(lowered_input):
            text = text[len(input_text):].lstrip(" :-,.\n")
        text = text.strip()

        last_text = text
        if text and not text.lower().startswith(lowered_input):
            return text

    return last_text


def tell_jokes(n: int = 3) -> List[str]:
    """Convenience helper: get n *different* jokes from the same model."""
    llm = build_llm()
    jokes, seen = [], set()
    for _ in range(n):
        joke = run_agent("Please tell me a lighthearted joke", llm=llm)
        # vary the seed for the next round trip as well
        llm = build_llm()
        if joke and joke not in seen:
            seen.add(joke)
            jokes.append(joke)
    return jokes


if __name__ == "__main__":
    for joke in tell_jokes(3):
        print(joke)
        print("-" * 60)
