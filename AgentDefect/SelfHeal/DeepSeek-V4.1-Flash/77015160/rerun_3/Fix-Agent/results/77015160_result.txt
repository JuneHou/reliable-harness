from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain_core.prompts import PromptTemplate
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


# The prompt MUST declare exactly the input variables that the "stuff" document
# chain of RetrievalQAWithSourcesChain feeds into it: {summaries} (the retrieved
# context) and {question}. Without this, LangChain silently falls back to its own
# built-in default prompt, which is why the object always printed/computed with
# the same default template and content.
PROMPT_TEMPLATE = """You are a helpful assistant that answers questions using only the provided context.

Use the following extracted parts of a long document to answer the question.
If you don't know the answer, just say "I don't know" and nothing else.
Never make up information.
Keep the answer concise and factual.

{summaries}

Question: {question}
Helpful Answer:"""

EXPECTED_INPUT_VARIABLES = {"summaries", "question"}


def build_prompt() -> PromptTemplate:
    prompt = PromptTemplate(
        template=PROMPT_TEMPLATE,
        input_variables=["summaries", "question"],
    )
    missing = EXPECTED_INPUT_VARIABLES - set(prompt.input_variables)
    if missing:
        raise ValueError(f"Custom prompt is missing required placeholders: {sorted(missing)}")
    return prompt


def run_agent():
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0, model_name="gpt-3.5-turbo")

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    prompt = build_prompt()

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm=llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        chain_type_kwargs={"prompt": prompt},  # <- makes the custom template actually used
        return_source_documents=True,
    )

    return chain


if __name__ == "__main__":
    chain = run_agent()

    # Inspect the prompt that the chain will really use, instead of printing the
    # whole chain object (which just shows LangChain's default/opaque repr).
    print("Active prompt template:\n")
    print(chain.combine_documents_chain.llm_chain.prompt.template)
    print("\nInput variables:", chain.combine_documents_chain.llm_chain.prompt.input_variables)

    # Actually run the chain with a question so the output matches expectations.
    result = chain.invoke({"question": "What is this document about?"})
    print("\nAnswer:", result["answer"])
    print("Sources:", result["sources"])
