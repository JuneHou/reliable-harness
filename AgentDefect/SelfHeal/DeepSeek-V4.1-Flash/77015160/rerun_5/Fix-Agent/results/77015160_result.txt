from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.chains.qa_with_sources.stuff_prompt import (
    CHAT_QUESTION_PROMPT,
    stuff_prompt,
)
from langchain.prompts import PromptTemplate, ChatPromptTemplate
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


# A custom question prompt. It must keep the placeholders that
# RetrievalQAWithSourcesChain feeds in: {summaries} and {question}.
# (If you don't pass one, LangChain silently falls back to its built-in
# CHAT_QUESTION_PROMPT / stuff_prompt, which is why the "default" template
# and its sample content always showed up.)
CUSTOM_QUESTION_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=(
        "Use only the following context to answer the question. "
        "If the answer is not in the context, say you don't know.\n\n"
        "Context:\n{summaries}\n\n"
        "Question: {question}\n"
        "Answer:"
    ),
)

# Custom "stuff" combine prompt: must keep {summaries} and {question}.
CUSTOM_COMBINE_PROMPT = ChatPromptTemplate.from_template(
    "Answer the question using only the provided documents. "
    "End the answer with a line 'SOURCES: <comma separated sources>'.\n\n"
    "Documents:\n{summaries}\n\n"
    "Question: {question}"
)


def run_agent(question: str = "What is this document about?") -> dict:
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        # Explicitly supply the templates instead of relying on the
        # framework defaults that were being printed/used.
        question_prompt=CUSTOM_QUESTION_PROMPT,
        combine_prompt=CUSTOM_COMBINE_PROMPT,
        return_source_documents=True,
    )

    # Actually run the chain, so the output is a real answer + sources,
    # not the chain's repr (which just echoed the default prompt text).
    return chain.invoke({"question": question})


if __name__ == "__main__":
    result = run_agent()
    print("ANSWER:", result["answer"])
    print("SOURCES:", result.get("sources"))

    # If you do want to inspect the prompt being used, print it explicitly
    # instead of printing the whole chain object.
    print("\nQUESTION PROMPT USED:\n", chain_prompt := CUSTOM_QUESTION_PROMPT.template
          if hasattr(CUSTOM_QUESTION_PROMPT, "template") else CUSTOM_QUESTION_PROMPT)
