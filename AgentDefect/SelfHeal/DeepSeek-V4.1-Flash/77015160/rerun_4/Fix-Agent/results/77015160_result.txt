import os

from dotenv import load_dotenv
from langchain.chains import RetrievalQAWithSourcesChain
from langchain_core.prompts import PromptTemplate
from langchain_openai import ChatOpenAI, OpenAIEmbeddings

try:  # langchain >= 0.1 keeps community integrations in their own package
    from langchain_community.vectorstores import FAISS
except ImportError:  # older langchain releases
    from langchain.vectorstores import FAISS


load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


# An explicit prompt is required. If you do not pass one, `from_chain_type`
# builds the chain with LangChain's built-in "stuff" template, which is exactly
# what kept showing up when the chain object was printed. The template
# variables must match what the qa-with-sources "stuff" chain feeds in:
#   {summaries} -> the retrieved document extracts
#   {question}  -> the user question
CUSTOM_TEMPLATE = """You are a precise question-answering assistant.
Answer the question using only the extracts below.
Write the answer as a single concise paragraph, then finish with one line
starting with "SOURCES:" that lists the source ids you actually used.

{summaries}

Question: {question}
Helpful Answer:"""

CUSTOM_PROMPT = PromptTemplate(
    template=CUSTOM_TEMPLATE,
    input_variables=["summaries", "question"],
)


def run_agent(question: str = "Which country is the first test document about?"):
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)
    retriever = db.as_retriever()

    # Inject the custom prompt through chain_type_kwargs so the combine
    # (stuff) chain stops falling back to its default template.
    try:
        chain = RetrievalQAWithSourcesChain.from_chain_type(
            llm=llm,
            chain_type="stuff",
            retriever=retriever,
            chain_type_kwargs={"prompt": CUSTOM_PROMPT},
            return_source_documents=True,
        )
    except (ValueError, TypeError):
        # Older releases do not forward `prompt`; build the chain first and
        # then replace the prompt object on its LLM chain.
        chain = RetrievalQAWithSourcesChain.from_chain_type(
            llm=llm,
            chain_type="stuff",
            retriever=retriever,
            return_source_documents=True,
        )
        chain.combine_documents_chain.llm_chain.prompt = CUSTOM_PROMPT

    # Show the template that is really used, instead of printing the whole
    # chain object (whose repr dumps the built-in default prompt).
    print("Prompt template in use:\n")
    print(chain.combine_documents_chain.llm_chain.prompt.template)
    print("-" * 60)

    # A chain has to be invoked with its inputs; printing it only shows its
    # (default) configuration and never produces an answer.
    result = chain.invoke({"question": question})

    print(f"Question: {question}")
    print(f"Answer:   {result['answer']}")
    print(f"Sources:  {result['sources']}")
    return result


if __name__ == "__main__":
    run_agent()
