import os

from dotenv import load_dotenv

from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import FAISS
from langchain.chains import RetrievalQAWithSourcesChain
from langchain_core.prompts import PromptTemplate

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


# The prompt for RetrievalQAWithSourcesChain MUST have exactly these two
# input variables: "summaries" (the retrieved context) and "question".
# It must also produce an "answer" and a "sources" section, because the chain
# parses those out of the LLM output.
CUSTOM_QA_WITH_SOURCES_PROMPT = PromptTemplate(
    template=(
        "Use the following excerpts of context to answer the question at the end.\n"
        "If you don't know the answer, just say that you don't know; do not try "
        "to make up an answer.\n\n"
        "{summaries}\n\n"
        "Question: {question}\n"
        "Helpful Answer (be concise):"
    ),
    input_variables=["summaries", "question"],
)


def build_chain():
    """Build (do NOT run) the retrieval chain."""
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = [
        "This is a test document about France.",
        "Another document about Germany.",
    ]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        # This is how a custom prompt is actually injected. Without it the chain
        # silently keeps using DEFAULT_QA_WITH_SOURCES_PROMPT.
        chain_type_kwargs={"prompt": CUSTOM_QA_WITH_SOURCES_PROMPT},
        # key the chain expects for the user question
        question_key="question",
    )
    return chain


def run_agent(question: str = "What is this document about?"):
    """Actually EXECUTE the chain and return its answer + sources.

    Previously the function returned the chain object itself, so printing it
    just dumped the chain's __repr__ (the default prompt template and its
    text) instead of any answer. That is why the output was always identical
    and never reflected a custom template.
    """
    chain = build_chain()

    result = chain.invoke({"question": question})

    answer = result.get("answer", "")
    sources = result.get("sources", "")

    return {
        "question": question,
        "answer": answer.strip(),
        "sources": sources.strip(),
    }


if __name__ == "__main__":
    question = "Which country is mentioned in the documents?"

    # Inspect the prompt that will really be used (instead of accidentally
    # printing the whole chain object).
    prompt_in_use = build_chain().combine_documents_chain.llm_chain.prompt
    print("Prompt template in use:")
    print(prompt_in_use.template)
    print("-" * 60)

    print(run_agent(question))
