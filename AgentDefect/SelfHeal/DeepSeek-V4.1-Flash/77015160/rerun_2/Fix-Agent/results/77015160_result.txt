from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.prompts import PromptTemplate
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# ---------------------------------------------------------------------------
# What was wrong:
#   `RetrievalQAWithSourcesChain.from_chain_type(...)` without a `prompt`
#   silently falls back to LangChain's hard-coded default stuff-prompt
#   ("Given the following extracted parts of a long document and a question,
#    create a final answer with references ("SOURCES") ...").  The script just
#   did `print(run_agent())`, which prints the *repr* of the chain object - and
#   that repr always dumps that default prompt/template.  Nothing was ever
#   executed, so there was no answer at all.
#
# Fix:
#   1. Provide an explicit prompt via `chain_type_kwargs={"prompt": PROMPT}`.
#      It MUST use the input variables the *sources* stuff-chain expects:
#      `summaries` (the retrieved context) and `question`, and it must ask the
#      model to return a "SOURCES:" line so the sources field is populated.
#   2. Actually invoke the chain with a question and print the answer/sources
#      instead of printing the chain object.
# ---------------------------------------------------------------------------

CUSTOM_PROMPT_TEMPLATE = """Use the following pieces of retrieved context to answer the question at the end.
If you don't know the answer, say that you don't know - do not make one up.
Answer in one or two concise sentences, and ALWAYS end your answer with a line
that starts with "SOURCES:" followed by the source ids you used.

QUESTION: {question}
=========
CONTEXT:
{summaries}
=========
FINAL ANSWER:"""

CUSTOM_PROMPT = PromptTemplate(
    template=CUSTOM_PROMPT_TEMPLATE,
    input_variables=["summaries", "question"],
)


def run_agent():
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm=llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        # this is what actually replaces the default prompt template
        chain_type_kwargs={"prompt": CUSTOM_PROMPT},
        # do not echo the (long) retrieved context back in the output
        return_source_documents=True,
    )

    return chain


if __name__ == "__main__":
    chain = run_agent()

    # Show the prompt that is really being used (should be OUR template,
    # not LangChain's default "Given the following extracted parts ...").
    print("=== Prompt used by the chain ===")
    print(chain.combine_documents_chain.llm_chain.prompt.template)

    # Actually run the chain for a question instead of printing the chain object.
    question = "Which country is the first document about?"
    result = chain({"question": question}, return_only_outputs=True)

    print("\n=== Question ===")
    print(question)
    print("\n=== Answer ===")
    print(result.get("answer", "").strip())
    print("\n=== Sources ===")
    print(result.get("sources", "").strip())
    print("\n=== Retrieved documents ===")
    for doc in result.get("source_documents", []):
        print(f"- [{doc.metadata.get('source', 'unknown')}] {doc.page_content}")
