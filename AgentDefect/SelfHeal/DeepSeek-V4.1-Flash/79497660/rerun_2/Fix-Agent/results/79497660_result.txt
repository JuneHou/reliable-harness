"""
Fixed example for: "How do I add custom columns (e.g. `external_id`) to the
Postgres / pgvector table that LlamaIndex creates, instead of burying those
fields inside the `metadata` JSON(B) column?"

Why the original output was wrong
---------------------------------
`PGVectorStore.from_params(...)` creates (and LlamaIndex's `add()` writes to)
a *fixed* schema:

    id | text | metadata (JSONB) | node_id | embedding | [text_search_tsv]

Every `Document.metadata` key - including `external_id` - is serialised into
the `metadata` JSON column. There is no parameter on `PGVectorStore` that
declares extra columns, so `external_id` never becomes a real column and it
cannot be indexed / joined / filtered on at the SQL level.

The fix
-------
1. Let `PGVectorStore` create its table (it does that in `__init__`), then
   `ALTER TABLE ... ADD COLUMN IF NOT EXISTS` for each custom column
   (plus an index) - see `ensure_custom_columns()`.
2. Keep the value flowing through node metadata (that is how the value reaches
   the store), and after ingestion move it into the dedicated column with one
   `UPDATE ... SET external_id = metadata ->> 'external_id'`
   - see `sync_custom_columns_from_metadata()`.
3. Query/filter on the real column instead of on `metadata->>'external_id'`
   - see `fetch_nodes_by_external_id()` and `list_table_columns()`.

The result: `external_id` is a genuine, typed, indexed column of the
LlamaIndex/pgvector table, and no longer only a key inside the metadata JSON.
"""

@contextlib.contextmanager
def pg_connection(dbname: str, *, autocommit: bool = True):
    """Yield a short-lived psycopg2 connection to `dbname`."""
    conn = psycopg2.connect(
        dbname=dbname,
        user=DB_USER,
        password=DB_PASSWORD,
        host=DB_HOST,
        port=DB_PORT,
    )
    conn.autocommit = autocommit
    try:
        yield conn
    finally:
        conn.close()


def ensure_database_exists(dbname: str) -> None:
    """Create the per-tenant database if it does not exist yet."""
    with pg_connection(ADMIN_DB) as conn, conn.cursor() as cur:
        cur.execute("SELECT 1 FROM pg_database WHERE datname = %s", (dbname,))
        if cur.fetchone() is None:
            cur.execute(sql.SQL("CREATE DATABASE {}").format(sql.Identifier(dbname)))
            print(f"[pgvector] created database '{dbname}'")


def _table_exists(conn, schema: str, table: str) -> bool:
    with conn.cursor() as cur:
        cur.execute(
            "SELECT to_regclass(%s) IS NOT NULL", (f'"{schema}"."{table}"',)
        )
        return bool(cur.fetchone()[0])


def ensure_custom_columns(
    vector_store: PGVectorStore,
    dbname: str,
    columns: Optional[Dict[str, str]] = None,
) -> None:
    """Add the custom columns (and an index per column) to the store's table."""
    columns = columns or CUSTOM_COLUMNS
    schema = vector_store.schema_name
    table = vector_store.table_name

    with pg_connection(dbname) as conn:
        # PGVectorStore normally creates the table in __init__; make sure it
        # exists before we try to ALTER it.
        if not _table_exists(conn, schema, table):
            try:  # private, but it is exactly what __init__ calls
                vector_store._initialize()  # type: ignore[attr-defined]
            except Exception as exc:  # pragma: no cover - defensive
                raise RuntimeError(
                    f"Could not create pgvector table {schema}.{table}: {exc}"
                ) from exc
            if not _table_exists(conn, schema, table):
                raise RuntimeError(
                    f"Table {schema}.{table} does not exist and could not be created."
                )

        with conn.cursor() as cur:
            for column, column_type in columns.items():
                cur.execute(
                    sql.SQL(
                        "ALTER TABLE {schema}.{table} "
                        "ADD COLUMN IF NOT EXISTS {col} {col_type}"
                    ).format(
                        schema=sql.Identifier(schema),
                        table=sql.Identifier(table),
                        col=sql.Identifier(column),
                        col_type=sql.SQL(column_type),
                    )
                )
                cur.execute(
                    sql.SQL(
                        "CREATE INDEX IF NOT EXISTS {idx} "
                        "ON {schema}.{table} ({col})"
                    ).format(
                        idx=sql.Identifier(f"idx_{table}_{column}"),
                        schema=sql.Identifier(schema),
                        table=sql.Identifier(table),
                        col=sql.Identifier(column),
                    )
                )
                print(
                    f"[pgvector] column '{column}' {column_type} ensured on "
                    f"{schema}.{table}"
                )


def sync_custom_columns_from_metadata(
    vector_store: PGVectorStore,
    dbname: str,
    columns: Optional[Dict[str, str]] = None,
) -> None:
    """Copy node metadata keys written by LlamaIndex into the real columns."""
    columns = columns or CUSTOM_COLUMNS
    schema = vector_store.schema_name
    table = vector_store.table_name

    with pg_connection(dbname) as conn, conn.cursor() as cur:
        for column in columns:
            cur.execute(
                sql.SQL(
                    """
                    UPDATE {schema}.{table}
                       SET {col} = metadata ->> %s
                     WHERE metadata ->> %s IS NOT NULL
                       AND ({col} IS DISTINCT FROM metadata ->> %s)
                    """
                ).format(
                    schema=sql.Identifier(schema),
                    table=sql.Identifier(table),
                    col=sql.Identifier(column),
                ),
                (column, column, column),
            )
            print(
                f"[pgvector] synced {cur.rowcount} row(s) of metadata['{column}'] "
                f"into column '{column}'"
            )

def list_table_columns(vector_store: PGVectorStore, dbname: str) -> List[Tuple[str, str]]:
    """Return [(column_name, data_type), ...] of the pgvector table."""
    schema = vector_store.schema_name
    table = vector_store.table_name
    with pg_connection(dbname) as conn, conn.cursor() as cur:
        cur.execute(
            """
            SELECT column_name, data_type
              FROM information_schema.columns
             WHERE table_schema = %s AND table_name = %s
             ORDER BY ordinal_position
            """,
            (schema, table),
        )
        return [(row[0], row[1]) for row in cur.fetchall()]


def fetch_nodes_by_external_id(
    vector_store: PGVectorStore, dbname: str, external_id: str
) -> List[Tuple[str, str, str]]:
    """Look a row up by the *column* (indexed), not by metadata->>'external_id'."""
    schema = vector_store.schema_name
    table = vector_store.table_name
    with pg_connection(dbname) as conn, conn.cursor() as cur:
        cur.execute(
            sql.SQL(
                "SELECT node_id, external_id, text FROM {schema}.{table} "
                "WHERE external_id = %s"
            ).format(
                schema=sql.Identifier(schema),
                table=sql.Identifier(table),
            ),
            (external_id,),
        )
        return cur.fetchall()
