            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
        # the fields LlamaIndex should store as real columns instead of JSON
        custom_columns={"external_id": "TEXT"},
        keep_metadata_copy=False,
    )

    return vector_store


def run_agent(request: Any) -> VectorStoreIndex:
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )

    # ------------------------------------------------------------------ checks
    print("Table columns:")
    for column in vector_store.describe_table():
        print(f"  {column['column']:<20} {column['type']}")

    print("\nRows (external_id is a column, no longer inside metadata_):")
    for row in vector_store.select_rows():
        print(f"  {row}")

    print(
        "\nLookup by column: node_ids for ext_002 =",
        vector_store.get_node_ids_by_external_id("ext_002"),
    )

    return index
