                        f"ON {self.table_ref} ({col})"
                    )

                    cur.execute(statement)

        super()._initialize()

                           AND metadata_ IS NOT NULL
                           AND metadata_ ? %s
                        """,

                    )

    def get_node_ids_by_column(self, column: str, value: Any) -> List[str]:

        if column not in self.custom_columns:

                    f"WHERE {_quote_ident(column)} = %s",

                )
                return [row[0] for row in cur.fetchall()]

    def delete_by_column(self, column: str, value: Any) -> None:
        if column not in self.custom_columns:
            raise ValueError(
                f"{column!r} is not a custom column. Known: {list(self.custom_columns)}"
            )
        with closing(self._connect()) as conn:
            with conn.cursor() as cur:
                cur.execute(
                    f"DELETE FROM {self.table_ref} WHERE {_quote_ident(column)} = %s",
                    (value,),
                )
            conn.commit()

    def describe_columns(self) -> List[Tuple[str, str]]:
        with closing(self._connect()) as conn:
            with conn.cursor() as cur:
                cur.execute(
                    """
                    SELECT column_name, data_type
                      FROM information_schema.columns
                     WHERE table_schema = %s AND table_name = %s
                     ORDER BY ordinal_position
                    """,

                )

                return cur.fetchall()

            return

        with closing(self._connect()) as conn:
            with conn.cursor() as cur:
                if not self._table_exists(cur):
                    # LlamaIndex has not created the table yet; retry on next call.
                    return

                statements: List[str] = []
                if self.schema_name:
                    statements.append(
                        f"CREATE SCHEMA IF NOT EXISTS {_quote_ident(self.schema_name)}"
                    )
                for column, sql_type in self.custom_columns.items():
                    col = _quote_ident(column)
                    statements.append(
                        f"ALTER TABLE {self.table_ref} ADD COLUMN IF NOT EXISTS {col} {sql_type}"
                    )
                    index_name = _quote_ident(
                        f"{self.table_name}_{column}_idx"[:63]
                    )
                    statements.append(
                        f"CREATE INDEX IF NOT EXISTS {index_name} "
                        f"ON {self.table_ref} ({col})"
                    )

                for statement in statements:
                    cur.execute(statement)
            conn.commit()

        self._custom_columns_ready = True

    def _initialize(self) -> None:
        """Run LlamaIndex' normal init (creates table/model), then add our columns."""
        super()._initialize()
        self.ensure_custom_columns()

    # ------------------------------------------------------------------ writes
    @staticmethod
    def _coerce(value: Any) -> Any:
        if value is None or isinstance(value, (str, int, float, bool)):
            return value
        return str(value)

    def _sync_custom_columns(self, nodes: Sequence[BaseNode]) -> None:
        """Copy ``node.metadata[column]`` into the real column for each node."""
        if not self.custom_columns or not nodes or not self.conn_kwargs:
            return

        self.ensure_custom_columns()

        columns = list(self.custom_columns)
        set_clause = ", ".join(f"{_quote_ident(c)} = %s" for c in columns)
        statement = (
            f"UPDATE {self.table_ref} SET {set_clause} WHERE node_id = %s"
        )

        rows: List[Tuple[Any, ...]] = []
        for node in nodes:
            metadata = node.metadata or {}
            values = [self._coerce(metadata.get(column)) for column in columns]
            if all(value is None for value in values):
                continue
            rows.append((*values, node.node_id))

        if not rows:
            return

        with closing(self._connect()) as conn:
            with conn.cursor() as cur:
                cur.executemany(statement, rows)
            conn.commit()

    def add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        ids = super().add(nodes, **add_kwargs)
        self._sync_custom_columns(nodes)
        return ids

    async def async_add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        ids = await super().async_add(nodes, **add_kwargs)
        self._sync_custom_columns(nodes)
        return ids

    # ------------------------------------------------------------------ reads
    def backfill_custom_columns(self) -> int:
        """Promote values for rows written before the columns existed."""
        if not self.custom_columns or not self.conn_kwargs:
            return 0
        self.ensure_custom_columns()

        updated = 0
        with closing(self._connect()) as conn:
            with conn.cursor() as cur:
                for column in self.custom_columns:
                    col = _quote_ident(column)
                    cur.execute(
                        f"""
                        UPDATE {self.table_ref}
                           SET {col} = metadata_ ->> %s
                         WHERE {col} IS NULL
                           AND metadata_ IS NOT NULL
                           AND metadata_ ? %s
                        """,
                        (column, column),
                    )
                    updated += cur.rowcount
            conn.commit()
        return updated

    def get_node_ids_by_column(self, column: str, value: Any) -> List[str]:
        """Plain SQL lookup on a first-class column (impossible via metadata JSON)."""
        if column not in self.custom_columns:
            raise ValueError(
                f"{column!r} is not a custom column. Known: {list(self.custom_columns)}"
            )
        with closing(self._connect()) as conn:
            with conn.cursor() as cur:
                cur.execute(
                    f"SELECT node_id FROM {self.table_ref} "
                    f"WHERE {_quote_ident(column)} = %s",
                    (value,),
                )
                return [row[0] for row in cur.fetchall()]

    def delete_by_column(self, column: str, value: Any) -> None:
        if column not in self.custom_columns:
            raise ValueError(
                f"{column!r} is not a custom column. Known: {list(self.custom_columns)}"
            )
        with closing(self._connect()) as conn:
            with conn.cursor() as cur:
                cur.execute(
                    f"DELETE FROM {self.table_ref} WHERE {_quote_ident(column)} = %s",
                    (value,),
                )
            conn.commit()

    def describe_columns(self) -> List[Tuple[str, str]]:
        with closing(self._connect()) as conn:
            with conn.cursor() as cur:
                cur.execute(
                    """
                    SELECT column_name, data_type
                      FROM information_schema.columns
                     WHERE table_schema = %s AND table_name = %s
                     ORDER BY ordinal_position
                    """,
                    (self.schema_name or "public", self.table_name),
                )
                return cur.fetchall()

    def fetch_rows(self, columns: Sequence[str]) -> List[Tuple[Any, ...]]:
        projection = ", ".join(_quote_ident(c) for c in columns)
        with closing(self._connect()) as conn:
            with conn.cursor() as cur:
                cur.execute(f"SELECT {projection} FROM {self.table_ref} ORDER BY id")
                return cur.fetchall()


def create_vector_store(request) -> CustomColumnPGVectorStore:
    """Vector store whose table has dedicated columns for our metadata fields."""
    ensure_database_exists(request.tenant)

    vector_store = CustomColumnPGVectorStore.from_connection_params(
        database=request.tenant,
        host=DB_HOST,
        port=DB_PORT,
        user=DB_USER,
        password=DB_PASSWORD,
        table_name=TABLE_NAME,
        schema_name=SCHEMA_NAME,
        embed_dim=EMBED_DIM,
        custom_columns=CUSTOM_COLUMNS,
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
    )

    # Create the table + custom columns up front so they exist before any insert.
    vector_store._initialize()

    return vector_store


def run_agent(request):
    vector_store = create_vector_store(request)

    documents = [
        Document(
            text="Sample document 1",
            metadata={"external_id": "ext_001", "source_system": "crm"},
        ),
        Document(
            text="Sample document 2",
            metadata={"external_id": "ext_002", "source_system": "crm"},
        ),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )

    return index, vector_store
