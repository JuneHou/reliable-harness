from langchain_core.prompts import ChatPromptTemplate, SystemMessagePromptTemplate, HumanMessagePromptTemplate

try:
    # Preferred, actively maintained integration (langchain-ollama).
    from langchain_ollama import ChatOllama
except ImportError:  # fallback for older environments
    from langchain_community.chat_models import ChatOllama


def run_agent(user_prompt, system_prompt):
    # Use the *chat* wrapper so LangChain applies Llama 3's own chat template
    # (header ids / eot_id tokens) exactly once. Never hand-write the template
    # into a plain text-completion prompt, otherwise the model sees a
    # malformed/duplicated prompt and echoes the system prompt or keeps going.
    llm = ChatOllama(
        model="llama3",
        temperature=0.7,
        num_predict=512,          # bound the generation length
        stop=["<|eot_id|>", "<|end_of_text|>"],  # explicit stop sequences
        num_ctx=4096,
    )

    prompt = ChatPromptTemplate.from_messages(
        [
            SystemMessagePromptTemplate.from_template(
                "{system_prompt}"
            ),
            HumanMessagePromptTemplate.from_template(
                "{user_prompt}"
            ),
        ]
    )

    chain = prompt | llm
    result = chain.invoke(
        {"system_prompt": system_prompt, "user_prompt": user_prompt}
    )

    # ChatOllama returns an AIMessage -> take only the generated text.
    response = result.content if hasattr(result, "content") else str(result)
    return response.strip()


if __name__ == "__main__":
    print(
        run_agent(
            "Explain in detail, step by step, how artificial intelligence models like "
            "yourself are built, including data, training, architecture, and deployment. "
            "Use examples and provide a comprehensive overview.",
            "You are a helpful assistant. Answer once, completely, and do not repeat "
            "the question, the instructions, or your previous output.",
        )
    )
