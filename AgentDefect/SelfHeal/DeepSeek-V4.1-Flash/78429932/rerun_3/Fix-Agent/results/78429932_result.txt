"""
Fixed example: single, complete Llama 3 response with LangChain + Ollama.

Why the original code produced repeated / runaway output
-------------------------------------------------------
`langchain_community.llms.Ollama` is a *text-completion* wrapper. It calls
Ollama's `/api/generate` endpoint, which:

  * does NOT apply the model's built-in chat template, and
  * does NOT know that `<|eot_id|>` (or `<|start_header_id|>`) ends a turn.

So the hand-written Llama-3 chat markup was just plain text to the model. When
generation reached the end of the assistant turn, nothing told the model to
stop, so it kept writing -- re-emitting the system prompt, new
`<|start_header_id|>user<|end_header_id|>` turns, etc.

The fix is to use the *chat* interface (`ChatOllama`, i.e. Ollama's
`/api/chat`), which applies the Llama 3 template itself and stops on the
correct end-of-turn tokens. No prompt markup should be hand-crafted, and the
system prompt must be passed as a `SystemMessage` rather than embedded in the
user text. Explicit stop sequences / generation limits are added as a
belt-and-braces guard.
"""

try:  # preferred, current package
    from langchain_ollama import ChatOllama
except ImportError:  # fallback for older installs
    from langchain_community.chat_models import ChatOllama

from langchain_core.messages import HumanMessage, SystemMessage


def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Send one prompt to Llama 3 and return exactly one complete answer."""

    llm = ChatOllama(
        model="llama3",
        # Deterministic-ish, coherent single answer
        temperature=0.7,
        top_p=0.9,
        # Hard cap so a runaway generation can never loop forever
        num_predict=1024,
        # Discourage the "repeat the prompt / keep going" behaviour
        repeat_penalty=1.1,
        repeat_last_n=256,
        # Stop as soon as the model emits an end-of-turn marker
        stop=["<|eot_id|>", "<|end_of_text|>", "<|start_header_id|>"],
        # Make sure we get one finished response, not a stream of chunks
        streaming=False,
    )

    # Roles are passed as typed messages -- LangChain/Ollama does the templating.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    response = llm.invoke(messages)

    # Response is an AIMessage; return only the generated text.
    return response.content


if __name__ == "__main__":
    print(
        run_agent(
            "Explain in detail, step by step, how artificial intelligence models "
            "like yourself are built, including data, training, architecture, and "
            "deployment. Use examples and provide a comprehensive overview.",
            "You are a helpful assistant.",
        )
    )
