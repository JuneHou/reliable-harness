"""Fixed LangChain + Ollama + Llama 3 example.

Why the original code produced bad output
-----------------------------------------
`langchain_community.llms.Ollama` is a *text-completion* wrapper: it sends the
string to Ollama's `/api/generate` endpoint. Ollama itself already wraps that
prompt in the model's chat template (defined in the llama3 Modelfile, which
contains `<|begin_of_text|>`, `<|start_header_id|>` ... `<|eot_id|>`).
So hand-writing the llama-3 special tokens in the prompt caused the template to
be applied a *second* time: the model saw a stray "<|begin_of_text|>" mid-prompt,
never received a real end-of-turn token, so it echoed the system prompt again and
kept rambling / continuing the "user"/"assistant" turns forever.

The fix has two parts
---------------------
1. Use the *chat* interface (`ChatOllama`) so LangChain passes structured
   messages (system / human) and Ollama applies the chat template exactly once.
2. Pass explicit stop tokens (`<|eot_id|>`, `<|end_of_text|>`) and bound the
   generation (`num_predict`) so a single `invoke()` call returns one complete,
   finite answer without repeating the system prompt or continuing to talk.
"""

from langchain_core.messages import HumanMessage, SystemMessage

# langchain-ollama >= 0.1 ships the maintained chat model.
try:
    from langchain_ollama import ChatOllama
except ImportError:  # older environments
    from langchain_community.chat_models import ChatOllama


def build_llm():
    return ChatOllama(
        model="llama3",
        temperature=0.7,
        # Do NOT inject <|begin_of_text|>/<|start_header_id|> yourself:
        # Ollama's llama3 template already adds them.
        stop=["<|eot_id|>", "<|end_of_text|>", "<|start_header_id|>"],
        # Guard against runaway generation ("keeps on writing").
        num_predict=1024,
        # Keep the model resident between calls (optional, speeds things up).
        keep_alive="5m",
    )


def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Run one prompt through the model and return a single complete response."""
    llm = build_llm()

    # Roles are expressed as message objects, not as raw chat-template text.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    response = llm.invoke(messages)

    # ChatOllama returns an AIMessage; take its content (a plain string).
    content = response.content
    if isinstance(content, list):  # multimodal-style content blocks
        content = "".join(
            part.get("text", "") if isinstance(part, dict) else str(part)
            for part in content
        )
    return content.strip()


if __name__ == "__main__":
    out = run_agent(
        "Explain in detail, step by step, how artificial intelligence models "
        "like yourself are built, including data, training, architecture, and "
        "deployment. Use examples and provide a comprehensive overview.",
        "You are a helpful assistant. Answer the user's question once, in full, "
        "and then stop.",
    )
    print(out)

    # ------------------------------------------------------------------
    # If you really must stay on the legacy text-completion wrapper, the
