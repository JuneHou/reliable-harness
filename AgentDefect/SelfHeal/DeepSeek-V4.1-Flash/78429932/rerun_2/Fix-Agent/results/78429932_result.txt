"""Corrected example: LangChain + Ollama + Llama 3 returning a single,
complete assistant response with no system-prompt echo / output repetition.

Root cause of the bug
---------------------
The original code used `langchain_community.llms.Ollama` (a *completion*
wrapper) and hand-wrote Llama 3 chat-template control tokens
(`<|begin_of_text|>`, `<|start_header_id|>`, `<|eot_id|>`, ...) inside the
prompt string.

Ollama applies Llama 3's chat template itself. Feeding already-templated text
means the control tokens are duplicated / nested, the model sees a malformed
conversation, and it responds by "finishing the document": it re-emits the
system prompt, re-opens headers, keeps generating further turns, and/or loops
on its own output.

Fixes
-----
1. Use a *chat* model (`ChatOllama`) and pass structured messages
   (`SystemMessage` / `HumanMessage`) instead of one raw string with manual
   special tokens. The template is then applied exactly once.
2. Provide explicit `stop` sequences so generation terminates at the end of the
   assistant turn instead of continuing into a fabricated next turn.
3. Bound/steady the generation with `num_predict`, `repeat_penalty`,
   `temperature` and an adequate `num_ctx`.
4. Return only the message content of the single `AIMessage`.
"""

from langchain_core.messages import SystemMessage, HumanMessage
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate

try:  # preferred, official integration package
    from langchain_ollama import ChatOllama
except ImportError:  # fallback for older setups
    from langchain_community.chat_models import ChatOllama

MODEL = "llama3"

# Tokens that mark the end of an assistant turn in the Llama 3 template.
# Ollama normally stops on <|eot_id|>; listing them explicitly (and also
# stopping on a new header) prevents the model from continuing the "chat".
STOP_TOKENS = [
    "<|eot_id|>",
    "<|end_of_text|>",
    "<|start_header_id|>",
    "\n<|start_header_id|>",
    "\n\n<|start_header_id|>",
    "user<|end_header_id|>",
]


def build_llm():
    return ChatOllama(
        model=MODEL,
        temperature=0.7,       # lower -> less rambling; 0 for determinism
        top_p=0.9,
        num_predict=1024,      # hard cap on generated tokens (no runaway loops)
        repeat_penalty=1.1,    # discourages verbatim repetition
        num_ctx=8192,          # large enough for the prompt + answer
        stop=STOP_TOKENS,
    )


def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Run one prompt and return exactly one complete assistant response."""
    llm = build_llm()

    # Chat model => roles are handled by the model's own template.
    # Never inject <|begin_of_text|>/<|start_header_id|>/<|eot_id|> by hand.
    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", "{system_prompt}"),
            ("human", "{user_prompt}"),
        ]
    )

    chain = prompt | llm | StrOutputParser()
    response = chain.invoke(
        {"system_prompt": system_prompt, "user_prompt": user_prompt}
    )

    # Defensive cleanup: drop anything the model may append after the turn ends.
    for marker in ("<|eot_id|>", "<|end_of_text|>", "<|start_header_id|>"):
        if marker in response:
            response = response.split(marker)[0]

    return response.strip()


# Equivalent, message-based version (no prompt template needed):
def run_agent_messages(user_prompt: str, system_prompt: str) -> str:
    llm = build_llm()
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]
    ai_message = llm.invoke(messages)
    return ai_message.content.strip()


if __name__ == "__main__":
    print(
        run_agent(
            "Explain in detail, step by step, how artificial intelligence models "
            "like yourself are built, including data, training, architecture, and "
            "deployment. Use examples and provide a comprehensive overview.",
            "You are a helpful assistant.",
        )
    )
