import os

from dotenv import find_dotenv, load_dotenv

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.memory import ConversationBufferMemory
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS

# Load environment variables from .env file
_ = load_dotenv(find_dotenv())

# ---------------------------------------------------------------------------
# Retrieval configuration
# ---------------------------------------------------------------------------
# OpenAI embeddings are normalized, and FAISS uses the Euclidean (L2) distance
# by default, so LangChain converts the distance into a relevance score in
# [0, 1] (1 = identical).  Documents scoring BELOW the threshold are dropped by
# the retriever, which means `get_relevant_documents(...)` returns [] when the
# question has nothing to do with the knowledge base (greetings, small talk,
# unrelated questions, ...).
SCORE_THRESHOLD = 0.75
TOP_K = 4

# The condense prompt must only REWRITE the follow-up question into a
# standalone question -- it must not try to answer it.
CONDENSE_QUESTION_TEMPLATE = """Given the following conversation and a follow up question, \
rephrase the follow up question to be a standalone question, in its original language.

Chat History:
{chat_history}
Follow Up Input: {question}
Standalone question:"""

# The QA prompt is explicitly restricted to the retrieved context so the model
# never invents an answer when the retriever returned nothing useful.
QA_TEMPLATE = """You are a helpful assistant that answers questions using ONLY the
information contained in the context below.

Rules:
- If the context does not contain the answer, reply exactly: "I don't know."
- If the question is a greeting, small talk, or otherwise unrelated to the context,
  reply exactly: "I don't know."
- Never use your own general knowledge and never make up an answer.

Context:
{context}

Question: {question}

Answer:"""

DOCS = [
    "Buildings are made out of brick",
    "Buildings are made out of wood",
    "Buildings are made out of stone",
    "Buildings are made out of atoms",
    "Buildings are made out of building materials",
    "Cars are made out of metal",
    "Cars are made out of plastic",
]

_QA_CACHE = {}


def get_retriever():
    """Build (once) and return the vector store retriever.

    The key fix: `search_type="similarity_score_threshold"` together with
    `score_threshold`.  A plain `vectorstore.as_retriever()` always returns the
    k nearest documents, no matter how irrelevant they are, which is why
    greetings / off-topic questions still produced documents.
    """
    if "retriever" not in _QA_CACHE:
        openai_api_key = os.getenv("OPENAI_API_KEY")
        if not openai_api_key:
            raise ValueError("OPENAI_API_KEY environment variable is not set")
        os.environ["OPENAI_API_KEY"] = openai_api_key

        embeddings = OpenAIEmbeddings()
        vectorstore = FAISS.from_texts(DOCS, embeddings)

        _QA_CACHE["retriever"] = vectorstore.as_retriever(
            search_type="similarity_score_threshold",
            search_kwargs={"score_threshold": SCORE_THRESHOLD, "k": TOP_K},
        )
    return _QA_CACHE["retriever"]


def get_qa_chain():
    """Build (once) and return the conversational retrieval chain."""
    if "qa" not in _QA_CACHE:
        openai_api_key = os.getenv("OPENAI_API_KEY")
        if not openai_api_key:
            raise ValueError("OPENAI_API_KEY environment variable is not set")
        os.environ["OPENAI_API_KEY"] = openai_api_key

        llm = ChatOpenAI(
            model_name="gpt-3.5-turbo",  # Name of the language model
            temperature=0,  # Parameter that controls the randomness of the responses
        )
        memory = ConversationBufferMemory(
            memory_key="chat_history",
            return_messages=True,
            output_key="answer",
        )
        _QA_CACHE["qa"] = ConversationalRetrievalChain.from_llm(
            llm,
            get_retriever(),
            condense_question_prompt=PromptTemplate.from_template(
                CONDENSE_QUESTION_TEMPLATE
            ),
            combine_docs_chain_kwargs={
                "prompt": PromptTemplate.from_template(QA_TEMPLATE)
            },
            memory=memory,
            return_source_documents=True,
        )
    return _QA_CACHE["qa"]


def run_agent(user_input):
    """Return the knowledge-base documents relevant to `user_input`.

    Returns an empty list when the query is not answered by the custom
    knowledge base (greetings, general questions, off-topic inputs).
    """
    retriever = get_retriever()
    qa = get_qa_chain()

    # IMPORTANT: query with the *actual* user input (the original code always
    # queried the hard-coded "what are cars made of?" and ignored the caller's
    # question entirely).
    query = (user_input or "").strip()
    if not query:
        return []

    documents = retriever.get_relevant_documents(query)
    if not documents:
        # Nothing above the similarity threshold -> not answerable from the
        # knowledge base, so don't call the LLM at all.
        return []

    result = qa({"question": query})
    answer = result.get("answer", "")

    # Secondary guard: even if some document barely passed the threshold, the
    # LLM may still say it does not know the answer.
    if "i don't know" in answer.lower() or "i do not know" in answer.lower():
        return []

    return documents


if __name__ == "__main__":
    user_input = "What are buildings made of?"
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result}")

    # Off-topic / greeting queries now yield an empty list
    print(run_agent("hello, how are you?"))  # -> []

OpenAI embeddings are unit-normalized, so for `"What are buildings made of?"` vs `"Buildings are made out of brick"` with cosine ≈ 0.85, the L2 distance is ≈ 0.55 and relevance ≈ 0.61 — **below 0.75**.
