"""
Retrieval-augmented QA over a small custom knowledge base.

Goal: `get_relevant_documents(query)` must return documents ONLY when the query
can be answered from the knowledge base, and an empty list for greetings,
small talk, or off-topic questions.

Fixes applied vs. the original snippet
--------------------------------------
1. The original code never used `user_input`: it hard-coded
   `query = "what are cars made of?"` and then ignored the chain result, so the
   function always answered the same question. The real query is now used.
2. The retriever was built with plain `as_retriever()` (no threshold at all), so
   similarity search ALWAYS returned the top-k documents - even for "hi" or
   "what is the weather like?". The retriever is now configured with
   `search_type="similarity_score_threshold"` so that documents below the
   threshold are dropped instead of being returned as noise.
3. `score_threshold` is expressed on LangChain's *relevance score* (clamped to
   [0, 1]), NOT on the raw cosine/L2 distance. With OpenAIEmbeddings +
   FAISS, unrelated text typically scores ~0.40-0.50 and on-topic text
   ~0.60-0.95, so `0.8` (the value many people copy/paste) filters out almost
   everything while `0.0` filters out nothing. A sane, tunable default is used.
4. A closed-book answer ("I don't know") or an empty/blank answer now also
   makes the function return an empty list, so the caller never receives
   off-topic documents.
5. Greetings / small talk are short-circuited before hitting the vector store,
   because short chit-chat sentences can still be *semantically* close enough to
   short documents to slip over a threshold.
6. `ConversationBufferMemory` is created with `return_messages=False` (the
   prompt templates format `{chat_history}` as plain text) and an explicit
   `output_key="answer"` (required when `return_source_documents=True`).
7. `retriever.get_relevant_documents(...)` raises `ValueError` in some LangChain
   versions when the threshold rejects every candidate - that is now handled
   and treated as "nothing relevant".
"""

import os
import re
from typing import List

from dotenv import load_dotenv, find_dotenv

from langchain.chat_models import ChatOpenAI
from langchain.chains import ConversationalRetrievalChain
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.memory import ConversationBufferMemory
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS

# Load environment variables from .env file
_ = load_dotenv(find_dotenv())


# --------------------------------------------------------------------------- #
# Knowledge base + tuned retrieval settings
# --------------------------------------------------------------------------- #
KNOWLEDGE_BASE = [
    "Buildings are made out of brick",
    "Buildings are made out of wood",
    "Buildings are made out of stone",
    "Buildings are made out of atoms",
    "Buildings are made out of building materials",
    "Cars are made out of metal",
    "Cars are made out of plastic",
]

# Relevance score threshold for FAISS + OpenAIEmbeddings (score is normalised to
# [0, 1], where 1.0 == identical vectors). Tune it for your own corpus:
#   * too high  -> valid answers are filtered out (empty list for real questions)
#   * too low   -> greetings/off-topic questions still return documents
DEFAULT_SCORE_THRESHOLD = 0.55
DEFAULT_K = 4

# Pure small talk can never be answered from a document store.
SMALL_TALK_RE = re.compile(
    r"^\W*("
    r"hi|hii+|hello|hey|hiya|yo|sup|what'?s up|"
    r"good (morning|afternoon|evening|night)|"
    r"how are you(\s+doing)?|how'?s it going|"
    r"who are you|what are you|what can you do|"
    r"thanks|thank you|thx|cheers|"
    r"bye|goodbye|see you|ok|okay|great|cool"
    r")\W*$",
    re.IGNORECASE,
)

NOT_FOUND_MARKERS = (
    "i don't know",
    "i do not know",
    "no information",
    "not in the context",
    "context does not",
)


# --------------------------------------------------------------------------- #
# Prompts
# --------------------------------------------------------------------------- #
# Used only to turn "and cars?" into a standalone question using the history.
CONDENSE_TEMPLATE = """Given the following conversation and a follow up question, \
rephrase the follow up question to be a standalone question.

Chat History:
{chat_history}
Follow Up Input: {question}
Standalone question:"""
CONDENSE_QUESTION_PROMPT = PromptTemplate.from_template(CONDENSE_TEMPLATE)

# Used to actually answer the question, strictly grounded in the retrieved
# context, so the LLM cannot hallucinate an off-topic answer.
QA_TEMPLATE = """This is a conversation with a human. Answer the question using \
ONLY the context below and the knowledge it contains.

If the context does not contain the answer, reply exactly: I don't know.
Never try to make up an answer.

Context:
{context}

Chat History:
{chat_history}

Question: {question}
Helpful answer:"""
QA_PROMPT = PromptTemplate(
    template=QA_TEMPLATE,
    input_variables=["context", "chat_history", "question"],
)


# --------------------------------------------------------------------------- #
# Building blocks
# --------------------------------------------------------------------------- #
def get_llm() -> ChatOpenAI:
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key:
        os.environ["OPENAI_API_KEY"] = openai_api_key
    return ChatOpenAI(model_name="gpt-3.5-turbo", temperature=0)


def build_retriever(
    score_threshold: float = DEFAULT_SCORE_THRESHOLD,
    k: int = DEFAULT_K,
):
    """FAISS retriever that returns [] instead of noise when nothing is relevant."""
    embeddings = OpenAIEmbeddings()
    vectorstore = FAISS.from_texts(KNOWLEDGE_BASE, embeddings)

    # <-- the actual fix: threshold-based similarity search instead of a plain
    #     top-k search which always returns something.
    return vectorstore.as_retriever(
        search_type="similarity_score_threshold",
        search_kwargs={"score_threshold": score_threshold, "k": k},
    )


def build_qa_chain(
    score_threshold: float = DEFAULT_SCORE_THRESHOLD,
    k: int = DEFAULT_K,
):
    llm = get_llm()
    retriever = build_retriever(score_threshold=score_threshold, k=k)

    memory = ConversationBufferMemory(
        memory_key="chat_history",
        return_messages=False,   # the prompts expect {chat_history} as text
        output_key="answer",     # needed when return_source_documents=True
    )

    qa = ConversationalRetrievalChain.from_llm(
        llm=llm,
        retriever=retriever,
        condense_question_prompt=CONDENSE_QUESTION_PROMPT,
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
        memory=memory,
        return_source_documents=True,
    )
    return qa, retriever


# --------------------------------------------------------------------------- #
# Public helpers
# --------------------------------------------------------------------------- #
def is_small_talk(query: str) -> bool:
    return bool(SMALL_TALK_RE.match((query or "").strip()))


def get_relevant_documents(
    query: str,
    retriever,
    score_threshold: float = DEFAULT_SCORE_THRESHOLD,
) -> List:
    """Return the documents that actually answer `query`, or [] if none do."""
    if not query or not query.strip():
        return []

    if is_small_talk(query):
        return []

    try:
        docs = retriever.get_relevant_documents(query.strip())
    except ValueError:
        # Every candidate scored below `score_threshold` in some versions.
        docs = []

    # Belt and braces: drop obviously empty documents.
    return [doc for doc in docs if getattr(doc, "page_content", "").strip()]


def run_agent(
    user_input: str,
    score_threshold: float = DEFAULT_SCORE_THRESHOLD,
    k: int = DEFAULT_K,
):
    """Answer `user_input` from the knowledge base, or return [] if impossible."""
    qa, retriever = build_qa_chain(score_threshold=score_threshold, k=k)

    docs = get_relevant_documents(user_input, retriever, score_threshold)
    if not docs:
        # Greeting, small talk or off-topic question -> nothing from the KB.
        return []

    result = qa({"question": user_input})

    answer = (result.get("answer") or "").strip()
    if not answer or any(m in answer.lower() for m in NOT_FOUND_MARKERS):
        # The LLM itself decided the context does not answer the question.
        return []

    return docs


if __name__ == "__main__":
    demo_queries = [
        "What are buildings made of?",   # -> building documents
        "what are cars made of?",        # -> car documents
        "hello, how are you?",           # -> []
        "Who won the 2022 world cup?",   # -> []
    ]

    for user_input in demo_queries:
        result = run_agent(user_input)
        print("\n=== User Query Result ===")
        print(f"Question: {user_input}")
        if not result:
            print("Answer: [] (nothing relevant in the knowledge base)")
        else:
            print("Answer (retrieved documents):")
            for doc in result:
                print(f"  - {doc.page_content}")
