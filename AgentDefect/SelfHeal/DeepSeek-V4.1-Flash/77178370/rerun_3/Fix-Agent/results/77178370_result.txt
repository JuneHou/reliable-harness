import os
from functools import lru_cache

from dotenv import load_dotenv, find_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import FAISS
from langchain.chains import ConversationalRetrievalChain
from langchain.memory import ConversationBufferMemory
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate

# Load environment variables from .env file
_ = load_dotenv(find_dotenv())

custom_template = """
This is conversation with a human. Answer the questions you get based on the knowledge you have.
If you don't know the answer, just say that you don't, don't try to make up an answer.
Chat History:
{chat_history}
Follow Up Input: {question}
"""
CUSTOM_QUESTION_PROMPT = PromptTemplate.from_template(custom_template)

DOCS = [
    "Buildings are made out of brick",
    "Buildings are made out of wood",
    "Buildings are made out of stone",
    "Buildings are made out of atoms",
    "Buildings are made out of building materials",
    "Cars are made out of metal",
    "Cars are made out of plastic",
]

# ---------------------------------------------------------------------------
# The important setting.
#
# `vectorstore.as_retriever()` (the default) is a plain "similarity" search:
# it ALWAYS returns the k nearest neighbours, no matter how irrelevant they
# are. That is why greetings / off-topic questions still came back with
# documents, and why a similarity threshold "did not work" - it was never
# applied.
#
# Fix: use search_type="similarity_score_threshold", which calls
# FAISS.similarity_search_with_relevance_scores() and drops every document
# whose relevance score is below `score_threshold`.
#
# FAISS reports relevance as 1 - distance / sqrt(2) for L2-normalised
# vectors, i.e. the cosine similarity between query and document, in [0, 1].
#   * on-topic query ("what are cars made of?")  -> ~0.6 - 0.9
#   * greeting / general question ("hi", "how are you") -> ~0.1 - 0.35
# So 0.5 splits the two populations. Raise it (0.6/0.7) to be stricter,
# lower it (0.4) if real knowledge-base questions are being dropped.
# ---------------------------------------------------------------------------
RELEVANCE_THRESHOLD = 0.5
TOP_K = 4

FALLBACK_ANSWER = (
    "I don't know. That question does not appear to be covered by my knowledge base."
)


@lru_cache(maxsize=1)
def build_retriever():
    """Build the FAISS store once and return a threshold-based retriever."""
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key:
        os.environ["OPENAI_API_KEY"] = openai_api_key

    embeddings = OpenAIEmbeddings()

    # normalize_L2=True is required: the relevance score of a FAISS index is
    # only meaningful (and bounded in [0, 1]) for L2-normalised vectors.
    vectorstore = FAISS.from_texts(DOCS, embeddings, normalize_L2=True)

    return vectorstore.as_retriever(
        search_type="similarity_score_threshold",
        search_kwargs={"k": TOP_K, "score_threshold": RELEVANCE_THRESHOLD},
    )


@lru_cache(maxsize=1)
def build_qa():
    """Conversational chain bound to the same threshold-based retriever."""
    llm = ChatOpenAI(
        model_name="gpt-3.5-turbo",  # Name of the language model
        temperature=0,  # Controls the randomness of the generated responses
    )
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    return ConversationalRetrievalChain.from_llm(
        llm,
        build_retriever(),
        condense_question_prompt=CUSTOM_QUESTION_PROMPT,
        memory=memory,
        return_source_documents=True,
    )


def _retrieve(retriever, question: str):
    """Invoke the retriever, supporting both old and new LangChain APIs."""
    invoke = getattr(retriever, "invoke", None)
    if callable(invoke):
        return invoke(question) or []
    return retriever.get_relevant_documents(question) or []


def run_agent(user_input):
    """
    Return the documents of the knowledge base that actually answer
    `user_input`, and an EMPTY LIST when the question is a greeting,
    a general question or otherwise off-topic.
    """
    retriever = build_retriever()

    # Use the real user question (the original code hard-coded
    # "what are cars made of?" here, so every call returned car docs).
    documents = _retrieve(retriever, user_input)

    # Belt and braces: never let empty/irrelevant results through.
    documents = [
        doc
        for doc in documents
        if doc is not None and getattr(doc, "page_content", "").strip()
    ]
    return documents


def answer_question(user_input):
    """Same filtering logic, but returns the final answer instead of docs."""
    qa = build_qa()
    result = qa({"question": user_input})

    source_documents = result.get("source_documents") or []
    if not source_documents:
        # Nothing cleared the relevance threshold -> the chain would happily
        # hallucinate from an empty/garbage context, so short-circuit.
        return {"answer": FALLBACK_ANSWER, "source_documents": []}

    return {
        "answer": result.get("answer", FALLBACK_ANSWER),
        "source_documents": source_documents,
    }


if __name__ == "__main__":
    user_input = "What are buildings made of?"
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Documents: {[doc.page_content for doc in result]}")
    print(f"Answer: {answer_question(user_input)['answer']}")

    # Off-topic inputs now yield an empty list instead of documents.
    for off_topic in ["hi", "how are you?", "what is the capital of France?"]:
        print(f"{off_topic!r} -> {[d.page_content for d in run_agent(off_topic)]}")
