"""Conversational retrieval over a small custom knowledge base.

`get_relevant_documents` returns documents ONLY when the question can be
answered from the knowledge base; greetings, chit-chat, general questions and
other off-topic inputs return an empty list (`[]`).

Why the original snippet never filtered anything
------------------------------------------------
1. `vectorstore.as_retriever()` was created with the *default*
   `search_type="similarity"`. That always returns the `k` nearest neighbours
   and never cuts anything off, so no similarity threshold was ever applied.
2. The threshold itself has to be checked on the *relevance score*
   (`similarity_search_with_relevance_scores`), because
   `similarity_search_with_score` returns a raw distance (smaller = closer);
   comparing a distance with a "similarity threshold" silently keeps
   everything. Embedding similarity is also compressed (unrelated sentences
   often sit around 0.7), so the threshold has to be tuned and combined with a
   semantic guard.
3. `result = qa({"question": query})` was thrown away and the return value was
   taken from `retriever.get_relevant_documents(query)` instead, so the
   conversational chain (and its prompt) had no influence on the result.
4. The prompt told the model to *"answer the questions you get based on the
   knowledge you have"* - i.e. it explicitly invited answers from general
   knowledge for greetings/off-topic questions.
5. `ConversationBufferMemory` was built without `output_key` although the
   chain returns several output keys (`answer`, `source_documents`).

What this version does
----------------------
* `KnowledgeBaseRetriever` implements `similarity_score_threshold` semantics:
  only documents whose relevance score >= `SCORE_THRESHOLD` are kept, and an
  empty list is returned when nothing qualifies.
* An extra LLM "can this be answered from the KB?" guard rejects general /
  off-topic questions that happen to have a high embedding similarity
  (greetings, "what is the capital of France?", ...).
* The chain answers with a strict "answer only from the retrieved context"
  prompt, and the caller returns `[]` when the model says it does not know.
"""

import os
from typing import Any, List, Optional

from dotenv import load_dotenv, find_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import FAISS
from langchain.chains import ConversationalRetrievalChain
from langchain.memory import ConversationBufferMemory
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate

# Retriever/Document/Manager classes moved around between langchain versions.
try:  # langchain >= 0.1 / langchain-core
    from langchain_core.callbacks import CallbackManagerForRetrieverRun
    from langchain_core.documents import Document
    from langchain_core.retrievers import BaseRetriever
except ImportError:  # langchain < 0.1
    from langchain.callbacks.manager import CallbackManagerForRetrieverRun
    from langchain.schema import BaseRetriever, Document


# Load environment variables from .env file
_ = load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY", "")

# ---------------------------------------------------------------------------
# Tunables: a document is "in the knowledge base" only above this score.
# Tune it once against a handful of on-topic / off-topic questions.
# ---------------------------------------------------------------------------
SCORE_THRESHOLD = 0.80
TOP_K = 3
NO_ANSWER = "I don't know."


CONDENSE_QUESTION_PROMPT = PromptTemplate.from_template(
    """Given the following conversation and a follow up question, rephrase the
follow up question to be a standalone question, in its original language.

Chat History:
{chat_history}
Follow Up Input: {question}
Standalone question:"""
)


QA_PROMPT = PromptTemplate.from_template(
    """You are an assistant for question-answering over a private knowledge base.
Answer the question using ONLY the pieces of context given below.
If the context does not contain the answer, reply exactly with "I don't know."
Do not use any outside knowledge and do not make anything up.

Context:
{context}

Chat History:
{chat_history}

Question: {question}
Helpful answer:"""
)


ANSWERABILITY_PROMPT = PromptTemplate.from_template(
    """You decide whether a question can be answered from a knowledge base.

Knowledge base excerpts:
---
{context}
---

Question: {question}

Does the knowledge base contain the information needed to answer the question?
Ignore greetings, small talk and general-knowledge questions: for those the
answer is "no". Answer with a single word: yes or no.
Answer:"""
)


def _is_no_answer(answer: str) -> bool:
    """True when the model did not answer from the knowledge base."""
    normalized = answer.strip().lower().replace("\u2019", "'")
    if not normalized:
        return True
    return (
        normalized.startswith("i don't know")
        or normalized.startswith("i do not know")
        or normalized.startswith("no answer")
    )


def _is_affirmative(verdict: str) -> bool:
    return verdict.strip().lower().lstrip("\"'` ").startswith("yes")


class KnowledgeBaseRetriever(BaseRetriever):
    """Retriever that returns [] instead of loosely related documents.

    It reproduces `search_type="similarity_score_threshold"` (explicitly
    filtering on the *relevance score*) and adds a semantic answerability
    guard so greetings / off-topic questions never reach the LLM.
    """

    vectorstore: Any
    llm: Any
    k: int = TOP_K
    score_threshold: float = SCORE_THRESHOLD

    def _get_relevant_documents(
        self,
        query: str,
        *,
        run_manager: Optional[CallbackManagerForRetrieverRun] = None,
    ) -> List[Document]:
        # 1) Score-based filtering: relevance score is in [0, 1], higher = better.
        scored = self.vectorstore.similarity_search_with_relevance_scores(
            query, k=self.k
        )
        candidates = [
            doc
            for doc, score in scored
            if score is not None and score >= self.score_threshold
        ]
        if not candidates:
            return []

        # 2) Semantic guard: the excerpts must actually answer the question.
        context = "\n".join(doc.page_content for doc in candidates)
        verdict = self.llm.predict(
            ANSWERABILITY_PROMPT.format(context=context, question=query)
        )
        if not _is_affirmative(verdict):
            return []

        return candidates


def build_vectorstore() -> FAISS:
    embeddings = OpenAIEmbeddings()
    docs = [
        "Buildings are made out of brick",
        "Buildings are made out of wood",
        "Buildings are made out of stone",
        "Buildings are made out of atoms",
        "Buildings are made out of building materials",
        "Cars are made out of metal",
        "Cars are made out of plastic",
    ]
    return FAISS.from_texts(docs, embeddings)


def build_retriever(vectorstore: FAISS, llm: ChatOpenAI) -> BaseRetriever:
    # Framework-native equivalent (no answerability guard):
    #   vectorstore.as_retriever(
    #       search_type="similarity_score_threshold",
    #       search_kwargs={"score_threshold": SCORE_THRESHOLD, "k": TOP_K},
    #   )
    return KnowledgeBaseRetriever(
        vectorstore=vectorstore,
        llm=llm,
        k=TOP_K,
        score_threshold=SCORE_THRESHOLD,
    )


def build_qa_chain() -> ConversationalRetrievalChain:
    llm = ChatOpenAI(
        model_name="gpt-3.5-turbo",  # Name of the language model
        temperature=0,  # Deterministic answers
    )
    vectorstore = build_vectorstore()
    retriever = build_retriever(vectorstore, llm)

    memory = ConversationBufferMemory(
        memory_key="chat_history",
        return_messages=True,
        output_key="answer",  # the chain returns more than one output key
    )

    return ConversationalRetrievalChain.from_llm(
        llm,
        retriever,
        condense_question_prompt=CONDENSE_QUESTION_PROMPT,
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
        memory=memory,
        return_source_documents=True,
    )


def run_agent(
    user_input: str, qa_chain: Optional[ConversationalRetrievalChain] = None
) -> List[Document]:
    """Return the KB documents that answer `user_input`, or [] if none do."""
    qa_chain = qa_chain or build_qa_chain()

    # `get_relevant_documents` is now threshold-filtered + guarded, so it
    # already returns [] for greetings / general / off-topic questions.
    relevant_docs = qa_chain.retriever.get_relevant_documents(user_input)
    if not relevant_docs:
        return []

    result = qa_chain({"question": user_input})

    # The model answered "I don't know." -> not answered from the KB.
    if _is_no_answer(result.get("answer", "")):
        return []

    return relevant_docs


if __name__ == "__main__":
    qa_chain = build_qa_chain()
    demo_queries = [
        "What are buildings made of?",
        "what are cars made of?",
        "hello!",
        "How are you doing today?",
        "What is the capital of France?",
    ]

    for user_input in demo_queries:
        result = run_agent(user_input, qa_chain)
        print("\n=== User Query Result ===")
        print(f"Question: {user_input}")
        print(f"Answer: {[doc.page_content for doc in result]}")
