# --------------------------------------------------------------------------- #
# Retriever that returns [] for anything not answered by the knowledge base
# --------------------------------------------------------------------------- #
class KnowledgeBaseRetriever(BaseRetriever):
    """Similarity filtering + LLM answerability check.

    Documents are returned ONLY when the query is answered by the knowledge
    base; otherwise an empty list is returned.
    """

    vectorstore: Any
    llm: Any = None
    k: int = DEFAULT_K
    score_threshold: float = DEFAULT_SCORE_THRESHOLD
    use_llm_relevance_check: bool = True

    def _get_relevant_documents(
        self,
        query: str,
        *,
        run_manager: Optional[CallbackManagerForRetrieverRun] = None,
    ) -> List[Document]:
        # 0) sanity check on the input
        if not query or not query.strip():
            return []

        # 1) greetings / small talk -> nothing
        if GREETING_PATTERN.match(query.strip()):
            return []

        # 2) embedding similarity with an explicit relevance score
        #    (score is normalised to [0, 1] by FAISS)
        scored = self.vectorstore.similarity_search_with_relevance_scores(
            query, k=self.k
        )
        docs = [doc for doc, score in scored if score >= self.score_threshold]
        if not docs:
            return []

        # 3) let the LLM decide whether the retrieved context really answers
        #    the question (embedding scores alone are not discriminative)
        if self.use_llm_relevance_check and self.llm is not None:
            context = "\n".join(f"- {doc.page_content}" for doc in docs)
            verdict = self.llm.predict(
                RELEVANCE_GRADER_PROMPT.format(question=query, context=context)
            )
            if not verdict.strip().upper().startswith("YES"):
                return []

        # 4) de-duplicate while preserving order
        unique: List[Document] = []
        seen = set()
        for doc in docs:
            if doc.page_content not in seen:
                seen.add(doc.page_content)
                unique.append(doc)
        return unique

def build_llm() -> ChatOpenAI:
    return ChatOpenAI(
        model_name="gpt-3.5-turbo",
        temperature=0,  # deterministic output
    )


def build_retriever(llm: ChatOpenAI) -> KnowledgeBaseRetriever:
    embeddings = OpenAIEmbeddings()
    vectorstore = FAISS.from_texts(KNOWLEDGE_BASE, embeddings, normalize_L2=True)

    # NOTE: the built-in equivalent of the filtering above is
    #   vectorstore.as_retriever(
    #       search_type="similarity_score_threshold",
    #       search_kwargs={"score_threshold": 0.5, "k": 4},
    #   )
    # but it raises/returns top-k when scores are all high, which is why the
    # extra LLM answerability check is added here.
    return KnowledgeBaseRetriever(
        vectorstore=vectorstore,
        llm=llm,
        k=DEFAULT_K,
        score_threshold=DEFAULT_SCORE_THRESHOLD,
        use_llm_relevance_check=True,
    )


def build_qa_chain(
    llm: ChatOpenAI,
    retriever: KnowledgeBaseRetriever,
    memory: ConversationBufferMemory,
) -> ConversationalRetrievalChain:
    return ConversationalRetrievalChain.from_llm(
        llm=llm,
        retriever=retriever,
        condense_question_prompt=CONDENSE_QUESTION_PROMPT,
        combine_docs_chain_kwargs={
            "prompt": QA_PROMPT.partial(no_answer=NO_ANSWER),
        },
        memory=memory,
        return_source_documents=True,
    )

# --------------------------------------------------------------------------- #
def run_agent(user_input: str):
    """Return the relevant knowledge-base documents for `user_input`.

    An empty list is returned for greetings, general questions and any other
    input that the custom knowledge base does not answer.
    """
    llm = build_llm()
    retriever = build_retriever(llm)

    # The answer, generated strictly from the retrieved context (or NO_ANSWER)
    memory = ConversationBufferMemory(
        memory_key="chat_history", return_messages=True, output_key="answer"
    )
    qa = build_qa_chain(llm, retriever, memory)
    answer = qa({"question": user_input})["answer"]

    # Documents only when the knowledge base actually answered the question
    if str(answer).strip().upper().startswith(NO_ANSWER):
        return []

    return retriever.get_relevant_documents(user_input)
