from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_verbose, set_debug
from langchain_core.tracers import ConsoleCallbackHandler
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# `set_verbose(True)` is the *legacy* verbose switch: it is picked up by old-style
# (Agent/Chain with `verbose=True`) objects, but it does NOT make an LCEL
# pipeline (`prompt | model | output_parser`) print its intermediate steps.
# The current way to get intermediate-step logging for any Runnable/LCEL chain is
# debug mode (global) and/or attaching a console callback handler to the run.
set_verbose(True)   # kept for any legacy components, harmless
set_debug(True)     # current global switch: logs every runnable step/token counts


def run_agent(topic: str) -> str:
    model = ChatOpenAI(openai_api_key=api_key)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)

    chain = prompt | model | output_parser

    # Attach the console tracer explicitly (via the run config) so the prompt,
    # the LLM request/response and the parser step are all printed, instead of
    # relying on the global verbose flag that LCEL ignores.
    config = {"callbacks": [ConsoleCallbackHandler()]}
    return chain.invoke({"topic": topic}, config=config)


if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
