from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser

# The global `set_verbose(True)` flag (langchain.globals) only affects legacy
# chain classes, and `verbose=True` is not a valid argument for `.invoke()` on
# LCEL runnables. The supported way to see every intermediate step of a
# Runnable/LCEL chain is to attach a callback handler -- such as
# ConsoleCallbackHandler -- to the chain run itself through the `config` dict.
from langchain_core.tracers import ConsoleCallbackHandler

from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def run_agent(topic: str, verbose: bool = True) -> str:
    model = ChatOpenAI(openai_api_key=api_key)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)
    chain = prompt | model | output_parser

    # Attach the console tracer to this specific run so each step
    # (prompt formatting, chat completion, output parsing, and the final
    # result) is logged to stdout.
    run_config = {"callbacks": [ConsoleCallbackHandler()]} if verbose else {}

    return chain.invoke({"topic": topic}, config=run_config)


if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
