"""Enable verbose logging of intermediate steps for a LangChain LCEL chain.

Why the original code didn't print anything:
    `set_verbose(True)` (and `verbose=True`) only affect legacy `Chain`
    objects, which expose a `verbose` attribute that is read when the chain
    is executed. An LCEL `RunnableSequence` (`prompt | model | parser`) does
    not look at that global flag at all, so nothing is logged.

The current, correct way to see intermediate steps for an LCEL chain is to
attach a tracer/callback handler through the `config` argument of
`invoke` (or bake it into the chain with `with_config`):

    from langchain_core.tracers import ConsoleCallbackHandler
    chain.invoke(inputs, config={"callbacks": [ConsoleCallbackHandler()]})

`langchain.globals.set_debug(True)` is the global equivalent (it enables the
console tracer for every runnable) and is the recommended replacement for
the deprecated global `set_verbose` when debugging LCEL.

Imports are also moved to the current, non-deprecated modules
(`langchain_openai`, `langchain_core`).
"""

from dotenv import load_dotenv
import os

from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.tracers import ConsoleCallbackHandler
from langchain_openai import ChatOpenAI

# Optional global switch: makes *every* runnable log its steps (equivalent to
# verbose=True for legacy chains, but works for LCEL). Kept off by default in
# favour of the explicit per-call callback handler below, which is the most
# reliable and scoped way to inspect intermediate steps.
# from langchain.globals import set_debug
# set_debug(True)

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def run_agent(topic: str) -> str:
    model = ChatOpenAI(openai_api_key=api_key, model="gpt-4o-mini")
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)

    chain = prompt | model | output_parser

    # Correct current way to get verbose intermediate-step output for LCEL:
    # pass a callback handler via the runnable config.
    # `with_config` permanently attaches the config to the chain; either form
    # works, this one keeps the handler next to the chain definition.
    chain = chain.with_config({"callbacks": [ConsoleCallbackHandler()]})

    return chain.invoke({"topic": topic})


if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
