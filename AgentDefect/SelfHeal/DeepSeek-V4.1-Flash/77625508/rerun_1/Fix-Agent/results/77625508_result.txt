"""
Fixed example: enable verbose / intermediate-step logging for an LCEL chain.

Why the original code printed nothing extra:
  * `set_verbose(True)` (and `Chain(verbose=True)` / `chain.verbose = True`) only
    affects the *legacy* `Chain` interface. LCEL chains (`prompt | model | parser`)
    are `Runnable`s, not `Chain`s, so the verbose flag never reaches them and no
    intermediate steps are logged.
  * The current way to log every intermediate step of an LCEL chain is either
      - `set_debug(True)`  -> global high-level logging of all callback-aware
        components (chain inputs/outputs, model inputs/outputs, etc.), or
      - attaching a callback handler to the run, e.g.
        `chain.invoke(inputs, config={"callbacks": [ConsoleCallbackHandler()]})`,
        which prints each step as it runs (best for a single, targeted chain).
"""

import os

from dotenv import load_dotenv

from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.tracers.stdout import ConsoleCallbackHandler

# `set_debug` lives in langchain_core.globals (re-exported from langchain.globals).
# Keep a fallback so the snippet works across LangChain versions.
try:
    from langchain_core.globals import set_debug
except ImportError:  # pragma: no cover - older installs
    from langchain.globals import set_debug

try:
    from langchain_openai import ChatOpenAI  # current package (langchain-openai)
except ImportError:  # pragma: no cover - older installs
    from langchain.chat_models import ChatOpenAI


load_dotenv()

api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Global debug flag: prints the inputs/outputs of every callback-aware
# component (RunnableSequence, ChatPromptTemplate formatting, ChatOpenAI, parser).
# This is what actually surfaces intermediate steps for LCEL chains.
set_debug(True)


def run_agent(topic: str) -> str:
    model = ChatOpenAI(openai_api_key=api_key, model="gpt-4o-mini", temperature=0)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)

    chain = prompt | model | output_parser

    # Alternatively / additionally: attach a console tracer to just this call so
    # each step (prompt -> LLM result -> parsed string) is printed verbosely.
    result = chain.invoke(
        {"topic": topic},
        config={"callbacks": [ConsoleCallbackHandler()]},
    )

    # LangSmith tracing (optional, most robust way to inspect LCEL runs):
    #   os.environ["LANGCHAIN_TRACING_V2"] = "true"
    #   os.environ["LANGCHAIN_API_KEY"] = "..."
    #   os.environ["LANGCHAIN_PROJECT"] = "my-project"
    return result


if __name__ == "__main__":
    output = run_agent("ice cream")
    print("Generated output:", output)
