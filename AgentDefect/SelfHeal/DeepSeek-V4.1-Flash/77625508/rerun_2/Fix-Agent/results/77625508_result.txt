"""Enable verbose / debug logging of intermediate steps for a LangChain LCEL chain.

Why the original code printed nothing extra
-------------------------------------------
1. ``set_verbose(True)`` only turns on verbose output for *legacy* ``Chain``
   subclasses (``LLMChain``, ``SequentialChain``, agents, ...).  A chain built
   with the ``|`` operator is a ``RunnableSequence`` from ``langchain_core``,
   and runnables do **not** consult that global flag -> no extra output.
2. The same is true for ``verbose=True`` passed to a chain/LLM: for LCEL the
   flag is not propagated down to the runnables being executed.
3. LangChain reads ``LANGCHAIN_VERBOSE`` / ``LANGCHAIN_DEBUG`` /
   ``LANGCHAIN_TRACING_V2`` at **import** time, so setting them *after*
   ``import langchain`` (e.g. via ``load_dotenv()`` placed below the imports)
   has no effect either.

The correct, current way for LCEL
---------------------------------
* Global switch: ``set_debug(True)`` (from ``langchain_core.globals`` /
  ``langchain.globals``).  Debug mode logs the input and output of *every*
  runnable in the graph - prompt, model and output parser included.
* Per-call switch: attach a callback handler explicitly, e.g.
  ``chain.invoke(inputs, config={"callbacks": [ConsoleCallbackHandler()]})``.
* For rich traces (latency, token usage, full nesting) use LangSmith:
  ``LANGCHAIN_TRACING_V2=true`` + ``LANGCHAIN_API_KEY=...`` in the env /
  ``.env`` file *before* importing langchain.
"""

import os

from dotenv import load_dotenv

# Load .env FIRST so anything LangChain reads at import time is available.
load_dotenv()

api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# --- imports (modern, non-deprecated module paths, with fallbacks) ----------
try:  # primary, current package
    from langchain_openai import ChatOpenAI
except ImportError:  # pragma: no cover - older langchain
    from langchain.chat_models import ChatOpenAI

from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate

# `set_debug` is the flag runnables actually look at.
try:
    from langchain_core.globals import set_debug
except ImportError:  # pragma: no cover
    from langchain.globals import set_debug

# A handler that prints every step (falls back to StdOutCallbackHandler).
try:
    from langchain_core.tracers.stdout import ConsoleCallbackHandler
except ImportError:  # pragma: no cover
    try:
        from langchain.callbacks.tracers import ConsoleCallbackHandler
    except ImportError:
        from langchain_core.callbacks import StdOutCallbackHandler as ConsoleCallbackHandler


def run_agent(topic: str) -> str:
    model = ChatOpenAI(openai_api_key=api_key, temperature=0)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)

    chain = prompt | model | output_parser

    # `config={"callbacks": [...]}` guarantees per-step logging for LCEL,
    # independently of any global flag.
    return chain.invoke(
        {"topic": topic},
        config={"callbacks": [ConsoleCallbackHandler()]},
    )


if __name__ == "__main__":
    # Global debug mode: prints START/END + inputs/outputs of each runnable.
    # (For an env-var-only setup use LANGCHAIN_DEBUG=true *before* importing
    # langchain instead of calling this function.)
    set_debug(True)

    result = run_agent("ice cream")
    print("Generated output:", result)
