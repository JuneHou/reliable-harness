import os
from langchain.llms import OpenAI
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()


def _dedupe(docs, key_fn=lambda d: d.page_content):
    """Keep only the first occurrence of each logically identical document.

    Chroma returns *neighbours*, not *distinct documents*: when a PDF page or a
    text chunk has near-duplicate copies in the index (overlapping splits,
    repeated headers/footers, pages loaded twice, etc.) the k-NN search
    happily returns the same text many times.  De-duplicating by content is
    what actually guarantees "unique" documents.
    """
    seen = set()
    unique_docs = []
    for doc in docs:
        key = key_fn(doc)
        if key in seen:
            continue
        seen.add(key)
        unique_docs.append(doc)
    return unique_docs


def build_vectorstore(pdf_path="paper2.pdf"):
    loader = UnstructuredFileLoader(pdf_path)
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    # Remove duplicate chunks *before* indexing so the same text is not stored
    # several times in the first place (overlap + repeated pages cause this).
    all_splits = _dedupe(all_splits)

    vectorstore = Chroma.from_documents(
        documents=all_splits, embedding=OpenAIEmbeddings()
    )
    return vectorstore, len(all_splits)


def run_agent(query, k=6, initial_fetch_k=50):
    """Return exactly `k` distinct documents that are relevant to `query`."""
    vectorstore, n_docs = build_vectorstore()

    # MMR ("max marginal relevance") first retrieves a large candidate pool
    # (fetch_k) and then picks chunks that are relevant *and* different from
    # each other, instead of returning near-identical neighbours like plain
    # "similarity" search does.
    fetch_k = min(max(initial_fetch_k, k), n_docs)
    retrieved_docs = []

    while True:
        retriever = vectorstore.as_retriever(
            search_type="mmr",
            search_kwargs={
                "k": k,
                "fetch_k": fetch_k,
                "lambda_mult": 0.5,  # 0 = max diversity, 1 = max similarity
            },
        )
        retrieved_docs = _dedupe(retriever.get_relevant_documents(query))

        # If de-duplication left us with fewer than k documents, widen the
        # candidate pool (and ultimately fall back to the whole index) so we
        # can still fill the result list with *distinct* chunks.
        if len(retrieved_docs) >= k or fetch_k >= n_docs:
            break
        fetch_k = min(fetch_k * 2, n_docs)

    return retrieved_docs[:k]


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)

    print(f"Retrieved {len(result)} unique documents")
    for i, doc in enumerate(result, start=1):
        start_index = doc.metadata.get("start_index")
        print(f"\n--- Document {i} (start_index={start_index}) ---")
        print(doc.page_content)
