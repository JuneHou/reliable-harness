import os

import dotenv
from langchain.llms import OpenAI
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma

dotenv.load_dotenv()


def run_agent(query):
    loader = UnstructuredFileLoader('paper2.pdf')
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    vectorstore = Chroma.from_documents(
        documents=all_splits, embedding=OpenAIEmbeddings()
    )

    # Why the old code printed the same document several times:
    #   search_type="similarity" just returns the k nearest neighbours of the query.
    #   Because RecursiveCharacterTextSplitter uses chunk_overlap=200, the chunks of
    #   the paper are almost identical to each other (the same paragraphs repeatedly
    #   appear inside consecutive chunks). All of those chunks have nearly the same
    #   embedding, so the top-6 nearest neighbours are often 6 copies/overlaps of the
    #   exact same passage and the answer "What is Toggle?" comes back 6 times.
    #
    # Fix: use MMR (maximal marginal relevance) instead of plain similarity search.
    #   MMR first fetches a larger candidate pool (fetch_k) and then iteratively picks
    #   the documents that are relevant to the query *and* dissimilar from each other,
    #   which yields k distinct passages.
    retriever = vectorstore.as_retriever(
        search_type="mmr",
        search_kwargs={"k": 6, "fetch_k": 50, "lambda_mult": 0.5},
    )

    retrieved_docs = retriever.get_relevant_documents(query)

    # Safety net: if the index still contains duplicated chunk texts, drop the
    # duplicates so that at most 6 truly unique documents are returned.
    unique_docs = []
    seen = set()
    for doc in retrieved_docs:
        key = doc.page_content.strip()
        if not key or key in seen:
            continue
        seen.add(key)
        unique_docs.append(doc)
        if len(unique_docs) == 6:
            break

    return unique_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
