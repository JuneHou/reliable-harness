import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv
dotenv.load_dotenv()


def run_agent(query, k=6):
    loader = UnstructuredFileLoader(
        'paper2.pdf'
    )
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    # De-duplicate chunks BEFORE indexing them: PDFs (headers, footers,
    # repeated paragraphs) often produce several chunks with identical text,
    # and every one of them would otherwise be returned by the retriever.
    seen_content = set()
    unique_splits = []
    for split in all_splits:
        key = split.page_content.strip()
        if not key or key in seen_content:
            continue
        seen_content.add(key)
        unique_splits.append(split)

    vectorstore = Chroma.from_documents(
        documents=unique_splits, embedding=OpenAIEmbeddings()
    )

    # "similarity" returns the k nearest chunks, which are frequently
    # near-identical neighbours from the same passage. MMR (max marginal
    # relevance) instead asks for a larger candidate pool (fetch_k) and picks
    # results that are both relevant *and* diverse, so we get k distinct
    # documents instead of the same one repeated.
    retriever = vectorstore.as_retriever(
        search_type="mmr",
        search_kwargs={"k": k, "fetch_k": max(4 * k, 20), "lambda_mult": 0.5},
    )

    retrieved_docs = retriever.get_relevant_documents(query)

    # Final safety net: keep only unique documents, preserving rank order.
    unique_docs = []
    seen_keys = set()
    for doc in retrieved_docs:
        key = (
            doc.metadata.get("source"),
            doc.metadata.get("page"),
            doc.metadata.get("start_index"),
            doc.page_content,
        )
        if key in seen_keys:
            continue
        seen_keys.add(key)
        unique_docs.append(doc)

    return unique_docs[:k]


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
