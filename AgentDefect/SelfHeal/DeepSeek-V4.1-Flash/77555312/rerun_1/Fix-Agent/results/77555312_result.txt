import os

import dotenv
import bs4
import langchain
from langchain.llms import OpenAI
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma

dotenv.load_dotenv()


def run_agent(query):
    loader = UnstructuredFileLoader('paper2.pdf')
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    # Chroma happily stores identical chunks, so overlapping splits (and any
    # repeated pages) end up as separate, equally-similar entries.  Because a
    # plain "similarity" search just returns the top-k nearest neighbours, the
    # same text can occupy several of the k slots.  Drop exact duplicates first.
    unique_splits = []
    seen_content = set()
    for split in all_splits:
        if split.page_content in seen_content:
            continue
        seen_content.add(split.page_content)
        unique_splits.append(split)

    vectorstore = Chroma.from_documents(
        documents=unique_splits,
        embedding=OpenAIEmbeddings(),
    )

    # Use MMR (maximal marginal relevance) instead of plain similarity: fetch a
    # larger candidate pool and select 6 results that are both relevant to the
    # query AND dissimilar from each other, so the six docs are distinct.
    retriever = vectorstore.as_retriever(
        search_type="mmr",
        search_kwargs={"k": 6, "fetch_k": 20, "lambda_mult": 0.5},
    )

    retrieved_docs = retriever.invoke(query)

    # Safety net: guarantee the caller gets six *unique* documents even if two
    # chunks happen to be near-identical.
    unique_docs = []
    seen_keys = set()
    for doc in retrieved_docs:
        key = (
            doc.metadata.get("source"),
            doc.metadata.get("start_index"),
            doc.page_content,
        )
        if key in seen_keys:
            continue
        seen_keys.add(key)
        unique_docs.append(doc)

    return unique_docs[:6]


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    for i, doc in enumerate(result, start=1):
        print(f"--- document {i} ---")
        print(doc)
