from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate

load_dotenv()

# The retriever only feeds the LLM the documents it returns, so with `k=1` the
# model only ever saw ONE CSV row (the "closest" one, e.g. only Author A) and
# therefore could not mention any other matching row (e.g. Author C).
# We also make the combine prompt explicit so the model lists EVERY matching
# row it is given instead of silently summarizing/skipping some of them.
QA_PROMPT = PromptTemplate(
    template="""You are answering questions about the rows of a movies CSV.
Use ONLY the rows in the context below. If several rows match the question,
list ALL of them (title, year, author/genre, ...) - never just one of them.
If the context does not contain the answer, say that you don't know.

Context (matching CSV rows):
{context}

Question: {question}

Answer (list every matching row):""",
    input_variables=["context", "question"],
)


def run_agent(prompt):

    embeddings = OpenAIEmbeddings()

    csvloader = CSVLoader("movies.csv", encoding="utf-8")
    csvdocuments = csvloader.load()

    search = Chroma.from_documents(csvdocuments, embeddings)

    # Retrieve ALL rows (or at least a much larger number than 1) so rows that
    # are slightly less "similar" but still relevant are not dropped.
    top_k = max(len(csvdocuments), 1)

    chain = ConversationalRetrievalChain.from_llm(
        ChatOpenAI(temperature=0),
        retriever=search.as_retriever(
            search_type="mmr",
            search_kwargs={"k": top_k, "fetch_k": top_k},
        ),
        return_source_documents=True,
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
    )

    # `chat_history` must be a list of (human, ai) tuples - an empty string made
    # the chain build a malformed history and lose context between turns.
    result = chain(
        {"question": prompt, "chat_history": []},
        return_only_outputs=True,
    )
    return result["answer"]


if __name__ == '__main__':
    print(run_agent("which movies are action"))
