from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate

load_dotenv()

# The retriever used to only give the LLM a single row (k=1), so any answer that
# required more than one matching record (e.g. "films in 1999" -> Author A AND
# Author C) was truncated to whatever single document happened to rank first.
# We now hand every row to the model and explicitly ask it to list all matches.
QA_TEMPLATE = """You are a helpful assistant that answers questions about a movies dataset.
Answer the question using ONLY the retrieved records in the context below.
The context may contain several records: you MUST consider all of them and list
every movie/author that matches the question, instead of stopping at the first match.
If the context does not contain the answer, say "I don't know based on the provided data."

Context:
{context}

Question: {question}

Answer:"""

QA_PROMPT = PromptTemplate(
    template=QA_TEMPLATE,
    input_variables=["context", "question"],
)

CONDENSE_TEMPLATE = """Given the following conversation and a follow up question,
rephrase the follow up question to be a standalone question.

Chat History:
{chat_history}
Follow Up Input: {question}
Standalone question:"""

CONDENSE_PROMPT = PromptTemplate.from_template(CONDENSE_TEMPLATE)


def build_chain(csv_path="movies.csv"):
    embeddings = OpenAIEmbeddings()

    csvloader = CSVLoader(csv_path)
    csvdocuments = csvloader.load()

    vectorstore = Chroma.from_documents(csvdocuments, embeddings)

    # Retrieve ALL rows instead of just one so nothing is silently dropped.
    num_docs = max(len(csvdocuments), 1)
    retriever = vectorstore.as_retriever(search_kwargs={"k": num_docs})

    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0),
        retriever=retriever,
        return_source_documents=True,
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
        condense_question_prompt=CONDENSE_PROMPT,
    )
    return chain


CHAIN = None


def run_agent(prompt, chat_history=None):
    global CHAIN
    if CHAIN is None:
        CHAIN = build_chain("movies.csv")

    # chat_history must be a list, not an empty string.
    if chat_history is None:
        chat_history = []

    result = CHAIN(
        {"question": prompt, "chat_history": chat_history},
        return_only_outputs=True,
    )
    return result["answer"]


if __name__ == '__main__':
    print(run_agent("which movies are action"))
    print(run_agent("which films were released in 1999 and who wrote them?"))
