from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate

load_dotenv()

# The default "stuff" prompt just summarizes whatever single chunk it was given.
# Make the instructions explicit: read EVERY retrieved row and enumerate all
# matches instead of reporting only the first one.
MOVIE_QA_PROMPT = PromptTemplate(
    input_variables=["context", "question"],
    template=(
        "You are a meticulous assistant that answers questions about a movies data set.\n"
        "Each line of the context below is one separate record from the CSV file, "
        "written as 'column: value' pairs.\n"
        "Read the ENTIRE context, collect every record that satisfies the question, "
        "and list all of them. Do not stop at the first match, do not summarize and "
        "do not invent records that are not in the context.\n"
        "Output one record per line in the form '<movie title> (<year>) - <author>'.\n"
        "If nothing in the context matches, answer exactly: 'No matching records found.'\n\n"
        "Context (retrieved records):\n{context}\n\n"
        "Question: {question}\n\n"
        "All matching movies:"
    ),
)

CONDENSE_QUESTION_PROMPT = PromptTemplate(
    input_variables=["chat_history", "question"],
    template=(
        "Given the following conversation and a follow up question, rephrase the "
        "follow up question to be a standalone question that keeps every filter "
        "(year, genre, author, ...) intact.\n\n"
        "Chat History:\n{chat_history}\n\n"
        "Follow Up Input: {question}\n"
        "Standalone question:"
    ),
)


def run_agent(prompt):
    embeddings = OpenAIEmbeddings()

    csvloader = CSVLoader("movies.csv")
    csvdocuments = csvloader.load()

    search = Chroma.from_documents(csvdocuments, embeddings)

    # BUG: search_kwargs={"k": 1} handed the LLM a single CSV row, so it could
    # physically never report more than one matching film. Retrieve the whole
    # (small) CSV so every matching row reaches the model.
    k = max(len(csvdocuments), 1)

    chain = ConversationalRetrievalChain.from_llm(
        ChatOpenAI(temperature=0.0),
        retriever=search.as_retriever(search_kwargs={"k": k}),
        condense_question_prompt=CONDENSE_QUESTION_PROMPT,
        combine_docs_chain_kwargs={"prompt": MOVIE_QA_PROMPT},
        return_source_documents=True,
        verbose=False,
    )

    # chat_history must be a list of (human, ai) tuples, not an empty string.
    result = chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return result["answer"]


if __name__ == '__main__':
    print(run_agent("which movies are action"))
