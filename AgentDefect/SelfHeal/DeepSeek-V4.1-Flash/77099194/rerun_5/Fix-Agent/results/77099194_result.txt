from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chains.question_answering import load_qa_chain
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate

load_dotenv()


def run_agent(prompt):
    embeddings = OpenAIEmbeddings()

    # Each CSV row becomes one Document (one movie + one author).
    csvloader = CSVLoader("movies.csv")
    csvdocuments = csvloader.load()

    search = Chroma.from_documents(csvdocuments, embeddings)

    # BUG FIX (main): k=1 only retrieved a single movie row, so the LLM could
    # only ever answer with "Author A" and never see Author C's row.
    # Retrieve the whole (small) catalog so every matching row is available.
    retriever = search.as_retriever(search_kwargs={"k": len(csvdocuments)})

    # BUG FIX: tell the model to report ALL matching rows instead of the first one.
    qa_prompt = PromptTemplate(
        template="""You are answering questions about a movies dataset.
The context below contains one row per movie, each row listing its title,
genre(s) and author.

Read EVERY row in the context. If more than one row satisfies the
question, you must list ALL of them - never return just the first match.

Context:
{context}

Question: {question}

Answer: list every matching movie together with its author
(e.g. "Movie X - Author A; Movie Y - Author C"):""",
        input_variables=["context", "question"],
    )

    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0),
        retriever=retriever,
        return_source_documents=True,
        combine_docs_chain_kwargs={"prompt": qa_prompt},
    )

    # BUG FIX: chat_history must be a list of (human, ai) tuples, not an empty string.
    result = chain(
        {"question": prompt, "chat_history": []},
        return_only_outputs=True,
    )
    return result["answer"]


if __name__ == '__main__':
    print(run_agent("which movies are action in 1999 and who are their authors?"))
