from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()


def run_agent(prompt):
    embeddings = OpenAIEmbeddings()

    csvloader = CSVLoader("movies.csv")
    csvdocuments = csvloader.load()  # one Document per CSV row

    search = Chroma.from_documents(csvdocuments, embeddings)

    # BUG: search_kwargs={"k": 1} handed the LLM a single CSV row, so any other
    # row that also satisfied the filter (e.g. another movie from 1999 written
    # by Author C) was never placed in the prompt and could never be reported.
    # FIX: retrieve (and stuff into the prompt) every row of the CSV, so the
    # model can aggregate all matching films/authors instead of only the
    # single nearest neighbour.
    k = len(csvdocuments) if csvdocuments else 1
    retriever = search.as_retriever(
        search_type="mmr",
        search_kwargs={"k": k, "fetch_k": k},
    )

    chain = ConversationalRetrievalChain.from_llm(
        ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # chat_history must be a list of (human, ai) tuples, not an empty string
    result = chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return result["answer"]


if __name__ == '__main__':
    # e.g. "which movies were released in 1999 and who wrote them?"
    print(run_agent("which movies were released in 1999 and who wrote them?"))
    print(run_agent("which movies are action"))
