import os
from glob import glob

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader, PyPDFLoader

_ = load_dotenv(find_dotenv())

COLLECTION_NAME = "langchain"


def _load_documents(document_paths):
    """Load and split every document, picking the loader from the file extension."""
    all_texts = []
    for doc_path in document_paths:
        if doc_path.lower().endswith(".pdf"):
            loader = PyPDFLoader(doc_path)
        else:
            loader = TextLoader(doc_path, encoding="utf-8")

        documents = loader.load()
        text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=1000, chunk_overlap=200
        )
        texts = text_splitter.split_documents(documents)
        print(f"Loaded {len(texts)} chunks from {doc_path}")
        all_texts.extend(texts)
    return all_texts


def run_agent(
    user_input,
    document_paths=None,
    persist_directory="chroma_db",
    collection_name=COLLECTION_NAME,
):
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY is not set (check your .env file).")
    os.environ["OPENAI_API_KEY"] = openai_api_key

    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    if document_paths:
        all_texts = _load_documents(document_paths)
        if not all_texts:
            raise ValueError("No text chunks were produced from the given documents.")

        # Build (and persist) the collection under an explicit, known name.
        docsearch = Chroma.from_documents(
            documents=all_texts,
            embedding=embeddings,
            persist_directory=persist_directory,
            collection_name=collection_name,
        )
    else:
        # FIX: all arguments are passed by KEYWORD.
        # The buggy call was `Chroma(persist_directory, embedding_function=embeddings)`,
        # where the first positional argument of Chroma is `collection_name`, so the
        # string "chroma_db" was used as the COLLECTION NAME (and persist_directory
        # stayed None). The resulting store had no documents, so the retriever
        # silently returned an empty list.
        docsearch = Chroma(
            collection_name=collection_name,
            embedding_function=embeddings,
            persist_directory=persist_directory,
        )

    # Sanity check: make sure the store really holds documents before querying.
    stored = docsearch.get()
    n_stored = len(stored.get("ids", []))
    print(f"Documents in collection '{collection_name}': {n_stored}")
    if n_stored == 0:
        raise RuntimeError(
            f"The collection '{collection_name}' in '{persist_directory}' is empty. "
            "Pass document_paths so the index can be built first."
        )

    retriever = docsearch.as_retriever(search_kwargs={"k": 5})
    docs = retriever.get_relevant_documents(user_input)
    print(f"Number of documents retrieved: {len(docs)}")

    return docs


if __name__ == "__main__":
    document_paths = glob("docs/*.txt") + glob("docs/*.pdf")
    query = "What is machine learning?"
    docs = run_agent(query, document_paths)

    print("\n=== User Query Result ===")
    print(f"Query: {query}")
    print(f"Number of relevant documents: {len(docs)}")

    for i, doc in enumerate(docs):
        print(f"\n--- Document {i + 1} ---")
        print(doc.page_content)

So the positional `"chroma_db"` was bound to **`collection_name`**, leaving `persist_directory=None` — i.e. the query-time store was an *ephemeral* client looking for a collection literally named `"chroma_db"`, while `from_documents` had persisted into directory `chroma_db` under the default collection `"langchain"`. - `from_documents(documents=..., embedding=..., persist_directory=..., collection_name=...)` — correct keyword names (`embedding`, not `embedding_function`, which is the naming used by `from_documents`).

- Explicit `collection_name` constant removes the implicit-default mismatch between construction and querying paths.
- `else` branch added so the store is only rebuilt when paths are given, rather than always re-instantiating from disk (the original redundant double assignment).
