import os
from glob import glob

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader

try:  # optional PDF support
    from langchain.document_loaders import PyPDFLoader
except ImportError:  # pragma: no cover
    PyPDFLoader = None

_ = load_dotenv(find_dotenv())

# The collection name MUST be identical when writing and when re-opening the
# store, otherwise you silently open an EMPTY collection (-> zero documents).
COLLECTION_NAME = "langchain"


def _load_and_split(document_paths, chunk_size=1000, chunk_overlap=200):
    """Load every document and split it into chunks."""
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=chunk_size, chunk_overlap=chunk_overlap
    )
    all_texts = []
    for doc_path in document_paths:
        ext = os.path.splitext(doc_path)[1].lower()
        if ext == ".pdf" and PyPDFLoader is not None:
            loader = PyPDFLoader(doc_path)
        else:
            try:
                loader = TextLoader(doc_path, encoding="utf-8")
            except TypeError:  # very old langchain versions have no encoding kwarg
                loader = TextLoader(doc_path)
        documents = loader.load()
        all_texts.extend(splitter.split_documents(documents))
    return all_texts


def run_agent(
    user_input,
    document_paths=None,
    persist_directory="chroma_db",
    collection_name=COLLECTION_NAME,
    k=5,
):
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY is not set in the environment / .env file.")

    os.environ["OPENAI_API_KEY"] = openai_api_key
    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    docsearch = None

    # ------------------------------------------------------------------
    # 1) Build / update the store when documents are supplied
    # ------------------------------------------------------------------
    if document_paths:
        texts = _load_and_split(document_paths)
        if texts:
            os.makedirs(persist_directory, exist_ok=True)
            docsearch = Chroma.from_documents(
                documents=texts,
                embedding=embeddings,
                collection_name=collection_name,      # same name on write ...
                persist_directory=persist_directory,  # ... and on read
            )
            if hasattr(docsearch, "persist"):
                docsearch.persist()  # older langchain releases require this

    # ------------------------------------------------------------------
    # 2) Re-open the persisted store.
    #    BUG WAS HERE: Chroma(persist_directory, embedding_function=...)
    #    passed the directory as the FIRST POSITIONAL argument, which is
    #    `collection_name`. So Chroma looked for a collection literally
    #    named "chroma_db", found/created an empty one and the retriever
    #    therefore returned 0 documents. Keyword args fix it.
    # ------------------------------------------------------------------
    if docsearch is None:
        docsearch = Chroma(
            collection_name=collection_name,
            embedding_function=embeddings,
            persist_directory=persist_directory,
        )

    retriever = docsearch.as_retriever(search_kwargs={"k": k})
    docs = retriever.get_relevant_documents(user_input)

    print(f"Number of documents retrieved: {len(docs)}")
    if not docs:
        print(
            "WARNING: the vector store returned no documents. "
            "Check that the persist_directory contains data and that the "
            "collection name matches the one used when indexing."
        )

    return docs


if __name__ == "__main__":
    document_paths = sorted(
        glob("docs/*.txt") + glob("docs/*.pdf")
    )
    query = "What is machine learning?"
    docs = run_agent(query, document_paths)

    print("\n=== User Query Result ===")
    print(f"Query: {query}")
    print(f"Number of relevant documents: {len(docs)}")

    for i, doc in enumerate(docs):
        print(f"\n--- Document {i + 1} ---")
        print(doc.page_content)
