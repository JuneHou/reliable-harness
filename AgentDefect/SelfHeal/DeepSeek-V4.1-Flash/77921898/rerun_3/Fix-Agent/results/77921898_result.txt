import os
from glob import glob

from dotenv import load_dotenv, find_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader, PyPDFLoader

_ = load_dotenv(find_dotenv())


def _load_documents(document_paths):
    """Load PDFs (and plain text files) into LangChain Documents."""
    documents = []
    for doc_path in document_paths:
        ext = os.path.splitext(doc_path)[1].lower()
        if ext == ".pdf":
            loader = PyPDFLoader(doc_path)
        elif ext in (".txt", ".md", ".csv"):
            loader = TextLoader(doc_path)
        else:
            raise ValueError(f"Unsupported document type: {doc_path}")
        documents.extend(loader.load())
    return documents


def run_agent(user_input, document_paths=None, persist_directory="chroma_db"):
    openai_api_key = os.getenv("OPENAI_API_KEY")
    os.environ["OPENAI_API_KEY"] = openai_api_key

    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    # IMPORTANT: the first positional argument of Chroma(...) is `collection_name`,
    # NOT `persist_directory`. Passing the directory positionally made the
    # vectorstore look into a collection literally named "chroma_db" (which is
    # empty) instead of the "langchain" collection created by from_documents,
    # so the retriever silently returned zero documents.
    collection_name = "langchain"

    docsearch = None

    if document_paths:
        documents = _load_documents(document_paths)

        if documents:
            text_splitter = RecursiveCharacterTextSplitter(
                chunk_size=1000, chunk_overlap=200
            )
            texts = text_splitter.split_documents(documents)

            docsearch = Chroma.from_documents(
                texts,
                embeddings,
                collection_name=collection_name,
                persist_directory=persist_directory,
            )
            # Make sure chunks are written to disk before querying.
            if hasattr(docsearch, "persist"):
                docsearch.persist()
        else:
            print(f"No documents could be loaded from: {document_paths}")

    # Always pass persist_directory (and collection_name) as KEYWORD arguments.
    if docsearch is None:
        docsearch = Chroma(
            collection_name=collection_name,
            embedding_function=embeddings,
            persist_directory=persist_directory,
        )

    retriever = docsearch.as_retriever(search_kwargs={"k": 5})
    docs = retriever.get_relevant_documents(user_input)
    print(f"Number of documents retrieved: {len(docs)}")

    if not docs:
        count = docsearch._collection.count() if hasattr(docsearch, "_collection") else "unknown"
        print(
            "WARNING: the retriever returned no documents. "
            f"Collection '{collection_name}' in '{persist_directory}' holds {count} chunk(s). "
            "Re-run with document_paths to (re)index your PDFs."
        )

    return docs


if __name__ == "__main__":
    # Include PDFs as well as text files.
    document_paths = sorted(glob("docs/*.pdf") + glob("docs/*.txt"))
    query = "What is machine learning?"
    docs = run_agent(query, document_paths, persist_directory="chroma_db")

    print("\n=== User Query Result ===")
    print(f"Query: {query}")
    print(f"Number of relevant documents: {len(docs)}")

    for i, doc in enumerate(docs):
        print(f"\n--- Document {i+1} ---")
        print(doc.page_content)

### Root cause identified (and correctly fixed)

The original bug is subtle and precisely diagnosed by the fix:

```python
docsearch = Chroma(persist_directory, embedding_function=embeddings)
```

`Chroma.__init__`'s first positional parameter is **`collection_name`**, not `persist_directory`. So the buggy code:
1. created/looked up a collection literally named `"chroma_db"` instead of `"langchain"` (the default collection `from_documents` had written to), and
2. silently dropped `persist_directory` to `None`, so it never even opened the on-disk store.

Result: an empty in-memory collection → `retriever.get_relevant_documents()` returns `[]` → `assertGreater(len(result), 0)` fails.

The fix addresses both halves:
- Explicit `collection_name="langchain"` used consistently for `from_documents` and for the read path.
- `persist_directory` passed as a **keyword** argument everywhere, eliminating the positional-arg trap.
- The redundant re-open is avoided by keeping the `docsearch` returned by `from_documents` (`if docsearch is None:` guard), so querying hits the freshly populated index.

### Additional changes — all sound

- **`_load_documents` helper**: extension-based loader dispatch. `PyPDFLoader` is importable from `langchain.document_loaders` (exposed via lazy `__getattr__`), and the test only globs `docs/*.txt`, so the PDF path is never exercised — no new import-time or runtime risk.
- **Defensive `hasattr(docsearch, "persist")`**: correct, since `Chroma.persist()` was removed/deprecated across langchain/chromadb versions where persistence is automatic.
- **`hasattr(docsearch, "_collection")`** guard around `count()`: safe, private attribute access is only for a diagnostic warning.
- **`if documents:` guard** and "unsupported type" `ValueError`: reasonable, and unreachable in this test.
- Signature of `run_agent(user_input, document_paths=None, persist_directory="chroma_db")` is unchanged, so the test's `run_agent(query, document_paths)` call still works with the default directory. Return type remains `list[Document]` with readable `page_content`.
- Extra `glob("docs/*.pdf")` only in `__main__` — irrelevant to the test, harmless.

### Residual (non-blocking) notes

- `os.environ["OPENAI_API_KEY"] = openai_api_key` would raise `TypeError` if the key is absent — pre-existing behavior, unchanged, and not triggered by the diff.
