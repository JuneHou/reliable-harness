import os
import random
import warnings

from dotenv import load_dotenv, find_dotenv

from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.vectorstores.base import VectorStoreRetriever
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]

DEFAULT_K = 4


def get_random_key(key_pool):
    usable = [k for k in key_pool if k]
    if not usable:
        raise ValueError(
            "No OpenAI API key available; set OPENAI_API_KEY in the environment/.env file."
        )
    return random.choice(usable)


def _matches(metadata, metadata_filter):
    """True if a chunk's metadata satisfies every (key, value) pair of the filter."""
    metadata = metadata or {}
    return all(metadata.get(key) == value for key, value in metadata_filter.items())


class FilteredVectorStoreRetriever(VectorStoreRetriever):
    """A VectorStoreRetriever that supports a *query-time* metadata filter.

    Why this is needed:
    `ConversationalRetrievalChain` never forwards arbitrary keys of the input
    dict (e.g. ``{"filter": {...}}``) to the retriever - it only calls
    ``retriever.get_relevant_documents(question)``.  Therefore a filter passed
    in the input dict is silently ignored and the whole collection is searched
    (the answer may then come from "paper1.pdf" instead of "paper2.pdf").

    This subclass keeps the filter on the retriever and forwards it to
    ``VectorStore.similarity_search(..., filter=...)`` (native Chroma ``where``
    filtering), so a *single* chain can serve every document/filter combination
    without duplicating the chain or modifying LangChain itself.
    """

    def set_filter(self, metadata_filter=None):
        """Set (or clear, when ``None``) the metadata filter used at query time."""
        search_kwargs = dict(self.search_kwargs or {})
        if metadata_filter is None:
            search_kwargs.pop("filter", None)
        else:
            search_kwargs["filter"] = dict(metadata_filter)
        self.search_kwargs = search_kwargs
        return self

    def clear_filter(self):
        return self.set_filter(None)

    def _get_relevant_documents(self, query, *, run_manager=None, **kwargs):
        search_kwargs = dict(self.search_kwargs or {})
        metadata_filter = search_kwargs.pop("filter", None)
        # a filter forwarded per call (retriever.get_relevant_documents(q, filter=...))
        # takes precedence over the one stored on the retriever
        metadata_filter = kwargs.pop("filter", metadata_filter)
        search_kwargs.update(kwargs)
        search_kwargs.setdefault("k", DEFAULT_K)

        if metadata_filter is None:
            return self.vectorstore.similarity_search(query, **search_kwargs)

        try:
            # Chroma translates this dict into a native "where" clause
            return self.vectorstore.similarity_search(
                query, filter=metadata_filter, **search_kwargs
            )
        except TypeError:
            # vector store backend does not accept a `filter` argument:
            # fall back to retrieving more chunks and filtering in Python
            fallback_kwargs = dict(search_kwargs)
            fallback_kwargs["k"] = max(fallback_kwargs.get("k", DEFAULT_K) * 5, DEFAULT_K * 5)
            docs = self.vectorstore.similarity_search(query, **fallback_kwargs)
            return [doc for doc in docs if _matches(doc.metadata, metadata_filter)][
                : search_kwargs.get("k", DEFAULT_K)
            ]


def build_vectorstore(pdf_paths, embeddings):
    """Load all PDFs into one Chroma store, tagging each chunk with its file name."""
    all_texts = []
    for pdf_path in pdf_paths:
        documents = PyPDFLoader(pdf_path).load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        # use the bare file name so filters like {'source': 'paper2.pdf'} match
        # even when absolute/relative paths are used for loading
        source = os.path.basename(pdf_path)
        for text in texts:
            text.metadata["source"] = source

        all_texts.extend(texts)

    return Chroma.from_documents(all_texts, embeddings)


def build_chain(pdf_paths, embeddings):
    """Build ONE conversational chain over all documents."""
    db = build_vectorstore(pdf_paths, embeddings)

    retriever = FilteredVectorStoreRetriever(
        vectorstore=db,
        search_kwargs={"k": DEFAULT_K},
    )

    qa = ConversationalRetrievalChain.from_llm(
        OpenAI(
            openai_api_key=get_random_key(OPENAI_API_KEY_POOL),
            temperature=0,
        ),
        retriever,
        verbose=True,
        chain_type="stuff",
        return_source_documents=True,
    )
    return qa


def ask(qa, query, chat_history=None, filter=None):
    """Ask a question, scoping retrieval to `filter` (e.g. {'source': 'paper2.pdf'})."""
    qa.retriever.set_filter(filter)

    # NOTE: do NOT put "filter" inside this dict - the chain ignores it.
    result = qa({"question": query, "chat_history": chat_history or []})

    if filter:
        mismatched = [
            doc for doc in result.get("source_documents", [])
            if not _matches(doc.metadata, filter)
        ]
        if mismatched:
            warnings.warn(
                f"Retriever returned chunks that do not match the filter {filter!r}: "
                f"{[d.metadata for d in mismatched]}"
            )

    return result


def run_agent(query, pdf_paths, filter=None):
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))
    qa = build_chain(pdf_paths, embeddings)
    return ask(qa, query, chat_history=[], filter=filter)
