"""
Fix summary
-----------
`ConversationalRetrievalChain` has exactly two input keys: ``question`` and
``chat_history``.  Anything else you put in the dict handed to the chain
(here: ``filter``) is silently dropped, so the retriever still performs an
*unfiltered* similarity search over every chunk of every PDF.  That is why the
answer still contains information coming from ``paper1.pdf`` even though the
user asked for ``paper2.pdf`` only.

The metadata filter has to reach the **vector store query**, which is the only
place Chroma understands it:

    db.similarity_search(query, filter={"source": "paper2.pdf"})

With a ``VectorStoreRetriever`` you do that through ``retriever.search_kwargs``
(they are forwarded verbatim to ``similarity_search``).  You therefore do *not*
have to patch LangChain and you do *not* have to build one chain per document:
build the chain once, and set/clear ``search_kwargs["filter"]`` on the (single)
retriever for each query.
"""

import os
import random
from typing import Any, Dict, List, Optional

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv("OPENAI_API_KEY"),
]


def get_random_key(key_pool: List[Optional[str]]) -> str:
    keys = [k for k in key_pool if k]
    if not keys:
        raise RuntimeError(
            "No OpenAI API key available - set OPENAI_API_KEY in your environment/.env"
        )
    return random.choice(keys)


def build_vectorstore(
    pdf_paths: List[str],
    embeddings: OpenAIEmbeddings,
    collection_name: str = "papers",
) -> Chroma:
    """Load every PDF into ONE collection, tagging each chunk with its source."""
    all_texts = []

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=1000, chunk_overlap=200
        )
        texts = text_splitter.split_documents(documents)

        for text in texts:
            # metadata is what the Chroma `filter` operates on
            text.metadata["source"] = pdf_path

        all_texts.extend(texts)

    return Chroma.from_documents(
        all_texts, embeddings, collection_name=collection_name
    )


def build_qa_chain(db: Chroma) -> ConversationalRetrievalChain:
    """Build the retriever + chain ONCE for the whole collection."""
    retriever = db.as_retriever(
        search_type="similarity",
        search_kwargs={"k": 4},
    )

    return ConversationalRetrievalChain.from_llm(
        OpenAI(
            openai_api_key=get_random_key(OPENAI_API_KEY_POOL),
            temperature=0,
        ),
        # NOTE: don't pass cache=True here; LLM caching is enabled globally
        # with set_llm_cache(InMemoryCache()) above.
        retriever,
        verbose=True,
        return_source_documents=True,
    )


def ask(
    qa: ConversationalRetrievalChain,
    question: str,
    chat_history: Optional[List] = None,
    metadata_filter: Optional[Dict[str, Any]] = None,
    k: int = 4,
) -> Dict[str, Any]:
    """Ask a question, optionally restricted to chunks matching `metadata_filter`.

    The filter is pushed down to the retriever (-> Chroma), instead of being
    handed to the chain, where it would be ignored.
    """
    search_kwargs: Dict[str, Any] = {"k": k}
    if metadata_filter:
        # Chroma expects the operator dict, e.g.
        #   {"source": "paper2.pdf"}  or  {"source": {"$in": [...]}}
        search_kwargs["filter"] = metadata_filter

    # Mutating the single retriever is fine for sequential use.  If you call
    # this concurrently, build one chain per request (cheap: it only wraps the
    # shared vector store) or use a runnable with configurable search_kwargs.
    qa.retriever.search_kwargs = search_kwargs

    return qa({"question": question, "chat_history": chat_history or []})


def run_agent(
    query: str,
    pdf_paths: List[str],
    metadata_filter: Optional[Dict[str, Any]] = None,
) -> Dict[str, Any]:
    embeddings = OpenAIEmbeddings(
        openai_api_key=get_random_key(OPENAI_API_KEY_POOL)
    )

    db = build_vectorstore(pdf_paths, embeddings)
    qa = build_qa_chain(db)

    return ask(qa, query, chat_history=[], metadata_filter=metadata_filter)


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]

    # Only chunks whose metadata["source"] == "paper2.pdf" may be retrieved.
    source_filter = {"source": "paper2.pdf"}

    result = run_agent(query, pdf_paths, metadata_filter=source_filter)

    print(result["answer"])
    print("\nRetrieved from:")
    for doc in result.get("source_documents", []):
        print(" -", doc.metadata.get("source"))
