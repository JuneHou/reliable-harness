"""Query-time metadata filtering for ConversationalRetrievalChain + Chroma.

Why the original code was wrong
-------------------------------
`qa({"question": ..., "chat_history": [], "filter": {"source": "paper2.pdf"}})`
does NOT filter anything: `Chain.prep_inputs` only validates the declared
`input_keys` (`question`, `chat_history`) and the extra `filter` key is never
forwarded to the retriever.  `VectorStoreRetriever` therefore runs a plain
similarity search over the WHOLE collection, so the answer can be built from
`paper1.pdf` chunks even though a filter was requested.

You do not have to modify LangChain, and you do not need a chain (or a
vector store, or new embeddings) per document.  Keep ONE chain and ONE
retriever, and change the retriever's metadata filter per question; the
retriever then passes it to Chroma's `where` clause
(`similarity_search(query, filter=...)`).
"""

import os
import random
from typing import Any, Dict, List, Optional

from dotenv import load_dotenv, find_dotenv

from langchain.cache import InMemoryCache
from langchain.chains import ConversationalRetrievalChain
from langchain.document_loaders import PyPDFLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.globals import set_llm_cache
from langchain.llms import OpenAI
from langchain.schema import BaseRetriever, Document
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]


def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


class FilterableVectorStoreRetriever(BaseRetriever):
    """Vector-store retriever whose metadata filter can be set at query time.

    `ConversationalRetrievalChain` calls `retriever.get_relevant_documents(query)`
    with the question only, so the filter cannot travel inside the chain input
    dict.  Instead the filter lives on the retriever and is handed to the
    vector store (`filter=` -> Chroma `where`) on every retrieval.
    """

    vectorstore: Any
    search_kwargs: Dict[str, Any] = {}
    metadata_filter: Optional[Dict[str, Any]] = None

    class Config:
        arbitrary_types_allowed = True

    def _effective_kwargs(self) -> Dict[str, Any]:
        kwargs = dict(self.search_kwargs or {})
        if self.metadata_filter:
            kwargs["filter"] = self.metadata_filter
        return kwargs

    def _get_relevant_documents(
        self, query: str, *, run_manager: Any = None, **kwargs: Any
    ) -> List[Document]:
        merged = dict(kwargs)
        merged.update(self._effective_kwargs())
        return self.vectorstore.similarity_search(query, **merged)

    async def _aget_relevant_documents(
        self, query: str, *, run_manager: Any = None, **kwargs: Any
    ) -> List[Document]:
        return self._get_relevant_documents(query, run_manager=run_manager, **kwargs)


def _load_vectorstore(pdf_paths: List[str], embeddings) -> Chroma:
    all_texts = []
    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        # Store the bare file name so filters such as {'source': 'paper2.pdf'}
        # match no matter whether a relative or an absolute path was given.
        file_name = os.path.basename(pdf_path)
        for text in texts:
            text.metadata["source"] = file_name
            text.metadata["file_name"] = file_name

        all_texts.extend(texts)

    return Chroma.from_documents(all_texts, embeddings)


_QA_CACHE: Dict[Any, ConversationalRetrievalChain] = {}


def get_qa_chain(pdf_paths: List[str]) -> ConversationalRetrievalChain:
    """Build (once) a chain holding a single index + single retriever."""
    key = tuple(os.path.abspath(p) for p in pdf_paths)
    if key not in _QA_CACHE:
        embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))
        db = _load_vectorstore(pdf_paths, embeddings)

        retriever = FilterableVectorStoreRetriever(
            vectorstore=db,
            search_kwargs={"k": 4},
        )

        _QA_CACHE[key] = ConversationalRetrievalChain.from_llm(
            llm=OpenAI(
                openai_api_key=get_random_key(OPENAI_API_KEY_POOL),
                cache=True,
                temperature=0,
            ),
            retriever=retriever,
            verbose=True,
            return_source_documents=True,
        )
    return _QA_CACHE[key]


def run_agent(
    query: str,
    pdf_paths: List[str],
    filter: Optional[Dict[str, Any]] = None,
    chat_history: Optional[List] = None,
):
    """Ask `query` using only chunks matching `filter` (metadata/where clause).

    The chain is created once per PDF set; the metadata filter is applied
    dynamically for this single question.
    """
    chat_history = chat_history or []

    qa = get_qa_chain(pdf_paths)
    retriever = qa.retriever
    assert isinstance(retriever, FilterableVectorStoreRetriever)

    previous_filter = retriever.metadata_filter
    retriever.metadata_filter = filter
    try:
        # NOTE: 'filter' is deliberately NOT put into this dict - the chain
        # would ignore it. It is applied by the retriever above.
        result = qa({"question": query, "chat_history": chat_history})
    finally:
        retriever.metadata_filter = previous_filter

    sources = {
        doc.metadata.get("source", "unknown")
        for doc in result.get("source_documents", [])
    }
    print("source documents used:", sorted(sources))

    if filter and filter.get("source"):
        expected = os.path.basename(filter["source"])
        if sources and sources != {expected}:
            print(
                f"WARNING: expected only '{expected}' but retrieved {sorted(sources)}; "
                "check that the metadata key/value exists in the index."
            )

    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]

    # Applied at query time, on a single shared chain/index.
    metadata_filter = {"source": "paper2.pdf"}

    result = run_agent(query, pdf_paths, filter=metadata_filter)
    print(result["answer"])
