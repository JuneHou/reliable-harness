"""
Persisting a LangChain ParentDocumentRetriever (parents + child chunks).

Why the original output did not match expectations
--------------------------------------------------
1. The docstore was an `InMemoryStore`, so the parent documents only lived in
   the Python process. As soon as the script ended they were gone: the Chroma
   collection still contained the *child chunks*, but the parents they point to
   no longer existed. Restoring the retriever therefore required re-running
   `add_documents` every time, and querying the raw vectorstore only ever gave
   back child chunks (never the full parent documents).
2. `max_marginal_relevance_search` was called directly on the vectorstore,
   which returns the *child chunks* stored in Chroma, not the parent documents
   the retriever is supposed to hand back.

Fix
---
* Replace `InMemoryStore` with a persistent key/value doc store backed by
  `LocalFileStore` (`create_kv_docstore`), stored next to the Chroma DB.
  Now both halves of the retriever are on disk:
      - child chunks  -> Chroma (persist_directory)
      - parent docs   -> LocalFileStore docstore (persist_directory)
* Build the retriever from those persistent locations, and only ingest the
  documents when the docstore/collection is empty (idempotent load-or-create).
* Query through the retriever (`get_relevant_documents`) so the full parent
  documents are returned instead of raw child chunks.
"""

import os

import nltk
from dotenv import load_dotenv

from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import LocalFileStore, create_kv_docstore
from langchain.schema import Document

load_dotenv()

PERSIST_DIRECTORY = "./store_location"
CHROMA_SUBDIR = "chroma_db_child"
DOCSTORE_SUBDIR = "docstore"


def get_documents():
    return [
        Document(
            page_content=(
                "Artificial Intelligence (AI) is technology that enables computers "
                "and machines to simulate human intelligence."
            )
        ),
        Document(
            page_content=(
                "Machine learning (ML) is a type of Artificial Intelligence (AI) that "
                "teaches computers to find patterns in large datasets and make decisions "
                "or predictions without being explicitly programmed for every scenario, "
                "learning and improving from experience, much like humans do."
            )
        ),
        Document(
            page_content=(
                "RAG most commonly refers to Retrieval-Augmented Generation, an AI "
                "technique that improves large language models (LLMs) by letting them "
                "access and use external, up-to-date information (like documents or "
                "databases) to generate more accurate, context-aware, and factual "
                "answers, preventing hallucinations and tailoring responses to specific "
                "knowledge bases without needing costly retraining."
            )
        ),
    ]


def get_retriever(persist_directory=PERSIST_DIRECTORY, hf_embed_path="./hf_cache"):
    """Rebuild the retriever from the on-disk vectorstore + on-disk docstore."""
    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path,
    )

    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory=os.path.join(persist_directory, CHROMA_SUBDIR),
    )

    # ---- the actual fix: a persistent docstore instead of InMemoryStore ----
    os.makedirs(os.path.join(persist_directory, DOCSTORE_SUBDIR), exist_ok=True)
    fs = LocalFileStore(os.path.join(persist_directory, DOCSTORE_SUBDIR))
    store = create_kv_docstore(fs)  # ByteStore -> KVStore[Document]

    child_splitter = NLTKTextSplitter(chunk_size=400)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter,
        # parent_splitter=None -> whole documents are kept as parents
    )

    return retriever, vectorstore


def run_agent(query, store_location=PERSIST_DIRECTORY):
    # NLTKTextSplitter needs the punkt tokenizer data (punkt_tab on newer NLTK).
    for resource in ("punkt", "punkt_tab"):
        try:
            nltk.download(resource, quiet=True)
        except Exception:
            pass

    retriever, vectorstore = get_retriever(store_location)

    # Load-or-create: only ingest when nothing has been persisted yet, so the
    # documents are not duplicated on every run and `add_documents` is skipped
    # once the store exists.
    already_stored = list(retriever.docstore.yield_keys())
    if not already_stored:
        docs = get_documents()
        # Stable ids make re-ingestion idempotent as well.
        retriever.add_documents(docs, ids=[f"doc-{i}" for i in range(len(docs))])
        try:
            vectorstore.persist()  # no-op on recent Chroma versions
        except Exception:
            pass

    # Query through the retriever so the *parent* documents are returned
    # (the raw vectorstore would only give back the child chunks).
    matching_doc = retriever.get_relevant_documents(query)
    return matching_doc


if __name__ == "__main__":
    query = "What is RAG?"
    result = run_agent(query)
    for doc in result:
        print(doc.page_content)
        print("-" * 40)
