"""
Persist a LangChain ParentDocumentRetriever (parents + child chunks) so it can be
restored later WITHOUT re-running `add_documents`.

Root cause of the output mismatch in the original code:
  * `ParentDocumentRetriever` splits the docs into *child chunks* and stores the
    chunks in the vector store (`Chroma`, collection "full_documents") while the
    *parent* documents are kept in the doc store (`InMemoryStore`).
  * The original code then threw the retriever away and queried the Chroma
    collection directly with `max_marginal_relevance_search(...)`. That only
    returns the persisted CHILD CHUNKS, so the user never sees the full parent
    documents they expected.
  * `InMemoryStore` is ephemeral: as soon as the process exits (or the object is
    garbage collected) the parent documents are gone, so the retriever can never
    be restored and `add_documents` must be re-run every time.

Fix:
  * Replace `InMemoryStore` with a persistent byte store backed by disk:
    `create_kv_docstore(LocalFileStore("./docstore_parents"))`.
  * Keep Chroma (with `persist_directory`) for the child chunks.
  * Rebuild the `ParentDocumentRetriever` from those two persisted stores and call
    `retriever.invoke(query)` (MMR search) so the PARENT documents are returned.
  * Only ingest when the doc store is empty, otherwise just restore.
"""

import os

import nltk
from dotenv import load_dotenv

from langchain_chroma import Chroma
from langchain_core.documents import Document
from langchain_core.storage import LocalFileStore, create_kv_docstore
from langchain_huggingface import HuggingFaceEmbeddings
from langchain_text_splitters import NLTKTextSplitter

# ParentDocumentRetriever lives in langchain.retrievers for <1.0 and in
# langchain_classic for >=1.0; support both.
try:
    from langchain.retrievers import ParentDocumentRetriever
except ImportError:  # pragma: no cover
    from langchain_classic.retrievers import ParentDocumentRetriever

load_dotenv()

HF_EMBED_PATH = "./hf_cache"
CHROMA_DIR = "./chroma_db_child"          # persisted child chunks
DOCSTORE_DIR = "./docstore_parents"       # persisted parent documents
COLLECTION_NAME = "full_documents"

DOCS = [
    Document(
        page_content="Artificial Intelligence (AI) is technology that enables "
        "computers and machines to simulate human intelligence."
    ),
    Document(
        page_content="Machine learning (ML) is a type of Artificial Intelligence "
        "(AI) that teaches computers to find patterns in large datasets and make "
        "decisions or predictions without being explicitly programmed for every "
        "scenario, learning and improving from experience, much like humans do."
    ),
    Document(
        page_content="RAG most commonly refers to Retrieval-Augmented Generation, "
        "an AI technique that improves large language models (LLMs) by letting them "
        "access and use external, up-to-date information (like documents or "
        "databases) to generate more accurate, context-aware, and factual answers, "
        "preventing hallucinations and tailoring responses to specific knowledge "
        "bases without needing costly retraining."
    ),
]


def ensure_nltk_data() -> None:
    """NLTKTextSplitter needs punkt (and punkt_tab on recent NLTK versions)."""
    for package in ("punkt", "punkt_tab"):
        try:
            nltk.download(package, quiet=True)
        except Exception:
            # punkt_tab does not exist on older NLTK releases - ignore.
            pass


def get_embedding_function() -> HuggingFaceEmbeddings:
    return HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=HF_EMBED_PATH,
        encode_kwargs={"normalize_embeddings": True},
    )


def get_parent_docstore(path: str = DOCSTORE_DIR):
    """Persistent doc store on disk: parent documents survive process restarts."""
    os.makedirs(path, exist_ok=True)
    return create_kv_docstore(LocalFileStore(path))


def build_retriever(
    embedding_function: HuggingFaceEmbeddings | None = None,
) -> ParentDocumentRetriever:
    """Rebuild the retriever purely from the persisted stores (no ingestion)."""
    embedding_function = embedding_function or get_embedding_function()

    # Child chunks are already persisted here by previous runs.
    vectorstore = Chroma(
        collection_name=COLLECTION_NAME,
        embedding_function=embedding_function,
        persist_directory=CHROMA_DIR,
    )

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=get_parent_docstore(),          # <- persisted, not InMemoryStore
        child_splitter=NLTKTextSplitter(chunk_size=400),
        search_type="mmr",                       # keep the user's MMR behaviour
        search_kwargs={"k": 3},
    )
    return retriever


def docstore_is_empty(retriever: ParentDocumentRetriever) -> bool:
    try:
        return next(iter(retriever.docstore.yield_keys()), None) is None
    except StopIteration:  # pragma: no cover
        return True


def run_agent(query, force_ingest: bool = False):
    """Ingest once, then restore from disk on subsequent calls."""
    ensure_nltk_data()

    retriever = build_retriever()

    # `add_documents` writes child chunks to Chroma and parent docs to the
    # LocalFileStore, so it is only needed when the store is still empty.
    if force_ingest or docstore_is_empty(retriever):
        retriever.add_documents(DOCS)

    # Simulate a later run / new process: rebuild everything from disk only.
    restored_retriever = build_retriever()

    # Returns the FULL PARENT DOCUMENTS (not the raw child chunks).
    matching_docs = restored_retriever.invoke(query)
    return matching_docs


if __name__ == "__main__":
    query = "What is RAG?"
    for i, doc in enumerate(run_agent(query), start=1):
        print(f"--- Parent document {i} ---")
        print(doc.page_content)
