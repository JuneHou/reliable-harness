import os

from dotenv import load_dotenv

from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import LocalFileStore, create_kv_docstore
from langchain.schema import Document

import nltk

load_dotenv()

# --- persisted locations (survive process restarts) ---
CHROMA_DIR = "./chroma_db_child"      # child chunks + embeddings (vectorstore)
DOCSTORE_DIR = "./docstore"           # parent documents (key-value docstore)
HF_EMBED_PATH = "./hf_cache"


def build_persistent_retriever(embedding_function, child_splitter):
    """Build a ParentDocumentRetriever whose *both* halves are persisted on disk.

    * child chunks  -> Chroma with a persist_directory
    * parent docs   -> (LocalFileStore + create_kv_docstore) instead of InMemoryStore
    """
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory=CHROMA_DIR,
    )

    os.makedirs(DOCSTORE_DIR, exist_ok=True)
    fs = LocalFileStore(DOCSTORE_DIR)          # same docstore is reloaded on every run
    store = create_kv_docstore(fs)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter,
        search_kwargs={"k": 3},
    )
    return retriever


def run_agent(query, store_location="./store_location"):
    nltk.download("punkt", quiet=True)

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining.")
    ]

    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=HF_EMBED_PATH,
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    retriever = build_persistent_retriever(embedding_function, child_splitter)

    # Only index the first time. On later runs the parents are read back from
    # ./docstore and the child chunks from ./chroma_db_child, so
    # add_documents() does not have to be called again.
    already_indexed = any(True for _ in retriever.docstore.yield_keys())
    if not already_indexed:
        retriever.add_documents(docs)

    # Explicit persist for older langchain versions (newer Chroma auto-persists).
    persist = getattr(retriever.vectorstore, "persist", None)
    if callable(persist):
        persist()

    # Use the retriever itself: it returns the *full parent documents*
    # that the matching child chunks map to, not the bare/truncated chunks.
    matching_docs = retriever.invoke(query)
    return matching_docs


if __name__ == "__main__":
    query = "What is RAG?"
    result = run_agent(query)
    for doc in result:
        print(doc.page_content)
        print("-" * 80)

def build_documents():
    """Stable ids make re-ingestion idempotent instead of duplicating data."""
    return [
        Document(page_content=d["text"], metadata={"id": d["id"]})
        for d in RAW_DOCS
    ]


def _download_nltk_data():
    for pkg in ("punkt", "punkt_tab"):
        try:
            nltk.download(pkg, quiet=True)
        except Exception:  # offline / already available
            pass


def build_retriever():
    """(Re)create the retriever from the persisted Chroma + docstore dirs.

    Nothing is re-embedded here: the Chroma collection is loaded from disk and
    the parent documents are loaded from the on-disk KV store, so the
    child-chunk -> parent-document mapping still resolves.

        collection_name=COLLECTION_NAME,
        embedding_function=embedding_function,
        persist_directory=CHROMA_DIR,

    store = create_kv_docstore(LocalFileStore(DOCSTORE_DIR))

        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter,
        id_key="id",                       # deterministic doc ids
        search_type="mmr",                 # keep the MMR behaviour the user wanted
        search_kwargs={"k": TOP_K},
    )
    return retriever, vectorstore, store


def docstore_is_populated(store):
    return next(iter(store.yield_keys()), None) is not None


def run_agent(query, rebuild=False, store_location=DOCSTORE_DIR):
    """First call ingests + persists; later calls only load from disk."""
    _download_nltk_data()

    retriever, vectorstore, store = build_retriever()

    if rebuild or not docstore_is_populated(store):

        # the child chunks (with their doc_id pointers) into Chroma.
        retriever.add_documents(build_documents(), add_to_docstore=True)

    # expanded back into their parent documents.
    matching_docs = retriever.invoke(query)
    return matching_docs
