import os

from dotenv import load_dotenv
import nltk

# --- version tolerant imports ---------------------------------------------
try:  # newest home of HuggingFaceEmbeddings
    from langchain_huggingface import HuggingFaceEmbeddings
except ImportError:  # pragma: no cover
    try:
        from langchain_community.embeddings import HuggingFaceEmbeddings
    except ImportError:
        from langchain.embeddings import HuggingFaceEmbeddings

try:
    from langchain_text_splitters import NLTKTextSplitter
except ImportError:  # pragma: no cover
    from langchain.text_splitter import NLTKTextSplitter

try:
    from langchain_chroma import Chroma
except ImportError:  # pragma: no cover
    from langchain_community.vectorstores import Chroma

try:
    from langchain.retrievers import ParentDocumentRetriever
except ImportError:  # pragma: no cover
    from langchain.retrievers.parent_documents import ParentDocumentRetriever

# FIX 1: a *disk backed* key/value docstore instead of InMemoryStore,
#        so the parent documents survive across processes.
from langchain.storage import LocalFileStore, create_kv_docstore

try:
    from langchain_core.documents import Document
except ImportError:  # pragma: no cover
    from langchain.schema import Document


def _download_nltk_data():
    """NLTKTextSplitter needs 'punkt' (and 'punkt_tab' on recent NLTK)."""
    for pkg in ("punkt", "punkt_tab"):
        try:
            nltk.download(pkg, quiet=True)
        except Exception:
            pass


def _collection_count(vectorstore):
    """Number of stored child chunks in the Chroma collection."""
    try:
        return vectorstore._collection.count()
    except Exception:
        try:
            return len(vectorstore.get().get("ids", []))
        except Exception:
            return 0


def _docstore_is_empty(docstore, probe=1):
    """True when the persisted docstore has no parent document yet."""
    try:
        for _ in docstore.yield_keys():
            return False
        return True
    except Exception:
        return True


def build_retriever(
    store_location="./store_location",
    chroma_dir="./chroma_db_child",
    hf_embed_path="./hf_cache",
):
    """Rebuild the retriever from disk only: vectorstore + docstore."""
    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path,
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    # child chunks (already persisted by Chroma thanks to persist_directory)
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory=chroma_dir,
    )

    # FIX 1 (cont.): parents are now stored on disk under ./store_location
    file_store = LocalFileStore(store_location)
    docstore = create_kv_docstore(file_store)

    # FIX 2: search the *parent* documents (not raw child chunks) and keep
    #        the MMR behaviour the user wanted, via the retriever itself.
    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=docstore,
        child_splitter=child_splitter,
        id_key="doc_id",
        search_type="mmr",
        search_kwargs={"k": 3},
    )
    return retriever, vectorstore, docstore


def run_agent(query, store_location="./store_location"):
    load_dotenv()
    _download_nltk_data()

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."),
    ]

    retriever, vectorstore, docstore = build_retriever(store_location=store_location)

    # FIX 3: only index on the very first run. On later runs both the child
    #        chunks (Chroma) and the parent docs (LocalFileStore) are reused,
    #        so add_documents() is skipped entirely.
    if _docstore_is_empty(docstore) or _collection_count(vectorstore) == 0:
        retriever.add_documents(docs)
        # older langchain versions need an explicit persist for Chroma
        if hasattr(vectorstore, "persist"):
            try:
                vectorstore.persist()
            except Exception:
                pass
    else:
        print("[info] loaded existing index - skipping add_documents()")

    # FIX 2 (cont.): retrieve parent documents through the retriever, so the
    #               output is the original full documents rather than the
    #               child chunks returned by a raw MMR search on the store.
    matching_doc = retriever.invoke(query)
    return matching_doc


if __name__ == "__main__":
    query = "What is RAG?"
    result = run_agent(query)
    for i, doc in enumerate(result, start=1):
        print(f"--- parent document {i} (id={doc.id!r}) ---")
        print(doc.page_content)
