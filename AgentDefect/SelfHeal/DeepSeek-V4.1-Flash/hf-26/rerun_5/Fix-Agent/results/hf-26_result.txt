# Fixed: the extra text before the final "Assistant:" answer appears because a
# *text-generation* LLM (`HuggingFaceEndpoint(..., task="text-generation")`) was fed
# a *chat* prompt. A raw text-completion endpoint receives the rendered chat prompt
# as a flat string ("System: ... Human: ... Assistant:") and simply continues that
# text, so Zephyr's chat markers / the whole conversation get echoed back and the
# real answer only shows up after the last "Assistant:".
#
# Fix: wrap the endpoint in `ChatHuggingFace`, which applies the model's chat
# template (and the proper stop tokens), so only the assistant reply is returned.
# A locally defined chat prompt (system + human) is used so the variables line up
# with the model's expected roles, and a light sanitizer removes any residual
# echoed role markers.
#
# https://discuss.huggingface.co/t/getting-additional-response-from-my-rag-using-huggingfaceendpoint-inference/145964
import os

from dotenv import load_dotenv
from langchain.chains.combine_documents import create_stuff_documents_chain
from langchain.chains.retrieval import create_retrieval_chain
from langchain_community.vectorstores import FAISS
from langchain_core.documents import Document
from langchain_core.prompts import ChatPromptTemplate
from langchain_huggingface import (
    ChatHuggingFace,
    HuggingFaceEndpoint,
    HuggingFaceEmbeddings,
)

load_dotenv()

api_token = os.getenv("HUGGINGFACEHUB_API_TOKEN")
if not api_token:
    raise ValueError("HUGGINGFACEHUB_API_TOKEN not found in .env file")
os.environ["HUGGINGFACEHUB_API_TOKEN"] = api_token

SYSTEM_TEMPLATE = (
    "You are an assistant for question-answering tasks. "
    "Use the following pieces of retrieved context to answer the question. "
    "If you don't know the answer, just say that you don't know. "
    "Answer with the answer only: do not repeat the question, the context, "
    "the conversation, or any role markers such as 'System:', 'Human:' or "
    "'Assistant:'.\n\n"
    "Context:\n{context}"
)


def clean_answer(text: str) -> str:
    """Strip echoed prompt/role markers and keep only the final answer."""
    if not text:
        return ""
    answer = text.strip()
    # If the model echoed the chat transcript, keep what follows the last
    # assistant marker.
    for marker in ("<|assistant|>", "Assistant:"):
        if marker in answer:
            answer = answer.rsplit(marker, 1)[-1]
    # Drop any leftover end-of-turn tokens.
    for token in ("</s>", "<|user|>", "<|system|>", "<|endoftext|>"):
        answer = answer.replace(token, " ")
    # Collapse blank lines produced by the removal above.
    answer = "\n".join(line.strip() for line in answer.splitlines() if line.strip())
    return answer.strip()


def run_agent(question: str) -> str:
    documents = [
        Document(
            page_content="Transformers.js uses ONNX Runtime to run models in the browser.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="ONNX Runtime is a cross-platform inference engine.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="Transformers.js enables running transformer models client-side.",
            metadata={"source": "docs"},
        ),
    ]

    embeddings = HuggingFaceEmbeddings(model_name="sentence-transformers/all-MiniLM-L6-v2")
    vectorstore = FAISS.from_documents(documents, embeddings)
    retriever = vectorstore.as_retriever(search_kwargs={"k": 2})

    llm = HuggingFaceEndpoint(
        repo_id="HuggingFaceH4/zephyr-7b-beta",
        task="text-generation",
        temperature=0.5,
        max_new_tokens=512,
        repetition_penalty=1.03,
    )
    # Key fix: use a chat model wrapper, not the raw text-generation LLM, so the
    # chat template (and stop tokens) are applied by the integration itself.
    chat_model = ChatHuggingFace(llm=llm, temperature=0.5, max_new_tokens=512)

    retrieval_qa_chat_prompt = ChatPromptTemplate.from_messages(
        [
            ("system", SYSTEM_TEMPLATE),
            ("human", "{input}"),
        ]
    )

    combine_docs_chain = create_stuff_documents_chain(chat_model, retrieval_qa_chat_prompt)
    rag_chain = create_retrieval_chain(retriever, combine_docs_chain)

    result = rag_chain.invoke({"input": question})
    return clean_answer(result["answer"])


if __name__ == "__main__":
    query = "Which runtime does Transformers.js uses?"
    response = run_agent(query)
    print(f"Input: {query}")
    print(f"answer: {response}")
