import os

from dotenv import load_dotenv
from langchain import hub
from langchain.chains.combine_documents import create_stuff_documents_chain
from langchain.chains.retrieval import create_retrieval_chain
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain_community.vectorstores import FAISS
from langchain_core.documents import Document
from langchain_huggingface import ChatHuggingFace, HuggingFaceEndpoint

load_dotenv()

api_token = os.getenv("HUGGINGFACEHUB_API_TOKEN")
if not api_token:
    raise ValueError("HUGGINGFACEHUB_API_TOKEN not found in .env file")
os.environ["HUGGINGFACEHUB_API_TOKEN"] = api_token


def run_agent(question: str):
    documents = [
        Document(
            page_content="Transformers.js uses ONNX Runtime to run models in the browser.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="ONNX Runtime is a cross-platform inference engine.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="Transformers.js enables running transformer models client-side.",
            metadata={"source": "docs"},
        ),
    ]

    embeddings = HuggingFaceEmbeddings(model_name="sentence-transformers/all-MiniLM-L6-v2")
    vectorstore = FAISS.from_documents(documents, embeddings)
    retriever = vectorstore.as_retriever(search_kwargs={"k": 2})

    # 1) Raw text-generation endpoint (no chat formatting of its own).
    llm = HuggingFaceEndpoint(
        repo_id="huggingfaceh4/zephyr-7b-beta",
        task="text-generation",
        temperature=0.5,
        max_new_tokens=1024,
        huggingfacehub_api_token=api_token,
        # Make sure the model stops instead of writing further turns
        # ("Human:", "Assistant:", ...) after answering.
        stop_sequences=["\nHuman:", "\nSystem:", "</s>"],
    )

    # 2) Wrap it in ChatHuggingFace so the chat prompt is rendered with the
    #    model's chat template (<|system|> ... <|user|> ... <|assistant|>)
    #    instead of being flattened into "System: ... / Human: ..." text.
    #    That flattened text is what the model kept continuing, producing the
    #    extra text before the final "Assistant:" answer.
    chat_model = ChatHuggingFace(llm=llm, temperature=0.5, max_new_tokens=1024)

    retrieval_qa_chat_prompt = hub.pull("langchain-ai/retrieval-qa-chat")

    # 3) Pass the *chat* model to the documents chain.
    combine_docs_chain = create_stuff_documents_chain(chat_model, retrieval_qa_chat_prompt)
    rag_chain = create_retrieval_chain(retriever, combine_docs_chain)

    result = rag_chain.invoke({"input": question})
    answer = result["answer"]

    # Safety net: strip any leftover role markers / verbose payload the model
    # may still emit (e.g. a trailing "Assistant:" prefix).
    if "Assistant:" in answer:
        answer = answer.rsplit("Assistant:", 1)[-1]
    return answer.strip()


if __name__ == "__main__":
    query = "Which runtime does Transformers.js uses?"
    response = run_agent(query)
    print(f"Input: {query}")
    print(f"answer: {response}")

### 1. Diff review — diagnosis is correct

The original bug was real and well understood in the diff:

- `create_stuff_documents_chain(llm, retrieval_qa_chat_prompt)` was given a **plain `HuggingFaceEndpoint` (an `LLM`, not a chat model)** together with a **chat prompt** (`langchain-ai/retrieval-qa-chat`).
- In the `BaseLLM` path, LangChain has to flatten the `ChatPromptValue` into a plain string via `to_string()`, producing text like `System: ...\nHuman: ...\nAssistant:`. The model then *continues that transcript*, emitting further `Human:`/`Assistant:` turns — exactly the "additional response" noise reported in the linked forum thread.
- The fix addresses the root cause: wrap the endpoint in `ChatHuggingFace` so the model's own chat template (`<|system|>…<|user|>…<|assistant|>`) is applied, pass the chat model to `create_stuff_documents_chain`, and add `stop_sequences` to halt at role markers.
- The trailing `if "Assistant:" in answer: answer = answer.rsplit("Assistant:", 1)[-1]` plus `.strip()` is a defensive cleanup and cannot throw on a `str`.
