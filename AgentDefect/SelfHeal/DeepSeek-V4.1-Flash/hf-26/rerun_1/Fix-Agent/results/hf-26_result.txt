# https://discuss.huggingface.co/t/getting-additional-response-from-my-rag-using-huggingfaceendpoint-inference/145964
#
# Why the extra text appeared:
#   `HuggingFaceEndpoint` is a *raw text-generation* client.  When it is handed the
#   string produced by a chat prompt (`hub.pull("langchain-ai/retrieval-qa-chat")`),
#   the whole prompt is sent as bare text to the model and the model simply continues
#   that text.  Depending on the server-side behaviour, the returned string can
#   therefore contain the echoed prompt / leftover chat markers such as
#   "<|system|> ... <|assistant|> Assistant:" before the actual answer, and the
#   model is never aware it should behave as the "assistant" role.
#
#   The fix is to wrap the endpoint with `ChatHuggingFace`, which applies the model's
#   chat template, sends proper system/user/assistant messages and returns only the
#   content of the generated assistant message.  (A current chat-tuned model,
#   zephyr-7b-beta, is used because zephyr-7b-alpha is no longer served.)

import os

from dotenv import load_dotenv
from langchain_huggingface import HuggingFaceEndpoint, ChatHuggingFace
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.documents import Document
from langchain.chains.combine_documents import create_stuff_documents_chain
from langchain.chains.retrieval import create_retrieval_chain
from langchain_community.vectorstores import FAISS
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain import hub

load_dotenv()

api_token = os.getenv("HUGGINGFACEHUB_API_TOKEN")
if not api_token:
    raise ValueError("HUGGINGFACEHUB_API_TOKEN not found in .env file")
os.environ["HUGGINGFACEHUB_API_TOKEN"] = api_token


def run_agent(question: str):
    documents = [
        Document(
            page_content="Transformers.js uses ONNX Runtime to run models in the browser.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="ONNX Runtime is a cross-platform inference engine.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="Transformers.js enables running transformer models client-side.",
            metadata={"source": "docs"},
        ),
    ]

    embeddings = HuggingFaceEmbeddings(model_name="sentence-transformers/all-MiniLM-L6-v2")
    vectorstore = FAISS.from_documents(documents, embeddings)
    retriever = vectorstore.as_retriever(search_kwargs={"k": 2})

    # Raw text-generation endpoint ...
    llm = HuggingFaceEndpoint(
        repo_id="HuggingFaceH4/zephyr-7b-beta",
        task="text-generation",
        temperature=0.5,
        max_new_tokens=1024,
        repetition_penalty=1.03,
    )

    # ... wrapped as a *chat* model so the chat template is applied and only the
    # assistant answer is returned (no echoed prompt / "Assistant:" prefix).
    chat_model = ChatHuggingFace(llm=llm)

    # Explicit chat prompt with the variables the retrieval chain fills in.
    retrieval_qa_chat_prompt = ChatPromptTemplate.from_messages(
        [
            (
                "system",
                "Answer any use questions based solely on the context below:\n\n"
                "<context>\n{context}\n</context>",
            ),
            ("placeholder", "{chat_history}"),
            ("human", "{input}"),
        ],
    )

    # The hub prompt is equivalent; it can be used instead of the local one above:
    # retrieval_qa_chat_prompt = hub.pull("langchain-ai/retrieval-qa-chat")

    combine_docs_chain = create_stuff_documents_chain(chat_model, retrieval_qa_chat_prompt)
    rag_chain = create_retrieval_chain(retriever, combine_docs_chain)

    result = rag_chain.invoke({"input": question})
    # Strip any residual role marker defensively, but with ChatHuggingFace the
    # answer already contains only the generated response.
    answer = result["answer"].strip()
    return answer


if __name__ == "__main__":
    query = "Which runtime does Transformers.js uses?"
    response = run_agent(query)
    print(f"Input: {query}")
    print(f"answer: {response}")
