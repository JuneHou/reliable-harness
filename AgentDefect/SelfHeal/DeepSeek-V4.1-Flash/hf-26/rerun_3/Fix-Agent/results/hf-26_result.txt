# Fixed version:
# The extra text before the final "Assistant:" answer happened because the
# retrieval-qa-chat prompt is a *chat* prompt (system/human roles) but it was
# being stuffed straight into a raw text-generation endpoint
# (HuggingFaceEndpoint). The endpoint receives the whole rendered transcript
# ("System: ... Human: ... Assistant:") as plain text and simply *continues*
# it, so the generated text contains the echoed prompt/instructions before it
# finally writes the answer for the Assistant turn.
#
# Fixes applied:
#   1. Wrap the endpoint with ChatHuggingFace so the model's own chat template
#      is applied (roles handled properly, no prompt echoed back).
#   2. Use an explicit, well-formed ChatPromptTemplate (system + human) with a
#      correct {context}/{input} placeholders and short "answer only" instruction.
#   3. Use a supported instruct/chat model (zephyr-7b-beta instead of the
#      decommissioned alpha) and sane generation params.
#   4. Sanitize the returned answer as a safety net for leftover scaffolding.

import os

from dotenv import load_dotenv

from langchain_huggingface import HuggingFaceEndpoint, ChatHuggingFace
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.documents import Document
from langchain_core.output_parsers import StrOutputParser
from langchain.chains.combine_documents import create_stuff_documents_chain
from langchain.chains.retrieval import create_retrieval_chain
from langchain_community.vectorstores import FAISS
from langchain_community.embeddings import HuggingFaceEmbeddings

load_dotenv()

api_token = os.getenv("HUGGINGFACEHUB_API_TOKEN")
if not api_token:
    raise ValueError("HUGGINGFACEHUB_API_TOKEN not found in .env file")
os.environ["HUGGINGFACEHUB_API_TOKEN"] = api_token


SYSTEM_PROMPT = (
    "You are an assistant for question-answering tasks. "
    "Use the following pieces of retrieved context to answer the question. "
    "If you don't know the answer, just say that you don't know. "
    "Answer in one or two short sentences. "
    "Do not repeat the question, do not restate these instructions and do not "
    "write 'Assistant:' or 'Answer:' in your reply.\n\n"
    "Context:\n{context}"
)


def _clean_answer(text: str) -> str:
    """Remove any echoed prompt scaffolding the model may still emit."""
    if not text:
        return ""
    cleaned = text.strip()
    for marker in ("Assistant:", "Answer:"):
        if marker in cleaned:
            cleaned = cleaned.rsplit(marker, 1)[-1]
    # Drop a repeated restatement of the question/system block if present.
    for marker in ("</s>", "<|assistant|>", "<|system|>", "<|user|>"):
        cleaned = cleaned.replace(marker, "")
    return cleaned.strip()


def run_agent(question: str) -> str:
    documents = [
        Document(
            page_content="Transformers.js uses ONNX Runtime to run models in the browser.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="ONNX Runtime is a cross-platform inference engine.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="Transformers.js enables running transformer models client-side.",
            metadata={"source": "docs"},
        ),
    ]

    embeddings = HuggingFaceEmbeddings(model_name="sentence-transformers/all-MiniLM-L6-v2")
    vectorstore = FAISS.from_documents(documents, embeddings)
    retriever = vectorstore.as_retriever(search_kwargs={"k": 2})

    # Raw text-generation endpoint ...
    endpoint = HuggingFaceEndpoint(
        repo_id="huggingfaceh4/zephyr-7b-beta",
        task="text-generation",
        temperature=0.5,
        max_new_tokens=256,
        repetition_penalty=1.03,
        huggingfacehub_api_token=api_token,
    )

    # ... wrapped as a chat model so the chat template/roles are applied and the
    # prompt is not echoed back into the answer.
    llm = ChatHuggingFace(llm=endpoint)

    retrieval_qa_chat_prompt = ChatPromptTemplate.from_messages(
        [
            ("system", SYSTEM_PROMPT),
            ("human", "{input}"),
        ]
    )

    combine_docs_chain = create_stuff_documents_chain(
        llm,
        retrieval_qa_chat_prompt,
        output_parser=StrOutputParser(),
    )
    rag_chain = create_retrieval_chain(retriever, combine_docs_chain)

    result = rag_chain.invoke({"input": question})
    return _clean_answer(result["answer"])


if __name__ == "__main__":
    query = "Which runtime does Transformers.js uses?"
    response = run_agent(query)
    print(f"Input: {query}")
    print(f"answer: {response}")
