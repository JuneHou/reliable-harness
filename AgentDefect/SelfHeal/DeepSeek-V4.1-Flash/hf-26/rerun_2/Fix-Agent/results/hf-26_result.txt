# https://discuss.huggingface.co/t/getting-additional-response-from-my-rag-using-huggingfaceendpoint-inference/145964
#
# Why the extra text showed up:
#  * `HuggingFaceEndpoint(task="text-generation")` is a *raw text completion*
#    model, not a chat model. `create_stuff_documents_chain` therefore flattens
#    whatever prompt it receives into ONE plain string and the model simply
#    *continues* that string.
#  * The prompt used (`hub.pull("langchain-ai/retrieval-qa-chat")`) plus the
#    default chat template of `zephyr-7b-alpha` leaves the model in "repeat the
#    conversation" mode: it re-prints the instructions / context / question that
#    it was given and only *then* writes "Assistant: <real answer>".
#  * So `result["answer"]` contained the echoed prompt + "Assistant:" before the
#    actual answer.
#
# The fix:
#  1. Use an explicit `ChatPromptTemplate` with clear roles and an explicit
#     instruction to answer only with the final answer (no preamble, no
#     restating the question/context, no role labels).
#  2. End the prompt with a single, unambiguous answer cue.
#  3. Defensively sanitize the raw completion, so that even if the endpoint
#     echoes the prompt (or the model still emits "Assistant:"), only the final
#     answer is returned.

import os
import re

from dotenv import load_dotenv

from langchain_huggingface import HuggingFaceEndpoint
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.documents import Document
from langchain.chains.combine_documents import create_stuff_documents_chain
from langchain.chains.retrieval import create_retrieval_chain
from langchain_community.vectorstores import FAISS
from langchain_community.embeddings import HuggingFaceEmbeddings

load_dotenv()

api_token = os.getenv("HUGGINGFACEHUB_API_TOKEN")
if not api_token:
    raise ValueError("HUGGINGFACEHUB_API_TOKEN not found in .env file")
os.environ["HUGGINGFACEHUB_API_TOKEN"] = api_token


# ---------------------------------------------------------------------------
# Output sanitizing helpers
# ---------------------------------------------------------------------------
# Role / turn markers a text-generation model may echo or re-emit.
_TURN_MARKERS = (
    "<|assistant|>",
    "<|system|>",
    "<|user|>",
    "assistant:",
    "answer:",
)

_LEADING_ROLE_RE = re.compile(
    r"^\s*(?:<\|assistant\|>|assistant\s*:|a\s*:)\s*", re.IGNORECASE
)


def _looks_like_echoed_prompt(text: str, question: str) -> bool:
    """True when the model re-printed the prompt it was given."""
    lowered = text.lower()
    if question and question.strip().lower() in lowered:
        return True
    return any(marker in lowered for marker in ("<|system|>", "<|user|>", "context:", "human:"))


def clean_answer(raw_answer: str, question: str = "") -> str:
    """Return only the final answer produced by the model.

    Strips an echoed prompt and any leading role label ("Assistant:") so that
    the caller sees just the answer text.
    """
    if not raw_answer:
        return raw_answer

    text = raw_answer.strip()

    # 1. If the prompt was echoed back, drop everything up to and including the
    #    last turn marker that belongs to the echoed prompt.
    if _looks_like_echoed_prompt(text, question):
        lowered = text.lower()
        cut_at = -1
        marker_len = 0
        for marker in _TURN_MARKERS:
            idx = lowered.rfind(marker)
            if idx > cut_at:
                cut_at = idx
                marker_len = len(marker)
        if cut_at != -1:
            candidate = text[cut_at + marker_len:].strip()
            if candidate:
                text = candidate

    # 2. Drop any residual leading role label, e.g. "Assistant: foo".
    previous = None
    while previous != text:
        previous = text
        text = _LEADING_ROLE_RE.sub("", text).strip()

    return text


def run_agent(question: str) -> str:
    documents = [
        Document(
            page_content="Transformers.js uses ONNX Runtime to run models in the browser.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="ONNX Runtime is a cross-platform inference engine.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="Transformers.js enables running transformer models client-side.",
            metadata={"source": "docs"},
        ),
    ]

    embeddings = HuggingFaceEmbeddings(model_name="sentence-transformers/all-MiniLM-L6-v2")
    vectorstore = FAISS.from_documents(documents, embeddings)
    retriever = vectorstore.as_retriever(search_kwargs={"k": 2})

    llm = HuggingFaceEndpoint(
        repo_id="huggingfaceh4/zephyr-7b-alpha",
        task="text-generation",
        temperature=0.1,
        max_new_tokens=1024,
    )

    # Explicit, well-scoped prompt: a single human turn with everything the
    # model needs and an instruction to output the answer only.
    retrieval_qa_chat_prompt = ChatPromptTemplate.from_messages(
        [
            (
                "human",
                "You are a helpful assistant for question-answering tasks.\n"
                "Use ONLY the pieces of retrieved context below to answer the question. "
                "If you don't know the answer, say that you don't know.\n"
                "Reply with the final answer only: do not repeat the question, "
                "do not repeat the context, do not add any preamble, and do not "
                "include role labels such as 'Assistant:'.\n\n"
                "Context:\n{context}\n\n"
                "Question: {input}\n\n"
                "Answer:",
            )
        ]
    )

    combine_docs_chain = create_stuff_documents_chain(llm, retrieval_qa_chat_prompt)
    rag_chain = create_retrieval_chain(retriever, combine_docs_chain)

    result = rag_chain.invoke({"input": question})
    return clean_answer(result["answer"], question)


if __name__ == "__main__":
    query = "Which runtime does Transformers.js uses?"
    response = run_agent(query)
    print(f"Input: {query}")
    print(f"answer: {response}")
