"""Corrected version of the GPT-4 PDF / MongoDB Atlas vector search demo.

Why the original returned an empty list
---------------------------------------
`atlas_collection` was a plain Python list (`atlas_collection = []`) that shadowed
the real `MongoDBAtlasVectorSearch` class with a dummy stub, so:

  * nothing was actually written to MongoDB (the list is not a Collection),
  * no connection to the cluster/database/collection was ever made,
  * no Atlas Vector Search index (`vector_index`) was ever created / confirmed
    to be *queryable*,
  * and `similarity_search` was executed against that in-memory stub, which can
    only ever return `[]`.

The fix is to route the writes and the reads to the *same* real MongoDB Atlas
collection and to the *same*, fully-built vector search index, with an embedding
field / dimension that matches the embedding model, and to wait until Atlas has
finished building the index before querying it.
"""

from dotenv import load_dotenv

import os
import time

from pymongo import MongoClient

from langchain_community.document_loaders import PyPDFLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings

# langchain-mongodb >= 0.1 exports the vector store and the index helper.
try:
    from langchain_mongodb import MongoDBAtlasVectorSearch
    from langchain_mongodb.index import create_vector_search_index

    _HAS_INDEX_HELPER = True
except ImportError:  # older installations
    from langchain_community.vectorstores import MongoDBAtlasVectorSearch

    create_vector_search_index = None
    _HAS_INDEX_HELPER = False


EMBEDDING_MODEL = "text-embedding-ada-002"
EMBEDDING_DIMENSIONS = 1536          # must match EMBEDDING_MODEL

VECTOR_INDEX_NAME = "vector_index"
EMBEDDING_FIELD = "embedding"        # field the vector index points at
TEXT_FIELD = "text"

MONGODB_URI_ENV = "MONGODB_ATLAS_CLUSTER_URI"
DB_NAME_ENV = "MONGODB_DB_NAME"
COLLECTION_NAME_ENV = "MONGODB_COLLECTION_NAME"


def get_env(name, default=None):
    value = os.getenv(name, default)
    if value is None or value == "":
        raise ValueError(f"Environment variable {name} not found")
    return value


def get_collection():
    """Return the *real* MongoDB Atlas collection used for both write and read.

    This is what the original code was missing: a Python list can never be a
    MongoDB collection, therefore nothing was persisted and every similarity
    search came back empty.
    """
    uri = get_env(MONGODB_URI_ENV)
    db_name = get_env(DB_NAME_ENV, "langchain_db")
    collection_name = get_env(COLLECTION_NAME_ENV, "gpt4_paper")

    client = MongoClient(uri)
    return client[db_name][collection_name]


def wait_until_index_is_queryable(collection, index_name, timeout=300, interval=5):
    """Atlas builds vector indexes asynchronously.

    Querying an index that is still in the BUILDING state (or that was never
    created at all) is another classic reason for an empty result set, so poll
    the index status until it is `queryable`.
    """
    deadline = time.time() + timeout
    while time.time() < deadline:
        try:
            indexes = list(collection.list_search_indexes())
        except Exception:
            indexes = []

        for index in indexes:
            if index.get("name") != index_name:
                continue
            if index.get("queryable"):
                return True
            if index.get("status") == "FAILED":
                raise RuntimeError(
                    f"Vector search index '{index_name}' failed to build: {index}"
                )
        time.sleep(interval)
    return False


def ensure_vector_index(collection, index_name=VECTOR_INDEX_NAME):
    """Create the vector search index if needed and make sure it is usable."""
    exists = any(
        index.get("name") == index_name
        for index in collection.list_search_indexes()
    )

    if not exists:
        definition = {
            "name": index_name,
            "type": "vectorSearch",
            "fields": [
                {
                    "type": "vector",
                    "path": EMBEDDING_FIELD,
                    "numDimensions": EMBEDDING_DIMENSIONS,
                    "similarity": "cosine",
                },
                {"type": "filter", "path": "source"},
            ],
        }

        if _HAS_INDEX_HELPER:
            create_vector_search_index(
                collection=collection,
                index_name=index_name,
                dimensions=EMBEDDING_DIMENSIONS,
                path=EMBEDDING_FIELD,
                similarity="cosine",
                wait_until_complete=60,
            )
        else:
            collection.database.command(
                {
                    "createSearchIndexes": collection.name,
                    "indexes": [definition],
                }
            )

    if not wait_until_index_is_queryable(collection, index_name):
        raise RuntimeError(
            f"Vector search index '{index_name}' on "
            f"'{collection.database.name}.{collection.name}' is not queryable yet. "
            "Wait for Atlas to finish building it."
        )


def run_agent(pdf_url):
    load_dotenv()

    if os.getenv("OPENAI_API_KEY") is None:
        raise ValueError("OPENAI_API_KEY not found")

    # 1) Real MongoDB Atlas collection (not an empty Python list).
    collection = get_collection()

    # 2) Same embedding model is used for indexing and for the query.
    embeddings = OpenAIEmbeddings(
        model=EMBEDDING_MODEL,
        disallowed_special=(),
    )

    # 3) Load + chunk the PDF.
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100,
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()

    docs = text_splitter.split_documents(data)
    if not docs:
        raise ValueError(f"No text could be extracted from {pdf_url}")

    # 4) The vector search index must exist and be queryable before searching.
    ensure_vector_index(collection, VECTOR_INDEX_NAME)

    # 5) Only embed/insert when the collection does not already hold the vectors,
    #    so repeated runs query the data instead of duplicating it.
    already_indexed = (
        collection.count_documents({EMBEDDING_FIELD: {"$exists": True}}, limit=1) > 0
    )

    if already_indexed:
        vector_search = MongoDBAtlasVectorSearch(
            collection=collection,
            embedding=embeddings,
            index_name=VECTOR_INDEX_NAME,
            embedding_key=EMBEDDING_FIELD,
            text_key=TEXT_FIELD,
            relevance_score_fn="cosine",
        )
    else:
        vector_search = MongoDBAtlasVectorSearch.from_documents(
            documents=docs,
            embedding=embeddings,
            collection=collection,
            index_name=VECTOR_INDEX_NAME,
            embedding_key=EMBEDDING_FIELD,
            text_key=TEXT_FIELD,
            relevance_score_fn="cosine",
        )

    query = "What were the compute requirements for training GPT 4"

    results = vector_search.similarity_search(query, k=4)

    if not results:
        raise RuntimeError(
            "Similarity search returned no documents. Check that the documents "
            f"were inserted into {collection.database.name}.{collection.name} "
            f"and that the Atlas index '{VECTOR_INDEX_NAME}' is queryable and "
            f"indexes '{EMBEDDING_FIELD}' with {EMBEDDING_DIMENSIONS} dimensions."
        )

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"

    docs_found = run_agent(url)

    print(f"Result: {len(docs_found)} document(s)")
    for i, doc in enumerate(docs_found, start=1):
        print(f"\n--- Result {i} ---")
        print(doc.page_content)
