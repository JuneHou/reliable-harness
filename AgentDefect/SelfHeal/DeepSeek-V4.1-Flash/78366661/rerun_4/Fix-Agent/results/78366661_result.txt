"""Query a GPT-4 report PDF through MongoDB Atlas Vector Search.

Why the original script printed an empty list
---------------------------------------------
The documents (and therefore the embeddings) definitely reached the collection,
but an Atlas *Vector Search index* is not created automatically when you write
documents with ``MongoDBAtlasVectorSearch.from_documents``.  The search stage
``$vectorSearch`` fails silently (returning zero candidates) when:

* no search index with the requested ``index_name`` exists on the collection,
* the index exists but is still building / not yet queryable,
* the index is defined on a different ``path`` (e.g. ``embedding``) or a
  different ``numDimensions`` than the embedding model actually produces,
* the query is executed against a different collection/database than the one
  that holds the data.

So the real fix is to use a real MongoClient/collection (not an in-memory
list), explicitly create the vector search index with the right definition and
wait until it is queryable, and only then run ``similarity_search``.
"""

from __future__ import annotations

import os
import time

from dotenv import load_dotenv
from pymongo import MongoClient

from langchain_community.document_loaders import PyPDFLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_mongodb import MongoDBAtlasVectorSearch

VECTOR_INDEX_NAME = "vector_index"
EMBEDDING_MODEL = "text-embedding-3-small"
EMBEDDING_DIMENSIONS = 1536  # must match the model AND the Atlas index definition


def get_atlas_collection():
    """Connect to MongoDB Atlas and return the collection used by the vector store."""
    load_dotenv()

    mongo_uri = os.getenv("MONGODB_ATLAS_CLUSTER_URI") or os.getenv("MONGODB_URI")
    if not mongo_uri:
        raise ValueError("MONGODB_ATLAS_CLUSTER_URI is not set")

    db_name = os.getenv("MONGODB_DB_NAME", "langchain_db")
    collection_name = os.getenv("MONGODB_COLLECTION_NAME", "gpt4_report")

    client = MongoClient(mongo_uri)
    return client[db_name][collection_name]


def _index_is_queryable(index: dict) -> bool:
    """Handle both old (``queryable``) and new (``status``) driver formats."""
    if "queryable" in index:
        return bool(index["queryable"])
    return str(index.get("status", "")).upper() == "READY"


def ensure_vector_search_index(
    collection,
    index_name: str = VECTOR_INDEX_NAME,
    dimensions: int = EMBEDDING_DIMENSIONS,
    path: str = "embedding",
    timeout_seconds: int = 300,
) -> None:
    """Create the Atlas Vector Search index if needed and wait until it can be queried.

    This is the step that was missing: without a searchable ``vectorSearch``
    index, ``similarity_search`` returns ``[]`` even though the data is stored.
    """
    existing = {idx.get("name"): idx for idx in collection.list_search_indexes()}
    if index_name not in existing:
        collection.create_search_index(
            {
                "name": index_name,
                "type": "vectorSearch",
                "definition": {
                    "fields": [
                        {
                            "type": "vector",
                            "path": path,
                            "numDimensions": dimensions,
                            "similarity": "cosine",
                        }
                    ]
                },
            }
        )

    deadline = time.time() + timeout_seconds
    while time.time() < deadline:
        indexes = {idx.get("name"): idx for idx in collection.list_search_indexes()}
        current = indexes.get(index_name)
        if current is not None and _index_is_queryable(current):
            return
        time.sleep(3)

    raise TimeoutError(
        f"Atlas vector search index {index_name!r} did not become queryable "
        f"within {timeout_seconds} seconds"
    )


def run_agent(pdf_url: str, query: str):
    """Load the PDF, store its chunks with embeddings and run a vector search."""
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key is None:
        raise ValueError("OPENAI_API_KEY not found")

    embeddings = OpenAIEmbeddings(
        model=EMBEDDING_MODEL,
        disallowed_special=(),
        api_key=openai_api_key,
    )

    atlas_collection = get_atlas_collection()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100,
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()
    docs = text_splitter.split_documents(data)

    # 1) Make sure a searchable vector index exists *before* querying.
    ensure_vector_search_index(
        collection=atlas_collection,
        index_name=VECTOR_INDEX_NAME,
        dimensions=EMBEDDING_DIMENSIONS,
        path="embedding",
    )

    # 2) Store the chunks + embeddings in Atlas.
    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=embeddings,
        collection=atlas_collection,
        index_name=VECTOR_INDEX_NAME,
    )

    # 3) Now the $vectorSearch stage has an index to use, so results are returned.
    results = vector_search.similarity_search(query, k=4)

    if not results:
        raise RuntimeError(
            "Vector search returned no documents. Check that the index "
            f"{VECTOR_INDEX_NAME!r} is defined on 'embedding' with "
            f"numDimensions={EMBEDDING_DIMENSIONS} and is in READY state, "
            "and that documents were written to the same collection."
        )

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"
    query = "What were the compute requirements for training GPT 4"

    results = run_agent(url, query)

    print("Result:", results)
    for i, doc in enumerate(results, start=1):
        print(f"--- chunk {i} ---")
        print(doc.page_content[:500])
