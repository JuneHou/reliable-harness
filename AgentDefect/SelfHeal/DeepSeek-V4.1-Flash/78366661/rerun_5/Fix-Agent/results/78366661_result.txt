from dotenv import load_dotenv
import os

from langchain_community.document_loaders import PyPDFLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import MongoDBAtlasVectorSearch

from pymongo import MongoClient


def run_agent(pdf_url):
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key is None:
        raise ValueError("OPENAI_API_KEY not found")

    # The vector store must be backed by a REAL MongoDB Atlas collection.
    # Passing a Python list (e.g. `atlas_collection = []`) silently swallows the
    # documents: the embeddings are never persisted and no Atlas `$vectorSearch`
    # index is ever queried, so `similarity_search` always returns `[]`.
    mongodb_uri = os.getenv("MONGODB_ATLAS_CLUSTER_URI")
    if mongodb_uri is None:
        raise ValueError("MONGODB_ATLAS_CLUSTER_URI not found")

    db_name = os.getenv("MONGODB_DB_NAME", "langchain_db")
    collection_name = os.getenv("MONGODB_COLLECTION_NAME", "langchain_collection")

    client = MongoClient(mongodb_uri)
    atlas_collection = client[db_name][collection_name]

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()

    docs = text_splitter.split_documents(data)

    if not docs:
        raise ValueError("No text chunks were extracted from the PDF")

    vector_search_index = "vector_index"

    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(disallowed_special=()),
        collection=atlas_collection,
        index_name=vector_search_index
    )

    query = "What were the compute requirements for training GPT 4"

    # k must be passed explicitly; the default only returns a few documents.
    results = vector_search.similarity_search(query, k=4)

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"

    results = run_agent(url)

    print("Result:", results)
