from dotenv import load_dotenv
import os

from pymongo import MongoClient

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import MongoDBAtlasVectorSearch


def get_atlas_collection():
    """Connect to MongoDB Atlas and return the target collection.

    The vector store MUST be backed by a real pymongo collection that holds the
    documents/embeddings; an empty list (or any other placeholder) cannot be
    queried and always yields zero results.
    """
    mongodb_uri = os.getenv("MONGODB_ATLAS_CLUSTER_URI")
    if mongodb_uri is None:
        raise ValueError("MONGODB_ATLAS_CLUSTER_URI not found")

    db_name = os.getenv("MONGODB_DB_NAME", "langchain_db")
    collection_name = os.getenv("MONGODB_COLLECTION_NAME", "documents")

    client = MongoClient(mongodb_uri)
    collection = client[db_name][collection_name]

    # Optional sanity check: warn if the index was not created in Atlas yet.
    return collection


def run_agent(pdf_url):
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key is None:
        raise ValueError("OPENAI_API_KEY not found")

    atlas_collection = get_atlas_collection()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()

    docs = text_splitter.split_documents(data)

    vector_search_index = "vector_index"

    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(disallowed_special=()),
        collection=atlas_collection,
        index_name=vector_search_index
    )

    # Make sure the writes are visible before querying (Atlas is eventually
    # consistent for freshly created indexes/collections).
    count = atlas_collection.count_documents({})
    print("Documents stored in Atlas collection:", count)
    if count == 0:
        raise RuntimeError(
            "No documents were stored in the Atlas collection. "
            "Check the connection string, database/collection names and "
            "that the vector_index exists in Atlas."
        )

    query = "What were the compute requirements for training GPT 4"

    results = vector_search.similarity_search(query, k=4)

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"

    results = run_agent(url)

    print("Result:", results)
