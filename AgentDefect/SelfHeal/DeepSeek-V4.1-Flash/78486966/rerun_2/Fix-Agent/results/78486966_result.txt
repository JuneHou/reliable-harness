from pathlib import Path

# Fix: CharacterTextSplitter defaults to separator="\n\n". The input text has no
# blank line in it, so the separator is never found and the splitter returns the
# whole text as ONE chunk (silently ignoring chunk_size/chunk_overlap).
# Use RecursiveCharacterTextSplitter (splits on ["\n\n", "\n", " ", ""] in order
# and only falls back when needed) so 30/5 really is honoured, and also fall back
# to the empty-string separator so any text can always be cut at chunk_size.
from langchain_text_splitters import RecursiveCharacterTextSplitter


def run_agent(text_raw):
    text_raw = " ".join(line.strip() for line in text_raw.strip().splitlines())

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=30,
        chunk_overlap=5,
        separators=["\n\n", "\n", " ", ""],  # "" guarantees hard splitting
        keep_separator=True,
    )
    chunks = text_splitter.split_text(text_raw)
    return chunks


def save_chunks_locally(chunks, file_path="chunks.txt"):
    """Save the chunks to disk with plain Python (no extra dependencies)."""
    path = Path(file_path).resolve()
    with path.open("w", encoding="utf-8") as f:
        for i, chunk in enumerate(chunks):
            f.write(f"[chunk {i}] len={len(chunk)}\n{chunk}\n\n")
    return path


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""

    chunks = run_agent(text_raw)

    for i, chunk in enumerate(chunks):
        print(f"chunk {i} (len={len(chunk)}): {chunk!r}")
    print(f"\n\n {len(chunks)}")

    saved_to = save_chunks_locally(chunks)
    print(f"chunks saved to: {saved_to}")

    # Also verify the file that was written
    print(Path(saved_to).read_text(encoding="utf-8"))
