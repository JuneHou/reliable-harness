import os

from langchain_text_splitters import RecursiveCharacterTextSplitter


def run_agent(text_raw):
    """Split a long text into ~30 character chunks with 5 characters of overlap.

    `CharacterTextSplitter` defaults to `separator="\n\n"`, and it only splits on
    that single separator. Because the sample text contains no blank lines, the
    splitter produced exactly one huge piece and then (since that piece was
    smaller than the "look at the whole text" merge step) returned it as a
    single chunk. Using `RecursiveCharacterTextSplitter` (or explicitly setting
    `separator=" "`) makes the text actually get split into many small chunks.
    """
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=30,
        chunk_overlap=5,
        length_function=len,
        separators=["\n\n", "\n", " ", ""],
        keep_separator=False,
        strip_whitespace=True,
    )
    chunks = text_splitter.split_text(text_raw)
    return chunks


def save_chunks(chunks, path="water_chunks.txt"):
    """Save the resulting chunks locally, one chunk per line."""
    with open(path, "w", encoding="utf-8") as f:
        for i, chunk in enumerate(chunks, start=1):
            f.write(f"{i}: {chunk}\n")
    return os.path.abspath(path)


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)
    for i, chunk in enumerate(chunks, start=1):
        print(f"{i}: ({len(chunk)}) {chunk!r}")

    print(f'\n\n {len(chunks)} chunks')

    saved_to = save_chunks(chunks)
    print(f"Chunks saved to: {saved_to}")
