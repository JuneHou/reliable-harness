from pathlib import Path

from langchain_text_splitters import RecursiveCharacterTextSplitter


def run_agent(text_raw: str) -> list[str]:
    """Split a long text into ~30 character chunks with 5 characters of overlap.

    `CharacterTextSplitter` splits on the separator "\\n\\n" by default and only
    *merges* the pieces afterwards. The sample text is one single block (it only
    contains single "\\n" characters), so there is nothing to merge and the whole
    text is returned as ONE chunk, no matter what `chunk_size` is set to.

    `RecursiveCharacterTextSplitter` actually enforces `chunk_size`: it tries the
    separators in order (paragraph -> line -> sentence -> word -> character) and
    recurses into any piece that is still too long.
    """
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=30,
        chunk_overlap=5,
        length_function=len,
        separators=["\n\n", "\n", ". ", " ", ""],
    )
    return text_splitter.split_text(text_raw)


def save_chunks(chunks: list[str], file_path: str = "chunks.txt") -> Path:
    """Save the chunks locally using plain Python (no extra dependencies)."""
    path = Path(file_path)
    with path.open("w", encoding="utf-8") as fh:
        for index, chunk in enumerate(chunks, start=1):
            fh.write(f"--- chunk {index} (len={len(chunk)}) ---\n{chunk}\n\n")
    return path


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""

    chunks = run_agent(text_raw)
    for i, chunk in enumerate(chunks, start=1):
        print(f"[{i:02d}] ({len(chunk):2d}) {chunk!r}")

    print(f"\n\n {len(chunks)} chunks")

    saved_to = save_chunks(chunks)
    print(f"Chunks saved to: {saved_to.resolve()}")
