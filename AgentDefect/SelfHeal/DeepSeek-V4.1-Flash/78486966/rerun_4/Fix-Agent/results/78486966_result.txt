from langchain_text_splitters import CharacterTextSplitter


def run_agent(text_raw):
    """Split text into chunks of 30 characters with 5 characters of overlap."""
    text_splitter = CharacterTextSplitter(
        # Default separator is "\n\n" and this text has no blank lines, so the
        # splitter produced ONE chunk (with a "chunk longer than chunk_size"
        # warning). An empty separator makes it split on every character and
        # then re-merge greedily up to chunk_size -> exact 30-char chunks.
        separator="",
        chunk_size=30,
        chunk_overlap=5,
        length_function=len,
        keep_separator=False,
        strip_whitespace=False,
    )
    chunks = text_splitter.split_text(text_raw.strip())
    return chunks


def save_chunks(chunks, file_path="chunks.txt"):
    """Save the chunks locally using plain Python (no extra dependency)."""
    with open(file_path, "w", encoding="utf-8") as file_handler:
        file_handler.write(f"Total chunks: {len(chunks)}\n\n")
        for index, chunk in enumerate(chunks, start=1):
            file_handler.write(f"--- Chunk {index} ({len(chunk)} chars) ---\n")
            file_handler.write(chunk + "\n\n")
    return file_path


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)

    for i, chunk in enumerate(chunks, start=1):
        print(f"[{i}] ({len(chunk)}) {chunk!r}")

    print(f"\n\nNumber of chunks: {len(chunks)}")

    saved_path = save_chunks(chunks)
    print(f"Chunks saved to: {saved_path}")
