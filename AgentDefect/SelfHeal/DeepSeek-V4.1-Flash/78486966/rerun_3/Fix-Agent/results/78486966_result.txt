import os

from langchain_text_splitters import RecursiveCharacterTextSplitter


def run_agent(text_raw):
    """Split the input text into ~30 character chunks with 5 characters of overlap.

    CharacterTextSplitter only splits on its ``separator`` (default "\\n\\n"),
    so a text without blank lines is returned as ONE big chunk even though
    ``chunk_size`` is 30.  RecursiveCharacterTextSplitter (with the default
    separator list ending in "") actually enforces chunk_size for any text,
    while still trying to keep words/sentences intact.
    """
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=30,
        chunk_overlap=5,
        length_function=len,
        separators=["\n\n", "\n", " ", ""],
    )
    chunks = text_splitter.split_text(text_raw.strip())
    return chunks


def save_chunks(chunks, path="chunks.txt"):
    """Save the generated chunks locally, one block per chunk."""
    with open(path, "w", encoding="utf-8") as f:
        for i, chunk in enumerate(chunks, start=1):
            f.write(f"--- Chunk {i} (len={len(chunk)}) ---\n")
            f.write(chunk + "\n\n")
    return os.path.abspath(path)


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)

    for i, chunk in enumerate(chunks, start=1):
        print(f"[{i}] ({len(chunk)} chars) {chunk!r}")

    print(f"\n\n {len(chunks)} chunks")

    saved_to = save_chunks(chunks)
    print(f"Chunks saved to: {saved_to}")
