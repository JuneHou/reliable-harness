from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000

# BUG FIX 1 (the real cause of "it keeps following the first prompt"):
# the conversation history must contain BOTH sides of the dialogue
# (user turns *and* the model's own answers), and it must be built from real
# chat message objects. The old code only ever appended ("user", text) tuples
# and never stored the model's reply, so after a few requests the model was
# given a list of consecutive "user" messages with no assistant turns.
# Ollama/llama3.2 merges consecutive user messages, so the very first
# instruction dominates and later "stop" instructions look like a
# continuation of the same request -> the model keeps answering the old prompt.
# The CLI works because it always sends the full turn history (user + assistant).
messages = [
    SystemMessage(
        content=(
            "You are a helpful assistant. Answer only the most recent user "
            "message and follow the latest instruction, even when it asks you "
            "to stop, change topic or end the conversation."
        )
    )
]

# BUG FIX 2: temperature 1.1 makes llama3.2 wildly creative / ignore
# instructions. Use a low value so instructions are actually followed.
llm = ChatOllama(
    model=MODEL,
    temperature=0.2,
    stream=True,
)


def get_number_of_tokens(messages):
    """Count tokens of the whole history (works with message objects)."""
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        return sum(len(encoding.encode(str(m.content))) for m in messages)
    except Exception:
        return sum(len(str(m.content)) for m in messages) // 4


def trim_history():
    """Drop the oldest turn (keeping the system prompt) until we fit the limit.

    BUG FIX 3: the old loop computed the token count once and then deleted
    messages[1] forever, so it could never terminate (or crashed with
    IndexError). The count has to be recomputed on every iteration.
    """
    while len(messages) > 2 and get_number_of_tokens(messages) >= MODEL_MAX_TOKENS:
        del messages[1]


@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_text = (request.form.get('text') or '').strip()
    if not user_text:
        return "", 400

    messages.append(HumanMessage(content=user_text))
    trim_history()

    try:
        ai_msg = llm.invoke(messages)
    except Exception as exc:  # don't leave a dangling user turn behind
        messages.pop()
        return f"Model error: {exc}", 500

    # BUG FIX 4: store the model's answer, otherwise the next request has no
    # assistant turn and the model repeats the first prompt again.
    messages.append(AIMessage(content=ai_msg.content))
    trim_history()

    return ai_msg.content


@app.route('/reset', methods=['POST'])
def reset():
    """Start a new conversation (so the old prompt is not carried over)."""
    del messages[1:]
    return "ok"


@app.route('/')
def index():
    return render_template('index.html')


if __name__ == '__main__':
    app.run(debug=True, port=5000)
