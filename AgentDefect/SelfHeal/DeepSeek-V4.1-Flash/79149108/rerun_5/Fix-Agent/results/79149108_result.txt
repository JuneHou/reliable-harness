from flask import Flask, render_template, request, session
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)
app.secret_key = "change-me-in-production"

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000        # context window of the model
MAX_OUTPUT_TOKENS = 512          # how long a single answer may be (num_predict)

SYSTEM_PROMPT = (
    "You are a helpful assistant. Always follow the user's most recent "
    "instruction, even if it contradicts something said earlier. If the user "
    "asks you to stop, says 'stop', 'end the story', or tells you the role-play "
    "is over, you must immediately stop the previous behaviour and reply "
    "briefly instead of continuing."
)

# ---------------------------------------------------------------------------
# Conversation state
# ---------------------------------------------------------------------------
# The history has to contain BOTH the user messages AND the assistant replies,
# otherwise the model never sees what it already said. Without the AI turns the
# model only "remembers" the very first user prompt (the role-play / persona
# instruction) and keeps answering it, which is exactly the buggy behaviour.
conversations = {}   # session id -> list[BaseMessage]
conversations["default"] = [SystemMessage(content=SYSTEM_PROMPT)]


def get_history(session_id: str):
    if session_id not in conversations:
        conversations[session_id] = [SystemMessage(content=SYSTEM_PROMPT)]
    return conversations[session_id]


def get_number_of_tokens(messages):
    """Count tokens of a list of LangChain messages."""
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        return sum(len(encoding.encode(str(m.content))) for m in messages)
    except Exception:
        total_chars = sum(len(str(m.content)) for m in messages)
        return total_chars // 4


def trim_history(messages, budget):
    """Drop the OLDEST turns (never the system prompt) until we fit the budget.

    The original code did `while tokens >= MAX: del messages[1]` but never
    recomputed `number_of_tokens` inside the loop (infinite loop) and could
    delete an assistant turn while keeping the matching user turn, leaving a
    broken role sequence. Here we always remove complete user/assistant pairs.
    """
    while len(messages) > 2 and get_number_of_tokens(messages) >= budget:
        del messages[1]                      # oldest turn after the system prompt
        if len(messages) > 1 and isinstance(messages[1], AIMessage):
            del messages[1]


# ---------------------------------------------------------------------------
# Model
# ---------------------------------------------------------------------------
# temperature 1.1 makes the model random and prone to endless continuation;
# a lower temperature follows instructions ("stop") far more reliably and
# num_predict caps runaway answers.
llm = ChatOllama(
    model=MODEL,
    temperature=0.3,
    num_predict=MAX_OUTPUT_TOKENS,
    stream=False,
)


@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    session_id = session.get('id', 'default')
    history = get_history(session_id)

    # 1. store the new user turn as a proper HumanMessage
    history.append(HumanMessage(content=request.form['text']))

    # 2. keep the prompt inside the context window (recomputes every iteration)
    trim_history(history, MODEL_MAX_TOKENS)

    # 3. call the model with the FULL history (user + assistant turns)
    ai_msg = llm.invoke(history)

    # 4. >>> the actual fix <<<: store the answer in the history so the model
    #    sees its own previous output on the next call and can react to
    #    follow-up instructions such as "stop".
    history.append(AIMessage(content=ai_msg.content))

    return ai_msg.content


@app.route('/reset', methods=['POST'])
def reset():
    session_id = session.get('id', 'default')
    conversations[session_id] = [SystemMessage(content=SYSTEM_PROMPT)]
    return "reset"


@app.route('/')
def index():
    return render_template('index.html')


if __name__ == '__main__':
    app.run(debug=True, port=5000)
