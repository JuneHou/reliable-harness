from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000
messages = []

llm = ChatOllama(
    model=MODEL,
    temperature=1.1,
    stream=True
)


def get_number_of_tokens(messages):
    """Count tokens for the whole conversation."""
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total_tokens = 0
        for m in messages:
            content = m.content if hasattr(m, "content") else str(m)
            total_tokens += len(encoding.encode(content))
        return total_tokens
    except Exception:
        total_chars = 0
        for m in messages:
            content = m.content if hasattr(m, "content") else str(m)
            total_chars += len(content)
        return total_chars // 4


@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    # 1) Add the user's new turn as a proper message object.
    messages.append(HumanMessage(content=request.form['text']))

    # 2) Trim OLD turns (recompute each iteration so the loop terminates),
    #    but always keep the most recent user message.
    while len(messages) > 1 and get_number_of_tokens(messages) >= MODEL_MAX_TOKENS:
        del messages[0]

    # 3) Get the model's reply.
    ai_msg = llm.invoke(messages)

    # 4) Append the assistant reply back into the history.
    #    This is the key fix: without it the conversation only contains
    #    user turns, so the model re-reads the first prompt and never
    #    receives a signal to stop. With the reply stored, the model sees
    #    a normal alternating user/assistant dialogue and honors later
    #    instructions such as "stop".
    messages.append(AIMessage(content=ai_msg.content))

    return ai_msg.content


@app.route('/')
def index():
    return render_template('index.html')


if __name__ == '__main__':
    app.run(debug=True, port=5000)
