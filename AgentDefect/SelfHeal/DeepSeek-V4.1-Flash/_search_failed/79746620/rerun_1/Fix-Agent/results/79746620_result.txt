"""
Metadata filtering on "null" values in LlamaIndex.

Why the original code returns nothing
-------------------------------------
`None` is not a filterable metadata value in LlamaIndex:

* The in-memory `SimpleVectorStore` flattens node metadata before storing it
  (`node_to_metadata_dict` -> `_flatten_metadata`), and only values that are
  `str` / `int` / `float` (or lists of those) survive. A `None` value is
  dropped, so the key does not exist in the stored metadata at all.
* The filter evaluation (`_filter_metadata`) then short-circuits with
  "key not in metadata" -> `False`, which is why `MetadataFilter(value=None)`
  never matches anything, even though the docstore still shows
  `{"start_date": None}`.

So the observed behaviour is expected, not a crash bug. To make "null"
metadata filterable you must index a *comparable* placeholder (a sentinel
string) instead of `None`, and filter on that same placeholder.

The fix below:
1. Encodes `None` metadata values into a string sentinel (`__NULL__`).
2. Builds the filter with the sentinel instead of `None`.
3. Decodes the sentinel back to `None` when reporting results, so the
   semantics stay "this field was null".
"""

import os

from dotenv import load_dotenv

from llama_index.core import Settings, VectorStoreIndex
from llama_index.core.embeddings import MockEmbedding
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    FilterOperator,
    MetadataFilter,
    MetadataFilters,
)

# String placeholder used to index "null" metadata values.
NULL_SENTINEL = "__NULL__"


def encode_metadata(metadata: dict) -> dict:
    """Replace `None` values with a filterable string sentinel."""
    return {
        key: (NULL_SENTINEL if value is None else value)
        for key, value in metadata.items()
    }


def decode_metadata(metadata: dict) -> dict:
    """Turn the sentinel back into `None` for display/comparison purposes."""
    return {
        key: (None if value == NULL_SENTINEL else value)
        for key, value in metadata.items()
    }


def null_value_filters(key: str) -> MetadataFilters:
    """Filter that matches nodes whose `key` metadata value was `None`."""
    return MetadataFilters(
        filters=[
            MetadataFilter(
                key=key,
                operator=FilterOperator.EQ,
                value=NULL_SENTINEL,  # <- sentinel, NOT None
            )
        ]
    )


def get_embed_model():
    """Use OpenAI when a key is available, otherwise a local mock model."""
    load_dotenv()
    if os.getenv("OPENAI_API_KEY"):
        from llama_index.embeddings.openai import OpenAIEmbedding

        return OpenAIEmbedding(model="text-embedding-3-small")
    return MockEmbedding(embed_dim=8)


def run_agent(query):
    # Set up embedding model
    Settings.embed_model = get_embed_model()

    # Create node; the None value is stored as a filterable sentinel
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=encode_metadata({"start_date": None}),
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes (raw, sentinel-encoded):")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print("Index nodes (decoded):")
    print([decode_metadata(node.metadata) for node in doc_index.docstore.docs.values()])
    print()

    # Filter on the sentinel value instead of None
    retriever = doc_index.as_retriever(
        filters=null_value_filters("start_date"), similarity_top_k=1
    )
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, decode_metadata(node.metadata)) for node in nodes])
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved (None is not filterable)")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
