import os
import nltk
from dotenv import load_dotenv

from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
# Persistent key/value docstore (replaces the ephemeral InMemoryStore)
from langchain.storage import LocalFileStore, create_kv_docstore
from langchain.schema import Document

CHROMA_DIR = "./chroma_db_child"      # persists the CHILD chunks (vectors + text)
DOCSTORE_DIR = "./docstore_db"        # persists the PARENT documents
COLLECTION_NAME = "full_documents"
HF_EMBED_PATH = "./hf_cache"


def _ensure_nltk_data():
    """NLTKTextSplitter needs punkt (and punkt_tab on newer NLTK versions)."""
    for pkg in ("punkt", "punkt_tab"):
        try:
            nltk.download(pkg, quiet=True)
        except Exception:
            pass


def get_documents():
    return [
        Document(
            page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."
        ),
        Document(
            page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."
        ),
        Document(
            page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."
        ),
    ]


def build_retriever(child_splitter=None):
    """Rebuild the retriever from persisted state (no add_documents needed)."""
    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=HF_EMBED_PATH,
    )

    # Child chunks -> Chroma, persisted on disk via persist_directory
    vectorstore = Chroma(
        collection_name=COLLECTION_NAME,
        embedding_function=embedding_function,
        persist_directory=CHROMA_DIR,
    )

    # Parent documents -> persistent file-backed KV docstore (was InMemoryStore)
    fs = LocalFileStore(DOCSTORE_DIR)
    store = create_kv_docstore(fs)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter or NLTKTextSplitter(chunk_size=400),
        # k applies to the child-chunk search; the retriever then returns parents
        search_kwargs={"k": 3},
        # search_type="mmr",  # enable if you want MMR over the child chunks
    )
    return retriever, vectorstore, store


def _is_already_indexed(vectorstore):
    try:
        return vectorstore._collection.count() > 0
    except Exception:
        return False


def run_agent(query, store_location=DOCSTORE_DIR):
    load_dotenv()
    _ensure_nltk_data()

    retriever, vectorstore, store = build_retriever()

    # Index only the first time; afterwards both stores are loaded from disk.
    if not _is_already_indexed(vectorstore):
        retriever.add_documents(get_documents())  # child chunks -> Chroma, parents -> LocalFileStore
        try:
            vectorstore.persist()  # explicit persist for older langchain versions
        except AttributeError:
            pass  # newer versions auto-persist

    # IMPORTANT: query the retriever, not the raw vectorstore, so the
    # PARENT documents are returned instead of the child chunks.
    if hasattr(retriever, "invoke"):
        return retriever.invoke(query)
    return retriever.get_relevant_documents(query)


if __name__ == "__main__":
    query = "What is RAG?"
    result = run_agent(query)

    for i, doc in enumerate(result, start=1):
        print(f"--- Parent document {i} ---")
        print(doc.page_content)
        print()
