import os

import nltk
from dotenv import load_dotenv

from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import LocalFileStore          # persistent byte store
from langchain.schema import Document

# create_kv_docstore turns a byte store (LocalFileStore) into a persisted BaseStore
try:
    from langchain.storage._lc_store import create_kv_docstore
except ImportError:  # langchain 1.x layout
    from langchain_classic.storage._lc_store import create_kv_docstore

load_dotenv()

CHROMA_DIR = "./chroma_db_child"
COLLECTION_NAME = "full_documents"
STORE_LOCATION = "./store_location"   # parent documents live here on disk
HF_EMBED_PATH = "./hf_cache"


def _get_punkt():
    """Make sure NLTK's sentence tokenizer data is available."""
    for resource in ("punkt", "punkt_tab"):
        try:
            nltk.download(resource, quiet=True)
        except Exception:
            pass


def build_embedding_function():
    return HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=HF_EMBED_PATH,
    )


def build_retriever(embedding_function, store_location=STORE_LOCATION):
    """Rebuild the retriever from on-disk state (child chunks + parent docs)."""
    vectorstore = Chroma(
        collection_name=COLLECTION_NAME,
        embedding_function=embedding_function,
        persist_directory=CHROMA_DIR,          # child chunks are persisted here
    )

    # >>> the fix: a *persisted* docstore instead of InMemoryStore() <<<
    byte_store = LocalFileStore(store_location)
    docstore = create_kv_docstore(byte_store)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=docstore,
        child_splitter=NLTKTextSplitter(chunk_size=400),
        # same behaviour as the previous max_marginal_relevance_search(query, k=3)
        search_type="mmr",
        search_kwargs={"k": 3},
    )
    return retriever


def docstore_is_empty(store_location=STORE_LOCATION):
    """True when nothing has been ingested yet (or the folder is missing)."""
    if not os.path.isdir(store_location):
        return True
    return not any(not name.startswith(".") for name in os.listdir(store_location))


def ingest_documents(retriever, docs):
    """One-off ingestion: parent docs -> docstore, child chunks -> Chroma."""
    retriever.add_documents(docs)

    persist = getattr(retriever.vectorstore, "persist", None)
    if callable(persist):                      # older chroma versions need this
        persist()

    if hasattr(retriever.docstore, "yield_keys"):
        try:
            list(retriever.docstore.yield_keys())
        except Exception:
            pass


def retrieve(retriever, query):
    """Query through the retriever so PARENT documents are returned."""
    try:
        return retriever.invoke(query)
    except AttributeError:
        return retriever.get_relevant_documents(query)


def run_agent(query, store_location=STORE_LOCATION):
    _get_punkt()

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."),
    ]

    embedding_function = build_embedding_function()
    retriever = build_retriever(embedding_function, store_location)

    # Only ingest the first time; on later runs the retriever is restored
    # purely from ./store_location (parents) and ./chroma_db_child (children).
    if docstore_is_empty(store_location):
        ingest_documents(retriever, docs)

    # Return full parent documents, not the raw child chunks stored in Chroma.
    matching_doc = retrieve(retriever, query)
    return matching_doc


if __name__ == "__main__":
    query = "What is RAG?"
    result = run_agent(query)
    print(result)
