    """Configure the global llama-index settings exactly once."""
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:

            model=MODEL_NAME,

    Settings.context_window = CONTEXT_WINDOW
    Settings.num_output = NUM_OUTPUT
    Settings.chunk_size = CHUNK_SIZE
    Settings.chunk_overlap = CHUNK_OVERLAP

    """Load (or build and persist) the vector index over the example documents."""
    if os.path.isdir(PERSIST_DIR) and os.listdir(PERSIST_DIR):
        storage_context = StorageContext.from_defaults(persist_dir=PERSIST_DIR)

def _configure_settings():
    """Configure the global llama-index settings exactly once."""
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise RuntimeError(
            "OPENAI_API_KEY is not set - add it to your .env file before running."
        )
    os.environ["OPENAI_API_KEY"] = api_key
    openai.api_key = api_key

    if not isinstance(getattr(Settings, "llm", None), OpenAI):
        Settings.llm = OpenAI(
            temperature=0.3,
            model=MODEL_NAME,
            max_tokens=NUM_OUTPUT,
        )
    Settings.context_window = CONTEXT_WINDOW
    Settings.num_output = NUM_OUTPUT
    Settings.chunk_size = CHUNK_SIZE
    Settings.chunk_overlap = CHUNK_OVERLAP


def create_vector_index(path=DOCS_PATH):
    """Load (or build and persist) the vector index over the example documents."""
    if os.path.isdir(PERSIST_DIR) and os.listdir(PERSIST_DIR):
        storage_context = StorageContext.from_defaults(persist_dir=PERSIST_DIR)
        return load_index_from_storage(storage_context)

    docs = SimpleDirectoryReader(path).load_data()
    if not docs:
        raise ValueError(
            f"No documents found in '{path}'. Add the example NDA files there first."
        )

    vector_index = VectorStoreIndex.from_documents(docs)
    # FIX: persist_dir must be a directory, not a file called 'vectorIndex.json'.
    vector_index.storage_context.persist(persist_dir=PERSIST_DIR)
    return vector_index


def build_query_engine(vector_index, similarity_top_k=4, response_mode="compact"):
    """Query engine that drafts documents from the examples instead of refusing."""
    return vector_index.as_query_engine(
        similarity_top_k=similarity_top_k,
        response_mode=response_mode,
        text_qa_template=DRAFT_QA_TEMPLATE,
        refine_template=DRAFT_REFINE_TEMPLATE,
    )


def run_agent(user_input, docs_path=DOCS_PATH):
    """Answer/draft from the example documents at docs_path."""
    _configure_settings()
    vector_index = create_vector_index(docs_path)
    query_engine = build_query_engine(vector_index)

    response = query_engine.query(user_input)
    result = str(response.response).strip()

    if not result:
        # Explicit, actionable message instead of a silent/empty refusal.
        result = (
            "The model returned an empty response. Check that the example "
            f"documents in '{docs_path}' actually loaded and that your "
            "OPENAI_API_KEY has available quota."
        )
    return result
