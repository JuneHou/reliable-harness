import json
import os

from langchain_text_splitters import CharacterTextSplitter


def run_agent(text_raw, chunk_size=30, chunk_overlap=5):
    """Split `text_raw` into chunks of at most `chunk_size` chars with
    `chunk_overlap` chars of overlap between neighbouring chunks.

    Why the original code returned a single chunk:
      CharacterTextSplitter's default `separator` is "\\n\\n" (two newlines).
      A splitter only splits on the separator, so if the text contains no
      "\\n\\n" the whole document is emitted as ONE chunk, no matter how small
      `chunk_size` is (LangChain only warns "Created a chunk of size N, which
      is longer than the specified 30" and carries on).

    The fix:
      1. Normalise the input (single newlines / stray whitespace are not
         paragraph separators here, so they must not block splitting).
      2. Use a separator that actually exists in the text -- a single space --
         so the text can be broken into word-sized pieces that are then merged
         back up to <= chunk_size with the requested overlap.
    """
    # Normalise whitespace so the splitter works on plain, space separated text.
    text_raw = " ".join(text_raw.split())

    text_splitter = CharacterTextSplitter(
        separator=" ",              # <- key change: default "\\n\\n" prevented any split
        chunk_size=chunk_size,
        chunk_overlap=chunk_overlap,
        length_function=len,
        is_separator_regex=False,
        strip_whitespace=True,
    )
    chunks = text_splitter.split_text(text_raw)
    return chunks


def save_chunks(chunks, path="chunks.json"):
    """Save the chunks locally (JSON keeps them easy to reload)."""
    directory = os.path.dirname(os.path.abspath(path))
    os.makedirs(directory, exist_ok=True)
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(chunks, fh, ensure_ascii=False, indent=2)

    # Bonus: a plain text dump with section markers, one chunk per line.
    txt_path = os.path.splitext(path)[0] + ".txt"
    with open(txt_path, "w", encoding="utf-8") as fh:
        for i, chunk in enumerate(chunks, start=1):
            fh.write(f"--- chunk {i} ({len(chunk)} chars) ---\n{chunk}\n")

    return os.path.abspath(path), os.path.abspath(txt_path)


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)

    for i, chunk in enumerate(chunks, start=1):
        print(f"[{i}] ({len(chunk)} chars) {chunk!r}")

    print(f"\nTotal chunks: {len(chunks)}")

    json_path, txt_path = save_chunks(chunks, "output/chunks.json")
    print(f"Saved chunks to: {json_path}")
    print(f"Saved chunks to: {txt_path}")
