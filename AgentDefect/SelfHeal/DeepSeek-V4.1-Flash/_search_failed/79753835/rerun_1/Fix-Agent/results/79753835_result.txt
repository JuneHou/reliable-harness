"""Reproduce the exact token count used by the BGE-M3 embedding model.

Why the original snippet reported two different numbers:

1. `TokenCountingHandler()` was created without a `tokenizer`, so it fell back to
   LlamaIndex's default global tokenizer (a tiktoken / GPT-2 style BPE).  That
   tokenizer has nothing to do with BGE-M3 (an XLM-R / sentencepiece model), so
   `total_embedding_token_count` could never match `AutoTokenizer(text)`.

2. The callback manager was created *after* the embedding model and assigned to
   `Settings` only afterwards.  `TransformComponent`/`BaseEmbedding` bind
   `self.callback_manager` from `Settings.callback_manager` at construction time,
   so the model kept an empty callback manager and the counter stayed at 0 (or
   was only updated occasionally).  The callback machinery (and the global
   setting) must exist *before* the embed model is instantiated, and the callback
   manager should also be passed explicitly to the embed model.

3. The comparison tokenization did not apply the same preprocessing as the model
   (`max_length` truncation / special tokens), so even with a matching tokenizer
   long inputs could disagree.

The corrected code builds the counter with BGE-M3's own tokenizer, wires the
callback manager into `Settings` first and into the embed model explicitly, and
uses the very same tokenization function on both sides - so both numbers are
identical and equal to what the model actually consumes.
"""

import os
from typing import List

from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from transformers import AutoTokenizer

MODEL_NAME = "BAAI/bge-m3"
CACHE_DIR = "./embeddings"
CHUNK_SIZE = 512


def create_tokenizer(model_name: str = MODEL_NAME):
    """Load the exact tokenizer that the embedding model uses (same cache dir)."""
    os.makedirs(CACHE_DIR, exist_ok=True)
    return AutoTokenizer.from_pretrained(model_name, cache_dir=CACHE_DIR)


def make_token_count_fn(tokenizer, max_length: int = CHUNK_SIZE):
    """Tokenize exactly like the embedder does (special tokens + truncation)."""

    def _count(string: str) -> List[int]:
        return tokenizer(
            string,
            add_special_tokens=True,
            truncation=True,
            max_length=max_length,
        )["input_ids"]

    return _count


def create_embedding_model(tokenizer, callback_manager, chunk_size: int = CHUNK_SIZE):
    """Create the embed model already bound to the token-counting callback manager."""
    os.makedirs(CACHE_DIR, exist_ok=True)
    return HuggingFaceEmbedding(
        model_name=MODEL_NAME,
        tokenizer_name=MODEL_NAME,      # make sure the model uses BGE-M3's tokenizer
        max_length=chunk_size,          # same truncation limit used when counting
        cache_folder=CACHE_DIR,
        callback_manager=callback_manager,
    )


def run_agent(text: str):
    chunk_size = CHUNK_SIZE

    # 1) Tokenizer first - it defines both the model input and the counting.
    tokenizer = create_tokenizer()
    token_count_fn = make_token_count_fn(tokenizer, max_length=chunk_size)

    # 2) Build the callback machinery BEFORE the embed model exists, and set it globally.
    token_counter = TokenCountingHandler(tokenizer=token_count_fn)
    callback_manager = CallbackManager([token_counter])
    Settings.callback_manager = callback_manager

    # 3) Now create the embed model so it inherits that callback manager
    #    (and assign it explicitly as well, to be safe).
    embed_model = create_embedding_model(tokenizer, callback_manager, chunk_size)
    embed_model.callback_manager = callback_manager
    Settings.embed_model = embed_model

    # Make sure we only measure the call we care about.
    token_counter.reset_counts()

    # 4) Run the embedding; the EMBEDDING event is now handled by our counter.
    Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # 5) Same preprocessing, same tokenizer -> same number.
    token_count = len(
        tokenizer(
            text,
            add_special_tokens=True,
            truncation=True,
            max_length=chunk_size,
        )["input_ids"]
    )

    return embedding_tokens, token_count


if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")

    assert embedding_tokens == token_count, (
