            "stored in its own column instead of inside the metadata JSON blob."
        ),
    )

    _custom_columns_ready = PrivateAttr(default=False)

    def __init__(self, **kwargs: Any) -> None:
        super().__init__(**kwargs)
        # LlamaIndex has created the table by now -> add our columns.
        self._ensure_custom_columns()

    # ------------------------------------------------------------------ helpers
    @property
    def _qualified_table(self) -> str:
        schema = getattr(self, "schema_name", SCHEMA_NAME) or SCHEMA_NAME
        return f'"{schema}"."{self.table_name}"'

    @property
    def _sync_engine(self) -> Engine:
        engine = getattr(self, "_engine", None)
        if engine is not None:
            return engine
        # Fallback for integration versions that keep the engine elsewhere.
        connection_string = getattr(self, "connection_string", None) or getattr(
            self, "_connection_string", None
        )
        engine = create_engine(connection_string)
        self._engine = engine
        return engine

    def _ensure_custom_columns(self) -> bool:
        """Create the custom columns if the table already exists."""
        if not self.custom_columns:
            return False
        if self._custom_columns_ready:
            return True

        with self._sync_engine.begin() as conn:
            table_exists = conn.execute(
                text("SELECT to_regclass(:tbl)"), {"tbl": self._qualified_table}
            ).scalar()
            if table_exists is None:
                # Table not created yet (e.g. perform_setup=False); retry on add().
                return False
            for column, sql_type in self.custom_columns.items():
                conn.execute(
                    text(
                        f'ALTER TABLE {self._qualified_table} '
                        f'ADD COLUMN IF NOT EXISTS "{column}" {sql_type}'
                    )
                )
        self._custom_columns_ready = True
        return True

    def _strip_promoted_keys(
        self, nodes: Sequence[BaseNode]
    ) -> List[BaseNode]:
        """Copy nodes without the promoted keys, so they are not duplicated in JSON."""
        keys = set(self.custom_columns)
        if not keys:
            return list(nodes)

        stripped: List[BaseNode] = []
        for node in nodes:
            metadata = node.metadata or {}
            promoted = keys & set(metadata)
            if not promoted:
                stripped.append(node)
                continue
            clone = copy.deepcopy(node)
            clone.metadata = {k: v for k, v in metadata.items() if k not in promoted}
            stripped.append(clone)
        return stripped

    def _write_promoted_columns(self, nodes: Sequence[BaseNode]) -> None:
        if not self.custom_columns or not self._ensure_custom_columns():
            return

        assignments = ", ".join(f'"{c}" = :v_{c}' for c in self.custom_columns)
        stmt = text(
            f"UPDATE {self._qualified_table} SET {assignments} WHERE node_id = :node_id"
        )

        payloads = []
        for node in nodes:
            metadata = node.metadata or {}
            if not (set(self.custom_columns) & set(metadata)):
                continue
            payload: Dict[str, Any] = {
                f"v_{c}": metadata.get(c) for c in self.custom_columns
            }
            payload["node_id"] = node.node_id
            payloads.append(payload)

        if not payloads:
            return

        with self._sync_engine.begin() as conn:
            for payload in payloads:
                conn.execute(stmt, payload)

    def _attach_promoted_columns(self, nodes: Sequence[BaseNode]) -> None:
        if not nodes or not self.custom_columns or not self._ensure_custom_columns():
            return

        columns = ", ".join(f'"{c}"' for c in self.custom_columns)
        stmt = text(
            f"SELECT node_id, {columns} FROM {self._qualified_table} "
            "WHERE node_id IN :node_ids"
        ).bindparams(bindparam("node_ids", expanding=True))

        with self._sync_engine.connect() as conn:
            rows = (
                conn.execute(stmt, {"node_ids": [n.node_id for n in nodes]})
                .mappings()
                .all()
            )
        by_node_id = {row["node_id"]: row for row in rows}

        for node in nodes:
            row = by_node_id.get(node.node_id)
            if row is None:
                continue
            for column in self.custom_columns:
                node.metadata[column] = row[column]

    # ------------------------------------------------------------ vector store API
    def add(self, nodes: Sequence[BaseNode], **add_kwargs: Any) -> List[str]:
        """Insert nodes; promoted metadata keys go to their own columns."""
        ids = super().add(self._strip_promoted_keys(nodes), **add_kwargs)
        self._write_promoted_columns(nodes)
        return ids

    async def async_add(
        self, nodes: Sequence[BaseNode], **add_kwargs: Any
    ) -> List[str]:
        # Keep the custom-column logic on the same code path when
        # BasePydanticVectorStore dispatches asynchronously.
        return await asyncio.to_thread(self.add, nodes, **add_kwargs)

    def query(
        self, query: VectorStoreQuery, **kwargs: Any
    ) -> VectorStoreQueryResult:
        result = super().query(query, **kwargs)
        self._attach_promoted_columns(result.nodes or [])
        return result

def build_table_model(
    base: Any,
    table_name: str,
    embed_dim: int,
    schema_name: Optional[str] = SCHEMA_NAME,
    custom_columns: Optional[Dict[str, Column]] = None,
    use_jsonb: bool = False,
    hybrid_search: bool = False,
    text_search_config: str = "english",
) -> Any:
    """Mirror of `llama_index.vector_stores.postgres.utils.get_data_model`,
    extended with the columns requested in `custom_columns`."""
    custom_columns = custom_columns or {}
    metadata_type = JSONB if use_jsonb else JSON

    class AbstractData(base):  # type: ignore[misc, valid-type]
        __abstract__ = True

        id = Column(Integer, primary_key=True, autoincrement=True)
        text = Column(Text, nullable=True)
        metadata_ = Column("metadata_", metadata_type, nullable=True)
        node_id = Column(Text, nullable=True)
        embedding = Column(Vector(embed_dim))
        if hybrid_search:
            text_search_tsv = Column(
                TSVector,
                Computed(f"to_tsvector('{text_search_config}', text)", persisted=True),
            )

    namespace: Dict[str, Any] = {
        "__tablename__": table_name,
        "__table_args__": {"schema": schema_name} if schema_name else {},
    }
    for name, column in custom_columns.items():
        namespace[name] = Column(name, column.type, nullable=True)

    return type(f"data_{table_name}", (AbstractData,), namespace)

class PGVectorStoreWithCustomColumns(PGVectorStore):
    """`PGVectorStore` whose table has real columns for chosen metadata fields."""

    _custom_column_types: Dict[str, Column] = PrivateAttr(default_factory=dict)
    _keep_metadata_copy: bool = PrivateAttr(default=True)

    def __init__(
        self,
        custom_columns: Optional[Dict[str, Column]] = None,
        keep_metadata_copy: bool = False,
        **kwargs: Any,
    ) -> None:
        # `keep_metadata_copy=False` -> the value lives ONLY in its own column
        # (exactly what was asked for).  Set it to True if you also want the
        # value kept in `metadata_` so that LlamaIndex MetadataFilters keep
        # working out of the box.
        super().__init__(**kwargs)

        """`ALTER TABLE ... ADD COLUMN IF NOT EXISTS` for every custom column."""
        dialect = postgresql.dialect()
        clauses = [
            f'ADD COLUMN IF NOT EXISTS "{name}" '
            f"{column.type.compile(dialect=dialect)}"
            for name, column in self._custom_column_types.items()
        ]
        ddl = f"ALTER TABLE {self._quoted_table_name()} " + ", ".join(clauses)
        with self._engine.begin() as connection:
            connection.execute(sql_text(ddl))

        """Replace the auto-generated model with one that knows the new columns."""
        base = declarative_base()
        self._base = base
        self._table_class = build_table_model(
            base=base,
            table_name=self.table_name,
            embed_dim=self.embed_dim,
            schema_name=self.schema_name,
            custom_columns=self._custom_column_types,
            use_jsonb=self.use_jsonb,
            hybrid_search=self.hybrid_search,
            text_search_config=self.text_search_config,
        )

        # Keeps the custom columns even if the store is lazily (re)initialised.
        super()._initialize(*args, **kwargs)
        if getattr(self, "_custom_column_types", None):
            self._ensure_custom_columns()
            self._rebuild_table_model()

        super_builder = getattr(super(), "_node_to_table_row", None)
        if super_builder is not None:
            row = super_builder(node)
        else:  # defensive fallback if the internal helper is renamed

                node_id=node.node_id,
                embedding=node.get_embedding(),
                text=node.get_content(metadata_mode=MetadataMode.NONE),

                    node,
                    remove_text=True,
                    flat_metadata=self.flat_metadata,
                ),
            )

        if not self._custom_column_types:
            return row

        # Promote the metadata fields into their dedicated columns.
        for name in self._custom_column_types:
            setattr(row, name, node.metadata.get(name))

        if not self._keep_metadata_copy:
            metadata = dict(row.metadata_ or {})
            for name in self._custom_column_types:
                metadata.pop(name, None)
            row.metadata_ = metadata

        return row

Indexes the 2 documents **first** — LlamaIndex writes each row with its own fixed column list (`id, node_id, text, metadata_, embedding`), so `external_id` is serialized into the `metadata_` JSON blob (the fix's own docstring states this explicitly: *"Any extra field you attach to `Document.metadata` therefore ends up inside the `metadata_` JSON blob"*).
3. Then runs `ensure_custom_columns` (`ALTER TABLE ... ADD COLUMN external_id TEXT`, index, and a `BEFORE INSERT OR UPDATE` trigger that copies `metadata_->>'external_id'` into the new column).
4. Then `sync_custom_columns` backfills existing rows.

### Why it fails the tests

**Test `test_external_id_saved_in_column` has two assertions:**
```python
self.assertIsNotNone(external_id)          # ✅ passes (column gets backfilled)
self.assertNotIn("external_id", metadata)  # ❌ FAILS
```

The fix **adds** a real column but **never removes the key from `Document.metadata`**. Because `run_agent` still passes `metadata={"external_id": "ext_001"}`, LlamaIndex's `node_to_metadata_dict(node, remove_text=True)` (whether flat or nested) serializes `external_id` into the JSONB `metadata_`. Nothing in the fixed code ever does `metadata_ = metadata_ - 'external_id'` or pops the key from the node before insertion. So the document ends up with `external_id` in **both** places, and `assertNotIn("external_id", metadata)` fails.

The fix is essentially solving a *different problem* (promoting metadata to columns via trigger) rather than the problem the test specifies (moving `external_id` **out of** `metadata_` **into** a dedicated column). That is a requirements mismatch, not a cosmetic one.

**Test `test_expected_external_ids`** would pass (backfill populates `ext_001` / `ext_002`), but one passing test doesn't rescue the suite.

### Additional technical problems

| Issue | Detail |
|---|---|
| Ordering flaw | DDL/trigger is installed *after* ingestion, so the trigger never fires for the rows under test — correctness relies solely on the separate `UPDATE` backfill. |
| Non-idempotent semantics | Since `metadata_` keeps the key, every subsequent `sync_custom_columns` call re-writes the column from JSON, meaning `metadata_` remains the source of truth rather than the column. |
| Connection leak | `with psycopg2.connect(...) as conn:` only commits/rolls back the transaction — it does **not** close the connection. `ensure_custom_columns`, `sync_custom_columns`, and `fetch_rows` each leak a connection per call, called twice per test run. |
| Coupling / fragile assumptions | `_metadata_column` sniffs `information_schema` for `metadata_`/`metadata`, and `sync_custom_columns` raises `RuntimeError` if not found; `customize_query` assumes `table.c.external_id` exists and returns `query.add_columns(...)` — dead code in this path but an undocumented contract change. `fetch_rows` orders by `node_id` while the test orders by `id`. |
| Dead/unused imports | `BaseNode`, `Tuple`, `List`, `Any`, `Sequence` partially unused; trivial but indicative. |
| API usage | `PGVectorStore.from_params(..., schema_name=, perform_setup=, use_jsonb=)` is valid for current `llama-index-vector-stores-postgres`, and `psycopg2.connect(**kwargs)` with `dbname` is valid — so API usage is *not* the blocker. The blocker is behavior. |

### What a correct fix would look like
Keep the dedicated column, but **also strip the key from the persisted metadata**, e.g. in the same backfill:
```sql
UPDATE data_vectorstore
   SET external_id = metadata_ ->> 'external_id',
       metadata_   = metadata_ - 'external_id'

```
(or pop `external_id` from `node.metadata` before `from_documents` and attach it via a post-insert `UPDATE`). The submitted fix does the first half and omits the second half entirely, which is precisely the half the test checks.

def _metadata_column(cur, schema_name: str, table_name: str) -> Optional[str]:

    cur.execute(

def ensure_custom_columns(
    database: str,
    table_name: str = TABLE_NAME,
    schema_name: str = SCHEMA_NAME,
    columns: Dict[str, str] = CUSTOM_COLUMNS,
) -> None:
    """
    Add the requested columns to the LlamaIndex table and install a trigger that
    keeps them in sync with ``metadata_`` for every row the vector store writes.
    """
    qualified = _qualified(schema_name, table_name)
    trigger_name = f"trg_sync_{table_name}_custom_columns"
    function_name = f"sync_{table_name}_custom_columns"

    with psycopg2.connect(**connection_params(database)) as conn:
        with conn.cursor() as cur:
            cur.execute("SELECT to_regclass(%s)", (f"{schema_name}.{table_name}",))
            if cur.fetchone()[0] is None:
                raise RuntimeError(
                    f"Table {qualified} does not exist yet - index at least one "
                    "document (or call ensure_table) before adding custom columns."
                )

            # 2) real columns (+ index for filtering by them)
            for column, pg_type in columns.items():
                cur.execute(
                    f"ALTER TABLE {qualified} "
                    f"ADD COLUMN IF NOT EXISTS {_ident(column)} {pg_type}"
                )
                cur.execute(
                    f"CREATE INDEX IF NOT EXISTS "
                    f"{_ident('idx_' + table_name + '_' + column)} "
                    f"ON {qualified} ({_ident(column)})"
                )

            metadata_col = _metadata_column(cur, schema_name, table_name)
            if metadata_col is None:
                raise RuntimeError(
                    f"No metadata column found on {qualified}; cannot sync custom "
                    "columns from metadata."
                )

            assignments = "\n            ".join(
                f"NEW.{_ident(column)} := COALESCE("
                f"NEW.{_ident(metadata_col)}->>'{column}', NEW.{_ident(column)});"
                for column in columns
            )
            cur.execute(
                f"""
                CREATE OR REPLACE FUNCTION {_ident(function_name)}()
                RETURNS trigger
                LANGUAGE plpgsql AS $$
                BEGIN
            {assignments}
                    RETURN NEW;
                END;
                $$;
                """
            )
            cur.execute(
                f"DROP TRIGGER IF EXISTS {_ident(trigger_name)} ON {qualified}"
            )
            cur.execute(
                f"""
                CREATE TRIGGER {_ident(trigger_name)}
                BEFORE INSERT OR UPDATE ON {qualified}
                FOR EACH ROW
                EXECUTE FUNCTION {_ident(function_name)}();
                """
            )
        conn.commit()


# --------------------------------------------------------------------------- #
# Step 4: backfill rows that were indexed before the columns existed
# --------------------------------------------------------------------------- #
def sync_custom_columns(
    database: str,
    table_name: str = TABLE_NAME,
    schema_name: str = SCHEMA_NAME,
    columns: Dict[str, str] = CUSTOM_COLUMNS,
) -> int:
    qualified = _qualified(schema_name, table_name)
    with psycopg2.connect(**connection_params(database)) as conn:
        with conn.cursor() as cur:
            metadata_col = _metadata_column(cur, schema_name, table_name)
            if metadata_col is None:
                raise RuntimeError(f"No metadata column found on {qualified}")

            assignments = ", ".join(
                f"{_ident(column)} = {_ident(metadata_col)}->>'{column}'"
                for column in columns
            )
            cur.execute(
                f"UPDATE {qualified} SET {assignments} "
                f"WHERE {_ident(metadata_col)} IS NOT NULL"
            )
            updated = cur.rowcount
        conn.commit()
    return updated

"""Store selected document fields in real Postgres columns instead of the metadata JSON.

Why the original code did not match expectations
------------------------------------------------
`llama_index.vector_stores.postgres.PGVectorStore` creates (and expects) a *fixed*
table layout::

    id | node_id | text | metadata_ (json/jsonb) | embedding (vector)

It has no `metadata_columns=` / `custom_columns=` option (unlike LangChain's PG
vector store), and `_node_to_table_row()` only forwards
`node_to_metadata_dict(node)` into `metadata_`.  Consequently `external_id`
ends up inside the `metadata_` JSON column and *no* `external_id` column is
ever created - exactly the reported mismatch.

Fix
---
A thin subclass of `PGVectorStore` that:
  1. adds the requested columns to the LlamaIndex-created table
     (`ALTER TABLE ... ADD COLUMN IF NOT EXISTS`), idempotently;
  2. on `add()` lifts the promoted metadata keys into those columns and removes
     them from the JSON metadata, so the value is stored exactly once;
  3. on `query()` reads the columns back for the returned nodes.

# these become real columns instead of living in the `metadata_` JSON column
CUSTOM_COLUMNS: Dict[str, str] = {
    "external_id": "VARCHAR(255)",
}


class PGVectorStoreWithCustomColumns(PGVectorStore):
    """`PGVectorStore` that promotes selected metadata keys to table columns."""

    custom_columns: Dict[str, str] = Field(
        default_factory=dict,
        description=(
            "Mapping of node-metadata key -> Postgres column type. Each key is "
            "stored in its own column instead of inside the metadata JSON blob."
        ),
    )

    _custom_columns_ready = PrivateAttr(default=False)

    def __init__(self, **kwargs: Any) -> None:
        super().__init__(**kwargs)
        # LlamaIndex has created the table by now -> add our columns.
        self._ensure_custom_columns()

    # ------------------------------------------------------------------ helpers
    @property
    def _qualified_table(self) -> str:
        schema = getattr(self, "schema_name", SCHEMA_NAME) or SCHEMA_NAME
        return f'"{schema}"."{self.table_name}"'

    @property
    def _sync_engine(self) -> Engine:
        engine = getattr(self, "_engine", None)
        if engine is not None:
            return engine
        # Fallback for integration versions that keep the engine elsewhere.
        connection_string = getattr(self, "connection_string", None) or getattr(
            self, "_connection_string", None
        )
        engine = create_engine(connection_string)
        self._engine = engine
        return engine

    def _ensure_custom_columns(self) -> bool:
        """Create the custom columns if the table already exists."""
        if not self.custom_columns:
            return False
        if self._custom_columns_ready:
            return True

        with self._sync_engine.begin() as conn:
            table_exists = conn.execute(
                text("SELECT to_regclass(:tbl)"), {"tbl": self._qualified_table}
            ).scalar()
            if table_exists is None:
                # Table not created yet (e.g. perform_setup=False); retry on add().
                return False
            for column, sql_type in self.custom_columns.items():
                conn.execute(
                    text(
                        f'ALTER TABLE {self._qualified_table} '
                        f'ADD COLUMN IF NOT EXISTS "{column}" {sql_type}'
                    )
                )
        self._custom_columns_ready = True
        return True

    def _strip_promoted_keys(
        self, nodes: Sequence[BaseNode]
    ) -> List[BaseNode]:
        """Copy nodes without the promoted keys, so they are not duplicated in JSON."""
        keys = set(self.custom_columns)
        if not keys:
            return list(nodes)

        stripped: List[BaseNode] = []
        for node in nodes:
            metadata = node.metadata or {}
            promoted = keys & set(metadata)
            if not promoted:
                stripped.append(node)
                continue
            clone = copy.deepcopy(node)
            clone.metadata = {k: v for k, v in metadata.items() if k not in promoted}
            stripped.append(clone)
        return stripped

    def _write_promoted_columns(self, nodes: Sequence[BaseNode]) -> None:
        if not self.custom_columns or not self._ensure_custom_columns():
            return

        assignments = ", ".join(f'"{c}" = :v_{c}' for c in self.custom_columns)
        stmt = text(
            f"UPDATE {self._qualified_table} SET {assignments} WHERE node_id = :node_id"
        )

        payloads = []
        for node in nodes:
            metadata = node.metadata or {}
            if not (set(self.custom_columns) & set(metadata)):
                continue
            payload: Dict[str, Any] = {
                f"v_{c}": metadata.get(c) for c in self.custom_columns
            }
            payload["node_id"] = node.node_id
            payloads.append(payload)

        if not payloads:
            return

        with self._sync_engine.begin() as conn:
            for payload in payloads:
                conn.execute(stmt, payload)

    def _attach_promoted_columns(self, nodes: Sequence[BaseNode]) -> None:
        if not nodes or not self.custom_columns or not self._ensure_custom_columns():
            return

        columns = ", ".join(f'"{c}"' for c in self.custom_columns)
        stmt = text(
            f"SELECT node_id, {columns} FROM {self._qualified_table} "
            "WHERE node_id IN :node_ids"
        ).bindparams(bindparam("node_ids", expanding=True))

        with self._sync_engine.connect() as conn:
            rows = (
                conn.execute(stmt, {"node_ids": [n.node_id for n in nodes]})
                .mappings()
                .all()
            )
        by_node_id = {row["node_id"]: row for row in rows}

        for node in nodes:
            row = by_node_id.get(node.node_id)
            if row is None:
                continue
            for column in self.custom_columns:
                node.metadata[column] = row[column]

    # ------------------------------------------------------------ vector store API
    def add(self, nodes: Sequence[BaseNode], **add_kwargs: Any) -> List[str]:
        """Insert nodes; promoted metadata keys go to their own columns."""
        ids = super().add(self._strip_promoted_keys(nodes), **add_kwargs)
        self._write_promoted_columns(nodes)
        return ids

    async def async_add(
        self, nodes: Sequence[BaseNode], **add_kwargs: Any
    ) -> List[str]:
        # Keep the custom-column logic on the same code path when
        # BasePydanticVectorStore dispatches asynchronously.
        return await asyncio.to_thread(self.add, nodes, **add_kwargs)

    def query(
        self, query: VectorStoreQuery, **kwargs: Any
    ) -> VectorStoreQueryResult:
        result = super().query(query, **kwargs)
        self._attach_promoted_columns(result.nodes or [])
        return result

    index = VectorStoreIndex.from_documents(
        documents, storage_context=storage_context
    )
    return index
