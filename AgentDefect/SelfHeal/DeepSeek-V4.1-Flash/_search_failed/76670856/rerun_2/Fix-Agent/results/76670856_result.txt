from dotenv import load_dotenv
from pathlib import Path
import json
import os
import sys

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.document_loaders import JSONLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.indexes import VectorstoreIndexCreator
from langchain.indexes.vectorstore import VectorStoreIndexWrapper
from langchain.prompts import PromptTemplate
from langchain.vectorstores import Chroma

load_dotenv()

# The JSON key that actually holds the review text.
CONTENT_KEY = "text"

REVIEWS = [
    "Great hotel, excellent service and comfortable rooms.",
    "I had a terrible experience at this hotel. The room was dirty and the staff was rude.",
    "Highly recommended! The hotel has a beautiful view and the staff is friendly.",
    "Average hotel. The room was okay, but nothing special.",
    "I absolutely loved my stay at this hotel. The amenities were top notch.",
    "Disappointing experience. The hotel was overpriced for the quality provided.",
    "The hotel exceeded my expectations. The room was spacious and clean.",
    "Avoid this hotel at all costs! The customer service was horrendous.",
    "Fantastic hotel with a great location. I would definitely stay here again.",
    "Not a bad hotel, but there are better options available in the area.",
]

# The default ConversationalRetrievalChain prompt is written for a generic
# "answer from the retrieved context" task.  For an opinion/review dataset the
# model must (a) only use the reviews it is given and (b) use *all* of them for
# counting/listing questions instead of only the few most similar ones.
QA_PROMPT = PromptTemplate(
    template=(
        "You are an assistant that answers questions about a hotel review dataset.\n"
        "Use ONLY the review excerpts in the context below. Never invent reviews, "
        "ratings, numbers or details that are not in the context.\n"
        "The context contains every review of the dataset, so when the question asks "
        "for a list, a count or an overall summary, enumerate/summarise all of the "
        "reviews you were given instead of only the most relevant ones.\n"
        "If the answer is not contained in the context, say that you don't know.\n\n"
        "Context:\n{context}\n\n"
        "Question: {question}\n"
        "Helpful answer:"
    ),
    input_variables=["context", "question"],
)


def _write_review_file(review_file: Path) -> None:
    """Write the sample dataset, giving every review a stable, explicit id."""
    payload = {
        "reviews": [
            {"id": index + 1, "text": text}
            for index, text in enumerate(REVIEWS)
        ]
    }
    review_file.write_text(json.dumps(payload, indent=2))


def _review_metadata(record: dict, metadata: dict) -> dict:
    """Keep useful metadata (id) alongside the default 'source' metadata."""
    metadata["review_id"] = record.get("id")
    return metadata


def _ensure_text_content(documents, content_key: str = CONTENT_KEY):
    """Guarantee that each Document.page_content is the plain review text.

    Depending on the installed langchain version the JSONLoader may fall back to
    putting the whole JSON object into ``page_content``.  Feeding the LLM a
    serialised dict (``{"text": "..."}``) instead of the review itself is one of
    the reasons the generated answers do not match the data, so normalise it.
    """
    for document in documents:
        content = document.page_content
        if isinstance(content, dict):
            document.page_content = str(content.get(content_key, content))
        elif isinstance(content, str) and content.lstrip().startswith("{"):
            try:
                parsed = json.loads(content)
            except ValueError:
                continue
            if isinstance(parsed, dict) and content_key in parsed:
                document.page_content = str(parsed[content_key])
    return documents


def run_agent(queries):
    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)

    review_file = data_dir / "review.json"

    if not review_file.exists():
        _write_review_file(review_file)

    # Enable to save to disk & reuse the model (for repeated queries on the same data)
    PERSIST = False

    # Number of reviews in the file. It is used as the retrieval depth so that
    # every review is handed to the LLM (the default retriever only returns the
    # 4 most similar documents, which makes "list/count all reviews" questions
    # come back incomplete or plainly wrong).
    payload = json.loads(review_file.read_text())
    num_reviews = len(payload.get("reviews", []))
    top_k = max(num_reviews, 1)

    if PERSIST and os.path.exists("persist"):
        print("Reusing index...\n")
        vectorstore = Chroma(
            persist_directory="persist",
            embedding_function=OpenAIEmbeddings(),
        )
        index = VectorStoreIndexWrapper(vectorstore=vectorstore)
    else:
        # jq_schema ".reviews[]" yields one object per review and content_key
        # "text" picks the review string out of that object, i.e. every review
        # becomes its own Document whose page_content is the review text.
        # (Equivalent alternative: jq_schema=".reviews[].text" with no content_key.)
        loader = JSONLoader(
            file_path=str(review_file),
            jq_schema=".reviews[]",
            content_key=CONTENT_KEY,
            metadata_func=_review_metadata,
        )

        documents = _ensure_text_content(loader.load())

        if PERSIST:
            index = VectorstoreIndexCreator(
                vectorstore_kwargs={"persist_directory": "persist"}
            ).from_documents(documents)
        else:
            index = VectorstoreIndexCreator().from_documents(documents)

        # keep the retrieval depth in sync with the number of ingested chunks
        top_k = max(len(documents), 1)

    retriever = index.vectorstore.as_retriever(search_kwargs={"k": top_k})

    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model="gpt-3.5-turbo", temperature=0),
        retriever=retriever,
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
    )

    chat_history = []
    results = []
    for query in queries:
        if query in ["quit", "q", "exit"]:
            break
        result = chain({"question": query, "chat_history": chat_history})
        print(result["answer"])
        results.append(result["answer"])
        chat_history.append((query, result["answer"]))
    return results


if __name__ == "__main__":
    run_agent(sys.argv[1:])
