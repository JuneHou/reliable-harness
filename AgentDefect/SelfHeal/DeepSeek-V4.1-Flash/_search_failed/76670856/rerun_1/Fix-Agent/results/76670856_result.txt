"""Conversational Q&A over a JSON review file - corrected version.

Why the original code returned wrong answers
--------------------------------------------
1. The retriever was built with ``index.vectorstore.as_retriever()`` and no
   ``search_kwargs``. That silently defaults to ``k=4``, so only the 4 most
   "similar" reviews ever reached the LLM. Questions such as
   "how many total texts are in the JSON file?" cannot be answered from a
   similarity top-4 (and count/aggregate questions are not a similarity
   problem at all), hence the "I don't know the answer".
2. ``ConversationalRetrievalChain`` rephrases every follow-up question with a
   second (default-temperature) LLM before retrieving. From the second turn
   onwards the question that actually reached the retriever/LLM was a
   rewritten, drifting variant ("condense question" corruption), which is why
   the first answer looked fine and every later answer looked wrong.
3. Both LLMs were non-deterministic (no ``temperature=0``) and the QA prompt
   was the generic chat prompt, which never told the model to stay grounded in
   the provided reviews, nor how to answer counting questions.
4. ``JSONLoader``/``jq_schema`` itself is correct here, but depending on the
   langchain version the parsed record can leak into ``page_content`` as the
   whole dict / ``str(dict)`` instead of just the ``text`` value, so the model
   sees ``{'text': '...'}`` noise (and duplicated key names) in the context.

Fixes applied
-------------
* Every review is loaded as its own ``Document`` with *clean* review text
  (``content_key='text'`` verified, plus normalisation and a deterministic
  fallback that reads the JSON directly if ``jq``/``JSONLoader`` misbehaves).
* The retriever is given ``k = number of reviews`` so the full corpus is in
  context (no silent 4-document truncation).
* ``rephrase_question=False`` so the user's literal question is used, with the
  chat history handed to the QA prompt instead of being used to rewrite it.
* Deterministic models (``temperature=0``) and an explicit grounded prompt
  that knows the total number of reviews and how to count/quote them.
* Documents are formatted as ``[Review N] ...`` via ``document_prompt`` (from
  metadata, so the embeddings stay clean) to make counting and attribution
  reliable.
"""

from dotenv import load_dotenv
from pathlib import Path
import json
import os
import sys

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.document_loaders import JSONLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.indexes.vectorstore import VectorStoreIndexWrapper
from langchain.prompts import PromptTemplate
from langchain.schema import Document
from langchain.vectorstores import Chroma

load_dotenv()

SAMPLE_REVIEWS = [
    "Great hotel, excellent service and comfortable rooms.",
    "I had a terrible experience at this hotel. The room was dirty and the staff was rude.",
    "Highly recommended! The hotel has a beautiful view and the staff is friendly.",
    "Average hotel. The room was okay, but nothing special.",
    "I absolutely loved my stay at this hotel. The amenities were top notch.",
    "Disappointing experience. The hotel was overpriced for the quality provided.",
    "The hotel exceeded my expectations. The room was spacious and clean.",
    "Avoid this hotel at all costs! The customer service was horrendous.",
    "Fantastic hotel with a great location. I would definitely stay here again.",
    "Not a bad hotel, but there are better options available in the area.",
]


def _write_sample_file(review_file: Path) -> None:
    payload = {"reviews": [{"text": text} for text in SAMPLE_REVIEWS]}
    review_file.write_text(json.dumps(payload, indent=2))


def _clean_content(raw) -> str:
    """Return only the review text.

    Some langchain versions ignore ``content_key`` and put the whole record
    (a dict, or the string representation of it) into ``page_content``.
    """
    if isinstance(raw, dict):
        raw = raw.get("text", "")
    elif isinstance(raw, str):
        stripped = raw.strip()
        if stripped.startswith("{") and stripped.endswith("}"):
            try:
                obj = json.loads(stripped)
            except json.JSONDecodeError:
                obj = None
            if isinstance(obj, dict) and "text" in obj:
                raw = obj["text"]
        else:
            raw = stripped

    return " ".join(str(raw or "").split())


def load_review_documents(review_file: Path):
    """Load one Document per review, with clean page_content."""
    documents = []

    try:
        loader = JSONLoader(
            file_path=str(review_file),
            jq_schema=".reviews[]",      # one jq match per review
            content_key="text",          # page_content <- record["text"]
        )
        raw_documents = loader.load()
    except Exception as exc:  # jq missing, schema/version mismatch, ...
        print(f"JSONLoader failed ({exc}); falling back to direct JSON parsing.")
        raw_documents = []

    for position, document in enumerate(raw_documents, start=1):
        text = _clean_content(document.page_content)
        if text:
            metadata = dict(document.metadata or {})
            metadata["review_id"] = position
            documents.append(Document(page_content=text, metadata=metadata))

    if documents:
        return documents

    # Deterministic fallback: read the JSON ourselves so the pipeline still
    # indexes exactly one clean document per review.
    payload = json.loads(review_file.read_text())
    for position, record in enumerate(payload.get("reviews", []), start=1):
        text = _clean_content(record)
        if text:
            documents.append(
                Document(
                    page_content=text,
                    metadata={
                        "source": str(review_file),
                        "seq_num": position,
                        "review_id": position,
                    },
                )
            )
    return documents


def build_qa_prompt(num_reviews: int) -> PromptTemplate:
    template = """You are a careful assistant that answers questions about a JSON file
containing exactly {num_reviews} hotel reviews.

Rules:
- Use ONLY the numbered reviews in the context below to answer.
- Each context item is prefixed with "[Review N]"; that N is the review number.
- If the question asks how many reviews/texts there are, answer from the
  numbered items you were given and the stated total of {num_reviews}.
- When asked about a specific review, quote its text and give its number.
- Never invent reviews and never mention reviews that are not in the context.
- If the context genuinely does not contain the answer, reply: I don't know.

Reviews:
{context}

Chat history (for reference only, do not repeat it verbatim):
{chat_history}

Question: {question}
Answer:"""

    prompt = PromptTemplate(
        template=template,
        input_variables=["context", "chat_history", "question", "num_reviews"],
    )
    return prompt.partial(num_reviews=str(num_reviews))


def build_document_prompt() -> PromptTemplate:
    return PromptTemplate(
        input_variables=["page_content", "review_id"],
        template="[Review {review_id}] {page_content}",
    )


def run_agent(queries):
    if isinstance(queries, str):
        queries = [queries]

    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)

    review_file = data_dir / "review.json"
    if not review_file.exists():
        _write_sample_file(review_file)

    # Enable to save to disk & reuse the model (for repeated queries on the same data)
    PERSIST = False
    embeddings = OpenAIEmbeddings()

    documents = load_review_documents(review_file)
    num_reviews = len(documents)
    if num_reviews == 0:
        raise RuntimeError(f"No reviews could be loaded from {review_file}")

    if PERSIST and os.path.exists("persist"):
        print("Reusing index...\n")
        vectorstore = Chroma(persist_directory="persist", embedding_function=embeddings)
        index = VectorStoreIndexWrapper(vectorstore=vectorstore)
    else:
        # Build the store directly from the documents: one review == one vector,
        # no text splitter fragmenting (and therefore losing) reviews.
        vectorstore = Chroma.from_documents(
            documents,
            embeddings,
            persist_directory="persist" if PERSIST else None,
        )
        if PERSIST:
            vectorstore.persist()
        index = VectorStoreIndexWrapper(vectorstore=vectorstore)

    try:
        total_vectors = index.vectorstore._collection.count()
    except Exception:
        total_vectors = num_reviews

    # KEY FIX: retrieve *every* review instead of the default top-4.
    k = max(num_reviews, total_vectors, 1)
    retriever = index.vectorstore.as_retriever(search_kwargs={"k": k})

    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model="gpt-3.5-turbo", temperature=0),
        retriever=retriever,
        condense_question_llm=ChatOpenAI(model="gpt-3.5-turbo", temperature=0),
        # Do not let a second LLM rewrite the user's question; keep the chat
        # history in the prompt instead. This is what stopped the follow-up
        # questions from silently drifting into wrong answers.
        rephrase_question=False,
        combine_docs_chain_kwargs={
            "prompt": build_qa_prompt(num_reviews),
            "document_prompt": build_document_prompt(),
        },
        return_source_documents=True,
    )

    chat_history = []
    results = []
    for query in queries:
        if query in ["quit", "q", "exit"]:
            break
        if not query or not query.strip():
            continue

        result = chain({"question": query, "chat_history": chat_history})
        answer = result["answer"].strip()
        print(answer)
        results.append(answer)
        chat_history.append((query, answer))

    return results


if __name__ == "__main__":
    demo_queries = [
        "How many total texts are in the JSON file?",
        "Summarize all of the reviews.",
        "Which review complains about rude staff?",
    ]
    run_agent(sys.argv[1:] or demo_queries)
