"""Load an Excel file with a LangChain ReAct agent and get the resulting
pandas DataFrame back in the *main* script - not just the agent's text summary.
"""

import dotenv
import pandas as pd
from langchain import hub
from langchain.agents import AgentExecutor, create_react_agent
from langchain_community.tools import ShellTool
from langchain_core.tools import Tool, tool
from langchain_experimental.utilities import PythonREPL
from langchain_openai import ChatOpenAI

dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# Shared state
# ---------------------------------------------------------------------------
# The agent runs inside LangChain and only hands back a string. To be able to
# read the DataFrame from the calling script we keep a single shared object
# that both the tools and `__main__` look at.
AGENT_STATE = {}

# One persistent namespace used by the PythonREPL tool: every variable the agent
# assigns (e.g. `df = ...`) stays alive after the agent finishes, and we can
# inspect it from `__main__`.
SHARED_NAMESPACE = {"pd": pd}


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and store the transformed pandas DataFrame.

    The DataFrame is saved in AGENT_STATE['df'] (and mirrored into the shared
    namespace as `df`) so the calling script can reuse it directly after the
    agent finishes. Only a short text summary is returned to the LLM.
    """
    try:
        df = pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    except (TypeError, ValueError):
        # Some sheets already have the right column layout.
        df = pd.read_excel(filepath, skiprows=2).reset_index(drop=True)
    except FileNotFoundError:
        return f"File not found: {filepath}"

    objects = df.select_dtypes(include=["object"])
    result = objects if not objects.empty else df

    AGENT_STATE["df"] = result
    AGENT_STATE["filepath"] = filepath
    SHARED_NAMESPACE["df"] = result

    return (
        f"Loaded '{filepath}' into a DataFrame named 'df' with shape "
        f"{result.shape} and columns {list(result.columns)}.\n"
        f"Preview:\n{result.head().to_string()}"
    )


def run_agent(user_prompt, file_name="data.json"):
    """Run the ReAct agent and return its response dict.

    After this call the loaded DataFrame is available through AGENT_STATE
    (see `get_dataframe`).
    """
    llm = ChatOpenAI(model="gpt-4o-mini")

    # IMPORTANT: reuse the shared namespace instead of a private one, otherwise
    # anything assigned inside the REPL is lost when the agent returns.
    python_repl = PythonREPL(_globals=SHARED_NAMESPACE, _locals=SHARED_NAMESPACE)
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description=(
            "Run Python commands in a REPL environment. The pandas library is already "
            "imported as 'pd' and the loaded DataFrame is available as 'df'; keep that "
            "variable name so the main program can access it."
        ),
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(
        agent=shell_agent,
        tools=tools_list,
        verbose=True,
        handle_parsing_errors=True,
        return_intermediate_steps=True,
        max_iterations=10,
    )

    response = agent_executor.invoke({"input": user_prompt})

    # Fallback: if the tool was not called but the agent created `df` by hand in
    # the REPL, pick it up from the shared namespace.
    if "df" not in AGENT_STATE:
        candidate = SHARED_NAMESPACE.get("df")
        if isinstance(candidate, pd.DataFrame):
            AGENT_STATE["df"] = candidate

    return response


def get_dataframe():
    """Return the DataFrame produced by the agent (raises if there is none)."""
    df = AGENT_STATE.get("df")
    if not isinstance(df, pd.DataFrame):
        raise RuntimeError(
            "The agent did not produce a DataFrame. Inspect the agent output above."
        )
    return df


if __name__ == "__main__":
    path = "excel_file.xlsx"
    user_prompt = (
        f"Load the Excel file located at {path} using the load_excel_response tool "
        "and assign the result to a variable named df. "
        "Do not print the entire file, just confirm that it was loaded."
    )

    result = run_agent(user_prompt)
    print(result["output"])

    # The actual data, as a plain local variable in the main script.
    df = get_dataframe()
    print(df.shape)
    print(df.head())

# ---------------------------------------------------------------------------

# ---------------------------------------------------------------------------
# The agent runs in its own "world": every tool call happens inside
# langchain/langchain_experimental, and whatever the agent creates (e.g. `df`)
# dies with the agent run.  To be able to read the result from the MAIN script
# we give the tools an explicit, shared namespace (a plain dict living in the
# main module) instead of letting them create throw-away local variables.
AGENT_NAMESPACE: dict = {}


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file, store the transformed DataFrame in the shared
    namespace as the variable `df`, and return a short JSON description."""
    df = pd.read_excel(
        filepath, skiprows=2, names=["responses"]
    ).reset_index(drop=True)
    df = df.select_dtypes(include=["object"])

    # Keep the real DataFrame in the shared namespace (NOT only a string copy)
    AGENT_NAMESPACE["df"] = df

    # Tools must return text, otherwise the agent cannot put it in the
    # "Observation" of the ReAct loop -> return a serialisable description.
    return json.dumps(
        {
            "shape": list(df.shape),
            "columns": [str(c) for c in df.columns],
            "preview": df.head().astype(str).to_dict(orient="records"),
            "note": "The full DataFrame is bound to the variable `df`.",
        }
    )


def get_dataframe(name: str = "df") -> pd.DataFrame:
    """Return the DataFrame the agent produced, as a normal local variable."""
    if name not in AGENT_NAMESPACE or not isinstance(

    ):
        raise RuntimeError(
            f"`{name}` was not created by the agent. "
            f"Available objects: {sorted(AGENT_NAMESPACE)}"
