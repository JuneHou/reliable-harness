from dotenv import load_dotenv
import os

from langchain_community.document_loaders import PyPDFLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_mongodb import MongoDBAtlasVectorSearch
from pymongo import MongoClient


def run_agent(pdf_url):
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key is None:
        raise ValueError("OPENAI_API_KEY not found")

    # A real MongoDB Atlas connection string is required: the previous code passed a
    # plain Python list (`atlas_collection = []`) as the "collection", so documents were
    # never actually persisted in Atlas and the vector search had nothing to query,
    # which is why `similarity_search` always returned an empty array.
    mongodb_uri = os.getenv("MONGODB_ATLAS_CLUSTER_URI")
    if mongodb_uri is None:
        raise ValueError("MONGODB_ATLAS_CLUSTER_URI not found")

    mongo_client = MongoClient(mongodb_uri)
    atlas_collection = mongo_client["langchain_db"]["vector_search"]

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()

    docs = text_splitter.split_documents(data)

    vector_search_index = "vector_index"

    # `from_documents` writes the embedded chunks into the Atlas collection
    # using the exact fields expected by the Atlas Vector Search index
    # (embedding / text / metadata), so a matching index named
    # "vector_index" on that collection must exist for queries to return hits.
    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(
            disallowed_special=(),
            model="text-embedding-ada-002"
        ),
        collection=atlas_collection,
        index_name=vector_search_index
    )

    query = "What were the compute requirements for training GPT 4"

    results = vector_search.similarity_search(query)

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"

    results = run_agent(url)

    print("Number of results:", len(results))
    for doc in results:
        print("Result:", doc.page_content[:500])
        print("Metadata:", doc.metadata)
