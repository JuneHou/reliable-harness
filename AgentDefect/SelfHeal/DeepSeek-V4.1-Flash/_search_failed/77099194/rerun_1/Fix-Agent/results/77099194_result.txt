from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate

load_dotenv()

# The QA prompt must make it clear that several CSV rows (one document per row)
# are provided and that ALL of them have to be considered, otherwise the model
# tends to answer with just the first matching row it sees.
QA_TEMPLATE = """You are a data analyst answering questions about a movie database.
Each document in the context below is one row of the CSV file, so the context can
contain many rows (one movie per row).

Read EVERY row in the context before answering and do not stop at the first
matching row. If several rows satisfy the question, list every one of them
(and say so explicitly).

Context:
{context}

Question: {question}

Answer (list all matching rows found in the context):"""

QA_PROMPT = PromptTemplate(
    template=QA_TEMPLATE,
    input_variables=["context", "question"],
)


def run_agent(prompt):
    embeddings = OpenAIEmbeddings()

    # Load one document per CSV row
    csvloader = CSVLoader(file_path="movies.csv", encoding="utf-8")
    csvdocuments = csvloader.load()

    search = Chroma.from_documents(csvdocuments, embeddings)

    # BUG FIX: with k=1 only the single most similar row was passed to the LLM,
    # so only "Author A" (the row that happened to rank first) was reported and
    # every other matching row (e.g. "Author C") was silently dropped.
    # Retrieve as many rows as the index holds so the whole relevant context is
    # available to the model.
    retriever = search.as_retriever(
        search_type="similarity",
        search_kwargs={"k": len(csvdocuments)},
    )

    chain = ConversationalRetrievalChain.from_llm(
        ChatOpenAI(temperature=0.0),
        retriever=retriever,
        return_source_documents=True,
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
    )

    # BUG FIX: chat_history must be a list of (human, ai) tuples (or message
    # objects), not an empty string.
    result = chain(
        {"question": prompt, "chat_history": []},
        return_only_outputs=True,
    )
    return result["answer"]


if __name__ == '__main__':
    print(run_agent("which movies are action"))
