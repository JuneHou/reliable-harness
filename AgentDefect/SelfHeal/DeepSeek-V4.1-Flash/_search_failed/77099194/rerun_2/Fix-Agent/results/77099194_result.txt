"""
Why the original code returned only "Author A":

1. `CSVLoader` stores ONE Document per CSV row, and
   `search.as_retriever(search_kwargs={"k": 1})` then hands the LLM
   **exactly one row** (the single closest embedding match).
   So even if several rows (e.g. Author A *and* Author C for 1999 films)
   match the question, only the first one ever reaches the model and the
   rest are silently dropped -> the answer is truncated/incomplete.

2. `chat_history` was passed as the string "" instead of a list of
   (human, ai) tuples, which makes the condense-question step ill-defined.

3. The default "stuff" prompt has no instruction to enumerate *every*
   matching row, so the model is free to answer with just the one row it got.

Fix: retrieve enough documents (k = number of rows in this small CSV),
use a proper empty list for chat_history, and use an explicit prompt that
requires the model to scan all context rows and list every match.
"""

from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.prompts import PromptTemplate
from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()

CSV_PATH = "movies.csv"

# Prompt that forces the model to use ALL retrieved rows instead of just one.
QA_PROMPT = PromptTemplate.from_template(
    """You are answering questions about a movies database.
Every line of the context below is exactly one row of the CSV file
(it contains the movie title, the release year, the author/director, etc.).

Read EVERY row of the context before answering.
List ALL rows that satisfy the condition in the question - there can be
several authors/movies that match, and every one of them must be reported.
Do not stop after the first match, and do not invent rows that are not in
the context. If no row matches, say that no matching row exists.

Context:
{context}

Question: {question}

Complete answer (list every matching movie together with all of its columns):"""
)


def run_agent(prompt, chat_history=None):
    """Answer a question over movies.csv.

    chat_history must be a list of (human, ai) tuples, e.g. []
    """
    if chat_history is None:
        chat_history = []

    embeddings = OpenAIEmbeddings()

    csvloader = CSVLoader(file_path=CSV_PATH, encoding="utf-8")
    csvdocuments = csvloader.load()

    search = Chroma.from_documents(csvdocuments, embeddings)

    # FIX: k=1 only ever kept a single CSV row, so only the first matching
    # movie/author reached the LLM. Retrieve enough rows to cover the file.
    k = max(1, min(len(csvdocuments), 20))

    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0),
        retriever=search.as_retriever(
            search_type="similarity",
            search_kwargs={"k": k},
        ),
        return_source_documents=True,
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
    )

    result = chain({"question": prompt, "chat_history": chat_history})

    # Optional: helps you verify that all the rows really were retrieved.
    for doc in result.get("source_documents", []):
        print(doc.metadata, doc.page_content)

    return result["answer"]


if __name__ == "__main__":
    print(run_agent("which movies are action"))
