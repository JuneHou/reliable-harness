"""
Why the original output looked wrong
------------------------------------
``RetrievalQAWithSourcesChain.from_chain_type(...)`` does NOT accept a bare
``prompt=`` argument. The prompt actually lives on the *inner* documents chain
(``chain.combine_documents_chain.llm_chain.prompt``). Therefore:

1. ``print(chain)`` just dumps the object repr, which always shows the built-in
   default templates (``stuff_prompt.PROMPT`` with
   ``input_variables=['summaries', 'question']`` etc.). Printing a chain object
   can never be "blank" nor show your custom template unless you actually
   inject one.
2. Because no prompt was ever supplied, the default template is *also* the one
   used at query time - which is why every call produced the same generic
   "Given the following extracted parts of a long document..." behaviour.
3. The chain was never invoked with a real question (``chain({"question": ...})``),
   so the user only ever saw the chain repr.

Fix: build an explicit ``PromptTemplate`` whose input_variables match what the
``stuff`` QA-with-sources chain expects (``summaries`` + ``question``), hand it
over through ``chain_type_kwargs={"prompt": ...}`` (or via
``from_llm(question_prompt=..., combine_prompt=...)``), then actually invoke the
chain with a question and inspect the *inner* prompt to verify it was applied.
"""

import os

from dotenv import load_dotenv

from langchain_openai import ChatOpenAI, OpenAIEmbeddings

# Robust imports across langchain versions
try:  # langchain >= 0.2 / langchain_classic
    from langchain.chains import RetrievalQAWithSourcesChain
except ImportError:  # pragma: no cover
    from langchain_classic.chains import RetrievalQAWithSourcesChain
try:
    from langchain.prompts import PromptTemplate
except ImportError:  # pragma: no cover
    from langchain_core.prompts import PromptTemplate
try:
    from langchain.vectorstores import FAISS
except ImportError:  # pragma: no cover
    from langchain_community.vectorstores import FAISS


load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


# ---------------------------------------------------------------------------
# 1. Explicit custom prompt for the "stuff" QA-with-sources chain.
#    The input variables MUST be exactly ["summaries", "question"]:
#    ``summaries`` is the retrieved, source-annotated context and ``question``
#    is the user query. The model must keep emitting "SOURCES: ..." because
#    RetrievalQAWithSourcesChain parses that section out of the answer.
# ---------------------------------------------------------------------------
CUSTOM_QA_TEMPLATE = """You are a precise assistant. Answer the question using ONLY the
provided summaries. If the answer is not contained in them, reply
"Not found in the provided documents." Keep the answer short.

Summaries:
{summaries}

Question: {question}

Answer (state the facts from the summaries only):
SOURCES:"""

CUSTOM_PROMPT = PromptTemplate(
    template=CUSTOM_QA_TEMPLATE,
    input_variables=["summaries", "question"],  # <- must match the chain
)


def run_agent():
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = [
        "This is a test document about France.",
        "Another document about Germany.",
    ]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm=llm,                       # pass by keyword: llm is not the 1st arg in all versions
        chain_type="stuff",
        retriever=db.as_retriever(),
        return_source_documents=True,
        # >>> the actual fix: inject the custom prompt <<<
        chain_type_kwargs={"prompt": CUSTOM_PROMPT},
    )
    return chain


def run_agent_from_llm():
    """Alternative (more explicit) way: control the question and combine prompts."""
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = [
        "This is a test document about France.",
        "Another document about Germany.",
    ]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]
    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    question_template = """Use the following context to answer the question at the end.
Only use information from the context. End with "SOURCES: <sources>".

{summaries}

Question: {question}
Helpful answer:"""
    question_prompt = PromptTemplate(
        template=question_template,
        input_variables=["summaries", "question"],
    )

    return RetrievalQAWithSourcesChain.from_llm(
        llm=llm,
        retriever=db.as_retriever(),
        return_source_documents=True,
        question_prompt=question_prompt,
        # combine_prompt=... can also be set when several document batches
        # have to be merged (map_reduce / refine style chains).
    )


if __name__ == "__main__":
    chain = run_agent()

    # Direct evidence that the custom template replaced the default one:
    inner_prompt = chain.combine_documents_chain.llm_chain.prompt
    print("--- prompt actually used by the chain ---")
    print(inner_prompt.template)
    assert inner_prompt.template == CUSTOM_QA_TEMPLATE, "custom prompt was not applied"

    # The chain must be *invoked* with a question - printing it alone shows only
    # the default template repr, which is what caused the confusion.
    result = chain.invoke({"question": "Which country is described in doc1?"})
    print("\n--- answer ---")
    print(result["answer"])
    print("\n--- sources ---")
    print(result["sources"])
