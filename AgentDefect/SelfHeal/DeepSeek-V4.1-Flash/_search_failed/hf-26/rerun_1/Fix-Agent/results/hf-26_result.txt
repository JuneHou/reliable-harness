# https://discuss.huggingface.co/t/getting-additional-response-from-my-rag-using-huggingfaceendpoint-inference/145964
#
# ROOT CAUSE
# ----------
# `create_stuff_documents_chain` was given a `HuggingFaceEndpoint`, i.e. a *raw
# text-generation (completion) model*. A completion model has no notion of chat
# roles, so LangChain renders the whole `ChatPromptTemplate`
# ("System: ... Human: ... Assistant:") into one long plain string and sends it
# to the model as the prompt.
#
# Serverless text-generation endpoints do not understand/conform to
# Zephyr's ChatML template when fed a raw string, and the endpoint echoes the
# prompt back, so the returned text looks like:
#
#   System: You are an assistant for question-answering tasks...
#   Context: ...
#   Human: Which runtime does Transformers.js uses?
#   Assistant: <the real answer>
#
# i.e. "extra text" before the final "Assistant:" answer.
#
# FIX
# ---
# Wrap the endpoint in `ChatHuggingFace` (it was imported but never used!). It
# is a *chat* model: it applies the tokenizer's chat template
# (Zephyr's `<|system|> ... <|user|> ... <|assistant|>`) to the incoming
# messages, so the model only generates the assistant turn and the answer comes
# back clean. The prompt is also declared explicitly as chat messages with the
# correct roles/placeholders instead of relying on a hub prompt being
# stringified.

import os

from dotenv import load_dotenv

from langchain.chains.combine_documents import create_stuff_documents_chain
from langchain.chains.retrieval import create_retrieval_chain
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain_community.vectorstores import FAISS
from langchain_core.documents import Document
from langchain_core.prompts import ChatPromptTemplate
from langchain_huggingface import ChatHuggingFace, HuggingFaceEndpoint

load_dotenv()

api_token = os.getenv("HUGGINGFACEHUB_API_TOKEN")
if not api_token:
    raise ValueError("HUGGINGFACEHUB_API_TOKEN not found in .env file")
os.environ["HUGGINGFACEHUB_API_TOKEN"] = api_token

REPO_ID = "huggingfaceh4/zephyr-7b-alpha"

SYSTEM_TEMPLATE = (
    "You are an assistant for question-answering tasks. "
    "Use the following pieces of retrieved context to answer the question. "
    "If you don't know the answer, just say that you don't know. "
    "Use three sentences maximum and keep the answer concise.\n\n"
    "Context:\n{context}"
)


def build_chat_model() -> ChatHuggingFace:
    """Text-generation endpoint wrapped as a *chat* model."""
    endpoint = HuggingFaceEndpoint(
        repo_id=REPO_ID,
        task="text-generation",
        temperature=0.5,
        max_new_tokens=1024,
    )
    # ChatHuggingFace applies the model's chat template, so only the assistant
    # turn is generated (no echoed system/human prompt in the output).
    return ChatHuggingFace(llm=endpoint, model_id=REPO_ID)


def build_prompt() -> ChatPromptTemplate:
    return ChatPromptTemplate.from_messages(
        [
            ("system", SYSTEM_TEMPLATE),
            ("human", "{input}"),
        ]
    )


def run_agent(question: str) -> str:
    documents = [
        Document(
            page_content="Transformers.js uses ONNX Runtime to run models in the browser.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="ONNX Runtime is a cross-platform inference engine.",
            metadata={"source": "docs"},
        ),
        Document(
            page_content="Transformers.js enables running transformer models client-side.",
            metadata={"source": "docs"},
        ),
    ]

    embeddings = HuggingFaceEmbeddings(model_name="sentence-transformers/all-MiniLM-L6-v2")
    vectorstore = FAISS.from_documents(documents, embeddings)
    retriever = vectorstore.as_retriever(search_kwargs={"k": 2})

    chat_model = build_chat_model()
    prompt = build_prompt()

    combine_docs_chain = create_stuff_documents_chain(chat_model, prompt)
    rag_chain = create_retrieval_chain(retriever, combine_docs_chain)

    result = rag_chain.invoke({"input": question})

    # ChatHuggingFace returns only the generated assistant message, so
    # `result["answer"]` is the bare answer (no "System:/Human:/Assistant:" echo).
    return result["answer"].strip()


if __name__ == "__main__":
    query = "Which runtime does Transformers.js uses?"
    response = run_agent(query)
    print(f"Input: {query}")
    print(f"answer: {response}")
