"""Fixed version of the CSV question/answering script.

Why the original version returned the wrong output and was so expensive
----------------------------------------------------------------------
* ``chain_type="map_reduce"`` does not reuse the ``stuff`` prompt. Out of the
  box it uses the generic QA map-reduce prompts (``QUESTION_PROMPT`` /
  ``COMBINE_PROMPT``):
      map   : "Use the following portion ... Return any relevant text verbatim."
      reduce: "... create a final answer with references (SOURCES). If you
               don't know the answer, just say that you don't know."
  For an exhaustive question such as "tell me the name of every driver who is
  German" the per-document answers are lossy and the final reduce/collapse step
  drops names (or answers "I don't know"), which is exactly why the answer
  differs from the one produced by ``stuff``.
* ``search_kwargs={"k": 1000}`` retrieves *every* row of the CSV, and
  ``map_reduce`` calls the LLM once per retrieved document -> one map call per
  CSV row (thousands of OpenAI calls per query, per file). That is where the
  unexpectedly high API cost comes from, and it is also what makes the
  collapse step lose information.

What this version changes
-------------------------
1. CSV rows are grouped into a handful of larger documents, so the map/reduce
   chain makes tens of LLM calls instead of thousands (much cheaper) while
   still seeing every row (so "every ..." questions stay exhaustive).
2. Explicit, purpose-built map/reduce prompts are passed through
   ``chain_type_kwargs`` using the correct keys for the map_reduce chain
   (``question_prompt`` and ``combine_prompt`` with the input variables
   ``context``/``question`` and ``summaries``/``question``). They force the
   model to quote every relevant record verbatim and to merge all partial
   answers without dropping or summarising items away.
3. The vectorstore/retriever is built once per file and ``k`` is the number of
   documents that actually exist, instead of the meaningless ``k=1000``.
"""

import os
import warnings

from dotenv import load_dotenv, find_dotenv

from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.document_loaders import CSVLoader
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.llms import OpenAI
from langchain.prompts import PromptTemplate
from langchain.vectorstores import Chroma

try:  # older langchain
    from langchain.docstore.document import Document
except ImportError:  # newer langchain
    from langchain_core.documents import Document


warnings.filterwarnings('ignore')

load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY") or ""

# Number of CSV rows packed into a single document. Keeping this reasonably
# large is what keeps the number of (paid) map calls low.
ROWS_PER_CHUNK = 40

# Map step prompt: extract every relevant record verbatim from one chunk.
MAP_PROMPT = PromptTemplate(
    input_variables=["context", "question"],
    template=(
        "You are given a part of a CSV file. Each line of the text below is one "
        "record of that file.\n\n"
        "Read the text and extract EVERY fact from it that is relevant to answering "
        "the question. Quote the relevant records verbatim, one record per line. "
        "Do not summarise, do not paraphrase, do not interpret and do not drop any "
        "matching record.\n"
        "If nothing in this part of the file is relevant, answer exactly: NONE\n\n"
        "Text:\n{context}\n\n"
        "Question: {question}\n"
        "Relevant records (verbatim, one per line, or NONE):"
    ),
)

# Reduce/collapse step prompt: merge all partial answers without losing items.
COMBINE_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=(
        "You are given several extracts taken from the same CSV file. Each extract "
        "contains only records that are relevant to the question.\n\n"
        "Combine all of them into one complete answer to the question.\n"
        "Rules:\n"
        "- keep every item that appears in ANY of the extracts,\n"
        "- remove exact duplicates only,\n"
        "- never summarise away, shorten or omit an item,\n"
        "- never add information that is not present in the extracts.\n"
        "If the extracts contain no relevant information at all, answer exactly: NONE\n\n"
        "Extracts:\n{summaries}\n\n"
        "Question: {question}\n"
        "Complete answer:"
    ),
)


def load_chunked_documents(path, rows_per_chunk=ROWS_PER_CHUNK):
    """Load a CSV file and pack its rows into a few larger documents."""
    loader = CSVLoader(file_path=path, encoding="utf-8")
    rows = loader.load()

    if not rows:
        return []

    documents = []
    for start in range(0, len(rows), rows_per_chunk):
        group = rows[start:start + rows_per_chunk]
        content = "\n".join(row.page_content.replace("\n", "; ") for row in group)
        documents.append(
            Document(
                page_content=content,
                metadata={
                    "source": path,
                    "rows": f"{start}-{start + len(group) - 1}",
                },
            )
        )
    return documents


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    for file in files:
        if not os.path.exists(file):
            print(f"Skipping missing file: {file}")
            continue

        print("=====================================")
        print(file)
        print("=====================================")

        with get_openai_callback() as cb:
            llm = OpenAI(temperature=0, batch_size=20)

            # Group rows so that map_reduce performs a few calls, not one per row.
            docs = load_chunked_documents(file)
            if not docs:
                print(f"No rows found in {file}")
                continue

            embeddings = OpenAIEmbeddings()

            # Build the vectorstore (index) once per file.
            db = Chroma.from_documents(docs, embeddings)

            # k must be a sensible number of documents: retrieving every row
            # (k=1000) is what made map_reduce both slow/expensive and lossy.
            retriever = db.as_retriever(
                search_type="similarity",
                search_kwargs={"k": len(docs)},
            )

            for query in queries:
                # Custom map/reduce prompts: this is the actual fix for the
                # wrong output, chain_type_kwargs expects `question_prompt`
                # and `combine_prompt` for the map_reduce chain type.
                qa_chain = RetrievalQA.from_chain_type(
                    llm=llm,
                    chain_type="map_reduce",
                    retriever=retriever,
                    chain_type_kwargs={
                        "question_prompt": MAP_PROMPT,
                        "combine_prompt": COMBINE_PROMPT,
                    },
                    verbose=True,
                )

                print(query)
                result = qa_chain.run(query)
                results.append(result)

                print(result)

            # Cost report for the whole file (index + all queries).
            print(cb)

    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German"
    ]
    res = run_agent(queries)
    print(res)

# vague "summary" of the retrieved chunks.
QA_PROMPT = PromptTemplate(
    template="""You are given rows retrieved from a CSV file of driver records.
Answer the question using ONLY the information contained in those rows.
If the answer cannot be found in the rows, reply exactly: I don't know.

{context}

Question: {question}
Answer (when the question asks for a list, list EVERY matching entry found in the rows):""",
    input_variables=["context", "question"],
)


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    # Embeddings only need to be instantiated once and reused.
    embeddings = OpenAIEmbeddings()

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:

            loader = CSVLoader(file_path=file, encoding='utf-8')
            docs = loader.load()

            # create the vectorestore to use as the index
            db = Chroma.from_documents(docs, embeddings)

            # CSVLoader creates ONE Document per CSV row. Because of that:
            #  * search_kwargs={"k": 1000} fed (up to) a thousand rows to the
            #    chain, and a map_reduce chain issues one LLM call per document
            #    (N map calls + combine calls) -> the huge API bill.
            #  * every map call only saw a single row, so questions that need
            #    all rows at once ("every driver who is German", "how many ...")
            #    were answered with "None"/"not relevant" per row and the
            #    reduce step then produced no answer at all.
            # Retrieve a small, bounded number of the most relevant rows and
            # answer them in a single ("stuff") prompt instead.
            k = min(len(docs), 5) if docs else 1
            retriever = db.as_retriever(
                search_type="similarity",
                search_kwargs={"k": k},
            )

            for query in queries:
                # "stuff" puts all retrieved rows in a single prompt, so the
                # model can aggregate over them (map_reduce processed each row
                # in isolation and lost the information needed to answer).
                qa_stuff = RetrievalQA.from_chain_type(
                    llm=OpenAI(temperature=0, batch_size=20),
                    chain_type="stuff",
                    retriever=retriever,
                    chain_type_kwargs={"prompt": QA_PROMPT},
                    verbose=True,
                )

                print(query)
                result = qa_stuff.run(query)
                results.append(result)

                print(result)

            print(cb)
    return results
