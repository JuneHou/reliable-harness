# ---------------------------------------------------------------------------
# `map_reduce` uses TWO different prompts and each one expects very specific
# input variables:
#
#   * question_prompt (the "map" step) -> input vars:  context, question
#   * combine_prompt  (the "reduce" step) -> input vars: summaries, question
#
# The LangChain defaults are written for long *narrative* documents:
#
#   map   : "Use the following portion of a long document to see if any of
#            the text is relevant to answer the question.
#            Return any relevant text verbatim. ..."
#   reduce: "... If you don't know the answer, just say that you don't know.
#            Don't try to make up an answer. ..."
#
# With a CSV loader every retrieved row is a tiny document, so most map steps
# answer "no relevant text", and the reduce step then concludes "I don't
# know" (or drops rows while summarizing).  That is exactly why the very same
# query answered fine with chain_type="stuff" (a single prompt that simply
# stuffs all retrieved rows) but produced wrong/incomplete output with
# chain_type="map_reduce".
#
# The prompts below are explicit about the task, keep the row information
# verbatim in the map step, and force the reduce step to list every match
# instead of summarizing them away.
# ---------------------------------------------------------------------------
MAP_PROMPT = PromptTemplate(
    input_variables=["context", "question"],
    template=(
        "The text below is a single row of a CSV file containing information "
        "about Formula 1 drivers.\n"
        "Decide whether this row helps to answer the question.\n"
        "If it does, copy the relevant information verbatim - keep every "
        "name, nationality and value exactly as written.\n"
        "If it does not, reply with exactly: NO RELEVANT INFORMATION\n\n"
        "Row:\n"
        "{context}\n\n"
        "Question: {question}\n"
        "Relevant information:"
    ),
)

REDUCE_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=(
        "Below are pieces of information extracted from the rows of a CSV "
        "file about Formula 1 drivers. Any line that says "
        "'NO RELEVANT INFORMATION' must be ignored.\n\n"
        "Question: {question}\n\n"
        "Extracted information:\n"
        "{summaries}\n\n"
        "Answer the question using ONLY the information above.\n"
        "Rules:\n"
        "1. If the question asks for the names of drivers matching a "
        "condition, list EVERY matching driver, one per line.\n"
        "2. Never omit a match and never invent a driver that is not present "
        "in the extracted information.\n"
        "3. If nothing matches, say that you don't know.\n\n"
        "Final answer:"
    ),
)


def build_qa_chain(file):
    """Load a CSV file, embed it and build a map_reduce RetrievalQA chain."""
    loader = CSVLoader(file_path=file, encoding='utf-8')
    docs = loader.load()

    embeddings = OpenAIEmbeddings()

    # create the vectorstore to use as the index
    db = Chroma.from_documents(docs, embeddings)

    # NOTE: the original code used search_kwargs={"k": 1000}, i.e. it fed the
    # *whole* CSV to the chain for every question.  With map_reduce the map
    # step runs one LLM call per retrieved document, so k=1000 meant ~1000
    # prompts (= a huge number of input tokens and calls) for a single query.
    # Retrieving a handful of relevant rows is both correct and far cheaper.
    retriever = db.as_retriever(search_type="similarity", search_kwargs={"k": 6})

    return RetrievalQA.from_chain_type(
        llm=OpenAI(temperature=0, batch_size=20, max_tokens=-1),
        chain_type="map_reduce",
        retriever=retriever,
        # the custom prompts must be handed to the map_reduce chain through
        # chain_type_kwargs, using the keys it actually accepts
        chain_type_kwargs={
            "question_prompt": MAP_PROMPT,
            "combine_prompt": REDUCE_PROMPT,
        },
        verbose=True,
    )


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:

            # build the vectorstore / chain once per file, not once per query

            for query in queries:
                print(query)
                # chain_type="map_reduce" + the prompts above returns the full
                # list of matching drivers instead of a truncated summary
                result = qa_chain.run(query)
                results.append(result)

                print(result)

            print(cb)
    return results

def normalise_driver_list(raw_answer: str) -> str:
    """Turn the LLM's free-form combine output into a clean, deduped name list."""
    if not raw_answer or not raw_answer.strip():
        return "No matching drivers found."

    # Drop the sentinel value the map step emits for non-matching rows.
    cleaned = re.sub(r"\bno match(es)?\b", "", raw_answer, flags=re.IGNORECASE)
    cleaned = re.sub(r"^.*?:\s*", "", cleaned, count=1) if cleaned.lower().startswith(
        ("answer", "final answer", "drivers", "names")
    ) else cleaned

    seen, names = set(), []
    for part in re.split(r"[,;\n\r]+", cleaned):
        name = _BULLET.sub("", part).strip(" \t.-\"'*:()")
        if not name or _NOISE.match(name):
            continue
        key = name.casefold()
        if key in seen:
            continue
        seen.add(key)
        names.append(name)

    if not names:
        return "No matching drivers found."
    return ", ".join(names)

import os
import warnings

from dotenv import load_dotenv, find_dotenv

from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.prompts import PromptTemplate

warnings.filterwarnings('ignore')

load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


# ---------------------------------------------------------------------------
# `map_reduce` uses TWO different prompts and each one expects very specific
# input variables:
#
#   * question_prompt (the "map" step) -> input vars:  context, question
#   * combine_prompt  (the "reduce" step) -> input vars: summaries, question
#
# The LangChain defaults are written for long *narrative* documents:
#
#   map   : "Use the following portion of a long document to see if any of
#            the text is relevant to answer the question.
#            Return any relevant text verbatim. ..."
#   reduce: "... If you don't know the answer, just say that you don't know.
#            Don't try to make up an answer. ..."
#
# With a CSV loader every retrieved row is a tiny document, so most map steps
# answer "no relevant text", and the reduce step then concludes "I don't
# know" (or drops rows while summarizing).  That is exactly why the very same
# query answered fine with chain_type="stuff" (a single prompt that simply
# stuffs all retrieved rows) but produced wrong/incomplete output with
# chain_type="map_reduce".
#
# The prompts below are explicit about the task, keep the row information
# verbatim in the map step, and force the reduce step to list every match
# instead of summarizing them away.
# ---------------------------------------------------------------------------
MAP_PROMPT = PromptTemplate(
    input_variables=["context", "question"],
    template=(
        "The text below is a single row of a CSV file containing information "
        "about Formula 1 drivers.\n"
        "Decide whether this row helps to answer the question.\n"
        "If it does, copy the relevant information verbatim - keep every "
        "name, nationality and value exactly as written.\n"
        "If it does not, reply with exactly: NO RELEVANT INFORMATION\n\n"
        "Row:\n"
        "{context}\n\n"
        "Question: {question}\n"
        "Relevant information:"
    ),
)

REDUCE_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=(
        "Below are pieces of information extracted from the rows of a CSV "
        "file about Formula 1 drivers. Any line that says "
        "'NO RELEVANT INFORMATION' must be ignored.\n\n"
        "Question: {question}\n\n"
        "Extracted information:\n"
        "{summaries}\n\n"
        "Answer the question using ONLY the information above.\n"
        "Rules:\n"
        "1. If the question asks for the names of drivers matching a "
        "condition, list EVERY matching driver, one per line.\n"
        "2. Never omit a match and never invent a driver that is not present "
        "in the extracted information.\n"
        "3. If nothing matches, say that you don't know.\n\n"
        "Final answer:"
    ),
)


def build_qa_chain(file):
    """Load a CSV file, embed it and build a map_reduce RetrievalQA chain."""
    loader = CSVLoader(file_path=file, encoding='utf-8')
    docs = loader.load()

    embeddings = OpenAIEmbeddings()

    # create the vectorstore to use as the index
    db = Chroma.from_documents(docs, embeddings)

    # NOTE: the original code used search_kwargs={"k": 1000}, i.e. it fed the
    # *whole* CSV to the chain for every question.  With map_reduce the map
    # step runs one LLM call per retrieved document, so k=1000 meant ~1000
    # prompts (= a huge number of input tokens and calls) for a single query.
    # Retrieving a handful of relevant rows is both correct and far cheaper.
    retriever = db.as_retriever(search_type="similarity", search_kwargs={"k": 6})

    return RetrievalQA.from_chain_type(
        llm=OpenAI(temperature=0, batch_size=20, max_tokens=-1),
        chain_type="map_reduce",
        retriever=retriever,
        # the custom prompts must be handed to the map_reduce chain through
        # chain_type_kwargs, using the keys it actually accepts
        chain_type_kwargs={
            "question_prompt": MAP_PROMPT,
            "combine_prompt": REDUCE_PROMPT,
        },
        verbose=True,
    )


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:

            # build the vectorstore / chain once per file, not once per query
            qa_chain = build_qa_chain(file)

            for query in queries:
                print(query)
                # chain_type="map_reduce" + the prompts above returns the full
                # list of matching drivers instead of a truncated summary
                result = qa_chain.run(query)
                results.append(result)

                print(result)

            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German"
    ]
    res = run_agent(queries)
    print(res)
