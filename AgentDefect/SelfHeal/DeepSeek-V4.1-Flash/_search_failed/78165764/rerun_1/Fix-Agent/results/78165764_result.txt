"""RAG endpoint that returns full-sentence, conversational answers.

Why the original output looked fragmented/non-conversational:
- The prompt was a bare "answer the question based on the following context"
  instruction with no chat roles. ChatOpenAI (gpt-3.5-turbo) is a chat-tuned
  model: when it is fed a single, terse, document-QA-style user turn it happily
  replies with terse, extractive fragments ("Inception", "$200 million") that
  look like a search snippet rather than a chatbot answer. llama2-family models
  follow the same short prompt much more "loosely", so they happen to answer in
  flowing prose.
- The retriever was wired straight into the "context" slot, so the model
  received raw serialized `Document(page_content=..., metadata={...})` objects
  instead of clean text, which further pushes the model into snippet-extraction
  mode.

The fix keeps the rest of the setup identical and makes the chat model behave
conversationally by (a) giving it proper chat roles with a system persona and
explicit answer-shape instructions, and (b) handing it cleanly formatted text
as context.
"""

from dotenv import load_dotenv
import os
import pandas as pd
from fastapi import FastAPI  # noqa: F401  (import kept from the original setup)
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

# --- Conversational prompt -------------------------------------------------
# Same context/question variables, but expressed as chat messages so the
# model is steered into a chatbot voice instead of extractive QA fragments.
ANSWER_TEMPLATE = """You are a friendly, helpful assistant that talks to the user \
like a chatbot in a natural conversation.

Use the context below to answer the user's question.

Context:
{context}

Rules for your answer:
- Reply in complete, grammatical sentences written as flowing prose (or a short
  paragraph), never as a bare word, name, number, or list fragment.
- Speak directly to the user in a conversational tone: briefly state the answer,
  then explain the relevant details from the context that justify it.
- If the context contains several related items, mention them naturally in your
  sentences rather than dumping them raw.
- If the context does not contain the answer, say so honestly in a friendly
  sentence instead of guessing.

Question: {question}

Answer:"""

prompt = ChatPromptTemplate.from_messages(
    [
        (
            "system",
            "You are a conversational assistant. Always respond to the user in "
            "full, natural sentences as part of an ongoing chat, even when the "
            "underlying information comes from retrieved documents.",
        ),
        ("human", ANSWER_TEMPLATE),
    ]
)


def format_docs(docs):
    """Turn retrieved Documents into clean, plain text for the prompt.

    Passing the retriever directly would hand the model serialized
    `Document(page_content=..., metadata={...})` blobs, which nudges it toward
    snippet-style answers.
    """
    return "\n\n".join(doc.page_content.strip() for doc in docs)


def build_chain():
    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings(),
    )

    retriever = vectorstore.as_retriever(search_kwargs={"k": 4})

    chain = (
        {
            "context": retriever | format_docs,
            "question": RunnablePassthrough(),
        }
        | prompt
        | model
        | StrOutputParser()
    )
    return chain


app = FastAPI(
    title="LangChain Server",
    version="1.0",
    description="Spin up a simple api server using LangChain's Runnable interfaces",
)

_chain = None


def run_agent(query):
    global _chain
    if _chain is None:
        _chain = build_chain()

    result = _chain.invoke(query)
    return result


if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
