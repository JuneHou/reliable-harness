from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_debug, set_verbose
from langchain.callbacks.tracers import ConsoleCallbackHandler
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# `set_verbose(True)` only turns on the (deprecated) `verbose` flag of individual
# chains/LLMs; LCEL chains built with the `|` operator are plain Runnables and do
# not honour that flag, so nothing is printed. The current way to see every
# intermediate step of an LCEL chain is the global *debug* setting, which enables
# verbose logging of all Runnable steps (or an explicit ConsoleCallbackHandler).
set_verbose(True)
set_debug(True)


def run_agent(topic: str) -> str:
    model = ChatOpenAI(openai_api_key=api_key)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)
    chain = prompt | model | output_parser
    # Attaching the console callback handler explicitly guarantees step-by-step
    # tracing of the LCEL chain regardless of global settings.
    return chain.invoke(
        {"topic": topic},
        config={"callbacks": [ConsoleCallbackHandler()]},
    )


if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
