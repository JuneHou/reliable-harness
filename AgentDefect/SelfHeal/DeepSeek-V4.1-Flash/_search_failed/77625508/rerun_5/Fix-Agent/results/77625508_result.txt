"""Enable verbose logging of intermediate steps for a LangChain LCEL chain.

Key point: in current LangChain versions, `set_verbose(True)` / `verbose=True`
only toggle high-level tracing for legacy chains and do not print the
intermediate steps (prompt formatting, LLM request/response, parser output) of
LCEL / Runnable sequences. To actually see those steps you must either

  1) enable debug mode globally with `set_debug(True)`, or
  2) (recommended, scoped) attach a console callback handler to the run via
     `config={"callbacks": [...]}`.
"""

from dotenv import load_dotenv
import os

from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.tracers.stdout import ConsoleCallbackHandler
from langchain_openai import ChatOpenAI

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def run_agent(topic: str) -> str:
    model = ChatOpenAI(openai_api_key=api_key)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)
    chain = prompt | model | output_parser

    # Correct, current way to log every intermediate step of an LCEL chain:
    # pass a callback handler through the run `config` instead of relying on
    # `set_verbose(True)` / `verbose=True`.
    config = {
        "callbacks": [ConsoleCallbackHandler()],
        "run_name": "joke_chain",
    }
    return chain.invoke({"topic": topic}, config=config)


if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
