import os
import random
from typing import Dict, List, Optional

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]


def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


def build_retriever(db: Chroma,
                    metadata_filter: Optional[Dict] = None,
                    k: int = 4) -> VectorStoreRetriever:
    """Build a retriever whose search_kwargs carry the metadata filter.

    The filter MUST live in the retriever's `search_kwargs`; a `filter`
    entry passed to the chain itself is silently dropped by
    ConversationalRetrievalChain, which is why the unfiltered documents
    ended up being used for the answer.
    """
    search_kwargs: Dict = {"k": k}
    if metadata_filter:
        search_kwargs["filter"] = metadata_filter
    return VectorStoreRetriever(vectorstore=db, search_kwargs=search_kwargs)


def build_chain(db: Chroma) -> ConversationalRetrievalChain:
    embeddings_key = get_random_key(OPENAI_API_KEY_POOL)
    llm_key = get_random_key(OPENAI_API_KEY_POOL)
    llm = OpenAI(openai_api_key=llm_key, cache=True, temperature=0)
    retriever = build_retriever(db)
    return ConversationalRetrievalChain.from_llm(
        llm,
        retriever,
        verbose=True,
        return_source_documents=True,
    )


def run_agent(query: str,
              pdf_paths: List[str],
              metadata_filter: Optional[Dict] = None) -> Dict:
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    all_texts = []
    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        source_name = os.path.basename(pdf_path)
        for text in texts:
            # Store both the basename and the full path so the filter used below
            # ('paper2.pdf') actually matches the stored metadata.
            text.metadata["source"] = source_name
            text.metadata["source_path"] = pdf_path

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    qa = build_chain(db)

    # Apply the metadata filter DYNAMICALLY at query time by mutating the
    # retriever's search_kwargs - no need for a separate chain per document
    # and no need to patch LangChain itself.
    if metadata_filter is not None:
        qa.retriever.search_kwargs["filter"] = metadata_filter
    else:
        qa.retriever.search_kwargs.pop("filter", None)

    # Do NOT pass "filter" here: ConversationalRetrievalChain only forwards
    # inputs that its input keys define, so the extra key would be ignored.
    result = qa({"question": query, "chat_history": []})

    sources = [doc.metadata.get("source") for doc in result.get("source_documents", [])]
    result["sources"] = sorted(set(s for s in sources if s))
    return result

def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


class FilterableVectorStoreRetriever(VectorStoreRetriever):
    """VectorStoreRetriever whose metadata filter can be changed at query time.

    Why this is needed:
    ``ConversationalRetrievalChain`` only feeds ``question`` (and
    ``chat_history``) into the retriever.  Anything else you put in the input
    dict - e.g. ``qa({"question": ..., "chat_history": [], "filter": {...}})`` -
    is silently dropped, so the ``filter`` never reaches Chroma and the answer
    keeps coming from every document.  The filter therefore has to travel
    through the retriever itself; keeping it as a mutable attribute means a
    single chain (and a single vector store) can serve *all* documents while
    the filter is swapped for each call.  No LangChain modification required.
    """

    metadata_filter: Optional[Dict[str, Any]] = None

    def _get_relevant_documents(self, query, *, run_manager=None):
        search_kwargs = dict(self.search_kwargs or {})
        if self.metadata_filter:
            # Chroma expects the metadata filter under the ``filter`` keyword.
            search_kwargs["filter"] = self.metadata_filter
        return self.vectorstore.search(
            query,
            self.search_type,
            **search_kwargs,
        )


def build_pipeline(pdf_paths):
    """Build the vector store + retriever + chain exactly once."""
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            # metadata is what the dynamic filter will match against
            text.metadata["source"] = pdf_path

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    retriever = FilterableVectorStoreRetriever(
        vectorstore=db,
        search_type="similarity",
        search_kwargs={"k": 4},
        metadata_filter=None,
    )

    qa = ConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever,
        verbose=True,
        return_source_documents=True,
    )

    return qa, retriever


def run_agent(query, pdf_paths, metadata_filter=None, chat_history=None):
    """Ask a question, optionally restricted to one document at query time."""
    qa, retriever = build_pipeline(pdf_paths)

    # Dynamic, per-question filter: set it right before invoking the chain.
    # {'source': 'paper2.pdf'} means "only answer from paper2.pdf".
    retriever.metadata_filter = metadata_filter

    result = qa({"question": query, "chat_history": chat_history or []})

    print(result['answer'])

    sources = [doc.metadata.get("source") for doc in result.get("source_documents", [])]
    print(f"\nSources used: {sources}")
    if metadata_filter:
        expected = metadata_filter.get("source")
        if expected and sources and any(src != expected for src in sources):
            print(f"WARNING: retriever returned chunks outside of {expected!r}")
    else:
        print("No filter applied: answer may use every document.")

    return result
