import uuid
from typing import Any, Dict, List, Optional

# Callback plumbing (mirrors llama_index.core.callbacks)
# ---------------------------------------------------------------------------
class CBEventType:
    LLM = "llm"


class EventPayload:
    MESSAGES = "messages"
    RESPONSE = "response"
    COMPLETION = "completion"


class BaseCallbackHandler:
    def start_trace(self, trace_id: Optional[str] = None) -> None:
        return None

    def end_trace(self, trace_id: Optional[str] = None, trace_map=None) -> None:
        return None

    def on_event_start(self, event_type, payload=None, event_id="", parent_id="", **kwargs) -> None:
        return None

    def on_event_end(self, event_type, payload=None, event_id="", parent_id="", **kwargs) -> None:
        return None


class OpenAIFineTuningHandler(BaseCallbackHandler):
    """Records LLM inputs/outputs and dumps them as OpenAI finetuning JSONL."""

    def __init__(self) -> None:
        # One entry per completed LLM call:
        #   {"messages": [...], "response": "..."}
        self.events: List[Dict[str, Any]] = []
        # Payloads of in-flight LLM calls, keyed by event id. LlamaIndex
        # sends the input messages on event *start* and the response on
        # event *end*, so the two have to be paired up.
        self._start_events: Dict[str, Dict[str, Any]] = {}

    # -- tracing ----------------------------------------------------------
    def start_trace(self, trace_id: Optional[str] = None) -> None:
        return None

    def end_trace(self, trace_id: Optional[str] = None, trace_map=None) -> None:
        return None

    # -- event hooks ------------------------------------------------------
    def on_event_start(self, event_type, payload=None, event_id="", parent_id="", **kwargs) -> None:
        if event_type == CBEventType.LLM:
            self._start_events[event_id] = dict(payload or {})

    def on_event_end(self, event_type, payload=None, event_id="", parent_id="", **kwargs) -> None:
        if event_type != CBEventType.LLM:
            return

        payload = payload or {}
        start_payload = self._start_events.pop(event_id, {})

        messages = payload.get(EventPayload.MESSAGES) or start_payload.get(EventPayload.MESSAGES) or []
        response = payload.get(EventPayload.RESPONSE)
        if response is None:
            response = payload.get(EventPayload.COMPLETION)

        response_text = self._as_text(response)
        if not messages or not response_text:
            # Never store an empty (unusable) finetuning example.
            return

        self.events.append(
            {
                EventPayload.MESSAGES: [self._as_message_dict(m) for m in messages],
                EventPayload.RESPONSE: response_text,
            }
        )

    # -- conversion helpers ----------------------------------------------
    @staticmethod
    def _as_message_dict(message: Any) -> Dict[str, str]:
        if isinstance(message, dict):
            role = message.get("role", "user")
            content = message.get("content", "")
        else:
            role = getattr(message, "role", "user")
            content = getattr(message, "content", message)
        # MessageRole enum -> plain string
        role = getattr(role, "value", role)
        return {
            "role": str(role).lower(),
            "content": "" if content is None else str(content),
        }

    @staticmethod
    def _as_text(response: Any) -> str:
        if response is None:
            return ""
        if isinstance(response, str):
            return response
        message = getattr(response, "message", None)
        if message is not None:
            return OpenAIFineTuningHandler._as_message_dict(message)["content"]
        text = getattr(response, "text", None)
        if text is not None:
            return str(text)
        return str(response)

    # -- output -----------------------------------------------------------
    def get_finetuning_events(self) -> List[Dict[str, Any]]:
        """Return the captured events in OpenAI chat finetuning format."""
        examples: List[Dict[str, Any]] = []
        for event in self.events:
            messages = list(event[EventPayload.MESSAGES])
            messages.append({"role": "assistant", "content": event[EventPayload.RESPONSE]})
            examples.append({"messages": messages})
        return examples

    def save_finetuning_events(self, path: str) -> int:
        """Write one JSON object per line and report the real example count."""
        examples = self.get_finetuning_events()

        directory = os.path.dirname(os.path.abspath(path))
        if directory:
            os.makedirs(directory, exist_ok=True)

        with open(path, "w", encoding="utf-8") as f:
            for example in examples:
                f.write(json.dumps(example, ensure_ascii=False) + "\n")

        print(f"Wrote {len(examples)} examples to {path}")
        return len(examples)


class CallbackManager:

    def add_handler(self, handler: BaseCallbackHandler) -> None:
        self.handlers.append(handler)

    def start_trace(self, trace_id: Optional[str] = None) -> None:
        for handler in self.handlers:

    def end_trace(self, trace_id: Optional[str] = None, trace_map=None) -> None:
        for handler in self.handlers:
            handler.end_trace(trace_id, trace_map)

    def on_event_start(self, event_type, payload=None, event_id="", parent_id="") -> str:

    @property

    def __str__(self) -> str:

class MockLLM:
    """Stand-in for an OpenAI chat model that reports LLM events."""

    def __init__(self, callback_manager: Optional[CallbackManager] = None) -> None:
        self.callback_manager = callback_manager or CallbackManager([])
        self.model = "mock-llm"

    def chat(self, messages: List[ChatMessage]) -> ChatResponse:
        payload = {EventPayload.MESSAGES: messages}
        event_id = self.callback_manager.on_event_start(CBEventType.LLM, payload)

        question = messages[-1].content if messages else ""
        response = ChatResponse(ChatMessage("assistant", f"Answer to: {question}"))

        self.callback_manager.on_event_end(
            CBEventType.LLM, {EventPayload.RESPONSE: response}, event_id=event_id
        )
        return response

class TextNode:

class QueryEngine:
    def __init__(self, index, similarity_top_k: int = 2, llm=None, callback_manager=None) -> None:
        self.index = index
        self.similarity_top_k = similarity_top_k
        self.llm = llm or Settings.llm or MockLLM()
        self.callback_manager = callback_manager or Settings.callback_manager
        # The LLM must report to the very same callback manager, otherwise
        # the finetuning handler sees no LLM events at all.
        self.llm.callback_manager = self.callback_manager

    def query(self, question: str) -> ChatResponse:
        nodes = self.index.retrieve(question, similarity_top_k=self.similarity_top_k)
        context = "\n".join(node.text for node in nodes)

        messages = [
            ChatMessage("system", "Answer the question using only the provided context."),
            ChatMessage("user", f"Context:\n{context}\n\nQuestion: {question}"),
        ]
        return self.llm.chat(messages)


class VectorStoreIndex:
    def __init__(self, documents: Optional[List[str]] = None) -> None:
        self.documents = list(documents or [])
        self.callback_manager: CallbackManager = Settings.callback_manager

    @staticmethod
    def from_documents(docs, callback_manager=None) -> "VectorStoreIndex":
        index = VectorStoreIndex([d if isinstance(d, str) else getattr(d, "text", str(d)) for d in (docs or [])])
        index.callback_manager = callback_manager or Settings.callback_manager
        return index

    def retrieve(self, question: str, similarity_top_k: int = 2) -> List[TextNode]:
        return [TextNode(doc) for doc in self.documents[:similarity_top_k]]

    def as_query_engine(self, similarity_top_k: int = 2, llm=None, callback_manager=None) -> QueryEngine:
        return QueryEngine(
            index=self,
            similarity_top_k=similarity_top_k,
            llm=llm,
            callback_manager=callback_manager or self.callback_manager,
        )

class FineTuningExample:

        questions: List[str] = []
        with open(questions_path, "r", encoding="utf-8") as f:
            for line in f:
                question = line.strip()
                if question:
                    questions.append(question)
        return questions

    def jsonl_generation(self) -> int:
        """Generate the JSONL file of fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        # 1) Build the handler + callback manager ...
        self.finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([self.finetuning_handler])

        # 2) ... and register it *globally* and on the LLM. This is the actual
        #    fix: previously the manager was only attached to an unused LLM
        #    object, so no LLM event ever reached the handler.
        Settings.callback_manager = callback_manager
        self.llm = MockLLM(callback_manager=callback_manager)
        Settings.llm = self.llm  # setter also propagates the callback manager

        questions = self._read_questions()

        try:
            index = VectorStoreIndex.from_documents(
                self.documents, callback_manager=callback_manager
            )
            query_engine = index.as_query_engine(
                similarity_top_k=2, llm=self.llm, callback_manager=callback_manager
            )
            for question in questions:
                query_engine.query(question)
        except Exception as exc:  # generation errors should not hide the events
            print(f"An error occurred: {exc}")
        finally:
            out_path = os.path.join(self.dataset_path, "finetuning_events.jsonl")
            num_examples = self.finetuning_handler.save_finetuning_events(out_path)
            if num_examples == 0:
                print(
                    "No LLM events were captured. Make sure the finetuning handler's "
                    "callback manager is attached to the LLM/query engine that runs "
                    "the queries (e.g. Settings.callback_manager = callback_manager)."
                )

        return num_examples

    def __repr__(self) -> str:  # pragma: no cover - debugging helper
        return f"ChatMessage(role={self.role!r}, content={self.content!r})"


class ChatResponse:
    def __init__(self, message: ChatMessage) -> None:
        self.message = message

    @property
    def text(self) -> str:
        return self.message.content

    def __str__(self) -> str:
        return self.message.content


class MockLLM:
    """Stand-in for an OpenAI chat model that reports LLM events."""

    def __init__(self, callback_manager: Optional[CallbackManager] = None) -> None:
        self.callback_manager = callback_manager or CallbackManager([])
        self.model = "mock-llm"

    def chat(self, messages: List[ChatMessage]) -> ChatResponse:
        payload = {EventPayload.MESSAGES: messages}
        event_id = self.callback_manager.on_event_start(CBEventType.LLM, payload)

        question = messages[-1].content if messages else ""
        response = ChatResponse(ChatMessage("assistant", f"Answer to: {question}"))

        self.callback_manager.on_event_end(
            CBEventType.LLM, {EventPayload.RESPONSE: response}, event_id=event_id
        )
        return response


# ---------------------------------------------------------------------------
# Index / query engine
# ---------------------------------------------------------------------------
class TextNode:
    def __init__(self, text: str) -> None:
        self.text = text


class QueryEngine:
    def __init__(self, index, similarity_top_k: int = 2, llm=None, callback_manager=None) -> None:
        self.index = index
        self.similarity_top_k = similarity_top_k
        self.llm = llm or Settings.llm or MockLLM()
        self.callback_manager = callback_manager or Settings.callback_manager
        # The LLM must report to the very same callback manager, otherwise
        # the finetuning handler sees no LLM events at all.
        self.llm.callback_manager = self.callback_manager

    def query(self, question: str) -> ChatResponse:
        nodes = self.index.retrieve(question, similarity_top_k=self.similarity_top_k)
        context = "\n".join(node.text for node in nodes)

        messages = [
            ChatMessage("system", "Answer the question using only the provided context."),
            ChatMessage("user", f"Context:\n{context}\n\nQuestion: {question}"),
        ]
        return self.llm.chat(messages)


class VectorStoreIndex:
    def __init__(self, documents: Optional[List[str]] = None) -> None:
        self.documents = list(documents or [])
        self.callback_manager: CallbackManager = Settings.callback_manager

    @staticmethod
    def from_documents(docs, callback_manager=None) -> "VectorStoreIndex":
        index = VectorStoreIndex([d if isinstance(d, str) else getattr(d, "text", str(d)) for d in (docs or [])])
        index.callback_manager = callback_manager or Settings.callback_manager
        return index

    def retrieve(self, question: str, similarity_top_k: int = 2) -> List[TextNode]:
        return [TextNode(doc) for doc in self.documents[:similarity_top_k]]

    def as_query_engine(self, similarity_top_k: int = 2, llm=None, callback_manager=None) -> QueryEngine:
        return QueryEngine(
            index=self,
            similarity_top_k=similarity_top_k,
            llm=llm,
            callback_manager=callback_manager or self.callback_manager,
        )

# Fine-tuning data generation
# ---------------------------------------------------------------------------
class FineTuningExample:
    def __init__(self, dataset_path: str, documents: Optional[List[str]] = None) -> None:
        self.dataset_path = dataset_path
        self.documents = list(documents or [])
        self.llm = None
        self.finetuning_handler: Optional[OpenAIFineTuningHandler] = None

    def _read_questions(self) -> List[str]:
        questions_path = os.path.join(self.dataset_path, "train_questions.txt")
        questions: List[str] = []
        with open(questions_path, "r", encoding="utf-8") as f:
            for line in f:
                question = line.strip()
                if question:
                    questions.append(question)
        return questions

    def jsonl_generation(self) -> int:
        """Generate the JSONL file of fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        # 1) Build the handler + callback manager ...
        self.finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([self.finetuning_handler])

        # 2) ... and register it *globally* and on the LLM. This is the actual
        #    fix: previously the manager was only attached to an unused LLM
        #    object, so no LLM event ever reached the handler.
        Settings.callback_manager = callback_manager
        self.llm = MockLLM(callback_manager=callback_manager)
        Settings.llm = self.llm  # setter also propagates the callback manager

        questions = self._read_questions()

        try:
            index = VectorStoreIndex.from_documents(
                self.documents, callback_manager=callback_manager
            )
            query_engine = index.as_query_engine(
                similarity_top_k=2, llm=self.llm, callback_manager=callback_manager
            )
            for question in questions:
                query_engine.query(question)
        except Exception as exc:  # generation errors should not hide the events
            print(f"An error occurred: {exc}")
        finally:
            out_path = os.path.join(self.dataset_path, "finetuning_events.jsonl")
            num_examples = self.finetuning_handler.save_finetuning_events(out_path)
            if num_examples == 0:
                print(
                    "No LLM events were captured. Make sure the finetuning handler's "
                    "callback manager is attached to the LLM/query engine that runs "
                    "the queries (e.g. Settings.callback_manager = callback_manager)."
                )

        return num_examples


def prepare_test_data(questions_path: str) -> None:
    directory = os.path.dirname(os.path.abspath(questions_path))
    os.makedirs(directory, exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path: str) -> int:
    dataset_path = os.path.dirname(os.path.abspath(questions_path))
    ft = FineTuningExample(
        dataset_path=dataset_path,
        documents=[
            "LlamaIndex is a data framework for LLM applications.",
            "Fine-tuning adapts a base model to a specific task.",
        ],
    )
    return ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
