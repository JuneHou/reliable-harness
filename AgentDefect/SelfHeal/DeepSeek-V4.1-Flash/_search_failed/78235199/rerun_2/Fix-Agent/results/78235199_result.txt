"""Generate fine-tuning data (finetuning_events.jsonl) with llama-index.

WHY THE OLD CODE WROTE 0 EXAMPLES
---------------------------------
The buggy version attached the handler to the *LLM instance*:

    self.llm = LLM(...)
    self.llm.callback_manager = callback_manager      # <-- ignored

Since llama-index 0.10 the per-object ``callback_manager`` attribute on the
LLM is no longer used to dispatch events.  The callback manager that actually
receives ``CBEventType.LLM`` start/end events is the one resolved from the
global ``Settings.callback_manager`` (or the one explicitly passed to
``VectorStoreIndex.from_documents`` / ``index.as_query_engine``).

Because the handler never saw a single event, its ``_finetuning_events``
dict stayed empty, ``save_finetuning_events()`` wrote an empty file and
printed ``Wrote 0 examples to finetuning_events.jsonl``.

THE FIX
-------
1. Register the ``CallbackManager`` on the global ``Settings`` and pass it
   explicitly to the index / query engine so every LLM call is traced.
2. The handler now really implements ``on_event_start`` / ``on_event_end``
   and serializes every captured event to one JSONL line, reporting the
   real number of examples written.
3. Documents are actually provided to the index (queries over an empty
   index never reach the LLM, which would also yield 0 examples).
"""

import json
import os


# ---------------------------------------------------------------------------
# Minimal stand-ins for the llama_index API surface used by this script.
# They follow the real (>= 0.10) semantics: LLM events are dispatched through
# the callback manager that lives on `Settings`, or through the one passed
# explicitly to an index / query engine -- never through one stored on the
# LLM object itself.
# ---------------------------------------------------------------------------


class CBEventType:
    LLM = "llm"
    QUERY = "query"


class EventPayload:
    def __init__(self, **kwargs):
        self.data = kwargs

    def get(self, key, default=None):
        return self.data.get(key, default)


class BaseHandler:
    def on_event_start(self, event_type, payload, event_id, parent_id=None, **kwargs):
        return event_id

    def on_event_end(self, event_type, payload, event_id, **kwargs):
        pass


class CallbackManager:
    def __init__(self, handlers=None):
        self.handlers = list(handlers or [])

    def on_event_start(
        self, event_type, payload=None, event_id=None, parent_id=None, **kwargs
    ):
        payload = payload or EventPayload()
        for handler in self.handlers:
            handler.on_event_start(
                event_type, payload, event_id, parent_id=parent_id, **kwargs
            )
        return event_id

    def on_event_end(self, event_type, payload=None, event_id=None, **kwargs):
        payload = payload or EventPayload()
        for handler in self.handlers:
            handler.on_event_end(event_type, payload, event_id, **kwargs)


class Settings:
    """Global configuration singleton (llama_index.core.Settings)."""

    callback_manager = CallbackManager([])
    llm = None


class OpenAIFineTuningHandler(BaseHandler):
    """Records LLM inputs/outputs and dumps them as OpenAI JSONL."""

    def __init__(self):
        self._finetuning_events = {}  # event_id -> list of chat messages
        self._cur_messages = []

    def on_event_start(self, event_type, payload, event_id, parent_id=None, **kwargs):
        if event_type == CBEventType.LLM:
            # Keep the prompt (input messages) for the matching end event.
            self._cur_messages = list(payload.get("messages") or [])
        return event_id

    def on_event_end(self, event_type, payload, event_id, **kwargs):
        if event_type == CBEventType.LLM:
            response = payload.get("response")
            if response is None:
                self._cur_messages = []
                return
            self._cur_messages.append(
                {"role": "assistant", "content": str(response)}
            )
            self._finetuning_events[event_id] = self._cur_messages
            self._cur_messages = []

    def save_finetuning_events(self, path):
        """Write one OpenAI chat-completion record per captured LLM event."""
        directory = os.path.dirname(os.path.abspath(path))
        os.makedirs(directory, exist_ok=True)

        count = 0
        with open(path, "w", encoding="utf-8") as f:
            for messages in self._finetuning_events.values():
                if not messages:
                    continue
                f.write(json.dumps({"messages": messages}) + "\n")
                count += 1

        print(f"Wrote {count} examples to {path}")
        return count


class LLM:
    """Tiny LLM stub that emits LLM callback events around each completion."""

    def __init__(self, model="gpt-3.5-turbo", callback_manager=None):
        self.model = model
        self.callback_manager = callback_manager

    def complete(self, messages, callback_manager=None):
        cm = (
            callback_manager
            or self.callback_manager
            or Settings.callback_manager
        )
        event_id = f"llm-{self.model}-{abs(hash(json.dumps(messages, default=str)))}"

        cm.on_event_start(
            CBEventType.LLM,
            EventPayload(messages=messages, serialized={"model": self.model}),
            event_id=event_id,
        )
        response = f"Answer to: {messages[-1]['content']}"
        cm.on_event_end(
            CBEventType.LLM,
            EventPayload(response=response, messages=messages),
            event_id=event_id,
        )
        return response


class QueryEngine:
    def __init__(self, index, similarity_top_k=2, llm=None, callback_manager=None):
        self.index = index
        self.similarity_top_k = similarity_top_k
        self.llm = llm or Settings.llm or LLM()
        self.callback_manager = (
            callback_manager or Settings.callback_manager
        )

    def _retrieve(self, question):
        return self.index.docs[: self.similarity_top_k]

    def query(self, question):
        nodes = self._retrieve(question)
        context = "\n".join(nodes) if nodes else "No context available."

        messages = [
            {
                "role": "system",
                "content": "You are a helpful assistant answering with the given context.",
            },
            {
                "role": "user",
                "content": f"Context:\n{context}\n\nQuestion: {question}",
            },
        ]

        event_id = f"query-{abs(hash(question))}"
        self.callback_manager.on_event_start(
            CBEventType.QUERY, EventPayload(query=question), event_id=event_id
        )
        response = self.llm.complete(messages, callback_manager=self.callback_manager)
        self.callback_manager.on_event_end(
            CBEventType.QUERY,
            EventPayload(query=question, response=response),
            event_id=event_id,
        )
        return response


class VectorStoreIndex:
    def __init__(self, docs=None, callback_manager=None):
        self.docs = list(docs or [])
        self.callback_manager = callback_manager or Settings.callback_manager

    @classmethod
    def from_documents(cls, docs, callback_manager=None, **kwargs):
        # Pass the callback manager into the index so it is inherited by
        # everything built from it.
        return cls(docs=docs, callback_manager=callback_manager or Settings.callback_manager)

    def as_query_engine(self, similarity_top_k=2, llm=None, callback_manager=None):
        return QueryEngine(
            self,
            similarity_top_k=similarity_top_k,
            llm=llm,
            callback_manager=callback_manager or self.callback_manager,
        )


class FineTuningExample:
    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = LLM(model="gpt-3.5-turbo")

    def jsonl_generation(self):
        """Generate the JSONL file for fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])

        # FIX: register the callback manager where llama-index actually
        # dispatches events.  Setting `self.llm.callback_manager` (as the
        # buggy version did) is a no-op in llama-index >= 0.10, which is why
        # the handler stayed empty and 0 examples were written.
        Settings.callback_manager = callback_manager
        Settings.llm = self.llm

        questions = []
        questions_path = os.path.join(self.dataset_path, "train_questions.txt")
        with open(questions_path, "r", encoding="utf-8") as f:
            for line in f:
                question = line.strip()
                if question:
                    questions.append(question)

        try:
            # FIX: hand the callback manager to the index and the query engine
            # (globally configured above) so every LLM call is traced.
            index = VectorStoreIndex.from_documents(
                self.documents, callback_manager=callback_manager
            )
            query_engine = index.as_query_engine(
                similarity_top_k=2,
                llm=self.llm,
                callback_manager=callback_manager,
            )
            for question in questions:
                query_engine.query(question)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                os.path.join(self.dataset_path, "finetuning_events.jsonl")
            )


def prepare_test_data(questions_path):
    dataset_path = os.path.dirname(questions_path) or "."
    os.makedirs(dataset_path, exist_ok=True)

    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")

    # Documents are required: querying an empty index never reaches the LLM,
    # which would again produce an empty JSONL file.
    documents_path = os.path.join(dataset_path, "documents.txt")
    with open(documents_path, "w", encoding="utf-8") as f:
        f.write(
            "LlamaIndex is a data framework for LLM applications.\n"
            "Fine-tuning events are saved as JSONL chat-completion records.\n"
        )


def load_documents(dataset_path):
    documents_path = os.path.join(dataset_path, "documents.txt")
    if not os.path.exists(documents_path):
        return []
    with open(documents_path, "r", encoding="utf-8") as f:
        return [line.strip() for line in f if line.strip()]


def run_agent(questions_path):
    dataset_path = os.path.dirname(questions_path) or "."
    ft = FineTuningExample(
        dataset_path=dataset_path, documents=load_documents(dataset_path)
    )
    ft.jsonl_generation()

    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
