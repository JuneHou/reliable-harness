import os

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import FAISS
from langchain.chains import ConversationalRetrievalChain
from langchain.memory import ConversationBufferMemory
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate

# Load environment variables from .env file
_ = load_dotenv(find_dotenv())

KNOWLEDGE_BASE = [
    "Buildings are made out of brick",
    "Buildings are made out of wood",
    "Buildings are made out of stone",
    "Buildings are made out of atoms",
    "Buildings are made out of building materials",
    "Cars are made out of metal",
    "Cars are made out of plastic",
]

CUSTOM_TEMPLATE = """
This is conversation with a human. Answer the questions you get based on the knowledge you have.
If you don't know the answer, just say that you don't, don't try to make up an answer.
Chat History:
{chat_history}
Follow Up Input: {question}
"""

# A chunk must reach this relevance score (0..1, cosine-like) to count as
# "answered from the custom knowledge base". Anything below it (greetings,
# small talk, off-topic questions) is dropped -> empty list.
SCORE_THRESHOLD = 0.4
TOP_K = 4


def build_qa():
    """Build the vector store, threshold retriever and conversational chain."""
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise RuntimeError("OPENAI_API_KEY is not set (check your .env file)")
    os.environ["OPENAI_API_KEY"] = openai_api_key

    llm = ChatOpenAI(
        model_name="gpt-3.5-turbo",  # Name of the language model
        temperature=0,               # Parameter that controls the randomness of the generated responses
    )
    embeddings = OpenAIEmbeddings()

    # normalize_L2=True stores unit vectors, so FAISS' euclidean distance can be
    # mapped to a cosine-like relevance score and SCORE_THRESHOLD is meaningful
    # regardless of the embedding magnitudes.
    vectorstore = FAISS.from_texts(KNOWLEDGE_BASE, embeddings, normalize_L2=True)

    # The important part: `score_threshold` is IGNORED by `as_retriever()` unless
    # search_type="similarity_score_threshold" is given explicitly. With the
    # default search_type="similarity" the retriever always returns TOP_K
    # documents, which is why greetings/off-topic questions were answered from
    # the vector store instead of returning an empty list.
    retriever = vectorstore.as_retriever(
        search_type="similarity_score_threshold",
        search_kwargs={"score_threshold": SCORE_THRESHOLD, "k": TOP_K},
    )

    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)

    qa = ConversationalRetrievalChain.from_llm(
        llm,
        retriever,
        condense_question_prompt=PromptTemplate.from_template(CUSTOM_TEMPLATE),
        memory=memory,
        return_source_documents=True,  # needed to know whether the KB answered
    )
    return qa, retriever


def run_agent(user_input):
    """Return the relevant documents (empty list when the KB cannot answer)."""
    qa, retriever = build_qa()

    # Query the chain with the actual user question (the original code answered a
    # hard-coded "what are cars made of?" and then overwrote the answer with raw
    # retriever output, which is why the result never matched the question).
    result = qa({"question": user_input})
    answer = (result.get("answer") or "").strip()
    source_docs = result.get("source_documents") or []

    # Nothing passed the similarity threshold -> greeting / general / off-topic
    # input, so no documents are relevant.
    if not source_docs:
        return [], answer

    # Extra guard: the LLM may refuse even with documents in context.
    refusal_markers = ("i don't know", "i do not know", "don't know the answer",
                       "not sure", "cannot answer", "can't answer")
    if any(marker in answer.lower() for marker in refusal_markers):
        return [], answer

    # Equivalent direct call on the retriever (also returns [] for off-topic).
    relevant_docs = retriever.get_relevant_documents(user_input)
    if not relevant_docs:
        return [], answer

    return relevant_docs, answer


if __name__ == "__main__":
    for user_input in (
        "What are buildings made of?",
        "hello",
        "what is the capital of France?",
    ):
        docs, answer = run_agent(user_input)
        print("\n=== User Query Result ===")
        print(f"Question: {user_input}")
        print(f"Answer: {answer}")
        print(f"Documents: {docs}")
