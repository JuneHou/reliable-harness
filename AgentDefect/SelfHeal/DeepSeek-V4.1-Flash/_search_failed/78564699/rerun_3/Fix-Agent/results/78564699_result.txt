import json
import os
import time

from llama_index.llms.llama_cpp import LlamaCPP

# ---------------------------------------------------------------------------
# Why the original snippet "hangs" during inference
# ---------------------------------------------------------------------------
# With `n_gpu_layers=0` every token must be produced on the CPU, and
# `llama-cpp-python` then spawns one worker per *logical* CPU because
# `n_threads` defaults to multiprocessing.cpu_count(). Combined with:
#   * a 13B Q4_0 model (~7.5 GB of weights) plus a 3900-token KV-cache, which
#     does not fit in the RAM of a typical CPU-only box and pushes the process
#     into swap (thrashing => no visible progress), and
#   * 256 requested new tokens,
# `llm.complete()` never reaches the point of emitting/returning an answer, so
# the script looks frozen forever right after model initialisation.
#
# Fix: use a GGUF model/quantisation actually sized for CPU inference, keep the
# context window small, cap the number of new tokens, and pin `n_threads` to
# the number of *physical* cores (oversubscribing threads on ggml's spin
# barrier is a known way to stall generation). The Llama-2 chat template is
# supplied explicitly so the model receives the format it was fine-tuned on.
# ---------------------------------------------------------------------------

MODEL_URL = (
    "https://huggingface.co/TheBloke/Llama-2-7B-chat-GGUF/resolve/main/"
    "llama-2-7b-chat.Q4_K_M.gguf"
)
MODEL_KEY = "llama-2-7b-chat"

PATHS_FILE = "./paths.json"
CONTEXT_WINDOW = 2048        # was 3900 -> much smaller KV-cache, fits in RAM
MAX_NEW_TOKENS = 128         # was 256 -> short, CPU-friendly answers

DEFAULT_SYSTEM_PROMPT = (
    "You are a helpful, respectful and honest assistant. Always answer as "
    "helpfully as possible, while being safe."
)


def _physical_cores() -> int:
    """Number of physical cores (fallback: half of the logical CPUs).

    Feeding ggml more threads than physical cores is the classic cause of
    generation appearing to stall, so never use cpu_count() directly here.
    """
    cores = None
    try:
        import psutil  # optional dependency

        cores = psutil.cpu_count(logical=False)
    except Exception:
        cores = None
    if not cores:
        cores = max(1, (os.cpu_count() or 2) // 2)
    return max(1, int(cores))


def build_llama2_prompt(messages) -> str:
    """Render messages with the Llama-2 chat template."""
    system_prompt = DEFAULT_SYSTEM_PROMPT
    for message in messages:
        if getattr(message, "role", None) == "system" and message.content:
            system_prompt = message.content

    prompt = f"<s>[INST] <<SYS>>\n{system_prompt}\n<</SYS>>\n\n"
    for message in messages:
        role = getattr(message, "role", "user")
        content = getattr(message, "content", "")
        if role in ("assistant", "ai", "model"):
            prompt += f"{content} </s><s>[INST] "
        elif role != "system":
            prompt += f"{content} [/INST] "
    return prompt


def wrap_completion(completion: str) -> str:
    return (
        f"<s>[INST] <<SYS>>\n{DEFAULT_SYSTEM_PROMPT}\n<</SYS>>\n\n"
        f"{completion} [/INST] "
    )


def resolve_model() -> dict:
    """Use the local GGUF from paths.json when available, else the URL."""
    if os.path.exists(PATHS_FILE):
        try:
            with open(PATHS_FILE, "r") as f:
                paths = json.load(f)
        except (json.JSONDecodeError, OSError):
            paths = {}

        local_path = paths.get(MODEL_KEY)
        if local_path and os.path.exists(local_path):
            return {"model_path": local_path, "model_url": None}

    # Let llama-cpp-python download / cache the model itself.
    return {"model_path": None, "model_url": MODEL_URL}


def build_llm() -> LlamaCPP:
    model_args = resolve_model()

    return LlamaCPP(
        model_url=model_args["model_url"],
        model_path=model_args["model_path"],
        temperature=0.1,
        max_new_tokens=MAX_NEW_TOKENS,
        context_window=CONTEXT_WINDOW,
        model_kwargs={
            "n_gpu_layers": 0,          # CPU only
            "n_threads": _physical_cores(),
            "n_threads_batch": _physical_cores(),
            "n_batch": 256,
            "use_mlock": False,         # don't pin RAM, avoid OOM/swap stalls
            "seed": 0,
        },
        messages_to_prompt=build_llama2_prompt,
        completion_to_prompt=wrap_completion,
        verbose=True,
    )


def run_agent(query, llm=None):
    if llm is None:
        llm = build_llm()

    started = time.time()
    response = llm.complete(query)
    elapsed = time.time() - started

    print(f"[info] generated in {elapsed:.1f}s", flush=True)
    return str(response)


if __name__ == "__main__":
    query = "What is the capital of France?"
    # Build the model once and reuse it for subsequent queries.
    llm = build_llm()
    response = run_agent(query, llm=llm)
    print(response)
