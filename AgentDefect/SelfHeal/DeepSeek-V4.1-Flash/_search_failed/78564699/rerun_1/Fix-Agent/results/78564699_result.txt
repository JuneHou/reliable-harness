"""
Fixed version.

Why the original program appeared to "hang forever after initialisation"
-----------------------------------------------------------------------
`llama-cpp-python` (the engine behind `LlamaCPP`) defaults to
``n_threads = multiprocessing.cpu_count() // 2`` and to ``n_batch = 512``.
On a CPU-only machine that means only half of the available threads are
used (which is exactly the "CPU usage stuck at ~50 %" symptom), and a
13B Q4_0 Llama-2-chat model then generates at well under one token per
second.  With ``max_new_tokens=256`` a single `llm.complete()` call can
therefore take tens of minutes / forever, so nothing is ever printed and
the process looks frozen: it is *not* crashing, it is just running with a
handicapped thread pool (plus a KV cache for a 3.9k context on ~9.5 GB of
RAM, which pushes the machine into swapping).

Fixes applied
-------------
1. Explicitly pass ``n_threads`` / ``n_threads_batch`` = number of
   *physical* cores into ``model_kwargs`` so the whole CPU is used.
2. Keep everything on CPU (``n_gpu_layers=0``) but also set ``n_batch``,
   and ``use_mlock=True`` so the weights are not swapped out mid-run.
3. Stream tokens (`llm.stream_complete`) so progress is visible
   immediately instead of waiting for the full 256 tokens.
4. Return/print ``response.text`` (printing the raw `CompletionResponse`
   pydantic object yields a repr, not the answer the user expects).
"""

import json
import os

from llama_index.llms.llama_cpp import LlamaCPP

# A 13B Q4_0 model needs well over ~9 GB of RAM once the KV cache for a
# 3900 token context is allocated. If the machine is smaller than that the
# OS starts swapping and generation effectively stalls. A 7B model (or the
# Q4_K_M quant) keeps everything in RAM and turns the "hang" into a reply
# within seconds.
DEFAULT_MODEL_URL = (
    "https://huggingface.co/TheBloke/Llama-2-7B-chat-GGUF/resolve/main/"
    "llama-2-7b-chat.Q4_K_M.gguf"
)
MODEL_KEY = "llama-2-13b-chat"


def get_physical_cores() -> int:
    """llama-cpp-python defaults to cpu_count() // 2 threads (half the
    machine idle). Use the real number of physical cores instead."""
    try:
        import psutil  # optional dependency

        cores = psutil.cpu_count(logical=False)
        if cores:
            return int(cores)
    except Exception:
        pass
    logical = os.cpu_count() or 4
    # assume 2 logical threads per physical core when psutil is unavailable
    return max(1, logical // 2)


def resolve_model():
    """Pick the local GGUF file if paths.json points at an existing one,
    otherwise fall back to downloading the default model."""
    model_url, model_path = DEFAULT_MODEL_URL, None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
        candidate = paths.get(MODEL_KEY)
        if candidate and os.path.exists(candidate):
            model_url, model_path = None, candidate

    return model_url, model_path


def run_agent(query: str) -> str:
    model_url, model_path = resolve_model()
    num_threads = get_physical_cores()

    llm = LlamaCPP(
        model_url=model_url,
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={
            # --- the actual fix -------------------------------------------
            "n_gpu_layers": 0,                 # pure CPU inference
            "n_threads": num_threads,          # was implicitly cpu_count//2
            "n_threads_batch": num_threads,    # prompt processing threads
            "n_batch": 512,                    # sane batch size
            "use_mlock": True,                 # don't swap the weights out
            # ---------------------------------------------------------------
        },
        verbose=True,
    )

    print(
        f"[info] CPU-only inference using {num_threads} physical cores "
        f"(model: {model_path or model_url})",
        flush=True,
    )

    # Stream the answer: tokens are printed as they are produced, so the
    # program shows progress instead of looking frozen for minutes.
    chunks = []
    for chunk in llm.stream_complete(query):
        text = chunk.delta or ""
        chunks.append(text)
        print(text, end="", flush=True)
    print()

    return "".join(chunks)


if __name__ == "__main__":
    query = "What is the capital of France?"
    answer = run_agent(query)
    print("--- final answer ---")
    print(answer)
