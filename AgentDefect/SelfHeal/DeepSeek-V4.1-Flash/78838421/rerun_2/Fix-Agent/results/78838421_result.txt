import logging

from langchain_community.document_loaders import UnstructuredPDFLoader
from langchain_community.vectorstores import Chroma
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_core.prompts import ChatPromptTemplate, PromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough
from langchain.retrievers.multi_query import MultiQueryRetriever

# Prefer the dedicated Ollama package, fall back to the community wrappers.
try:
    from langchain_ollama import ChatOllama, OllamaEmbeddings
except ImportError:  # pragma: no cover - older installs
    from langchain_community.chat_models import ChatOllama
    from langchain_community.embeddings import OllamaEmbeddings

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s - %(levelname)s - %(message)s",
)
# Make MultiQueryRetriever log the generated queries (useful for debugging).
logging.getLogger("langchain.retrievers.multi_query").setLevel(logging.INFO)

EMBED_MODEL = "nomic-embed-text"
LOCAL_MODEL = "llama3"
# Ollama defaults to a 2048 token context window and silently truncates the
# prompt. Raise it so the whole retrieved context is actually seen by the LLM.
NUM_CTX = 8192
CHUNK_SIZE = 1000
CHUNK_OVERLAP = 200
RETRIEVER_K = 4


def format_docs(docs):
    """Turn retrieved Documents into clean plain text for the prompt."""
    return "\n\n".join(doc.page_content for doc in docs)


def build_chain(vector_db, llm):
    # ---------------------------------------------------------------- retriever
    QUERY_PROMPT = PromptTemplate(
        input_variables=["question"],
        template="""You are an AI language model assistant. Your task is to generate five
different versions of the given user question to retrieve relevant documents from
a vector database. By generating multiple perspectives on the user question, your
goal is to help the user overcome some of the limitations of the distance-based
similarity search.
Provide these alternative questions separated by newlines and nothing else.
Output exactly five lines, one question per line, without numbering or bullets.
Original question: {question}""",
    )

    # NOTE: signature is from_llm(llm=..., retriever=..., prompt=...).
    # The previous call passed the retriever first, which built a broken
    # retriever and produced empty / meaningless answers.
    retriever = MultiQueryRetriever.from_llm(
        llm=llm,
        retriever=vector_db.as_retriever(search_kwargs={"k": RETRIEVER_K}),
        prompt=QUERY_PROMPT,
        include_original=True,
    )
    logging.info("Retriever created")

    # ------------------------------------------------------------- RAG prompt
    # Explicit, role-based template with correct placeholders and instructions,
    # so the model answers the question instead of echoing the context.
    prompt = ChatPromptTemplate.from_messages(
        [
            (
                "system",
                "You are a helpful assistant for question answering. "
                "You answer ONLY from the retrieved context. "
                "Never use prior knowledge. If the context does not contain the "
                "answer, reply exactly: 'I don't know based on the provided document.'",
            ),
            (
                "human",
                "Retrieved context:\n"
                "---------------------\n"
                "{context}\n"
                "---------------------\n\n"
                "Question: {question}\n\n"
                "Answer the question directly and concisely (max 5 sentences), "
                "using only the context above.",
            ),
        ]
    )
    logging.info("RAG prompt created")

    chain = (
        {"context": retriever | format_docs, "question": RunnablePassthrough()}
        | prompt
        | llm
        | StrOutputParser()
    )
    logging.info("Chain created")
    return chain


def run_agent(question, local_path="./paper1.pdf"):
    try:
        # -------------------------------------------------------- load the PDF
        if not local_path:
            logging.error("Upload a PDF file")
            raise ValueError("No PDF file uploaded")

        loader = UnstructuredPDFLoader(file_path=local_path)
        data = loader.load()
        logging.info("Loading of PDF is done")

        if not data:
            raise ValueError("The PDF contained no extractable text")
        logging.info(f"First page content preview: {data[0].page_content[:500]}...")

        # --------------------------------------------------------- split/chunk
        # Smaller chunks keep the retrieved context inside the model's window.
        text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=CHUNK_SIZE, chunk_overlap=CHUNK_OVERLAP
        )
        chunks = text_splitter.split_documents(data)
        logging.info(f"Created {len(chunks)} chunks")

        # ---------------------------------------------------------- vector db
        logging.info("Creating Vector db")
        embedding_model = OllamaEmbeddings(model=EMBED_MODEL)
        print("Embedding", embedding_model)
        try:
            vector_db = Chroma.from_documents(
                documents=chunks,
                embedding=embedding_model,
                collection_name="local-rag",
            )
        except Exception as e:
            logging.error(f"Error creating vector db: {e}")
            raise  # do not continue with an undefined/broken store

        if vector_db:
            logging.info("Vector db verification successful")
        else:
            logging.error("Vector db creation failed")
            raise ValueError("Vector db creation failed")

        # ---------------------------------------------------------------- LLM
        llm = ChatOllama(model=LOCAL_MODEL, temperature=0, num_ctx=NUM_CTX)
        logging.info("LLM model loaded")

        chain = build_chain(vector_db, llm)

        # ------------------------------------------------------------- invoke
        response = chain.invoke(question)
        print(response)
        logging.info("Chain invoked")
        logging.info(f"Response: {response}")

        return response

    except Exception as e:
        logging.error(f"An error occurred: {e}")
        return None


if __name__ == "__main__":
    question = "What are the 5 pillars of global cooperation?"
    response = run_agent(question)
    print(f"\nFinal Output: {response}")
