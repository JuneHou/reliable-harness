import logging

from langchain_community.document_loaders import UnstructuredPDFLoader
from langchain_community.embeddings import OllamaEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain.prompts import ChatPromptTemplate, PromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_community.chat_models import ChatOllama
from langchain_core.runnables import RunnablePassthrough
from langchain.retrievers.multi_query import MultiQueryRetriever

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')


def run_agent(question):
    local_path = "./paper1.pdf"

    # ---------------------------------------------------------------
    # 1) Load the PDF
    # ---------------------------------------------------------------
    if not local_path:
        logging.error("Upload a PDF file")
        raise ValueError("No PDF file uploaded")

    loader = UnstructuredPDFLoader(file_path=local_path)
    data = loader.load()
    if not data:
        raise ValueError("The PDF was loaded but contains no text")
    logging.info("Loading of PDF is done")

    # Preview first page
    logging.info(f"First page content preview: {data[0].page_content[:500]}...")

    # ---------------------------------------------------------------
    # 2) Split and chunk
    #    IMPORTANT: chunks must be small enough to be (a) embedded
    #    meaningfully and (b) actually fit several of them inside the
    #    LLM prompt. 7500-character chunks make each retrieved chunk a
    #    very large, diluted blob, so the retrieved context is mostly
    #    irrelevant and the model answers "I don't know"/something
    #    unrelated. Use ~1000 chars with 200 overlap instead.
    # ---------------------------------------------------------------
    text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
    chunks = text_splitter.split_documents(data)
    if not chunks:
        raise ValueError("No chunks were created from the PDF")
    logging.info(f"Created {len(chunks)} chunks")

    # ---------------------------------------------------------------
    # 3) Create the vector store.
    #    The original code swallowed the exception and then referenced
    #    `vector_db`, which is never bound when embedding/store creation
    #    fails. That NameError was caught by the outer handler, so the
    #    program silently "stopped" right after the embedding phase and
    #    returned None. Bind the variable first and let real errors
    #    propagate so the failure is visible.
    # ---------------------------------------------------------------
    vector_db = None
    try:
        logging.info("Creating Vector db")
        embedding_model = OllamaEmbeddings(model="nomic-embed-text", show_progress=True)
        print("Embedding", embedding_model)
        vector_db = Chroma.from_documents(
            documents=chunks,
            embedding=embedding_model,
            collection_name="local-rag",
        )
        logging.info("Local db created successfully")
    except Exception as e:
        logging.error(f"Error creating vector db: {e}")
        raise

    if vector_db is None:
        raise ValueError("Vector db creation failed")
    logging.info("Vector db verification successful")

    # ---------------------------------------------------------------
    # 4) LLM from Ollama
    # ---------------------------------------------------------------
    local_model = "llama3"
    llm = ChatOllama(model=local_model, temperature=0)
    logging.info("LLM model loaded")

    # ---------------------------------------------------------------
    # 5) Multi-query prompt.
    #    The prompt must produce *exactly* one question per line with no
    #    preamble or numbering, otherwise the retriever's line parser
    #    feeds junk lines ("Here are five alternative questions:", "1)")
    #    into the vector store and the retrieved context is wrong.
    # ---------------------------------------------------------------
    QUERY_PROMPT = PromptTemplate(
        input_variables=["question"],
        template="""You are an AI language model assistant. Your task is to generate five
different versions of the given user question to retrieve relevant documents from
a vector database. By generating multiple perspectives on the user question, your
goal is to help the user overcome some of the limitations of the distance-based
similarity search.

Rules:
- Output only the five questions.
- One question per line.
- No numbering, no bullets, no preamble, no explanations, no blank lines.

Original question: {question}""",
    )
    logging.info("Query prompt created")

    retriever = MultiQueryRetriever.from_llm(
        vector_db.as_retriever(search_kwargs={"k": 4}),
        llm,
        prompt=QUERY_PROMPT,
        include_original=True,
    )
    logging.info("Retriever created")

    # ---------------------------------------------------------------
    # 6) RAG prompt.
    #    Make the instruction explicit and use proper chat roles instead
    #    of one ambiguous indented string, so the answer is actually
    #    grounded in the retrieved context (and correctly formatted).
    # ---------------------------------------------------------------
    template = """You are an assistant for question-answering tasks.
Answer the question using ONLY the following pieces of retrieved context.
If the context does not contain the answer, say that you don't know - do not
make anything up. Keep the answer concise and well structured.

Context:
{context}

Question: {question}

Answer:"""
    prompt = ChatPromptTemplate.from_template(template)
    logging.info("RAG prompt created")

    def format_docs(docs):
        return "\n\n".join(doc.page_content for doc in docs)

    chain = (
        {"context": retriever | format_docs, "question": RunnablePassthrough()}
        | prompt
        | llm
        | StrOutputParser()
    )
    logging.info("Chain created")

    response = chain.invoke(question)
    print(response)
    logging.info("Chain invoked")
    logging.info(f"Response: {response}")

    return response


if __name__ == "__main__":
    question = "What are the 5 pillars of global cooperation?"
    response = run_agent(question)
    print(f"\nFinal Output: {response}")
