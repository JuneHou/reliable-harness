import logging

from langchain_community.document_loaders import UnstructuredPDFLoader
from langchain_community.embeddings import OllamaEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain.prompts import ChatPromptTemplate, PromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_community.chat_models import ChatOllama
from langchain_core.runnables import RunnablePassthrough
from langchain_core.documents import Document
from langchain.retrievers.multi_query import MultiQueryRetriever

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

# MultiQueryRetriever logs the queries it generated at INFO level. Keeping this on makes
# the pipeline visible instead of appearing to silently "stop" after the embedding step.
logging.getLogger("langchain.retrievers.multi_query").setLevel(logging.INFO)

# ---------------------------------------------------------------------------
# Tuning values
#
# The original script used chunk_size=7500 with k=4 retrieved documents, i.e. a
# ~30.000 character (~8k token) context that was fed into `llama3` while Ollama's
# default context window (num_ctx) is only 2048 tokens.  Ollama silently truncates
# an over-long prompt (it does not raise), so the model ends up with an almost
# empty/irrelevant context and returns an empty string.  str output parser then
# yields "", nothing is printed after the embedding logs and run_agent() returns
# "" / None -> "embedding finished but the script stops before continuing".
#
# Fix: chunks that comfortably fit the (explicitly enlarged) context window,
# a bounded number of retrieved chunks and num_ctx set on the chat model.
# ---------------------------------------------------------------------------
CHUNK_SIZE = 1000
CHUNK_OVERLAP = 200
NUM_RESULTS = 4
NUM_CTX = 8192
LOCAL_MODEL = "llama3"
EMBEDDING_MODEL = "nomic-embed-text"


def format_docs(docs):
    """Turn retrieved Documents into clean, compact context text.

    Without this the prompt receives the repr of each Document
    (page_content=... metadata=...), which wastes context tokens and is one of
    the reasons the model answered badly / not at all.
    """
    return "\n\n".join(
        d.page_content.strip() if isinstance(d, Document) else str(d).strip()
        for d in docs
    )


def build_vector_db(chunks, embedding_model):
    vector_db = Chroma.from_documents(
        documents=chunks,
        embedding=embedding_model,
        collection_name="local-rag",
    )
    if vector_db is None or vector_db._collection.count() == 0:
        raise RuntimeError("Vector db creation failed: collection is empty")
    return vector_db


def run_agent(question):
    local_path = "./paper1.pdf"
    vector_db = None  # <- defined up-front so a failed build cannot raise UnboundLocalError
    try:
        # Local PDF file uploads
        if local_path:
            loader = UnstructuredPDFLoader(file_path=local_path)
            data = loader.load()
            if not data:
                raise ValueError(f"No text could be extracted from {local_path}")
            logging.info("Loading of PDF is done")
        else:
            logging.error("Upload a PDF file")
            raise ValueError("No PDF file uploaded")

        # Preview first page
        logging.info(f"First page content preview: {data[0].page_content[:500]}...")

        # Split and chunk (small enough to fit into the LLM context window)
        text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=CHUNK_SIZE, chunk_overlap=CHUNK_OVERLAP
        )
        logging.info("Text splitter created")
        chunks = text_splitter.split_documents(data)
        logging.info(f"Created {len(chunks)} chunks")
        if not chunks:
            raise ValueError("PDF produced no chunks, nothing to embed")

        logging.info("Creating Vector db")
        try:
            # NOTE: `show_progress` is deprecated/removed in recent
            # langchain-community releases and makes the constructor raise.
            embedding_model = OllamaEmbeddings(model=EMBEDDING_MODEL)
            print("Embedding", embedding_model)
            vector_db = build_vector_db(chunks, embedding_model)
            logging.info("Local db created successfully")
        except Exception as e:
            logging.error(f"Error creating vector db: {e}")
            raise  # do not silently continue with an undefined/empty store

        logging.info("Vector db verification successful")

        # LLM from Ollama.
        # num_ctx must be large enough for the retrieved context + prompt, otherwise
        # Ollama truncates the prompt and returns an empty completion.
        llm = ChatOllama(
            model=LOCAL_MODEL,
            temperature=0,
            num_ctx=NUM_CTX,
            num_predict=-1,
        )
        logging.info("LLM model loaded")

        QUERY_PROMPT = PromptTemplate(
            input_variables=["question"],
            template="""You are an AI language model assistant. Your task is to generate five
        different versions of the given user question to retrieve relevant documents from
        a vector database. By generating multiple perspectives on the user question, your
        goal is to help the user overcome some of the limitations of the distance-based
        similarity search. Provide these alternative questions separated by newlines.
        Original question: {question}""",
        )
        logging.info("Query prompt created")

        base_retriever = vector_db.as_retriever(search_kwargs={"k": NUM_RESULTS})
        retriever = MultiQueryRetriever.from_llm(
            retriever=base_retriever,
            llm=llm,
            prompt=QUERY_PROMPT,
            parser_key="lines",
        )
        logging.info("Retriever created")

        # RAG prompt: explicit system role + explicit output instruction so the
        # model always produces an answer instead of an empty/nonsense completion.
        template = """Answer the question based ONLY on the following context.
If the answer is not contained in the context, say "I don't know based on the provided document."

Context:
{context}

Question: {question}

Answer:"""
        prompt = ChatPromptTemplate.from_messages(
            [
                (
                    "system",
                    "You are a helpful assistant answering questions about a PDF document. "
                    "Use only the supplied context and be concise.",
                ),
                ("human", template),
            ]
        )
        logging.info("RAG prompt created")

        chain = (
            {"context": retriever | format_docs, "question": RunnablePassthrough()}
            | prompt
            | llm
            | StrOutputParser()
        )
        logging.info("Chain created")

        response = chain.invoke(question)
        logging.info("Chain invoked")

        # Validation: an empty answer is the symptom, not a valid result.
        if not response or not response.strip():
            logging.warning(
                "LLM returned an empty answer with the multi-query chain "
                "(check that Ollama is running and that num_ctx is big enough). "
                "Falling back to a plain similarity-search chain."
            )
            fallback_chain = (
                {"context": base_retriever | format_docs, "question": RunnablePassthrough()}
                | prompt
                | llm
                | StrOutputParser()
            )
            response = fallback_chain.invoke(question)

        response = (response or "").strip()
        if not response:
            raise RuntimeError(
                "The model returned an empty response. Verify `ollama serve` is running, "
                f"that the models `{LOCAL_MODEL}` and `{EMBEDDING_MODEL}` are pulled, "
                "and that the document actually contains the answer."
            )

        print(response)
        logging.info(f"Response: {response}")

        return response

    except Exception as e:
        logging.error(f"An error occurred: {e}")
        return None


if __name__ == "__main__":
    question = "What are the 5 pillars of global cooperation?"
    response = run_agent(question)
    print(f"\nFinal Output: {response}")
