import asyncio


async def run_agent(texts):
    """Run the chain synchronously and asynchronously so the timings can be compared.

    ``Chain.aapply`` is a coroutine function: calling it only builds a coroutine
    object, which must be awaited (or scheduled in the event loop) before any work
    is done.  Awaiting it here yields the same list-of-dicts structure that the
    synchronous ``Chain.apply`` returns.
    """

    import os
    from time import time

    import openai
    from dotenv import load_dotenv, find_dotenv
    from langchain.chains import LLMChain
    from langchain.chat_models import ChatOpenAI
    from langchain.prompts import ChatPromptTemplate

    _ = load_dotenv(find_dotenv())
    openai.api_key = os.getenv('OPENAI_API_KEY')

    llm = ChatOpenAI(temperature=0)

    prompt = ChatPromptTemplate.from_template(
        'Classify the text based on the Common European Framework of Reference '
        'for Languages (CEFR). Give a single value: {text}',
    )
    chain = LLMChain(llm=llm, prompt=prompt)

    # ---- synchronous version ------------------------------------------------
    start = time()
    res_a = chain.apply(texts)
    print(res_a)
    print(f"apply time taken: {time() - start:.2f} seconds")
    print()

    # ---- asynchronous version -----------------------------------------------
    # ``chain.aapply(...)`` returns a coroutine; it has to be awaited, otherwise
    # nothing runs and you would only see a coroutine object printed.
    start = time()
    res_aa = await chain.aapply(texts)
    print(res_aa)
    print(f"aapply time taken: {time() - start:.2f} seconds")
    print()

    assert [r["text"] for r in res_a] == [r["text"] for r in res_aa]

    return res_a


if __name__ == "__main__":
    texts = [
        {'text': 'Hallo, ich bin 25 Jahre alt.'},
        {'text': 'Wie geht es dir?'},
        {'text': 'In meiner Freizeit, spiele ich gerne Fussball.'}
    ]

    # The top level has no running event loop, so create one with asyncio.run.
    result = asyncio.run(run_agent(texts))
    print("\nReturned result:", result)
