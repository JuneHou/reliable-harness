import os
from typing import List

from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import PydanticOutputParser
from pydantic import BaseModel, Field

load_dotenv()


# --- Schema: a LIST of dictionaries (nested JSON), not a single flat dict -----
class MissingMention(BaseModel):
    """One identifier/text pair extracted from the error message."""

    identifier: str = Field(description="This is missing identifier.")
    text: str = Field(description="This is missing text.")


class MissingMentions(BaseModel):
    """Top-level object holding the collection of parsed mentions."""

    mentions: List[MissingMention] = Field(
        description=(
            "List of every identifier/text pair mentioned in the error message. "
            "Return one object per pair, in the order they appear."
        )
    )


def get_output_parser() -> PydanticOutputParser:
    """
    StructuredOutputParser.from_response_schemas() only describes *flat*
    top-level string fields, so it can never emit a list of nested objects.
    PydanticOutputParser emits a proper JSON schema (including arrays and
    nested objects) and validates the parsed result.
    """
    return PydanticOutputParser(pydantic_object=MissingMentions)


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    # parse() validates/coerces against the schema; model_dump() returns plain
    # nested dicts, e.g.
    # {"mentions": [{"identifier": "...", "text": "..."}, ...]}
    parsed = output_parser.parse(response.content)
    return parsed.model_dump()


if __name__ == "__main__":
    test_msg = (
        "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),"
        "AllOf(withId:identifier2, withText:text1),"
        "AllOf(withId:identifier3, withText:text1))"
    )
    PROMPT_TEMPLATE = """
    You are an android developer.
    Parse this error message and provide me identifiers & texts mentioned in error message.
    Return ALL of them as a JSON list under the "mentions" key.
    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
    print(result["mentions"])  # the list of dictionaries
