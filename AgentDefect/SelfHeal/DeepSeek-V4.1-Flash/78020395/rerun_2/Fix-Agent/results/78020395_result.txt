import os
import json
from typing import List, Dict, Any

from dotenv import load_dotenv
from pydantic import BaseModel, Field

from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
# NOTE: langchain.output_parsers.StructuredOutputParser only knows how to describe
# a *flat* object (one key/value pair per ResponseSchema). To get nested JSON
# (a list of dictionaries) you have to describe the shape with a nested model and
# use a JSON/Pydantic parser instead.
from langchain_core.output_parsers import JsonOutputParser

load_dotenv()


# ---------------------------------------------------------------------------
# 1. Describe the nested schema with Pydantic models
# ---------------------------------------------------------------------------
class MissingItem(BaseModel):
    """A single identifier/text pair found in the error message."""

    identifier: str = Field(description="This is missing identifier.")
    text: str = Field(description="This is missing text.")


class MissingItems(BaseModel):
    """Container whose single field is a LIST of nested objects."""

    missing_items: List[MissingItem] = Field(
        description=(
            "A JSON array with one entry per identifier/text pair mentioned in "
            "the error message. Never collapse the pairs into a single object."
        )
    )


# ---------------------------------------------------------------------------
# 2. Build a parser that emits/accepts nested JSON
# ---------------------------------------------------------------------------
def get_output_parser() -> JsonOutputParser:
    # JsonOutputParser(pydantic_object=...) turns the nested model into format
    # instructions like:
    #   {"missing_items": [{"identifier": "...", "text": "..."}, ...]}
    # and parses exactly that shape back into Python data.
    return JsonOutputParser(pydantic_object=MissingItems)


def _to_list_of_dicts(parsed: Any) -> List[Dict[str, Any]]:
    """Normalise whatever the parser returns into a plain list of dicts."""
    if isinstance(parsed, BaseModel):
        parsed = parsed.model_dump()

    # The parser normally returns {"missing_items": [...]}; be forgiving about
    # other key names / bare lists the model may produce.
    if isinstance(parsed, dict):
        for key in ("missing_items", "items", "results", "data", "output"):
            if isinstance(parsed.get(key), list):
                parsed = parsed[key]
                break
        else:
            if all(isinstance(v, dict) for v in parsed.values()) and parsed:
                parsed = list(parsed.values())
            else:
                parsed = [parsed]

    if not isinstance(parsed, list):
        parsed = [parsed]

    result: List[Dict[str, Any]] = []
    for item in parsed:
        if isinstance(item, BaseModel):
            result.append(item.model_dump())
        elif isinstance(item, dict):
            result.append(item)
    return result


# ---------------------------------------------------------------------------
# 3. Predict
# ---------------------------------------------------------------------------
def predict_result(msg: str, PROMPT_TEMPLATE: str) -> List[Dict[str, Any]]:
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    content = response.content

    # Parse the nested JSON; fall back to raw json.loads if the parser is picky.
    try:
        parsed = output_parser.parse(content)
    except Exception:
        start, end = content.find("{"), content.rfind("}")
        if start == -1 or end == -1:
            raise
        parsed = json.loads(content[start : end + 1])

    # -> list[dict], e.g.
    # [{"identifier": "identifier1", "text": "text1"},
    #  {"identifier": "identifier2", "text": "text1"},
    #  {"identifier": "identifier3", "text": "text1"}]
    return _to_list_of_dicts(parsed)


if __name__ == "__main__":
    test_msg = (
        "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),"
        "AllOf(withId:identifier2, withText:text1),"
        "AllOf(withId:identifier3, withText:text1))"
    )

    PROMPT_TEMPLATE = """
    You are an android developer.
    Parse this error message and provide me identifiers & texts mentioned in error message.
    Return them as a JSON array of objects, one object per identifier/text pair.
    --------
    Error message is {msg}
    --------
    {format_instructions}
    """

    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
    for item in result:
        print(item["identifier"], "->", item["text"])
