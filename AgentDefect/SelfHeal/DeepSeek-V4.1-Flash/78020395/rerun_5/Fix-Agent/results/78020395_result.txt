import json
import os
from typing import Any, Dict, List

from dotenv import load_dotenv
from langchain_core.output_parsers import JsonOutputParser
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.utils.json import parse_json_markdown
from langchain_openai import ChatOpenAI
from pydantic import BaseModel, Field

load_dotenv()


# ---------------------------------------------------------------------------
# Schema: one item (flat dict) ...
# ---------------------------------------------------------------------------
class MissingItem(BaseModel):
    identifier: str = Field(description="The identifier mentioned in the error message.")
    text: str = Field(description="The text mentioned in the error message.")


# ... and a container whose single field is a LIST of items, so the parser is
# able to produce nested JSON: {"items": [{"identifier": ...}, {...}]}
# (StructuredOutputParser + ResponseSchema can only ever describe a single
#  flat object, which is why the original code never returned a list.)
class MissingItems(BaseModel):
    items: List[MissingItem] = Field(
        default_factory=list,
        description="All identifier/text pairs found in the error message.",
    )


def get_output_parser():
    # JsonOutputParser takes a pydantic object as the *schema* (not as a
    # single-object contract) and can happily validate nested lists.
    return JsonOutputParser(pydantic_object=MissingItems)


def _normalise(parsed: Any) -> List[Dict[str, Any]]:
    """Force the parsed output into a list of dictionaries (nested JSON)."""
    if isinstance(parsed, list):
        raw_items = parsed
    elif isinstance(parsed, dict):
        raw_items = None
        for key in ("items", "results", "data", "matches", "identifiers"):
            if isinstance(parsed.get(key), list):
                raw_items = parsed[key]
                break
        if raw_items is None:
            # Already a flat dict -> wrap it so callers always get a list.
            raw_items = [parsed] if parsed else []
    elif parsed is None:
        raw_items = []
    else:
        raw_items = []

    result: List[Dict[str, Any]] = []
    for item in raw_items:
        if isinstance(item, BaseModel):
            item = item.model_dump()
        if isinstance(item, dict):
            result.append(
                {
                    "identifier": item.get("identifier"),
                    "text": item.get("text"),
                }
            )
    return result


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    content = response.content
    try:
        parsed = output_parser.parse(content)
    except Exception:
        # Model returned a bare JSON array / fenced block -> parse leniently.
        parsed = parse_json_markdown(content)

    return _normalise(parsed)


def predict_result_structured(msg, PROMPT_TEMPLATE):
    """Alternative (recommended on modern LangChain): let the model do the
    structured decoding with the same nested pydantic schema."""
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    ).with_structured_output(MissingItems)

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions="")
    parsed = model.invoke(message)
    if isinstance(parsed, BaseModel):
        parsed = parsed.model_dump()
    return _normalise(parsed)


if __name__ == "__main__":
    test_msg = (
        "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),"
        "AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"
    )
    PROMPT_TEMPLATE = """
    You are an android developer.
    Parse this error message and provide me identifiers & texts mentioned in error message.

    Return EVERY identifier/text pair you find, as a JSON object whose "items"
    key holds an array of objects, like this:
    {{"items": [{{"identifier": "identifier1", "text": "text1"}},
                {{"identifier": "identifier2", "text": "text2"}}]}}

    Do not return a single flat object; always return the list under "items".
    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(json.dumps(result, indent=2))
