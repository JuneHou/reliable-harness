import os
from typing import List

from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import PydanticOutputParser
from pydantic import BaseModel, Field

load_dotenv()


class MissingItem(BaseModel):
    """A single identifier/text pair mentioned in the error message."""

    identifier: str = Field(description="This is missing identifier.")
    text: str = Field(description="This is missing text.")


class MissingItems(BaseModel):
    """Wrapper object so the parser output is a list of nested dictionaries."""

    items: List[MissingItem] = Field(
        description="List of every identifier/text pair mentioned in the error message."
    )


def get_output_parser():
    # A Pydantic model with a List[...] field is required: StructuredOutputParser
    # only supports a *flat* dict (name -> value), it cannot express nested
    # objects or lists, which is why the original output was a single flat dict.
    output_parser = PydanticOutputParser(pydantic_object=MissingItems)
    return output_parser


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    # parse() returns the pydantic object -> dump it to plain nested dicts
    parsed = output_parser.parse(response.content)
    response_as_dict = parsed.model_dump()
    # Return the list of dictionaries directly (nested JSON)
    return response_as_dict["items"]


if __name__ == "__main__":
    test_msg = "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"
    PROMPT_TEMPLATE = """ 
    You are an android developer. 
    Parse this error message and provide me identifiers & texts mentioned in error message. 
    Return ALL of the pairs you find as a JSON list, one object per pair, each object
    having exactly the keys "identifier" and "text".
    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
