import os
from typing import List

from dotenv import load_dotenv
from pydantic import BaseModel, Field
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import PydanticOutputParser

load_dotenv()


class MissingIdentifierText(BaseModel):
    """One (identifier, text) pair extracted from the error message."""

    identifier: str = Field(description="This is missing identifier.")
    text: str = Field(description="This is missing text.")


class MissingItems(BaseModel):
    """Container that forces the model to return a *list* of objects."""

    items: List[MissingIdentifierText] = Field(
        description=(
            "A list of JSON objects. Each object has exactly two keys: "
            "'identifier' and 'text'. Add one object per AllOf(...) term "
            "found in the error message."
        )
    )


def get_output_parser():
    # PydanticOutputParser supports nested schemas (list of objects),
    # whereas StructuredOutputParser only supports a flat dict.
    output_parser = PydanticOutputParser(pydantic_object=MissingItems)
    return output_parser


def _as_dict(item):
    """pydantic v2 / v1 compatibility."""
    if hasattr(item, "model_dump"):
        return item.model_dump()
    return item.dict()


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(
        msg=msg,
        format_instructions=format_instructions,
    )
    response = model.invoke(message)

    parsed = output_parser.parse(response.content)

    # Return a real Python list of plain dictionaries (nested JSON shape).
    return [_as_dict(item) for item in parsed.items]


if __name__ == "__main__":
    test_msg = (
        "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),"
        "AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"
    )
    PROMPT_TEMPLATE = """
    You are an android developer.
    Parse this error message and provide me identifiers & texts mentioned in the error message.

    IMPORTANT OUTPUT RULES:
    - Return the data as a JSON list of objects, not a single flat object.
    - Each object must contain exactly the keys "identifier" and "text".
    - Emit one object for EVERY AllOf(...) term in the error message
      (e.g. withId:identifier1 -> identifier, withText:text1 -> text).
    - Do not merge the pairs into one object and do not add extra keys.

    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
