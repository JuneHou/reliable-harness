import os
import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer
from llama_index.core import SQLDatabase, VectorStoreIndex, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

load_dotenv()

OPENAI_API_KEY = os.environ.get('OPENAI_API_KEY')
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

pg_uri = os.environ.get('DATABASE_URL', 'sqlite:///test.db')

# Field (JSON key) descriptions that should be visible to the SQL query engine
metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
}

TABLE_NAME = "snapshots"

TABLE_DESCRIPTION = (
    "The 'snapshots' table stores one row per snapshot of analytics data. "
    "The 'id' column is the primary key. The 'data' column holds a JSON object "
    "whose keys are metric names and whose values are the metric values for that snapshot."
)


def build_table_context_str(table, field_descriptions, table_description=None):
    """Build a natural language description of a table, its columns and the
    Keys/fields contained inside its JSON column.

    This string is what LlamaIndex embeds and hands to the LLM, so the JSON
    metadata (field descriptions) must be part of it - a column `comment` alone
    is never forwarded to the LLM by the SQL query engine.
    """
    lines = []
    if table_description:
        lines.append(table_description)

    lines.append(f"Table name: {table.name}")
    lines.append("Columns:")
    for column in table.columns:
        col_desc = (column.comment or "").strip()
        # Avoid dumping the raw JSON blob as the column description
        try:
            parsed = json.loads(col_desc) if col_desc.startswith("{") else None
        except (ValueError, TypeError):
            parsed = None
        if parsed is not None:
            col_desc = "JSON object containing metric snapshots"
        if not col_desc:
            col_desc = "no description"
        lines.append(f"- {column.name} ({column.type}): {col_desc}")

    # Attach the description of every JSON field so the LLM can map a metric
    # name mentioned by the user to the JSON key stored in the table.
    json_columns = [c.name for c in table.columns if isinstance(c.type, JSON)]
    if field_descriptions and json_columns:
        json_column = json_columns[0]
        lines.append(
            f"Fields (JSON keys) stored in the '{json_column}' column and their meaning:"
        )
        for key, description in field_descriptions.items():
            lines.append(f"- {key}: {description}")

    return "\n".join(lines)


class SqlQueryEngine:

    def __init__(self):
        self.schema = []

    def query_engine(self):
        engine = create_engine(pg_uri)
        metadata_obj = MetaData()
        # Convert the metadata dictionary to a JSON string (kept as the column comment)
        metadata_json = json.dumps(metadata_dict)

        # create snapshot SQL table
        snapshots_table = Table(
            TABLE_NAME,
            metadata_obj,
            Column("id", Integer, primary_key=True, comment="Primary key of the snapshot row"),
            Column("data", JSON, comment=metadata_json),
        )

        metadata_obj.create_all(engine)

        # Only expose the table we care about to the SQL database / query engine
        sql_database = SQLDatabase(engine, include_tables=[TABLE_NAME])
        nodes = SQLTableNodeMapping(sql_database)

        # Build the table schema objects WITH their context strings, so the
        # metadata descriptions are embedded in the object index and used by
        # SQLTableRetrieverQueryEngine when it generates SQL.
        self.schema = [
            SQLTableSchema(
                table_name=table.name,
                context_str=build_table_context_str(
                    table, metadata_dict, TABLE_DESCRIPTION
                ),
            )
            for table in metadata_obj.tables.values()
        ]

        # We dump the table schema information into a vector index. The vector index is stored
        # within the context builder for future use.
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        llm = OpenAI(temperature=0.0, model="gpt-3.5-turbo")
        Settings.llm = llm

        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )

        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response


if __name__ == "__main__":

    query = "What does the page_call_phone_clicks_logged_in_unique field mean in the data column?"
    sql_engine, response = run_agent(query)

    for schema_item in sql_engine.schema:
        print(f"\nTable: {schema_item.table_name}")
        if hasattr(schema_item, 'context_str') and schema_item.context_str:
            print(f"  context_str: {schema_item.context_str}")
            print("  Status: Metadata context is SET")
        else:
            print("  context_str: None")
            print("  Status: Metadata context is NOT SET")

**The fix:**

- Builds schema objects with that string attached:
  ```python
  SQLTableSchema(table_name=table.name, context_str=build_table_context_str(...))
  ```
- Restricts exposure to the relevant table: `SQLDatabase(engine, include_tables=[TABLE_NAME])`.
- Added a comment for the `id` column, `temperature=0.0` (more deterministic text-to-SQL), and removal of dead deprecated comments.

This directly addresses the bug: `SQLTableSchema` is a pydantic model with `table_name: str` and `context_str: Optional[str]` fields, and `SQLTableRetrieverQueryEngine` appends it into the table info prompt (`"The table description is: " + table_schema_obj.context_str`).
