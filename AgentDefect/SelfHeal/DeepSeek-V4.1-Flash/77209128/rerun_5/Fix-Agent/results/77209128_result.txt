import os
import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer
from llama_index.core import SQLDatabase, VectorStoreIndex, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

load_dotenv()

OPENAI_API_KEY = os.environ.get('OPENAI_API_KEY')
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

pg_uri = os.environ.get('DATABASE_URL', 'sqlite:///test.db')

metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
}

TABLE_NAME = "snapshots"
DATA_COLUMN = "data"


def build_table_context(metadata: dict, table_name: str = TABLE_NAME,
                        data_column: str = DATA_COLUMN) -> str:
    """
    Build the natural-language description that will be attached to the table
    schema object (``SQLTableSchema.context_str``).

    This is the piece of text that actually reaches the LLM (both when the
    table is selected from the vector index and when the SQL is generated),
    so the JSON field descriptions must live here - a SQL comment on the
    column is never picked up by LlamaIndex.
    """
    lines = [
        f"The table `{table_name}` contains snapshot rows of analytics data.",
        f"Each row has an `id` (integer primary key) and a `{data_column}` column "
        f"which stores a JSON object of metric values.",
        "",
        f"Keys available inside the JSON `{data_column}` column of `{table_name}` "
        f"and their meaning:",
    ]
    for field_name, description in metadata.items():
        lines.append(f"- {field_name} (JSON key in `{data_column}`): {description}")
    lines.append("")
    lines.append(
        f"Example: to retrieve the value of a key use a JSON extraction such as "
        f"`SELECT {data_column} ->> '{next(iter(metadata))}' FROM {TABLE_NAME};` "
        f"(or `json_extract({data_column}, '$.{next(iter(metadata))}')` on SQLite)."
    )
    return "\n".join(lines)


def _schema_kwargs(table_name: str, context_str: str,
                   column_descriptions: dict) -> dict:
    """
    Build keyword arguments for `SQLTableSchema`, only adding the
    column-level fields that the installed LlamaIndex version actually
    supports (they were added in later releases).
    """
    kwargs = {"table_name": table_name, "context_str": context_str}
    fields = getattr(SQLTableSchema, "model_fields", None)
    if fields is None:
        fields = getattr(SQLTableSchema, "__fields__", {}) or {}
    if "col_descriptions" in fields:
        kwargs["col_descriptions"] = column_descriptions
    return kwargs


class SqlQueryEngine:

    def __init__(self):
        self.schema = []

    def query_engine(self):
        engine = create_engine(pg_uri)
        metadata_obj = MetaData()

        # Convert the metadata dictionary to a JSON string
        metadata_json = json.dumps(metadata_dict)

        # create snapshot SQL table
        # NOTE: the comment/JSON payload is only kept on the SQLAlchemy side for
        # reference; LlamaIndex does not read column comments. The descriptions
        # are passed explicitly through SQLTableSchema.context_str below.
        snapshots_table = Table(
            TABLE_NAME,
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
            comment=metadata_json,
        )

        metadata_obj.create_all(engine)

        sql_database = SQLDatabase(engine, include_tables=[TABLE_NAME])
        nodes = SQLTableNodeMapping(sql_database)

        # Human readable description of the JSON payload of the `data` column.
        table_context = build_table_context(metadata_dict)
        column_descriptions = {
            "id": "Integer primary key of the snapshot row.",
            "data": (
                "JSON object holding analytics metrics. Available keys: "
                + "; ".join(
                    f"{key} -> {value}" for key, value in metadata_dict.items()
                )
            ),
        }

        # We dump the table schema information into a vector index. The vector index
        # is stored within the context builder for future use.
        self.schema = []
        for table_name in sql_database.get_usable_table_names():
            self.schema.append(
                SQLTableSchema(
                    **_schema_kwargs(table_name, table_context, column_descriptions)
                )
            )

        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        llm = OpenAI(temperature=0, model="gpt-3.5-turbo")
        Settings.llm = llm

        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )

        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response


if __name__ == "__main__":

    query = "What does the page_call_phone_clicks_logged_in_unique field mean in the data column?"
    sql_engine, response = run_agent(query)

    for schema_item in sql_engine.schema:
        print(f"\nTable: {schema_item.table_name}")
        if hasattr(schema_item, 'context_str') and schema_item.context_str:
            print(f"  context_str: {schema_item.context_str}")
            print("  Status: Metadata context is SET")
        else:
            print("  context_str: None")
            print("  Status: Metadata context is NOT SET")
