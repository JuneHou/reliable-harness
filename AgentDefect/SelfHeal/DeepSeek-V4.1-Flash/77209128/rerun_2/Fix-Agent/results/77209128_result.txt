import os
import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer
from llama_index.core import SQLDatabase, VectorStoreIndex, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI
from llama_index.embeddings.openai import OpenAIEmbedding

load_dotenv()

OPENAI_API_KEY = os.environ.get('OPENAI_API_KEY')
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

pg_uri = os.environ.get('DATABASE_URL', 'sqlite:///test.db')

TABLE_NAME = "snapshots"

# JSON metadata: field name -> human readable description of the field that
# lives inside the `data` JSON column of the snapshots table.
metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
}


def build_table_context(table_name: str, field_descriptions: dict) -> str:
    """Build a human readable description of the table + its JSON fields.

    This string is what gets attached to the SQLTableSchema (``context_str``)
    and therefore what the retriever and the SQL generation prompt actually see.
    """
    sample_field = next(iter(field_descriptions))
    lines = [
        f'Table "{table_name}" stores snapshot records.',
        'Column "id" is the primary key.',
        'Column "data" holds a JSON object whose keys are metric field names.',
        "Field descriptions inside the \"data\" JSON column:",
    ]
    for field_name, description in field_descriptions.items():
        lines.append(f'- "{field_name}": {description}')
    lines.append(
        f'To read such a field use a JSON accessor, e.g. '
        f"data->>'{sample_field}' (SQLite/Postgres)."
    )
    return "\n".join(lines)


class SqlQueryEngine:

    def __init__(self):
        self.schema = []

    def query_engine(self):
        engine = create_engine(pg_uri)
        metadata_obj = MetaData()
        # Convert the metadata dictionary to a JSON string. This is kept as a
        # column comment (works on DBs that support comments, e.g. Postgres --
        # SQLite simply ignores it), but it is *not* what feeds the LLM.
        metadata_json = json.dumps(metadata_dict)
        # create snapshot SQL table
        snapshots_table = Table(
            TABLE_NAME,
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
        )

        metadata_obj.create_all(engine)

        # Restrict the SQL database to the table we care about so the retrieved
        # schema is deterministic.
        sql_database = SQLDatabase(engine, include_tables=[TABLE_NAME])
        nodes = SQLTableNodeMapping(sql_database)

        llm = OpenAI(temperature=0.0, model="gpt-3.5-turbo")
        Settings.llm = llm
        # The ObjectIndex needs an embedding model as well.
        Settings.embed_model = OpenAIEmbedding()

        # KEY FIX: the JSON field descriptions must be attached to the
        # SQLTableSchema through `context_str`. Previously the schema was built
        # with only `table_name`, so `context_str` stayed None: the vector index
        # (built from `str(SQLTableSchema)` == "<table>: <context_str>") and the
        # SQL prompt (which appends `context_str` to the table info) never saw
        # the metadata. A column comment alone is not enough -- and SQLite does
        # not persist column comments at all.
        table_context = build_table_context(TABLE_NAME, metadata_dict)

        self.schema = [
            SQLTableSchema(table_name=TABLE_NAME, context_str=table_context)
            for table_name in metadata_obj.tables.keys()
        ]

        # Dump the table schema information into a vector index so the right
        # table (and its context) is retrieved at query time.
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )

        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response


if __name__ == "__main__":

    query = "What does the page_call_phone_clicks_logged_in_unique field mean in the data column?"
    sql_engine, response = run_agent(query)

    for schema_item in sql_engine.schema:
        print(f"\nTable: {schema_item.table_name}")
        if getattr(schema_item, 'context_str', None):
            print(f"  context_str: {schema_item.context_str}")
            print("  Status: Metadata context is SET")
            # Also verify the text that is embedded for retrieval actually
            # contains the metadata.
            print(f"  Retrieval node text: {str(schema_item)[:200]}...")
        else:
            print("  context_str: None")
            print("  Status: Metadata context is NOT SET")
