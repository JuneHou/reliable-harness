import os
import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer
from llama_index.core import SQLDatabase, VectorStoreIndex, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

load_dotenv()

OPENAI_API_KEY = os.environ.get('OPENAI_API_KEY')
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

pg_uri = os.environ.get('DATABASE_URL', 'sqlite:///test.db')

metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
}


class SqlQueryEngine:

    def __init__(self):
        self.schema = []

    def query_engine(self):
        engine = create_engine(pg_uri)
        metadata_obj = MetaData()
        # Convert the metadata dictionary to a JSON string
        metadata_json = json.dumps(metadata_dict, indent=2)

        # create snapshot SQL table
        table_name = "snapshots"
        snapshots_table = Table(
            table_name,
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column(
                "data",
                JSON,
                comment=(
                    "JSON object mapping metric field names to their human "
                    f"readable descriptions: {metadata_json}"
                ),
            ),
        )

        metadata_obj.create_all(engine)

        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        # ------------------------------------------------------------------
        # FIX: the column `comment` (and the JSON metadata in general) is NOT
        # picked up by the text-to-SQL prompt. SQLTableRetrieverQueryEngine
        # only receives the schema of a table through the matching
        # SQLTableSchema object, using:
        #   * `context_str`       -> free-form table description that is
        #                            prepended to the CREATE TABLE statement
        #   * `col_descriptions`  -> per-column descriptions
        # Therefore the JSON metadata has to be attached to the
        # SQLTableSchema instances (and not only to the SQLAlchemy column).
        # ------------------------------------------------------------------
        field_lines = "\n".join(
            f"  - data.{field}: {description}"
            for field, description in metadata_dict.items()
        )
        table_context = (
            f"Table `{table_name}` holds daily snapshot rows.\n"
            "Column `id`: integer primary key identifying the snapshot row.\n"
            "Column `data`: JSON object whose keys are metric field names and "
            "whose values are the metric values for that row.\n"
            "The JSON `data` column contains the following fields:\n"
            f"{field_lines}"
        )
        col_descriptions = {
            "id": "Integer primary key identifying the snapshot row.",
            "data": (
                "JSON object keyed by metric field name. Available fields:\n"
                f"{field_lines}"
            ),
        }

        self.schema = []
        for tbl_name in metadata_obj.tables.keys():
            try:
                schema_item = SQLTableSchema(
                    table_name=tbl_name,
                    context_str=table_context,
                    col_descriptions=col_descriptions,
                )
            except Exception:
                # Older llama-index versions have no `col_descriptions`
                schema_item = SQLTableSchema(
                    table_name=tbl_name,
                    context_str=table_context,
                )
            self.schema.append(schema_item)

        # We dump the table schema information into a vector index. The vector
        # index is stored within the context builder for future use.
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        llm = OpenAI(temperature=0.9, model="gpt-3.5-turbo")
        # deprecated for new version
        # llm_predictor = LLMPredictor(llm=llm)
        # service_context = ServiceContext.from_defaults(llm_predictor=llm_predictor)
        Settings.llm = llm

        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )

        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response


if __name__ == "__main__":

    query = "What does the page_call_phone_clicks_logged_in_unique field mean in the data column?"
    sql_engine, response = run_agent(query)

    for schema_item in sql_engine.schema:
        print(f"\nTable: {schema_item.table_name}")
        if hasattr(schema_item, 'context_str') and schema_item.context_str:
            print(f"  context_str: {schema_item.context_str}")
            print("  Status: Metadata context is SET")
        else:
            print("  context_str: None")
            print("  Status: Metadata context is NOT SET")
        col_desc = getattr(schema_item, 'col_descriptions', None)
        if col_desc:
            print(f"  col_descriptions: {col_desc}")
