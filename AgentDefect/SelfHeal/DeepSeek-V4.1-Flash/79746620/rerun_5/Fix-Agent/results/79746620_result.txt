"""LlamaIndex: making "null" metadata values filterable.

Why the original code returns nothing
-------------------------------------
LlamaIndex metadata filters (`MetadataFilter` / `MetadataFilters`) only compare
concrete, vector-store friendly scalar values: `str`, `int`, `float` (or lists
of those). `None` is *not* a filterable value:

  * the default comparators treat `None` as "no value", so
    `FilterOperator.EQ` with `value=None` never produces a match;
  * several vector store backends drop / reject `None` metadata and translate
    filter values into native queries, where `None` means "unset filter".

So this behaviour is expected. To index and retrieve "null" metadata, store an
explicit sentinel (here the string "null") at indexing time and filter on that
same sentinel string.
"""

import os
from typing import Any, Dict, Optional

from dotenv import load_dotenv

from llama_index.core import Settings, VectorStoreIndex
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    FilterOperator,
    MetadataFilter,
    MetadataFilters,
)
from llama_index.embeddings.openai import OpenAIEmbedding

# Sentinel stored in place of `None`. Any string works, as long as the same
# value is used when writing the metadata and when building the filter.
NULL_SENTINEL = "null"


def make_filterable_metadata(metadata: Optional[Dict[str, Any]]) -> Dict[str, Any]:
    """Return metadata that LlamaIndex filters can actually match.

    `None` (and other non-scalar values) cannot be matched by metadata
    filters, so they are replaced with a sentinel string. Nested lists are
    normalized element-wise so they stay comparable too.
    """
    if metadata is None:
        return {}

    cleaned: Dict[str, Any] = {}
    for key, value in metadata.items():
        if value is None:
            cleaned[key] = NULL_SENTINEL
        elif isinstance(value, (list, tuple)):
            cleaned[key] = [
                NULL_SENTINEL if item is None else item for item in value
            ]
        elif isinstance(value, (str, int, float, bool)):
            cleaned[key] = value
        else:
            # Anything else (dicts, dates, custom objects, ...) is stringified
            # so that it remains a comparable scalar.
            cleaned[key] = str(value) if value is not None else NULL_SENTINEL
    return cleaned


def build_null_filter(key: str) -> MetadataFilters:
    """Build a filter that matches nodes whose `key` value is the null sentinel."""
    return MetadataFilters(
        filters=[
            MetadataFilter(
                key=key,
                operator=FilterOperator.EQ,
                value=NULL_SENTINEL,
            )
        ]
    )


def run_agent(query: str):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (uses OPENAI_API_KEY from .env when available,
    # otherwise fall back to a deterministic local mock so the example runs).
    if os.getenv("OPENAI_API_KEY"):
        Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")
    else:
        from llama_index.core.embeddings import MockEmbedding

        Settings.embed_model = MockEmbedding(embed_dim=8)

    # Create nodes. The `None` value is normalized to the "null" sentinel so it
    # becomes a first-class, filterable metadata value.
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=make_filterable_metadata({"start_date": None}),
    )
    node_02 = TextNode(
        text="This document has a real start date in the metadata",
        id_="node_02",
        metadata=make_filterable_metadata({"start_date": "2024-01-01"}),
    )

    # Create index
    doc_index = VectorStoreIndex([node_01, node_02])

    # Debug: check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Demonstrate the old (unsupported) attempt, for clarity only.
    null_with_none = MetadataFilters(
        filters=[
            MetadataFilter(
                key="start_date", operator=FilterOperator.EQ, value=None
            )
        ]
    )
    none_retriever = doc_index.as_retriever(
        filters=null_with_none, similarity_top_k=1
    )
    print("Retrieved with value=None (expected: empty):")
    print([n.node_id for n in none_retriever.retrieve(query)])
    print()

    # The working approach: filter on the stored sentinel.
    filters = build_null_filter("start_date")
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes (filtering on the 'null' sentinel):")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved (None is not filterable)")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
        for node in response:
            print(
                f" - {node.node_id}: start_date={node.metadata.get('start_date')!r} "
                f"(matched null sentinel {NULL_SENTINEL!r})"
            )
