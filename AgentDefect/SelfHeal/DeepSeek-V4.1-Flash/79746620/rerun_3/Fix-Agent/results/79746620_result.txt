"""
Filtering "null" metadata values with LlamaIndex — corrected example.

Why the original output did not match expectations
--------------------------------------------------
LlamaIndex evaluates metadata filters with plain equality comparison
(``metadata.get(key) == filter.value`` for ``FilterOperator.EQ``) and every
comparison short-circuits to ``False`` whenever either side is ``None`` —
``None`` is treated as "this key has no value", not as a value you can match.
On top of that, most vector-store backends cannot index ``None`` at all (the
value is simply absent from the stored metadata payload).

Therefore ``MetadataFilter(key="start_date", operator=FilterOperator.EQ,
value=None)`` can never return anything, even when the node metadata really
holds ``None``. That behaviour IS expected — ``None`` is not a filterable
metadata value.

How to index "null" metadata so it is retrievable
-------------------------------------------------
Materialise "null" as an explicit, indexable sentinel value (a non-empty
string such as ``"__NULL__"``) while normalising the metadata before
indexing, and then filter with that very same sentinel. Non-empty strings are
supported by every vector store / metadata filter implementation, so the
filter now matches exactly the documents whose value was originally ``None``.
"""

import os

from dotenv import load_dotenv

from llama_index.core import Settings, VectorStoreIndex
from llama_index.core.embeddings import MockEmbedding
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    FilterOperator,
    MetadataFilter,
    MetadataFilters,
)

# Sentinel used to materialise "null" metadata values so they stay indexable.
# (A non-empty string on purpose: empty strings / falsy values are dropped or
# coerced to None by several vector-store integrations.)
NULL_SENTINEL = "__NULL__"


def normalize_metadata(metadata: dict | None) -> dict:
    """Replace non-indexable metadata values (``None``) by an indexable sentinel."""
    normalized = {}
    for key, value in (metadata or {}).items():
        normalized[key] = NULL_SENTINEL if value is None else value
    return normalized


def configure_embed_model() -> None:
    """Set up the embedding model (falls back to a mock one without credentials)."""
    # Load environment variables from .env file
    load_dotenv()

    if os.getenv("OPENAI_API_KEY"):
        # Uses OPENAI_API_KEY from .env
        from llama_index.embeddings.openai import OpenAIEmbedding

        Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")
    else:
        # Keeps this example runnable (and verifiable) without an API key
        Settings.embed_model = MockEmbedding(embed_dim=8)


def run_agent(query):
    configure_embed_model()

    # Store "null" as an explicit sentinel instead of a raw None
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=normalize_metadata({"start_date": None}),
    )

    # Control node: a real value, which must NOT match the "null" filter
    node_02 = TextNode(
        text="This document has an actual start date in the metadata",
        id_="node_02",
        metadata=normalize_metadata({"start_date": "2024-01-01"}),
    )

    # Create index
    doc_index = VectorStoreIndex([node_01, node_02])

    # Debug: Check what's actually stored / indexed
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter by the "null" sentinel (NOT by value=None, which cannot match)
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=5)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
        # Sanity check: only the node whose "start_date" was None is returned
        assert [n.node_id for n in response] == ["node_01"], response
        print("Correct: the node with indexed 'null' metadata was retrieved.")
