"""
Why the original output was "wrong"
-----------------------------------
LlamaIndex treats a stored ``None`` metadata value as a *missing* value.  The
generic metadata-filter implementation used by the in-memory
``SimpleVectorStore`` (and the same is true for most real backends: Chroma,
PGVector, Qdrant, Milvus, ...) only compares a metadata entry when the stored
value is present and not ``None``:

    if filter.operator == FilterOperator.EQ:
        return lambda x: x.get(filter.key) is not None and x[filter.key] == filter.value

so a node whose ``start_date`` metadata is ``None`` can never match

    MetadataFilter(key="start_date", operator=FilterOperator.EQ, value=None)

and the retriever returns ``[]``.  That is *expected* behaviour, not a crash.

How to index "null" metadata so it can be retrieved by filters
--------------------------------------------------------------
Never persist ``None`` for a value you intend to filter on.  Persist an
explicit, non-null sentinel instead (optionally plus a boolean flag), then
filter on that sentinel / flag.
"""

import os

from dotenv import load_dotenv

from llama_index.core import Settings, VectorStoreIndex
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    FilterOperator,
    MetadataFilter,
    MetadataFilters,
)

# Explicit, comparable placeholder standing in for a "null" metadata value.
NULL_SENTINEL = "NULL"


def make_filterable_metadata(metadata):
    """Rewrite metadata so that ``None`` values become filterable.

    For each ``key`` with a ``None`` value we store:
      * ``key``                -> ``NULL_SENTINEL``  (a real, comparable value)
      * ``key + "__is_null"``  -> ``True``           (convenience boolean flag)

    Non-null values are preserved and get the flag set to ``False``.
    """
    filterable = {}
    for key, value in (metadata or {}).items():
        if value is None:
            filterable[key] = NULL_SENTINEL
            filterable[f"{key}__is_null"] = True
        else:
            filterable[key] = value
            filterable[f"{key}__is_null"] = False
    return filterable


def _configure_embed_model():
    """Use OpenAI embeddings when a key exists, else a local mock embedding.

    Embeddings play no role in metadata filtering, but the retriever still
    needs *some* embed model configured to run a query.
    """
    if os.getenv("OPENAI_API_KEY"):
        from llama_index.embeddings.openai import OpenAIEmbedding

        Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")
    else:
        from llama_index.core.embeddings import MockEmbedding

        Settings.embed_model = MockEmbedding(embed_dim=8)


def _retrieve(doc_index, query, filters):
    return doc_index.as_retriever(filters=filters, similarity_top_k=1).retrieve(query)


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()
    _configure_embed_model()

    # Store an explicit "null" marker instead of a raw None value
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=make_filterable_metadata({"start_date": None}),
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # --- 1) Filter on the sentinel value -------------------------------
    sentinel_filters = MetadataFilters(
        filters=[
            MetadataFilter(
                key="start_date",
                operator=FilterOperator.EQ,
                value=NULL_SENTINEL,
            )
        ]
    )
    nodes = _retrieve(doc_index, query, sentinel_filters)

    # --- 2) Equivalent filter using the boolean flag -------------------
    flag_filters = MetadataFilters(
        filters=[
            MetadataFilter(
                key="start_date__is_null",
                operator=FilterOperator.EQ,
                value=True,
            )
        ]
    )
    flag_nodes = _retrieve(doc_index, query, flag_filters)

    # --- 3) Sanity check: the inverse filter must NOT match ------------
    non_null_filters = MetadataFilters(
        filters=[
            MetadataFilter(
                key="start_date__is_null",
                operator=FilterOperator.EQ,
                value=False,
            )
        ]
    )
    non_null_nodes = _retrieve(doc_index, query, non_null_filters)

    # Show results
    print("Retrieved nodes (sentinel filter):")
    print([(node.node_id, node.metadata) for node in nodes])
    print()
    print("Retrieved nodes (boolean flag filter):")
    print([(node.node_id, node.metadata) for node in flag_nodes])
    print()
    print("Retrieved nodes (non-null flag filter):")
    print([(node.node_id, node.metadata) for node in non_null_nodes])
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved (a stored None value is not filterable with value=None)")
    else:
        print(
            f"\nResult: Found {len(response)} node(s) with 'null' start_date "
            f"metadata via the sentinel filter"
        )
