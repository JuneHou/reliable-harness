from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# ---------------------------------------------------------------------------
# Why the original result was empty (and why that is expected):
#
# LlamaIndex evaluates metadata filters by looking up the metadata value for
# the filter key. When the stored value is `None`, the lookup is
# indistinguishable from "key is missing", so `FilterOperator.EQ` with
# `value=None` evaluates to False and no node is ever returned. Additionally,
# most vector store backends drop or reject `null` payload values, so a `None`
# cannot reliably be indexed as a queryable payload in the first place.
#
# Fix: index "null" as an explicit sentinel value of a supported type (a
# string) and filter using that exact same value/type. Then EQ comparisons
# work and the key stays present in the payload for every backend.
# ---------------------------------------------------------------------------
NULL_SENTINEL = "null"


def normalize_metadata(metadata: dict) -> dict:
    """Convert None (or missing) metadata values into a filterable sentinel."""
    if not metadata:
        return {}
    return {
        key: (NULL_SENTINEL if value is None else value)
        for key, value in metadata.items()
    }


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node whose "null" metadata value is stored as a real, filterable value
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=normalize_metadata({"start_date": None}),
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter by the sentinel value that was actually indexed.
    # The filter value must match the stored value *and* its type.
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes


def demonstrate_is_empty(query):
    """Alternative approach: match a genuinely empty/None value with IS_EMPTY.

    Note: only some backends (e.g. the default SimpleVectorStore) support
    IS_EMPTY. It is kept optional so it can never break the primary result.
    """
    try:
        node = TextNode(
            text="This document has None in the metadata",
            id_="node_is_empty",
            metadata={"start_date": None},
        )
        index = VectorStoreIndex([node])
        filters = MetadataFilters(
            filters=[
                MetadataFilter(key="start_date", operator=FilterOperator.IS_EMPTY)
            ]
        )
        retrieved = index.as_retriever(filters=filters, similarity_top_k=1).retrieve(
            query
        )
        print(f"IS_EMPTY approach retrieved {len(retrieved)} node(s):")
        print([(n.node_id, n.metadata) for n in retrieved])
        return retrieved
    except Exception as exc:  # pragma: no cover - backend dependent
        print(f"IS_EMPTY approach unsupported by this vector store: {exc}")
        return []

    print()
    print("Secondary check (IS_EMPTY on a raw None value):")
    demonstrate_is_empty(query)

import os

from llama_index.core import Settings, VectorStoreIndex

)

# LlamaIndex (and most vector stores underneath it) treat a metadata value of
# `None` the same as a *missing* key: both the indexing path and the filter path
# short-circuit when the looked-up value is `None`.  Because of that, a node whose
# metadata is {"start_date": None} can never be matched by
# MetadataFilter(key="start_date", operator=FilterOperator.EQ, value=None):
# the "null" is simply not indexable / not filterable.
#
# The supported way to index a "null" value is to store an explicit, serializable
# sentinel (a string is safest across vector stores) instead of `None`, and then
# filter on that same sentinel.
NULL_SENTINEL = "__NULL__"


def normalize_metadata(metadata: dict) -> dict:
    """Replace `None` values with an indexable/filterable sentinel string."""
    return {key: (NULL_SENTINEL if value is None else value) for key, value in metadata.items()}


def make_node(node_id: str, text: str, metadata: dict) -> TextNode:
    """Build a TextNode whose metadata contains no `None` values."""
    return TextNode(text=text, id_=node_id, metadata=normalize_metadata(metadata))
