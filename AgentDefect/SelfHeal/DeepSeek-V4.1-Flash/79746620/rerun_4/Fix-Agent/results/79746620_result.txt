"""Metadata filtering for "null" metadata values in LlamaIndex.

Why the original code's output was unexpected
---------------------------------------------
`FilterOperator.EQ` with `value=None` does not match metadata that was indexed as
`None`. In LlamaIndex (and in essentially every vector-store backend it wraps),
a `None` metadata value is treated as "this key is absent": many backends drop
`None` fields when the metadata payload is serialized (Chroma/pgvector/Pinecone
etc. only index scalars/lists), and the generic in-memory comparison path also
never gets a concrete value to compare against. So the filter is effectively
"start_date == <nothing>", which never matches -> no nodes retrieved.

The supported way to make "null" filterable is to index an explicit sentinel
value instead of Python `None` (for example the string "null", an empty string,
or a dedicated marker), and then filter on that exact sentinel.

Notes / alternatives
--------------------
* `NULL_SENTINEL` below is a plain string so it is storable, comparable and
  indexable by every backend (including external vector stores).
* If your backend/LlamaIndex version supports it, `FilterOperator.IS_EMPTY`
  can also be used for "value is missing or empty", but a sentinel is portable.
* Always normalize metadata at indexing time (see `normalize_metadata`) so that
  writes and reads agree on what "null" means.
"""

from __future__ import annotations

import os
from typing import Any, Dict, List, Optional

from dotenv import load_dotenv
from llama_index.core import Settings, VectorStoreIndex
from llama_index.core.embeddings import MockEmbedding
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    FilterOperator,
    MetadataFilter,
    MetadataFilters,
)
from llama_index.embeddings.openai import OpenAIEmbedding

# Explicit, storable representation of a "null" metadata value.
NULL_SENTINEL = "null"


def normalize_metadata(metadata: Optional[Dict[str, Any]]) -> Dict[str, Any]:
    """Replace ``None`` metadata values with an explicit, filterable sentinel.

    Vector stores cannot reliably store/filter Python ``None``, so a metadata
    value of ``None`` must be converted into a concrete value before indexing.
    """
    if not metadata:
        return {}
    return {
        key: (NULL_SENTINEL if value is None else value)
        for key, value in metadata.items()
    }


def build_node(text: str, node_id: str, metadata: Dict[str, Any]) -> TextNode:
    """Create a TextNode whose metadata never contains ``None``."""
    return TextNode(
        text=text,
        id_=node_id,
        metadata=normalize_metadata(metadata),
    )


def run_agent(query: str) -> List[Any]:
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (uses OPENAI_API_KEY from .env when available,
    # otherwise a local mock embedding so the example stays runnable offline)
    if os.getenv("OPENAI_API_KEY"):
        Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")
    else:
        Settings.embed_model = MockEmbedding(embed_dim=8)

    # Create nodes: "null" is stored as an explicit sentinel, not as None
    node_01 = build_node(
        text="This document has None in the metadata",
        node_id="node_01",
        metadata={"start_date": None},
    )
    node_02 = build_node(
        text="This document has a real start date in the metadata",
        node_id="node_02",
        metadata={"start_date": "2024-01-01"},
    )

    # Create index
    doc_index = VectorStoreIndex([node_01, node_02])

    # Debug: check what is actually stored (no None values anymore)
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Try to filter by the "null" value using the sentinel
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=2)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved (None is not filterable)")
    else:
        print(f"\nResult: Found {len(response)} node(s)")

    # The node indexed with a "null" start_date must now be retrievable
    retrieved_ids = [node.node_id for node in response]
    assert "node_01" in retrieved_ids, (
        "Expected node_01 (null start_date, stored via sentinel) to be retrieved, "
        f"got {retrieved_ids}"
    )
    assert "node_02" not in retrieved_ids, (
        "Filtering on the null sentinel must not return nodes with a real value"
    )
    print("\nCheck passed: null metadata is indexed as a sentinel and is filterable.")
