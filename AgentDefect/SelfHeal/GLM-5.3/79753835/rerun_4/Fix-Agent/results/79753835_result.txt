from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer
import os


def create_embedding_model(_chunk_size=None):
    print('loading embeddings...')
    # The original if/else had two identical branches, so a single code path
    # is kept here: make sure the cache folder exists, then load BGE-M3.
    os.makedirs('./embeddings', exist_ok=True)
    _emb_model_name = "BAAI/bge-m3"
    _embed_model = HuggingFaceEmbedding(
        model_name=_emb_model_name,
        max_length=_chunk_size,
        cache_folder='./embeddings',
    )
    print('embeddings loaded')
    return _embed_model


def run_agent(text: str):
    chunk_size = 512
    model_name = "BAAI/bge-m3"

    # The exact tokenizer BGE-M3 uses internally (XLM-RoBERTa style BPE,
    # includes the <s> ... </s> special tokens).
    tokenizer = AutoTokenizer.from_pretrained(model_name)

    embed_model = create_embedding_model(_chunk_size=chunk_size)

    # ---------------------------------------------------------------------
    # FIX for the token-count mismatch:
    #
    # TokenCountingHandler() built *without* a tokenizer falls back to
    # LlamaIndex's GLOBAL tokenizer, which by default is a tiktoken encoding
    # for "gpt-3.5-turbo" (cl100k_base).  That tokenizer has nothing to do
    # with BGE-M3, so `total_embedding_token_count` (computed with
    # cl100k_base) came out lower than the count produced by BGE-M3's own
    # tokenizer.  The embedding model itself never reports a token count --
    # the callback handler tokenizes the text on its own.
    #
    # The handler counts tokens via `len(self.tokenizer(chunk))`, and calling
    # a HF tokenizer directly returns a dict-like BatchEncoding (whose len()
    # is the number of keys, not tokens), so a thin wrapper returning the
    # list of input ids is passed instead.  With the embedding model's own
    # tokenizer installed in the handler, the reported embedding token count
    # is identical to what the tokenizer alone produces, i.e. the count can
    # be replicated without actually running the embedding.
    # ---------------------------------------------------------------------
    def bge_tokenizer(text: str):
        return tokenizer(text)["input_ids"]

    token_counter = TokenCountingHandler(tokenizer=bge_tokenizer)
    callback_manager = CallbackManager([token_counter])

    # Register the callback manager before the embedding model so the
    # EMBEDDING events are guaranteed to be routed to the token counter.
    Settings.callback_manager = callback_manager
    embed_model.callback_manager = callback_manager
    Settings.embed_model = embed_model

    embedding_vector = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Plain tokenizer usage now reproduces the embedding model's token count.
    tokenized_text = tokenizer(text)
    token_count = len(tokenized_text['input_ids'])
    return embedding_tokens, token_count


if __name__ == "__main__":
