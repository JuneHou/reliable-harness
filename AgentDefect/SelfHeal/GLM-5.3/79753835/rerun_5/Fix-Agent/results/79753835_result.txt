from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer
import os

MODEL_NAME = "BAAI/bge-m3"


def create_embedding_model(_chunk_size=None, callback_manager=None):
    print('loading embeddings...')
    if os.path.exists('./embeddings/models--BAAI--bge-m3'):
        _emb_model_name = MODEL_NAME
        _embed_model = HuggingFaceEmbedding(model_name=_emb_model_name, max_length=_chunk_size,
                                            cache_folder='./embeddings',
                                            callback_manager=callback_manager)
    else:
        os.makedirs("./embeddings", exist_ok=True)
        _emb_model_name = MODEL_NAME
        _embed_model = HuggingFaceEmbedding(model_name=_emb_model_name, max_length=_chunk_size,
                                            cache_folder='./embeddings',
                                            callback_manager=callback_manager)
    print('embeddings loaded')
    return _embed_model


def run_agent(text: str):
    chunk_size = 512

    # Load the exact tokenizer that the embedding model (BGE-M3) itself uses.
    tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME)

    # The TokenCountingHandler has no knowledge of which embedding model is
    # being used. By default it falls back to the gpt2 (tiktoken) tokenizer,
    # which is why the reported embedding token count was different (lower)
    # than the count produced by BGE-M3's own tokenizer. Pass the model's
    # tokenizer to the handler so the counted embedding tokens match the
    # model's real tokenization (special tokens included).
    token_counter = TokenCountingHandler(tokenizer=tokenizer.encode)

    callback_manager = CallbackManager([token_counter])

    # Register the callback manager on Settings *before* the embedding model
    # is created, and also pass it directly to the embedding model, so the
    # token counting handler is guaranteed to receive the EMBEDDING events
    # fired by get_text_embedding() (otherwise the model keeps a default,
    # empty callback manager and no embedding tokens are counted).
    Settings.callback_manager = callback_manager

    embed_model = create_embedding_model(_chunk_size=chunk_size,
                                         callback_manager=callback_manager)

    Settings.embed_model = embed_model

    embedding_vector = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Tokenizing the text directly with the model's own tokenizer now
    # replicates exactly the token count the embedding process reports,
    # without having to run the embedding itself.
    tokenized_text = tokenizer(text)
    token_count = len(tokenized_text['input_ids'])
    return embedding_tokens, token_count


if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
    print(f"The tokenizer broke the text into {token_count} tokens")
