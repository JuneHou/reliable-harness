from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer
import os

MODEL_NAME = "BAAI/bge-m3"


def create_embedding_model(_chunk_size=None, callback_manager=None):
    print('loading embeddings...')
    if os.path.exists('./embeddings/models--BAAI--bge-m3'):
        _emb_model_name = MODEL_NAME
        _embed_model = HuggingFaceEmbedding(
            model_name=_emb_model_name,
            max_length=_chunk_size,
            cache_folder='./embeddings',
            # FIX (wiring): hand the callback manager holding the token
            # counter to the embedding model itself, so the
            # CBEventType.EMBEDDING events fired by get_text_embedding()
            # actually reach the TokenCountingHandler.
            callback_manager=callback_manager,
        )
    else:
        os.makedirs("./embeddings", exist_ok=True)
        _emb_model_name = MODEL_NAME
        _embed_model = HuggingFaceEmbedding(
            model_name=_emb_model_name,
            max_length=_chunk_size,
            cache_folder='./embeddings',
            callback_manager=callback_manager,
        )
    print('embeddings loaded')
    return _embed_model


def run_agent(text: str):
    chunk_size = 512

    # FIX (root cause): the TokenCountingHandler counts embedding tokens with
    # *its own* tokenizer, which defaults to the global gpt2/tiktoken
    # tokenizer -- NOT the tokenizer of the embedding model.  That is why
    # `total_embedding_token_count` was lower than the BGE-M3 tokenizer count.
    # Load the tokenizer that BGE-M3 actually uses and give it to the handler.
    tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME, cache_folder='./embeddings')

    def _bge_token_count(text_to_tokenize: str):
        # Tokenize exactly the way the embedding model does (special tokens
        # such as <s>/</s> included), so len() of the result is the model's
        # token count and matches len(tokenizer(text)['input_ids']).
        return tokenizer(text_to_tokenize)['input_ids']

    token_counter = TokenCountingHandler(tokenizer=_bge_token_count)
    callback_manager = CallbackManager([token_counter])

    # FIX (wiring): register the callback manager globally *before* creating
    # the embedding model (it is also passed to the model directly above), so
    # the embedding events are counted no matter how Settings propagates it.
    Settings.callback_manager = callback_manager

    embed_model = create_embedding_model(
        _chunk_size=chunk_size, callback_manager=callback_manager
    )
    Settings.embed_model = embed_model

    embedding_vector = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    tokenized_text = tokenizer(text)
    token_count = len(tokenized_text['input_ids'])
    return embedding_tokens, token_count


if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
    print(f"The tokenizer broke the text into {token_count} tokens")
