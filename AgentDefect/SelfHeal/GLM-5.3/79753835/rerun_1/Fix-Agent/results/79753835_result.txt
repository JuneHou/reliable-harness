"""
Fix for the token-count mismatch between llama-index's TokenCountingHandler
and the BAAI/bge-m3 tokenizer.

Root cause
----------
TokenCountingHandler does NOT tokenize with the embedding model's own
tokenizer. When none is supplied it counts with llama-index's global
tokenizer (Settings.tokenizer), which defaults to tiktoken's
"cl100k_base" encoding (see llama_index.core.utils.get_tokenizer).

The original code compared that number against
AutoTokenizer.from_pretrained("BAAI/bge-m3")(text)["input_ids"], which
  (a) uses a completely different (XLM-RoBERTa / sentencepiece) vocabulary,
      and
  (b) contains the special tokens <s> and </s>,
so it always came out higher than the number reported by the embedding
process.

Fix
---
Count with the exact same tokenizer the handler uses. The standalone count
then reproduces the number the embedding process reports -- no embedding
run required.
"""

import os

from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler


def create_embedding_model(_chunk_size=None):
    print('loading embeddings...')
    # The old if/else branches did exactly the same thing; one
    # behaviour-equivalent path is enough.
    os.makedirs("./embeddings", exist_ok=True)
    _emb_model_name = "BAAI/bge-m3"
    _embed_model = HuggingFaceEmbedding(model_name=_emb_model_name,
                                        max_length=_chunk_size,
                                        cache_folder='./embeddings')
    print('embeddings loaded')
    return _embed_model


def get_counting_tokenizer():
    """Return the tokenizer that TokenCountingHandler counts embedding
    tokens with.

    This is llama-index's global tokenizer -- tiktoken's "cl100k_base"
    encoding by default -- NOT the bge-m3 tokenizer. That default is the
    whole reason the two numbers in the original code disagreed.
    """
    try:
        from llama_index.core.utils import get_tokenizer
        return get_tokenizer()  # tiktoken "cl100k_base" .encode
    except ImportError:
        # Fallback: the global llama-index tokenizer (same tiktoken
        # encoding unless it has been explicitly overridden).
        return Settings.tokenizer


def count_tokens_without_embedding(text: str) -> int:
    """Replicate the token count the embedding process reports for `text`
    WITHOUT running any embedding.

    Just count with the very same tokenizer the TokenCountingHandler uses
    (tiktoken "cl100k_base" by default). Using the bge-m3 tokenizer here is
    what produced the higher, mismatched number in the original code:
    different vocabulary + the extra <s>/</s> special tokens.
    """
    return len(get_counting_tokenizer()(text))


def run_agent(text: str):
    chunk_size = 512
    embed_model = create_embedding_model(_chunk_size=chunk_size)

    # Use one tokenizer everywhere: hand it to the handler explicitly and
    # reuse the very same callable for the standalone count below, so both
    # numbers are computed identically by construction.
    counting_tokenizer = get_counting_tokenizer()
    token_counter = TokenCountingHandler(tokenizer=counting_tokenizer)
    callback_manager = CallbackManager([token_counter])

    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager
    # Belt and suspenders: make sure the embed model reports its
    # EMBEDDING events to our token counter.
    embed_model.callback_manager = callback_manager

    embedding_vector = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Standalone token count -- no bge-m3 tokenizer involved any more, so
    # this now reproduces exactly what the embedding process reports.
    token_count = count_tokens_without_embedding(text)

    return embedding_tokens, token_count


if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
    print(f"The tokenizer broke the text into {token_count} tokens")
    print(f"Counts match: {embedding_tokens == token_count}")

    # NOTE: if you instead want the number of tokens bge-m3 *itself* consumes

    #   true_token_count = len(model_count(text))
