import os
import uuid
from datetime import datetime
from dotenv import load_dotenv
import chromadb
from chromadb.utils import embedding_functions
from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import CharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings


load_dotenv()


config = {
    "OPENAI_API_KEY": os.getenv("OPENAI_API_KEY")
}

class configs:
    EMBEDDINGS_MODEL = "text-embedding-3-small"


class MockPDFFile:
    def __init__(self, filepath, filename):
        self.filepath = filepath
        self.filename = filename


def read_docs(pdf_file):
    pdf_loader = PyPDFLoader(pdf_file.filepath)
    pdf_documents = pdf_loader.load()

    text_splitter = CharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
    documents = text_splitter.split_documents(pdf_documents)
    
    return documents


def generate_and_store_embeddings(documents, pdf_file, user_id):
    client = chromadb.PersistentClient(path="./trained_db")
    collection = client.get_or_create_collection(
        "PDF_Embeddings",
        embedding_function=embedding_functions.OpenAIEmbeddingFunction(
            api_key=config["OPENAI_API_KEY"],
            model_name=configs.EMBEDDINGS_MODEL
        )
    )
    now = datetime.now()

    # FIX: this metadata must be attached to EVERY document chunk so that it
    # is stored on each individual embedding.
    # Previously it was only passed as `collection_metadata`, which merely
    # annotates the collection object itself and is NOT attached to the
    # stored documents/embeddings.  On top of that, PyPDFLoader sets each
    # chunk's `source` metadata to the *full file path*, while the retrieval
    # code filtered with the bare file name -- so `where={"source": filename}`
    # matched nothing and an empty result was returned.
    metadata = {
        "source": pdf_file.filename,
        "user": str(user_id),
        "created_at": now.strftime("%d/%m/%Y %H:%M:%S")
    }

    # Attach the metadata to every chunk (this is what actually gets stored
    # per embedding).  It also overwrites the loader's `source` (full path)
    # with the file name, so the value used in the `where` filter matches
    # exactly what is stored.
    for doc in documents:
        doc.metadata.update(metadata)

    ids = [str(uuid.uuid4()) for _ in range(len(documents))]

    try:
        vectordb = Chroma.from_documents(
                    documents,         
                    embedding=OpenAIEmbeddings(openai_api_key=config["OPENAI_API_KEY"], 
                    model=configs.EMBEDDINGS_MODEL),
                    persist_directory='./trained_db',
                    collection_name = collection.name, 
                    client = client,
                    ids = ids
                )
        if hasattr(vectordb, "persist"):
            try:
                vectordb.persist()
            except Exception:
                # chromadb's PersistentClient writes everything to disk
                # automatically (persist() is deprecated / a no-op in newer
                # versions), so it is safe to ignore failures here.
                pass
        
    except Exception as err:
        print(f"An error occured: {err=}, {type(err)=}")
        return {"answer": "An error occured while generating embeddings. Please check terminal for more details."}
    return vectordb


def get_pdf_embeddings(vectordb, filename, user_id=None):
    """Return ONLY the ids and metadata of the embeddings that belong to
    `filename` (optionally restricted to a single `user_id`).

    The `where` filter is evaluated inside ChromaDB, so we never fetch all
    ids/metadata of the collection, and `include=["metadatas"]` makes sure
    neither the embeddings nor the raw documents are downloaded either.
    """
    where = {"source": filename}
    if user_id is not None:
        where = {"$and": [{"source": filename}, {"user": str(user_id)}]}
    return vectordb.get(where=where, include=["metadatas"])


def delete_pdf_embeddings(vectordb, filename, user_id=None):
    """Delete every embedding that belongs to `filename` and return how many
    were removed."""
    result = get_pdf_embeddings(vectordb, filename, user_id=user_id)
    ids = result.get("ids") or []
    if ids:
        vectordb.delete(ids=ids)
    return len(ids)


def run_agent(pdf_path, user_id):
   
    pdf_file = MockPDFFile(pdf_path, os.path.basename(pdf_path))
    
   
    documents = read_docs(pdf_file) 
    print(f"    ✓ Loaded {len(documents)} document chunks")
    
   
    vectordb = generate_and_store_embeddings(documents, pdf_file, user_id)
    
    if isinstance(vectordb, dict):
       
        return None, pdf_file.filename
    
    
    return vectordb, pdf_file.filename

2. **Path vs filename mismatch**: `PyPDFLoader` sets each chunk's `source` metadata to the *full file path* (e.g., `./docs/test.pdf`), while the retrieval filter used `os.path.basename()` (just `test.pdf`). Even if metadata had been attached per-document, the filter wouldn't match.

✅ **No breaking changes**: The interface is preserved, `collection_metadata` removal is safe (it was non-functional for filtering), and `ids` parameter usage is correct.
