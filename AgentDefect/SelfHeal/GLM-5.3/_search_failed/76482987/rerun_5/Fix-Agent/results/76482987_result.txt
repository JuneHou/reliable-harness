import chromadb

def run_agent(documents):
    if(len(documents) !=2):
        print("number of documents should be 2")
        return
    chroma_client = chromadb.Client()
    collection = chroma_client.create_collection(name="my_collection")
    collection.add(
        embeddings=[[1.2, 2.3, 4.5], [6.7, 8.2, 9.2]],
        documents=documents,
        metadatas=[{"source": "my_source"}, {"source": "my_source"}],
        ids=["id1", "id2"]
    )   

    # ChromaDB's get() defaults to include=["metadatas", "documents"],
    # which is why embeddings came back as None even though they were stored.
    # Explicitly request embeddings in the 'include' parameter.
    return collection.get(include=["ids", "documents", "metadatas", "embeddings"])
