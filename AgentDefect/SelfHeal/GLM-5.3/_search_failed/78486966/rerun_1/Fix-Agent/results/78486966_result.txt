import json

from langchain_text_splitters import CharacterTextSplitter


def run_agent(text_raw):
    """Split text_raw into fixed-size character chunks (30 chars, 5-char overlap).

    Why the original code returned the wrong output:
    CharacterTextSplitter does NOT simply cut the text every `chunk_size`
    characters.  It first splits the text on `separator` -- whose DEFAULT is
    "\n\n" (a blank line) -- and only afterwards merges those pieces into
    chunks of at most `chunk_size` characters.  The input text contains no
    blank lines, so the default configuration returned ONE chunk holding the
    whole text (together with a "Created a chunk of size ..." warning)
    instead of the expected 30-character chunks.

    The fix:
    * separator=""            -> cut the text into individual characters so
                                  real fixed-size chunks can be built.
    * strip_whitespace=False  -> keep every chunk exactly 30 chars long (the
                                  default would trim spaces/newlines at the
                                  chunk borders, yielding 28-30 char chunks).
    * text_raw.strip()        -> drop the leading/trailing newlines that the
                                  triple-quoted string literal adds.
    """
    text_splitter = CharacterTextSplitter(
        separator="",
        chunk_size=30,
        chunk_overlap=5,
        strip_whitespace=False,
        is_separator_regex=False,
    )
    chunks = text_splitter.split_text(text_raw.strip())
    return chunks


def save_chunks_locally(chunks, filepath="chunks.json"):
    """Persist the resulting chunks locally.

    * chunks.json -> the chunk list as JSON (reload later with
                     json.load(open("chunks.json", encoding="utf-8"))).
    * chunks.txt  -> human-readable file, one chunk per block.
    """
    with open(filepath, "w", encoding="utf-8") as f:
        json.dump(chunks, f, indent=2, ensure_ascii=False)

    txt_filepath = filepath.rsplit(".", 1)[0] + ".txt"
    with open(txt_filepath, "w", encoding="utf-8") as f:
        for i, chunk in enumerate(chunks, start=1):
            f.write(f"--- chunk {i} ({len(chunk)} chars) ---\n{chunk}\n\n")

    return filepath, txt_filepath


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)

    # Show each chunk, its length and the 5-character overlap it shares with
    # the previous chunk, so the fixed-size / overlap behaviour is visible.
    for i, chunk in enumerate(chunks, start=1):
        print(f"chunk {i:>2} ({len(chunk):>2} chars): {chunk!r}")
        if i > 1:
            print(f"          5-char overlap with previous chunk: {chunks[i - 2][-5:]!r}")

    print(chunks)
    print(f'\n\n {len(chunks)}')

    json_path, txt_path = save_chunks_locally(chunks)
    print(f"Saved {len(chunks)} chunks locally to '{json_path}' and '{txt_path}'")
