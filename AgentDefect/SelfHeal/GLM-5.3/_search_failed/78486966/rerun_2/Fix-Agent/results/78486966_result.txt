import json

from langchain_text_splitters import CharacterTextSplitter


def _make_splitter():
    r"""Create a CharacterTextSplitter that produces fixed-size character chunks.

    Why the original code returned one single (un-split) chunk:

    * ``CharacterTextSplitter`` first splits the text on ``separator`` and only
      afterwards merges the pieces up to ``chunk_size``.  Its default separator
      is ``"\n\n"`` (a blank line).  The sample text only contains single
      newlines, so no split point was ever found and the whole text was
      returned as ONE chunk of ~290 characters.

    The fix:

    * ``separator=""``  -> the text is first cut into single characters, which
      are then merged into chunks of exactly ``chunk_size`` characters, keeping
      ``chunk_overlap`` characters shared between consecutive chunks.
    * ``strip_whitespace=False`` -> no characters are trimmed from the chunk
      borders, so every chunk is exactly 30 characters long (with the default
      ``True`` the chunk lengths would fluctuate between ~28-30).
    * ``length_function=len`` (and ``measure_chunk_size_by_tokens=False`` on
      langchain-text-splitters >= 1.0) -> the chunk size is counted in
      characters, not tokens.
    """
    kwargs = {
        "separator": "",
        "chunk_size": 30,
        "chunk_overlap": 5,
        "strip_whitespace": False,
        "length_function": len,
    }
    try:
        # langchain-text-splitters >= 1.0 measures chunk_size in tokens by
        # default; switch that off so "30" really means 30 characters.
        return CharacterTextSplitter(measure_chunk_size_by_tokens=False, **kwargs)
    except TypeError:
        # Older versions do not know this keyword argument.
        return CharacterTextSplitter(**kwargs)


def run_agent(text_raw):
    text_splitter = _make_splitter()
    chunks = text_splitter.split_text(text_raw)
    return chunks


def save_chunks_locally(chunks, json_path="chunks.json", txt_path="chunks.txt"):
    """Save the resulting chunks to the local file system."""
    # 1) Machine-readable copy: the full list of chunks as JSON.
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(chunks, f, ensure_ascii=False, indent=2)

    # 2) Human-readable copy: one numbered block per chunk.
    with open(txt_path, "w", encoding="utf-8") as f:
        for i, chunk in enumerate(chunks, start=1):
            f.write(f"--- chunk {i} ({len(chunk)} chars) ---\n{chunk}\n\n")


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)
    print(chunks)
    print(f'\n\n {len(chunks)}')

    # Quick verification: every full chunk is exactly 30 chars long and each
    # chunk starts with the last 5 characters of the previous chunk.
    for i, chunk in enumerate(chunks, start=1):
        print(f"chunk {i:2d} | len={len(chunk):2d} | {chunk!r}")

    save_chunks_locally(chunks)
    print("\nChunks saved locally to 'chunks.json' and 'chunks.txt'")
