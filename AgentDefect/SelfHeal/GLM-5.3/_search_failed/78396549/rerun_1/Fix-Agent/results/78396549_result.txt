import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage

load_dotenv()


def run_agent(prompt: str) -> str:
    """Invoke a ChatOpenAI model and return the assistant's text answer.

    Why the original code printed nothing useful:
      llm.invoke(...) on a ChatOpenAI instance does NOT return a plain
      string -- it returns an AIMessage object (a Pydantic model). The
      generated text lives inside its `content` attribute, which the
      original code never accessed, so printing the raw object showed
      something like "AIMessage(content='', ...)" or an opaque repr
      instead of the actual answer.

    Fixes applied:
      1. Access the response with dot-notation: response.content
      2. Keep the return type honest (a real str, matching -> str)
      3. Wrap the user prompt in a HumanMessage, the canonical
         LangChain chat-model input shape used in the official docs
         (https://python.langchain.com/docs/tutorials/)
    """
    api_key = os.getenv("OPENAI_API_KEY")
    llm = ChatOpenAI(openai_api_key=api_key)

    # Chat models expect a message (or list of messages), not a bare string.
    response = llm.invoke([HumanMessage(content=prompt)])

    # The AIMessage object holds the text in `.content`.
    return response.content


if __name__ == "__main__":
    user_prompt = "How can LangSmith help with testing?"
    answer = run_agent(user_prompt)
    print("LLM response:", answer)
