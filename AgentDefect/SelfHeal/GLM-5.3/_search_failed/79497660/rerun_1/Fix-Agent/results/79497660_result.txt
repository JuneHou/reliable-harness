"""
Fix: store `external_id` in a dedicated column of the PostgreSQL table used by
LlamaIndex's PGVectorStore instead of only inside the `metadata_` JSON column.

The stock PGVectorStore generates the table schema itself (id, node_id,
source_node_id, source_text, metadata_, embedding, text_) and serializes every
document metadata field into the `metadata_` JSON column.  To get custom
columns into the table schema we subclass PGVectorStore and

  * extend the generated table model with the columns configured in
    CUSTOM_COLUMNS (`external_id` here), and
  * write the values of those columns ourselves in `add()`.

An `ALTER TABLE ... ADD COLUMN IF NOT EXISTS` migration also takes care of
tables that were already created by a previous run of the script.
"""

import os
from typing import Any, Dict, List

import psycopg2
from dotenv import load_dotenv
from sqlalchemy import Column, Text, text as sa_text
from sqlalchemy.dialects.postgresql import insert as pg_insert

from llama_index.core import VectorStoreIndex, Document
from llama_index.core.schema import BaseNode, MetadataMode
from llama_index.core.vector_stores.utils import node_to_metadata_dict
from llama_index.vector_stores.postgres import PGVectorStore

load_dotenv()

# Metadata keys that are promoted to dedicated columns of the Postgres table
# instead of living only inside the `metadata_` JSON column.
CUSTOM_COLUMNS: Dict[str, Any] = {
    "external_id": Text,
}


def _append_custom_columns(table: Any) -> None:
    """Append the custom columns to an (already generated) SQLAlchemy table."""
    for name, type_ in CUSTOM_COLUMNS.items():
        if name not in table.columns:
            table.append_column(Column(name, type_))


def _sql_type_name(type_: Any) -> str:
    """SQL type name used in the ALTER TABLE statements."""
    try:
        return str(type_()) if isinstance(type_, type) else str(type_)
    except Exception:
        return "TEXT"


class CustomPGVectorStore(PGVectorStore):
    """PGVectorStore with custom columns (e.g. `external_id`) in the table schema.

    * `_get_table_class` extends the table model generated by PGVectorStore, so
      freshly created tables contain the custom columns.
    * `__init__` also extends the model for versions that build it through a
      module level helper, and migrates already existing tables with
      `ALTER TABLE ... ADD COLUMN IF NOT EXISTS`.
    * `add` writes the value of every custom column when nodes are inserted.
    """

    def _get_table_class(self) -> Any:
        """Extend the table model generated by PGVectorStore (older versions)."""
        table_class = super()._get_table_class()
        _append_custom_columns(table_class.__table__)
        return table_class

    def __init__(self, *args: Any, **kwargs: Any) -> None:
        super().__init__(*args, **kwargs)

        # Newer versions of the integration build the table model through a
        # module level helper instead of `_get_table_class`; extend the model
        # here as well (idempotent thanks to the guard in _append_custom_columns).
        _append_custom_columns(self._table_class.__table__)

        # Make sure the custom columns exist in the database as well: either
        # the table was just created without them, or it already existed from
        # an earlier run of the original script.
        self._ensure_custom_columns()

    def _sql_engine(self) -> Any:
        """Return the sync SQLAlchemy engine used by the store."""
        return getattr(self, "_engine", None) or getattr(self, "_sync_engine", None)

    def _qualified_table(self) -> str:
        """Return the schema-qualified name of the data table."""
        table = self._table_class.__table__
        schema = table.schema or getattr(self, "schema_name", None) or "public"
        return '"{}"."{}"'.format(schema, table.name)

    def _ensure_custom_columns(self) -> None:
        """Add the custom columns to the table if they do not exist yet."""
        engine = self._sql_engine()
        if engine is None:
            return
        qualified = self._qualified_table()
        try:
            with engine.begin() as connection:
                for name, type_ in CUSTOM_COLUMNS.items():
                    connection.execute(
                        sa_text(
                            "ALTER TABLE {} ADD COLUMN IF NOT EXISTS {} {}".format(
                                qualified, name, _sql_type_name(type_)
                            )
                        )
                    )
        except Exception as exc:  # best effort migration
            print("Warning: could not add custom columns: {}".format(exc))

    def add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        """Add nodes, writing the custom metadata keys to their own columns."""
        session_factory = getattr(self, "_session", None) or getattr(
            self, "_sync_session", None
        )
        if session_factory is None:
            raise AttributeError(
                "No SQLAlchemy session factory found on the vector store."
            )

        ids: List[str] = []
        with session_factory() as session:
            for node in nodes:
                ids.append(node.node_id)

                embedding = node.get_embedding()

                # Values for the dedicated columns, taken from the metadata.
                custom_values = {
                    name: node.metadata.get(name) for name in CUSTOM_COLUMNS
                }

                stmt = pg_insert(self._table_class).values(
                    node_id=node.node_id,
                    embedding=embedding,
                    metadata_=node_to_metadata_dict(
                        node, remove_text=True, flat_metadata=self.flat_metadata
                    ),
                    text_=node.get_content(metadata_mode=MetadataMode.NONE),
                    # `external_id` and friends go into their own columns.
                    **custom_values,
                )
                if getattr(self, "returns_id", False):
                    stmt = stmt.returning(self._table_class.id)
                cursor = session.execute(stmt)
                if getattr(self, "returns_id", False):
                    ids[-1] = cursor.scalar_one_or_none()

            session.commit()

        return ids


def create_vector_store(request):
    host = "localhost"
    password = "password"
    port = 5432
    user = "postgres"

    # Use the custom store so that `external_id` becomes a real column of the
    # generated table instead of a key inside the `metadata_` JSON column.
    vector_store = CustomPGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name="vectorstore",
        embed_dim=1536,
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
    )

    return vector_store


def run_agent(request):
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )

    return index


def verify_external_id_column(request):
    """Verify that `external_id` is stored in its own column of the table."""
    conn = psycopg2.connect(
        host="localhost",
        port=5432,
        dbname=request.tenant,
        user="postgres",
        password="password",
    )
    try:
        with conn.cursor() as cursor:
            # LlamaIndex prefixes the table name with "data_" by default.
            cursor.execute(
                "SELECT table_name FROM information_schema.tables "
                "WHERE table_schema = 'public' "
                "AND table_name IN ('data_vectorstore', 'vectorstore')"
            )
            rows = cursor.fetchall()
            if not rows:
                print("No vector store table found.")
                return
            table = rows[0][0]
            cursor.execute('SELECT external_id FROM "public"."{}"'.format(table))
            values = [row[0] for row in cursor.fetchall()]
            print("external_id column in {}: {}".format(table, values))
    finally:
        conn.close()
