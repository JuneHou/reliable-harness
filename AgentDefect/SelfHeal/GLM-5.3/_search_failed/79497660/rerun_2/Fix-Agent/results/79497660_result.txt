from llama_index.core import VectorStoreIndex, Document
from llama_index.vector_stores.postgres import PGVectorStore
from llama_index.core.vector_stores.types import Column, VectorDataType
import psycopg2
import os
from dotenv import load_dotenv

load_dotenv()

HOST = "localhost"
PASSWORD = "password"
PORT = 5432
USER = "postgres"
TABLE_NAME = "vectorstore"


def _drop_existing_table(request):
    """
    Drop any table left over from previous runs.

    Custom metadata columns are ONLY created when the table itself is created:
    PGVectorStore builds the table with SQLAlchemy's create_all(), which never
    ALTERs an already existing table. So if `data_vectorstore` was previously
    created without the `external_id` column, simply passing `metadata_cols`
    would have no effect. Dropping the table first guarantees the table is
    (re)created with the custom column in its schema.
    """
    conn = psycopg2.connect(
        dbname=request.tenant,
        user=USER,
        password=PASSWORD,
        host=HOST,
        port=PORT,
    )
    conn.autocommit = True
    with conn.cursor() as cursor:
        cursor.execute(f"DROP TABLE IF EXISTS public.data_{TABLE_NAME} CASCADE")
        cursor.execute(f'DROP TABLE IF EXISTS public."{TABLE_NAME}" CASCADE')
    conn.close()


def create_vector_store(request):
    # Make sure the table is (re)created with the custom column present.
    _drop_existing_table(request)

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=HOST,
        password=PASSWORD,
        port=PORT,
        user=USER,
        table_name=TABLE_NAME,
        embed_dim=1536,
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
        # THE FIX: declare custom columns that are added directly to the
        # PostgreSQL table schema instead of only living inside the metadata
        # JSON column. For every Column listed here, PGVectorStore:
        #   1. adds a real SQL column to the `data_<table_name>` table during
        #      table creation, and
        #   2. populates it from each node's metadata using the same key
        #      (here: "external_id").
        metadata_cols=[
            Column(
                "external_id",
                VectorDataType.TEXT,
                nullable=True,
            )
        ],
    )

    return vector_store


def run_agent(request):
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store
    )

    return index


def verify_external_id_column(request):
    """Confirm that external_id now lives in a dedicated table column."""
    conn = psycopg2.connect(
        dbname=request.tenant,
        user=USER,
        password=PASSWORD,
        host=HOST,
        port=PORT,
    )
    try:
        with conn.cursor() as cursor:
            cursor.execute(
                """
                SELECT column_name, data_type
                FROM information_schema.columns
                WHERE table_schema = 'public'
                  AND table_name = %s
                  AND column_name = 'external_id'
                """,
                (f"data_{TABLE_NAME}",),
            )
            row = cursor.fetchone()

            if row is None:
                print("FAILED: 'external_id' column does not exist in the table")
                return

            print(
                "OK: 'external_id' exists as a separate column "
                f"(type: {row[1]}) in table data_{TABLE_NAME}"
            )

            cursor.execute(f"SELECT external_id FROM data_{TABLE_NAME}")
            for (external_id,) in cursor.fetchall():
                print(f"  external_id column value: {external_id}")
    finally:
        conn.close()


if __name__ == "__main__":
    class Request:

    verify_external_id_column(request)
