import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()

PERSIST_DIRECTORY = "./chroma_db"
COLLECTION_NAME = "paper2"


def build_vectorstore():
    """Load, split, and index the document exactly ONCE.

    Original bug: Chroma.from_documents() was called inside run_agent(), so
    every query re-loaded the PDF, re-split it, and re-inserted ALL chunks
    into the same default collection. The collection kept growing with
    duplicate copies of every chunk, so the retriever's top-k results were
    dominated by identical documents.

    Fix: build the index once (module scope), persist it to disk under a
    named collection, and only ingest when the collection is empty.
    """
    embedding = OpenAIEmbeddings()

    # Persistent, named collection instead of an anonymous in-memory one.
    vectorstore = Chroma(
        collection_name=COLLECTION_NAME,
        embedding_function=embedding,
        persist_directory=PERSIST_DIRECTORY,
    )

    # Ingest only if the collection is empty -> prevents re-insertion on rerun.
    existing_ids = vectorstore.get().get("ids", [])
    if not existing_ids:
        loader = UnstructuredFileLoader('paper2.pdf')
        docs = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=1000, chunk_overlap=200, add_start_index=True
        )
        all_splits = text_splitter.split_documents(docs)

        # Deterministic IDs: even if ingestion ran again, Chroma would
        # upsert (replace) these IDs instead of appending duplicates.
        ids = [f"paper2-chunk-{i}" for i in range(len(all_splits))]
        vectorstore.add_documents(all_splits, ids=ids)

    return vectorstore


def deduplicate_documents(docs):
    """Safety net: keep only unique documents in the retrieval results."""
    seen = set()
    unique_docs = []
    for doc in docs:
        key = (doc.page_content, str(doc.metadata.get("start_index")))
        if key not in seen:
            seen.add(key)
            unique_docs.append(doc)
    return unique_docs


# Built once at the right scope (right frequency), NOT on every query.
VECTORSTORE = build_vectorstore()
RETRIEVER = VECTORSTORE.as_retriever(
    search_type="similarity", search_kwargs={"k": 6}
)


def run_agent(query):
    retrieved_docs = RETRIEVER.get_relevant_documents(query)
    # Guarantee unique documents are returned.
    retrieved_docs = deduplicate_documents(retrieved_docs)
    return retrieved_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
