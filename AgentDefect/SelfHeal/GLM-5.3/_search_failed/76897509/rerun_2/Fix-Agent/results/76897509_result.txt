import os
import random
from typing import Any, Dict
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.vectorstores.base import VectorStoreRetriever
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]


def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


class ConversationalRetrievalChainWithFilter(ConversationalRetrievalChain):
    """
    A ConversationalRetrievalChain that accepts a metadata 'filter' input key
    at query time and applies it to the retriever's search_kwargs, so the same
    chain instance can be reused with different metadata filters.

    Vanilla ConversationalRetrievalChain only understands 'question' and
    'chat_history' input keys; a 'filter' key is silently ignored, which made
    retrieval run over ALL documents instead of the requested subset.
    """

    def _call(self, inputs: Dict[str, Any], *args: Any, **kwargs: Any) -> Dict[str, Any]:
        # Remove the filter key so it is not treated as an unexpected prompt
        # variable by the condense-question / QA sub-chains.
        metadata_filter = inputs.pop("filter", None)

        if isinstance(metadata_filter, dict) and metadata_filter:
            # Inject the filter into the retriever's search kwargs; the
            # VectorStoreRetriever forwards search_kwargs to Chroma's
            # similarity_search, applying the metadata filter at query time.
            self.retriever.search_kwargs["filter"] = metadata_filter
        else:
            # No filter supplied -> retrieve over the whole collection.
            self.retriever.search_kwargs.pop("filter", None)

        return super()._call(inputs, *args, **kwargs)


def run_agent(query, pdf_paths):
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            text.metadata["source"] = pdf_path

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    retriever = VectorStoreRetriever(vectorstore=db)

    qa = ConversationalRetrievalChainWithFilter.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever,
        verbose=True,
        return_source_documents=True)

    filter = {'source': 'paper2.pdf'}  # it means the information from paper2.pdf should be used to answer the query.
    # if it answers from paper2.pdf, then the filtering is working correctly.

    # The 'filter' key is now consumed by ConversationalRetrievalChainWithFilter
    # and applied to the retriever before the documents are fetched, so only
    # chunks from paper2.pdf are retrieved and used to build the answer.
    result = qa({"question": query, "chat_history": [], "filter": filter})

    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    result = run_agent(query, pdf_paths)

    print(result['answer'])

    # Optional sanity check: verify the retrieved sources actually come from
    # the filtered document (paper2.pdf).
    sources = {doc.metadata.get('source') for doc in result.get('source_documents', [])}
    print(f"\n[Filter check] sources used: {sources}")
