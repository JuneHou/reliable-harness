import pandas as pd
from dotenv import load_dotenv

from langchain.chat_models import ChatOpenAI

try:
    from langchain.agents import create_pandas_dataframe_agent
except ImportError:
    # newer LangChain versions moved the pandas agent to langchain_experimental
    from langchain_experimental.agents import create_pandas_dataframe_agent

load_dotenv()


def run_agent(prompt):
    """
    Answer questions about the data stored in movies.csv.

    Why the previous version only ever returned one author:
      * CSVLoader() turns EVERY row of the CSV into its own Document.
      * The retriever was built with search_kwargs={"k": 1}, so only the single
        most similar row was handed to the LLM -- hence "author A" and nothing
        else, no matter how many authors actually have films from 1999.
      * Even with a larger k, similarity search returns only the rows closest
        to the question, never *all* rows matching a filter, so exhaustive
        questions ("which authors have films from 1999" / "which movies are
        action") could never be answered completely row-by-row.

    Fix:
      * Load the whole CSV into a pandas DataFrame and let the LLM run pandas
        code over it (e.g. df[df['year'] == 1999]).  Every row is visible at
        once, so the answer contains ALL matching authors/movies.
      * (If you keep the vector-store approach, you must at least raise k,
        e.g. search_kwargs={"k": 20}, but the DataFrame agent is the reliable
        way to answer filter/aggregation questions over a CSV.)
    """
    # temperature=0 -> deterministic, complete answers instead of a single
    # creative guess.
    llm = ChatOpenAI(temperature=0)

    # The entire table is available to the model at once.
    df = pd.read_csv("movies.csv")

    agent = create_pandas_dataframe_agent(
        llm,
        df,
        verbose=True,
        handle_parsing_errors=True,
    )

    return agent.run(prompt)


if __name__ == '__main__':
    print(run_agent("which movies are action"))
