from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer
import os


def create_embedding_model(_chunk_size=None):
    print('loading embeddings...')
    if os.path.exists('./embeddings/models--BAAI--bge-m3'):
        _emb_model_name = "BAAI/bge-m3"
        _embed_model = HuggingFaceEmbedding(model_name=_emb_model_name, max_length=_chunk_size,
                                            cache_folder='./embeddings')
    else:
        os.makedirs("./embeddings", exist_ok=True)
        _emb_model_name = "BAAI/bge-m3"
        _embed_model = HuggingFaceEmbedding(model_name=_emb_model_name, max_length=_chunk_size,
                                            cache_folder='./embeddings')
    print('embeddings loaded')
    return _embed_model


def count_tokens_without_embedding(text: str, tokenizer) -> int:
    """
    Replicates the token count that the embedding process reports for
    `text` WITHOUT actually running the embedding model.

    The embedding callback counts tokens with exactly this logic: the
    number of input_ids produced by the tokenizer that is configured on
    the TokenCountingHandler (special tokens such as <s>/</s> included).
    """
    return len(tokenizer(text)["input_ids"])


def run_agent(text: str):
    chunk_size = 512

    # Load the tokenizer of the embedding model FIRST.
    model_name = "BAAI/bge-m3"
    tokenizer = AutoTokenizer.from_pretrained(model_name)

    embed_model = create_embedding_model(_chunk_size=chunk_size)

    # ------------------------------------------------------------------
    # ROOT CAUSE OF THE MISMATCH
    #
    # TokenCountingHandler does NOT use the embedding model's tokenizer.
    # When no tokenizer is supplied it falls back to llama_index's
    # MockTokenizer, which simply chops the text into 4-character chunks
    # (i.e. ceil(len(text) / 4) "tokens").  That is why the reported
    # embedding token count was LOWER than the real BGE-M3 token count
    # produced by AutoTokenizer (which also emits the <s>/</s> special
    # tokens).
    #
    # FIX: pass the real BGE-M3 tokenizer to the handler so that it
    # counts tokens exactly like the model does.  The handler computes
    # len(self.tokenizer(text)), so the callable must return the list of
    # input_ids -- handing over the HF tokenizer object directly would
    # make len() return the number of dict keys (e.g. 2) instead of the
    # number of tokens.
    # ------------------------------------------------------------------
    token_counter = TokenCountingHandler(
        tokenizer=lambda t: tokenizer(t)["input_ids"]
    )
    callback_manager = CallbackManager([token_counter])

    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    embedding_vector = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Same counting logic the handler now uses, so this reproduces the
    # embedding token count exactly -- computed WITHOUT running the
    # embedding, which is what we wanted to replicate.
    token_count = count_tokens_without_embedding(text, tokenizer)
    return embedding_tokens, token_count

if __name__ == "__main__":

    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
