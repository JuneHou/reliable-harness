import os
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import FAISS
from langchain.chains import ConversationalRetrievalChain
from langchain.chains import LLMChain
from langchain.memory import ConversationBufferMemory
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate

# Load environment variables from .env file
_ = load_dotenv(find_dotenv())


def run_agent(user_input):
    openai_api_key = os.getenv('OPENAI_API_KEY')

    os.environ['OPENAI_API_KEY'] = openai_api_key

    custom_template = """
This is conversation with a human. Answer the questions you get based on the knowledge you have.
If you don't know the answer, just say that you don't, don't try to make up an answer.
Chat History:
{chat_history}
Follow Up Input: {question}
"""
    CUSTOM_QUESTION_PROMPT = PromptTemplate.from_template(custom_template)
    llm = ChatOpenAI(
        model_name="gpt-3.5-turbo",  # Name of the language model
        temperature=0  # Parameter that controls the randomness of the generated responses
    )
    embeddings = OpenAIEmbeddings()
    docs = [
        "Buildings are made out of brick",
        "Buildings are made out of wood",
        "Buildings are made out of stone",
        "Buildings are made out of atoms",
        "Buildings are made out of building materials",
        "Cars are made out of metal",
        "Cars are made out of plastic",
    ]
    vectorstore = FAISS.from_texts(docs, embeddings)
    retriever = vectorstore.as_retriever()
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)

    # return_source_documents=True makes the chain expose the documents that
    # were actually used to produce the answer, instead of blindly calling
    # retriever.get_relevant_documents() (which always returns the top-k most
    # similar docs, even for greetings / unrelated queries).
    qa = ConversationalRetrievalChain.from_llm(
        llm,
        retriever,
        condense_question_prompt=CUSTOM_QUESTION_PROMPT,
        memory=memory,
        return_source_documents=True
    )

    # Use the real user input (the buggy version hard-coded
    # "what are cars made of?", so the printed question never matched
    # the documents that were returned).
    result = qa({"question": user_input})
    answer = result["answer"]
    source_documents = result.get("source_documents", [])

    if not answer.strip():
        # No meaningful answer produced -> nothing came from the knowledge base
        return []

    # Similarity-score thresholds proved unreliable for telling real
    # knowledge-base questions apart from greetings / small talk. Instead, let
    # the LLM decide whether the answer it just produced is actually grounded
    # in the custom knowledge base documents.
    relevance_prompt = PromptTemplate.from_template(
        """You are given a user question, a chatbot answer, and the documents of a custom knowledge base.
Decide whether the chatbot's answer is based on the knowledge base documents.
If the answer is grounded in the documents, respond with only "yes".
If the answer is a greeting, small talk, or was made up without using the documents, respond with only "no".

Question: {question}
Answer: {answer}
Documents:
{documents}

Is the answer based on the documents? (yes/no):"""
    )
    relevance_chain = LLMChain(llm=llm, prompt=relevance_prompt)
    documents_text = "\n".join(
        f"- {doc.page_content}" for doc in source_documents
    ) or "- (no documents)"

    verdict = relevance_chain.run(
        question=user_input,
        answer=answer,
        documents=documents_text
    ).strip().lower()

    # Relevant documents are returned only when the bot's answer really comes
    # from the custom knowledge base; otherwise (greetings, chit-chat, or
    # "I don't know" answers) an empty list is returned.
    if verdict.startswith("y"):
        return source_documents
    return []


if __name__ == "__main__":
    # Related query: the answer comes from the knowledge base,
    # so the relevant source documents are returned.
    user_input = "What are buildings made of?"
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result}")

    # Unrelated query (a greeting): nothing comes from the knowledge base,
    # so an empty list is returned instead of the "most similar" documents.
    greeting = "Hello there!"
    greeting_result = run_agent(greeting)
    print("\n=== Unrelated Query Result ===")
    print(f"Question: {greeting}")
    print(f"Answer: {greeting_result}")
