from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node with None in metadata
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata={"start_date": None},
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # ------------------------------------------------------------------
    # WHY THE ORIGINAL CODE FAILED
    # ------------------------------------------------------------------
    # The original filter was:
    #
    #     MetadataFilter(
    #         key="start_date",
    #         operator=FilterOperator.EQ,
    #         value=None,
    #     )
    #
    # `FilterOperator.EQ` requires a concrete, comparable value
    # (str/int/float/list).  When `value=None` is passed, there is nothing
    # to compare against: LlamaIndex treats a `None` stored in a node's
    # metadata as an "empty"/null value rather than as a regular value that
    # can be equality-matched, so `EQ` with `None` silently returns zero
    # results even for nodes whose metadata field literally is `None`.
    # (Most vector store backends behave the same way, and several even
    # drop metadata keys whose value is None.)  In short: `None` is not a
    # filterable value for the equality operator.
    #
    # THE RECOMMENDED APPROACH
    # ------------------------
    # Use the dedicated `FilterOperator.IS_EMPTY` operator, which LlamaIndex
    # provides exactly for null/empty metadata filtering.  It matches nodes
    # whose metadata value for the key is None (or an empty string, empty
    # list, or empty dict).
    #
    # ALTERNATIVE WORKAROUND (if you must stick to `EQ`, e.g. because your
    # vector store does not support IS_EMPTY): store an explicit sentinel
    # string instead of a real None in the metadata and filter on it, e.g.
    #
    #     metadata={"start_date": "null"}
    #     MetadataFilter(key="start_date",
    #                    operator=FilterOperator.EQ,
    #                    value="null")
    # ------------------------------------------------------------------
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.IS_EMPTY,
        value=None,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes
