from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# ---------------------------------------------------------------------------
# Why the original code did not work:
#
# LlamaIndex metadata filters cannot match Python `None`.  Metadata filter
# values have to be concrete scalars (str/int/float/bool), so a
# `MetadataFilter(key=..., operator=FilterOperator.EQ, value=None)` never
# matches a node whose metadata field is None.  The node is stored fine, but
# the filter silently excludes it, so the retriever returns nothing.
#
# Recommended approach / workaround:
#
# Do not store `None` as a metadata value.  Normalize null values to a
# sentinel string when building nodes, and filter on that sentinel.  The
# helpers below make the conversion explicit and reversible.
# ---------------------------------------------------------------------------

NULL_SENTINEL = "None"


def to_sentinel(value):
    """Normalize a null (None) metadata value into a filterable sentinel."""
    return NULL_SENTINEL if value is None else value


def from_sentinel(value):
    """Convert the sentinel back into None when reading a metadata value."""
    return None if value == NULL_SENTINEL else value


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node with the sentinel string instead of Python None.
    # ("null" is stored as "None" so that it is actually filterable.)
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata={"start_date": to_sentinel(None)},
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter on the sentinel value (NOT on Python None)
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved (None is not filterable)")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
        # Recover the original "null" semantics for downstream use:
        start_date = from_sentinel(response[0].metadata["start_date"])
        print(f"start_date (sentinel converted back): {start_date!r}")
