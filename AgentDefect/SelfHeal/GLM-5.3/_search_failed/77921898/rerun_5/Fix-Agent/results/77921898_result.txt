import os
from dotenv import load_dotenv, find_dotenv
from glob import glob
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader


_ = load_dotenv(find_dotenv())


def run_agent(user_input, document_paths=None, persist_directory="chroma_db"):
    """
    Index the given documents into a persisted Chroma vector store (if any
    paths are supplied) and retrieve the most relevant documents for the
    user's query.

    Key fix:
    --------
    Previously the persisted store was re-opened with:

        docsearch = Chroma(persist_directory, embedding_function=embeddings)

    The FIRST POSITIONAL argument of the Chroma constructor is
    ``collection_name``, NOT ``persist_directory``.  That call therefore
    created a brand-new, empty collection literally *named* "chroma_db"
    (with no persistence directory configured), so the retriever queried a
    collection that never contained any documents -> 0 documents retrieved,
    while everything else in the pipeline still "worked".

    The fix is to pass the directory as a keyword argument
    (``persist_directory=...``) and to use the SAME collection name in both
    ``Chroma.from_documents`` and the reload, so the persisted vectors and
    their source metadata are actually read back.
    """
    openai_api_key = os.getenv('OPENAI_API_KEY')
    if not openai_api_key:
        raise ValueError(
            "OPENAI_API_KEY not found in environment variables. "
            "Set it in your .env file before running the agent."
        )

    os.environ['OPENAI_API_KEY'] = openai_api_key

    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    # Use one consistent collection name for both writing and reading.
    collection_name = "langchain_docs"

    if document_paths:
        all_texts = []
        for doc_path in document_paths:
            loader = TextLoader(doc_path)
            documents = loader.load()
            text_splitter = RecursiveCharacterTextSplitter(
                chunk_size=1000,
                chunk_overlap=200
            )
            texts = text_splitter.split_documents(documents)
            all_texts.extend(texts)

        if not all_texts:
            raise ValueError(
                f"No text could be extracted from the provided documents: "
                f"{document_paths}"
            )

        # Persist the index to disk under `persist_directory`.
        docsearch = Chroma.from_documents(
            all_texts,
            embeddings,
            collection_name=collection_name,
            persist_directory=persist_directory,
        )
    else:
        # Precondition check: without documents and without an existing
        # persisted store there is nothing to retrieve from.
        if not os.path.isdir(persist_directory):
            raise FileNotFoundError(
                f"No documents were provided and no existing vector store "
                f"was found at '{persist_directory}'. Provide "
                f"document_paths or a valid persist_directory."
            )

    # Re-open the persisted store.
    # `persist_directory` MUST be passed as a keyword argument, because the
    # first positional parameter of Chroma() is `collection_name`.
    docsearch = Chroma(
        collection_name=collection_name,
        persist_directory=persist_directory,
        embedding_function=embeddings,
    )

    retriever = docsearch.as_retriever(search_kwargs={"k": 5})
    docs = retriever.get_relevant_documents(user_input)
    print(f"Number of documents retrieved: {len(docs)}")

    if not docs:
        print(
            "Warning: no documents were retrieved. Verify that the persist "
            "directory contains data for collection "
            f"'{collection_name}' and that the same embedding model is used."
        )

    return docs


if __name__ == "__main__":

    document_paths = glob("docs/*.txt")
    if not document_paths:
        raise FileNotFoundError(
            "No .txt files found in ./docs — there is nothing to index. "
            "Add documents before running the agent."
        )

    query = "What is machine learning?"
    docs = run_agent(query, document_paths)

    print("\n=== User Query Result ===")
    print(f"Query: {query}")
    print(f"Number of relevant documents: {len(docs)}")

    for i, doc in enumerate(docs):
        print(f"\n--- Document {i+1} ---")
        print(f"Metadata: {doc.metadata}")
        print(doc.page_content)
