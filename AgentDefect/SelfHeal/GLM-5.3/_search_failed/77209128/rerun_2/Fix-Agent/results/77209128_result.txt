import os
import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer
from llama_index.core import SQLDatabase, VectorStoreIndex, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

load_dotenv()

OPENAI_API_KEY = os.environ.get('OPENAI_API_KEY')
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

pg_uri = os.environ.get('DATABASE_URL', 'sqlite:///test.db')

# Field-level metadata: a JSON dictionary mapping each key inside the
# table's 'data' JSON column to a human-readable description.
metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
}


class SqlQueryEngine:

    def __init__(self):
        self.schema = []

    def query_engine(self):
        engine = create_engine(pg_uri)
        metadata_obj = MetaData()

        # Convert the metadata dictionary to a JSON string
        metadata_json = json.dumps(metadata_dict, indent=2)

        # create snapshot SQL table
        table_name = "snapshots"
        snapshots_table = Table(
            table_name,
            metadata_obj,
            Column("id", Integer, primary_key=True),
            # NOTE: this SQLAlchemy column comment is purely DB-level
            # documentation. LlamaIndex does NOT read column comments, so
            # this alone never makes the metadata available to the
            # query engine (that was the bug).
            Column("data", JSON, comment=metadata_json),
        )

        metadata_obj.create_all(engine)

        sql_database = SQLDatabase(engine)
        # SQLTableNodeMapping converts SQLTableSchema objects into index nodes.
        table_node_mapping = SQLTableNodeMapping(sql_database)

        # ------------------------------------------------------------------
        # THE FIX: attach the metadata to the table via
        # SQLTableSchema.context_str (not via SQLAlchemy column comments).
        #
        # - SQLTableNodeMapping.to_node() embeds `context_str` into the node
        #   text ("Context of table X: ..."), so the VectorStoreIndex / object
        #   retriever can match queries against the field descriptions.
        # - SQLTableRetrieverQueryEngine._get_table_context() appends
        #   "The table description is: <context_str>" to the schema handed to
        #   the LLM, so the generated SQL can use the field meanings.
        # ------------------------------------------------------------------
        table_context_dict = {
            table_name: (
                f"The '{table_name}' table stores one row per snapshot. The "
                "'data' column is a JSON dictionary where each key is a metric "
                "field name and each value is the measurement for that field. "
                "The meaning of each field inside the 'data' JSON column is "
                "described below:\n" + metadata_json
            )
        }

        self.schema = []
        for tbl_name in metadata_obj.tables.keys():
            self.schema.append(
                SQLTableSchema(
                    table_name=tbl_name,
                    context_str=table_context_dict.get(tbl_name),
                )
            )

        llm = OpenAI(temperature=0, model="gpt-3.5-turbo")
        Settings.llm = llm

        # We dump the table schema information (now including the metadata
        # context) into a vector index. The vector index is stored
        # within the context builder for future use.
        index = ObjectIndex.from_objects(
            self.schema, table_node_mapping, VectorStoreIndex
        )

        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
            llm=llm,
        )

        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response


if __name__ == "__main__":

    query = "What does the page_call_phone_clicks_logged_in_unique field mean in the data column?"

            print("  Status: Metadata context is SET")
        else:
            print("  context_str: None")
            print("  Status: Metadata context is NOT SET")
