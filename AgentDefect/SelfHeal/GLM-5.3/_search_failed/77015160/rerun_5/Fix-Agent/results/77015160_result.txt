from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def run_agent(question: str = "Which countries are mentioned in the documents?"):
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # Custom, blank prompt template.
    # This replaces the built-in default template of the "stuff"
    # QA-with-sources chain, which ships with pre-filled few-shot example
    # content (e.g. the "Huey, Dewey, and Louie" example).
    #
    # IMPORTANT: for chain_type="stuff" the prompt MUST declare exactly these
    # input variables: "summaries" (the retrieved documents are stuffed here)
    # and "question" (the user's query). Mismatched/missing variables cause
    # the chain to fall back to or fail against the default template.
    custom_prompt = PromptTemplate(
        input_variables=["summaries", "question"],
        template=(
            "You are an assistant for question-answering tasks.\n"
            "Use the following pieces of retrieved context to answer the "
            "question. If you don't know the answer, just say that you "
            "don't know, don't try to make up an answer.\n\n"
            "Context:\n{summaries}\n\n"
            "Question: {question}\n"
            "Helpful Answer:"
        ),
    )

    # Pass the custom prompt through chain_type_kwargs so it overrides the
    # chain's built-in default prompt.
    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        chain_type_kwargs={"prompt": custom_prompt},
        return_source_documents=False,
    )

    # Actually invoke the chain with a question.
    # The original code printed the chain object itself, which only renders
    # its repr -- i.e. the default template with pre-filled example content --
    # instead of running the chain and returning an answer with sources.
    result = chain.invoke({"question": question})
    return result


if __name__ == "__main__":
    output = run_agent()
    print("Question:", output.get("question", ""))
    print("Answer:", output.get("answer", ""))
    print("Sources:", output.get("sources", ""))
