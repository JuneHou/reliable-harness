import random

from langchain_ollama import ChatOllama


def run_agent(input_text):
    # WHY THE ORIGINAL CODE RETURNED IDENTICAL OUTPUT EVERY TIME:
    #
    # temperature / top_p / repeat_penalty only reshape the probability
    # distribution that tokens are *sampled* from -- they do NOT make the
    # sampling process itself random. Ollama drives its token-sampling RNG
    # with the `seed` option, and with the default (fixed) seed the very same
    # prompt plus the very same options always draws the exact same token
    # sequence, so the model returns a byte-for-byte identical answer on every
    # run no matter how high temperature is set. (langchain_ollama documents
    # `seed` as: "Setting the same seed will make the model generate the same
    # text for the same prompt. (Default: 0)".)
    #
    # FIX: draw a fresh random seed for every request so each call samples
    # independently from the temperature-adjusted distribution, which yields
    # varied, non-deterministic answers. (If you ever need reproducibility
    # instead, pin the seed to a constant.)
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        seed=random.randint(1, 2**31 - 1),  # NEW: random seed per request
        timeout=120,
    )
    response = llm.invoke(input_text)
    return response.content
