import random

from langchain_ollama import ChatOllama


def run_agent(input_text):
    """
    Ask Ollama (via LangChain) a question and return the answer.

    Root cause of the original bug:
    -------------------------------
    temperature / top_p / repeat_penalty only shape the *sampling
    distribution* -- they do not control the randomness source. Ollama
    derives its sampling RNG from the `seed` option, and when that seed is
    fixed (or left at Ollama's default), generation is reproducible: the
    exact same prompt always produced the exact same tokens, which is why
    the model appeared to "ignore" the temperature.

    Fix:
    ----
    Explicitly pass a fresh, random `seed` on every request. Each call now
    gets a different RNG state, so the stochastic sampling parameters
    actually vary the output and the answers become non-deterministic.
    (A random per-call seed is portable: it does not depend on the Ollama
    server version's default-seed behaviour.)
    """
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        # NEW: per-call random seed -> non-deterministic sampling.
        # Must be a fresh value each run; a constant seed (e.g. seed=42)
        # would reintroduce the identical-response behaviour.
        seed=random.randint(0, 2**31 - 1),
        timeout=120,
    )
    response = llm.invoke(input_text)
    return response.content


if __name__ == "__main__":
    prompt = "Please tell me a lighthearted joke"

    # Run the same prompt several times to demonstrate that the answers
    # now vary instead of repeating verbatim.
    for attempt in range(1, 4):
        print(f"[run {attempt}] {run_agent(prompt)}")
        print("-" * 70)
