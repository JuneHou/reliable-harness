"""
Local RAG pipeline: chat with a local PDF using Ollama, LangChain and Chroma.

Why the original script "silently stopped" after the embedding step
-------------------------------------------------------------------
The nested ``try/except`` around the vector-store creation swallowed the real
error (typically the Ollama server not running, or the ``nomic-embed-text``
model not being pulled).  Because that exception was only logged and never
re-raised, ``vector_db`` was never assigned, so the very next statement
(``if vector_db:``) raised ``UnboundLocalError`` -- which was in turn caught
and swallowed by the outer ``except``.  The function then quietly returned
``None`` and the program ended with ``Final Output: None`` instead of an
answer, without any traceback.  This is why it looked like the script just
"stopped" after the embedding step without throwing any errors.

Fixes applied here:
  1. No exception is silently swallowed any more; failures are logged with
     their full traceback (``logger.exception``).
  2. ``vector_db`` is initialised up-front (no UnboundLocalError) and its
     presence is verified before it is used.
  3. Explicit precondition checks validate the external resources this script
     depends on (PDF file present, Ollama server reachable, required models
     available) *before* any expensive work runs, and missing models are
     pulled automatically when possible -- otherwise a clear, actionable
     error is raised (e.g. ``ollama serve`` / ``ollama pull ...``).
  4. ``logging.basicConfig(..., force=True)`` guarantees log messages are
     actually emitted even if another library already touched the root logger.
"""

import json
import logging
import os
import urllib.request

from langchain_community.document_loaders import UnstructuredPDFLoader
from langchain_community.embeddings import OllamaEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain.prompts import ChatPromptTemplate, PromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_community.chat_models import ChatOllama
from langchain_core.runnables import RunnablePassthrough
from langchain.retrievers.multi_query import MultiQueryRetriever

# ---------------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------------
PDF_PATH = "./paper1.pdf"
EMBEDDING_MODEL = "nomic-embed-text"
LLM_MODEL = "llama3"

# Make sure logging is really configured even if another library already
# installed handlers on the root logger (otherwise failures stay invisible).
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s - %(levelname)s - %(message)s",
    force=True,
)
logger = logging.getLogger(__name__)


def _ollama_base_url():
    """Return the Ollama endpoint that the LangChain clients will use."""
    base_url = os.environ.get("OLLAMA_HOST", "http://localhost:11434")
    if not base_url.startswith(("http://", "https://")):
        base_url = "http://" + base_url
    return base_url.rstrip("/")


def _ollama_request(path, payload=None, timeout=10):
    """Minimal helper for talking to the Ollama REST API."""
    url = _ollama_base_url() + path
    data = json.dumps(payload).encode("utf-8") if payload is not None else None
    request = urllib.request.Request(
        url,
        data=data,
        headers={"Content-Type": "application/json"} if data is not None else {},
        method="POST" if data is not None else "GET",
    )
    with urllib.request.urlopen(request, timeout=timeout) as response:
        return json.loads(response.read().decode("utf-8"))


def ensure_ollama_ready(required_models, auto_pull=True):
    """
    Precondition check for the external resources this script depends on.

    The original script failed silently at the embedding step because the
    Ollama server was unreachable / the required models were not pulled, and
    the resulting exception was swallowed by a bare ``except Exception``
    block.  This check makes such environment problems explicit *before* any
    expensive work (PDF parsing, embedding) is attempted, and tries to repair
    them by pulling the missing models.
    """
    # 1) Is the Ollama server running?
    try:
        info = _ollama_request("/api/tags", timeout=10)
    except Exception as exc:
        raise RuntimeError(
            f"Cannot reach the Ollama server at {_ollama_base_url()} ({exc}). "
            "Start it with `ollama serve` and run this script again."
        ) from exc

    installed = set()
    for model_info in info.get("models", []):
        name = model_info.get("name") or model_info.get("model") or ""
        installed.add(name.split(":")[0])

    # 2) Are the required models available?
    missing = [model for model in required_models if model not in installed]
    if not missing:
        return

    if not auto_pull:
        raise RuntimeError(
            f"Required Ollama model(s) missing: {', '.join(missing)}. "
            f"Pull them first with: ollama pull {' '.join(missing)}"
        )

    # 3) Best effort: pull the missing models so the pipeline can proceed.
    #    (stream=False -> Ollama answers once the download is complete, so no
    #    socket timeout may be used here.)
    for model in missing:
        logger.info(f"Pulling missing Ollama model '{model}' (this may take a while)...")
        try:
            _ollama_request("/api/pull", payload={"name": model, "stream": False}, timeout=None)
            logger.info(f"Model '{model}' is now available.")
        except Exception as exc:
            raise RuntimeError(
                f"Could not pull Ollama model '{model}' ({exc}). "
                f"Pull it manually with: ollama pull {model}"
            ) from exc


def run_agent(question):
    vector_db = None  # initialise: prevents UnboundLocalError if creation fails
    try:
        # -------------------------------------------------------------------
        # Precondition checks: fail loudly with actionable messages instead
        # of silently producing no output.
        # -------------------------------------------------------------------
        if not PDF_PATH:
            raise ValueError("No PDF file uploaded")

        if not os.path.isfile(PDF_PATH):
            raise FileNotFoundError(f"PDF file not found: {PDF_PATH}")

        ensure_ollama_ready(required_models=[EMBEDDING_MODEL, LLM_MODEL])

        # -------------------------------------------------------------------
        # Local PDF file upload
        # -------------------------------------------------------------------
        loader = UnstructuredPDFLoader(file_path=PDF_PATH)
        data = loader.load()
        if not data:
            raise ValueError(f"No content could be extracted from '{PDF_PATH}'.")
        logger.info("Loading of PDF is done")

        # Preview first page
        logger.info(f"First page content preview: {data[0].page_content[:500]}...")

        # -------------------------------------------------------------------
        # Split and chunk
        # -------------------------------------------------------------------
        text_splitter = RecursiveCharacterTextSplitter(chunk_size=7500, chunk_overlap=100)
        logger.info("Text splitter created")
        chunks = text_splitter.split_documents(data)
        logger.info(f"Created {len(chunks)} chunks")
        if not chunks:
            raise ValueError("Text splitting produced no chunks.")

        # -------------------------------------------------------------------
        # Vector DB
        #
        # NOTE: errors here are intentionally NOT swallowed any more.  In the
        # original code the nested try/except hid the real failure (usually
        # the Ollama server being down or 'nomic-embed-text' not pulled),
        # which left `vector_db` unbound and made the script silently stop
        # right after the embedding step.
        # -------------------------------------------------------------------
        logger.info("Creating Vector db")
        embedding_model = OllamaEmbeddings(
            model=EMBEDDING_MODEL,
            base_url=_ollama_base_url(),
            show_progress=True,
        )
        logger.info(f"Embedding model '{EMBEDDING_MODEL}' created")
        vector_db = Chroma.from_documents(
            documents=chunks,
            embedding=embedding_model,
            collection_name="local-rag",
        )
        logger.info("Local db created successfully")

        # Verify the vector db before using it (precondition check).
        if vector_db is None:
            raise ValueError("Vector db creation failed")

        # -------------------------------------------------------------------
        # LLM from Ollama
        # -------------------------------------------------------------------
        llm = ChatOllama(model=LLM_MODEL, base_url=_ollama_base_url())
        logger.info("LLM model loaded")

        QUERY_PROMPT = PromptTemplate(
            input_variables=["question"],
            template="""You are an AI language model assistant. Your task is to generate five
        different versions of the given user question to retrieve relevant documents from
        a vector database. By generating multiple perspectives on the user question, your
        goal is to help the user overcome some of the limitations of the distance-based
        similarity search. Provide these alternative questions separated by newlines.
        Original question: {question}""",
        )
        logger.info("Query prompt created")

        retriever = MultiQueryRetriever.from_llm(
            vector_db.as_retriever(),
            llm,
            prompt=QUERY_PROMPT,
        )
        logger.info("Retriever created")

        # RAG prompt
        template = """Answer the question based ONLY on the following context:
{context}
    Question: {question}
    """
        prompt = ChatPromptTemplate.from_template(template)
        logger.info("RAG prompt created")

        chain = (
            {"context": retriever, "question": RunnablePassthrough()}
            | prompt
            | llm
            | StrOutputParser()
        )
        logger.info("Chain created")

        response = chain.invoke(question)
        logger.info("Chain invoked")
        logger.info(f"Response: {response}")

        return response

    except Exception as e:
        # Log WITH the full traceback so the real root cause is always
        # visible instead of being swallowed (which caused the silent stop).
        logger.exception(f"An error occurred: {e}")
        return None


if __name__ == "__main__":
    question = "What are the 5 pillars of global cooperation?"
    response = run_agent(question)
    if response is not None:
        print(f"\nFinal Output: {response}")
