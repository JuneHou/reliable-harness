import json
import logging
import os
from urllib.error import HTTPError, URLError
from urllib.request import urlopen

from langchain_community.document_loaders import UnstructuredPDFLoader
from langchain_community.embeddings import OllamaEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain.prompts import ChatPromptTemplate, PromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_community.chat_models import ChatOllama
from langchain_core.runnables import RunnablePassthrough
from langchain.retrievers.multi_query import MultiQueryRetriever

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

# Central Ollama configuration, used consistently by every component below.
EMBEDDING_MODEL = "nomic-embed-text"
LLM_MODEL = "llama3"
PDF_PATH = "./paper1.pdf"


def _resolve_ollama_base_url():
    """Resolve the Ollama server URL (honours OLLAMA_BASE_URL / OLLAMA_HOST)."""
    url = (
        os.environ.get("OLLAMA_BASE_URL")
        or os.environ.get("OLLAMA_HOST")
        or "http://localhost:11434"
    )
    url = url.rstrip("/")
    if "://" not in url:
        url = "http://" + url
    return url


def get_available_ollama_models(base_url):
    """Return the model names available in the local Ollama instance.

    Raises ConnectionError with an actionable message when the Ollama server
    cannot be reached (the #1 reason the pipeline used to 'silently' die).
    """
    try:
        with urlopen(f"{base_url}/api/tags", timeout=5) as resp:
            payload = json.loads(resp.read().decode("utf-8"))
    except (URLError, HTTPError, OSError, ValueError) as e:
        raise ConnectionError(
            f"Cannot reach the Ollama server at {base_url} ({e}). "
            "Start it with: ollama serve"
        ) from e
    # Model names come back as "<model>:<tag>" - compare on the model part only.
    return {m.get("name", "").split(":")[0] for m in payload.get("models", [])}


def run_agent(question):
    try:
        # ---- Precondition check 0: the PDF must exist -------------------
        if not PDF_PATH or not os.path.exists(PDF_PATH):
            logging.error(f"PDF file not found: '{PDF_PATH}'. Upload a PDF file.")
            return None

        # ---- Precondition check 1: Ollama server up + models pulled -----
        # If the embedding model was never pulled ('ollama pull nomic-embed-text')
        # the embedding request fails; the inner except used to swallow that
        # error and the script appeared to just 'stop' after the embedding
        # step. Checking up front turns that silent stop into an actionable
        # message.
        base_url = _resolve_ollama_base_url()
        try:
            available_models = get_available_ollama_models(base_url)
        except ConnectionError as e:
            logging.error(str(e))
            return None

        missing_models = [
            m for m in (EMBEDDING_MODEL, LLM_MODEL) if m not in available_models
        ]
        if missing_models:
            logging.error(
                "The following models are not available in your local Ollama "
                f"instance: {missing_models}"
            )
            logging.error(f"Fix: run 'ollama pull {' '.join(missing_models)}' first.")
            return None
        logging.info(
            f"Ollama is running at {base_url} with the required models: "
            f"{EMBEDDING_MODEL}, {LLM_MODEL}"
        )

        # ---- Load the PDF ------------------------------------------------
        loader = UnstructuredPDFLoader(file_path=PDF_PATH)
        data = loader.load()
        if not data:
            logging.error("The PDF loaded but contains no pages.")
            return None
        logging.info("Loading of PDF is done")

        # Preview first page
        logging.info(f"First page content preview: {data[0].page_content[:500]}...")

        # ---- Split and chunk ----------------------------------------------
        text_splitter = RecursiveCharacterTextSplitter(chunk_size=7500, chunk_overlap=100)
        logging.info("Text splitter created")
        chunks = text_splitter.split_documents(data)
        if not chunks:
            logging.error("Text splitting produced no chunks - nothing to embed.")
            return None
        logging.info(f"Created {len(chunks)} chunks")

        # ---- Create the vector DB -----------------------------------------
        # FIX: initialise the variable so a failed creation can never lead to
        # an UnboundLocalError, and abort immediately (instead of continuing
        # in a broken state) when creation fails.
        vector_db = None
        logging.info("Creating Vector db")
        try:
            embedding_model = OllamaEmbeddings(
                model=EMBEDDING_MODEL, base_url=base_url, show_progress=True
            )
            logging.info(f"Embedding model ready: {embedding_model.model}")

            # Precondition check 2: verify the embedding endpoint really
            # works before bulk-embedding every chunk.
            probe = embedding_model.embed_query("connectivity check")
            if not probe:
                raise ValueError("embedding model returned an empty vector")
            logging.info(f"Embedding endpoint OK (embedding dimension: {len(probe)})")

            vector_db = Chroma.from_documents(
                documents=chunks,
                embedding=embedding_model,
                collection_name="local-rag"
            )
            logging.info("Local db created successfully")
        except Exception:
            # FIX: log the full traceback (logging.exception) instead of only
            # the message, and stop here - never fall through with a broken
            # or unbound vector store.
            logging.exception("Error creating vector db")
            logging.error(
                f"Hint: make sure 'ollama serve' is running and the embedding "
                f"model is pulled ('ollama pull {EMBEDDING_MODEL}')."
            )
            return None

        if vector_db:
            logging.info("Vector db verification successful")
        else:
            logging.error("Vector db creation failed")
            return None

        # ---- LLM from Ollama -----------------------------------------------
        llm = ChatOllama(model=LLM_MODEL, base_url=base_url)
        logging.info("LLM model loaded")

        QUERY_PROMPT = PromptTemplate(
            input_variables=["question"],
            template="""You are an AI language model assistant. Your task is to generate five
        different versions of the given user question to retrieve relevant documents from
        a vector database. By generating multiple perspectives on the user question, your
        goal is to help the user overcome some of the limitations of the distance-based
        similarity search. Provide these alternative questions separated by newlines.
        Original question: {question}""",
        )
        logging.info("Query prompt created")

        retriever = MultiQueryRetriever.from_llm(
            vector_db.as_retriever(),
            llm,
            prompt=QUERY_PROMPT
        )
        logging.info("Retriever created")

        # RAG prompt
        template = """Answer the question based ONLY on the following context:
{context}
    Question: {question}
    """
        prompt = ChatPromptTemplate.from_template(template)
        logging.info("RAG prompt created")

        chain = (
            {"context": retriever, "question": RunnablePassthrough()}
            | prompt
            | llm
            | StrOutputParser()
        )
        logging.info("Chain created")

        response = chain.invoke(question)
        logging.info("Chain invoked")
        logging.info(f"Response: {response}")

        return response

    except Exception:
        # FIX: log the full traceback so nothing can fail 'silently' anymore.
        logging.exception("An error occurred while running the RAG agent")
        return None


if __name__ == "__main__":
    question = "What are the 5 pillars of global cooperation?"
    response = run_agent(question)
    print(f"\nFinal Output: {response}")
    if response is None:
