import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_core.tools import tool
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # Correct parameter name is "max_results" (the previous "max_result" was
    # silently ignored by the tool schema).
    simple_search = TavilySearch(max_results=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key
    )

    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory
    )

    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, memory, config, user_input):
    """
    Send only the NEW user message to the agent.

    The MemorySaver checkpointer was passed to create_react_agent, so on every
    invocation the graph automatically loads the previously checkpointed
    messages for the thread_id in `config` and appends the new message through
    the `add_messages` reducer. The old code manually pulled the raw
    checkpoint data (memory.get(config) -> channel_values) and re-fed the whole
    history as graph input, which is not the intended API and corrupts /
    duplicates the message channel, so the agent appeared to lose its memory.
    With only the new message passed in, conversation memory is retained
    correctly across invocations.
    """
    input_message = HumanMessage(content=user_input)

    final_message = None
    for step in agent_executor.stream(
        {"messages": [input_message]},
        config,
        stream_mode="values",
    ):
        final_message = step["messages"][-1]

    return final_message.content


def main():
    agent_executor, memory, config = initialize_agent()
    response1 = run_agent(agent_executor, memory, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, memory, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
