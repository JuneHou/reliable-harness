import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_core.tools import tool
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # Fix: the TavilySearch parameter is `max_results` (plural), not `max_result`.
    # The old `max_result=2` was silently ignored, so the tool used its default
    # of 5 results instead of the intended 2.
    simple_search = TavilySearch(max_results=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key
    )

    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory
    )

    # A stable thread_id is what groups all turns of this conversation
    # together in the checkpointer.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, memory, config, user_input):
    # Do NOT manually rebuild the conversation history from the checkpointer.
    #
    # The old code used the deprecated, low-level `memory.get(config)` API and
    # dug into `state["channel_values"]["messages"]`, which is an internal /
    # unstable checkpoint format and does not reliably expose the prior
    # messages, so the agent lost its memory between invocations.
    #
    # With a checkpointer (MemorySaver) attached and a fixed `thread_id`,
    # LangGraph automatically loads the persisted thread state and the
    # `add_messages` reducer appends this new user message to the existing
    # history. Therefore we only need to send the NEW message.
    final_message = None
    for step in agent_executor.stream(
        {"messages": [HumanMessage(content=user_input)]},
        config,
        stream_mode="values",
    ):
        final_message = step["messages"][-1]

    return final_message.content


def main():
    agent_executor, memory, config = initialize_agent()
    # Both turns use the same thread_id ("agent003") and the same
    # MemorySaver-backed agent, so the second turn sees the first turn's
    # messages and the agent can answer "Bob".
    response1 = run_agent(agent_executor, memory, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, memory, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
