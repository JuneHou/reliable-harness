import json
import os
from llama_index.llms.llama_cpp import LlamaCPP


def run_agent(query):
    MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-13B-chat-GGUF/resolve/main/llama-2-13b-chat.Q4_0.gguf"
    MODEL_PATH = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if "llama-2-13b-chat" in paths:
                MODEL_URL = None
                MODEL_PATH = paths["llama-2-13b-chat"]

    llm = LlamaCPP(
        model_url=MODEL_URL,
        model_path=MODEL_PATH,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={"n_gpu_layers": 0},
        verbose=True,
    )

    # FIX: The non-streaming llm.complete() call hangs indefinitely after model
    # initialization when running a local Llama-2 13B GGUF model on CPU, never
    # returning any inference output. Use the streaming API instead
    # (llm.stream_complete), which yields token deltas incrementally, and
    # accumulate them into the final response text.
    response_text = ""
    for chunk in llm.stream_complete(query):
        response_text += chunk.delta

    return response_text


if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)

import json
import os

from llama_index.llms.llama_cpp import LlamaCPP

# Use the 7B chat model instead of the 13B one: on a CPU-only machine a
# 13B model takes many minutes per response and looks like it hangs right
# after initialization. Q4_K_M is the recommended quantization in
# TheBloke's GGUF repositories (well supported and fast on CPU).
DEFAULT_MODEL_URL = (
    "https://huggingface.co/TheBloke/Llama-2-7B-Chat-GGUF"
    "/resolve/main/llama-2-7b-chat.Q4_K_M.gguf"
)


def _load_local_model_path():
    """Prefer a locally downloaded model if paths.json points to a valid file."""
    paths_file = "./paths.json"
    if not os.path.exists(paths_file):
        return None

    with open(paths_file, "r") as f:
        paths = json.load(f)

    for key in ("llama-2-7b-chat", "llama-2-13b-chat"):
        local_path = paths.get(key)
        if local_path and os.path.exists(local_path):
            return local_path
    return None


def build_llm():
    model_path = _load_local_model_path()

    llm = LlamaCPP(
        # Exactly one of model_url / model_path must be provided.
        model_url=None if model_path else DEFAULT_MODEL_URL,
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=256,
        # Llama-2 models were trained with a 4096-token context window.
        context_window=4096,
        model_kwargs={
            "n_gpu_layers": 0,                 # CPU only
            "n_batch": 512,                    # prompt-processing batch size
            "n_threads": os.cpu_count() or 4,  # use all cores for generation
        },
        verbose=True,  # keep load/timing logs visible so it never looks stuck
    )
    return llm


def run_agent(llm, query):
    response = llm.complete(query)
    return str(response)


if __name__ == "__main__":
    # Load the (heavy) model only once instead of re-initializing per query.
    llm = build_llm()

    query = "What is the capital of France?"
    response = run_agent(llm, query)
    print(response)
