import os
from typing import List
from dotenv import load_dotenv
from pydantic import BaseModel, Field
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import PydanticOutputParser

load_dotenv()


class Element(BaseModel):
    """A single identifier / text pair parsed from the error message."""
    identifier: str = Field(description="This is missing identifier.")
    text: str = Field(description="This is missing text.")


class ParsedError(BaseModel):
    """Structured result: one entry per AllOf(...) group in the error message."""
    elements: List[Element] = Field(
        description="The list of identifier and text pairs found in the error message."
    )


def get_output_parser():
    # StructuredOutputParser only supports FLAT key/value schemas.
    # To get nested JSON (a list of dicts with identifier/text pairs),
    # we must use PydanticOutputParser with nested Pydantic models.
    output_parser = PydanticOutputParser(pydantic_object=ParsedError)
    return output_parser


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    response_as_dict = output_parser.parse(response.content)
    # Return the nested JSON the user expects: a list of {identifier, text} dicts.
    return [element.dict() for element in response_as_dict.elements]


if __name__ == "__main__":
    test_msg = "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"
    PROMPT_TEMPLATE = """ 
    You are an android developer. 
    Parse this error message and provide me identifiers & texts mentioned in the error message.
    Return one object per AllOf(...) group, each containing its identifier and its text.
    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
