from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_debug, set_verbose
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# NOTE: set_verbose(True) no longer shows intermediate steps for LCEL
# (RunnableSequence / pipe-composed) chains in LangChain >= 0.0.345.
# set_debug(True) is the correct flag to log the inputs and outputs of
# every step in the chain (prompt -> model -> output parser).
set_debug(True)
# Optionally keep verbose as well; it is harmless but not needed for
# showing the intermediate LCEL chain steps.
set_verbose(True)


def run_agent(topic: str) -> str:
    model = ChatOpenAI(openai_api_key=api_key)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)
    chain = prompt | model | output_parser
    return chain.invoke({"topic": topic})


if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
