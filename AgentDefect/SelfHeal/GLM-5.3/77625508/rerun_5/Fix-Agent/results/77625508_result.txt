from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_debug, set_verbose
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# NOTE: In LangChain 0.0.345, `set_verbose(True)` / `verbose=True` only apply to
# legacy (agent-executor style) chains. LCEL chains (prompt | model | parser)
# ignore the verbose flag. To see the intermediate steps (the inputs/outputs of
# every Runnable in the chain), debug logging must be enabled via set_debug(True).
set_debug(True)
# Keep verbose enabled as well for any legacy-style components.
set_verbose(True)

def run_agent(topic: str) -> str:
    model = ChatOpenAI(openai_api_key=api_key)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)
    chain = prompt | model | output_parser
    # With set_debug(True), invoking the chain now prints the intermediate
    # steps: the formatted prompt messages, the LLM request/response, and the
    # parsed output for each Runnable in the chain.
    return chain.invoke({"topic": topic})

if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
