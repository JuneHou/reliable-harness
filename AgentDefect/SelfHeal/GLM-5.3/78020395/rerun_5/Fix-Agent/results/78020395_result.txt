from dotenv import load_dotenv

class NestedStructuredOutputParser(StructuredOutputParser):
    """
    A StructuredOutputParser that asks the LLM for a *nested* JSON output:
    a JSON array (a list of dictionaries), one object per identifier/text pair,
    each object containing the defined response schemas.

    Why this is needed
    ------------------
    1. The default ``StructuredOutputParser.get_format_instructions()``
       renders the response schemas as a single FLAT JSON object:

           {"identifier": "...", "text": "..."}

       so the model is forced to return exactly one identifier and one text
       (all pairs get merged / overwritten into that one flat dict).
    2. The default ``StructuredOutputParser.parse()`` only accepts JSON
       objects and raises OutputParserException for JSON arrays.

    Fix
    ---
    - Override ``get_format_instructions()`` so the example schema shown to
      the model is an ARRAY OF OBJECTS.
    - Override ``parse()`` so a JSON array (list of dicts) is accepted.
    """

    def get_format_instructions(self) -> str:
        # Build an example object from the response schemas
        example_fields = []
        for schema in self.response_schemas:
            example_fields.append(
                f'    "{schema.name}": string  // {schema.description}'
            )
        example_object = ",\n".join(example_fields)

        return (
            'The output should be a markdown code snippet formatted in the '
            'following JSON schema, including the leading and trailing "```json" '
            'and "```":\n\n'
            "```json\n"
            "[\n"
            "    {\n"
            f"{example_object}\n"
            "    },\n"
            "    {\n"
            f"{example_object}\n"
            "    }\n"
            "]\n"
            "```\n\n"
            "IMPORTANT: the top-level value must be a JSON array (a list of "
            "objects). Return one object in the array for EVERY identifier & "
            "text pair found in the input, and each object must contain all of "
            "the keys described above. Do NOT return a single flat dictionary "
            "and do NOT merge multiple pairs into one object."
        )

    def parse(self, text: str) -> Any:
        # The default implementation only accepts JSON objects (dicts) and
        # raises OutputParserException for arrays; accept both here.
        try:
            return parse_json_markdown(text)
        except Exception as e:
            raise OutputParserException(
                f"Parsing text\n{repr(text)} raised following error:\n{e}"
            )


def get_output_parser():
    missing_id = ResponseSchema(name="identifier", description="This is missing identifier.")
    missing_text = ResponseSchema(name="text", description="This is missing text.")

    response_schemas = [missing_id, missing_text]
    output_parser = NestedStructuredOutputParser.from_response_schemas(response_schemas)
    return output_parser


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    response_as_dict = output_parser.parse(response.content)

    # Defensive normalization: if the model still returns a single flat dict,
    # wrap it into a list so the output shape is always a list of dicts.
    if isinstance(response_as_dict, dict):
        response_as_dict = [response_as_dict]

    return response_as_dict
