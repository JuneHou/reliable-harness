import os
import json
from typing import List
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain.output_parsers import PydanticOutputParser
from pydantic import BaseModel, Field

load_dotenv()


class Element(BaseModel):
    """A single identifier/text pair extracted from the error message."""
    identifier: str = Field(description="The withId value mentioned in the error message")
    text: str = Field(description="The withText value mentioned in the error message")


class ParsedResult(BaseModel):
    """A list of identifier/text pairs found in the error message."""
    elements: List[Element] = Field(description="List of identifier/text pairs found in the error message")


def get_output_parser():
    # StructuredOutputParser cannot express nested/repeating structures
    # (a list of {identifier, text} dicts), so use PydanticOutputParser
    # with a nested model instead to get the desired nested JSON output.
    output_parser = PydanticOutputParser(pydantic_object=ParsedResult)
    return output_parser


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    response_as_dict = output_parser.parse(response.content)
    # Return a plain nested JSON structure: a list of dicts with identifier/text
    return json.loads(response_as_dict.model_dump_json())["elements"]


if __name__ == "__main__":
    test_msg = "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"
    PROMPT_TEMPLATE = """ 
    You are an android developer. 
    Parse this error message and provide me identifiers & texts mentioned in error message.
    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(json.dumps(result, indent=2))
