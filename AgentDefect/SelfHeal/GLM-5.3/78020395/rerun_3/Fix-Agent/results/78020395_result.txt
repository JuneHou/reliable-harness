import os
from typing import List

from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import PydanticOutputParser

# ---------------------------------------------------------------------------
# Why the original code failed to produce nested output:
#
#   StructuredOutputParser.from_response_schemas() builds its format
#   instructions from a FLAT {name: description} mapping, so the LLM is only
#   ever told to emit a single flat dict, e.g.
#       {"identifier": "...", "text": "..."}
#   It cannot describe an array of objects, i.e. a list of
#       {"identifier": ..., "text": ...}
#   pairs. To get nested/structured JSON output we must instead describe the
#   whole structure with pydantic models and use PydanticOutputParser, which
#   serializes the complete (nested) JSON schema -- including the list of
#   identifier/text objects -- into the format instructions.
# ---------------------------------------------------------------------------

# Use LangChain's pydantic v1 shim when available (works on langchain-core
# 0.1/0.2/0.3), otherwise fall back to plain pydantic (newer langchain-core
# that dropped the shim and natively supports pydantic v2 models).
try:
    from langchain_core.pydantic_v1 import BaseModel, Field
except ImportError:  # langchain >= 1.0 / langchain-core without the v1 shim
    from pydantic import BaseModel, Field

load_dotenv()


class MissingElement(BaseModel):
    """One identifier/text pair found in the error message."""
    identifier: str = Field(description="The identifier mentioned in the error message, e.g. identifier1")
    text: str = Field(description="The text mentioned in the error message, e.g. text1")


class MissingElements(BaseModel):
    """All identifier/text pairs mentioned in the error message."""
    elements: List[MissingElement] = Field(
        description="Every identifier & text pair mentioned in the error "
                    "message, in order of appearance"
    )


def get_output_parser():
    # PydanticOutputParser replaces StructuredOutputParser so that the nested
    # structure (a list of {identifier, text} objects) is part of the schema
    # the LLM is asked to follow.
    return PydanticOutputParser(pydantic_object=MissingElements)


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    parsed = output_parser.parse(response.content)

    # Convert the parsed structure into the plain list of dictionaries the
    # caller expects:
    #   [{"identifier": "identifier1", "text": "text1"},
    #    {"identifier": "identifier2", "text": "text1"},
    #    ...]
    elements = parsed["elements"] if isinstance(parsed, dict) else parsed.elements

    result = []
    for element in elements:
        if isinstance(element, dict):
            result.append({"identifier": element["identifier"], "text": element["text"]})
        elif hasattr(element, "model_dump"):  # pydantic v2 object
            result.append(element.model_dump())
        else:  # pydantic v1 object
            result.append(element.dict())
    return result


if __name__ == "__main__":
    test_msg = "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"
    PROMPT_TEMPLATE = """ 
    You are an android developer. 
    Parse this error message and provide me identifiers & texts mentioend in error message. 
    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
