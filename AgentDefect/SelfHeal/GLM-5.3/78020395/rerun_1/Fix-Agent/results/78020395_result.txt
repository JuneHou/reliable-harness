import os
from typing import List

from dotenv import load_dotenv
from pydantic import BaseModel, Field

from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import PydanticOutputParser

load_dotenv()


class UiElement(BaseModel):
    """A single identifier + text pair extracted from the error message."""
    identifier: str = Field(description="This is missing identifier.")
    text: str = Field(description="This is missing text.")


class UiElementList(BaseModel):
    """The full set of identifier + text pairs found in the error message."""
    elements: List[UiElement] = Field(
        description="List of all identifier & text pairs mentioned in the error message"
    )


def get_output_parser():
    # StructuredOutputParser only supports FLAT dictionaries (each
    # ResponseSchema name maps to a single scalar value), so it can never
    # produce nested output such as a list of {identifier, text} objects.
    # PydanticOutputParser supports nested models and lists, which gives us
    # the nested JSON structure we need.
    output_parser = PydanticOutputParser(pydantic_object=UiElementList)
    return output_parser


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    # parse() returns a UiElementList pydantic object; convert it to plain
    # nested dicts: {"elements": [{"identifier": ..., "text": ...}, ...]}
    parsed = output_parser.parse(response.content)
    response_as_dict = parsed.model_dump() if hasattr(parsed, "model_dump") else parsed.dict()
    return response_as_dict


if __name__ == "__main__":
    test_msg = "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"
    PROMPT_TEMPLATE = """ 
    You are an android developer. 
    Parse this error message and provide me identifiers & texts mentioend in error message. 
    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
