from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node with None in metadata
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata={"start_date": None},
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # -------------------------------------------------------------------------
    # FIX: filtering on null (None) metadata values.
    #
    # The original filter was:
    #
    #     MetadataFilter(
    #         key="start_date",
    #         operator=FilterOperator.EQ,
    #         value=None,
    #     )
    #
    # That never matches anything. LlamaIndex's metadata filtering does not
    # support matching against a literal `None` value with `FilterOperator.EQ`:
    # an equality filter compares against the value actually stored for the
    # key, and null metadata values are not matched this way. As a result, a
    # node whose metadata is {"start_date": None} is silently filtered out and
    # the retriever returns no nodes even though the document exists in the
    # index (hence "No nodes retrieved").
    #
    # The supported way to filter for null/missing metadata values is the
    # dedicated `FilterOperator.IS_EMPTY` operator, which matches nodes where
    # the given key is missing, null, or empty:
    #
    #     MetadataFilter(
    #         key="start_date",
    #         operator=FilterOperator.IS_EMPTY,
    #     )
    #
    # Alternative workaround (required for vector stores that cannot store
    # nulls at all, e.g. Chroma/Pinecone): normalize `None` metadata values to
    # an explicit sentinel string (e.g. "null") when building the nodes, and
    # then filter with `FilterOperator.EQ` on that sentinel value:
    #
    #     metadata={"start_date": None if ... else ...}  ->
    #     metadata={"start_date": "null"}
    #     MetadataFilter(key="start_date",
    #                    operator=FilterOperator.EQ,
    #                    value="null")
    # -------------------------------------------------------------------------
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.IS_EMPTY,  # was: FilterOperator.EQ with value=None
        value=None,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes
