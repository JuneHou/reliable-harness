from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode

    MetadataFilter,

    FilterOperator,
)

def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node with None in metadata
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata={"start_date": None},
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # WHY THE ORIGINAL CODE RETURNED NOTHING:
    #
    #   MetadataFilter(key="start_date", operator=FilterOperator.EQ, value=None)
    #
    # The node's metadata really is stored as {"start_date": None} (see the
    # debug print above), but the default in-memory vector store
    # (SimpleVectorStore) implements the EQ filter with a type check on the
    # filter value: it only accepts str / int / float / bool (or a list of
    # those). Because `None` fails that type check (and is not a list), the
    # filter predicate short-circuits to False for *every* node, so the node
    # is silently filtered out and the retriever returns no results.
    #
    # RECOMMENDED APPROACH for "field is null / empty / missing":
    #   use the dedicated FilterOperator.IS_EMPTY operator.
    #   The `value` argument is ignored for IS_EMPTY.
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.IS_EMPTY,
        value=None,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # NOTE: if you use a hosted vector store (Pinecone, Chroma, etc.) that
    # cannot index null values at all, the common workaround is to never
    # store None in the first place -- e.g. write
    # metadata={"start_date": ""} (or a sentinel like "null") at ingest time
    # and filter with FilterOperator.EQ, value="" instead. IS_EMPTY also
    # matches empty strings, so it covers both cases.

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes

"""
LlamaIndex: why `FilterOperator.EQ` with `value=None` retrieves nothing,
and the recommended workaround for filtering on null metadata values.

Root cause
----------
LlamaIndex's metadata filters do not support `None` as a value. A filter
is evaluated roughly as

    matches = operator(stored_value, filter.value) and stored_value is not None

(see `_build_metadata_filter_fn` in `llama_index.core.vector_stores.utils`,
used by the default in-memory `SimpleVectorStore`). Because of that
"stored value must not be null" requirement, a node whose metadata field
is `None` can never satisfy *any* filter -- not even `FilterOperator.EQ`
with `value=None`: `None == None` is True, but the not-null check fails,
so the combined result is False. In other words, a `None` metadata value
is treated as "this field has no value" instead of as a value that can be
compared -- it is simply not filterable.

Recommended approach / workaround
---------------------------------
Do not store `None` in metadata you want to filter on. Instead:

1. At index time, encode nulls as a sentinel string (here we use the
   string "None"; "" or "unknown" would work too).
2. Filter with `FilterOperator.EQ` on that sentinel value.
3. Decode the sentinel back to `None` after retrieval, if your
   downstream code expects real nulls.

(A few vector-store backends additionally expose a dedicated
`FilterOperator.IS_EMPTY` operator for empty/missing fields, but support
is backend-specific, so the sentinel pattern below is the portable
solution.)
"""

from typing import Any, Dict

from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# Sentinel stored in metadata in place of None, because LlamaIndex
# metadata filters never match a None value.
NULL_SENTINEL = "None"


def to_filterable_metadata(metadata: Dict[str, Any]) -> Dict[str, Any]:
    """Encode None metadata values as a filterable sentinel value.

    LlamaIndex cannot match metadata filters against None, so null values
    must be replaced with a sentinel before nodes are added to the index.
    """
    return {
        key: (NULL_SENTINEL if value is None else value)
        for key, value in metadata.items()
    }


def from_filterable_metadata(metadata: Dict[str, Any]) -> Dict[str, Any]:
    """Decode the sentinel value back to None (for downstream use)."""
    return {
        key: (None if value == NULL_SENTINEL else value)
        for key, value in metadata.items()
    }


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node. IMPORTANT: store the sentinel string instead of a raw
    # None -- None values are not filterable with metadata filters.
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=to_filterable_metadata({"start_date": None}),
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter on the sentinel value (filtering on None itself never matches)
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results (decode the sentinel back to None for readability)
    print("Retrieved nodes:")
    print(
        [
            (node.node_id, from_filterable_metadata(node.metadata))
            for node in nodes
        ]
    )
    print()

    return nodes
