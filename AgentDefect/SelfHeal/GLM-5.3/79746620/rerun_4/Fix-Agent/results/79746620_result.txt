"""
LlamaIndex metadata filtering on null metadata values -- fixed version.

WHY THE ORIGINAL CODE RETRIEVES NOTHING
---------------------------------------
`MetadataFilter(key="start_date", operator=FilterOperator.EQ, value=None)` can
never match a node whose metadata field is `None`, because:

1. LlamaIndex strips `None` metadata values when node metadata is serialized
   for the vector store / persisted (`node_to_metadata_dict()` defaults to
   `remove_null=True`), so the stored node no longer has the `start_date` key
   at all.
2. `MetadataFilter` stringifies its value for every operator except the
   `is_*` operators, so `value=None` becomes the string "None", which can
   never compare equal to a missing / null field.
3. Most vector stores (Chroma, Pinecone, Postgres, ...) cannot express
   "field is null" filters either.

RECOMMENDED WORKAROUND
----------------------
Do not store `None` in node metadata. Normalize it to a sentinel string
("null" here) when the nodes are built, filter on that sentinel with
`FilterOperator.EQ`, and map the sentinel back to `None` after retrieval.
This works with every vector store.

(For documents where a key is completely *absent*, some stores also support
`FilterOperator.IS_EMPTY`, but that matches missing keys rather than keys
holding `None`, and it is not supported by every vector store.)
"""

from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# Sentinel that stands in for "no value" inside node metadata.
# Vector stores cannot store or filter on actual `None` values.
NULL_SENTINEL = "null"


def normalize_metadata(metadata):
    """Prepare metadata for indexing: None -> NULL_SENTINEL."""
    return {
        key: (NULL_SENTINEL if value is None else value)
        for key, value in metadata.items()
    }


def denormalize_metadata(metadata):
    """Convert stored metadata back: NULL_SENTINEL -> None."""
    return {
        key: (None if value == NULL_SENTINEL else value)
        for key, value in metadata.items()
    }


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node. `None` is normalized to the sentinel value so that the
    # "missing" value is actually stored in the vector store and can be
    # filtered on later.
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=normalize_metadata({"start_date": None}),  # -> {"start_date": "null"}
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print(
        [
            denormalize_metadata(node.metadata)
            for node in doc_index.docstore.docs.values()
        ]
    )
    print()

    # Filter on the sentinel value instead of None.
    # (FilterOperator.EQ with value=None can never match: None metadata values
    # are stripped from stored metadata and the filter value gets stringified
    # to "None".)
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, denormalize_metadata(node.metadata)) for node in nodes])
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
