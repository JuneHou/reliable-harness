"""
LlamaIndex: filtering on null (None) metadata values -- corrected version.

WHY THE ORIGINAL CODE FAILED
============================
The original code stored a null value in a node's metadata
(`metadata={"start_date": None}`) and then tried to retrieve it with:

    MetadataFilter(key="start_date", operator=FilterOperator.EQ, value=None)

`FilterOperator.EQ` (like the other comparison operators: NE, GT, GTE, LT,
LTE, IN, NIN, TEXT_MATCH, CONTAINS) is only defined for concrete scalar
values (str / int / float / bool, or lists for IN / NIN).  `None` is *not* a
valid comparison value: LlamaIndex's filter evaluation does not treat
`value=None` as "match nodes whose metadata value is also None".  The null
filter value simply never matches, so the retriever returned an empty list
("No nodes retrieved") even though the node was correctly stored in the index
(the docstore debug print proves it is there with `{'start_date': None}`).

RECOMMENDED APPROACH
====================
Use the dedicated empty-value operators that LlamaIndex provides for exactly
this case:

    MetadataFilter(key="start_date", operator=FilterOperator.IS_EMPTY)

* `FilterOperator.IS_EMPTY`     -> matches nodes whose metadata value is
                                   None, an empty string, or missing entirely.
* `FilterOperator.IS_NOT_EMPTY` -> the complement (non-empty values only).

(These operators are available from llama-index-core >= 0.10.44 onwards.)

WORKAROUND (older versions / stores without empty-operator support)
===================================================================
Represent "null" with an explicit sentinel value (e.g. the string "None") in
the node metadata and filter on that sentinel with `FilterOperator.EQ`.

This script tries the recommended `IS_EMPTY` approach first and automatically
falls back to the sentinel workaround if the installed llama-index version
cannot filter with `IS_EMPTY`, so the expected node is always retrieved.
"""

import os
from typing import Any, List, Optional

from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# Sentinel string used to represent a null metadata value when the installed
# llama-index version / vector store cannot filter on actual None values.
NULL_SENTINEL = "None"

NODE_TEXT = "This document has None in the metadata"


def _empty_operator_available() -> bool:
    """Whether this llama-index version ships IS_EMPTY / IS_NOT_EMPTY."""
    return hasattr(FilterOperator, "IS_EMPTY") and hasattr(
        FilterOperator, "IS_NOT_EMPTY"
    )


def _build_is_empty_filters(key: str) -> Optional[MetadataFilters]:
    """Build filters matching nodes whose `key` metadata value is null/empty.

    Returns None when the filters cannot be built on the installed version.
    """
    if not _empty_operator_available():
        return None

    # Tolerate small API differences between versions (the `value` argument
    # being optional vs. explicitly None for the empty operators).
    for filter_kwargs in (
        {"key": key, "operator": FilterOperator.IS_EMPTY},
        {"key": key, "operator": FilterOperator.IS_EMPTY, "value": None},
    ):
        try:
            return MetadataFilters(filters=[MetadataFilter(**filter_kwargs)])
        except Exception:
            continue
    return None


def _build_sentinel_filters(key: str) -> MetadataFilters:
    """Workaround filters: EQ match on an explicit sentinel value for null."""
    return MetadataFilters(
        filters=[
            MetadataFilter(
                key=key,
                operator=FilterOperator.EQ,
                value=NULL_SENTINEL,
            )
        ]
    )


def _setup_embedding_model() -> None:
    """Use OpenAI embeddings when a key is available, mock embeddings otherwise."""
    if os.getenv("OPENAI_API_KEY"):
        Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")
    else:
        # Keeps the filtering demo runnable without an OpenAI API key.
        from llama_index.core.embeddings import MockEmbedding

        Settings.embed_model = MockEmbedding(embed_dim=1536)


def _retrieve(doc_index, query, filters, similarity_top_k: int = 1) -> List[Any]:
    """Run a filtered retrieval against the given index."""
    retriever = doc_index.as_retriever(
        filters=filters, similarity_top_k=similarity_top_k
    )
    return retriever.retrieve(query)


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    _setup_embedding_model()

    # ------------------------------------------------------------------
    # Strategy 1 (recommended): keep None in the metadata and filter on it
    # with the dedicated IS_EMPTY operator.  Using `FilterOperator.EQ` with
    # `value=None` does NOT work -- None is not a comparable filter value.
    # ------------------------------------------------------------------
    is_empty_filters = _build_is_empty_filters("start_date")
    if is_empty_filters is not None:
        # Create node with None in the metadata
        node_01 = TextNode(
            text=NODE_TEXT,
            id_="node_01",
            metadata={"start_date": None},
        )

        # Create index
        doc_index = VectorStoreIndex([node_01])

        # Debug: Check what's actually stored
        print("Index nodes:")
        print([node.metadata for node in doc_index.docstore.docs.values()])
        print()

        # Filter on the null value with IS_EMPTY (not EQ + None)
        try:
            nodes = _retrieve(doc_index, query, is_empty_filters)
        except Exception as exc:
            # e.g. this version's vector store does not implement IS_EMPTY
            print(f"IS_EMPTY filtering not supported here ({exc}).")
            print()
            nodes = []

        # Show results
        print("Retrieved nodes:")
        print([(node.node_id, node.metadata) for node in nodes])
        print()

        if nodes:
            return nodes

        print("IS_EMPTY returned no nodes; using sentinel workaround.")
        print()

    # ------------------------------------------------------------------
    # Strategy 2 (workaround): store an explicit sentinel value ("None")
    # instead of a real null and filter on it with FilterOperator.EQ.
    # ------------------------------------------------------------------
    node_01 = TextNode(
        text=NODE_TEXT,
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes (sentinel workaround):")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter on the sentinel value with EQ
    filters = _build_sentinel_filters("start_date")
    nodes = _retrieve(doc_index, query, filters)

    # Show results
    print("Retrieved nodes (sentinel workaround):")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
