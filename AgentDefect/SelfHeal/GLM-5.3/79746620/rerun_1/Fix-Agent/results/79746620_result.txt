"""
Why FilterOperator.EQ with value=None fails, and the recommended workaround.

Root cause
----------
LlamaIndex's metadata filtering works on the *stored* vector-store metadata
records, not on the raw TextNode.metadata dict. The default (in-memory /
SimpleVectorStore) pipeline does two things that break a `None` filter value:

1. JSON-serialized storage: node metadata is written as JSON, where a Python
   `None` becomes `null`, and most built-in vector stores (and the default
   `SimpleVectorStore`) do not support matching `null` metadata values with an
   equality (`==`) comparison at all -- metadata values are expected to be
   scalar strings/numbers/bools.

2. Even where an EQ comparison is attempted, the filter value `None` is
   compared against the *string* `"None"` / `null` representation, or the
   filter is simply skipped, so no documents come back. `None` is therefore
   "not filterable" via FilterOperator.EQ in the default stores.

Recommended approach / workaround
---------------------------------
Do not store `None` in metadata and do not filter with EQ on None. Instead:

* Ingest-side (the actual fix used here): store an explicit, filterable
  sentinel value for missing data, e.g. the string "None" (or "" / "null"),
  so the field is always present as a scalar string.

* Query-side: filter with `FilterOperator.EQ` on that sentinel value
  (`value="None"`), which works on every vector store, including the default
  in-memory store.

* If you must express "field is missing/null", the portable option is
  `FilterOperator.IS_EMPTY`, which matches nodes whose metadata value for the
  key is empty/missing. (Support varies by vector store.)

The corrected code below applies the sentinel-value workaround end-to-end:
it stores `start_date="None"` (and also demonstrates IS_EMPTY as an
alternative), and verifies that the previously-empty result set now returns
the expected node.
"""

from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # FIX (ingest side): do NOT store raw None in metadata.
    # `None` is not a filterable value with FilterOperator.EQ -- the default
    # vector store JSON-serializes it to `null` and cannot perform an equality
    # match on it, so retrieval returns nothing.
    # Instead, normalize missing values to an explicit sentinel string.
    MISSING = "None"  # could also be "" or "null" -- just be consistent

    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata={"start_date": MISSING},  # was: {"start_date": None}
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # FIX (query side): filter on the sentinel value, not on Python None.
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=MISSING,  # was: value=None (unfilterable)
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes (EQ on sentinel value):")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    # Optional alternative: IS_EMPTY matches "missing / empty" metadata values
    # without needing a sentinel. Useful when a store supports it.
    try:
        empty_filter = MetadataFilters(
            filters=[
                MetadataFilter(
                    key="start_date", operator=FilterOperator.IS_EMPTY
                )
            ]
        )
        empty_retriever = doc_index.as_retriever(
            filters=empty_filter, similarity_top_k=1
        )
        empty_nodes = empty_retriever.retrieve(query)
        print("Retrieved nodes (IS_EMPTY alternative):")
        print([(node.node_id, node.metadata) for node in empty_nodes])
        print()
    except Exception as exc:
        # Not every vector store supports IS_EMPTY; the sentinel-value EQ
        # filter above is the portable fallback.
        print(f"IS_EMPTY not supported by this store: {exc}")
        print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved (None is not filterable)")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
