import json
import os


class FineTuningExampleData:
    """A single fine-tuning example captured from a query/response pair."""

    def __init__(self, query, response):
        self.query = query
        self.response = response

    def to_dict(self):
        # OpenAI chat-style fine-tuning record.
        return {
            "messages": [
                {"role": "system", "content": "You are a helpful assistant."},
                {"role": "user", "content": self.query},
                {"role": "assistant", "content": self.response},
            ]
        }


class OpenAIFineTuningHandler:
    """Collects fine-tuning events (query/response pairs) and writes them to a JSONL file.

    The handler records a fine-tuning example every time a query starts and a
    (non-empty) response is produced, and serializes those examples on save.
    """

    def __init__(self, callback_manager=None):
        self._examples = []
        self._pending_query = None
        self.callback_manager = callback_manager

    def on_query_start(self, query):
        """Called when a query begins; remembers the question being asked."""
        self._pending_query = query

    def on_query_end(self, response):
        """Called when the query finishes; stores the query/response pair."""
        if self._pending_query is not None and response:
            self._examples.append(
                FineTuningExampleData(self._pending_query, response)
            )
        self._pending_query = None

    @property
    def examples(self):
        return list(self._examples)

    def save_finetuning_events(self, path):
        """Write every collected example to `path`, one JSON object per line."""
        with open(path, "w", encoding="utf-8") as f:
            for example in self._examples:
                f.write(json.dumps(example.to_dict(), ensure_ascii=False) + "\n")
        print(f"Wrote {len(self._examples)} examples to {path}")


class CallbackManager:
    """Routes query events to all registered handlers."""

    def __init__(self, handlers):
        self.handlers = handlers

    def on_query_start(self, query):
        for handler in self.handlers:
            handler.on_query_start(query)

    def on_query_end(self, response):
        for handler in self.handlers:
            handler.on_query_end(response)


class VectorStoreIndex:
    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex()

    def as_query_engine(self, similarity_top_k, llm):
        # Return a real query engine wired to the LLM (and hence to the
        # callback manager / fine-tuning handler) instead of the bare index.
        return QueryEngine(index=self, similarity_top_k=similarity_top_k, llm=llm)


class QueryEngine:
    """Mock query engine that generates responses and notifies the LLM callbacks."""

    def __init__(self, index, similarity_top_k, llm):
        self.index = index
        self.similarity_top_k = similarity_top_k
        self.llm = llm

    def query(self, question):
        # Simulate retrieval + generation, producing a real response string.
        response = f"Answer to: {question}"

        # Fire the callback events so the fine-tuning handler records the
        # question/answer pair.
        self.llm.callback_manager.on_query_start(question)
        self.llm.callback_manager.on_query_end(response)
        return response


class FineTuningExample:

    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def _load_questions(self):
        """Read the questions file, failing loudly if it is missing/empty."""
        questions_path = f"{self.dataset_path}/train_questions.txt"
        if not os.path.exists(questions_path):
            raise FileNotFoundError(
                f"Questions file not found: {questions_path} - no fine-tuning "
                "events can be generated without input questions."
            )

        questions = []
        with open(questions_path, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if line:
                    questions.append(line)

        if not questions:
            raise ValueError(
                f"No questions found in {questions_path} - the fine-tuning "
                "events file would be empty."
            )
        return questions

    def jsonl_generation(self):
        """Generate JSONL file for fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        self.llm = type("LLM", (), {"callback_manager": callback_manager})()

        try:
            questions = self._load_questions()

            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                query_engine.query(question)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                f"{self.dataset_path}/finetuning_events.jsonl"
            )
        return finetuning_handler


def prepare_test_data(questions_path):

    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path):
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    return ft.jsonl_generation()


if __name__ == "__main__":

    print(f"Verified {len(lines)} fine-tuning events in {output_path}")
