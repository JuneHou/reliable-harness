import json
import os
import uuid


class CBEventType:
    """Types of callback events dispatched through the CallbackManager."""

    LLM_START = "llm_start"
    LLM_END = "llm_end"


class OpenAIFineTuningHandler:
    """
    Callback handler that records LLM query/response pairs so they can be
    exported as an OpenAI fine-tuning dataset (JSONL).

    Fixes:
      * Added storage for observed events (`_fine_tuning_handler`). Previously
        the handler recorded nothing, so it always "wrote 0 examples".
      * Implemented the callback interface (`on_event`) so the CallbackManager
        can actually deliver LLM events to this handler.
      * `save_finetuning_events` now serializes the recorded events instead of
        writing an empty file and hardcoding "Wrote 0 examples".
    """

    SYSTEM_MESSAGE = "You are a helpful assistant."

    def __init__(self):
        # FIX (Add New Attribute): storage for the observed fine-tuning events.
        self._fine_tuning_handler = []
        # FIX (Add New Attribute): prompts of LLM calls still in flight.
        self._pending_queries = {}

    # ------------------------------------------------------------------ #
    # Callback interface (invoked by CallbackManager)
    # ------------------------------------------------------------------ #
    def start_trace(self, trace_id=None):
        """No-op; kept for interface compatibility."""

    def end_trace(self, trace_id=None, trace_map=None):
        """No-op; kept for interface compatibility."""

    def on_event(self, event_type, payload=None, event_id=None, **kwargs):
        """Record LLM start/end events as fine-tuning examples."""
        payload = payload or {}

        if event_type == CBEventType.LLM_START:
            # Remember the prompts of the LLM call that just started.
            self._pending_queries[event_id] = list(payload.get("messages", []))

        elif event_type == CBEventType.LLM_END:
            query_messages = payload.get("messages")
            if query_messages is None:
                query_messages = self._pending_queries.pop(event_id, [])
            else:
                self._pending_queries.pop(event_id, None)

            response = payload.get("response", "")
            if query_messages:
                self._fine_tuning_handler.append(
                    self._build_example(query_messages, response)
                )

    # ------------------------------------------------------------------ #
    # Public API
    # ------------------------------------------------------------------ #
    def get_finetuning_events(self):
        """Return the fine-tuning examples recorded so far."""
        return self._fine_tuning_handler

    def save_finetuning_events(self, path):
        """Write every recorded example to `path` as JSONL."""
        events = self.get_finetuning_events()
        with open(path, "w", encoding="utf-8") as f:
            for example in events:
                f.write(json.dumps(example) + "\n")
        print(f"Wrote {len(events)} examples to {path}")

    # ------------------------------------------------------------------ #
    # Internals
    # ------------------------------------------------------------------ #
    def _build_example(self, messages, response):
        """Turn one LLM call into an OpenAI chat fine-tuning example."""
        example_messages = [{"role": "system", "content": self.SYSTEM_MESSAGE}]
        for message in messages:
            example_messages.append(
                {
                    "role": message.get("role", "user"),
                    "content": message.get("content", ""),
                }
            )
        example_messages.append({"role": "assistant", "content": response})
        return {"messages": example_messages}


class CallbackManager:
    """Holds callback handlers and dispatches events to all of them."""

    def __init__(self, handlers):
        # Accept a single handler as well as a list of handlers.
        if not isinstance(handlers, (list, tuple)):
            handlers = [handlers]
        self.handlers = list(handlers)

    def on_event(self, event_type, payload=None, event_id=None, **kwargs):
        """
        FIX (Addition of Operations): actually propagate events to the
        registered handlers. Previously events were silently dropped, so the
        fine-tuning handler never saw any LLM call and saved 0 examples.
        """
        for handler in self.handlers:
            on_event = getattr(handler, "on_event", None)
            if callable(on_event):
                handler.on_event(
                    event_type, payload=payload, event_id=event_id, **kwargs
                )


class OpenAI:
    """
    Minimal stand-in for `llama_index.llms.OpenAI`.

    FIX: the original code attached the callback manager to a bare object
    (`type("LLM", (), {...})()`) that the query engine ignored, so no LLM
    events were ever produced. A real LLM wrapper dispatches `llm_start` /
    `llm_end` events through its callback manager for every completion, which
    is what feeds the fine-tuning handler.
    """

    def __init__(self, callback_manager=None):
        self.callback_manager = callback_manager or CallbackManager([])

    def complete(self, messages, context=""):
        """Run one completion and notify the callback manager about it."""
        event_id = str(uuid.uuid4())

        self.callback_manager.on_event(
            CBEventType.LLM_START, payload={"messages": messages}, event_id=event_id
        )

        # --- mock completion (a real LLM calls the OpenAI API here) ---
        question = ""
        for message in reversed(messages):
            if message.get("role") == "user":
                question = message.get("content", "")
                break

        context_text = context.strip()
        if context_text:
            response = (
                f"Based on the retrieved context, the answer to '{question}' "
                f"is: {context_text}"
            )
        else:
            response = (
                f"There is no information about '{question}' in the indexed "
                "documents."
            )
        # -----------------------------------------------------------------

        self.callback_manager.on_event(
            CBEventType.LLM_END,
            payload={"messages": messages, "response": response},
            event_id=event_id,
        )
        return response


def _document_text(document):
    """Return the text of a document (plain string or object with `.text`)."""
    if isinstance(document, str):
        return document
    return getattr(document, "text", None) or str(document)


class VectorStoreIndex:
    """Minimal stand-in for `llama_index.VectorStoreIndex`."""

    def __init__(self, documents=None):
        self.documents = list(documents or [])
        self.llm = None
        self.similarity_top_k = 2

    @staticmethod
    def from_documents(docs):
        # FIX (Fix Data Access): keep the documents instead of dropping them.
        return VectorStoreIndex(docs)

    def as_query_engine(self, similarity_top_k, llm=None):
        # FIX (Add New Attribute): remember the LLM (and its callback manager)
        # instead of ignoring it. This is the LLM whose callbacks populate the
        # fine-tuning handler, so it must be the one used for querying.
        self.similarity_top_k = similarity_top_k
        self.llm = llm
        return self

    def query(self, question):
        """Run a query end-to-end, going through the LLM so events fire."""
        if self.llm is None:
            raise ValueError(
                "The query engine has no LLM. Without an LLM that carries the "
                "callback manager, no fine-tuning events can be recorded."
            )

        # Naive retrieval: the top-k indexed documents.
        retrieved = self.documents[: self.similarity_top_k]
        context = "\n".join(_document_text(doc) for doc in retrieved)

        messages = [{"role": "user", "content": question}]
        return self.llm.complete(messages, context=context)

    def __init__(self, dataset_path, documents):
