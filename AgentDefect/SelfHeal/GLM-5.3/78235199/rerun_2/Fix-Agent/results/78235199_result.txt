class FinetuningExample:
    """A single fine-tuning example in OpenAI chat format."""

    def __init__(self, messages=None, query=None, response=None):
        if messages is None:
            messages = []
            if query is not None:
                messages.append({"role": "user", "content": query})
            messages.append({"role": "assistant", "content": response or ""})
        self.messages = messages

    @classmethod
    def from_llm_events(cls, payload):
        """Build an example from the payload of an LLM end-event."""
        payload = payload or {}
        if isinstance(payload.get("messages"), list):
            messages = [dict(m) for m in payload["messages"]]
        else:
            messages = [{"role": "user", "content": payload.get("query", "")}]
        response = payload.get("response", "")
        if isinstance(response, dict):  # chat-message-like object
            response = response.get("content", "")
        messages.append({"role": "assistant", "content": response})
        return cls(messages=messages)

    def to_jsonl(self):
        """Convert the example to a JSON-serializable dict."""
        return {"messages": self.messages}

    @classmethod
    def write_jsonl(cls, examples, path):
        """Write all examples to `path`, one JSON object per line."""
        with open(path, "w", encoding="utf-8") as f:
            for example in examples:
                f.write(json.dumps(example.to_jsonl()) + "\n")

class CallbackManager:
    """Holds handlers and dispatches callback events to them.

    FIX: the buggy version only stored the handlers and never called any of
    their hooks, so handlers could never observe LLM calls.
    """

    def __init__(self, handlers=None):
        self._handlers = list(handlers) if handlers else []

    @property
    def handlers(self):
        return list(self._handlers)

    def add_handler(self, handler):
        if handler not in self._handlers:
            self._handlers.append(handler)

    def remove_handler(self, handler):
        if handler in self._handlers:
            self._handlers.remove(handler)

    def _for_each(self, method_name, *args, **kwargs):
        for handler in self._handlers:
            method = getattr(handler, method_name, None)
            if callable(method):
                method(*args, **kwargs)

    def on_event_start(self, event_type, payload=None, event_id=None, **kwargs):
        self._for_each("on_event_start", event_type, payload, event_id=event_id, **kwargs)
        return event_id

    def on_event_end(self, event_type, payload=None, event_id=None, **kwargs):
        self._for_each("on_event_end", event_type, payload, event_id=event_id, **kwargs)

    def start_trace(self, trace_id=None):
        self._for_each("start_trace", trace_id)

    def end_trace(self, trace_id=None):
        self._for_each("end_trace", trace_id)

class LLM:
    """Deterministic stand-in for `OpenAI(model=..., callback_manager=...)`.

    FIX: the buggy code built an empty stub object
    (`type("LLM", (), {"callback_manager": callback_manager})()`) that could
    not produce completions. The crucial property of this class is that every
    completion is announced through the callback manager, which is how the
    fine-tuning handler observes (question, answer) pairs.
    """

    def __init__(self, model="gpt-3.5-turbo-0613", callback_manager=None):
        self.model = model
        self.callback_manager = (
            callback_manager if callback_manager is not None else CallbackManager()
        )

    def predict(self, prompt):
        """Run a completion and fire the LLM callback events around it."""
        event_id = str(uuid.uuid4())
        self.callback_manager.on_event_start(
            CBEventType.LLM,
            payload={"messages": [{"role": "user", "content": prompt}]},
            event_id=event_id,
        )
        # Deterministic mock completion (no network access required).
        response = f"This is a generated answer to: {prompt}"
        self.callback_manager.on_event_end(
            CBEventType.LLM,
            payload={
                "messages": [{"role": "user", "content": prompt}],
                "response": response,
            },
            event_id=event_id,
        )
        return response

class ServiceContext:
    """Bundles the LLM and callback manager used by the index/query engine."""

    @classmethod
    def from_defaults(cls, llm=None, callback_manager=None):
        instance = cls()
        instance.llm = llm
        instance.callback_manager = callback_manager
        return instance


class VectorStoreIndex:
    def __init__(self, documents=None, service_context=None):
        self.documents = list(documents) if documents else []
        self._service_context = service_context

    @classmethod
    def from_documents(cls, documents, service_context=None):
        return cls(documents=documents, service_context=service_context)

    def as_query_engine(self, similarity_top_k=2, llm=None, service_context=None):
        return QueryEngine(
            index=self,
            similarity_top_k=similarity_top_k,
            llm=llm,
            service_context=service_context,
        )


class QueryEngine:
    def __init__(self, index, similarity_top_k=2, llm=None, service_context=None):
        self.index = index
        self.similarity_top_k = similarity_top_k
        context = service_context or getattr(index, "_service_context", None)
        effective_llm = llm or getattr(context, "llm", None)
        if effective_llm is None:
            raise ValueError(
                "QueryEngine needs an LLM. Pass the LLM (or a service_context "
                "containing it) that is wired to the fine-tuning callback "
                "manager, otherwise no events will be recorded."
            )
        # FIX: the query engine must use the LLM that carries the callback
        # manager holding the fine-tuning handler.
        self.llm = effective_llm

    def _retrieve(self, question):
        docs = self.index.documents[: self.similarity_top_k]
        chunks = []
        for doc in docs:
            text = doc.get("text") if isinstance(doc, dict) else getattr(doc, "text", None)
            if text:
                chunks.append(text)
        return "\n\n".join(chunks)

    def query(self, question):
        context = self._retrieve(question)
        if context:
            prompt = (
                "Context information is below.\n"
                "---------------------\n"
                f"{context}\n"
                "---------------------\n"
                "Given the context information and not prior knowledge, "
                f"answer the query.\nQuery: {question}\nAnswer: "
            )
        else:
            prompt = question
        # FIX: actually run the LLM (through its callback manager) instead of
        # returning None. This is what produces the fine-tuning events.
        return self.llm.predict(prompt)

    def clear_finetuning_events(self):
        self.finetuning_events = []

    def save_finetuning_events(self, path):
        """Write every recorded example to `path` as JSONL.

        FIX: the buggy version always wrote an empty string and printed
        "Wrote 0 examples". Now each recorded example is written as one JSON
        line and the real count is reported.
        """
        FinetuningExample.write_jsonl(self.finetuning_events, path)
        print(f"Wrote {len(self.finetuning_events)} examples to {path}")

class LLM:
    """Deterministic stand-in for `OpenAI(model=..., callback_manager=...)`.

    FIX: the buggy code built an empty stub object
    (`type("LLM", (), {"callback_manager": callback_manager})()`) that could
    not produce completions. The crucial property of this class is that every
    completion is announced through the callback manager, which is how the
    fine-tuning handler observes (question, answer) pairs.
    """

    def __init__(self, model="gpt-3.5-turbo-0613", callback_manager=None):
        self.model = model
        self.callback_manager = (
            callback_manager if callback_manager is not None else CallbackManager()
        )

    def predict(self, prompt):
        """Run a completion and fire the LLM callback events around it."""
        event_id = str(uuid.uuid4())
        self.callback_manager.on_event_start(
            CBEventType.LLM,
            payload={"messages": [{"role": "user", "content": prompt}]},
            event_id=event_id,
        )
        # Deterministic mock completion (no network access required).
        response = f"This is a generated answer to: {prompt}"
        self.callback_manager.on_event_end(
            CBEventType.LLM,
            payload={
                "messages": [{"role": "user", "content": prompt}],
                "response": response,
            },
            event_id=event_id,
        )
        return response


# ---------------------------------------------------------------------------
# Index / query engine plumbing
# ---------------------------------------------------------------------------

class ServiceContext:
    """Bundles the LLM and callback manager used by the index/query engine."""

    @classmethod
    def from_defaults(cls, llm=None, callback_manager=None):
        instance = cls()
        instance.llm = llm
        instance.callback_manager = callback_manager
        return instance


class VectorStoreIndex:
    def __init__(self, documents=None, service_context=None):
        self.documents = list(documents) if documents else []
        self._service_context = service_context

    @classmethod
    def from_documents(cls, documents, service_context=None):
        return cls(documents=documents, service_context=service_context)

    def as_query_engine(self, similarity_top_k=2, llm=None, service_context=None):
        return QueryEngine(
            index=self,
            similarity_top_k=similarity_top_k,
            llm=llm,
            service_context=service_context,
        )


class QueryEngine:
    def __init__(self, index, similarity_top_k=2, llm=None, service_context=None):
        self.index = index
        self.similarity_top_k = similarity_top_k
        context = service_context or getattr(index, "_service_context", None)
        effective_llm = llm or getattr(context, "llm", None)
        if effective_llm is None:
            raise ValueError(
                "QueryEngine needs an LLM. Pass the LLM (or a service_context "
                "containing it) that is wired to the fine-tuning callback "
                "manager, otherwise no events will be recorded."
            )
        # FIX: the query engine must use the LLM that carries the callback
        # manager holding the fine-tuning handler.
        self.llm = effective_llm

    def _retrieve(self, question):
        docs = self.index.documents[: self.similarity_top_k]
        chunks = []
        for doc in docs:
            text = doc.get("text") if isinstance(doc, dict) else getattr(doc, "text", None)
            if text:
                chunks.append(text)
        return "\n\n".join(chunks)

    def query(self, question):
        context = self._retrieve(question)
        if context:
            prompt = (
                "Context information is below.\n"
                "---------------------\n"
                f"{context}\n"
                "---------------------\n"
                "Given the context information and not prior knowledge, "
                f"answer the query.\nQuery: {question}\nAnswer: "
            )
        else:
            prompt = question
        # FIX: actually run the LLM (through its callback manager) instead of
        # returning None. This is what produces the fine-tuning events.
        return self.llm.predict(prompt)

class FineTuningExample:
    """Orchestrates question-answering and fine-tuning event collection."""

    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate a JSONL file of fine-tuning events from the questions."""
        os.makedirs(self.dataset_path, exist_ok=True)

        # 1) Handler + callback manager, mirroring the LlamaIndex recipe.
        callback_manager = CallbackManager()
        finetuning_handler = OpenAIFinetuningHandler(callback_manager=callback_manager)
        callback_manager.add_handler(finetuning_handler)

        # 2) FIX: build a real LLM bound to that callback manager. The buggy
        #    code created an empty stub object, so no LLM call could ever
        #    fire a callback event for the handler to record.
        self.llm = LLM(model="gpt-3.5-turbo-0613", callback_manager=callback_manager)

        questions_path = f"{self.dataset_path}/train_questions.txt"
        if not os.path.exists(questions_path):
            raise FileNotFoundError(f"Questions file not found: {questions_path}")

        questions = []
        with open(questions_path, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if line:
                    questions.append(line)

        try:
            # 3) FIX: pass the service context (LLM + callback manager) when
            #    building the query engine so the queries run through the LLM
            #    that is wired to the fine-tuning handler.
            service_context = ServiceContext.from_defaults(
                llm=self.llm, callback_manager=callback_manager
            )
            index = VectorStoreIndex.from_documents(
                self.documents, service_context=service_context
            )
            query_engine = index.as_query_engine(
                similarity_top_k=2, service_context=service_context
            )

            finetuning_handler.start_trace()
            try:
                for question in questions:
                    response = query_engine.query(question)
                    print(f"Question: {question}")
                    print(f"Response: {response}\n")
            finally:
                finetuning_handler.end_trace()
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                f"{self.dataset_path}/finetuning_events.jsonl"
            )


def prepare_test_data(questions_path):
    os.makedirs(os.path.dirname(questions_path), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path):
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()

    # Sanity check: the JSONL file must now contain one example per question.
    output_path = f"{dataset_path}/finetuning_events.jsonl"
    examples = []
    with open(output_path, "r", encoding="utf-8") as f:
        for line in f:
            if line.strip():
                examples.append(json.loads(line))
    print(f"Verified {len(examples)} fine-tuning examples in {output_path}")
    for example in examples:
        print(json.dumps(example))


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
