import json
import os


class FinetuningEvent:
    """A single LLM exchange (query + response) captured for fine-tuning."""

    def __init__(self, query, response):
        self.query = query
        self.response = response

    def to_dict(self):
        """Convert to the OpenAI chat fine-tuning JSONL format."""
        return {
            "messages": [
                {"role": "system", "content": "You are a helpful assistant."},
                {"role": "user", "content": str(self.query)},
                {"role": "assistant", "content": str(self.response)},
            ]
        }


class OpenAIFineTuningHandler:
    """Collects fine-tuning events while a query engine runs."""

    def __init__(self):
        # FIX: the buggy version had nowhere to store events, so it always
        # wrote an empty file and printed "Wrote 0 examples". This list
        # accumulates every captured LLM exchange.
        self.finetuning_handler = []

    def on_event_end(self, event_type, payload=None, event_id="", **kwargs):
        """Callback invoked by CallbackManager when an LLM call finishes.

        FIX: the signature must match what the callback manager dispatches
        (event_type, payload, event_id). Without this method (or with a
        mismatched signature) events are silently dropped and nothing is
        ever recorded.
        """
        if payload is None:
            return
        query = payload.get("query")
        response = payload.get("response")
        if query is not None and response is not None:
            self.finetuning_handler.append(
                FinetuningEvent(query=query, response=response)
            )

    def save_finetuning_events(self, path):
        """Write every collected event to `path`, one JSON object per line.

        FIX: the buggy version wrote an empty string and always reported
        0 examples. Now it serializes the accumulated events and reports
        the real count.
        """
        with open(path, "w", encoding="utf-8") as f:
            for event in self.finetuning_handler:
                f.write(json.dumps(event.to_dict(), ensure_ascii=False) + "\n")
        print(f"Wrote {len(self.finetuning_handler)} examples to {path}")


class CallbackManager:
    """Fans out callback events to the registered handlers."""

    def __init__(self, handlers):
        self.handlers = handlers

    # FIX: the buggy version only stored handlers and never dispatched
    # anything, so handlers were never notified of LLM events.
    def on_event_end(self, event_type, payload=None):
        for handler in self.handlers:
            if hasattr(handler, "on_event_end"):
                handler.on_event_end(event_type, payload=payload)


class LLM:
    """Mocked LLM whose callback manager fires the LLM_END events that
    OpenAIFineTuningHandler listens for.

    FIX: the buggy version built a bare dynamic object that only carried a
    `callback_manager` attribute. It could not generate a response or emit
    an event, so the fine-tuning handler stayed empty even though the query
    engine "worked". Now every completion dispatches `on_llm_end` carrying
    the query and response payloads.
    """

    def __init__(self, callback_manager):
        self.callback_manager = callback_manager

    def complete(self, question):
        response = f"Response to: {question}"
        if self.callback_manager is not None:
            self.callback_manager.on_event_end(
                "on_llm_end",
                payload={"query": question, "response": response},
            )
        return response


class VectorStoreIndex:

    def __init__(self):
        self.llm = None

    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex()

    def as_query_engine(self, similarity_top_k, llm):
        # FIX: keep a reference to the llm (and therefore its callback
        # manager) so the query engine stays wired to the handler while it
        # produces responses. The buggy version discarded the llm entirely.
        self.llm = llm
        return self

    def query(self, question):
        # FIX: route the query through the LLM so a response is produced
        # and an `on_llm_end` event (query + response) is dispatched to the
        # callback manager, where the fine-tuning handler records it.
        if self.llm is not None and hasattr(self.llm, "complete"):
            return self.llm.complete(question)
        return None


class FineTuningExample:

    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate JSONL file for fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        self.llm = LLM(callback_manager)

        questions = []
        with open(f'{self.dataset_path}/train_questions.txt', "r", encoding='utf-8') as f:
            for line in f:
                line = line.strip()
                if line:
                    questions.append(line)

        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                response = query_engine.query(question)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                f'{self.dataset_path}/finetuning_events.jsonl'
            )


def prepare_test_data(questions_path):
    os.makedirs(os.path.dirname(questions_path), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path):
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
