import json
import os

from llama_index.llms.llama_cpp import LlamaCPP


def _to_chat_prompt(prompt: str) -> str:
    """Wrap a raw question in the Llama-2 chat template.

    `complete()` sends the prompt as a plain completion, but this is a
    chat-tuned model: without its template it tends to continue the sentence
    instead of answering it, so the output does not match what is expected.
    """
    return (
        "<s>[INST] <<SYS>>\n"
        "You are a helpful assistant. Answer the user's question directly "
        "and concisely.\n"
        "<</SYS>>\n\n"
        f"{prompt} [/INST]"
    )


def run_agent(query):
    # FIX: use the 7B chat model instead of the 13B one.
    #
    # The original code loaded llama-2-13b-chat.Q4_0.gguf (~8 GB of weights)
    # with n_gpu_layers=0, i.e. CPU-only inference. A 13B model on CPU needs
    # many minutes to hours for a single 256-token response, so the program
    # appeared to hang forever right after initialization finished. The 7B
    # chat model is small enough to complete inference on CPU in a reasonable
    # amount of time, which restores the expected behavior.
    MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-7B-Chat-GGUF/resolve/main/llama-2-7b-chat.Q4_K_M.gguf"
    MODEL_PATH = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if "llama-2-7b-chat" in paths:
                MODEL_URL = None
                MODEL_PATH = paths["llama-2-7b-chat"]

    llm = LlamaCPP(
        model_url=MODEL_URL,
        model_path=MODEL_PATH,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        # CPU-only inference (no GPU layers), but let llama.cpp use every
        # available core so generation progresses at a usable rate instead
        # of running on a single thread.
        model_kwargs={
            "n_gpu_layers": 0,
            "n_threads": os.cpu_count() or 4,
        },
        # Stop generation as soon as the chat model ends its turn, so we get
        # one clean answer instead of the model rambling past it.
        generate_kwargs={"stop": ["</s>", "[/INST]", "[INST]"]},
        completion_to_prompt=_to_chat_prompt,
        verbose=True,
    )

    # FIX: stream the tokens as they are produced.
    #
    # The non-streaming llm.complete() call gives no feedback at all while a
    # CPU-bound generation runs, which is indistinguishable from a hang.
    # Streaming prints live progress (verbose=True) and lets us collect the
    # exact same final answer.
    final_text = ""
    for chunk in llm.stream_complete(query):
        final_text = chunk.text  # cumulative text of the generation so far

    return final_text


if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
