import json
import os
from llama_index.llms.llama_cpp import LlamaCPP


def run_agent(query):
    # FIX: The original code used the 13B model (llama-2-13b-chat.Q4_0.gguf)
    # with n_gpu_layers=0, i.e. pure CPU inference. A 13B model on CPU generates
    # tokens so slowly (often < 1 token/second) that generating 256 tokens takes
    # tens of minutes to hours, so after initialization the program sits silently
    # and appears to hang forever without ever printing an answer.
    #
    # Use the smaller Llama-2 7B chat model with the Q4_K_M quantization instead:
    # it is small/fast enough to finish inference on CPU in a reasonable time and
    # actually return the expected response.
    MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-7b-Chat-GGUF/resolve/main/llama-2-7b-chat.Q4_K_M.gguf"
    MODEL_PATH = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if "llama-2-7b-chat" in paths:
                MODEL_URL = None
                MODEL_PATH = paths["llama-2-7b-chat"]

    llm = LlamaCPP(
        model_url=MODEL_URL,
        model_path=MODEL_PATH,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        # Run fully on CPU and use all available cores so generation does not
        # stall on a single thread.
        model_kwargs={"n_gpu_layers": 0, "n_threads": os.cpu_count()},
        verbose=True,
    )

    response = llm.complete(query)

    return response


if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
