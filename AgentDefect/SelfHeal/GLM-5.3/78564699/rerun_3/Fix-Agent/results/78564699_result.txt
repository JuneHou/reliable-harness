import json
import os
from llama_index.llms.llama_cpp import LlamaCPP


def run_agent(query):
    # FIX (Use Different Model):
    # The original code loaded "TheBloke/Llama-2-13B-chat-GGUF" in the legacy
    # Q4_0 quantization and forced CPU-only inference (n_gpu_layers=0).
    # A 13B Q4_0 model needs ~10+ GB of RAM once the KV cache for a ~3900
    # token context is allocated, so on a CPU-only machine the process starts
    # swapping right after initialization: complete() is invoked but no token
    # is ever produced and the program appears to hang indefinitely.
    # Q4_0 is also a legacy quantization format that has known compatibility
    # problems with llama-cpp-python builds.
    #
    # Switch to the smaller Llama-2-7B-chat model with the recommended
    # Q4_K_M quantization (~4 GB). It fits in RAM on a normal CPU-only
    # machine and starts generating tokens immediately.
    MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-7B-Chat-GGUF/resolve/main/llama-2-7b-chat.Q4_K_M.gguf"
    MODEL_PATH = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            # Look up the local copy of the (working) 7B chat model.
            if "llama-2-7b-chat" in paths:
                MODEL_URL = None
                MODEL_PATH = paths["llama-2-7b-chat"]

    # Precondition check: if a local path is configured but the file is
    # missing, fall back to downloading the working model instead of
    # silently failing later during inference.
    if MODEL_PATH is not None and not os.path.exists(MODEL_PATH):
        MODEL_PATH = None
        MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-7B-Chat-GGUF/resolve/main/llama-2-7b-chat.Q4_K_M.gguf"

    llm = LlamaCPP(
        model_url=MODEL_URL,
        model_path=MODEL_PATH,
        temperature=0.1,
        max_new_tokens=256,
        # Llama-2 was trained with a 4096 token context window,
        # so 3900 is a safe value (leave room for the prompt).
        context_window=3900,
        # CPU-only inference: no layers offloaded to the GPU.
        model_kwargs={"n_gpu_layers": 0},
        verbose=True,
    )

    response = llm.complete(query)

    return response


if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
