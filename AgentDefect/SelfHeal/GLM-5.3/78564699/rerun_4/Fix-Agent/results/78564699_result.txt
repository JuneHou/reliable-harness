import json
import os

from llama_index.llms.llama_cpp import LlamaCPP
from llama_index.llms.llama_cpp.llama_utils import (
    messages_to_prompt,
    completion_to_prompt,
)


def run_agent(query):
    # Use the recommended K-quant (Q4_K_M) instead of the legacy Q4_0
    # quantization. Q4_0 is a deprecated legacy quant known to produce
    # corrupted / non-terminating output with Llama-2 13B chat models,
    # which makes generation appear to hang forever with no usable output.
    MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-13B-chat-GGUF/resolve/main/llama-2-13b-chat.Q4_K_M.gguf"
    MODEL_PATH = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if "llama-2-13b-chat" in paths:
                MODEL_URL = None
                MODEL_PATH = paths["llama-2-13b-chat"]

    llm = LlamaCPP(
        model_url=MODEL_URL,
        model_path=MODEL_PATH,
        temperature=0.1,
        max_new_tokens=256,
        # llama2 has a context window of 4096 tokens, but we set it lower
        # to allow for some wiggle room
        context_window=3900,
        # run fully on CPU
        model_kwargs={"n_gpu_layers": 0},
        # Llama-2 *chat* models expect the "[INST] <<SYS>>...<</SYS>>...[/INST]"
        # chat template. Without these formatters the raw query string is fed
        # straight into the chat-tuned model, which then never emits its
        # end-of-sequence token, rambles until max_new_tokens is exhausted and
        # never returns a usable answer (looks like an infinite hang).
        messages_to_prompt=messages_to_prompt,
        completion_to_prompt=completion_to_prompt,
        verbose=True,
    )

    response = llm.complete(query)

    return response


if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
