import json
import os

from llama_index.llms.llama_cpp import LlamaCPP
from llama_index.llms.llama_cpp.utils import (
    completion_to_prompt,
    messages_to_prompt,
)


def run_agent(query):
    # -------------------------------------------------------------------------
    # Why the original code appeared to "hang forever":
    #
    #   Nothing was actually deadlocked.  A 13B-parameter Llama-2 model (Q4_0)
    #   running entirely on the CPU (`n_gpu_layers: 0`) only generates roughly
    #   1-3 tokens/second.  `llm.complete()` blocks until the *whole* answer
    #   (up to `max_new_tokens=256`) has been generated, so the process sat
    #   there for many minutes with no visible output -- which looks exactly
    #   like an infinite hang.
    #
    # The fix:
    #   1. Use the smaller / much faster 7B chat model, which runs at a usable
    #      speed on CPU.
    #   2. Stream the answer with `llm.stream_complete()` so tokens are
    #      printed as they are generated -- progress is always visible and
    #      the program clearly terminates.
    # -------------------------------------------------------------------------

    # Smaller model (7B instead of 13B): the 13B model on CPU was far too
    # slow and made the program look like it was hanging.
    MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-7B-Chat-GGUF/resolve/main/llama-2-7b-chat.Q4_0.gguf"
    MODEL_PATH = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if "llama-2-7b-chat" in paths:
                MODEL_URL = None
                MODEL_PATH = paths["llama-2-7b-chat"]

    llm = LlamaCPP(
        model_url=MODEL_URL,
        model_path=MODEL_PATH,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        # Format the prompt correctly for a Llama-2 *chat* model so the
        # answer is a coherent reply instead of raw text continuation.
        messages_to_prompt=messages_to_prompt,
        completion_to_prompt=completion_to_prompt,
        model_kwargs={"n_gpu_layers": 0},  # CPU only
        verbose=True,
    )

    # Stream the response token by token instead of blocking silently until
    # the full completion is ready.  This makes generation progress visible
    # and guarantees the program produces output and finishes.
    response_text = ""
    for chunk in llm.stream_complete(query):
        print(chunk.delta, end="", flush=True)
        response_text += chunk.delta
    print()  # newline after the streamed answer

    return response_text


if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
