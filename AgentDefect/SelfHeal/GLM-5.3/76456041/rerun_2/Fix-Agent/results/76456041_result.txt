import os
import warnings
from typing import List

from dotenv import load_dotenv, find_dotenv

from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.document_loaders import CSVLoader
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.llms import OpenAI
from langchain.prompts import PromptTemplate
from langchain.schema import Document
from langchain.vectorstores import Chroma

warnings.filterwarnings('ignore')

load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")

# ---------------------------------------------------------------------------
# FIX 1 (input data / cost):
# CSVLoader returns ONE Document per CSV row.  Feeding those row-level
# documents to a "map_reduce" chain makes LangChain issue one LLM call per
# row -- that is why the full file was so slow and expensive -- and every map
# call only ever saw a single row, so the combined answer came back wrong or
# incomplete.  Grouping several rows into one document gives the map step
# enough context to actually answer the question and cuts the number of LLM
# calls by roughly ROWS_PER_DOCUMENT.
# ---------------------------------------------------------------------------
ROWS_PER_DOCUMENT = 25

# ---------------------------------------------------------------------------
# FIX 2 (prompts / correctness):
# The default map_reduce prompts are written for "locate relevant text in a
# long document" (the map prompt asks to return text verbatim, and the
# default combine prompt even ends with "Final answer in French:"), which is
# why map_reduce did not answer the question correctly while "stuff" did.
# Both prompts are replaced with prompts that actually answer the question
# and properly merge the partial answers.
# ---------------------------------------------------------------------------
QUESTION_PROMPT = PromptTemplate(
    input_variables=["context", "question"],
    template=(
        "The following is a set of rows taken from a CSV file.\n\n"
        "CSV rows:\n{context}\n\n"
        "Using only the rows above, answer the question as completely as "
        "possible. List every row that matches the question; if none of the "
        "rows match, reply with exactly \"No matching rows\".\n"
        "Question: {question}\n"
        "Answer:"
    ),
)

COMBINE_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=(
        "The following are partial answers, each produced from a different "
        "part of the same CSV file.\n\n"
        "Partial answers:\n{summaries}\n\n"
        "Combine all of the partial answers into a single final answer to the "
        "question. Keep every item that appears in any partial answer, remove "
        "duplicates, ignore any partial answer that says \"No matching rows\", "
        "answer in the same language as the question, and do not invent "
        "information.\n"
        "Question: {question}\n"
        "Final answer:"
    ),
)


def group_rows_into_documents(
    docs: List[Document],
    rows_per_document: int = ROWS_PER_DOCUMENT,
) -> List[Document]:
    """Group the row-level Documents created by CSVLoader into larger
    Documents so that the map step of a map_reduce chain receives several
    rows of context per LLM call instead of a single row."""
    grouped: List[Document] = []
    for start in range(0, len(docs), rows_per_document):
        batch = docs[start:start + rows_per_document]
        grouped.append(
            Document(
                page_content="\n\n".join(d.page_content for d in batch),
                metadata={
                    "source": batch[0].metadata.get("source", ""),
                    "rows": f"{start + 1}-{start + len(batch)}",
                },
            )
        )
    return grouped


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:

            loader = CSVLoader(file_path=file, encoding='utf-8')
            docs = loader.load()

            # Work on grouped rows instead of individual rows: one LLM call
            # per row is what made map_reduce expensive and unreliable.
            docs = group_rows_into_documents(docs)

            embeddings = OpenAIEmbeddings()

            # create the vectorstore to use as the index
            db = Chroma.from_documents(docs, embeddings)
            # expose this index in a retriever interface.
            # Questions such as "Tell me the name of every driver who is
            # German" can only be answered from the whole file, so retrieve
            # every document that was indexed (k = number of documents).
            retriever = db.as_retriever(
                search_type="similarity",
                search_kwargs={"k": max(len(docs), 1)},
            )

            # Build the QA chain once per file (not once per query) and pass
            # the custom map/combine prompts through chain_type_kwargs.
            qa_stuff = RetrievalQA.from_chain_type(
                llm=OpenAI(temperature=0, batch_size=20),
                chain_type="map_reduce",
                retriever=retriever,
                chain_type_kwargs={
                    "question_prompt": QUESTION_PROMPT,
                    "combine_prompt": COMBINE_PROMPT,
                },
                verbose=True
            )

            for query in queries:
                print(query)
                result = qa_stuff.run(query)
                results.append(result)

                print(result)

            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German"
    ]
    res = run_agent(queries)
