import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.prompts import PromptTemplate
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings

warnings.filterwarnings('ignore')

load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    # -------------------------------------------------------------------------
    # FIX: custom prompts for the "map_reduce" chain.
    #
    # CSVLoader turns EVERY ROW of the CSV into its own small document. The
    # default map_reduce prompts are written for a different use case -
    # locating the relevant part of ONE long document:
    #
    #   map (default):    "Use the following portion of a long document to see
    #                      if any of the text is relevant ... Return any
    #                      relevant text verbatim."
    #   reduce (default): "Given the following extracted parts of a long
    #                      document and a question, create a final answer ..."
    #                      (it even contains an example about contract law and
    #                      tells the model to answer "I don't know" if unsure).
    #
    # Applied to hundreds of single-row documents, the map step just echoes raw
    # rows back and the reduce step is fed a misleading prompt, which is why
    # map_reduce returns a wrong answer for the full CSV while "stuff" (a
    # normal question-answering prompt) works fine on a small subset. It also
    # makes the map step return long verbatim rows, which is expensive.
    #
    # The prompts below match what is actually being done:
    #   - map:    answer/extract from each individual row (short answers only),
    #   - reduce: merge all per-row answers into one final complete answer.
    #
    # The placeholder variables must be exactly the ones the chain expects:
    #   question_prompt -> {context} (a single row)      + {question}
    #   combine_prompt  -> {summaries} (per-row answers) + {question}
    #   collapse_prompt -> {summaries} (used when the per-row answers do not
    #                      fit into the model's context window)
    # -------------------------------------------------------------------------
    question_prompt = PromptTemplate(
        input_variables=["context", "question"],
        template=(
            "The following is one row of a CSV file:\n\n"
            "{context}\n\n"
            "Question: {question}\n"
            "If this row contains the information the question asks for, "
            "extract only that information (for example the driver's full "
            "name). If it does not, reply with exactly: None\n"
            "Answer:"
        ),
    )

    combine_prompt = PromptTemplate(
        input_variables=["summaries", "question"],
        template=(
            "The following are answers that were extracted from the individual "
            "rows of a CSV file:\n\n"
            "{summaries}\n\n"
            "Question: {question}\n"
            "Combine every extracted answer into a single final, complete "
            "answer to the question. Ignore all entries that say 'None', "
            "remove duplicates, and do not add any information that is not "
            "contained in the extracted answers. If none of the rows contained "
            "the answer, say that you don't know.\n"
            "Final answer:"
        ),
    )

    collapse_prompt = PromptTemplate(
        input_variables=["summaries"],
        template=(
            "The following are partial answers extracted from the rows of a "
            "CSV file:\n\n"
            "{summaries}\n\n"
            "Merge them into one consolidated answer, keeping every distinct "
            "piece of information and removing duplicates and entries that "
            "say 'None'. Do not add any new information.\n"
            "Consolidated answer:"
        ),
    )

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:

            loader = CSVLoader(file_path=file, encoding='utf-8')
            docs = loader.load()

            embeddings = OpenAIEmbeddings()

            # create the vectorestore to use as the index
            # (a distinct collection per file so that the two CSVs can never
            #  end up mixed together in the same collection)
            db = Chroma.from_documents(
                docs, embeddings, collection_name=file.replace('.', '_')
            )
            # expose this index in a retriever interface
            # k has to cover every row, otherwise exhaustive questions such as
            # "every driver who is German" can never be answered completely;
            # it is bounded by the number of rows (and the original cap of
            # 1000) so we never ask the vector store for more documents than
            # it actually contains.
            retriever = db.as_retriever(
                search_type="similarity",
                search_kwargs={"k": min(len(docs), 1000)},
            )

            # build the QA chain once per file (it does not depend on the query)
            qa_chain = RetrievalQA.from_chain_type(
                llm=OpenAI(temperature=0, batch_size=20),
                chain_type="map_reduce",
                retriever=retriever,
                chain_type_kwargs={
                    "question_prompt": question_prompt,
                    "combine_prompt": combine_prompt,
                    "collapse_prompt": collapse_prompt,
                    # collapse the per-row answers whenever they would not fit
                    # into the context window of the model (large CSV files)
                    "token_max": 3000,
                },
                verbose=True
            )

            for query in queries:
                print(query)
                result = qa_chain.run(query)
                results.append(result)

                print(result)

            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German"
    ]
    res = run_agent(queries)
