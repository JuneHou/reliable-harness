import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.prompts import PromptTemplate

warnings.filterwarnings('ignore')

load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")

# Context budget of the default OpenAI() completion model (text-davinci-003,
# 4096 tokens). A safety margin is kept for the question, the surrounding
# prompt and the answer itself.
MODEL_MAX_CONTEXT_TOKENS = 4096
PROMPT_SAFETY_MARGIN_TOKENS = 1000


def _estimate_num_tokens(text):
    """Rough token estimate (~4 characters per token for OpenAI models)."""
    return max(1, len(text) // 4)


# Prompts for the "refine" chain.
# The default map_reduce map step only asks the model to "return any relevant
# text verbatim" for one single CSV row at a time, and the combine step then
# has to merge hundreds of those independent fragments - that is where drivers
# get lost and the answer stops matching the data. These prompts instead
# explicitly accumulate the answer: everything that has already been found
# must be kept and new matches must be appended.
QUESTION_PROMPT = PromptTemplate(
    input_variables=["context_str", "question"],
    template=(
        "Context information is below.\n"
        "---------------------\n"
        "{context_str}\n"
        "---------------------\n"
        "Given the context information and not prior knowledge, "
        "answer the question: {question}\n"
    ),
)

REFINE_PROMPT = PromptTemplate(
    input_variables=["context_str", "existing_answer", "question"],
    template=(
        "The original question is as follows: {question}\n"
        "We have provided an existing answer: {existing_answer}\n"
        "We have the opportunity to refine the existing answer"
        "(only if needed) with some more context below.\n"
        "------------\n"
        "{context_str}\n"
        "------------\n"
        "Given the new context, refine the original answer to better answer "
        "the question. Keep every item that is already part of the existing "
        "answer and append any new item that matches the question. "
        "If the context isn't useful, return the original answer."
    ),
)


def _pick_chain_type(docs):
    """Return (chain_type, chain_type_kwargs) suited to the amount of data.

    * "stuff" puts all documents into a single prompt: one cheap LLM call and
      a correct answer, but it only works while the whole file still fits into
      the model's context window (that is why "stuff" worked fine on the
      subset of the data).
    * "refine" loops over the documents one by one and iteratively updates
      the answer, so the information of EVERY row is taken into account and
      no driver is dropped - this is the chain type designed for questions
      that need information from all documents.
    """
    total_doc_tokens = sum(_estimate_num_tokens(d.page_content) for d in docs)

    if total_doc_tokens <= (MODEL_MAX_CONTEXT_TOKENS - PROMPT_SAFETY_MARGIN_TOKENS):
        # Everything fits into one prompt -> cheapest and most accurate option.
        return "stuff", {}

    # Too much data for a single prompt -> refine the answer over every
    # document so that nothing is lost when combining.
    return "refine", {
        "question_prompt": QUESTION_PROMPT,
        "refine_prompt": REFINE_PROMPT,
    }


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:

            loader = CSVLoader(file_path=file, encoding='utf-8')
            docs = loader.load()

            embeddings = OpenAIEmbeddings()

            # create the vectorstore to use as the index
            db = Chroma.from_documents(docs, embeddings)
            # expose this index in a retriever interface.
            # The question has to be answered over the WHOLE file, so every
            # row has to come back: k is set to the number of documents.
            # (The previous k=1000 silently dropped rows as soon as a file
            # contained more than 1000 documents.)
            retriever = db.as_retriever(
                search_type="similarity",
                search_kwargs={"k": len(docs)},
            )

            # NOTE: "map_reduce" was the wrong chain type for this kind of
            # question ("every driver who ..."):
            #   * its map step runs the question against each single CSV row
            #     (k=1000 -> up to 1000 LLM calls per query plus the combine
            #     calls), which is what made it so expensive, and
            #   * its reduce/combine step then merges hundreds of independent
            #     fragments and loses entries along the way, which is why the
            #     final answer did not match the data.
            # "stuff" is used when the file fits into the context window
            # (correct and cheap), "refine" is used for the full file.
            chain_type, chain_type_kwargs = _pick_chain_type(docs)

            qa_chain = RetrievalQA.from_chain_type(
                llm=OpenAI(temperature=0, batch_size=20),
                chain_type=chain_type,
                chain_type_kwargs=chain_type_kwargs,
                retriever=retriever,
                verbose=True,
            )

            for query in queries:
                print(query)
                result = qa_chain.run(query)
                results.append(result)

                print(result)

            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German"
    ]
    res = run_agent(queries)
