import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.prompts import PromptTemplate
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings

warnings.filterwarnings('ignore')

load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


# ---------------------------------------------------------------------------
# Why the original code gave wrong answers on the full CSV
# ---------------------------------------------------------------------------
# CSVLoader creates ONE document per CSV row, so the full file becomes several
# hundred tiny documents.  With chain_type="map_reduce" the question is sent
# to the LLM once per row (the "map" step - this is where the huge cost comes
# from) and the partial answers are then merged by a "combine" step.
#
# The DEFAULT map/combine prompts are generic QA prompts ("Use the following
# pieces of context to answer the question ... If you don't know the answer,
# just say that you don't know").  Applied to a single CSV row at a time they
# make the model answer "I don't know" for most rows.  The combine step then
# receives a wall of partial answers that is longer than the model's context
# window, so LangChain recursively collapses/re-summarises it and names get
# lost, the "just say you don't know" wording biases it towards incomplete
# answers, and the default max_tokens=256 cuts the final list short.
# ("stuff" works on the small file simply because every row then fits into a
# single prompt, so the model can see all rows at once.)
#
# The fix: give the map step an explicit extraction prompt (return the name of
# the driver when the row matches the question, otherwise NONE) and the
# combine step an explicit merging prompt (keep every name that was found),
# raise max_tokens so long lists are not truncated, and retrieve every row.
# ---------------------------------------------------------------------------

QUESTION_PROMPT_TEMPLATE = """The text below is a single record taken from a CSV file that contains racing drivers.
Read this one record and answer the question using only the information it contains.
If this record does not match what the question asks for, reply with the single word NONE.
Never invent information that is not in the record.

Record:
{context}

Question: {question}
Answer (the driver's full name if this record matches the question, otherwise NONE):"""

QUESTION_PROMPT = PromptTemplate(
    template=QUESTION_PROMPT_TEMPLATE,
    input_variables=["context", "question"],
)

COMBINE_PROMPT_TEMPLATE = """The text below lists the answers that were extracted from every record of a CSV file of racing drivers (one answer per record; NONE means that the record did not match the question).
Merge all of these answers into one final, complete answer to the question.
Include EVERY driver name that was extracted - do not summarise, do not shorten the list and do not add anyone who is not mentioned.
Answer with a plain comma-separated list of names.
If every extracted answer is NONE, say that you don't know.

Question: {question}

Extracted answers:
{summaries}

Final answer:"""

COMBINE_PROMPT = PromptTemplate(
    template=COMBINE_PROMPT_TEMPLATE,
    input_variables=["summaries", "question"],
)


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:

            loader = CSVLoader(file_path=file, encoding='utf-8')
            docs = loader.load()

            embeddings = OpenAIEmbeddings()

            # create the vectorestore to use as the index
            db = Chroma.from_documents(docs, embeddings)
            # expose this index in a retriever interface.
            # k is set to the number of rows so that the similarity search
            # returns EVERY row - exhaustive questions ("every driver who is
            # German") must not silently drop rows.
            retriever = db.as_retriever(
                search_type="similarity",
                search_kwargs={"k": len(docs)},
            )

            for query in queries:
                qa_chain = RetrievalQA.from_chain_type(
                    llm=OpenAI(
                        temperature=0,
                        batch_size=20,
                        # default is 256, which truncates long lists of names
                        max_tokens=800,
                    ),
                    chain_type="map_reduce",
                    chain_type_kwargs={
                        # map step: extract the answer from a single record
                        "question_prompt": QUESTION_PROMPT,
                        # combine step: merge everything that was extracted
                        "combine_prompt": COMBINE_PROMPT,
                    },
                    retriever=retriever,
                    verbose=True,
                )

                print(query)
                result = qa_chain.run(query)
                results.append(result)

                print(result)

            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German"
    ]
    res = run_agent(queries)
