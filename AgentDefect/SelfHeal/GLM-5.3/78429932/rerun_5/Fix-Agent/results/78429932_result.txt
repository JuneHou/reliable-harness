def run_agent(user_prompt, system_prompt):
    # Use the chat-style wrapper: Ollama applies the correct Llama 3 chat

    llm = ChatOllama(model="llama3")

    # Properly structured messages with valid roles (system / user).

    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # A single chat completion: one response, no repeated system text.
    response = llm.invoke(messages)

    # Return only the assistant's message content, not the whole object.
    return response.content
