from langchain_community.chat_models import ChatOllama
from langchain_core.messages import HumanMessage, SystemMessage


def run_agent(user_prompt, system_prompt):
    # Use ChatOllama (a chat model) instead of the raw-completion Ollama wrapper.
    # Do NOT manually embed Llama 3 special tokens (<|begin_of_text|>,
    # <|start_header_id|>, <|eot_id|>, ...): Ollama applies the model's chat
    # template automatically. Injecting them by hand makes the model treat the
    # tokens as literal conversation turns, causing it to generate multiple
    # responses and repeat the system prompt content.
    llm = ChatOllama(model="llama3")

    # Send properly structured messages with valid roles so Ollama formats
    # them correctly into a single system + user exchange.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    response = llm.invoke(messages)
    # Return only the assistant's reply text (a single, formatted response).
    return response.content


if __name__ == "__main__":
    print(run_agent("Explain in detail, step by step, how artificial intelligence models like yourself are built, including data, training, architecture, and deployment. Use examples and provide a comprehensive overview.", "You are a helpful assistant."))
