from langchain_ollama import ChatOllama
from langchain_core.messages import SystemMessage, HumanMessage


def run_agent(user_prompt, system_prompt):
    # Llama 3 is a chat model, so it must be invoked through a chat interface
    # (ChatOllama), not a completion-style LLM wrapper. The chat interface sends
    # structured messages with valid roles, and the Ollama server applies the
    # Llama 3 chat template (<|begin_of_text|>, <|start_header_id|>, <|eot_id|>...)
    # exactly once. Manually embedding those special tokens in the prompt (as the
    # buggy version did) makes Ollama wrap the text in a SECOND template, so the
    # model sees repeated system sections and emits multiple/duplicated responses.
    llm = ChatOllama(model="llama3")

    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    response = llm.invoke(messages)

    # A single, properly formatted AIMessage is returned.
    return response.content


if __name__ == "__main__":

        "You are a helpful assistant.",
