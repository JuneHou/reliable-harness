from langchain_ollama import ChatOllama
from langchain_core.messages import SystemMessage, HumanMessage


def run_agent(user_prompt, system_prompt):
    # Use the chat interface (ChatOllama) rather than the raw completion-style
    # Ollama LLM wrapper. The chat wrapper applies Llama 3's chat template
    # exactly once, server-side, so we must NOT hand-write the special tokens
    # (<|begin_of_text|>, <|start_header_id|>, <|eot_id|>) ourselves.
    # Hand-writing them makes Ollama apply the template twice, so the model
    # treats the tokens as ordinary text, completes the user's turn itself and
    # keeps generating extra turns -> multiple responses / repeated system
    # prompt content.
    llm = ChatOllama(
        model="llama3",
        num_predict=2048,   # bound generation to a single complete answer
        temperature=0.1,    # deterministic, single focused response
    )

    # Send properly structured message objects with valid roles.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # Single invoke -> single response, already extracted from the chat turn.
    response = llm.invoke(messages)
    return response.content


if __name__ == "__main__":
