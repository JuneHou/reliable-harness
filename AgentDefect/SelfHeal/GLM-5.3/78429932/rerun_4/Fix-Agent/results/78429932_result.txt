from langchain_ollama import ChatOllama
from langchain_core.messages import SystemMessage, HumanMessage


def run_agent(user_prompt, system_prompt):
    # Use the chat model so Ollama applies the llama3 chat template
    # itself. We must NOT manually inject special tokens like
    # <|begin_of_text|>, <|start_header_id|>, <|eot_id|> -- doing so makes
    # the model treat them as literal text and mimic the template,
    # producing repeated system content and multiple assistant turns.
    llm = ChatOllama(model="llama3")

    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # Properly structured messages with valid roles; the framework adds
    # the correct template formatting, so we get a single clean response.
    response = llm.invoke(messages)
    return response.content
