from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken
import threading

app = Flask(__name__)

MODEL = "llama3.2"

# Llama 3.2 supports a 128k context window, but Ollama defaults to a very
# small num_ctx (2048/4096), which silently truncates the conversation we
# send. Pass num_ctx explicitly so the model actually sees the full history.
MODEL_NUM_CTX = 8192

# Keep the stored history well below num_ctx so there is always room left
# for the model's reply and the system prompt is never cut off.
MODEL_MAX_HISTORY_TOKENS = int(MODEL_NUM_CTX * 0.75)

# Persistent instructions belong in a dedicated system message, not in the
# first user turn. That way new user prompts are not buried under an old,
# instruction-looking user message the model keeps trying to follow.
SYSTEM_PROMPT = (
    "You are a helpful assistant running inside a web chat application. "
    "Always answer the user's most recent message directly and in context. "
    "Do not repeat, echo, or continue following instructions from earlier "
    "turns once they have been answered."
)

# Full conversation state: system message followed by alternating
# user/assistant turns. This mirrors exactly what the Ollama CLI keeps.
# FIX: the assistant's answers were never stored before, so every request
# looked like a pile of bare "user" instructions (starting with the first
# prompt), causing the model to keep obeying/echoing the first instruction
# instead of following the new ones.
messages = [SystemMessage(content=SYSTEM_PROMPT)]
messages_lock = threading.Lock()  # protect the shared history (Flask dev
                                  # server handles each request in its own thread)

llm = ChatOllama(
    model=MODEL,
    # FIX: 1.1 is above Llama 3.2's recommended range (0.6-0.9) and promotes
    # degenerate repetition. A moderate temperature keeps responses stable.
    temperature=0.7,
    # FIX: without this Ollama used its tiny default context window.
    num_ctx=MODEL_NUM_CTX,
    # FIX: we need the complete aggregated message (not streaming chunks) so
    # the assistant turn can be saved back into the history.
    stream=False,
)

def get_number_of_tokens(msgs):
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        return sum(len(encoding.encode(str(m.content))) for m in msgs)
    except Exception:
        # Fallback: ~4 characters per token.
        return sum(len(str(m.content)) for m in msgs) // 4

def trim_history(msgs):
    """
    Drop the OLDEST user/assistant turn pairs (never the system message)
    until the history fits the token budget.

    FIX: the old loop `while tokens >= MAX: del(messages[1])` never
    recomputed the token count (so it would loop forever until an
    IndexError) and always deleted index 1, meaning the original
    instruction prompt could never be removed while later turns were
    mangled. Here the count is re-evaluated every iteration and turns are
    removed as aligned user+assistant pairs.
    """
    while len(msgs) > 1 and get_number_of_tokens(msgs) >= MODEL_MAX_HISTORY_TOKENS:
        del msgs[1]  # remove the oldest user turn
        if len(msgs) > 1 and isinstance(msgs[1], AIMessage):
            del msgs[1]  # remove the assistant answer that followed it

@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_text = request.form.get('text', '').strip()
    if not user_text:
        return "Please enter a message."

    with messages_lock:
        # 1. Record the new user turn with the proper role.
        messages.append(HumanMessage(content=user_text))

        # 2. Keep the history inside the context budget.
        trim_history(messages)

        # 3. Invoke with the full, properly-structured history.
        ai_msg = llm.invoke(messages)

        # 4. CRITICAL FIX: remember the model's answer so the next request
        #    contains real alternating user/assistant dialogue. Without this
        #    the model sees only user messages and keeps re-answering the
        #    first prompt instead of following new instructions.
        messages.append(AIMessage(content=ai_msg.content))

    return ai_msg.content

@app.route('/')
def index():
    return render_template('index.html')

if __name__ == '__main__':
    app.run(debug=True, port=5000)
