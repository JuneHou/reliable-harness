from flask import Flask, render_template, request
from ollama import chat
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000
# Keep the conversation bounded well below the context limit to leave room
# for the model's response. Using only the most recent history also keeps the
# Ollama context window (num_ctx) small enough to run quickly.
MAX_HISTORY_MESSAGES = 40  # 20 user turns + 20 assistant replies

messages = []  # alternating: [SystemMessage?, HumanMessage, AIMessage, HumanMessage, ...]

# FIX 1: temperature must stay within Meta's recommended range for
# Llama-3 family models (0.6 - 0.9). A value > 1.0 (like the old 1.1)
# produces degraded, repetitive output that keeps re-executing earlier
# instructions instead of following the new ones.
# FIX 2: pass num_ctx explicitly instead of relying on Ollama's default
# (2048), so the full prompt history plus the reply actually fits.
# FIX 3: stream=False -- the endpoint returns one complete string, so there
# is no reason to request a streaming response.
llm = ChatOllama(
    model=MODEL,
    temperature=0.8,
    num_ctx=8192,
    stream=False,
)


def get_number_of_tokens(msgs):
    """Estimate tokens, including roles/formatting overhead (chat template tags),
    and re-compute safely instead of relying on a stale count."""
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total_tokens = 0
        for m in msgs:
            # +4 accounts for role name and surrounding template tags
            total_tokens += len(encoding.encode(str(m.content))) + 4
        return total_tokens
    except Exception:
        total_chars = sum(len(str(m.content)) for m in msgs)
        # ~4 chars per token is the standard fallback estimate
        return total_chars // 4


def trim_history():
    """Drop the oldest turns (never the system message) until the history fits."""
    while get_number_of_tokens(messages) >= MODEL_MAX_TOKENS and len(messages) > 1:
        # Remove the oldest non-system message (keep the SystemMessage at index 0)
        del messages[1]


@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_text = request.form['text']
    messages.append(HumanMessage(content=user_text))

    # Keep the conversation short enough to fit the context window.
    trim_history()
    # Also cap the number of messages retained (simple sliding window).
    while len(messages) > MAX_HISTORY_MESSAGES:
        del messages[1]

    # The whole point of the fix: send a properly structured conversation
    # (system + strictly alternating HumanMessage/AIMessage turns) so the
    # model answers the *latest* user message instead of looping over the
    # first one's instructions.
    ai_msg = llm.invoke(messages)

    # FIX 4 (the main behavioral fix): store the model's reply in the
    # conversation. Without the assistant turns, the history is a flat list
    # of consecutive user messages; Llama-3's chat template then makes the
    # model "complete" the missing assistant turns itself, so it keeps
    # repeating the first prompt's instructions and ignores new prompts.
    messages.append(AIMessage(content=ai_msg.content))

    return ai_msg.content


@app.route('/')
def index():
    return render_template('index.html')


if __name__ == '__main__':
    app.run(debug=True, port=5000)
