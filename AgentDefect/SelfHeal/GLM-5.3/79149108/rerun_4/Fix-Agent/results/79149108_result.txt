from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000

# Ollama truncates the prompt to its own context window (num_ctx), which
# defaults to only 2048/4096 tokens regardless of the model's 128k capability.
# If the conversation outgrows num_ctx, messages silently vanish from the
# model's view. Raise num_ctx explicitly and trim the history ourselves.
OLLAMA_NUM_CTX = 8192
RESPONSE_TOKEN_BUDGET = 1024
MAX_HISTORY_TOKENS = OLLAMA_NUM_CTX - RESPONSE_TOKEN_BUDGET

SYSTEM_PROMPT = (
    "You are a helpful assistant in a web chat. Always answer the user's "
    "most recent message. When the user gives a new instruction, treat it "
    "as overriding any earlier instruction."
)

# CRITICAL FIX #1: the conversation history must hold proper message objects
# for BOTH sides of the conversation (user *and* assistant turns), alternating.
# The original code only ever appended user messages as raw tuples, so the
# model never saw its own replies: from its point of view the first prompt was
# never answered, which is why it kept repeating that first prompt's
# instructions instead of following the new ones.
messages = [SystemMessage(content=SYSTEM_PROMPT)]

llm = ChatOllama(
    model=MODEL,
    # CRITICAL FIX #2: temperature=1.1 is above Llama3.2's recommended range
    # and produced erratic, repetitive, low-instruction-following output.
    # 0.7 keeps replies coherent and instruction-following.
    temperature=0.7,
    # CRITICAL FIX #3: make the real Ollama context window explicit so the
    # whole history actually reaches the model.
    num_ctx=OLLAMA_NUM_CTX,
    # CRITICAL FIX #4: we return one complete answer, not a stream, so
    # streaming must be disabled (previously stream=True was mixed with
    # .invoke()/.content).
    stream=False,
)


def get_number_of_tokens(msgs):
    """Approximate token count for the message history.

    tiktoken's GPT-3.5 encoding is a good enough proxy for Llama; fall back
    to a chars/4 estimate if the encoding cannot be loaded.
    """
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        return sum(len(encoding.encode(m.content)) for m in msgs)
    except Exception:
        return sum(len(m.content) for m in msgs) // 4


def trim_history():
    """Drop the OLDEST user/assistant pairs (never the system prompt) until
    the history fits the context budget.

    The original loop computed the token count once OUTSIDE the `while` and
    then only deleted entries, so it could never terminate; worse, it deleted
    index 1 (everything after the first message), which kept only the very
    first prompt -- exactly the "stuck on the first instructions" symptom.
    Recomputing the count on every iteration fixes both problems.
    """
    while get_number_of_tokens(messages) >= MAX_HISTORY_TOKENS:
        start = 1 if isinstance(messages[0], SystemMessage) else 0
        # Always keep the system message plus at least the newest user turn.
        if len(messages) - start <= 1:
            break
        # Remove one user/assistant pair so turns stay alternating.
        del messages[start:start + 2]


@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_text = (request.form.get('text') or '').strip()
    if not user_text:
        return ""

    messages.append(HumanMessage(content=user_text))
    trim_history()

    # CRITICAL FIX #5: pass proper LangChain message objects. The original
    # code passed raw ("role", text) tuples, which ChatOllama does not expect.
    ai_msg = llm.invoke(messages)
    answer = ai_msg.content
    if not isinstance(answer, str):  # defensive: some models return lists
        answer = "".join(
            part if isinstance(part, str) else str(part) for part in answer
        )

    # CRITICAL FIX #6: record the assistant's reply in the history. Without
    # this turn-taking, every request looks like one long unanswered stream
    # of user instructions, so the model keeps re-answering the first one.
    messages.append(AIMessage(content=answer))

    return answer


@app.route('/')
def index():
    return render_template('index.html')


if __name__ == '__main__':
    app.run(debug=True, port=5000)
