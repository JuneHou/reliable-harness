from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000
messages = []

llm = ChatOllama(
    model=MODEL,
    temperature=0.7,
    stream=True,
    num_ctx=8192,
)

def get_number_of_tokens(messages):
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total_tokens = 0
        for message in messages:
            total_tokens += len(encoding.encode(str(message.content)))
        return total_tokens
    except:
        total_chars = sum(len(str(message.content)) for message in messages)
        return total_chars // 4

def trim_history():
    """Drop the oldest *conversation turns* (never index 0, which is the
    system prompt) until we are back under the model's context limit."""
    number_of_tokens = get_number_of_tokens(messages)
    while number_of_tokens >= MODEL_MAX_TOKENS and len(messages) > 1:
        del messages[1]  # keep the system prompt at index 0
        number_of_tokens = get_number_of_tokens(messages)

@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_text = request.form.get('text', '')

    # Build a proper message history.
    # Keep an explicit system prompt pinned at index 0 so the model's core
    # instructions are always present and are never the first thing trimmed.
    if not messages or not isinstance(messages[0], SystemMessage):
        messages.insert(0, SystemMessage(
            content="You are a helpful assistant. Answer the user's latest question directly "
                    "and concisely. Do not repeat or re-answer earlier questions."
        ))

    messages.append(HumanMessage(content=user_text))
    trim_history()

    try:
        ai_msg = llm.invoke(messages)
    except Exception as e:
        return f"Error calling the model: {e}", 500

    # CRITICAL FIX: store the model's reply so the next request can see it.
    # Without this the conversation looks like an endless stream of user
    # turns with no answers, which is what made the model keep
    # re-answering/repeating the first prompt's instructions.
    messages.append(AIMessage(content=ai_msg.content))

    return ai_msg.content

@app.route('/')
def index():
    return render_template('index.html')

if __name__ == '__main__':
    app.run(debug=True, port=5000)
