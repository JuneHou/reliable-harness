from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"

# Ollama does not serve llama3.2 with its full 128k theoretical context by
# default; it uses a small window (num_ctx, commonly 2048/4096/8192).
# Keep the token budget and num_ctx in sync so history is never silently
# truncated by the server.
MODEL_MAX_TOKENS = 8192

# Persistent instructions belong in a proper system message (which is what
# the Ollama CLI effectively does), NOT inside the first user prompt.
SYSTEM_PROMPT = "You are a helpful, concise assistant."
messages = [SystemMessage(content=SYSTEM_PROMPT)]

llm = ChatOllama(
    model=MODEL,
    temperature=0.7,          # 1.1 was too hot -> erratic/repetitive output
    num_ctx=MODEL_MAX_TOKENS, # context window that actually fits our budget
    stream=True,
)


def get_number_of_tokens(msgs):
    """Approximate token count for a list of LangChain message objects."""
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        return sum(len(encoding.encode(m.content)) for m in msgs)
    except Exception:
        # Fallback: rough estimate (~4 characters per token)
        return sum(len(str(m.content)) for m in msgs) // 4


def trim_history():
    """
    Drop the oldest conversation turns (never the system prompt) while the
    history is over budget. The token count is recomputed on every iteration
    so the loop always terminates.
    """
    while len(messages) > 1 and get_number_of_tokens(messages) >= MODEL_MAX_TOKENS:
        del messages[1]  # messages[0] is the system prompt and must survive


@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_text = (request.form.get('text') or '').strip()
    if not user_text:
        return "No text provided.", 400

    # Proper LangChain message object with an explicit role.
    messages.append(HumanMessage(content=user_text))
    trim_history()

    try:
        ai_msg = llm.invoke(messages)
    except Exception as exc:
        # Roll the user turn back so a failed request doesn't poison history.
        messages.pop()
        return f"Error talking to the model: {exc}", 500

    # ------------------------------------------------------------------
    # THE key fix: store the assistant's reply in the conversation history.
    #
    # Previously the app only ever appended ("user", text) tuples and never
    # recorded the model's answers, so the model was shown a wall of
    # back-to-back user messages with no assistant turns (and in a data
    # format ChatOllama does not properly understand). It therefore kept
    # completing the FIRST prompt's instructions over and over instead of
    # following the new ones -- exactly the difference from a real chat
    # session in the Ollama CLI, where every assistant reply is fed back
    # into the model as context.
    # ------------------------------------------------------------------
    messages.append(AIMessage(content=ai_msg.content))

    return ai_msg.content


@app.route('/')
def index():
    return render_template('index.html')


if __name__ == '__main__':
    app.run(debug=True, port=5000)
