Why the script previously "stopped silently" after the embedding step:
  1. The nested try/except around Chroma.from_documents swallowed the real
     exception (Ollama server not running, model not pulled, or a
     chromadb/sqlite3 incompatibility) and only logged a bare message.

     statement (`if vector_db:`) raised UnboundLocalError, which was again
     swallowed by the outer handler -> run_agent() quietly returned None and
     the expected RAG answer was never produced.
  3. logging.error(f"...{e}") hides the traceback, so the failure looked
     like the script simply stopped.

The corrected version surfaces the root cause with a full traceback,
verifies preconditions (PDF present, Ollama up, models pulled) up front,
never references a possibly-unbound variable, and prefers the maintained
langchain_ollama / langchain_chroma packages when available.
"""

import json
import logging
import os
import sqlite3 as _stdlib_sqlite3
import sys
from pathlib import Path
from urllib.request import urlopen

# chromadb requires sqlite3 >= 3.35.0. Some Python builds (notably on Windows)
# bundle an older sqlite, which makes Chroma fail. Swap in pysqlite3 when needed
# and available. This must run before chromadb gets imported by LangChain.
if _stdlib_sqlite3.sqlite_version_info < (3, 35, 0):
    try:
        import pysqlite3

        sys.modules["sqlite3"] = sys.modules.pop("pysqlite3")
    except ImportError:
        # pysqlite3-binary not installed; chromadb will report it clearly.
        pass

# Prefer the maintained packages; fall back to the deprecated community ones
# for older environments.
try:
    from langchain_chroma import Chroma
except ImportError:
    from langchain_community.vectorstores import Chroma

try:
    from langchain_ollama import ChatOllama, OllamaEmbeddings
except ImportError:
    from langchain_community.chat_models import ChatOllama
    from langchain_community.embeddings import OllamaEmbeddings

from langchain.prompts import ChatPromptTemplate, PromptTemplate
from langchain.retrievers.multi_query import MultiQueryRetriever
from langchain_community.document_loaders import UnstructuredPDFLoader
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough
from langchain_text_splitters import RecursiveCharacterTextSplitter

logging.basicConfig(level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s")
logger = logging.getLogger(__name__)

OLLAMA_BASE_URL = os.environ.get("OLLAMA_BASE_URL", "http://localhost:11434")
EMBEDDING_MODEL = "nomic-embed-text"
LLM_MODEL = "llama3"


def check_ollama_ready():
    """Fail fast with an actionable message if the Ollama server or a required
    model is missing. Previously these problems only surfaced deep inside
    Chroma.from_documents and were swallowed, which looked like a silent stop."""
    try:
        with urlopen(f"{OLLAMA_BASE_URL}/api/tags", timeout=5) as response:
            payload = json.loads(response.read().decode("utf-8"))
    except (OSError, ValueError) as exc:
        raise RuntimeError(
            f"Cannot reach the Ollama server at {OLLAMA_BASE_URL}: {exc}. "
            "Is Ollama running? Start it with `ollama serve` and try again."
        ) from exc

    installed = {model.get("name", "") for model in payload.get("models", [])}
    for model in (EMBEDDING_MODEL, LLM_MODEL):
        if not any(name == model or name.split(":", 1)[0] == model for name in installed):
            raise RuntimeError(
                f"Model '{model}' is not installed in your local Ollama instance "
                f"(installed models: {sorted(name for name in installed if name)}). "
                f"Download it with: ollama pull {model}"
            )
    logger.info("Ollama is running and both required models are available")


def load_and_split_pdf(local_path):
    """Load the PDF and split it into chunks, validating each precondition."""
    pdf_path = Path(local_path)
    if not pdf_path.is_file():
        raise FileNotFoundError(
            f"PDF file not found: {pdf_path.resolve()}. Place the file next to "
            "this script or pass a valid path."
        )

    loader = UnstructuredPDFLoader(file_path=str(pdf_path))
    data = loader.load()
    logger.info("Loading of PDF is done")
    if not data or not data[0].page_content.strip():
        raise ValueError(f"No text could be extracted from {pdf_path.name}")
    logger.info(f"First page content preview: {data[0].page_content[:500]}...")

    text_splitter = RecursiveCharacterTextSplitter(chunk_size=7500, chunk_overlap=100)
    logger.info("Text splitter created")
    chunks = text_splitter.split_documents(data)
    if not chunks:
        raise ValueError("Text splitting produced no chunks; check the PDF content.")
    logger.info(f"Created {len(chunks)} chunks")
    return chunks


def build_vector_db(chunks):
    """Embed the chunks into Chroma. Errors here are no longer swallowed:
    they propagate with a full traceback so the real cause is visible."""
    embedding_model = OllamaEmbeddings(model=EMBEDDING_MODEL, base_url=OLLAMA_BASE_URL)
    logger.info("Creating vector db (embedding %d chunks via Ollama)...", len(chunks))
    vector_db = Chroma.from_documents(
        documents=chunks,
        embedding=embedding_model,
        collection_name="local-rag",
    )
    if vector_db is None:
        raise RuntimeError("Chroma.from_documents failed: no vector store was returned.")
    logger.info("Local db created successfully")
    logger.info("Vector db verification successful")
    return vector_db


def run_agent(question, local_path="./paper1.pdf"):
    # Fail fast on missing prerequisites instead of dying silently later.
    check_ollama_ready()
    chunks = load_and_split_pdf(local_path)
    vector_db = build_vector_db(chunks)

    # LLM from Ollama
    llm = ChatOllama(model=LLM_MODEL, base_url=OLLAMA_BASE_URL)
    logger.info("LLM model loaded")

    QUERY_PROMPT = PromptTemplate(
        input_variables=["question"],
        template="""You are an AI language model assistant. Your task is to generate five
different versions of the given user question to retrieve relevant documents from
a vector database. By generating multiple perspectives on the user question, your
goal is to help the user overcome some of the limitations of the distance-based
similarity search. Provide these alternative questions separated by newlines.
Original question: {question}""",
    )
    logger.info("Query prompt created")

    retriever = MultiQueryRetriever.from_llm(
        retriever=vector_db.as_retriever(),
        llm=llm,
        prompt=QUERY_PROMPT,
    )
    logger.info("Retriever created")

    # RAG prompt
    template = """Answer the question based ONLY on the following context:
{context}
Question: {question}
"""
    prompt = ChatPromptTemplate.from_template(template)
    logger.info("RAG prompt created")

    chain = (
        {"context": retriever, "question": RunnablePassthrough()}
        | prompt
        | llm
        | StrOutputParser()
    )
    logger.info("Chain created")

    response = chain.invoke(question)
    logger.info("Chain invoked")
    logger.info(f"Response: {response}")

    return response


if __name__ == "__main__":
    question = "What are the 5 pillars of global cooperation?"
    try:
        response = run_agent(question)
        print(f"\nFinal Output: {response}")
    except Exception:
        # Log the FULL traceback instead of swallowing the error, so the actual
        # root cause is visible rather than the script just "stopping".
        logger.exception("RAG pipeline failed")
        sys.exit(1)
