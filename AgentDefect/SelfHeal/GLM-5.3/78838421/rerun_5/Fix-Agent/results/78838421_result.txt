import logging
from langchain_community.document_loaders import UnstructuredPDFLoader
from langchain_community.embeddings import OllamaEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain.prompts import ChatPromptTemplate, PromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_community.chat_models import ChatOllama
from langchain_core.runnables import RunnablePassthrough
from langchain.retrievers.multi_query import MultiQueryRetriever

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')


def run_agent(question):
    local_path = "./paper1.pdf"

    # FIX: initialize vector_db up front so it is always defined, even if the
    # embedding / Chroma creation step raises an exception. Previously, when the
    # inner try/except swallowed the error, the following "if vector_db:" check
    # raised an UnboundLocalError, which the outer except silently converted
    # into "return None" (the script appeared to just stop after embedding).
    vector_db = None

    try:
        # Local PDF file uploads
        if local_path:
            loader = UnstructuredPDFLoader(file_path=local_path)
            data = loader.load()
            logging.info("Loading of PDF is done")
        else:
            logging.error("Upload a PDF file")
            raise ValueError("No PDF file uploaded")

        # Preview first page
        logging.info(f"First page content preview: {data[0].page_content[:500]}...")

        # Split and chunk
        text_splitter = RecursiveCharacterTextSplitter(chunk_size=7500, chunk_overlap=100)
        logging.info("Text splitter created")
        chunks = text_splitter.split_documents(data)
        logging.info(f"Created {len(chunks)} chunks")

        logging.info("Creating Vector db")
        try:
            embedding_model = OllamaEmbeddings(model="nomic-embed-text", show_progress=True)
            logging.info(f"Embedding model created: {embedding_model}")

            # This is the step that actually talks to the Ollama server. If the
            # server is down or the model is missing, an exception is raised here.
            vector_db = Chroma.from_documents(
                documents=chunks,
                embedding=embedding_model,
                collection_name="local-rag"
            )
            logging.info("Local db created successfully")
        except Exception as e:
            # FIX: log the full traceback AND re-raise. The previous code only
            # logged a single line and continued, which left vector_db unbound
            # and caused a confusing UnboundLocalError downstream.
            logging.exception(
                f"Error creating vector db: {e}. Make sure the Ollama server is "
                f"running (e.g. 'ollama serve') and the embedding model is "
                f"available (e.g. 'ollama pull nomic-embed-text')."
            )
            raise

        if vector_db:
            logging.info("Vector db verification successful")
        else:
            logging.error("Vector db creation failed")
            # FIX: actually raise instead of only logging, so the pipeline
            # stops with a clear message instead of continuing with None.
            raise ValueError("Vector db creation failed")

        # LLM from Ollama
        local_model = "llama3"
        llm = ChatOllama(model=local_model)
        logging.info("LLM model loaded")

        QUERY_PROMPT = PromptTemplate(
            input_variables=["question"],
            template="""You are an AI language model assistant. Your task is to generate five
        different versions of the given user question to retrieve relevant documents from
        a vector database. By generating multiple perspectives on the user question, your
        goal is to help the user overcome some of the limitations of the distance-based
        similarity search. Provide these alternative questions separated by newlines.
        Original question: {question}""",
        )
        logging.info("Query prompt created")

        retriever = MultiQueryRetriever.from_llm(
            vector_db.as_retriever(),
            llm=llm,
            prompt=QUERY_PROMPT
        )
        logging.info("Retriever created")

        # RAG prompt
        template = """Answer the question based ONLY on the following context:
{context}
    Question: {question}
    """
        prompt = ChatPromptTemplate.from_template(template)
        logging.info("RAG prompt created")

        chain = (
            {"context": retriever, "question": RunnablePassthrough()}
            | prompt
            | llm
            | StrOutputParser()
        )
        logging.info("Chain created")

        # FIX: explicit handling around the LLM/chain invocation so that any
        # failure (server down, model missing, bad LLM output) surfaces with a
        # full traceback and an actionable message instead of failing silently.
        try:
            response = chain.invoke(question)
        except Exception as e:
            logging.exception(
                f"Error invoking RAG chain: {e}. Make sure the chat model is "
                f"available (e.g. 'ollama pull {local_model}')."
            )
            raise

        logging.info("Chain invoked")
        logging.info(f"Response: {response}")

        return response

    except Exception as e:
        # FIX: use logging.exception so the full traceback is printed; the
        # failure is now clearly visible instead of silently returning None.
        logging.exception(f"An error occurred: {e}")
        return None


if __name__ == "__main__":
    question = "What are the 5 pillars of global cooperation?"
    response = run_agent(question)
    print(f"\nFinal Output: {response}")

def _installed_ollama_models(client):
    """Return the set of model names (without tags) available in the local Ollama instance."""
    response = client.list()
    if isinstance(response, dict):
        models = response.get("models", [])
    else:
        models = getattr(response, "models", [])
    names = set()
    for m in models:
        if isinstance(m, dict):
            name = m.get("model") or m.get("name") or ""
        else:
            name = getattr(m, "model", None) or getattr(m, "name", "") or ""
        if name:
            names.add(name.split(":")[0])
    return names


def check_ollama_ready(required_models):
    """
    Precondition check for the whole pipeline: the Ollama server must be
    reachable and every required model must be available locally (missing
    models are pulled automatically). Returns True when the environment is
    ready, False otherwise.
    """
    try:
        import ollama
    except ImportError:
        logging.error("The 'ollama' python package is not installed. Run: pip install ollama")
        return False

    try:
        client = ollama.Client()
        installed = _installed_ollama_models(client)
    except Exception as e:
        logging.error(
            "Cannot reach the Ollama server. Start it with 'ollama serve' "
            f"(default: http://localhost:11434). Details: {e}"
        )
        return False

    for model in required_models:
        if model not in installed:
            logging.info(f"Model '{model}' is not present locally. Pulling it (this can take a while)...")
            try:
                client.pull(model)
            except Exception as e:
                logging.error(f"Failed to pull model '{model}'. Run 'ollama pull {model}' manually. Details: {e}")
                return False
    return True
