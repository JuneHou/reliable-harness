from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    # Fixed prompt: add explicit instructions telling the model HOW to answer,
    # so ChatOpenAI responds with full, chatbot-like conversational sentences
    # instead of terse fragments. The original prompt only said "Answer the
    # question based on the following context", which lets gpt-3.5-turbo reply
    # with short extracted snippets rather than natural sentences.
    ANSWER_TEMPLATE = """You are a helpful, friendly chatbot assistant.
Answer the question based on the following context:
    {context}

Question: {question}

Instructions for your answer:
- Respond with full, complete, conversational sentences, like a chatbot talking to a user.
- Do not answer with only a single word, a short phrase, or a raw snippet from the context.
- Directly address the user and give a natural, flowing answer to their question.
- If the context does not contain the answer, say politely in a full sentence that you could not find the answer.
"""

    prompt = ChatPromptTemplate.from_template(ANSWER_TEMPLATE)

    chain = (
        {
            "context": retriever,
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)
    
    return result

if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
