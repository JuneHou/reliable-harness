from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

def format_docs(docs):
    return "\n\n".join(doc.page_content for doc in docs)

def run_agent(query):

    # Use a more capable model so it reliably follows the conversational
    # style instructions instead of returning terse/extractive answers.
    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-4o-mini",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    # Add an explicit SystemMessage with the assistant persona and an explicit
    # instruction to answer in full, conversational sentences, plus the exact
    # {context} and {question} placeholders the chain requires.
    ANSWER_TEMPLATE = """You are a helpful, friendly chatbot assistant. \
Always answer the question in full, complete, conversational sentences, \
the way a chatbot would naturally speak to a user. Do not reply with only \
keywords, short fragments, or raw text copied from the context. \
Use only the facts found in the following context to build your answer.

Context:
{context}

Question: {question}

Answer:"""

    prompt = ChatPromptTemplate.from_messages([
        ("system", ANSWER_TEMPLATE),
        ("human", "{question}"),
    ])

    chain = (
        {
            "context": (retriever | format_docs),
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)

    return result

if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
