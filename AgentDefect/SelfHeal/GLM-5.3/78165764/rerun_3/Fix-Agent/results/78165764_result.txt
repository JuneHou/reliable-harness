from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    # FIXED PROMPT:
    # The original prompt only said "Answer the question based on the following
    # context" with no instructions about *how* to answer. Because gpt-3.5-turbo
    # is highly prompt-steerable, it replies minimally/extractively (often just a
    # word or phrase), whereas Llama2 chat-style runs produce full conversational
    # sentences. The fix is to explicitly instruct the model to respond like a
    # chatbot: using full, complete, conversational sentences that address the
    # user directly, instead of replying with a bare fact.
    ANSWER_TEMPLATE = """You are a friendly, helpful chatbot assistant.
Answer the question based on the following context:

{context}

Question: {question}

Instructions:
- Always respond in full, complete sentences (like a chatbot talking to a user).
- Be conversational and natural; address the user directly.
- Do NOT reply with just a single word, a bare fact, a number, or a short phrase.
- If the context contains the answer, state it in a clear, complete sentence.
- If the context does not contain the answer, say so politely in a full sentence.
"""

    prompt = ChatPromptTemplate.from_template(ANSWER_TEMPLATE)

    chain = (
        {
            "context": retriever,
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)

    return result

if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
