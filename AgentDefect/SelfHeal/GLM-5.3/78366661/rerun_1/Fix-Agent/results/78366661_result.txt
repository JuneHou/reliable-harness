"""
MongoDB Atlas Vector Search + LangChain RAG pipeline.

Why the original code always printed `Result: []` even though documents were
"stored":

  * A local mock `MongoDBAtlasVectorSearch` class was defined inside the file,
    whose `similarity_search()` unconditionally returned an empty list, and
  * `atlas_collection` was a plain, empty Python list -- not a real MongoDB
    Atlas collection -- so no embeddings were ever persisted and no real
    `$vectorSearch` query was ever executed.

The fix:
  1. Use the REAL `MongoDBAtlasVectorSearch` from the `langchain_mongodb`
     package (`pip install langchain-mongodb pymongo`).
  2. Pass a genuine Atlas collection obtained from `pymongo.MongoClient`
     using the `MONGODB_ATLAS_URI` connection string.
  3. Before searching, verify the Atlas Search vector index exists and has
     finished building. A missing / still-building / differently-named index
     makes `similarity_search()` return `[]` even when the documents are
     stored correctly, so we fail loudly instead of silently.
"""

import os
import time

from dotenv import load_dotenv

from pymongo import MongoClient

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_mongodb import MongoDBAtlasVectorSearch

# Dimensions produced by OpenAIEmbeddings' default model
# (text-embedding-ada-002 / text-embedding-3-small). This MUST match the
# `numDimensions` declared in the Atlas Vector Search index definition.
EMBEDDING_DIMENSIONS = 1536


def _ensure_search_index_ready(collection, index_name, timeout=120):
    """Make sure the Atlas vector-search index exists and is READY.

    If the index is missing, still building, or named differently from
    `index_name`, `similarity_search()` returns an empty list even though the
    documents were stored successfully -- the exact symptom being debugged.
    """
    deadline = time.time() + timeout

    while True:
        try:
            indexes = list(collection.list_search_indexes())
        except Exception as exc:
            # pymongo too old, or cluster has no Search nodes: we cannot
            # introspect the index, so warn and let the query speak for itself.
            print(
                f"WARNING: could not inspect Atlas Search indexes ({exc}); "
                "skipping index readiness check."
            )
            return

        index = next(
            (i for i in indexes if i.get("name") == index_name), None
        )

        if index is None:
            raise RuntimeError(
                f"Atlas Search index '{index_name}' was not found on "
                f"'{collection.database.name}.{collection.name}'. Create it "
                "in the Atlas UI (Database > Search Indexes > Create Index) "
                "as a JSON 'vectorSearch' index:\n"
                "{\n"
                '  "fields": [{\n'
                '    "type": "vector",\n'
                '    "path": "embedding",\n'
                f'    "numDimensions": {EMBEDDING_DIMENSIONS},\n'
                '    "similarity": "cosine"\n'
                "  }]\n"
                "}\n"
                "Note: newly created indexes take up to ~1 minute to build."
            )

        if index.get("status") in (None, "READY"):
            return

        if time.time() > deadline:
            raise RuntimeError(
                f"Atlas Search index '{index_name}' is still "
                f"'{index.get('status')}' after {timeout}s."
            )

        # Index is still BUILDING -- wait and retry.
        time.sleep(2)


def run_agent(pdf_url):
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key is None:
        raise ValueError("OPENAI_API_KEY not found")

    mongodb_uri = os.getenv("MONGODB_ATLAS_URI")
    if mongodb_uri is None:
        raise ValueError(
            "MONGODB_ATLAS_URI not found. Set it to your Atlas connection "
            "string, e.g. "
            "'mongodb+srv://<user>:<password>@<cluster>.mongodb.net/'"
        )

    # A REAL Atlas collection -- not a plain Python list. `from_documents`
    # embeds and persists every chunk into this collection, and
    # `similarity_search` runs a real $vectorSearch aggregation against it.
    db_name = os.getenv("MONGODB_DB", "langchain_db")
    collection_name = os.getenv("MONGODB_COLLECTION", "test")
    vector_search_index = "vector_index"

    client = MongoClient(mongodb_uri)
    atlas_collection = client[db_name][collection_name]

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100,
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()

    docs = text_splitter.split_documents(data)

    # The REAL vector store from langchain_mongodb: embeds every chunk with
    # OpenAIEmbeddings, upserts it into Atlas, and exposes a working
    # similarity_search backed by Atlas Vector Search.
    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(disallowed_special=()),
        collection=atlas_collection,
        index_name=vector_search_index,
    )

    # Fail loudly (instead of silently returning []) if the vector index is
    # missing or still building.
    _ensure_search_index_ready(atlas_collection, vector_search_index)

    query = "What were the compute requirements for training GPT 4"

    results = vector_search.similarity_search(query)

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"

    results = run_agent(url)

    print(f"Result ({len(results)} documents):")
    for doc in results:
        print(doc.page_content[:500])
        print("---")
