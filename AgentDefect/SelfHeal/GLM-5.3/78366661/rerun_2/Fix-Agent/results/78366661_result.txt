from dotenv import load_dotenv
import os

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings

# Import the REAL MongoDB Atlas vector store integration
# (requires: pip install langchain-mongodb pymongo)
from langchain_mongodb import MongoDBAtlasVectorSearch
from pymongo import MongoClient
from pymongo.errors import PyMongoError


def run_agent(pdf_url):
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key is None:
        raise ValueError("OPENAI_API_KEY not found")

    # The connection string to the MongoDB Atlas cluster must be provided
    mongodb_atlas_uri = os.getenv("MONGODB_ATLAS_URI")
    if mongodb_atlas_uri is None:
        raise ValueError(
            "MONGODB_ATLAS_URI not found. Set it to your Atlas connection "
            "string, e.g. "
            "'mongodb+srv://<user>:<password>@<cluster>.mongodb.net/?retryWrites=true&w=majority'"
        )

    db_name = os.getenv("MONGODB_DB_NAME", "langchain_db")
    collection_name = os.getenv("MONGODB_COLLECTION_NAME", "test")

    # Create a REAL MongoDB collection object backed by the Atlas cluster.
    # Previously this was a plain Python list ([]), so documents could never
    # be stored in (or searched from) MongoDB.
    client = MongoClient(mongodb_atlas_uri)
    atlas_collection = client[db_name][collection_name]

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()

    docs = text_splitter.split_documents(data)

    vector_search_index = "vector_index"

    # Sanity check: an empty result set is very often caused by the Atlas
    # Search index not existing (or not finished building) on the collection.
    try:
        index_names = [
            idx.get("name")
            for idx in atlas_collection.list_search_indexes()
        ]
        if vector_search_index not in index_names:
            print(
                f"WARNING: Atlas search index '{vector_search_index}' was not "
                f"found on {db_name}.{collection_name}. Create it in the Atlas "
                "UI (Search tab) with the 'vectorSearch' type, a field named "
                "'embedding', and dimensions matching your embedding model "
                "(1536 for text-embedding-ada-002 / text-embedding-3-small)."
            )
    except (PyMongoError, AttributeError, TypeError):
        # list_search_indexes requires a recent pymongo/Atlas tier; do not
        # block execution if it is unavailable.
        pass

    # Use the genuine MongoDBAtlasVectorSearch.from_documents. This computes
    # embeddings for each chunk and inserts them into the Atlas collection.
    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(disallowed_special=()),
        collection=atlas_collection,
        index_name=vector_search_index
    )

    query = "What were the compute requirements for training GPT 4"

    # Perform a real vector similarity search against Atlas.
    results = vector_search.similarity_search(query, k=4)

    if not results:
        print(
            "WARNING: similarity_search returned no documents. Verify that: "
            "(1) the Atlas vector search index finished building, "
            "(2) the index name matches, and "
            "(3) documents were inserted into the same database/collection."
        )

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"

    results = run_agent(url)

    print("Result:", results)
