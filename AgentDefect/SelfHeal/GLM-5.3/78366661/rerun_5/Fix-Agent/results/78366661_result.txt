from dotenv import load_dotenv
import os

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_mongodb import MongoDBAtlasVectorSearch
from pymongo import MongoClient


def run_agent(pdf_url):
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key is None:
        raise ValueError("OPENAI_API_KEY not found")

    # A real MongoDB Atlas connection string is required so that the
    # documents are actually stored in (and searched from) Atlas.
    atlas_connection_string = os.getenv("MONGODB_ATLAS_URI")
    if atlas_connection_string is None:
        raise ValueError("MONGODB_ATLAS_URI not found")

    db_name = os.getenv("MONGODB_DB_NAME", "langchain_db")
    collection_name = os.getenv("MONGODB_COLLECTION_NAME", "test")

    # Use a real MongoDB collection object, not an empty Python list.
    client = MongoClient(atlas_connection_string)
    atlas_collection = client[db_name][collection_name]

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()

    docs = text_splitter.split_documents(data)

    if not docs:
        raise ValueError("No documents were extracted from the PDF")

    vector_search_index = "vector_index"

    # Use the real MongoDBAtlasVectorSearch from langchain_mongodb,
    # which actually persists the embedded documents and performs a
    # real $vectorSearch aggregation against the Atlas Search index.
    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(disallowed_special=()),
        collection=atlas_collection,
        index_name=vector_search_index,
    )

    query = "What were the compute requirements for training GPT 4"

    # Retrieve the top-k most similar chunks from Atlas.
    results = vector_search.similarity_search(query, k=4)

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"

    results = run_agent(url)

    print("Result:", results)
