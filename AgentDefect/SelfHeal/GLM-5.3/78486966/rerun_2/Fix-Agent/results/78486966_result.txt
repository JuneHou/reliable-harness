from langchain_text_splitters import CharacterTextSplitter

import json
from pathlib import Path


def run_agent(text_raw):
    # BUG (why output didn't match expectations):
    #   CharacterTextSplitter's DEFAULT separator is "\n\n" (a blank line).
    #   It first splits the text on that separator and only then merges the
    #   pieces up to chunk_size. The sample text has no blank lines, so the
    #   entire text was returned as ONE giant chunk (~340 chars) together with
    #   the warning "Created a chunk of size 340, which is longer than the
    #   specified 30".
    #
    # FIX:
    #   Set separator="" so the splitter works character by character. The
    #   merger then packs the characters into fixed-size chunks of exactly
    #   chunk_size=30 characters, with chunk_overlap=5 characters shared
    #   between consecutive chunks.
    text_splitter = CharacterTextSplitter(
        separator="",
        chunk_size=30,
        chunk_overlap=5,
    )
    chunks = text_splitter.split_text(text_raw)
    return chunks


def save_chunks_locally(chunks, directory="."):
    """Save the resulting chunks locally so they can be reused later.

    - chunks.json : the exact list of chunks (reload with json.load)
    - chunks.txt  : human-readable version, one chunk per section
    """
    directory = Path(directory)
    directory.mkdir(parents=True, exist_ok=True)

    json_path = directory / "chunks.json"
    txt_path = directory / "chunks.txt"

    # Structured save (exact chunk boundaries are preserved).
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(chunks, f, ensure_ascii=False, indent=2)

    # Human-readable save.
    with open(txt_path, "w", encoding="utf-8") as f:
        for i, chunk in enumerate(chunks, start=1):
            f.write(f"--- Chunk {i} ({len(chunk)} chars) ---\n{chunk}\n\n")

    return json_path, txt_path
