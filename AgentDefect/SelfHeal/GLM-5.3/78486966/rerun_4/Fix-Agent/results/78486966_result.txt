import json
import os

from langchain_text_splitters import CharacterTextSplitter


def run_agent(text_raw):
    # CharacterTextSplitter first splits the text on `separator` and only
    # then merges those pieces up to `chunk_size`. With the default
    # separator ("\n\n") the text is never broken apart (it only contains
    # single newlines), so one oversized chunk covering the whole text is
    # returned instead of fixed-size character chunks.
    #
    # To get true fixed-size character chunks:
    #   - separator=""             -> split on every single character
    #   - length_function=len      -> chunk_size is counted in characters
    #   - is_separator_regex=False -> treat the separator as a literal string
    text_splitter = CharacterTextSplitter(
        separator="",
        chunk_size=30,            # each chunk holds at most 30 characters
        chunk_overlap=5,          # consecutive chunks share 5 characters
        length_function=len,
        is_separator_regex=False,
    )
    # Remove the leading/trailing newlines of the raw text so chunk
    # boundaries line up with real characters (no padding whitespace).
    chunks = text_splitter.split_text(text_raw.strip())
    return chunks


def save_chunks_locally(chunks, directory="chunks"):
    """Save the resulting chunks locally.

    - one .txt file per chunk  (chunks/chunk_000.txt, chunks/chunk_001.txt, ...)
    - a single chunks.json file holding the full list of chunks
    """
    os.makedirs(directory, exist_ok=True)

    # One file per chunk
    for i, chunk in enumerate(chunks):
        file_path = os.path.join(directory, f"chunk_{i:03d}.txt")
        with open(file_path, "w", encoding="utf-8") as f:
            f.write(chunk)

    # All chunks together in a single JSON file
    json_path = os.path.join(directory, "chunks.json")
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(chunks, f, ensure_ascii=False, indent=2)

    return directory


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)

    print(chunks)
    for i, chunk in enumerate(chunks):
        print(f"chunk {i} ({len(chunk)} chars): {chunk!r}")
    print(f"\n\n {len(chunks)}")

    out_dir = save_chunks_locally(chunks)
    print(f"\nChunks saved locally to '{out_dir}' "
          f"(individual .txt files plus chunks.json).")
