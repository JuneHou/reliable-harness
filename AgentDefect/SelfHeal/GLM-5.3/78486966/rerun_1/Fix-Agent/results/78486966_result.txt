import json
import os

from langchain_text_splitters import CharacterTextSplitter


def run_agent(text_raw):
    """Split text into fixed-size character chunks with overlapping segments.

    Root cause of the mismatch: CharacterTextSplitter defaults to
    separator="\\n\\n", which first splits the text on blank lines and only
    then merges the pieces up to chunk_size. The result is paragraph-sized
    chunks, NOT fixed 30-character windows.

    Fix: set separator="" so the text is cut at the character level, and the
    splitter then merges the characters into chunks of exactly `chunk_size`
    characters, with `chunk_overlap` characters shared between consecutive
    chunks.
    """
    try:
        text_splitter = CharacterTextSplitter(
            separator="",
            chunk_size=30,
            chunk_overlap=5,
            is_separator_regex=False,
        )
    except TypeError:
        # Older versions of langchain_text_splitters do not accept
        # is_separator_regex; fall back to the plain parameter set.
        text_splitter = CharacterTextSplitter(
            separator="",
            chunk_size=30,
            chunk_overlap=5,
        )

    chunks = text_splitter.split_text(text_raw)
    return chunks


def save_chunks_locally(chunks, txt_path="chunks.txt", json_path="chunks.json"):
    """Persist the resulting chunks to local files.

    - chunks.txt : one human-readable block per chunk (with index and length)
    - chunks.json: the full list of chunks in machine-readable form
    """
    with open(txt_path, "w", encoding="utf-8") as f:
        for i, chunk in enumerate(chunks, start=1):
            f.write(f"[Chunk {i}] ({len(chunk)} chars)\n{chunk}\n\n")

    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(
            {"num_chunks": len(chunks), "chunks": chunks},
            f,
            ensure_ascii=False,
            indent=2,
        )

    return os.path.abspath(txt_path), os.path.abspath(json_path)


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)

    for i, chunk in enumerate(chunks, start=1):
        print(f"[Chunk {i}] ({len(chunk)} chars): {chunk!r}")

    print(f"\n\nTotal chunks: {len(chunks)}")

    txt_path, json_path = save_chunks_locally(chunks)
    print(f"Chunks saved locally to:\n  - {txt_path}\n  - {json_path}")
