import json

from langchain_text_splitters import CharacterTextSplitter

# Local files where the resulting chunks will be stored
CHUNKS_JSON_PATH = "chunks.json"
CHUNKS_TXT_PATH = "chunks.txt"


def run_agent(text_raw):
    """Split the raw text into fixed-size character chunks with overlap.

    Root cause of the bug:
        CharacterTextSplitter's DEFAULT separator is "\\n\\n". The splitter
        first cuts the text on that separator and only then packs the pieces
        into chunks of `chunk_size`. The sample text contains no double
        newline, so the whole text stayed a single (over-sized) piece and
        LangChain merely printed the warning:
            "Created a chunk of size ..., which is longer than the
             specified 30"
        and returned ONE chunk instead of many 30-character chunks.

    The fix:
        Pass separator="" so the splitter works on the raw character
        stream. It then produces chunks of exactly `chunk_size` characters
        that overlap by `chunk_overlap` characters.
    """
    text_splitter = CharacterTextSplitter(
        separator="",          # <-- key fix: split on every character,
        chunk_size=30,         #     giving fixed-size 30-char chunks
        chunk_overlap=5,       # 5 characters shared between neighbours
        is_separator_regex=False,
        keep_separator=True,
    )
    chunks = text_splitter.split_text(text_raw)
    return chunks


def save_chunks_locally(chunks, json_path=CHUNKS_JSON_PATH, txt_path=CHUNKS_TXT_PATH):
    """Save the resulting chunks locally so they can be reused later."""
    # 1) JSON file: one structured list element per chunk
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(chunks, f, ensure_ascii=False, indent=2)

    # 2) Plain-text file: one chunk per line
    with open(txt_path, "w", encoding="utf-8") as f:
        f.write("\n".join(chunks))

    print(f"\nSaved {len(chunks)} chunks to '{json_path}' and '{txt_path}'")


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    # Trim the surrounding blank lines so the first/last chunks are not
    # made up of (stripped-away) newlines.
    chunks = run_agent(text_raw.strip())

    print(chunks)
    print(f'\n\n {len(chunks)}')

    # Show each chunk with its length to verify the fixed size / overlap
    for i, chunk in enumerate(chunks, start=1):
        print(f"chunk {i:02d} (len={len(chunk):02d}): {chunk!r}")

    # Persist the chunks locally (JSON + plain text)
    save_chunks_locally(chunks)
