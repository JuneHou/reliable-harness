import json
import os

from langchain_text_splitters import CharacterTextSplitter

CHUNK_SIZE = 30     # every chunk should hold exactly 30 characters
CHUNK_OVERLAP = 5   # consecutive chunks should share 5 characters


def run_agent(text_raw):
    """
    Split `text_raw` into fixed-size character chunks (30 chars each) with a
    5-character overlap, and save the resulting chunks locally.

    Why the original code produced the wrong output
    ------------------------------------------------
    CharacterTextSplitter FIRST splits the text on its `separator` and only
    afterwards packs those pieces into chunks of `chunk_size`.  The default
    separator is "\\n\\n" (a blank line => paragraph level).  Every paragraph
    of the input text is longer than 30 characters, so the splitter could
    never cut inside a paragraph: it simply returned one (over-sized) chunk
    per paragraph and logged warnings such as
        "Created a chunk of size 118, which is longer than the specified 30".
    Result: 4 paragraph-sized chunks instead of many 30-character chunks,
    and the requested 5-character overlap never appeared (the overlap logic
    removes whole paragraph splits, leaving nothing behind).

    The fix
    -------
    * separator=""          -> split at every single character, so that
                               chunk_size / chunk_overlap now operate on
                               characters, producing fixed-size chunks
                               (chunk N+1 starts 5 chars before chunk N ends).
    * strip_whitespace=False -> do not strip leading/trailing whitespace of a
                               chunk, otherwise chunks that start or end with
                               a space/newline would be shorter than 30.
    """
    # (optional cleanup) remove the leading/trailing newline that comes from
    # the triple-quoted string, otherwise it becomes the first character of
    # the first chunk and the last character of the last chunk.
    text_raw = text_raw.strip()

    text_splitter = CharacterTextSplitter(
        separator="",                 # character-level splitting (the key fix)
        chunk_size=CHUNK_SIZE,
        chunk_overlap=CHUNK_OVERLAP,
        strip_whitespace=False,       # keep chunks exactly 30 chars long
        is_separator_regex=False,
    )
    chunks = text_splitter.split_text(text_raw)

    # ------------------------------------------------------------------
    # Save the resulting chunks locally
    # ------------------------------------------------------------------
    out_dir = os.path.dirname(os.path.abspath(__file__))

    # 1) JSON file - keeps the list structure, easy to load back later.
    json_path = os.path.join(out_dir, "chunks.json")
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(chunks, f, indent=2, ensure_ascii=False)

    # 2) Plain-text file - one chunk per line (human readable).
    txt_path = os.path.join(out_dir, "chunks.txt")
    with open(txt_path, "w", encoding="utf-8") as f:
        for i, chunk in enumerate(chunks, start=1):
            f.write(f"chunk {i:02d} (len={len(chunk)}): {chunk}\n")

    return chunks


def load_chunks(json_path="chunks.json"):
    """Load previously saved chunks back from the local JSON file."""
    with open(json_path, encoding="utf-8") as f:
        return json.load(f)


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)

    print(chunks)
    print(f'\n\n {len(chunks)}')

    # Sanity check: fixed-size 30-char chunks with a 5-char overlap
    # (all chunks except the final one have exactly 30 characters).
    for i, chunk in enumerate(chunks, start=1):
        print(f"chunk {i:02d} | len={len(chunk):2d} | {chunk!r}")
    for prev, curr in zip(chunks, chunks[1:]):
        overlap = prev[len(prev) - CHUNK_OVERLAP:]
        print(f"overlap: {overlap!r} shared with next chunk -> {curr.startswith(overlap)}")
