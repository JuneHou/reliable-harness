from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


# Custom, blank prompt template that replaces LangChain's default STUFF_PROMPT.
# The default prompt ships with pre-filled example content (e.g. a sample question
# about contract law and sample "English law" context), which is why the chain
# always displayed that template instead of an empty/custom one.
#
# IMPORTANT: the prompt must
#   1) declare exactly the input variables the "stuff" chain injects:
#      "summaries" (the retrieved document content) and "question", and
#   2) instruct the model to end its response with a "SOURCES:" section, because
#      RetrievalQAWithSourcesChain parses its output by splitting on "SOURCES:"
#      to separate the answer from the source list.
CUSTOM_QA_WITH_SOURCES_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=(
        "You are a helpful assistant that answers questions using only the "
        "provided context.\n\n"
        "Context:\n"
        "---------\n"
        "{summaries}\n"
        "---------\n\n"
        "Question: {question}\n\n"
        "Answer the question using only the context above. If the answer is not "
        "in the context, say you don't know. Do not make up an answer.\n"
        "Always end your response with a line in exactly this format:\n"
        "SOURCES: <comma-separated source names, or NONE>\n\n"
        "Answer:"
    ),
)


def run_agent(question: str = "What is the test document about?"):
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        # Override the default (example-filled) prompt here:
        chain_type_kwargs={"prompt": CUSTOM_QA_WITH_SOURCES_PROMPT},
    )

    # Invoke the chain with a question so we get the actual result
    # (an answer plus parsed sources) instead of the chain's repr,
    # which dumps the default template with pre-filled example content.
    return chain.invoke({"question": question})


if __name__ == "__main__":
    result = run_agent("What is the test document about?")
    print("ANSWER:", result.get("answer"))
    print("SOURCES:", result.get("sources"))
