from langchain_openai import ChatOpenAI

try:
    # Newer installs (langchain_openai package owns the embeddings class)
    from langchain_openai import OpenAIEmbeddings
except ImportError:  # pragma: no cover
    # Fallback for older langchain installs
    from langchain.embeddings.openai import OpenAIEmbeddings

from langchain.chains import RetrievalQAWithSourcesChain
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


# ---------------------------------------------------------------------------
# Custom prompt for the "stuff" chain.
#
# Why the user was seeing a "default template with pre-filled example
# content": `RetrievalQAWithSourcesChain.from_chain_type()` builds its own
# internal chain (LLMChain) using the library's built-in default prompt
# (STUFF_PROMPT), which contains the example/hint text baked in.  The only
# way to replace it is to pass your own PromptTemplate through the
# `chain_type_kwargs` argument of `from_chain_type()`.
#
# IMPORTANT: for chain_type="stuff", the prompt MUST declare the exact input
# variables the stuff documents chain feeds it: "summaries" and "question".
# ---------------------------------------------------------------------------
CUSTOM_QA_WITH_SOURCES_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=(
        "You are an assistant for question-answering over retrieved "
        "documents.\n"
        "Use the following pieces of retrieved context to answer the "
        "question.\n"
        "If you don't know the answer, just say that you don't know. "
        "Do not make up an answer.\n"
        "Always end your answer with a line listing the sources you used, "
        "in the form 'SOURCES: <source1>, <source2>'.\n\n"
        "Content:\n"
        "{summaries}\n\n"
        "Question: {question}\n"
        "Helpful Answer:"
    ),
)


def run_agent(question: str = "What are the test documents about?"):
    """
    Build the retrieval QA-with-sources chain AND actually run it on a
    question, returning the chain's result dict.

    The original bug: this function only *returned the chain object* and the
    caller did `print(run_agent())`, which prints the chain's repr — i.e. the
    serialized chain internals, including the default prompt template with
    its pre-filled example content — instead of an actual answer.  A chain
    must be *invoked* with a question to produce output.
    """
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        # Override the default (example-filled) prompt with our own:
        chain_type_kwargs={"prompt": CUSTOM_QA_WITH_SOURCES_PROMPT},
    )

    # Actually run the chain with a question (invoke the chain, don't print it)
    try:
        # Preferred modern API
        result = chain.invoke({"question": question})
    except AttributeError:
        # Fallback for older LangChain versions without .invoke()
        result = chain({"question": question})

    return result


if __name__ == "__main__":
    try:
        result = run_agent("What are the test documents about?")
        # RetrievalQAWithSourcesChain returns {"answer": ..., "sources": ...}
        print("ANSWER :", result.get("answer", result))
        print("SOURCES:", result.get("sources", "N/A"))
    except Exception as exc:
        print(f"Failed to run the chain: {exc}")
