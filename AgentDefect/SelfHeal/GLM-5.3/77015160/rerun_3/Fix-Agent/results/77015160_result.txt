"""
Corrected version of the RetrievalQAWithSourcesChain example.

Why the original code showed a default template with pre-filled example content:
1. `RetrievalQAWithSourcesChain.from_chain_type(..., chain_type="stuff")` silently
   builds the combine step with the built-in default prompt
   (`langchain/chains/qa_with_sources/stuff_prompt.py`), which is hard-coded with
   few-shot example content (contract-law / "capital of France" examples and fake
   "SOURCES: 28 29 30 ..." citations). Unless you override it via
   `chain_type_kwargs={"prompt": ...}`, that default template is what gets used.
2. `run_agent()` returned the chain object itself and `print(run_agent())` printed
   the chain's repr (i.e., the default template) instead of actually running the
   chain with a question.

The fix:
- Supply a clean, custom prompt (no pre-filled examples) via `chain_type_kwargs`.
  Its input variables must match exactly what the stuff combine-chain expects:
  `summaries` (the retrieved documents) and `question`.
- Actually invoke the chain with a question and print the answer/sources.
"""

from typing import Optional

from dotenv import load_dotenv

from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain.prompts import PromptTemplate
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.vectorstores import FAISS

import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def build_custom_prompt() -> PromptTemplate:
    """
    Build a clean prompt that overrides the default stuff prompt.

    IMPORTANT: the input variables must be exactly ["summaries", "question"].
    - "summaries" is where the retrieved document contents are inserted
      (this is the combine-document chain's `document_variable_name`).
    - "question" is the user question passed when invoking the chain.
    """
    template = (
        "You are a helpful assistant that answers questions using only the "
        "provided context documents.\n"
        "If the context does not contain the answer, say that you don't know. "
        "Do not make up an answer.\n"
        "Always end your answer with a 'SOURCES:' line listing the source "
        "names you used.\n\n"
        "CONTEXT:\n"
        "{summaries}\n\n"
        "QUESTION: {question}\n\n"
        "ANSWER (end with a 'SOURCES:' line):"
    )
    return PromptTemplate(
        input_variables=["summaries", "question"],
        template=template,
    )


def run_agent(
    question: str = "What is this collection of documents about?",
    prompt: Optional[PromptTemplate] = None,
):
    """
    Build the retrieval chain with a custom (blank/example-free) prompt and
    run it against a real question.

    Args:
        question: The question to ask the chain.
        prompt: Optional custom PromptTemplate. Must expose the input
            variables "summaries" and "question". If None, the clean
            default prompt above is used (NOT LangChain's example-filled one).
    """
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    # Use the maintained langchain_openai embeddings class
    # (instead of the deprecated langchain.embeddings.openai module).
    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        # Override the default stuff prompt (the one pre-filled with
        # contract-law / capital-of-France example content):
        chain_type_kwargs={"prompt": prompt or build_custom_prompt()},
    )

    # Actually run the chain with a question instead of printing the chain
    # object itself (which merely renders the default template).
    try:
        result = chain.invoke({"question": question})
    except AttributeError:
        # Fallback for older langchain versions without `invoke`.
        result = chain({"question": question})

    return {
        "question": question,
        "answer": str(result.get("answer", "")).strip(),
        "sources": str(result.get("sources", "")).strip(),
    }


if __name__ == "__main__":
    output = run_agent()
    print("Question:", output["question"])
    print("Answer:  ", output["answer"])
    print("Sources: ", output["sources"])
