from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# ---------------------------------------------------------------------------
# Custom prompt that replaces LangChain's default "stuff" prompt.
#
# The default prompt (langchain.chains.prompt_template.SUFFICIENT_CONTEXT /
# QA with sources default) contains pre-filled example content, which is what
# the user was seeing. To override it we must pass our own PromptTemplate via
# chain_type_kwargs={"prompt": ...}.
#
# Requirements for the custom template to work with chain_type="stuff":
#   - input_variables must be exactly ["context", "question"] ("context" is
#     the document_variable_name the StuffDocumentsChain stuffs docs into),
#   - the model must be instructed to end its answer with a "SOURCES:" line,
#     because RetrievalQAWithSourcesChain parses sources from that line.
# ---------------------------------------------------------------------------
CUSTOM_QA_WITH_SOURCES_PROMPT = PromptTemplate(
    input_variables=["context", "question"],
    template=(
        "You are a helpful assistant that answers questions using ONLY the "
        "context provided below.\n\n"
        "Context:\n{context}\n\n"
        "Question: {question}\n\n"
        "Instructions:\n"
        "- Answer the question based only on the context above.\n"
        "- If the context does not contain the answer, say that you don't know.\n"
        "- End your answer with a line that starts with 'SOURCES:' followed by "
        "a comma-separated list of the source names you used.\n"
        "Example of the expected final line format:\n"
        "SOURCES: doc1, doc2"
    ),
)


def run_agent(question: str = "Which countries are mentioned in the documents?"):
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        # Override the default (pre-filled example) template with our own:
        chain_type_kwargs={"prompt": CUSTOM_QA_WITH_SOURCES_PROMPT},
    )

    # The bug: previously the chain OBJECT was returned and printed, which
    # rendered LangChain's default template (with its pre-filled example
    # content) instead of an actual answer. The chain must be INVOKED with a
    # question to produce real output.
    result = chain.invoke({"question": question})
    return result


if __name__ == "__main__":
    result = run_agent("Which countries are mentioned in the documents?")

    # Print the actual chain output (answer + sources), not the chain object.
    answer = (result.get("answer") or result.get("output_text") or "").strip()
    sources = (result.get("sources") or "").strip()

    print("Question: Which countries are mentioned in the documents?")
    print("\nAnswer:")
    print(answer if answer else "(no answer returned)")
    print("\nSources:")
    print(sources if sources else "(none)")
