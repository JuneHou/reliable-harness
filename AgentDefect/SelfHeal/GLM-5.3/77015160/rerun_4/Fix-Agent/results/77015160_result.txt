from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.vectorstores import FAISS
from langchain.prompts import PromptTemplate
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


# ------------------------------------------------------------------
# Custom prompt that REPLACES the chain's built-in default template.
#
# Why the default template showed up before:
#   RetrievalQAWithSourcesChain(chain_type="stuff") falls back to
#   LangChain's hard-coded default prompt, which is pre-filled with
#   illustrative example content (a fake "question about sports" etc.).
#   Since no custom prompt was supplied AND the script printed the
#   chain object itself (never invoking it), the only "output" seen was
#   that default, example-filled template.
#
# How to override it:
#   Pass a custom PromptTemplate through `chain_type_kwargs`.
#   The stuff-with-sources chain requires exactly these input variables:
#     - "summaries"  -> the retrieved documents (document_variable_name)
#     - "question"   -> the user's question
#   The placeholder names must match, or the chain will fail to run.
# ------------------------------------------------------------------
CUSTOM_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=(
        "You are a helpful assistant that answers questions strictly "
        "from the provided context.\n\n"
        "Context:\n\"\"\"\n{summaries}\n\"\"\"\n\n"
        "Question: {question}\n\n"
        "Instructions:\n"
        "1. Answer the question using ONLY the context above.\n"
        "2. If the context does not contain the answer, reply that you "
        "do not know.\n"
        "3. On the final line, list the sources you used in the exact "
        "format:\n"
        "SOURCES: source1, source2\n"
        "If no sources were used, output only 'SOURCES:'."
    ),
)


def run_agent(query: str = "What countries are mentioned in the documents?"):
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    # Use the non-deprecated langchain_openai embeddings class (the old
    # langchain.embeddings.openai path was removed), otherwise the script
    # fails before producing any output at all.
    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        # Override the default (example-filled) prompt here:
        chain_type_kwargs={
            "prompt": CUSTOM_PROMPT,
            # Must match the {summaries} placeholder in CUSTOM_PROMPT:
            "document_variable_name": "summaries",
        },
        # Optional but useful for verifying which docs were retrieved:
        return_source_documents=True,
    )

    # IMPORTANT: actually invoke the chain with a question.
    # (Previously the chain object itself was printed, which only shows
    # the default template's repr instead of a real answer.)
    result = chain.invoke({"question": query})

    # Normalize the result into a plain, printable output.
    return {
        "question": query,
        "answer": result.get("answer", "").strip(),
        "sources": result.get("sources", "").strip(),
        "source_documents": [
            {
                "content": doc.page_content,
                "source": doc.metadata.get("source", "unknown"),
            }
            for doc in result.get("source_documents", [])
        ],
    }


if __name__ == "__main__":
    output = run_agent("What countries are mentioned in the documents?")
    print("Question:", output["question"])
    print("Answer:", output["answer"])
    print("Sources:", output["sources"])
    print("Retrieved documents:")
    for doc in output["source_documents"]:
        print(f"  - [{doc['source']}] {doc['content']}")

#
# Why this is needed:
# RetrievalQAWithSourcesChain (chain_type="stuff") is built via
# load_qa_with_sources_chain(), whose default "stuff" PROMPT ships with

#   - "summaries": where the retrieved (formatted) documents are injected
#   - "question": the user's question

QA_WITH_SOURCES_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=(
        "You are given the following documents, each with its source:\n\n"
        "{summaries}\n\n"

        "If the answer is not contained in the documents, say that you don't know. "

        "Answer:"
    ),
)

# (default is "Content: {page_content}\nSource: {source}").

    input_variables=["page_content", "source"],
