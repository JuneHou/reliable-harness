import os
import json
from dotenv import load_dotenv
from sqlalchemy import (
    create_engine,
    MetaData,
    Table,
    Column,
    JSON,
    Integer,
    select,
    func,
)
from llama_index.core import SQLDatabase, VectorStoreIndex, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

load_dotenv()

OPENAI_API_KEY = os.environ.get('OPENAI_API_KEY')
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

pg_uri = os.environ.get('DATABASE_URL', 'sqlite:///test.db')

# Field-level metadata: every key is a field inside the JSON "data" column of the
# "snapshots" table, and every value is a human-readable description of that field.
metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
}

TABLE_NAME = "snapshots"

# Sample data used to seed the table once (only when it is still empty) so the
# query engine can return values alongside the metadata descriptions.
SAMPLE_ROW = {"page_call_phone_clicks_logged_in_unique": 2}


def build_table_context(table_name: str):
    """
    Build the extra context string that is attached to a table's SQLTableSchema.

    NOTE (the original bug): LlamaIndex never reads SQL column comments, so
    ``Column(..., comment=...)`` does NOT expose metadata to the LLM. The only
    supported way is ``SQLTableSchema(table_name=..., context_str=...)``:

      * ``SQLTableNodeMapping._from_object`` embeds ``context_str`` into the
        TextNode that is indexed by the vector store (improving retrieval), and
      * ``SQLTableRetrieverQueryEngine.get_table_context`` appends it to the
        table schema handed to the LLM for text-to-SQL and response synthesis.

    Also note that ``metadata_dict`` is keyed by *field names inside the JSON
    "data" column*, not by table name, so the context must be composed into a
    description of that column rather than looked up by table name.
    """
    if table_name != TABLE_NAME:
        return None

    field_descriptions = "\n".join(
        f"- {field}: {description}" for field, description in metadata_dict.items()
    )
    return (
        f"The '{table_name}' table stores snapshot records. Its 'data' column is a "
        "JSON column: each key of the JSON is a metric field name and each value is "
        "that metric's value. Descriptions of the fields inside the 'data' JSON "
        f"column:\n{field_descriptions}"
    )


class SqlQueryEngine:

    def __init__(self):
        self.schema = []

    def query_engine(self):
        engine = create_engine(pg_uri)
        metadata_obj = MetaData()
        # Serialize the metadata dictionary to a JSON string. This is kept only as
        # a column comment so the descriptions persist in the database schema
        # itself; by itself it NEVER reaches the LLM (that was the original bug).
        metadata_json = json.dumps(metadata_dict)
        # create snapshot SQL table
        table_name = TABLE_NAME
        snapshots_table = Table(
            table_name,
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
        )

        metadata_obj.create_all(engine)

        # Seed one sample row, but only if the table is still empty (existing
        # data is never overwritten). This gives the query engine real values to
        # return in addition to the metadata-driven explanation.
        with engine.begin() as conn:
            row_count = conn.execute(
                select(func.count()).select_from(snapshots_table)
            ).scalar()
            if not row_count:
                conn.execute(snapshots_table.insert().values(data=SAMPLE_ROW))

        sql_database = SQLDatabase(engine, include_tables=[TABLE_NAME])
        nodes = SQLTableNodeMapping(sql_database)

        self.schema = []
        for tbl_name in metadata_obj.tables.keys():
            # FIX: attach the field metadata to each table schema via `context_str`.
            # This is what makes the metadata visible to the query engine: the
            # SQLTableNodeMapping embeds it in the indexed node text and
            # SQLTableRetrieverQueryEngine injects it into the LLM prompt together
            # with the table schema. Without it, context_str stays None and the
            # LLM never sees the column descriptions.
            self.schema.append(
                SQLTableSchema(
                    table_name=tbl_name,
                    context_str=build_table_context(tbl_name),
                )
            )

        # We dump the table schema information into a vector index. The vector index is stored
        # within the context builder for future use.
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)
        # Use temperature=0: text-to-SQL generation must be deterministic/reliable,
        # not "creative" (0.9 produced unstable, incorrect queries).
        llm = OpenAI(temperature=0, model="gpt-3.5-turbo")
        #depricated for new version
        # llm_predictor = LLMPredictor(llm=llm)
        # service_context = ServiceContext.from_defaults(llm_predictor=llm)
        Settings.llm = llm

        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )

        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response


if __name__ == "__main__":

    query = "What does the page_call_phone_clicks_logged_in_unique field mean in the data column?"
    sql_engine, response = run_agent(query)

    for schema_item in sql_engine.schema:
        print(f"\nTable: {schema_item.table_name}")
        if hasattr(schema_item, 'context_str') and schema_item.context_str:
            print(f"  context_str: {schema_item.context_str}")
            print("  Status: Metadata context is SET")
        else:
            print("  context_str: None")
            print("  Status: Metadata context is NOT SET")
