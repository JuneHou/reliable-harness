import os
import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer
from llama_index.core import SQLDatabase, VectorStoreIndex, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

load_dotenv()

OPENAI_API_KEY = os.environ.get('OPENAI_API_KEY')
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

pg_uri = os.environ.get('DATABASE_URL', 'sqlite:///test.db')

# Field-level metadata: descriptions of the keys stored inside the JSON
# "data" column of the snapshots table.
metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
}


class SqlQueryEngine:

    def __init__(self):
        self.schema = []

    def query_engine(self):
        engine = create_engine(pg_uri)
        metadata_obj = MetaData()

        # create snapshot SQL table
        table_name = "snapshots"
        snapshots_table = Table(
            table_name,
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON),
        )

        metadata_obj.create_all(engine)

        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        # Build the table schema objects and attach the field metadata via
        # `context_str`.
        #
        # NOTE: putting the metadata into a SQL column *comment*
        # (e.g. Column("data", JSON, comment=json.dumps(metadata_dict)))
        # does NOT make it visible to LlamaIndex -- column comments are never
        # read by the query engine (and are silently ignored by SQLite).
        # `context_str` is the supported way to attach extra context to a
        # table schema: it gets embedded into the ObjectIndex and is injected
        # into the text-to-SQL prompt by SQLTableRetrieverQueryEngine.
        self.schema = []
        for tbl_name in metadata_obj.tables.keys():
            context_str = None
            if metadata_dict:
                context_str = (
                    "The 'data' column of this table stores a JSON object. "
                    "Descriptions of its fields:\n"
                    + json.dumps(metadata_dict, indent=2)
                )
            self.schema.append(
                SQLTableSchema(table_name=tbl_name, context_str=context_str)
            )

        # We dump the table schema information (now including the metadata
        # context) into a vector index. The vector index is stored
        # within the context builder for future use.
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)
        llm = OpenAI(temperature=0.9, model="gpt-3.5-turbo")
        # depricated for new version
        # llm_predictor = LLMPredictor(llm=llm)
        # service_context = ServiceContext.from_defaults(llm_predictor=llm)
        Settings.llm = llm

        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )

        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response


if __name__ == "__main__":

    query = "What does the page_call_phone_clicks_logged_in_unique field mean in the data column?"
    sql_engine, response = run_agent(query)

    for schema_item in sql_engine.schema:
        print(f"\nTable: {schema_item.table_name}")
        if hasattr(schema_item, 'context_str') and schema_item.context_str:
            print(f"  context_str: {schema_item.context_str}")
            print("  Status: Metadata context is SET")
        else:
            print("  context_str: None")
            print("  Status: Metadata context is NOT SET")

## Analysis

**Root cause of the bug:** The buggy code attempted to attach field metadata via a SQL column `comment` (`Column("data", JSON, comment=metadata_json)`). This has two fatal problems:
1. SQLAlchemy column comments are silently ignored by SQLite (the default fallback DB).
2. More fundamentally, LlamaIndex never reads column comments — `SQLTableNodeMapping`/`SQLTableRetrieverQueryEngine` don't inspect DB comment metadata.

Meanwhile, the `SQLTableSchema` objects were constructed with only `table_name`, so `schema_item.context_str` was always `None` → the test's `assertIsNotNone(schema_item.context_str)` fails.

**What the fix does:**
- Passes `context_str=...` to `SQLTableSchema(...)` — this is the documented, supported mechanism for attaching extra table context in LlamaIndex (it gets embedded into the ObjectIndex nodes and injected into the text-to-SQL prompt).
- Removes the dead column `comment` parameter.
- Renames the loop variable `table_name` → `tbl_name` to avoid shadowing the outer variable (good hygiene).
- Preserves `run_agent`'s return signature `(sql_engine, response)` and all other behavior.
