import random

from langchain_ollama import ChatOllama


def build_llm() -> ChatOllama:
    """
    Build a ChatOllama instance configured for non-deterministic generation.

    Why the original code always returned the same answer:
    ------------------------------------------------------
    Ollama's token sampling is *fully reproducible when the RNG `seed` is
    pinned*. Per Ollama's docs, `seed` "sets the random number seed to use
    for generation. Setting this to a specific number will make the model
    generate the same text for the same prompt."

    `temperature` and `top_p` only reshape the token probability
    distribution -- they do NOT reseed the sampler. So with the seed left at
    its default/pinned value, an identical prompt + identical parameters
    decode to byte-for-byte identical text on every call, no matter how high
    the temperature is.

    The fix:
    --------
    1. Give every request a fresh, randomly generated `seed` so each call
       samples along a different path.
    2. Widen the sampling space (top_k) so the sampler has room to explore
       the reshaped distribution.

    NOTE: build_llm() must be called once per request so the seed is
    re-randomized; reusing a single instance with one fixed seed would
    reintroduce the deterministic behaviour.
    """
    return ChatOllama(
        model="qwen2.5:14b",
        # Sampling parameters -- kept creative but sane.
        temperature=0.7,
        top_p=0.95,
        top_k=40,
        num_ctx=2048,
        repeat_penalty=1.2,
        timeout=120,
        # *** The actual fix ***
        # A different seed per request => different random sampling path
        # => varied, non-deterministic answers for the same prompt.
        seed=random.randint(0, 2**31 - 1),
    )


def run_agent(input_text: str) -> str:
    """Invoke the Ollama-backed chat model and return its text content."""
    llm = build_llm()
    response = llm.invoke(input_text)
    return response.content


if __name__ == "__main__":
    prompt = "Please tell me a lighthearted joke"

    # Ask the SAME prompt several times to demonstrate that the responses
    # now vary run-to-run instead of being identical every time.
    for i in range(3):
        print(f"[{i + 1}] {run_agent(prompt)}")
        print("-" * 60)
