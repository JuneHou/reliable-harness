import random

from langchain_ollama import ChatOllama


def run_agent(input_text):
    # Root cause: Ollama's `seed` option defaults to a fixed value (0).
    # With a fixed seed, the sampler RNG is initialized identically on every
    # request, so the same prompt always yields the exact same token sequence,
    # which makes `temperature` / `top_p` appear to have no effect.
    #
    # Fix: generate a fresh random seed for each invocation so sampling is
    # genuinely stochastic and temperature/top_p produce varied answers.
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        seed=random.randint(0, 2**32 - 1),  # new random seed per call
        timeout=120,
    )
    response = llm.invoke(input_text)
    return response.content


if __name__ == '__main__':
