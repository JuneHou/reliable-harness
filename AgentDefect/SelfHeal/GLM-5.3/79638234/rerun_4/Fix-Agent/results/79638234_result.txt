import random

from langchain_ollama import ChatOllama


def run_agent(input_text):
    """
    Query an Ollama model through LangChain and return a *varied* answer.

    Root cause of the original bug:
        Ollama initializes its token-sampling PRNG from the `seed` option.
        When no seed is sent (or it is pinned to a fixed value), the RNG starts
        from the same state on every request, so the same prompt with the same
        temperature/top_p always produces the exact same tokens -- temperature
        only reshapes the probability distribution, it does not change *which*
        token is drawn.

    Fix:
        Pass a fresh, randomly generated `seed` on every request. A new seed
        reinitializes the sampler each time, making the outputs non-deterministic
        while keeping the desired temperature / top_p sampling behavior.
    """
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        # Fresh random seed per request -> different sampling RNG state each
        # call, so identical prompts now yield different answers.
        seed=random.randint(0, 2**32 - 1),
        timeout=120,
    )
    response = llm.invoke(input_text)
    return response.content


if __name__ == "__main__":
    # Run several times to demonstrate that responses now vary.
    for i in range(3):
        print(f"--- run {i + 1} ---")
        print(run_agent("Please tell me a lighthearted joke"))
        print()
