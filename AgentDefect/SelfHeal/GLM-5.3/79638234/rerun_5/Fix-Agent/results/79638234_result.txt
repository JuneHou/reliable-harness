import random

from langchain_ollama import ChatOllama


def run_agent(input_text):
    # Root cause of the identical responses:
    # Ollama's generation endpoint seeds its sampler deterministically when no
    # `seed` is supplied (the default seed of 0 is treated as a fixed seed by the
    # llama.cpp runner), so the RNG state is identical for every request.
    # `temperature`, `top_p`, and `repeat_penalty` only reshape the probability
    # distribution tokens are sampled from -- they do NOT change the RNG state.
    # With a fixed seed, the exact same tokens are therefore selected every time,
    # and the same prompt returns the exact same answer.
    #
    # Fix:
    # Send a fresh, randomly chosen `seed` with every request. Ollama then
    # reseeds its sampler for that call, and sampling over the
    # temperature-shaped distribution produces varied, non-deterministic output.
    # Note: the seed is generated per call (not once at import time), otherwise
    # it would still be a fixed seed and the output would remain deterministic.
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        seed=random.randint(0, 2**31 - 1),  # random seed per request -> varied answers
        timeout=120,
    )
    response = llm.invoke(input_text)
    return response.content


if __name__ == '__main__':
    prompt = "Please tell me a lighthearted joke"

    # Run the same prompt a few times to confirm the answers now vary.
    answers = [run_agent(prompt) for _ in range(3)]
    for i, answer in enumerate(answers, start=1):
        print(f"Answer {i}: {answer}")
        print("-" * 60)
