import random

from langchain_ollama import ChatOllama


def run_agent(input_text: str) -> str:
    """
    Invoke a locally served Ollama chat model with varied, non-deterministic output.

    Root cause of the original bug:
    Ollama's sampling RNG is seeded from the `seed` option, which defaults to a
    fixed value (0) when not provided. With a fixed seed, temperature/top_p
    sampling draws the exact same token sequence for an identical prompt every
    time, so the response is byte-for-byte identical on each call -- even with
    temperature=0.7. `temperature` controls how wide the sampling distribution
    is, but `seed` controls which deterministic draw is taken from it.

    Fix:
    Pass a fresh random `seed` on every request so each generation samples
    independently from the temperature/top_p distribution.
    """
    llm = ChatOllama(
        model="qwen2.5:14b",
        # Sampling options forwarded to the Ollama API `options` object:
        temperature=0.7,        # diversity of token selection
        top_p=0.95,             # nucleus sampling
        num_ctx=2048,           # context window size
        repeat_penalty=1.2,     # penalize repeated tokens
        timeout=120,
        # --- The key fix: a new random seed per request => non-deterministic output ---
        # Use a positive 32-bit int, which is what Ollama's seed option expects.
        seed=random.randint(0, 2**31 - 1),
    )

    response = llm.invoke(input_text)
    return response.content


if __name__ == "__main__":
    prompt = "Please tell me a lighthearted joke"

    # Demonstrate that repeated calls with the same prompt now produce
    # different, varied answers instead of one identical response.
    for i in range(3):
        print(f"--- Response {i + 1} ---")
        print(run_agent(prompt))
        print()
