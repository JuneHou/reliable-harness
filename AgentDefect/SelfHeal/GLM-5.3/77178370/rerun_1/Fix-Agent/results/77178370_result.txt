import os
from typing import Any, List

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import FAISS
from langchain.chains import ConversationalRetrievalChain, LLMChain
from langchain.memory import ConversationBufferMemory
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate
from langchain.schema import BaseRetriever, Document
from langchain.callbacks.manager import CallbackManagerForRetrieverRun

# Load environment variables from .env file
_ = load_dotenv(find_dotenv())


class CustomRetriever(BaseRetriever):
    """Retriever that returns the relevant documents only when the answer to
    the question can actually be found in the custom knowledge base.

    A similarity-score threshold proved unreliable, so an LLM is used to
    decide whether the retrieved context really answers the question.
    Unrelated inputs (e.g. greetings) therefore return an empty list.
    """

    # New attributes added to the retriever: the wrapped vector-store
    # retriever and the LLM chain that performs the relevance check.
    base_retriever: Any
    relevance_chain: Any

    def _get_relevant_documents(
        self, query: str, *, run_manager: CallbackManagerForRetrieverRun
    ) -> List[Document]:
        # 1. Fetch the usual top-k candidates from the vector store.
        candidate_docs = self.base_retriever.get_relevant_documents(query)
        if not candidate_docs:
            return []

        # 2. Let the LLM decide whether the knowledge base actually contains
        #    the answer to the question (instead of an unreliable similarity
        #    score threshold).
        context = "\n".join(doc.page_content for doc in candidate_docs)
        decision = self.relevance_chain.predict(
            question=query, context=context
        ).strip().lower()

        # 3. Return the documents only when the answer really comes from the
        #    custom knowledge base; otherwise return an empty list (e.g. for
        #    greetings such as "hi").
        if "yes" in decision:
            return candidate_docs
        return []


def run_agent(user_input):
    openai_api_key = os.getenv('OPENAI_API_KEY')

    os.environ['OPENAI_API_KEY'] = openai_api_key

    custom_template = """
This is conversation with a human. Answer the questions you get based on the knowledge you have.
If you don't know the answer, just say that you don't, don't try to make up an answer.
Chat History:
{chat_history}
Follow Up Input: {question}
"""
    CUSTOM_QUESTION_PROMPT = PromptTemplate.from_template(custom_template)
    llm = ChatOpenAI(
        model_name="gpt-3.5-turbo",  # Name of the language model
        temperature=0  # Parameter that controls the randomness of the generated responses
    )
    embeddings = OpenAIEmbeddings()
    docs = [
        "Buildings are made out of brick",
        "Buildings are made out of wood",
        "Buildings are made out of stone",
        "Buildings are made out of atoms",
        "Buildings are made out of building materials",
        "Cars are made out of metal",
        "Cars are made out of plastic",
    ]
    vectorstore = FAISS.from_texts(docs, embeddings)
    base_retriever = vectorstore.as_retriever()

    # LLM chain that decides whether the question is answered by the custom
    # knowledge base (Yes) or is unrelated small talk such as a greeting (No).
    relevance_template = PromptTemplate(
        input_variables=["question", "context"],
        template=(
            "You are checking whether a user question can be answered from a "
            "custom knowledge base.\n\n"
            "Knowledge base:\n{context}\n\n"
            "Question: {question}\n\n"
            "Can the question be answered with the information contained in "
            "the knowledge base above? Small talk such as greetings (for "
            "example 'hi' or 'hello') can NOT be answered from it.\n"
            "Answer with a single word, either Yes or No:"
        ),
    )
    relevance_chain = LLMChain(llm=llm, prompt=relevance_template)

    # Wrap the plain vector-store retriever in the custom retriever so that it
    # only returns documents when the answer really comes from the knowledge
    # base and returns an empty list for unrelated queries.
    retriever = CustomRetriever(
        base_retriever=base_retriever,
        relevance_chain=relevance_chain,
    )

    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    qa = ConversationalRetrievalChain.from_llm(
        llm,
        retriever,
        condense_question_prompt=CUSTOM_QUESTION_PROMPT,
        memory=memory
    )

    # Use the user's actual input instead of a hard-coded query so that the
    # answer and the returned documents match the question that was asked.
    query = user_input
    qa_result = qa({"question": query})
    print(f"Bot answer: {qa_result['answer']}")

    # The retriever now returns the relevant documents only when the answer
    # really comes from the custom knowledge base, and an empty list for
    # unrelated queries such as greetings.
    result = retriever.get_relevant_documents(query)

    return result


if __name__ == "__main__":
    user_input = "What are buildings made of?"
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result}")

    # Unrelated query (a greeting): the retriever returns an empty list.
    greeting = "hi, how are you?"
    greeting_result = run_agent(greeting)
    print("\n=== Greeting Result ===")
    print(f"Question: {greeting}")
    print(f"Answer: {greeting_result}")
