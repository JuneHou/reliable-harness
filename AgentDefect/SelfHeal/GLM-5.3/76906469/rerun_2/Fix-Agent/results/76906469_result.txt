from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Memory only works with *conversational* agent types, because only their
# prompt templates contain a {chat_history} placeholder.
# The memory_key must match that placeholder ("chat_history"), and since we
# are using a chat model (ChatOpenAI) with the chat conversational ReAct
# agent, the memory must return message objects, not strings
# (return_messages=True), otherwise the prompt formatting fails.
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True
)

agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    # CHAT_ZERO_SHOT_REACT_DESCRIPTION ignores attached memory: its prompt
    # has no {chat_history} variable, so the conversation history is never
    # injected and every call is stateless. To enable memory support we must
    # use a conversational ReAct agent instead.
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False
)

def run_agent(prompt: str) -> str:
    return agent_test.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    result2 = run_agent("what is the last question I asked?")
    print(result2)
