from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Memory must return message objects (return_messages=True) because the
# CHAT_CONVERSATIONAL_REACT_DESCRIPTION prompt uses a MessagesPlaceholder
# for "chat_history" that expects BaseMessage instances, not plain strings.
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True
)

# CHAT_ZERO_SHOT_REACT_DESCRIPTION does not support memory: its prompt
# (hwchase17/chat-zero-shot-react-description) has no "chat_history"
# placeholder, so an attached ConversationBufferMemory is simply ignored
# and the agent stays stateless. CHAT_CONVERSATIONAL_REACT_DESCRIPTION
# is the chat-model ReAct agent whose prompt includes "chat_history",
# which is what actually enables conversation memory.
agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False
)

def run_agent(prompt: str) -> str:
    return agent_test.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    result2 = run_agent("what is the last question I asked?")
    print(result2)
