from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# FIX (part 2): The conversational chat agent renders "chat_history" through a
# MessagesPlaceholder, so the memory must return message objects rather than a
# single concatenated string. Without return_messages=True, memory loading
# fails with a Pydantic ValidationError ("value is not a valid list").
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True
)

# FIX (part 1): CHAT_ZERO_SHOT_REACT_DESCRIPTION is a single-turn agent whose
# prompt has no "chat_history" input variable, so any memory attached to it is
# silently ignored (history never reaches the LLM). To enable memory support,
# use the conversational chat-model agent, whose prompt includes the
# conversation history:
agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False
)

def run_agent(prompt: str) -> str:
    return agent_test.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    result2 = run_agent("what is the last question I asked?")
    print(result2)
