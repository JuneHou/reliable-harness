from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# FIX 1: `return_messages=True` is required for chat-model based agents.
# The conversational chat agent's prompt holds `chat_history` in a
# MessagesPlaceholder, which expects a list of message objects instead of the
# plain string that ConversationBufferMemory returns by default.
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True,
)

# FIX 2: CHAT_ZERO_SHOT_REACT_DESCRIPTION is not a memory-aware agent type.
# Its ReAct prompt has no `chat_history` variable, so any memory attached to
# the executor is silently ignored (the history is loaded but never injected
# into the prompt). CHAT_CONVERSATIONAL_REACT_DESCRIPTION is the chat-model
# ReAct agent whose prompt includes a `chat_history` placeholder, which is
# what actually enables the attached ConversationBufferMemory.
agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False
)

def run_agent(prompt: str) -> str:
    return agent_test.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    result2 = run_agent("what is the last question I asked?")
    print(result2)
