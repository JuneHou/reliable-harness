def run_agent(prompt: str) -> str:
    api_key = os.getenv("OPENAI_API_KEY")
    llm = ChatOpenAI(openai_api_key=api_key)

    # ChatOpenAI.invoke() returns an AIMessage object, not a plain string.

    # alongside metadata (additional_kwargs, response_metadata, usage_metadata).
    # Accessing .content ensures we return the clean generated text.
    response = llm.invoke(prompt)
    return response.content
