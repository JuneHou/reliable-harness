import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

load_dotenv()


def run_agent(prompt: str) -> str:
    api_key = os.getenv("OPENAI_API_KEY")
    llm = ChatOpenAI(openai_api_key=api_key)

    # llm.invoke(...) does NOT return a plain string; it returns an
    # AIMessage object. That is why the getting-started tutorial examples
    # appear to "produce no output": only the object (or its repr) shows up
    # instead of the expected answer text. The actual assistant text is
    # stored in the `.content` attribute, so we extract it here.
    response = llm.invoke(prompt)
    return response.content


if __name__ == "__main__":
    user_prompt = "How can LangSmith help with testing?"
    answer = run_agent(user_prompt)
    print("LLM response:", answer)
