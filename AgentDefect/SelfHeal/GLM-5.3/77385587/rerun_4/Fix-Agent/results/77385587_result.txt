import os
import pickle

from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import EncoderBackedStore, LocalFileStore
from langchain.schema import Document
import nltk


def run_agent(query, store_location="./store_location"):
    """Answer `query` with a ParentDocumentRetriever whose parent documents
    AND child chunks are both persisted to disk.

    - Child chunks live in Chroma (persist_directory="./chroma_db_child").
    - Parent (full) documents live in a LocalFileStore-backed docstore at
      `store_location`. This replaces the non-persistent InMemoryStore, which
      died with the process and forced documents to be re-added every run.

    First run: documents are added and written to disk.
    Later runs: the whole structure is restored from disk and nothing is
    re-added.
    """

    nltk.download("punkt")

    hf_embed_path = "./hf_cache"

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining.")
    ]

    # Deterministic document ids: the same document always maps to the same
    # docstore key, which lets us detect on later runs what has already been
    # persisted (instead of re-adding everything with fresh random UUIDs).
    doc_ids = [f"doc_{i}" for i in range(len(docs))]

    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    # Child chunks: persisted on disk by Chroma via persist_directory.
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child"
    )

    # Parent documents: persist them to disk too.
    # LocalFileStore is a key/value store backed by the local file system
    # (one file per key, created automatically). EncoderBackedStore adapts it
    # so we can store Document objects under plain string keys.
    store = EncoderBackedStore(
        LocalFileStore(root_path=store_location),
        key_encoder=lambda key: key,
        value_serializer=lambda value: pickle.dumps(value),
        value_deserializer=lambda value: pickle.loads(value),
    )

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter
    )

    # Only add the documents that are not in the persistent docstore yet.
    already_stored = store.mget(doc_ids)
    docs_to_add = [
        doc for doc, stored in zip(docs, already_stored) if stored is None
    ]
    ids_to_add = [
        doc_id for doc_id, stored in zip(doc_ids, already_stored)
        if stored is None
    ]

    if docs_to_add:
        retriever.add_documents(docs_to_add, ids=ids_to_add)
        # Some chromadb versions require an explicit persist() call while
        # newer ones persist automatically - support both.
        try:
            if hasattr(vectorstore, "persist"):
                vectorstore.persist()
        except Exception:
            pass

    # Re-open the persisted child-chunk vectorstore, proving the child chunks
    # survive across runs (the small chunks are the child-level view).
    vector_db = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child"
    )

    matching_doc = vector_db.max_marginal_relevance_search(query, k=3)

    # Return the FULL parent documents. ParentDocumentRetriever searches the
    # child chunks and then fetches the corresponding parent documents from
    # the (now persistent) docstore, so the complete structure is restored
    # without re-adding any documents.
    full_docs = retriever.get_relevant_documents(query)

    return full_docs


if __name__ == "__main__":
    query = "What is RAG?"
    result = run_agent(query)
    print(result)
