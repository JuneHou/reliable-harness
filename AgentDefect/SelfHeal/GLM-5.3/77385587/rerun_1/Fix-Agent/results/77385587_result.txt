import json
import os

from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import LocalFileStore
from langchain.storage.encoder import EncoderBackedStore
from langchain.schema import Document
import nltk

# Metadata key ParentDocumentRetriever uses to link child chunks -> parent doc id
DOC_ID_KEY = "doc_id"


def _serialize(obj):
    """Turn Document objects into JSON-serializable dicts."""
    if isinstance(obj, Document):
        return {"page_content": obj.page_content, "metadata": obj.metadata}
    if isinstance(obj, (list, tuple)):
        return [_serialize(o) for o in obj]
    raise TypeError(f"Object of type {type(obj)} is not JSON serializable")


def _deserialize(raw):
    """Turn the stored JSON back into Document object(s)."""
    if isinstance(raw, bytes):
        raw = raw.decode("utf-8")
    data = json.loads(raw)
    if isinstance(data, list):
        return [
            Document(page_content=d.get("page_content", ""), metadata=d.get("metadata", {}))
            for d in data
        ]
    return Document(
        page_content=data.get("page_content", ""),
        metadata=data.get("metadata", {}),
    )


def build_persistent_docstore(root_path: str) -> EncoderBackedStore:
    """
    Persistent replacement for InMemoryStore().

    LocalFileStore keeps everything on disk, and EncoderBackedStore converts
    the Document objects to/from JSON so they can be written to those files.
    Together with Chroma's persist_directory, BOTH the parent documents and
    the child chunks survive a restart of the process.
    """
    return EncoderBackedStore(
        store=LocalFileStore(root_path),
        key_encoder=lambda k: k.decode() if isinstance(k, bytes) else k,
        serializer=lambda obj: json.dumps(obj, default=_serialize).encode("utf-8"),
        deserializer=_deserialize,
    )


def run_agent(query, store_location="./store_location"):

    load_dotenv()
    nltk.download("punkt", quiet=True)

    hf_embed_path = "./hf_cache"

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining.")
    ]

    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    # Child chunks (vectors) are persisted on disk by Chroma itself.
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory=os.path.join(store_location, "chroma_db_child")
    )

    # THE FIX: parent documents are stored in a file-backed docstore instead
    # of InMemoryStore(), so they persist between runs and the whole
    # parent <-> child structure can be restored without re-adding documents.
    store = build_persistent_docstore(os.path.join(store_location, "docstore"))

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter
    )

    # Ingest only on the very first run; on every later run the full structure
    # (child chunks in Chroma + parent docs in the file docstore) is simply
    # restored from disk.
    try:
        docstore_has_docs = len(list(store.yield_keys())) > 0
    except Exception:
        docstore_has_docs = False

    try:
        chroma_count = vectorstore._collection.count()
    except Exception:
        chroma_count = 0

    if not docstore_has_docs or chroma_count == 0:
        retriever.add_documents(docs)
        # Older langchain versions require an explicit flush of Chroma.
        if hasattr(vectorstore, "persist"):
            try:
                vectorstore.persist()
            except Exception:
                pass

    # MMR search over the *child* chunks (persisted in Chroma) ...
    matching_sub_docs = vectorstore.max_marginal_relevance_search(query, k=3)

    # ... then return the FULL parent documents fetched from the persistent
    # docstore (previously the raw child chunks were returned, which did not
    # match the expected output of a parent-document retriever).
    parent_docs = []
    seen_ids = set()
    for sub_doc in matching_sub_docs:
        parent_id = sub_doc.metadata.get(DOC_ID_KEY)
        if parent_id is None or parent_id in seen_ids:
            continue
        seen_ids.add(parent_id)
        parents = store.mget([parent_id])[0]
        if parents is None:
            continue
        if isinstance(parents, list):
            parent_docs.extend(parents)
        else:
            parent_docs.append(parents)

    return parent_docs


if __name__ == "__main__":
    query = "What is RAG?"
    result = run_agent(query)
    print(result)
    for doc in result:
        print("-" * 80)
        print(doc.page_content)
