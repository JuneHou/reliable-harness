from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.schema import Document
from langchain.storage import EncoderBackedStore, LocalFileStore
from langchain.storage.encoders import doc_to_json, json_to_doc
import nltk


def _build_persistent_docstore(root_path):
    """
    Build a *persistent* docstore for the PARENT documents.

    The bug: `InMemoryStore` keeps the parent documents only in RAM, so once
    the process exits the docstore is empty. The full structure of the
    ParentDocumentRetriever (child chunks in the vectorstore + parent
    documents in the docstore) could never be restored, and the documents
    had to be re-added (and re-embedded) on every run.

    The fix: back the docstore with a `LocalFileStore` (one file per parent
    document on disk, under `root_path`) wrapped in an `EncoderBackedStore`
    that knows how to serialize/deserialize `Document` objects. The parent
    documents now survive across runs and can be reloaded together with the
    Chroma child chunks without re-adding anything.
    """
    fs_store = LocalFileStore(root_path)

    # Preferred: official convenience helper that turns any ByteStore into a
    # Document-backed store (Document values are JSON-encoded on disk).
    try:
        from langchain.storage._lc_store import create_kv_docstore

        return create_kv_docstore(fs_store)
    except Exception:
        pass

    # Fallback: build the equivalent encoder-backed document store by hand.
    try:
        return EncoderBackedStore(
            store=fs_store,
            key_encoder_key="str",
            value_serializer=doc_to_json,
            value_deserializer=json_to_doc,
        )
    except TypeError:
        # Very old versions of EncoderBackedStore take a callable encoder.
        return EncoderBackedStore(
            fs_store,
            lambda key: key.encode("utf-8"),
            doc_to_json,
            json_to_doc,
        )


def _is_already_indexed(vectorstore, docstore):
    """
    Check whether both halves of the ParentDocumentRetriever were already
    persisted (child chunks in the Chroma vectorstore + parent documents in
    the docstore), so documents are NOT re-added (duplicated) on subsequent
    runs.
    """
    try:
        child_count = vectorstore._collection.count()
    except Exception:
        child_count = 0

    try:
        parent_count = len(list(docstore.yield_keys()))
    except Exception:
        parent_count = 0

    return child_count > 0 and parent_count > 0


def run_agent(query, store_location="./store_location", vectorstore_location="./chroma_db_child"):
    nltk.download("punkt")

    hf_embed_path = "./hf_cache"

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining.")
    ]

    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    # Child chunks: persisted on disk through Chroma's persist_directory.
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory=vectorstore_location
    )

    # Parent documents: persisted on disk through a file-backed docstore.
    # This replaces the non-persistent InMemoryStore.
    docstore = _build_persistent_docstore(store_location)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=docstore,
        child_splitter=child_splitter,
        search_kwargs={"k": 3}
    )

    # Index the documents only on the first run. On every later run the
    # child chunks (Chroma) and the parent documents (LocalFileStore) are
    # both restored from disk, without re-adding the documents.
    if not _is_already_indexed(vectorstore, docstore):
        retriever.add_documents(docs)
        # Older chroma/langchain versions need an explicit persist() call;
        # newer versions persist automatically.
        if hasattr(vectorstore, "persist"):
            try:
                vectorstore.persist()
            except Exception:
                pass

    # Query through the retriever instead of the raw child-chunk
    # vectorstore: child chunks are matched in Chroma and the *full parent
    # documents* are fetched back from the persistent docstore, so the
    # complete original documents are returned.
    matching_doc = retriever.get_relevant_documents(query)
    return matching_doc


if __name__ == "__main__":
    load_dotenv()
    query = "What is RAG?"
    result = run_agent(query)
    print(result)
