import json

from dotenv import load_dotenv

from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import EncoderBackedStore, LocalFileStore
from langchain.schema import Document

import nltk


def serialize_document(doc: Document) -> bytes:
    """Serialize a parent Document to JSON bytes so it can be stored on disk."""
    return json.dumps(
        {
            "page_content": doc.page_content,
            "metadata": doc.metadata,
        }
    ).encode("utf-8")


def deserialize_document(payload: bytes) -> Document:
    """Rebuild a parent Document from the JSON bytes stored on disk."""
    parsed = json.loads(payload.decode("utf-8"))
    return Document(
        page_content=parsed["page_content"],
        metadata=parsed.get("metadata", {}),
    )


def build_persistent_docstore(store_location: str) -> EncoderBackedStore:
    """
    Persistent replacement for the non-persistent InMemoryStore.

    LocalFileStore saves each parent document as a file under `store_location`,
    while EncoderBackedStore handles the (de)serialization between the
    `Document` objects used by ParentDocumentRetriever and raw bytes.
    """
    return EncoderBackedStore(
        store=LocalFileStore(store_location),
        # doc ids (uuid strings) are safe to use directly as file names
        key_encoder=lambda key: key,
        value_serializer=serialize_document,
        value_deserializer=deserialize_document,
    )


def run_agent(query, store_location="./store_location"):
    nltk.download("punkt")

    hf_embed_path = "./hf_cache"
    # Child chunks are persisted to disk by Chroma via persist_directory.
    chroma_path = "./chroma_db_child"

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."),
    ]

    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path,
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    # Child chunks -> persisted to disk by Chroma (persist_directory).
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory=chroma_path,
    )

    # Parent documents -> persisted to disk in `store_location`
    # (this replaces the non-persistent InMemoryStore).
    store = build_persistent_docstore(store_location)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter,
    )

    # Ingest the documents only on the very first run. From then on, both the
    # child chunks (Chroma) and the parent documents (docstore) are restored
    # from disk, so nothing is re-added or duplicated.
    # NOTE: if you previously ran the InMemoryStore version, delete the
    # ./chroma_db_child folder once to remove the duplicated chunks it
    # accumulated from repeated re-ingestion.
    if not list(store.yield_keys()):
        retriever.add_documents(docs)

    # Full structure restored from disk: the retriever finds the matching
    # child chunks in the persisted vectorstore and returns their FULL parent
    # documents looked up in the persisted docstore.
    parent_documents = retriever.get_relevant_documents(query)

    # Child chunks can still be searched directly in the persisted vectorstore.
    vector_db = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory=chroma_path,
    )
    matching_doc = vector_db.max_marginal_relevance_search(query, k=3)

    return matching_doc, parent_documents


if __name__ == "__main__":
    load_dotenv()
    query = "What is RAG?"
    matching_doc, parent_documents = run_agent(query)

    print("Child chunks (restored from the persisted Chroma collection):")
    for doc in matching_doc:
        print("-", doc.page_content)

    print("\nFull parent documents (restored from the persistent docstore):")
    for doc in parent_documents:
        print("-", doc.page_content)
