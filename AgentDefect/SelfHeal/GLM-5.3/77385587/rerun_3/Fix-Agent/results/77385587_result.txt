import os
import json

from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import EncoderBackedStore, LocalFileStore
from langchain.schema import Document
import nltk


def get_docstore(store_path):
    """
    Build a persistent (file-backed) docstore that keeps the parent documents
    on disk.

    The original code used ``InMemoryStore``, so every parent document was
    lost as soon as the process exited and the whole structure of the
    ParentDocumentRetriever had to be rebuilt (documents re-added) on the
    next run.

    ``LocalFileStore`` writes every docstore entry to a file inside
    ``store_path`` and ``EncoderBackedStore`` takes care of (de-)serializing
    the ``Document`` objects to/from JSON, so the parent documents survive
    across runs - just like the child chunks stored in the persisted Chroma
    collection.
    """
    # Make sure the target folder exists before creating the file store.
    os.makedirs(store_path, exist_ok=True)
    root = LocalFileStore(store_path)

    docstore = EncoderBackedStore(
        store=root,
        # Docstore keys are normally plain strings (uuid hex values), but some
        # code paths hand over bytes, so handle both.
        key_encoder=lambda k: k.decode("utf-8") if isinstance(k, bytes) else str(k),
        serializer=lambda v: json.dumps(v.dict()).encode("utf-8"),
        deserializer=lambda v: Document(**json.loads(v.decode("utf-8"))),
    )
    return docstore


def run_agent(query, store_location="./store_location"):
    """
    Index the documents (only on the first run) and query them.

    Both parts of the ParentDocumentRetriever are persisted:
      * child chunks  -> Chroma collection stored in "./chroma_db_child"
      * parent docs   -> file-backed docstore stored in "store_location"

    On every subsequent call the full structure is restored from disk, so the
    documents do NOT have to be added again.
    """
    nltk.download("punkt")

    hf_embed_path = "./hf_cache"

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining.")
    ]

    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    # Child chunks: Chroma persists them to disk in "./chroma_db_child".
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child"
    )

    # Parent documents: replaced the volatile InMemoryStore with a
    # persistent, file-backed docstore (JSON files under store_location).
    store = get_docstore(store_location)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter
    )

    # Add the documents only if the docstore is still empty. Otherwise the
    # full structure is restored from disk (re-adding would duplicate the
    # child chunks inside the Chroma collection on every run).
    if len(list(store.yield_keys())) == 0:
        retriever.add_documents(docs)

    vector_db = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child"
    )

    # Child chunks coming from the persisted vectorstore (as before).
    matching_doc = vector_db.max_marginal_relevance_search(query, k=3)

    # Full parent documents looked up in the persistent docstore: this only
    # works because the docstore is now persisted instead of in-memory.
    try:
        parent_docs = retriever.invoke(query)
    except AttributeError:
        # Older langchain versions do not expose the Runnable interface.
        parent_docs = retriever.get_relevant_documents(query)

    return matching_doc, parent_docs


if __name__ == "__main__":
    load_dotenv()
    query = "What is RAG?"
    child_docs, parent_docs = run_agent(query)

    print("=== Child chunks (persisted in Chroma: ./chroma_db_child) ===")
    for doc in child_docs:
        print(doc)
        print("-" * 80)

    print("\n=== Parent documents (restored from the persisted docstore) ===")
    for doc in parent_docs:
        print(doc)
        print("-" * 80)
