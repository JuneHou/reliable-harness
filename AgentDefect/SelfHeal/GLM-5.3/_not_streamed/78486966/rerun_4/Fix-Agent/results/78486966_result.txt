import json
from pathlib import Path

# FIX: Use RecursiveCharacterTextSplitter instead of CharacterTextSplitter.
# CharacterTextSplitter only splits on a single separator (default "\n\n").
# The input text has no blank lines, so no split point was ever found and the
# whole text was returned as one chunk. RecursiveCharacterTextSplitter falls
# back through ["\n\n", "\n", " ", ""] so oversized pieces are actually broken
# down to chunk_size with the requested chunk_overlap.
from langchain_text_splitters import RecursiveCharacterTextSplitter

# Directory where the chunks will be saved locally.
OUTPUT_DIR = Path("chunks_output")


def save_chunks_locally(chunks, output_dir=OUTPUT_DIR):
    """Persist the chunks to disk in several convenient formats."""
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    # 1) Plain text: one chunk per file (chunk_001.txt, chunk_002.txt, ...).
    for idx, chunk in enumerate(chunks, start=1):
        (output_dir / f"chunk_{idx:03d}.txt").write_text(chunk, encoding="utf-8")

    # 2) JSON: all chunks in a single structured file.
    with open(output_dir / "chunks.json", "w", encoding="utf-8") as f:
        json.dump(chunks, f, ensure_ascii=False, indent=2)

    # 3) Plain text list: all chunks in one file, separated by blank lines.
    with open(output_dir / "chunks.txt", "w", encoding="utf-8") as f:
        f.write("\n\n".join(chunks))

    return output_dir


def run_agent(text_raw):
    # FIX (parameter values): keep the requested chunk_size=30 / chunk_overlap=5
    # and let the recursive splitter enforce them via its fallback separators.
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=30,
        chunk_overlap=5,
        separators=["\n\n", "\n", " ", ""],  # explicit, mirrors defaults
        length_function=len,
        keep_separator=False,
        add_start_index=True,
    )
    chunks = text_splitter.split_text(text_raw)

    # Clean up leading/trailing whitespace on each chunk so blank lines don't
    # turn into phantom "chunks" of pure whitespace.
    chunks = [chunk.strip() for chunk in chunks if chunk.strip()]
    return chunks


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""

    chunks = run_agent(text_raw)

    print(chunks)
    print(f"\n\n {len(chunks)}")

    # Save the resulting chunks locally.
    output_path = save_chunks_locally(chunks)
    print(f"\nSaved {len(chunks)} chunks to: {output_path.resolve()}")
