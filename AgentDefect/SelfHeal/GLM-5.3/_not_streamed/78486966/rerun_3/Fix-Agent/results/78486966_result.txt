import json

from langchain_text_splitters import RecursiveCharacterTextSplitter


def run_agent(text_raw):
    # BUG: CharacterTextSplitter only splits on a single separator (its
    # default separator is "\n\n") and it never force-splits a segment that
    # exceeds chunk_size. Since this text contains no "\n\n", the entire text
    # was returned as one chunk.
    # FIX: Use RecursiveCharacterTextSplitter, which tries a hierarchy of
    # separators ("\n\n", "\n", " ", "") and therefore actually enforces
    # chunk_size / chunk_overlap, producing ~30-character chunks with a
    # 5-character overlap.
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=30,
        chunk_overlap=5,
        length_function=len,
        add_start_index=False,
    )
    chunks = text_splitter.split_text(text_raw)
    return chunks


def save_chunks(chunks, json_path="chunks.json", txt_path="chunks.txt"):
    """Save the resulting chunks locally.

    - chunks.json : machine-readable list of chunks (easy to reload later)
    - chunks.txt  : human-readable, one chunk per line
    """
    # Save as JSON so the chunks can be reloaded exactly as they were split.
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(chunks, f, ensure_ascii=False, indent=2)

    # Also save as a plain text file for easy inspection.
    with open(txt_path, "w", encoding="utf-8") as f:
        for i, chunk in enumerate(chunks, start=1):
            f.write(f"[Chunk {i}]\n{chunk}\n\n")


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)
    print(chunks)
    print(f'\n\n {len(chunks)}')

    # Persist the chunks locally
    save_chunks(chunks, json_path="chunks.json", txt_path="chunks.txt")
    print("Chunks saved to chunks.json and chunks.txt")
