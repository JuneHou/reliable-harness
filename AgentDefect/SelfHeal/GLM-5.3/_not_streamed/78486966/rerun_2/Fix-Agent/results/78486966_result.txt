import json
import os

from langchain_text_splitters import CharacterTextSplitter


def run_agent(text_raw):
    # BUG FIX: CharacterTextSplitter splits ONLY on the given `separator`.
    # The default separator is "\n\n" (a blank line), and this text contains
    # none, so the splitter could never break it up and the entire text was
    # returned as a single oversized chunk.
    # Changing the separator to "\n" makes the splitter split on every
    # newline, which lets chunk_size=30 and chunk_overlap=5 actually be
    # enforced as the user expects.
    text_splitter = CharacterTextSplitter(
        separator="\n",
        chunk_size=30,
        chunk_overlap=5,
        is_separator_regex=False,
    )
    chunks = text_splitter.split_text(text_raw)
    return chunks


def save_chunks_locally(chunks, path="chunks.json"):
    """Save the resulting chunks to a local file so they can be reused later."""
    with open(path, "w", encoding="utf-8") as f:
        json.dump(chunks, f, ensure_ascii=False, indent=2)
    return path


def load_chunks(path="chunks.json"):
    """Load previously saved chunks back from disk."""
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)
    print(chunks)
    print(f'\n\n {len(chunks)}')

    # Save the resulting chunks locally and verify they round-trip correctly.
    saved_path = save_chunks_locally(chunks)
    print(f"\nSaved {len(chunks)} chunks to {os.path.abspath(saved_path)}")

    loaded_chunks = load_chunks(saved_path)
    print(f"Loaded {len(loaded_chunks)} chunks back from disk:")
    for i, chunk in enumerate(loaded_chunks, start=1):
        print(f"{i}: {chunk!r}")
