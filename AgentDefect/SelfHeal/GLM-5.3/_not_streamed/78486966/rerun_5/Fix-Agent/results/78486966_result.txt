import json
from pathlib import Path

from langchain_text_splitters import CharacterTextSplitter


def run_agent(text_raw):
    # ------------------------------------------------------------------
    # BUG FIX: why the whole text came back as a single chunk
    # ------------------------------------------------------------------
    # CharacterTextSplitter does NOT cut the text at arbitrary positions.
    # It works in two stages:
    #   1) split the text on `separator`  (default: "\n\n" -- a blank line)
    #   2) merge the resulting pieces back together into chunks of
    #      up to `chunk_size` characters (with `chunk_overlap` between them)
    #
    # The source text contains only single newlines between sentences and no
    # blank lines, so the default separator "\n\n" never matches. The entire
    # text therefore becomes one single "piece" which is larger than
    # chunk_size=30, and it is returned as one oversized chunk (LangChain
    # logs: "Created a chunk of size ..., which is longer than the specified 30").
    #
    # Fix: pass an explicit separator.
    #   - separator=""  -> split on individual characters, so chunks are
    #                      EXACTLY chunk_size chars with chunk_overlap overlap
    #   - separator=" " -> split on words, chunks <= chunk_size, word-aligned
    #                      (use this instead if you want clean word boundaries)
    # ------------------------------------------------------------------
    text_splitter = CharacterTextSplitter(
        separator="",              # <-- the fix: character-level splitting
        chunk_size=30,
        chunk_overlap=5,
        is_separator_regex=False,
    )

    # Strip the leading/trailing newlines that come from the triple-quoted
    # string literal, so the first and last chunks contain real text.
    chunks = text_splitter.split_text(text_raw.strip())
    return chunks


def save_chunks_locally(chunks, txt_path="chunks.txt", json_path="chunks.json"):
    """Persist the resulting chunks locally so they can be reused later."""
    # 1) Plain-text file: each chunk separated by a delimiter line
    Path(txt_path).write_text("\n---\n".join(chunks), encoding="utf-8")

    # 2) JSON file: structured storage, easy to load back
    #    (e.g. before embedding into a vector store)
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(chunks, f, ensure_ascii=False, indent=2)

    return txt_path, json_path
