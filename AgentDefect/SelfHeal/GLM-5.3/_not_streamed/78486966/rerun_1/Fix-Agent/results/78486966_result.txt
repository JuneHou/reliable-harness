import json
from pathlib import Path

from langchain_text_splitters import RecursiveCharacterTextSplitter

CHUNKS_JSON_PATH = "chunks.json"
CHUNKS_TXT_PATH = "chunks.txt"


def run_agent(text_raw):
    """
    Split the raw text into chunks of at most 30 characters with a
    5-character overlap.

    Why the original code failed:
    CharacterTextSplitter splits on a SINGLE separator (default "\\n\\n").
    The input text contains no blank lines, so no separator was ever found,
    the whole text became one "split", and -- because CharacterTextSplitter
    cannot subdivide an oversized split -- the entire text was returned as a
    single chunk, ignoring chunk_size / chunk_overlap.

    RecursiveCharacterTextSplitter recursively tries several separators
    ("\\n\\n" -> "\\n" -> " " -> "") so it can always break the text down far
    enough to honor chunk_size=30 with chunk_overlap=5.
    """
    if text_raw is None:
        return []

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=30,
        chunk_overlap=5,
        separators=["\n\n", "\n", " ", ""],
    )
    chunks = text_splitter.split_text(text_raw)
    return chunks


def save_chunks_locally(chunks, json_path=CHUNKS_JSON_PATH, txt_path=CHUNKS_TXT_PATH):
    """
    Persist the resulting chunks to local files:

    * JSON file: a plain list of chunks (easy to re-load programmatically).
    * Text file: a human-readable listing with per-chunk metadata.
    """
    try:
        Path(json_path).write_text(
            json.dumps(chunks, indent=2, ensure_ascii=False),
            encoding="utf-8",
        )
    except OSError as exc:
        print(f"Could not save chunks to {json_path!r}: {exc}")

    try:
        with open(txt_path, "w", encoding="utf-8") as f:
            f.write(f"Total chunks: {len(chunks)}\n\n")
            for i, chunk in enumerate(chunks, start=1):
                f.write(f"[Chunk {i}] ({len(chunk)} chars)\n{chunk}\n\n")
    except OSError as exc:
        print(f"Could not save chunks to {txt_path!r}: {exc}")


if __name__ == "__main__":
    text_raw = """
Water is life's matter and matrix, mother, and medium. There is no life without water.
Save water, secure the future.
Conservation is the key to a sustainable water supply.
Every drop saved today is a resource for tomorrow.
Let's work together to keep our rivers flowing and our oceans blue.
"""
    chunks = run_agent(text_raw)

    # Verify the fix: every chunk now respects chunk_size=30.
    print(chunks)
    print(f"\n\n {len(chunks)} chunks")
    print(f"max chunk length: {max(len(c) for c in chunks)} chars")

    # Save the chunks locally.
    save_chunks_locally(chunks)
    print(
        f"Saved {len(chunks)} chunks to '{CHUNKS_JSON_PATH}' "
        f"and '{CHUNKS_TXT_PATH}'."
    )
