import os
from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import LocalFileStore
from langchain.storage.encoder import EncoderBackedStore
from langchain.schema import Document
import nltk


def _docstore_key_encoder(key: str) -> str:
    """Encode a docstore key so it is filesystem safe (no '/' etc.)."""
    import base64

    return base64.urlsafe_b64encode(key.encode("utf-8")).decode("utf-8")


def _make_persistent_docstore(root_path: str):
    """
    Build a docstore backed by LocalFileStore (files on disk) that persists
    both parent documents and child chunks across sessions.

    Documents are stored as JSON, so they can be reloaded with Document(_dict)
    without having to re-add the original documents.
    """
    import json

    base_store = LocalFileStore(root_path)

    def key_encoder(key: str) -> str:
        return _docstore_key_encoder(key)

    def value_serializer(value) -> bytes:
        # Document -> JSON bytes (works for any Document-like object)
        if hasattr(value, "dict"):
            return json.dumps(value.dict()).encode("utf-8")
        if hasattr(value, "page_content"):
            return json.dumps(
                {
                    "page_content": value.page_content,
                    "metadata": getattr(value, "metadata", {}),
                }
            ).encode("utf-8")
        raise TypeError(f"Cannot serialize docstore value: {type(value)}")

    def value_deserializer(serialized: bytes):
        payload = json.loads(serialized.decode("utf-8"))
        # Payload may be a full Document dump or a simple page_content dict
        if "page_content" in payload and "type" in payload and payload.get("type") == "Document":
            return Document(**payload)
        if "page_content" in payload:
            return Document(
                page_content=payload["page_content"],
                metadata=payload.get("metadata", {}),
            )
        raise ValueError(f"Cannot deserialize docstore payload: {payload}")

    return EncoderBackedStore(
        store=base_store,
        key_encoder=key_encoder,
        value_serializer=value_serializer,
        value_deserializer=value_deserializer,
    )


def run_agent(query, store_location="./store_location"):
    """
    The root cause of the mismatch: ParentDocumentRetriever's docstore was an
    InMemoryStore, so parent documents (and any child chunk docs) disappeared
    when the process ended, while the Chroma vectorstore was persisted on
    disk. After a restart, the vectorstore still returned matching child
    chunks, but looking up their parent IDs in the empty in-memory docstore
    failed / returned nothing, so results did not match expectations.

    Fix: replace the ephemeral InMemoryStore with a persistent, file-backed
    docstore (LocalFileStore + EncoderBackedStore) so both the parent
    documents and the child chunks survive across sessions without re-adding
    documents.
    """
    nltk.download("punkt")

    hf_embed_path = "./hf_cache"

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining.")
    ]

    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child"
    )

    # --- FIX: persistent docstore instead of InMemoryStore ---
    store = _make_persistent_docstore(store_location)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter
    )

    # Only add documents if the persistent docstore is empty (first run).
    # On subsequent runs the parent docs are already saved on disk, so we do
    # not re-add them (which would create duplicate child chunks).
    doc_ids = list(store.yield_keys())
    if not doc_ids:
        retriever.add_documents(docs)

    vector_db = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child"
    )

    matching_doc = vector_db.max_marginal_relevance_search(query, k=3)

    # Demonstrate that the parent document can be resolved from the persisted
    # docstore across sessions (i.e., retrieval via the ParentDocumentRetriever
    # works without re-adding documents).
    parent_docs = retriever.get_relevant_documents(query)
    if parent_docs:
        print("Parent documents resolved from persistent docstore:")
        for d in parent_docs:
            print("-", d.page_content[:80], "...")

    return matching_doc


if __name__ == "__main__":
    load_dotenv()
    query = "What is RAG?"
    result = run_agent(query)
    print(result)
