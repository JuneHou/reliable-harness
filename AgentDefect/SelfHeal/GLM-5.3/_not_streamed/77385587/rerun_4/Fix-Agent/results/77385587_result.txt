import os
import pickle

from dotenv import load_dotenv

from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import EncoderBackedStore, LocalFileStore
from langchain.schema import Document
import nltk

CHROMA_PERSIST_DIR = "./chroma_db_child"


def build_persistent_docstore(store_location: str) -> EncoderBackedStore:
    """
    Build a docstore for the parent documents that persists to disk.

    The original code used ``InMemoryStore()``, which is ephemeral: as soon as
    the process exits, every parent document is lost. The child chunks survive
    (Chroma has a ``persist_directory``), but on the next session the
    ParentDocumentRetriever can no longer map child chunks back to their
    parent documents, and re-adding the documents duplicates the chunks that
    are already stored in Chroma.

    ``LocalFileStore`` is a persistent key/value store, but it can only hold
    raw bytes. Wrapping it in an ``EncoderBackedStore`` that pickles the
    ``Document`` objects on write and unpickles them on read yields a
    persistent ``BaseStore[str, Document]`` that can be used directly as the
    ``docstore`` of ``ParentDocumentRetriever`` (passing a bare LocalFileStore
    as docstore raises a TypeError because mset expects bytes).
    """
    os.makedirs(store_location, exist_ok=True)
    return EncoderBackedStore(
        store=LocalFileStore(store_location),
        key_encoder=lambda key: key,
        value_encoder=pickle.dumps,
        value_decoder=pickle.loads,
    )


def ensure_nltk_punkt() -> None:
    """Download the NLTK punkt tokenizer only when it is not present yet."""
    try:
        nltk.data.find("tokenizers/punkt")
    except LookupError:
        nltk.download("punkt", quiet=True)


def run_agent(query, store_location="./store_location"):
    ensure_nltk_punkt()

    hf_embed_path = "./hf_cache"

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining.")
    ]

    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path,
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    # Child chunks: persisted on disk by Chroma thanks to persist_directory.
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory=CHROMA_PERSIST_DIR,
    )

    # Parent documents: persisted on disk through the LocalFileStore-backed
    # docstore (this replaces the ephemeral InMemoryStore and finally makes
    # use of the store_location parameter).
    store = build_persistent_docstore(store_location)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter,
        search_kwargs={"k": 3},
    )

    # Only index the documents on the very first run. Both storage layers are
    # persistent now, so re-running the script must NOT re-add (and therefore
    # duplicate) the documents.
    child_count = len(vectorstore.get().get("ids", []))
    parent_count = len(list(store.yield_keys()))
    if child_count == 0 and parent_count == 0:
        retriever.add_documents(docs)
        print("Indexed documents into the vectorstore and the persistent docstore.")
    else:
        print(
            f"Loaded existing index from disk ({child_count} child chunks, "
            f"{parent_count} parent documents) - skipping re-ingestion."
        )

    # Retrieve through the retriever so that BOTH stores are used: the child
    # chunks are fetched from the persisted Chroma collection and the
    # corresponding parent documents are fetched back from the persisted
    # docstore. This now works identically on the first run and on every
    # subsequent session without re-adding anything.
    if hasattr(retriever, "invoke"):
        matching_doc = retriever.invoke(query)
    else:  # older langchain versions
        matching_doc = retriever.get_relevant_documents(query)

    return matching_doc


if __name__ == "__main__":
    load_dotenv()
    query = "What is RAG?"
    result = run_agent(query)
    for doc in result:
        print(doc.page_content)
        print("-" * 80)
