import os
from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import LocalFileStore, EncoderByteStore
from langchain.schema import Document
import nltk


def run_agent(query, store_location="./store_location"):

    # Download the punkt tokenizer once (guard so it is not redone on every call)
    try:
        nltk.data.find("tokenizers/punkt")
    except LookupError:
        nltk.download("punkt")

    hf_embed_path = "./hf_cache"

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining.")
    ]

    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child"
    )

    # --- FIX: replace the ephemeral InMemoryStore with a persistent docstore ---
    # LocalFileStore persists key -> value (bytes) pairs on disk, so the parent
    # documents survive across sessions. Because ParentDocumentRetriever stores
    # Document objects in the docstore, we need a byte store that can serialize
    # and deserialize them: EncoderByteStore wraps LocalFileStore with a JSON
    # (json.dumps / json.loads) encoder.
    os.makedirs(store_location, exist_ok=True)
    file_store = LocalFileStore(root_path=store_location)
    store = EncoderByteStore(store=file_store)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter
    )

    # Only (re-)add documents if they are not already in the persistent docstore.
    # A simple precondition check: if the persisted docstore already holds our
    # parent documents, skip ingestion entirely so documents are not re-added
    # (duplicated) on every run.
    existing_ids = [k.decode() if isinstance(k, bytes) else str(k) for k in store.yield_keys()]
    if not existing_ids:
        retriever.add_documents(docs)
    else:
        print(f"Docstore already populated with {len(existing_ids)} parent document(s); skipping re-ingestion.")

    # Persist the Chroma vectorstore (child chunks + parent ids) to disk
    vectorstore.persist()

    vector_db = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child"
    )

    # --- FIX: use the ParentDocumentRetriever itself for retrieval ---
    # The retriever first matches small child chunks in the vectorstore, then
    # looks up the parent documents in the (now persistent) docstore. This is
    # what makes the persistence meaningful: matching parent documents are
    # returned after a restart instead of raw ephemeral child chunks.
    matching_doc = retriever.get_relevant_documents(query, k=3)

    return matching_doc


if __name__ == "__main__":
    query = "What is RAG?"
    result = run_agent(query)
    for doc in result:
        print(doc.page_content)
        print("-" * 80)
