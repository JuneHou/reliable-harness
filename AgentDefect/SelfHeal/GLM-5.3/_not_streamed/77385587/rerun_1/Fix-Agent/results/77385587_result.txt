import os
import pickle
import warnings

from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import EncoderBackedStore, LocalFileStore
from langchain.schema import Document
import nltk


def _build_persistent_docstore(store_location):
    """Build a ParentDocumentRetriever docstore that survives restarts.

    The bug: ``InMemoryStore`` is ephemeral, so every session started with an
    empty docstore, forcing the documents to be re-added (and duplicating the
    child chunks in Chroma at the same time).

    The fix: back the docstore with a ``LocalFileStore``, which persists
    (key, value) pairs as files on disk.  ``LocalFileStore`` is a ByteStore
    (its values must be ``bytes``), while ``ParentDocumentRetriever`` stores
    ``Document`` objects, so we wrap it in an ``EncoderBackedStore`` that
    pickles each Document on write and unpickles it on read.
    """
    return EncoderBackedStore(
        store=LocalFileStore(store_location),
        # Docstore keys are uuid hex strings, which are valid file names;
        # decode bytes defensively just in case.
        key_encoder=lambda key: key.decode("utf-8") if isinstance(key, bytes) else key,
        value_serializer=lambda doc: pickle.dumps(doc),
        value_deserializer=lambda blob: pickle.loads(blob),
    )


def _docstore_is_empty(store):
    """True if the persistent docstore holds no parent documents yet."""
    try:
        return sum(1 for _ in store.yield_keys()) == 0
    except Exception:
        # If the store cannot be inspected, assume it is empty so documents
        # are (re-)added and retrieval keeps working.
        return True


def _indexed_child_count(vectorstore):
    """Number of child chunks already indexed in Chroma (-1 if unknown)."""
    try:
        return int(vectorstore._collection.count())
    except Exception:
        return -1


def _persist_vectorstore(vectorstore):
    """Flush Chroma to disk (older versions require an explicit persist call)."""
    try:
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            vectorstore.persist()
    except Exception:
        # Newer chromadb/langchain versions persist automatically.
        pass


def _make_retriever(vectorstore, store, child_splitter, k=3):
    """Create the ParentDocumentRetriever in a version-tolerant way."""
    for extra_kwargs in (
        {"child_search_kwargs": {"k": k}},
        {"search_kwargs": {"k": k}},
        {},
    ):
        try:
            return ParentDocumentRetriever(
                vectorstore=vectorstore,
                docstore=store,
                child_splitter=child_splitter,
                **extra_kwargs,
            )
        except Exception:
            continue
    raise RuntimeError("Could not initialize ParentDocumentRetriever")


def run_agent(query, store_location="./store_location"):

    load_dotenv()
    nltk.download("punkt")

    hf_embed_path = "./hf_cache"

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),
        Document(page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."),
        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining.")
    ]

    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    # Child chunks are persisted by Chroma through persist_directory, so this
    # same call transparently re-opens the existing collection on every
    # subsequent run (no need to build a second Chroma instance).
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child"
    )

    # FIX: persist the parent documents on disk instead of using the
    # ephemeral InMemoryStore().  Both the parent documents and the child
    # chunks now survive across sessions.
    store = _build_persistent_docstore(store_location)

    retriever = _make_retriever(vectorstore, store, child_splitter, k=3)

    # FIX: only ingest the corpus once.  On later runs the parent documents
    # are reloaded from the persistent docstore and the child chunks from
    # Chroma, so re-adding would only create duplicates in both stores.
    if _docstore_is_empty(store) or _indexed_child_count(vectorstore) == 0:
        retriever.add_documents(docs)
        _persist_vectorstore(vectorstore)

    # FIX: retrieve through the retriever itself so the persisted docstore is
    # actually used: the child chunks are matched in Chroma, and the
    # corresponding (full) parent documents are looked up in the docstore.
    try:
        matching_doc = retriever.invoke(query)
    except AttributeError:
        # Older langchain versions only expose get_relevant_documents().
        matching_doc = retriever.get_relevant_documents(query)

    return matching_doc


if __name__ == "__main__":
    query = "What is RAG?"
    result = run_agent(query)
    print(result)
