from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# ---------------------------------------------------------------------------
# Root cause
# ----------
# LlamaIndex's default in-memory vector store (SimpleVectorStore) serializes
# node metadata when nodes are inserted (see `node_to_metadata_dict` in
# `llama_index.core.vector_stores.utils`, which JSON-encodes the metadata
# dict). A metadata value of Python ``None`` is not preserved as a value the
# filter layer can compare against, and ``MetadataFilter.value`` is not meant
# to carry ``None`` either. As a result, an ``EQ`` filter built with
# ``value=None`` silently matches *no* documents -- even documents whose
# metadata field was explicitly set to ``None``.
#
# Fix
# ---
# Encode "null / missing" as an explicit sentinel string when indexing, and
# filter on that sentinel with ``FilterOperator.EQ``. This keeps the value
# round-trippable through the store's metadata serialization and gives the
# filter an actual value to compare against (it is also the portable pattern
# for production vector stores, which have no notion of a null metadata value).
# ---------------------------------------------------------------------------
NULL_VALUE = "null"


def normalize_metadata(metadata: dict) -> dict:
    """Convert ``None`` metadata values into the filterable NULL_VALUE sentinel.

    Use this when building nodes so that "missing"/null fields can be
    retrieved later with an EQ metadata filter.
    """
    return {
        key: NULL_VALUE if value is None else value
        for key, value in metadata.items()
    }


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node, encoding the missing start_date as the sentinel string so
    # that it survives indexing and can be matched by an EQ metadata filter.
    # (Storing a raw ``None`` makes the field unfilterable.)
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=normalize_metadata({"start_date": None}),
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter for the sentinel value that represents a null start_date.
    # (Filtering with value=None matches nothing, because None is not a
    # storable/filterable metadata value.)
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_VALUE,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
