import os
import random
from typing import Any, Dict, List

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.vectorstores.base import VectorStoreRetriever
from langchain.document_loaders import PyPDFLoader
from langchain.schema import Document
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]


def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


class FilterableConversationalRetrievalChain(ConversationalRetrievalChain):
    """A ConversationalRetrievalChain that accepts a metadata filter per query.

    Problem: a plain ConversationalRetrievalChain silently ignores a "filter"
    key passed in its input dict, so retrieval runs over the ENTIRE vector
    store (paper1.pdf AND paper2.pdf) instead of only the requested documents.

    Fix: this subclass pops the optional "filter" input and injects it into the
    vector store retriever's search_kwargs right before retrieval happens, so a
    single chain instance can be re-used with a different metadata filter at
    every query (no need to build a separate chain per filter).
    """

    def _apply_filter(self, inputs: Dict[str, Any]) -> None:
        """Pop the "filter" input and apply it to the retriever's search_kwargs."""
        metadata_filter = inputs.pop("filter", None)
        search_kwargs = getattr(self.retriever, "search_kwargs", None)
        if search_kwargs is None:
            # Retriever is not a VectorStoreRetriever -> nothing to filter on.
            return
        if metadata_filter:
            search_kwargs["filter"] = metadata_filter
        else:
            # Make sure a filter from a previous query does not leak into this one.
            search_kwargs.pop("filter", None)

    def _get_docs(self, inputs: Dict[str, Any], **kwargs: Any) -> List[Document]:
        self._apply_filter(inputs)
        return super()._get_docs(inputs, **kwargs)

    async def _aget_docs(self, inputs: Dict[str, Any], **kwargs: Any) -> List[Document]:
        self._apply_filter(inputs)
        return await super()._aget_docs(inputs, **kwargs)


def run_agent(query, pdf_paths, metadata_filter=None):
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            text.metadata["source"] = pdf_path

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    retriever = VectorStoreRetriever(vectorstore=db)

    qa = FilterableConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever,
        verbose=True,
        return_source_documents=True)

    # The filter is now honoured at query time: only chunks whose metadata
    # matches this filter are retrieved and used to answer the question.
    if metadata_filter is None:
        # Default of the original script: answer only from paper2.pdf.
        metadata_filter = {'source': 'paper2.pdf'}
    # if it answers from paper2.pdf, then the filtering is working correctly.

    result = qa({
        "question": query,
        "chat_history": [],
        "filter": metadata_filter,
    })

    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    result = run_agent(query, pdf_paths, metadata_filter={'source': 'paper2.pdf'})

    print(result['answer'])

class ChromaRetriever(BaseRetriever):
    """Retriever that applies a metadata 'where' filter on Chroma at query time.

    ConversationalRetrievalChain ignores any extra 'filter' key passed in the
    input dict, so filtering must be done inside get_relevant_documents.
    """

    vectorstore: Chroma
    search_kwargs: Dict[str, Any] = {}

    class Config:
        arbitrary_types_allowed = True

    def get_relevant_documents(self, query: str, **kwargs: Any) -> List[Document]:
        """Retrieve documents from the vectorstore, applying the metadata filter.

        The Chroma wrapper's similarity_search accepts a `filter` (a.k.a.
        `where`) argument that is forwarded to the underlying Chroma query, so
        only documents whose metadata matches are considered for the LLM.
        """
        search_kwargs = self.search_kwargs.copy()
        search_kwargs.update(kwargs)
        docs = self.vectorstore.similarity_search(
            query=query,
            k=search_kwargs.pop("k", 4),
            filter=search_kwargs.pop("filter", None),
        )
        return docs

    async def aget_relevant_documents(self, query: str, **kwargs: Any) -> List[Document]:
        return self.get_relevant_documents(query, **kwargs)


def run_agent(query, pdf_paths, filter=None):
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            text.metadata["source"] = pdf_path

        verbose=True,
        return_source_documents=True)

    result = qa({"question": query, "chat_history": []})

    return result
