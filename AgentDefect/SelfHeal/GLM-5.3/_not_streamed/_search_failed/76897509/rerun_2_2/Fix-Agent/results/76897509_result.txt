_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]

def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


class FilteredConversationalRetrievalChain(ConversationalRetrievalChain):
    """ConversationalRetrievalChain that honours a metadata filter passed per query.

    The stock chain only reads "question" and "chat_history" from its inputs, so
    a "filter" key passed to qa({...}) is silently dropped (extra input keys are
    allowed but never used) and the retrieval runs unfiltered over every chunk
    in the vector store, i.e. the answer can be built from any indexed document
    (paper1.pdf included) instead of only the requested one.

    This subclass intercepts the chain inputs, extracts "filter" and injects it
    into the retriever's search_kwargs. VectorStoreRetriever forwards its
    search_kwargs to Chroma.similarity_search(query, k, filter=...), so the
    filter is applied at retrieval time on every call -- there is no need to
    build a separate chain/retriever instance per document filter.
    """

    def _apply_filter(self, inputs: Dict[str, Any]) -> None:
        """Move the per-query "filter" from the chain inputs onto the retriever."""
        filter = inputs.pop("filter", None)
        search_kwargs = getattr(self.retriever, "search_kwargs", None)
        if search_kwargs is None:
            # This retriever does not support search_kwargs; nothing to do.
            return
        if filter:
            # Handed down to Chroma as the metadata "where" clause, e.g.
            # {"source": "paper2.pdf"} -> only paper2.pdf chunks are returned.
            search_kwargs["filter"] = filter
        else:
            # No filter requested for this query: make sure a previously applied
            # filter does not leak into this (or a later) call.
            search_kwargs.pop("filter", None)

    def _get_docs(
        self,
        *args: Any,
        run_manager: Optional[CallbackManagerForChainRun] = None,
        **kwargs: Any,
    ) -> List[Document]:
        # The signature of _get_docs differs across langchain versions
        # (_get_docs(inputs) / _get_docs(new_question, inputs) /
        #  _get_docs(new_question, inputs, run_manager=...)), so locate the
        # inputs dict in a signature-agnostic way.
        inputs = None
        for arg in args:
            if isinstance(arg, dict):
                inputs = arg
                break
        if inputs is None:
            inputs = kwargs.get("inputs")

        # Apply the metadata filter (if any) before the actual retrieval happens.
        if isinstance(inputs, dict) and "filter" in inputs:
            self._apply_filter(inputs)

        # Forward to the parent implementation, passing run_manager through only
        # if the installed version's _get_docs actually expects it.
        parent = super()._get_docs
        if "run_manager" in inspect.signature(parent).parameters:
            kwargs["run_manager"] = (
                run_manager or CallbackManagerForChainRun.get_noop_manager()
            )
        else:
            kwargs.pop("run_manager", None)
        return parent(*args, **kwargs)


def run_agent(query, pdf_paths, filter=None):
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))
    
    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()
        
        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)
        
        for text in texts:
            text.metadata["source"] = pdf_path
        
        all_texts.extend(texts)
    
    db = Chroma.from_documents(all_texts, embeddings)
    
    qa = FilteredConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        VectorStoreRetriever(vectorstore=db),
        verbose=True,
        return_source_documents=True)

    # The metadata filter is now applied dynamically at query time: only chunks
    # whose metadata matches the filter (here source == "paper2.pdf") are
    # retrieved, so the answer must come from paper2.pdf.
    result = qa({"question": query, "chat_history": [], "filter": filter})
    
    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    filter = {'source': 'paper2.pdf'} #it means the information from paper2.pdf should be used to answer the query.
    #if it answers from paper2.pdf, then the filtering is working correctly.

    result = run_agent(query, pdf_paths, filter=filter)
    
    print(result['answer'])

    # Sanity check: every retrieved source document must come from paper2.pdf.
    if result.get("source_documents") is not None:
        sources = sorted({
            doc.metadata.get("source", "<unknown>")
            for doc in result["source_documents"]
        })
        print(f"\nSources used to answer: {sources}")
