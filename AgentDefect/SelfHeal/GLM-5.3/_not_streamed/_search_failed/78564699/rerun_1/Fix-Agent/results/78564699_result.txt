import json
import os
from llama_index.llms.llama_cpp import LlamaCPP


def run_agent(query):
    # Use the smaller 7B model instead of 13B.
    # A 13B Q4_0 model is far too slow for CPU-only inference
    # (n_gpu_layers=0), which makes generation appear to hang
    # indefinitely without producing any output. The 7B Q4_0
    # quantized model is small enough to generate tokens at a
    # usable speed on CPU, so inference actually completes.
    MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-7B-chat-GGUF/resolve/main/llama-2-7b-chat.Q4_0.gguf"
    MODEL_PATH = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if "llama-2-7b-chat" in paths:
                MODEL_URL = None
                MODEL_PATH = paths["llama-2-7b-chat"]

    llm = LlamaCPP(
        model_url=MODEL_URL,
        model_path=MODEL_PATH,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={"n_gpu_layers": 0},
        verbose=True,
    )

    response = llm.complete(query)

    return response

if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
