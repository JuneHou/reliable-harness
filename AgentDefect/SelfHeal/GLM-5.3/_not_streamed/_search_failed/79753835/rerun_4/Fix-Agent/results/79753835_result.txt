from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer
import os

MODEL_NAME = "BAAI/bge-m3"


def create_embedding_model(_chunk_size=None):
    print('loading embeddings...')
    os.makedirs("./embeddings", exist_ok=True)
    _embed_model = HuggingFaceEmbedding(
        model_name=MODEL_NAME,
        max_length=_chunk_size,
        cache_folder='./embeddings',
    )
    print('embeddings loaded')
    return _embed_model


def get_model_tokenizer():
    """Load the tokenizer BGE-M3 itself tokenizes with.

    HuggingFaceEmbedding wraps sentence-transformers, which tokenizes with
    AutoTokenizer.from_pretrained(MODEL_NAME), so this is the very same
    tokenizer the embedding pipeline feeds to the model.
    """
    return AutoTokenizer.from_pretrained(MODEL_NAME, cache_folder='./embeddings')


def tokenize_like_the_model(tokenizer, max_length):
    """Build the tokenizer callable handed to TokenCountingHandler.

    TokenCountingHandler counts tokens as ``len(tokenizer(text))`` and, when no
    tokenizer is supplied, silently falls back to LlamaIndex's *global*
    tokenizer -- tiktoken's cl100k_base (the gpt-3.5-turbo encoding).  That is
    why the reported embedding token count never matched the BGE-M3 tokenizer.

    This reproduces the pre-processing HuggingFaceEmbedding
    (sentence-transformers) applies to each text before running the model:
      * the model's own tokenizer (not cl100k_base),
      * special tokens are added (<s> ... </s>), so they are counted too,
      * truncation="longest_first" up to the max_seq_length that was passed to
        HuggingFaceEmbedding(max_length=...),
      * a single text is padded to nothing, so padding can be ignored here.

    A plain list of ids is returned so that len() really is the token count
    (handing over the raw HF tokenizer object would make len() operate on a
    BatchEncoding instead of on the token ids).
    """
    def _tokenize(text: str) -> list:
        return tokenizer.encode(
            text,
            add_special_tokens=True,
            truncation="longest_first",
            max_length=max_length,
        )

    return _tokenize


def count_tokens(text: str, tokenizer, max_length=None) -> int:
    """Standalone (cheap) token count -- same pre-processing, no embedding run."""
    encoded = tokenizer(
        text,
        add_special_tokens=True,
        truncation="longest_first",
        max_length=max_length,
    )
    return len(encoded['input_ids'])


def run_agent(text: str):
    chunk_size = 512

    tokenizer = get_model_tokenizer()

    # FIX: count embedding tokens with the model's own tokenizer.  Without
    # this, TokenCountingHandler() uses the global tiktoken cl100k_base
    # tokenizer, which is why the two numbers differed.
    token_counter = TokenCountingHandler(
        tokenizer=tokenize_like_the_model(tokenizer, chunk_size)
    )
    callback_manager = CallbackManager([token_counter])

    embed_model = create_embedding_model(_chunk_size=chunk_size)

    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    embedding_vector = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    token_count = count_tokens(text, tokenizer, max_length=chunk_size)
    return embedding_tokens, token_count


if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
    print(f"The tokenizer broke the text into {token_count} tokens")
