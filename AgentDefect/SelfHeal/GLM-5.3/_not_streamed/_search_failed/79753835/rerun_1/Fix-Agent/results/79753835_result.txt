from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer
import os


def create_embedding_model(_chunk_size=None):
    print('loading embeddings...')
    if os.path.exists('./embeddings/models--BAAI--bge-m3'):
        _emb_model_name = "BAAI/bge-m3"
        _embed_model = HuggingFaceEmbedding(model_name=_emb_model_name, max_length=_chunk_size,
                                            cache_folder='./embeddings')
    else:
        os.makedirs("./embeddings", exist_ok=True)
        _emb_model_name = "BAAI/bge-m3"
        _embed_model = HuggingFaceEmbedding(model_name=_emb_model_name, max_length=_chunk_size,
                                            cache_folder='./embeddings')
    print('embeddings loaded')
    return _embed_model


def count_tokens_standalone(text: str, chunk_size: int = 512) -> int:
    """Count tokens exactly the way BGE-M3 does, without running the embedding.

    Reproduces the pre-processing that HuggingFaceEmbedding/BGE-M3 applies to
    the text before encoding it:
      1. The text is stripped (BGE-M3's sentence preprocessing strips leading
         and trailing whitespace).
      2. The tokenizer is called with add_special_tokens=True, which adds the
         [CLS] and [SEP] special tokens around the text. This is what the
         embedding model's forward pass actually sees, so its token count is
         the "raw" tokenization count + 2.
    """
    model_name = "BAAI/bge-m3"
    tokenizer = AutoTokenizer.from_pretrained(model_name, cache_folder='./embeddings')

    # Reproduce the model's preprocessing: strip the text
    processed_text = text.strip()

    # Tokenize with the special tokens added, exactly like the embedding model does
    tokenized_text = tokenizer(processed_text, add_special_tokens=True,
                               max_length=chunk_size, truncation=True)
    token_count = len(tokenized_text['input_ids'])
    return token_count


def run_agent(text: str):
    chunk_size = 512
    embed_model = create_embedding_model(_chunk_size=chunk_size)

    # Create the model's tokenizer once, so the token counter uses the SAME
    # tokenizer as the embedding model instead of the default (gpt-2/tiktoken)
    model_name = "BAAI/bge-m3"
    tokenizer = AutoTokenizer.from_pretrained(model_name, cache_folder='./embeddings')

    token_counter = TokenCountingHandler(tokenizer=tokenizer)
    callback_manager = CallbackManager([token_counter])

    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    embedding_vector = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    token_count = count_tokens_standalone(text, chunk_size=chunk_size)
    return embedding_tokens, token_count

if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
    print(f"The tokenizer broke the text into {token_count} tokens")
