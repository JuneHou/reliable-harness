"""
Fix for: "external_id is stored inside the metadata JSON column, but should
be a separate column of the Postgres table created by PGVectorStore".

Root cause
----------
`PGVectorStore` always creates a FIXED table schema:

    id | text | node_id | embedding | metadata_

and serializes *all* node metadata (including `external_id`) into the
`metadata_` JSON(B) column. Putting `external_id` into `Document.metadata`
therefore only ends up inside the JSON blob - it never becomes a real,
queryable column of the table.

Fix
---
Subclass `PGVectorStore` (`CustomPGVectorStore`) so custom fields become
first-class table columns:

1. `_ensure_custom_columns()`: after the parent creates its standard table,
   run `ALTER TABLE ... ADD COLUMN IF NOT EXISTS external_id TEXT` (plus an
   index on the new column).
2. `add()` / `aadd()`: let the parent insert the standard columns, then write
   the custom column value (taken from the node metadata) with a plain SQL
   UPDATE. By default the custom field is stripped from the JSON metadata so
   it is stored *only* as a column.
3. `query()` / `aquery()`: read the custom column back and re-attach it to
   the retrieved nodes' metadata, so retrieval keeps working end to end.

The custom store is handed to `VectorStoreIndex.from_documents` through a
`StorageContext` (the documented way that works across llama-index versions).
"""

import psycopg2

load_dotenv()

PASSWORD = os.getenv("PG_PASSWORD", "password")

# Custom columns that must exist as REAL table columns instead of being
# buried inside the `metadata_` JSON(B) column:  name -> SQL type.
CUSTOM_COLUMNS: Dict[str, str] = {
    "external_id": "TEXT",
}


class CustomPGVectorStore(PGVectorStore):
    """PGVectorStore whose table has extra, first-class columns (e.g.
    `external_id`) in addition to the standard id/text/node_id/embedding/
    metadata_ columns."""

    custom_columns: Dict[str, str] = CUSTOM_COLUMNS

    # If True the custom fields are ALSO kept inside the `metadata_` JSON
    # column. The requirement was "a separate column *instead of* the
    # metadata JSON", hence the default False.
    store_custom_columns_in_metadata: bool = False

    # ------------------------------------------------------------------ #
    # construction
    # ------------------------------------------------------------------ #
    def __init__(self, *args: Any, **kwargs: Any) -> None:
        # The parent constructor creates the standard table
        # (id, text, node_id, embedding, metadata_) ...
        super().__init__(*args, **kwargs)
        # ... and we extend it with our custom columns.
        self._ensure_custom_columns()

    # ------------------------------------------------------------------ #
    # helpers
    # ------------------------------------------------------------------ #
    @property
    def _sync_engine(self) -> Any:
        """The synchronous SQLAlchemy engine of the vector store."""
        engine = getattr(self, "_engine", None) or getattr(self, "engine", None)
        if engine is None:
            raise RuntimeError("No SQLAlchemy engine available on the store.")
        return engine

    @property
    def _raw_table_name(self) -> str:
        return getattr(self, "table_name", None) or getattr(
            self, "_table_name", "vectorstore"
        )

    @property
    def _qualified_table(self) -> str:
        schema = getattr(self, "schema_name", None) or "public"
        return f'"{schema}"."{self._raw_table_name}"'

    def _ensure_custom_columns(self) -> None:
        """Add the custom columns (and indexes) to the table created by
        PGVectorStore. Idempotent - safe to run on every start-up."""
        if not self.custom_columns:
            return
        with self._sync_engine.begin() as conn:
            for column, sql_type in self.custom_columns.items():
                conn.execute(
                    text(
                        f"ALTER TABLE {self._qualified_table} "
                        f"ADD COLUMN IF NOT EXISTS {column} {sql_type}"
                    )
                )
                # A real column can - and should - be indexed.
                conn.execute(
                    text(
                        f"CREATE INDEX IF NOT EXISTS "
                        f"idx_{self._raw_table_name}_{column} "
                        f"ON {self._qualified_table} ({column})"
                    )
                )

    def _custom_values(self, node: BaseNode) -> Dict[str, Any]:
        """Extract the custom-column values from a node's metadata."""
        values: Dict[str, Any] = {}
        metadata = node.metadata or {}
        for column in self.custom_columns:
            if column in metadata:
                values[column] = metadata[column]
        return values

    def _nodes_without_custom_columns(self, nodes: List[BaseNode]) -> List[BaseNode]:
        """Return node copies whose metadata no longer contains the custom
        fields, so those values are stored ONLY as table columns and are not
        duplicated inside the `metadata_` JSON(B) column."""
        if self.store_custom_columns_in_metadata:
            return nodes

        result: List[BaseNode] = []
        for node in nodes:
            if not self._custom_values(node):
                result.append(node)
                continue
            if hasattr(node, "model_copy"):  # pydantic v2
                node_copy = node.model_copy()
            else:  # pydantic v1
                node_copy = node.copy()
            node_copy.metadata = {
                key: value
                for key, value in (node.metadata or {}).items()
                if key not in self.custom_columns
            }
            result.append(node_copy)
        return result

    def _write_custom_columns(self, nodes: List[BaseNode]) -> None:
        """Persist the custom-column values for the given (original) nodes."""
        assignments = []
        for node in nodes:
            values = self._custom_values(node)
            if values:
                assignments.append({"id": node.node_id, **values})
        if not assignments:
            return

        with self._sync_engine.begin() as conn:
            for params in assignments:
                set_clause = ", ".join(
                    f"{column} = :{column}"
                    for column in self.custom_columns
                    if column in params
                )
                stmt = text(
                    f"UPDATE {self._qualified_table} "
                    f"SET {set_clause} WHERE id = :id"
                )
                conn.execute(stmt, params)

    def _read_custom_columns(self, ids: List[str]) -> Dict[str, Dict[str, Any]]:
        """Read the custom-column values for the given row (node) ids."""
        columns = list(self.custom_columns)
        if not ids or not columns:
            return {}
        stmt = text(
            f"SELECT id, {', '.join(columns)} FROM {self._qualified_table} "
            f"WHERE id = ANY(:ids)"
        )
        mapping: Dict[str, Dict[str, Any]] = {}
        with self._sync_engine.begin() as conn:
            for row in conn.execute(stmt, {"ids": list(ids)}).mappings().all():
                mapping[row["id"]] = {column: row[column] for column in columns}
        return mapping

    def _hydrate_custom_columns(self, result: VectorStoreQueryResult) -> None:
        """Re-attach the custom column values onto the retrieved nodes."""
        if not result.nodes:
            return
        ids = result.ids or [node.node_id for node in result.nodes]
        values_by_id = self._read_custom_columns(ids)
        for node in result.nodes:
            if node.metadata is None:
                node.metadata = {}
            for column, value in values_by_id.get(node.node_id, {}).items():
                if value is not None:
                    node.metadata[column] = value

    # ------------------------------------------------------------------ #
    # write path (insert)
    # ------------------------------------------------------------------ #
    def add(self, nodes: List[BaseNode], use_async: bool = False) -> List[str]:
        # 1) Standard insert of text / embedding / node_id / metadata_ ...
        stripped_nodes = self._nodes_without_custom_columns(nodes)
        ids = super().add(stripped_nodes, use_async=use_async)
        # 2) ... plus the custom columns, filled from the node metadata.
        self._write_custom_columns(nodes)
        return ids

    async def aadd(self, nodes: List[BaseNode]) -> List[str]:
        stripped_nodes = self._nodes_without_custom_columns(nodes)
        ids = await super().aadd(stripped_nodes)
        self._write_custom_columns(nodes)
        return ids

    # ------------------------------------------------------------------ #
    # read path (search)
    # ------------------------------------------------------------------ #
    def query(
        self, query: VectorStoreQuery, **kwargs: Any
    ) -> VectorStoreQueryResult:
        result = super().query(query, **kwargs)
        self._hydrate_custom_columns(result)
        return result

    async def aquery(
        self, query: VectorStoreQuery, **kwargs: Any
    ) -> VectorStoreQueryResult:
        result = await super().aquery(query, **kwargs)
        self._hydrate_custom_columns(result)
        return result

    # ------------------------------------------------------------------ #
    # convenience API unlocked by the real column
    # ------------------------------------------------------------------ #
    def get_node_ids_by_external_id(self, external_id: str) -> List[str]:
        """Look up node ids directly on the external_id column."""
        stmt = text(
            f"SELECT id FROM {self._qualified_table} "
            f"WHERE external_id = :external_id"
        )
        with self._sync_engine.begin() as conn:
            return [
                row[0]
                for row in conn.execute(
                    stmt, {"external_id": external_id}
                ).fetchall()
            ]

    def delete_by_external_id(self, external_ids: List[str]) -> None:
        """Delete rows directly on the external_id column. Also removes
        legacy rows that only carry the value inside the metadata_ JSON."""
        if not external_ids:
            return
        stmt = text(
            f"DELETE FROM {self._qualified_table} "
            f"WHERE external_id = ANY(:external_ids) "
            f"OR metadata_ ->> 'external_id' = ANY(:external_ids)"
        )
        with self._sync_engine.begin() as conn:
            conn.execute(stmt, {"external_ids": list(external_ids)})

def print_table_schema(database_name, table_name="vectorstore"):
    """Print the columns of the vector store table to verify the fix."""
    conn = get_connection(database_name)
    cursor = conn.cursor()
    cursor.execute(
        """
        SELECT column_name, data_type
        FROM information_schema.columns
        WHERE table_name = %s
        ORDER BY ordinal_position;
        """,
        (table_name,),
    )
    rows = cursor.fetchall()
    cursor.close()
    conn.close()

    print(f"Columns in the '{table_name}' table:")
    for column_name, data_type in rows:
        print(f"  - {column_name} ({data_type})")
