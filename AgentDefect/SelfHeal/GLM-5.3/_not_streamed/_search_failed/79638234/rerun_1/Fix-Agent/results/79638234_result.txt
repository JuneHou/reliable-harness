import random

from langchain_ollama import ChatOllama


def run_agent(input_text):
    """Invoke the local Ollama model via LangChain and return its response text.

    Why the original code always returned the identical response:
    ----------------------------------------------------------------
    Ollama pins the RNG used for token sampling when the generation seed is
    fixed (the LangChain docs state: "Setting this to a specific number will
    make the model generate the same text for the same prompt"), and recent
    Ollama versions (notably the new engine) treat the default/leftover seed
    as a request for deterministic generation.  With a pinned seed, the same
    prompt + the same options always sample exactly the same tokens, so
    `temperature`, `top_p` and `repeat_penalty` only reshape the probability
    distribution but never change *which* tokens are actually picked between
    runs -- which is why every call returned a word-for-word identical joke.

    The fix: draw a *different* random seed for every request.  Combined with
    temperature > 0, this restores varied, stochastic outputs.
    """
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        timeout=120,
        # Fresh random seed per call -> different sampling decisions each
        # time.  The range 0..2**31-1 stays inside Ollama's expected integer
        # range for the seed option.
        seed=random.randint(0, 2**31 - 1),
    )
    response = llm.invoke(input_text)
    return response.content


if __name__ == '__main__':
    prompt = "Please tell me a lighthearted joke"
    # Run the *same* prompt several times to verify the outputs now vary.
    for i in range(3):
        print(f"--- Run {i + 1} ---")
        print(run_agent(prompt))
        print()
