import os
from itertools import combinations

from dotenv import load_dotenv

from langchain.document_loaders import CSVLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.runnable import RunnablePassthrough
from langchain.schema.output_parser import StrOutputParser

load_dotenv()

# Similarity thresholds for classifying control pairs (cosine similarity, 0..1).
# Pairs with score >= DUPLICATE_THRESHOLD are treated as duplicates,
# pairs with score >= SIMILAR_THRESHOLD (but below duplicate) as similar.
SIMILAR_THRESHOLD = 0.80
DUPLICATE_THRESHOLD = 0.95


def cosine_similarity(vec_a, vec_b):
    """Cosine similarity between two embedding vectors (pure Python, no extra deps)."""
    dot = sum(a * b for a, b in zip(vec_a, vec_b))
    norm_a = sum(a * a for a in vec_a) ** 0.5
    norm_b = sum(b * b for b in vec_b) ** 0.5
    if norm_a == 0 or norm_b == 0:
        return 0.0
    return dot / (norm_a * norm_b)


def find_similar_and_duplicate_pairs(similar_threshold=SIMILAR_THRESHOLD,
                                     duplicate_threshold=DUPLICATE_THRESHOLD):
    """Detect similar and duplicate control pairs by comparing every control
    with every other control (pairwise), which is what the user actually wants.

    NOTE: CSVLoader already produces ONE document per CSV row, so each control
    stays intact. The CharacterTextSplitter from the original code was removed
    because it can cut a control row in half and corrupt the pair detection.
    """
    loader = CSVLoader(file_path="control.csv")
    data = loader.load()

    if len(data) < 2:
        return [], []

    embeddings = OpenAIEmbeddings()
    vectors = embeddings.embed_documents([doc.page_content for doc in data])

    similar_pairs = []
    duplicate_pairs = []

    # Compare each control with every other control (not with the query!).
    for i, j in combinations(range(len(vectors)), 2):
        score = cosine_similarity(vectors[i], vectors[j])
        pair = (
            data[i].page_content.replace("\n", " ").strip(),
            data[j].page_content.replace("\n", " ").strip(),
            round(score, 4),
        )
        if score >= duplicate_threshold:
            duplicate_pairs.append(pair)
        elif score >= similar_threshold:
            similar_pairs.append(pair)

    return similar_pairs, duplicate_pairs


def format_pairs(pairs):
    if not pairs:
        return "- None found"
    return "\n".join(
        f"- Pair {idx}: Control A: {a} | Control B: {b} | Similarity score: {score}"
        for idx, (a, b, score) in enumerate(pairs, start=1)
    )


def run_agent(query):
    # 1. Compute the actual similar/duplicate pairs (deterministic, threshold-based).
    similar_pairs, duplicate_pairs = find_similar_and_duplicate_pairs()

    # 2. Build the context from the detected pairs so the LLM answers about pairs,
    #    not about arbitrary chunks retrieved by similarity to the question text.
    context = (
        f"SIMILAR CONTROL PAIRS (similarity score >= {SIMILAR_THRESHOLD}):\n"
        f"{format_pairs(similar_pairs)}\n\n"
        f"DUPLICATE CONTROL PAIRS (similarity score >= {DUPLICATE_THRESHOLD}):\n"
        f"{format_pairs(duplicate_pairs)}"
    )

    # 3. FIX (Change Prompt): the prompt now explicitly requires a LIST of
    #    similar and duplicate control pairs with their similarity scores.
    #    The old instructions ("use three sentences maximum", "say you don't
    #    know") were the reason the expected list output never appeared.
    template = """You are an assistant that analyzes a control register and identifies similar and duplicate controls.

You are given pairs of controls that were detected as similar or duplicates of each other based on their embedding cosine similarity score.

Answer the question using ONLY the pairs provided in the context below.

Return the answer as a list. For every pair, output one bullet line in this exact format:
- [DUPLICATE] Control: "<control A>" | Control: "<control B>" | Similarity score: <score>
- [SIMILAR] Control: "<control A>" | Control: "<control B>" | Similarity score: <score>

List all DUPLICATE pairs first, then all SIMILAR pairs.
If no pairs were found, answer exactly: "No similar or duplicate controls were found."

Question: {question}

Context:
{context}

Answer:
"""

    prompt = ChatPromptTemplate.from_template(template)
    llm = ChatOpenAI(temperature=0, model="gpt-3.5-turbo", verbose=True)

    rag_chain = (
        # The context is the computed pair list (constant for this run),
        # not a retriever that searches for chunks similar to the question.
        {"context": lambda _: context, "question": RunnablePassthrough()}
        | prompt
        | llm
        | StrOutputParser()
    )

    output = rag_chain.invoke(query)
    return output


if __name__ == "__main__":
    # FIX: the query now matches the intended task instead of an unrelated
    # movie-recommendation style question.
    query = "List all similar and duplicate control pairs with their similarity scores."
    result = run_agent(query)
    print(result)
