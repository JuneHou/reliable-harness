import os
from dotenv import load_dotenv

from langchain_community.document_loaders import CSVLoader
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.runnable import RunnablePassthrough
from langchain.schema.output_parser import StrOutputParser

load_dotenv()

# NOTE: Chroma's similarity_search_with_score() returns a DISTANCE (L2 by
# default), where LOWER means MORE SIMILAR -- it is not a similarity score
# where higher is better. A pair is considered similar/duplicate when its
# distance is <= this threshold. Tune this value for your embedding model
# (OpenAIEmbeddings L2 distances for near-duplicates are typically small).
SIMILARITY_DISTANCE_THRESHOLD = 0.35


def load_rows(file_path="control.csv"):
    """Load the CSV so that each row becomes exactly one Document.

    No CharacterTextSplitter is applied: splitting CSV rows (chunk_size=500,
    chunk_overlap=50) breaks/merges individual rows, destroying the
    row-level granularity required to compare controls pairwise.
    """
    loader = CSVLoader(file_path=file_path)
    rows = loader.load()
    # Give every row a stable identifier so pairs can be deduplicated.
    for i, doc in enumerate(rows):
        doc.metadata["row_id"] = i
    return rows


def find_similar_pairs(rows, threshold=SIMILARITY_DISTANCE_THRESHOLD, top_k=None):
    """Detect similar/duplicate pairs by comparing EVERY row against the
    whole collection with a distance threshold.

    A single-query top-k retriever (the original approach) can only fetch a
    few documents relevant to one query string; it can never enumerate all
    similar/duplicate pairs inside the dataset. This is why the original
    pipeline failed to produce the expected list of pairs.
    """
    if not rows:
        return []

    vectorstore = Chroma.from_documents(documents=rows, embedding=OpenAIEmbeddings())

    # Compare each row against the full collection so no pair is missed.
    k = len(rows) if top_k is None else min(top_k, len(rows))

    pairs = {}
    for row in rows:
        hits = vectorstore.similarity_search_with_score(row.page_content, k=k)
        for doc, distance in hits:
            # Chroma returns a distance: smaller = more similar.
            if distance > threshold:
                continue  # too far apart -> not a similar/duplicate pair
            i, j = row.metadata["row_id"], doc.metadata["row_id"]
            if i == j:
                continue  # skip self-match
            key = tuple(sorted((i, j)))
            if key not in pairs or distance < pairs[key]:
                pairs[key] = distance

    # Convert back to (row_a, row_b, distance) and rank most-similar first.
    result = [(rows[i], rows[j], d) for (i, j), d in pairs.items()]
    result.sort(key=lambda item: item[2])
    return result


def format_pairs(pairs):
    """Render the detected pairs (with their distance score) as text."""
    lines = []
    for idx, (doc_a, doc_b, distance) in enumerate(pairs, start=1):
        lines.append(
            f"- Pair {idx} (distance={distance:.4f}, lower = more similar):\n"
            f"    Control A: {doc_a.page_content}\n"
            f"    Control B: {doc_b.page_content}"
        )
    return "\n".join(lines)


def run_agent(query):
    # Step 1: deterministic pair detection on the CSV data itself.
    rows = load_rows()
    pairs = find_similar_pairs(rows)

    if not pairs:
        return "No similar or duplicate control pairs were found below the similarity threshold."

    pairs_text = format_pairs(pairs)

    # Step 2: the LLM only formats/presents the already-computed pairs.
    # The original prompt was a generic QA template ("use three sentences
    # maximum, keep the answer concise") that never asked for a list of
    # pairs -- it forced the model to summarize, so the expected list of
    # similar/duplicate pairs could never appear in the output.
    template = """You are an assistant that analyses control data to find similar and duplicate control pairs.

Below is the verified list of control pairs whose embedding distance is below the similarity threshold (lower distance = more similar). This list was computed directly from the dataset.

Your task is to output the list of similar and duplicate control pairs, identifying each control by its name and reporting its distance score.

Rules:
- List EVERY pair from the provided list; do not skip, merge, or summarize pairs.
- Do not limit the length of your answer to three sentences; a complete list is required.
- Do not invent pairs that are not in the provided list.
- If the list is empty, say that no similar or duplicate pairs were found.

Question: {question}

Detected pairs (ordered by most similar first):
{pairs}

Answer:

"""

    prompt = ChatPromptTemplate.from_template(template)
    llm = ChatOpenAI(temperature=0, model="gpt-3.5-turbo")

    rag_chain = (
        {
            "question": RunnablePassthrough(),
            "pairs": lambda _ignored: pairs_text,
        }
        | prompt
        | llm
        | StrOutputParser()
    )
    return rag_chain.invoke(query)


if __name__ == "__main__":
    # The query must match the intended task: listing similar/duplicate
    # control pairs, not a generic question about the CSV contents.
    query = "List the similar and duplicate control pairs in the dataset based on the similarity score threshold, including their names and scores."
    result = run_agent(query)
    print(result)
