import os
from dotenv import load_dotenv

from langchain.document_loaders import CSVLoader
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.runnable import RunnablePassthrough
from langchain.schema.output_parser import StrOutputParser

load_dotenv()

# Similarity score threshold: pairs scoring above this are considered
# "similar"; pairs scoring above the (higher) duplicate threshold are
# considered near-duplicates. Tune these to your dataset.
SIMILARITY_THRESHOLD = 0.80
DUPLICATE_THRESHOLD = 0.92

def run_agent(query):
    # NOTE: no text splitting. CSVLoader already produces one Document
    # per CSV row, which must stay intact so control pairs can be formed.
    loader = CSVLoader(file_path="control.csv")
    data = loader.load()

    vectorstore = Chroma.from_documents(documents=data, embedding=OpenAIEmbeddings())

    # Explicitly compute similarity scores with a large enough k so that
    # all candidate rows are considered, then filter by threshold.
    k = max(len(data), 1)
    docs_and_scores = vectorstore.similarity_search_with_score(query, k=k)

    # Chroma returns distance (lower = more similar). Convert to a
    # cosine-similarity-like score in [0, 1] (higher = more similar)
    # so thresholds behave as expected.
    scored_rows = []
    for doc, distance in docs_and_scores:
        similarity = 1.0 - distance
        scored_rows.append((doc.page_content, similarity))

    # Build the similarity / duplicate pair listing.
    similar_pairs = [r for r in scored_rows if r[1] >= SIMILARITY_THRESHOLD]
    duplicate_pairs = [r for r in scored_rows if r[1] >= DUPLICATE_THRESHOLD]

    context = "\n\n".join(
        f"[similarity score: {score:.3f}] {text}" for text, score in scored_rows
    )

    # Prompt changed: instead of forcing "three sentences maximum" and
    # "if you don't know, just say you don't know" (which suppressed the
    # expected output), it now explicitly requests the structured list of
    # similar and duplicate control pairs with their scores.
    template = """You are an assistant that analyzes a list of controls with their similarity scores.

    Based on the retrieved context below, output a list of similar control pairs and duplicate control pairs.

    Rules:
    - Controls with a similarity score >= {similarity_threshold} are considered SIMILAR.
    - Controls with a similarity score >= {duplicate_threshold} are considered DUPLICATES.
    - For each control, state its Name, Description, and its similarity score.
    - Present the answer as two clearly labeled lists: "Similar Pairs" and "Duplicate Pairs".
    - If no controls meet a threshold, explicitly state that the list is empty.

    Question: {question}

    Context:
    {context}

    Answer:
    """

    prompt = ChatPromptTemplate.from_template(template)
    llm = ChatOpenAI(temperature=0, model="gpt-3.5-turbo", verbose=True)

    rag_chain = (
        {
            "context": lambda _: context,
            "question": RunnablePassthrough(),
            "similarity_threshold": lambda _: SIMILARITY_THRESHOLD,
            "duplicate_threshold": lambda _: DUPLICATE_THRESHOLD,
        }
        | prompt
        | llm
        | StrOutputParser()
    )

    output = rag_chain.invoke(query)

    # Deterministic pair summary computed directly from the scores, so the
    # expected output is produced even if the model paraphrases.
    lines = []
    lines.append("=== Similar Control Pairs (score >= %.2f) ===" % SIMILARITY_THRESHOLD)
    if similar_pairs:
        for text, score in similar_pairs:
            lines.append(f"- {text} (score: {score:.3f})")
    else:
        lines.append("None found.")

    lines.append("")
    lines.append("=== Duplicate Control Pairs (score >= %.2f) ===" % DUPLICATE_THRESHOLD)
    if duplicate_pairs:
        for text, score in duplicate_pairs:
            lines.append(f"- {text} (score: {score:.3f})")
    else:
        lines.append("None found.")

    deterministic_summary = "\n".join(lines)

    return output + "\n\n" + deterministic_summary

if __name__ == "__main__":
    query = "List the similar and duplicate control pairs from the control data with their similarity scores."
    result = run_agent(query)
    print(result)
