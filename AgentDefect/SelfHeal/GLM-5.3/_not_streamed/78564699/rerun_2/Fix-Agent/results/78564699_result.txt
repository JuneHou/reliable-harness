import json
import os
from llama_index.llms.llama_cpp import LlamaCPP


def run_agent(query):
    # Use the 7B model instead of the 13B model.
    # The 13B model (~7.3 GB at Q4_0) is too heavy for CPU-only inference
    # (n_gpu_layers=0 due to limited VRAM): generation is so slow that
    # llm.complete() appears to hang indefinitely and never returns.
    # The 7B model (~3.8 GB at Q4_0) is roughly half the size and produces
    # a response on CPU within a reasonable time.
    MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-7B-Chat-GGUF/resolve/main/llama-2-7b-chat.Q4_0.gguf"
    MODEL_PATH = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if "llama-2-7b-chat" in paths:
                MODEL_URL = None
                MODEL_PATH = paths["llama-2-7b-chat"]

    llm = LlamaCPP(
        model_url=MODEL_URL,
        model_path=MODEL_PATH,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={"n_gpu_layers": 0},
        verbose=True,
    )

    response = llm.complete(query)

    return response

if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
