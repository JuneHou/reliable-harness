import json
import os
from llama_index.llms.llama_cpp import LlamaCPP


def run_agent(query):
    # FIX: The Llama-2 13B chat model (llama-2-13b-chat.Q4_0.gguf, ~7.4 GB) is
    # far too heavy for CPU-only inference (n_gpu_layers=0). It loads and
    # initializes successfully, but generation becomes extremely slow / starts
    # swapping once memory pressure builds, so llm.complete() appears to hang
    # indefinitely. Switch to the much smaller and faster 7B chat model, which
    # runs comfortably on CPU and completes inference in a reasonable time.
    MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-7B-chat-GGUF/resolve/main/llama-2-7b-chat.Q4_0.gguf"
    MODEL_PATH = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            # Prefer a locally downloaded copy of the (smaller) 7B model.
            if "llama-2-7b-chat" in paths:
                MODEL_URL = None
                MODEL_PATH = paths["llama-2-7b-chat"]

    llm = LlamaCPP(
        model_url=MODEL_URL,
        model_path=MODEL_PATH,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        # generate_kwargs are forwarded to llama-cpp-python's completion call.
        # Explicit stop sequences ensure generation terminates cleanly once the
        # answer is produced instead of rambling until the token limit.
        generate_kwargs={"stop": ["</s>", "[INST]"]},
        # model_kwargs are forwarded to the llama_cpp.Llama(...) constructor.
        # Keep n_gpu_layers at 0 for CPU-only execution, and make sure the
        # underlying model is actually created with a context (n_ctx) large
        # enough for the prompt plus generated tokens -- LlamaIndex's
        # context_window alone is not passed to the model, and a context
        # overflow is another common cause of stalled generation.
        model_kwargs={"n_gpu_layers": 0, "n_ctx": 3900},
        verbose=True,
    )

    response = llm.complete(query)

    return response


if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
