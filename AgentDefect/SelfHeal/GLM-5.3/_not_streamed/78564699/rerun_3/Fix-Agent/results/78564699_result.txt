import json
import os
from llama_index.llms.llama_cpp import LlamaCPP


def run_agent(query):
    # Fix: the previous code used Llama-2 13B (llama-2-13b-chat.Q4_0.gguf).
    # Running a 13B model entirely on CPU (n_gpu_layers=0) is far too slow --
    # each token can take seconds, so llm.complete() appears to hang
    # indefinitely even though model initialization succeeds.
    # Use the smaller Llama-2 7B chat model, which is tractable for
    # CPU-only inference and returns a response in reasonable time.
    MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-7B-Chat-GGUF/resolve/main/llama-2-7b-chat.Q4_0.gguf"
    MODEL_PATH = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if "llama-2-7b-chat" in paths:
                MODEL_URL = None
                MODEL_PATH = paths["llama-2-7b-chat"]

    llm = LlamaCPP(
        model_url=MODEL_URL,
        model_path=MODEL_PATH,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={"n_gpu_layers": 0},
        verbose=True,
    )

    response = llm.complete(query)

    return response

if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
