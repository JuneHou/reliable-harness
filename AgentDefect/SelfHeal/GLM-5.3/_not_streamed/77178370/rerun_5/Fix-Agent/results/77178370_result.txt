import os
from typing import Any, Dict, List

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import FAISS
from langchain.chains import ConversationalRetrievalChain
from langchain.memory import ConversationBufferMemory
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate
from langchain.retrievers import ContextualCompressionRetriever
from langchain.retrievers.document_compressors import LLMChainFilter
from langchain.schema import Document

# Load environment variables from .env file
_ = load_dotenv(find_dotenv())


def run_agent(user_input: str) -> Dict[str, Any]:
    """Run the conversational retrieval chatbot.

    Returns a dict with:
      - "answer": the chatbot's answer to the user's question
      - "source_documents": the knowledge-base documents that were actually
        used to answer the question. This list is EMPTY when the question
        does not require the external knowledge base (e.g. "hello" or
        "what's 2+2"), i.e. when the answer does not come from the docs.
    """
    openai_api_key = os.getenv('OPENAI_API_KEY')
    if not openai_api_key:
        raise ValueError(
            "OPENAI_API_KEY not found. Please set it in your .env file."
        )

    os.environ['OPENAI_API_KEY'] = openai_api_key

    custom_template = """
This is conversation with a human. Answer the questions you get based on the knowledge you have.
If you don't know the answer, just say that you don't, don't try to make up an answer.
Chat History:
{chat_history}
Follow Up Input: {question}
"""
    CUSTOM_QUESTION_PROMPT = PromptTemplate.from_template(custom_template)

    llm = ChatOpenAI(
        model_name="gpt-3.5-turbo",  # Name of the language model
        temperature=0  # Parameter that controls the randomness of the generated responses
    )
    embeddings = OpenAIEmbeddings()

    docs = [
        "Buildings are made out of brick",
        "Buildings are made out of wood",
        "Buildings are made out of stone",
        "Buildings are made out of atoms",
        "Buildings are made out of building materials",
        "Cars are made out of metal",
        "Cars are made out of plastic",
    ]
    vectorstore = FAISS.from_texts(docs, embeddings)

    # FIX 1 (why the output did not match expectations):
    # A plain vector-store retriever ALWAYS returns its top-k nearest
    # documents, no matter how unrelated the question is. That is why general
    # questions like "hello" or "what's 2+2" still returned "source"
    # documents. We therefore first fetch a few candidate documents...
    base_retriever = vectorstore.as_retriever(search_kwargs={"k": 4})

    # ...and then pass them through an LLM-based relevance filter, which only
    # keeps a document when it is genuinely relevant to the question.
    # (Similarity-score thresholds are not reliable with FAISS + OpenAI
    # embeddings, so the LLM decides relevance instead.)
    relevance_filter = LLMChainFilter.from_llm(llm)

    retriever = ContextualCompressionRetriever(
        base_compressor=relevance_filter,
        base_retriever=base_retriever,
    )

    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)

    qa = ConversationalRetrievalChain.from_llm(
        llm,
        retriever,
        condense_question_prompt=CUSTOM_QUESTION_PROMPT,
        memory=memory
    )

    # FIX 2: answer the question the user actually asked. The previous code
    # ignored `user_input` and hard-coded "what are cars made of?", so the
    # printed question ("What are buildings made of?") did not match what
    # was answered/retrieved.
    result = qa({"question": user_input})
    answer = result["answer"]

    # FIX 3: keep the chain's answer instead of overwriting it, and obtain
    # the source documents through the relevance-filtered retriever.
    # get_relevant_documents() now returns the knowledge-base documents only
    # when the answer really comes from the custom knowledge base, and
    # returns an empty list for general questions that don't need the docs.
    source_documents: List[Document] = retriever.get_relevant_documents(user_input)

    return {
        "answer": answer,
        "source_documents": source_documents,
    }


if __name__ == "__main__":
    # 1) A question that must be answered from the custom knowledge base:
    #    -> answer + the matching source documents.
    user_input = "What are buildings made of?"
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result['answer']}")
    print(
        "Source Documents:",
        [doc.page_content for doc in result['source_documents']],
    )

    # 2) A general question that does not need the knowledge base:
    #    -> answer, but NO source documents (empty list).
    general_input = "hello"
    general_result = run_agent(general_input)
    print("\n=== General Query Result ===")
    print(f"Question: {general_input}")
    print(f"Answer: {general_result['answer']}")
    print(
        "Source Documents:",
        [doc.page_content for doc in general_result['source_documents']],
    )
