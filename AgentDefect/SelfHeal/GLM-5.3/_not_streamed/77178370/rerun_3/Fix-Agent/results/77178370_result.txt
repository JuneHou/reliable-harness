import os
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import FAISS
from langchain.chains import ConversationalRetrievalChain 
from langchain.memory import ConversationBufferMemory
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate

# Load environment variables from .env file
_ = load_dotenv(find_dotenv())


def run_agent(user_input):
    openai_api_key = os.getenv('OPENAI_API_KEY')

    os.environ['OPENAI_API_KEY'] = openai_api_key

    custom_template = """
This is conversation with a human. Answer the questions you get based on the knowledge you have.
If you don't know the answer, just say that you don't, don't try to make up an answer.
Chat History:
{chat_history}
Follow Up Input: {question}
"""
    CUSTOM_QUESTION_PROMPT = PromptTemplate.from_template(custom_template)
    llm = ChatOpenAI(
        model_name="gpt-3.5-turbo",  # Name of the language model
        temperature=0  # Parameter that controls the randomness of the generated responses
    )
    embeddings = OpenAIEmbeddings()
    docs = [
        "Buildings are made out of brick",
        "Buildings are made out of wood",
        "Buildings are made out of stone",
        "Buildings are made out of atoms",
        "Buildings are made out of building materials",
        "Cars are made out of metal",
        "Cars are made out of plastic",
    ]
    vectorstore = FAISS.from_texts(docs, embeddings)

    # FIX 1: use a relevance-score threshold retriever instead of the plain
    # "similarity" retriever. A plain retriever ALWAYS returns the top-k
    # closest documents, even for chit-chat like "hello" or "what's 2+2".
    # With a score threshold, only documents that are genuinely relevant to
    # the question are returned; general queries fall below the threshold and
    # therefore retrieve no documents (an empty result).
    retriever = vectorstore.as_retriever(
        search_type="similarity_score_threshold",
        search_kwargs={"k": 4, "score_threshold": 0.5},
    )

    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    # FIX 2: ask the chain to return the source documents it actually used,
    # so we don't have to re-run retrieval ourselves (and possibly get
    # documents the bot never looked at).
    qa = ConversationalRetrievalChain.from_llm(
        llm,
        retriever,
        condense_question_prompt=CUSTOM_QUESTION_PROMPT,
        memory=memory,
        return_source_documents=True,
    )

    # FIX 3: use the actual user input (the old code ignored it and always
    # asked the hardcoded question "what are cars made of?").
    # The chain internally calls the retriever's get_relevant_documents on
    # the condensed standalone question.
    result = qa({"question": user_input})

    # FIX 4: do NOT overwrite the chain result with a separate
    # retriever.get_relevant_documents() call. The chain's "source_documents"
    # are exactly the documents the bot accessed for its answer. For inputs
    # that have nothing to do with the knowledge base ("hello", "how are
    # you", "what's 2+2", ...) no document passes the relevance threshold,
    # so this list is empty and we return an empty result.
    source_documents = result.get("source_documents") or []

    return source_documents


if __name__ == "__main__":
    user_input = "What are buildings made of?"
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    if result:
        print("Answer came from the knowledge base. Source documents:")
        for i, doc in enumerate(result, start=1):
            print(f"  [{i}] {doc.page_content}")
    else:
        print("No source documents: the answer did not come from the knowledge base.")
