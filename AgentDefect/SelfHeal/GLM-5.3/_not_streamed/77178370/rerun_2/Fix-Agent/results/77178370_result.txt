class KnowledgeBaseRetriever(BaseRetriever):
    """
    Custom retriever that returns the source documents of the custom knowledge
    base ONLY when the question actually has to be answered with them.

    - Questions about the knowledge base (e.g. "what are cars made of?")
      return the matching source documents.
    - General questions that do not require the external documents
      (e.g. "hello" or "what's 2+2?") return an empty list.
    """

    vectorstore: FAISS               # the custom knowledge base
    llm: ChatOpenAI                  # decides whether the knowledge base is needed
    knowledge_base_texts: List[str]  # raw texts stored in the knowledge base
    k: int = 4                       # number of documents to return

    def _ask_llm(self, prompt: str) -> str:
        """Call the LLM and return its answer as plain text (version safe)."""
        if hasattr(self.llm, "predict"):
            return self.llm.predict(prompt)
        return self.llm.invoke(prompt).content

    def _needs_knowledge_base(self, question: str) -> bool:
        """Use the LLM to decide if the question requires the knowledge base."""
        classification_prompt = PromptTemplate.from_template(
            """You decide whether a user question has to be answered with a
specific custom knowledge base or not.

Custom knowledge base:
{knowledge_base}

Only questions whose answer can be found in the knowledge base above need it.
General questions such as greetings (e.g. "hello"), small talk, or general
knowledge questions (e.g. "what's 2+2?") do NOT need it.

User question: {question}

Does the question need the custom knowledge base? Reply with exactly one word: yes or no.
Answer:"""
        )
        answer = self._ask_llm(
            classification_prompt.format(
                knowledge_base="\n".join(
                    f"- {text}" for text in self.knowledge_base_texts
                ),
                question=question,
            )
        )
        return answer.strip().lower().startswith("yes")

    def _get_relevant_documents(self, query: str, **kwargs) -> List[Document]:
        # General questions must NOT return any knowledge-base documents.
        if not self._needs_knowledge_base(query):
            return []

        # The answer comes from the custom knowledge base,
        # so its source documents are returned.
        return self.vectorstore.similarity_search(query, k=self.k)


def run_agent(user_input):
    openai_api_key = os.getenv('OPENAI_API_KEY')

    os.environ['OPENAI_API_KEY'] = openai_api_key

    custom_template = """
This is conversation with a human. Answer the questions you get based on the knowledge you have.
If you don't know the answer, just say that you don't, don't try to make up an answer.
Chat History:
{chat_history}
Follow Up Input: {question}
"""
    CUSTOM_QUESTION_PROMPT = PromptTemplate.from_template(custom_template)
    llm = ChatOpenAI(
        model_name="gpt-3.5-turbo",  # Name of the language model
        temperature=0  # Parameter that controls the randomness of the generated responses
    )
    embeddings = OpenAIEmbeddings()
    docs = [
        "Buildings are made out of brick",
        "Buildings are made out of wood",
        "Buildings are made out of stone",
        "Buildings are made out of atoms",
        "Buildings are made out of building materials",
        "Cars are made out of metal",
        "Cars are made out of plastic",
    ]
    vectorstore = FAISS.from_texts(docs, embeddings)

    # Use the custom retriever instead of vectorstore.as_retriever() so that
    # get_relevant_documents() only returns documents when the answer really
    # comes from the custom knowledge base.
    retriever = KnowledgeBaseRetriever(
        vectorstore=vectorstore,
        llm=llm,
        knowledge_base_texts=docs,
    )
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    qa = ConversationalRetrievalChain.from_llm(
        llm,
        retriever,
        condense_question_prompt=CUSTOM_QUESTION_PROMPT,
        memory=memory
    )

    # Answer the question that was actually asked (user_input) instead of a
    # hardcoded query, and keep the answer instead of overwriting it.
    result = qa({"question": user_input})

    # get_relevant_documents() now returns the source documents only when the
    # answer comes from the custom knowledge base, and an empty list for
    # general questions such as "hello" or "what's 2+2?".
    source_documents = retriever.get_relevant_documents(user_input)

    return {
        "answer": result.get("answer", ""),
        "source_documents": source_documents,
    }

class KnowledgeBaseRetriever(BaseRetriever):
    """
    Custom retriever that returns the source documents of the custom
    knowledge base only when the answer to the question actually has to come
    from that knowledge base.

    ``get_relevant_documents(query)`` returns:
      * the relevant source documents -> when the question needs the custom
        knowledge base (e.g. "what are cars made of?");
      * an empty list -> for general queries that do not require the
        external documents (e.g. "hello" or "what's 2+2?").

    A plain vector-store retriever cannot do this, because it always returns
    the top-k most similar documents no matter what the query looks like.
    This retriever therefore adds two extra steps:
      1. an LLM "router" decides whether the question needs the knowledge
         base at all -- if it does not, nothing is retrieved;
      2. for knowledge-base questions, only documents whose semantic
         similarity (cosine similarity of the embeddings of the question and
         of the document) reaches a threshold are returned.
    """

    vectorstore: Any  # FAISS store built from the custom knowledge base
    embeddings: Any  # embeddings used to compute the similarity scores
    llm: Any  # LLM used to decide whether the knowledge base is needed
    knowledge_base_description: str = (
        "short facts about what buildings and cars are made of"
    )
    relevance_threshold: float = 0.3  # minimum cosine similarity to keep a doc
    k: int = 4  # number of documents fetched from the vector store

    def _needs_knowledge_base(self, query: str) -> bool:
        """Ask the LLM whether answering `query` requires the custom
        knowledge base (greetings / general-knowledge questions do not)."""
        decision_prompt = (
            "You are a router for a chatbot that has access to a custom "
            "knowledge base.\n"
            f"The knowledge base contains: {self.knowledge_base_description}.\n\n"
            "Decide whether answering the user message below needs information "
            "from that knowledge base:\n"
            "- greetings or small talk (for example 'hello') do NOT need it;\n"
            "- questions that can be answered from general knowledge alone "
            "(for example \"what's 2+2?\") do NOT need it;\n"
            "- questions about the topics stored in the knowledge base DO "
            "need it.\n\n"
            f"User message: {query}\n"
            "Reply with a single word: yes or no:"
        )
        try:
            answer = self.llm.predict(decision_prompt)
        except Exception:
            # If the classification call fails, fall back to retrieving.
            return True

        first_token = (
            answer.strip().lower().split()[0].strip(".,!?")
            if answer.strip()
            else ""
        )
        return first_token != "no"

    def _get_relevant_documents(
        self,
        query: str,
        run_manager: Any = None,
        **kwargs: Any,
    ) -> List[Document]:
        """Return the source documents only when the answer to `query` has
        to come from the custom knowledge base."""
        # 1) General queries ("hello", "what's 2+2?") do not need the
        #    external documents, so an empty result is returned instead of
        #    the always-present top-k documents of a plain retriever.
        if not self._needs_knowledge_base(query):
            return []

        # 2) Knowledge-base question: keep only the documents that are
        #    semantically relevant to the question.
        candidate_docs = self.vectorstore.similarity_search(query, k=self.k)
        query_vector = self.embeddings.embed_query(query)

        relevant_docs: List[Document] = []
        for doc in candidate_docs:
            doc_vector = self.embeddings.embed_query(doc.page_content)
            similarity = self._cosine_similarity(query_vector, doc_vector)
            if similarity >= self.relevance_threshold:
                relevant_docs.append(doc)
        return relevant_docs

    @staticmethod
    def _cosine_similarity(vector_a: List[float], vector_b: List[float]) -> float:
        """Cosine similarity between two embedding vectors."""
        dot_product = sum(a * b for a, b in zip(vector_a, vector_b))
        norm_a = math.sqrt(sum(a * a for a in vector_a))
        norm_b = math.sqrt(sum(b * b for b in vector_b))
        if norm_a == 0 or norm_b == 0:
            return 0.0
        return dot_product / (norm_a * norm_b)
