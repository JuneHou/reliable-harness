import os
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import FAISS
from langchain.chains import ConversationalRetrievalChain
from langchain.memory import ConversationBufferMemory
from langchain.chat_models import ChatOpenAI
from langchain.prompts import PromptTemplate

# Load environment variables from .env file
_ = load_dotenv(find_dotenv())


def run_agent(user_input):
    openai_api_key = os.getenv('OPENAI_API_KEY')

    os.environ['OPENAI_API_KEY'] = openai_api_key

    custom_template = """
This is conversation with a human. Answer the questions you get based on the knowledge you have.
If you don't know the answer, just say that you don't, don't try to make up an answer.
Chat History:
{chat_history}
Follow Up Input: {question}
"""
    CUSTOM_QUESTION_PROMPT = PromptTemplate.from_template(custom_template)
    llm = ChatOpenAI(
        model_name="gpt-3.5-turbo",  # Name of the language model
        temperature=0  # Parameter that controls the randomness of the generated responses
    )
    embeddings = OpenAIEmbeddings()
    docs = [
        "Buildings are made out of brick",
        "Buildings are made out of wood",
        "Buildings are made out of stone",
        "Buildings are made out of atoms",
        "Buildings are made out of building materials",
        "Cars are made out of metal",
        "Cars are made out of plastic",
    ]
    vectorstore = FAISS.from_texts(docs, embeddings)

    # ------------------------------------------------------------------
    # FIX: build the retriever with a similarity-score threshold so that
    # only documents that are genuinely relevant to the user's question
    # are returned. General / small-talk queries such as "hello" or
    # "what's 2+2?" are not similar enough to any document in the
    # knowledge base, so they fall below the threshold and
    # get_relevant_documents() correctly returns an empty list.
    #
    # The default retriever (plain "similarity" search) always returns
    # the top-k nearest documents no matter how low their similarity
    # score is, which is why unrelated queries previously still
    # returned "source" documents.
    # ------------------------------------------------------------------
    retriever = vectorstore.as_retriever(
        search_type="similarity_score_threshold",
        search_kwargs={"score_threshold": 0.5},
    )

    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    qa = ConversationalRetrievalChain.from_llm(
        llm,
        retriever,
        condense_question_prompt=CUSTOM_QUESTION_PROMPT,
        memory=memory
    )

    # ------------------------------------------------------------------
    # FIX: use the actual user input instead of the hard-coded
    # "what are cars made of?" query, so the retrieval (and the answer)
    # always correspond to the question that was asked.
    # ------------------------------------------------------------------
    query = user_input
    answer = qa({"question": query})

    # Because of the score threshold above, this now returns the source
    # documents only when the question actually matches the custom
    # knowledge base, and an empty list for general queries.
    result = retriever.get_relevant_documents(query)

    print(f"Answer: {answer['answer']}")

    return result


if __name__ == "__main__":
    # Question that requires the custom knowledge base:
    # get_relevant_documents returns the matching source documents.
    user_input = "What are buildings made of?"
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Source documents: {[doc.page_content for doc in result]}")

    # General question that does NOT require the knowledge base:
    # get_relevant_documents now returns an empty list.
    general_input = "what's 2+2?"
    result = run_agent(general_input)
    print("\n=== General Query Result ===")
    print(f"Question: {general_input}")
    print(f"Source documents: {[doc.page_content for doc in result]}")
