import os
import uuid
from datetime import datetime
from dotenv import load_dotenv
import chromadb
from chromadb.utils import embedding_functions
from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import CharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings


load_dotenv()


config = {
    "OPENAI_API_KEY": os.getenv("OPENAI_API_KEY")
}

class configs:
    EMBEDDINGS_MODEL = "text-embedding-3-small"


class MockPDFFile:
    def __init__(self, filepath, filename):
        self.filepath = filepath
        self.filename = filename


def read_docs(pdf_file):
    pdf_loader = PyPDFLoader(pdf_file.filepath)
    pdf_documents = pdf_loader.load()

    text_splitter = CharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
    documents = text_splitter.split_documents(pdf_documents)

    return documents


def generate_and_store_embeddings(documents, pdf_file, user_id):
    client = chromadb.PersistentClient(path="./trained_db")
    collection = client.get_or_create_collection(
        "PDF_Embeddings",
        embedding_function=embedding_functions.OpenAIEmbeddingFunction(
            api_key=config["OPENAI_API_KEY"],
            model_name=configs.EMBEDDINGS_MODEL
        )
    )
    now = datetime.now()

    # Per-file metadata that we want to be able to filter on later.
    metadata = {
        "source": pdf_file.filename,
        "user": str(user_id),
        "created_at": now.strftime("%d/%m/%Y %H:%M:%S")
    }

    # FIX: `collection_metadata` only describes the collection itself and is
    # NOT stored with the individual documents/embeddings, which is why the
    # `where={"source": filename}` filter previously returned empty results.
    # The metadata must be attached to each individual Document so it is
    # stored per-embedding and becomes filterable via `where`.
    for doc in documents:
        doc.metadata = {**doc.metadata, **metadata}

    ids = [str(uuid.uuid4()) for _ in range(len(documents))]

    try:
        vectordb = Chroma.from_documents(
            documents,
            embedding=OpenAIEmbeddings(
                openai_api_key=config["OPENAI_API_KEY"],
                model=configs.EMBEDDINGS_MODEL
            ),
            persist_directory='./trained_db',
            collection_name=collection.name,
            client=client,
            ids=ids,
        )

        # Persistence is automatic with chromadb.PersistentClient; only call
        # persist() if the installed langchain version still supports it.
        if hasattr(vectordb, "persist"):
            try:
                vectordb.persist()
            except Exception:
                pass

    except Exception as err:
        print(f"An error occured: {err=}, {type(err)=}")
        return {"answer": "An error occured while generating embeddings. Please check terminal for more details."}
    return vectordb


def run_agent(pdf_path, user_id):

    pdf_file = MockPDFFile(pdf_path, os.path.basename(pdf_path))


    documents = read_docs(pdf_file)
    print(f"    ✓ Loaded {len(documents)} document chunks")


    vectordb = generate_and_store_embeddings(documents, pdf_file, user_id)

    if isinstance(vectordb, dict):

        return None, pdf_file.filename


    return vectordb, pdf_file.filename


if __name__ == "__main__":

    pdf_path = None
    user_id = "test_user_123"


    if os.path.exists("docs"):
        for file in os.listdir("docs"):
            if file.endswith(".pdf"):
                pdf_path = os.path.join("docs", file)
                break

    if not pdf_path:
        print("No PDF file found in docs folder")
        exit(1)


    vectordb, filename = run_agent(pdf_path, user_id)

    if vectordb is None:
        print("Agent failed to create embeddings")
        exit(1)


    # Now that "source" is stored on each document's metadata, this filtered
    # query returns only this file's embeddings (ids + metadata), e.g.:
    #   {'ids': [...], 'metadatas': [{'source': filename, ...}, ...], ...}
    result = vectordb.get(where={"source": filename})

### Root Cause Analysis
The bug was in `generate_and_store_embeddings()`: the per-file metadata (`source`, `user`, `created_at`) was passed via the `collection_metadata` parameter of `Chroma.from_documents()`. In Chroma, `collection_metadata` only describes the **collection object itself** — it is **not** attached to individual documents/embeddings. Consequently, `vectordb.get(where={"source": filename})` matched nothing, and `len(result["ids"])` was `0`, failing the test.

### What the Fix Does
1. **Attaches metadata to each Document** (`for doc in documents: doc.metadata = {**doc.metadata, **metadata}`) — this is the correct approach. Chroma's `where` filter operates on per-record metadata (confirmed by Chroma docs: "Metadata - filter documents based on metadata using where clause in either `Collection.query()` or `Collection.get()`"). Merging with `{**doc.metadata, **metadata}` preserves PyPDFLoader's `page` metadata while overwriting `source` with the basename — which exactly matches the `filename` returned by `run_agent` (`os.path.basename(pdf_path)`), so the test's filter key aligns with the stored value. All metadata values are strings/ints, which are valid Chroma metadata types.
2. **Removes the ineffective `collection_metadata` parameter** — cleaner and no longer misleading.
3. **Guards the `persist()` call** with `hasattr()` and a try/except — correct defensive handling, since `persist()` was removed in `langchain-chroma` 0.4.x (confirmed by docs/issues) and persistence is automatic with `chromadb.PersistentClient`.
4. **Preserves the `run_agent` return signature** `(vectordb, pdf_file.filename)`, matching the test's unpacking.

- `validate_fixed_code` → **ACCEPT** (structure and output format match expectations).

- `Chroma.from_documents(documents, embedding, persist_directory, collection_name, client, ids)` remains valid usage; the LangChain Chroma `get()` proxies to `chromadb.Collection.get()` and returns a dict containing `ids`, so `result["ids"]` is populated.
