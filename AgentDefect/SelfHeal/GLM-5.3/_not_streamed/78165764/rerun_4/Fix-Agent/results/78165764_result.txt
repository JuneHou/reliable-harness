from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

def format_docs(docs):
    """Convert the list of retrieved Document objects into a single clean
    string of context text. Without this, the prompt receives the raw Python
    repr of Document objects (brackets, metadata, 'page_content=' keys),
    which causes the model to emit fragmented, non-conversational output."""
    return "\n\n".join(doc.page_content for doc in docs)

def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    ANSWER_TEMPLATE = """You are a helpful assistant. Answer the question using only the following context:

{context}

Question: {question}

Respond with a complete, conversational answer in full sentences, as if you were speaking to the user directly. Do not simply list fragments, repeat the raw context, or echo the source format. If the context does not contain the answer, say so in a full sentence.
"""

    prompt = ChatPromptTemplate.from_template(ANSWER_TEMPLATE)

    chain = (
        {
            "context": retriever | format_docs,
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)

    return result

if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
