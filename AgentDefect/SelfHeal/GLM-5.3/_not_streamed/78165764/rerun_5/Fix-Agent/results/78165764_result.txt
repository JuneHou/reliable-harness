from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()


def format_docs(docs):
    """Join retrieved documents into clean, plain-text context so the model
    receives readable text instead of raw Document objects (which print as
    fragmented page-content snippets and encourage fragmented echoes)."""
    return "\n\n".join(doc.page_content for doc in docs)


def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    # Explicit prompt: tell the model exactly how to shape its output --
    # a single, complete, conversational answer in full sentences, never a
    # raw repetition of the retrieved context.
    ANSWER_TEMPLATE = """You are a helpful assistant. Answer the question based only on the following context:

{context}

Question: {question}

Instructions:
- Respond with one concise, conversational answer written in complete sentences.
- Synthesize the relevant information from the context and answer the question directly.
- Do not repeat, list, or quote the raw context.
"""

    prompt = ChatPromptTemplate.from_template(ANSWER_TEMPLATE)

    chain = (
        {
            "context": retriever | format_docs,
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)

    return result

if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
