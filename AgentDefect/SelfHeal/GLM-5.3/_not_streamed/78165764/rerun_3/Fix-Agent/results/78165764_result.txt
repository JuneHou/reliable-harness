from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()


def format_docs(docs):
    """Convert retrieved Document objects into clean, readable text.

    Without this, the raw Python list of Document objects is interpolated
    into the prompt (e.g. "[Document(page_content='...', metadata={...})]"),
    which is noisy and pushes the model toward fragmented, keyword-style
    answers instead of proper sentences.
    """
    return "\n\n".join(doc.page_content for doc in docs)


def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    # The original template ("Answer the question based on the following
    # context") told the model WHAT to answer but not HOW to answer.
    # GPT-3.5-turbo follows instructions very literally, so it returned
    # terse, fragmented facts. Llama2 appears more conversational only
    # because its chat template injects a default assistant persona.
    # Fix: add an explicit system prompt that shapes the response style.
    SYSTEM_TEMPLATE = (
        "You are a helpful assistant for question-answering tasks. "
        "Use the following pieces of retrieved context to answer the question. "
        "If you don't know the answer, just say that you don't know. "
        "Always answer in complete, conversational sentences that flow "
        "naturally, as if you were talking to a person. "
        "Do not answer with isolated keywords, fragments, or bullet lists.\n\n"
        "Context:\n{context}"
    )

    prompt = ChatPromptTemplate.from_messages([
        ("system", SYSTEM_TEMPLATE),
        ("human", "{question}"),
    ])

    chain = (
        {
            "context": retriever | format_docs,
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)
    
    return result

if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
