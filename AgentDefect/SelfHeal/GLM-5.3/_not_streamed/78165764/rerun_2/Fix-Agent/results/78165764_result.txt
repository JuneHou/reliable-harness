from fastapi import FastAPI
from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-4o-mini",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple api server using LangChain's Runnable interfaces",
    )

    # Use an explicit system message to set persona/tone, and make the
    # answer-format instructions unambiguous so the model replies with a
    # complete, conversational sentence instead of echoing raw context
    # fragments.
    SYSTEM_TEMPLATE = """You are a helpful assistant that answers user questions \
using only the provided context.

Rules:
- Always answer in complete, grammatically correct sentences.
- Respond in a natural, conversational tone as if speaking to the user.
- Do NOT simply repeat or list fragments of the context.
- Do NOT use bullet points, lists, or single keywords.
- If the answer is not contained in the context, reply politely that you \
don't have enough information to answer."""

    ANSWER_TEMPLATE = """Context:
    {context}

    Question: {question}

    Using only the context above, answer the question in a single complete \
sentence in a conversational tone:"""

    prompt = ChatPromptTemplate.from_messages([
        ("system", SYSTEM_TEMPLATE),
        ("human", ANSWER_TEMPLATE),
    ])

    chain = (
        {
            "context": retriever,
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)

    return result

if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
