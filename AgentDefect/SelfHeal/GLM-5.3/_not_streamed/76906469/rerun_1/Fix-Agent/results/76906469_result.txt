from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# ---------------------------------------------------------------------------
# Why the agent could not remember anything:
#
# `AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION` builds its prompt from the
# "hwchase17/react-chat" template. That prompt's only input variables are
# ["input", "agent_scratchpad"] -- it has NO "chat_history" slot. The
# AgentExecutor still saves every turn into the ConversationBufferMemory and
# even merges the loaded "chat_history" value into the chain inputs, but since
# the prompt template has no such variable, the value is silently dropped when
# the prompt is rendered. The LLM never sees the history, so the agent cannot
# recall previous turns. (This agent type does not support chat memory.)
#
# Fix 1: switch to the agent type that is actually designed for chat models
#   with conversational memory: `CHAT_CONVERSATIONAL_REACT_DESCRIPTION`
#   (ConversationalChatAgent). Its prompt contains a
#   `MessagesPlaceholder(variable_name="chat_history")`, so the memory's
#   content is genuinely injected into every LLM call.
#
# Fix 2: that MessagesPlaceholder expects a *list of message objects*
#   (HumanMessage / AIMessage), not a plain string. Therefore the memory must
#   be created with `return_messages=True`; otherwise
#   ConversationBufferMemory returns a concatenated string and the history is
#   not usable by the placeholder.
# ---------------------------------------------------------------------------
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True,
)

agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False
)

def run_agent(prompt: str) -> str:
    return agent_test.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    result2 = run_agent("what is the last question I asked?")
    print(result2)
