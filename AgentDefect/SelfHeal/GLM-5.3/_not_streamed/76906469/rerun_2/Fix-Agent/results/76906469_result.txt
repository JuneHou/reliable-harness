from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# FIX 2: `return_messages=True` makes the memory expose `chat_history` as a LIST of
# BaseMessage objects. The conversational chat agent renders `chat_history` through a
# MessagesPlaceholder, which requires a list; the default (`return_messages=False`)
# returns a plain concatenated string and would break/be ignored by the chat prompt.
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True,
)

agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    # FIX 1: CHAT_ZERO_SHOT_REACT_DESCRIPTION (ZeroShotChatAgent / hwchase17 react-chat)
    # has NO {chat_history} placeholder in its prompt -- only `input` and
    # `agent_scratchpad` -- so any attached memory is silently ignored and the agent
    # cannot recall previous turns. Use the conversational chat agent
    # (CHAT_CONVERSATIONAL_REACT_DESCRIPTION / ConversationalChatAgent), whose prompt
    # contains a `chat_history` MessagesPlaceholder (matching memory_key="chat_history")
    # and which is explicitly designed to hold a conversation and remember prior turns.
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False
