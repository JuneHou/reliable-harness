from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Memory requirements for the conversational chat agent:
#  1) memory_key must be "chat_history" -- this is the name of the
#     placeholder variable in the agent's prompt template, so the
#     buffer contents actually get injected into the prompt.
#  2) return_messages=True -- the CHAT_CONVERSATIONAL_REACT_DESCRIPTION
#     prompt uses a MessagesPlaceholder, so the memory must return the
#     raw message objects (HumanMessage/AIMessage) instead of a single
#     pre-formatted string.
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True,
)

# NOTE: AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION is a single-turn,
# stateless zero-shot agent. Its prompt template has NO "chat_history"
# placeholder, so any memory attached to it is silently ignored and the
# agent cannot recall previous turns. To get conversational memory, we
# use AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION, which is the
# chat-model agent whose prompt DOES include a chat_history placeholder.
agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False
)


def run_agent(prompt: str) -> str:
    return agent_test.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    # Now that the agent's prompt actually receives the chat history,
    # this follow-up question is answered correctly with the previous
    # turn's content instead of being "forgotten".
    result2 = run_agent("what is the last question I asked?")
    print(result2)
