from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

from langchain_openai import ChatOpenAI
from langchain.memory import ConversationBufferMemory

# `initialize_agent` lives in `langchain` (<=0.2) but was moved to
# `langchain_classic` in newer versions, so import it from wherever it exists.
try:
    from langchain.agents import initialize_agent, AgentType, load_tools
except ImportError:
    from langchain_classic.agents import initialize_agent, AgentType, load_tools

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# ---------------------------------------------------------------------------
# FIX:
# AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION uses a ReAct prompt that contains
# NO `{chat_history}` placeholder. Because the prompt has no such input
# variable, `initialize_agent` never wires the attached memory into the prompt,
# so the ConversationBufferMemory is silently ignored and the agent cannot see
# any previous turns (which is why it cannot answer "what is the last question
# I asked?").
#
# AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION is the chat-model agent whose
# prompt DOES contain a `chat_history` MessagesPlaceholder. With this agent
# type, the memory passed to `initialize_agent` is actually injected into the
# prompt on every call, so the agent can recall earlier questions and answers.
#
# `return_messages=True` is required because the chat conversational agent's
# prompt expects `chat_history` as a list of chat messages rather than a plain
# string, and `memory_key` must be "chat_history" to match the placeholder.
# ---------------------------------------------------------------------------
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True,
)

agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False,
)


def run_agent(prompt: str) -> str:
    return agent_test.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    # Now that the conversation memory is actually injected into the prompt,
    # the agent correctly recalls the previous question.
    result2 = run_agent("what is the last question I asked?")
    print(result2)
