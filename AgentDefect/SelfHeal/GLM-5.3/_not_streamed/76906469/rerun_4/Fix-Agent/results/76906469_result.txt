from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# FIX 1: return_messages=True is required for chat-model-based conversational
# agents. Their prompt uses a MessagesPlaceholder for "chat_history", which
# expects a list of chat messages rather than a plain string. Without this,
# the memory contents cannot be rendered into the chat-model prompt.
memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True
)

# FIX 2: The original AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION prompt template
# only defines the variables {tools}, {tool_names} and {input} -- it has no
# {chat_history} placeholder. Therefore, even though a ConversationBufferMemory
# was attached, its contents were never injected into the LLM prompt, so the
# agent could not recall previous turns. Switching to
# CHAT_CONVERSATIONAL_REACT_DESCRIPTION (the conversational ReAct agent
# optimized for chat models) uses a prompt that includes the chat_history
# variable, which matches the memory_key above and enables memory recall.
agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False
)

def run_agent(prompt: str) -> str:
    return agent_test.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    result2 = run_agent("what is the last question I asked?")
    print(result2)
