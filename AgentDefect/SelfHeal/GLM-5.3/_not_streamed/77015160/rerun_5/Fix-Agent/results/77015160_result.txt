from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def run_agent():
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # --- FIX ---
    # Previously the chain was built with only `chain_type="stuff"` and no
    # `chain_type_kwargs`, so LangChain silently used its built-in default
    # prompt from `langchain.chains.qa_with_sources.prompt`. Printing the
    # chain therefore exposed that hardcoded, pre-filled default template.
    #
    # Supplying our own PromptTemplate via `chain_type_kwargs={"prompt": ...}`
    # makes the chain use (and display) OUR template instead.
    #
    # Note: a "stuff" documents chain fills exactly these two variables, so
    # the custom prompt MUST declare and use `summaries` (retrieved docs)
    # and `question` (user query).
    custom_prompt = PromptTemplate(
        input_variables=["summaries", "question"],
        template=(
            "You are a helpful assistant. Answer the question using only the "
            "context provided below.\n\n"
            "Context:\n{summaries}\n\n"
            "Question: {question}\n\n"
            "Answer:"
        ),
    )

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        chain_type_kwargs={"prompt": custom_prompt},
    )

    return chain


if __name__ == "__main__":
    chain = run_agent()

    # The chain now carries the custom prompt instead of the hardcoded default.
    print("Custom prompt in use:")
    print(chain.combine_documents_chain.llm_chain.prompt.template)

    # Demonstrate the chain actually running with the custom prompt.
    result = chain({"question": "What is the first document about?"})
    print("\nAnswer:", result["answer"])
    print("Sources:", result["sources"])
