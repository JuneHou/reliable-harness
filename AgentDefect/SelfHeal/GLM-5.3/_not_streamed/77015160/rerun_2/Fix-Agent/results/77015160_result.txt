from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def run_agent():
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # The hardcoded template shown when printing the chain is simply LangChain's
    # built-in default for the "stuff" QA-with-sources chain type. To make the
    # chain use (and display) our own template instead of the default one, we
    # must build an explicit PromptTemplate and hand it to the chain through
    # `chain_type_kwargs={"prompt": ...}`.
    custom_template = (
        "Use the following pieces of context to answer the question at the end. "
        "If you don't know the answer, just say that you don't know, "
        "don't try to make up an answer.\n"
        "Return the answer together with a list of the relevant sources "
        'as a "SOURCES" part.\n\n'
        "{summaries}\n\n"
        "Question: {question}\n"
        "Answer:"
    )

    custom_prompt = PromptTemplate(
        template=custom_template,
        input_variables=["summaries", "question"],
    )

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        chain_type_kwargs={"prompt": custom_prompt},
    )

    return chain


if __name__ == "__main__":
    chain = run_agent()

    # Printing the chain now reflects our custom prompt template instead of
    # the framework's pre-filled default.
    print(chain)

    # Verify the custom prompt is actually wired into the chain.
    print("\n--- Prompt actually used by the chain ---")
    print(chain.combine_documents_chain.llm_chain.prompt.template)
