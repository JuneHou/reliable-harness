from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def run_agent():
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # Custom prompt for the "stuff" combine-documents chain.
    # It must expose the exact input variables the chain expects:
    #   - "summaries": the concatenated contents of the retrieved documents
    #   - "question":  the user's question
    # It must also instruct the model to end with a "SOURCES:" section,
    # because RetrievalQAWithSourcesChain parses that from the answer.
    custom_prompt = PromptTemplate(
        input_variables=["summaries", "question"],
        template=(
            "You are a helpful assistant. Use the following pieces of context "
            "to answer the question at the end. If you don't know the answer, "
            "just say that you don't know. Don't try to make up an answer.\n\n"
            "Context:\n{summaries}\n\n"
            "Question: {question}\n"
            'Helpful Answer (end with a line of the form "SOURCES: source1, source2"):'
        ),
    )

    # Optional: custom per-document prompt used to build the "summaries" text.
    custom_document_prompt = PromptTemplate(
        input_variables=["page_content", "source"],
        template="Content: {page_content}\nSource: {source}",
    )

    # Pass the custom prompts through chain_type_kwargs so the chain uses
    # (and prints) our own template instead of the hardcoded default one.
    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        chain_type_kwargs={
            "prompt": custom_prompt,
            "document_prompt": custom_document_prompt,
        },
    )

    return chain


if __name__ == "__main__":
    print(run_agent())
