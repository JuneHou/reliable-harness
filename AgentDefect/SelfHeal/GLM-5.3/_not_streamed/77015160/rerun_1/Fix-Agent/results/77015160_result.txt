from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


# The root cause of the "hardcoded default prompt" issue:
# 1) Printing the chain object (print(chain)) only prints the chain's repr,
#    which serializes its internal state -- including the DEFAULT prompt
#    template that from_chain_type auto-builds for the inner StuffDocumentsChain.
# 2) No custom prompt is ever supplied, because from_chain_type auto-constructs
#    the document chain; the only way to override the prompt is through
#    chain_type_kwargs={"prompt": ...}.
#
# Fix: define an explicit custom prompt template (with the required
# {summaries} and {question} input variables) and pass it in chain_type_kwargs,
# then actually invoke the chain with a question instead of printing its repr.

CUSTOM_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=(
        "You are a helpful assistant. Use the following pieces of context "
        "to answer the question at the end.\n\n"
        "Context:\n{summaries}\n\n"
        "Question: {question}\n"
        "Helpful Answer:"
    ),
)


def run_agent():
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # Supply our own prompt so the chain does NOT fall back to the default one.
    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        chain_type_kwargs={"prompt": CUSTOM_PROMPT},
    )

    return chain


if __name__ == "__main__":
    chain = run_agent()

    # Show the prompt actually in use -- it now reflects the custom template.
    print("=== Custom prompt in use ===")
    print(chain.combine_documents_chain.llm_chain.prompt.template)

    # Run the chain for real: printing the chain object only prints its repr
    # (the serialized internal state), never an answer.
    print("\n=== Invoking the chain ===")
    result = chain({"question": "What is the test document about?"})
    print("Answer:", result["answer"])
    print("Sources:", result["sources"])
