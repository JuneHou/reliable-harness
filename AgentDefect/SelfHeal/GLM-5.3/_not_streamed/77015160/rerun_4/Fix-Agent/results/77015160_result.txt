from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Custom prompt for the QA-with-sources "stuff" chain.
#
# Root cause of the reported behavior:
#   RetrievalQAWithSourcesChain.from_chain_type() builds a StuffDocumentsChain
#   via load_qa_with_sources_chain(), which falls back to the library's built-in
#   default template (langchain/chains/qa_with_sources/stuff_prompt.py). That
#   default ships with hard-coded example content, which is why printing the
#   chain displayed a pre-filled prompt instead of a blank/custom one.
#
# Fix:
#   Pass our own PromptTemplate through `chain_type_kwargs={"prompt": ...}` so
#   the chain is constructed with our template instead of the library default.
#   The template must use exactly the input variables the underlying
#   StuffDocumentsChain expects:
#     - "summaries": the retrieved documents
#     - "question":  the user's actual question
prompt_template = (
    "Use the following pieces of context to answer the question at the end. "
    "If you don't know the answer, just say that you don't know, don't try to "
    "make up an answer.\n\n"
    "{summaries}\n\n"
    "Question: {question}\n"
    "Helpful Answer:"
)

PROMPT = PromptTemplate(
    template=prompt_template,
    input_variables=["summaries", "question"],
)


def run_agent():
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # chain_type_kwargs is forwarded to load_qa_with_sources_chain(), so the
    # "prompt" key overrides the built-in default template for chain_type="stuff".
    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        chain_type_kwargs={"prompt": PROMPT},
    )

    return chain


if __name__ == "__main__":
    print(run_agent())
