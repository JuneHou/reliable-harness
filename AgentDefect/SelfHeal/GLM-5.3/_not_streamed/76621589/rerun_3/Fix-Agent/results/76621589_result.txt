import asyncio


async def run_aapply(chain, texts):
    """Execute the chain's asynchronous apply method and return real results.

    `chain.aapply()` is a coroutine: calling it directly only creates a
    coroutine object. It must be awaited inside an async function that is
    driven by an event loop (started here via asyncio.run from sync code).
    """
    from time import time

    start = time()
    res_aa = await chain.aapply(texts)  # await the coroutine to get results
    print(res_aa)
    print(f"aapply time taken: {time() - start:.2f} seconds")

    return res_aa


def run_agent(texts):

    import os
    from time import time

    import openai
    from dotenv import load_dotenv, find_dotenv
    from langchain.chains import LLMChain
    from langchain.chat_models import ChatOpenAI
    from langchain.prompts import ChatPromptTemplate

    _ = load_dotenv(find_dotenv())
    openai.api_key = os.getenv('OPENAI_API_KEY')

    llm = ChatOpenAI(temperature=0)

    prompt = ChatPromptTemplate.from_template(
        'Classify the text based on the Common European Framework of Reference '
        'for Languages (CEFR). Give a single value: {text}',
    )
    chain = LLMChain(llm=llm, prompt=prompt)

    # --- Synchronous apply ---
    start = time()
    res_a = chain.apply(texts)
    print(res_a)
    print(f"apply time taken: {time() - start:.2f} seconds")
    print()

    # --- Asynchronous apply ---
    # Bridge the sync context into an event loop so the coroutine returned by
    # chain.aapply() is actually awaited and its results are collected.
    res_aa = asyncio.run(run_aapply(chain, texts))

    print(f"Both calls finished: apply={len(res_a)} results, "
          f"aapply={len(res_aa)} results")

    return res_a


if __name__ == "__main__":
    texts = [
        {'text': 'Hallo, ich bin 25 Jahre alt.'},
        {'text': 'Wie geht es dir?'},
        {'text': 'In meiner Freizeit, spiele ich gerne Fussball.'}
    ]

    result = run_agent(texts)
    print("\nReturned result:", result)
