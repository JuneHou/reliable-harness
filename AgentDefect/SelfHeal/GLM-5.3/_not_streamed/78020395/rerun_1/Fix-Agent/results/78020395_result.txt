import os
from typing import List
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain_core.pydantic_v1 import BaseModel, Field
from langchain_core.output_parsers import PydanticOutputParser


load_dotenv()


class MatchedElement(BaseModel):
    """A single identifier/text pair extracted from the error message."""
    identifier: str = Field(description="The element identifier, e.g. identifier1")
    text: str = Field(description="The element text, e.g. text1")


class ErrorMatchList(BaseModel):
    """A list of matched elements extracted from the error message."""
    items: List[MatchedElement] = Field(description="List of matched elements with identifier and text")


def get_output_parser():
    return PydanticOutputParser(pydantic_object=ErrorMatchList)


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    response_as_dict = output_parser.parse(response.content)
    return response_as_dict


if __name__ == "__main__":
    test_msg = "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"
    PROMPT_TEMPLATE = """ 
    You are an android developer. 
    Parse this error message and provide me a list of identifiers & texts mentioned in error message. 
    Every AllOf(...) group corresponds to one separate match object with its own identifier and text. 
    Return one object per AllOf group in the order they appear in the error message. 
    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
