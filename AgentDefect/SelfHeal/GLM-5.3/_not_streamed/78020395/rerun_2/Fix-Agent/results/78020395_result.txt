import os
from typing import List

from dotenv import load_dotenv
from pydantic import BaseModel, Field

from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import PydanticOutputParser

load_dotenv()


class IdentifierTextPair(BaseModel):
    """A single identifier/text pair extracted from the error message."""
    identifier: str = Field(description="This is missing identifier.")
    text: str = Field(description="This is missing text.")


class ParsedErrorContent(BaseModel):
    """All identifier/text pairs found in the error message."""
    elements: List[IdentifierTextPair] = Field(
        description="The list of identifier/text pairs mentioned in the error message."
    )


def get_output_parser():
    # StructuredOutputParser/ResponseSchema can only describe a single flat
    # dictionary (one value per schema name), so it cannot produce nested JSON
    # like a list of {identifier, text} objects. PydanticOutputParser with a
    # nested model (a list of sub-models) is used instead.
    output_parser = PydanticOutputParser(pydantic_object=ParsedErrorContent)
    return output_parser


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    parsed = output_parser.parse(response.content)

    # Convert the parsed pydantic objects into the expected nested JSON output:
    # a list of dictionaries, e.g. [{"identifier": ..., "text": ...}, ...]
    result = []
    for pair in parsed.elements:
        if hasattr(pair, "model_dump"):  # pydantic v2
            result.append(pair.model_dump())
        else:                             # pydantic v1 fallback
            result.append(pair.dict())
    return result


if __name__ == "__main__":
    test_msg = "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"
    PROMPT_TEMPLATE = """ 
    You are an android developer. 
    Parse this error message and provide me identifiers & texts mentioned in error message. 
    Return every identifier/text pair found in the error message as a separate entry 
    in the "elements" list.
    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
