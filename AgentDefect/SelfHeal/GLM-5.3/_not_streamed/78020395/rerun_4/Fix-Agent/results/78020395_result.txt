import os
from typing import List

from dotenv import load_dotenv
from pydantic import BaseModel, Field
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import PydanticOutputParser


load_dotenv()


class MissingElement(BaseModel):
    """A single missing UI element parsed from the error message."""
    identifier: str = Field(description="The missing identifier (e.g. withId value)")
    text: str = Field(description="The missing text (e.g. withText value)")


class MissingElements(BaseModel):
    """The full list of missing elements extracted from the error message."""
    missing_elements: List[MissingElement] = Field(
        description="List of missing elements, each with an identifier and a text"
    )


def get_output_parser():
    # StructuredOutputParser only supports a single flat dict of named
    # ResponseSchemas, so it can never express "a list of dicts with
    # identifier/text pairs".  PydanticOutputParser lets us define a nested
    # (list-of-objects) JSON schema instead, which is what the user wants.
    return PydanticOutputParser(pydantic_object=MissingElements)


def predict_result(msg, PROMPT_TEMPLATE):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    # PydanticOutputParser returns a MissingElements object;
    # convert it to plain dicts (list of {"identifier": ..., "text": ...})
    # so the printed output matches the expected nested JSON structure.
    parsed = output_parser.parse(response.content)
    return [element.model_dump() for element in parsed.missing_elements]


if __name__ == "__main__":
    test_msg = "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"
    PROMPT_TEMPLATE = """
    You are an android developer.
    Parse this error message and provide me identifiers & texts mentioned in error message.
    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
