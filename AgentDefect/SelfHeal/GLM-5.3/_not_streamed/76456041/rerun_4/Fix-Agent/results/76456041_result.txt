import os
import warnings
import pandas as pd
from dotenv import load_dotenv, find_dotenv
from langchain.callbacks import get_openai_callback
from langchain.llms import OpenAI
from langchain_experimental.agents import create_pandas_dataframe_agent

warnings.filterwarnings('ignore')

load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:

            # Load the CSV as a real pandas DataFrame.
            # The map_reduce chain over a vector store treated the CSV as
            # unstructured text: each chunk only saw a few rows, so
            # aggregation-style questions ("list every German driver",
            # "what is the oldest birthdate") could not be answered reliably
            # and the reduce step merged lossy summaries. The pandas agent
            # instead computes the exact answer over the full DataFrame.
            df = pd.read_csv(file, encoding='utf-8')

            # Create the agent that generates and runs pandas code against df.
            # This is also dramatically cheaper: no embedding of every CSV row
            # and no per-chunk map-reduce LLM calls for every question.
            agent = create_pandas_dataframe_agent(
                llm=OpenAI(temperature=0),
                df=df,
                verbose=True,
                handle_parsing_errors=True,
            )

            for query in queries:
                print(query)
                result = agent.run(query)
                results.append(result)
                print(result)

            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German",
        "What is the birthdate of the oldest driver?"
    ]
    res = run_agent(queries)
