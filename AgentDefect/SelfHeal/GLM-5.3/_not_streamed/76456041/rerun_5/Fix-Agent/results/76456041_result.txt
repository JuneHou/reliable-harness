import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.prompts import PromptTemplate
from langchain.schema import Document

warnings.filterwarnings('ignore')

load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")

# ---------------------------------------------------------------------------
# How many raw CSV rows (i.e. CSVLoader Documents) are packed into a single
# chunk.  CSVLoader returns ONE document per row, which makes map_reduce
# issue (and bill) one LLM call *per row* -- this is the main reason the
# full-CSV run was so expensive.  Packing ~100 rows per chunk cuts the
# number of map calls (and embeddings) by roughly 100x, while giving the
# map step enough rows at once to extract several drivers per call.
# ---------------------------------------------------------------------------
ROWS_PER_CHUNK = 100
RETRIEVER_K = 1000  # must be >= number of chunks so NOTHING is left out


# ---------------------------------------------------------------------------
# The default map_reduce prompts are written for summarising long documents:
#   map:   "Use the following portion of a long document to see if any of
#           the text is useful in answering the question..."
#   reduce: "If you don't know the answer, just say that you don't know.
#           DON'T make up an answer."
# With one-driver-per-chunk CSV data those prompts make the model answer
# "the context does not mention..." or list only a couple of names, and the
# combine step can never aggregate (count drivers, find the oldest
# birthdate, ...).  We replace both prompts with explicit extraction /
# aggregation instructions.
# ---------------------------------------------------------------------------
MAP_PROMPT = PromptTemplate(
    input_variables=["context", "question"],
    template=(
        "The following rows come from a CSV file that describes drivers.\n\n"
        "Context:\n{context}\n\n"
        "Question: {question}\n\n"
        "Look at EVERY row above and extract only the information the "
        "question asks for (for example, the name of each driver whose "
        "nationality is the one requested, or every birthdate).  "
        "Return the extracted values as a comma-separated list.  "
        "If nothing in these rows matches, reply with exactly: NONE"
    ),
)

REDUCE_PROMPT = PromptTemplate(
    input_variables=["context", "question"],
    template=(
        "The following are partial answers extracted from different parts "
        "of a CSV file that describes drivers.\n\n"
        "Partial answers:\n{context}\n\n"
        "Question: {question}\n\n"
        "Combine ALL of the partial answers into ONE final answer that "
        "covers the ENTIRE dataset: merge the lists, remove duplicates, and "
        "do NOT drop any value.  If the question asks for a count, count "
        "the combined values; if it asks for an extreme value (e.g. the "
        "oldest birthdate), compute it from the combined values.  Ignore "
        "any partial answer that is NONE.  Final answer:"
    ),
)

CHAIN_TYPE_KWARGS = {
    "question_prompt": MAP_PROMPT,    # the "map" step
    "combine_prompt": REDUCE_PROMPT,  # the "reduce" step
}


def group_rows_into_chunks(docs, rows_per_chunk=ROWS_PER_CHUNK):
    """Merge the one-document-per-row output of CSVLoader into larger
    chunks so map_reduce runs on a handful of documents instead of one
    LLM call per row (this is what made the full-CSV version so costly)."""
    chunks = []
    for i in range(0, len(docs), rows_per_chunk):
        batch = docs[i:i + rows_per_chunk]
        chunks.append(
            Document(
                page_content="\n".join(d.page_content for d in batch),
                metadata={
                    "source": batch[0].metadata.get("source", ""),
                    "row_start": i,
                },
            )
        )
    return chunks


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:

            loader = CSVLoader(file_path=file, encoding='utf-8')
            docs = loader.load()

            # pack the rows into a few large chunks (major cost reduction)
            chunks = group_rows_into_chunks(docs)

            embeddings = OpenAIEmbeddings()

            # vector store over the CHUNKS, not the individual rows
            db = Chroma.from_documents(chunks, embeddings)

            # k must cover every chunk so the whole dataset is aggregated
            retriever = db.as_retriever(
                search_type="similarity",
                search_kwargs={"k": RETRIEVER_K}
            )

            # build the chain ONCE and reuse it for every query
            qa_stuff = RetrievalQA.from_chain_type(
                llm=OpenAI(temperature=0, batch_size=20),
                chain_type="map_reduce",
                retriever=retriever,
                chain_type_kwargs=CHAIN_TYPE_KWARGS,
                verbose=True
            )

            for query in queries:
                print(query)
                result = qa_stuff.run(query)
                results.append(result)

                print(result)

            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German"
    ]
    res = run_agent(queries)
