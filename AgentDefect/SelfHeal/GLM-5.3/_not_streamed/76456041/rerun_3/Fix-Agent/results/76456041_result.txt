import os
import uuid
import warnings

from dotenv import load_dotenv, find_dotenv

from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.docstore.document import Document
from langchain.document_loaders import CSVLoader
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.llms import OpenAI
from langchain.prompts import PromptTemplate
from langchain.vectorstores import Chroma

warnings.filterwarnings('ignore')

load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")

# ---------------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------------
# CSVLoader produces ONE document per CSV row and the map_reduce chain makes
# ONE LLM call per document.  Feeding it row-by-row documents therefore meant
# one API call per row of the full CSV (the source of the huge bill), and each
# map step only ever saw a single isolated row.  Packing ROWS_PER_DOCUMENT
# rows into one document cuts the number of map calls (and embeddings) by that
# factor while every chunk still stays comfortably inside the model context.
ROWS_PER_DOCUMENT = 20

# ---------------------------------------------------------------------------
# Custom map / reduce prompts  (the actual bug fix)
#
# The built-in map_reduce prompts are purely extractive: the map step is told
# to "return any relevant text verbatim" and the combine step merely re-answers
# from that raw text.  That is fine for lookups but fails for aggregation
# questions ("name every German driver", "oldest birthdate"): no step is ever
# asked to count rows or compare dates, so the reduce step receives a wall of
# unprocessed rows, truncates them and returns a wrong/partial answer.  It only
# *appeared* to work with chain_type="stuff" on the small sample file because
# there the whole table fitted into a single prompt.
#
# The replacement prompts must use exactly the input variables that the
# map_reduce chain expects:
#     map (question) step -> ["context", "question"]
#     combine (reduce) step -> ["summaries", "question"]
# ---------------------------------------------------------------------------
QUESTION_PROMPT = PromptTemplate(
    input_variables=["context", "question"],
    template=(
        "The following is a chunk of rows taken from a CSV table of drivers:\n"
        "---------------------\n{context}\n---------------------\n"
        "Using ONLY the rows in this chunk, produce the partial result needed "
        "to answer the question:\n"
        "- If the question asks which rows match a condition (for example a "
        "given nationality), list EVERY matching row of this chunk together "
        "with the fields needed to answer (name, nationality, birthdate, ...).\n"
        "- If the question asks for a count, list the matching rows AND state "
        "how many of them are in this chunk.\n"
        "- If the question asks for an extreme value (oldest/youngest driver, "
        "min/max of a column), list the candidate rows of this chunk together "
        "with the value that must be compared.\n"
        "Never invent rows that are not present in the chunk.\n"
        "If nothing in this chunk is relevant, reply exactly with: "
        "No relevant rows in this chunk.\n"
        "Question: \"{question}\"\n"
        "Partial result for this chunk:"
    ),
)

COMBINE_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=(
        "The following are partial results, each computed on a different "
        "chunk of the same CSV table:\n"
        "---------------------\n{summaries}\n---------------------\n"
        "Combine them into one single final answer to the question:\n"
        "merge the matching rows reported by all chunks, remove duplicates so "
        "that every driver is listed and counted only once, then actually "
        "COMPUTE the requested aggregate (the total count of the merged rows, "
        "or the extreme value) instead of just repeating the extracted text.\n"
        "Question: \"{question}\"\n"
        "Final answer:"
    ),
)


def batch_documents(row_documents, rows_per_doc=ROWS_PER_DOCUMENT):
    """Merge the one-row-per-document output of CSVLoader into batch documents.

    Keeps the number of map-step LLM calls (one per document) low while each
    chunk remains a well-formed, self-describing table excerpt.
    """
    batched, chunk = [], []
    for doc in row_documents:
        chunk.append(doc.page_content)
        if len(chunk) >= rows_per_doc:
            batched.append(Document(page_content="\n".join(chunk)))
            chunk = []
    if chunk:
        batched.append(Document(page_content="\n".join(chunk)))
    return batched


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:

            loader = CSVLoader(file_path=file, encoding='utf-8')
            docs = loader.load()

            if not docs:
                print(f"{file}: no rows found, skipping.")
                continue

            # Cost fix: batch the rows so the map step runs once per batch
            # instead of once per row (~ROWS_PER_DOCUMENT times fewer LLM
            # calls, and fewer embeddings as well).
            batched_docs = batch_documents(docs)

            embeddings = OpenAIEmbeddings()

            # A unique collection name per file (and per run) keeps the
            # in-memory Chroma collections of drivers.csv and
            # drivers_full.csv separate; with the shared default collection
            # name, rows of the first file would leak into the answers for
            # the second one and inflate counts.
            collection_name = (
                f"qa_{os.path.splitext(os.path.basename(file))[0]}"
                f"_{uuid.uuid4().hex[:8]}"
            )

            # create the vectorstore to use as the index (over the batches)
            db = Chroma.from_documents(
                batched_docs, embeddings, collection_name=collection_name
            )

            # Correctness fix: aggregation questions need the COMPLETE table,
            # so retrieve every chunk (k = number of chunks).  A small k would
            # silently drop rows and produce wrong counts / wrong extremes.
            retriever = db.as_retriever(
                search_type="similarity",
                search_kwargs={"k": len(batched_docs)},
            )

            # Build the chain once per file (not once per query) and hand
            # map_reduce the aggregation-aware prompts defined above instead
            # of the default "extract relevant text" prompts.
            qa_chain = RetrievalQA.from_chain_type(
                llm=OpenAI(temperature=0),
                chain_type="map_reduce",
                retriever=retriever,
                chain_type_kwargs={
                    "question_prompt": QUESTION_PROMPT,
                    "combine_prompt": COMBINE_PROMPT,
                },
                verbose=True,
            )

            for query in queries:
                print(query)
                result = qa_chain.run(query)
                results.append(result)

                print(result)

            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German",
        "What is the birthdate of the oldest driver?",
    ]
    res = run_agent(queries)
