import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.callbacks import get_openai_callback
from langchain.llms import OpenAI

warnings.filterwarnings('ignore')

load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def run_agent(queries):
    results = []

    files = ['drivers.csv', 'drivers_full.csv']

    # The CSV agent lives in langchain_experimental in newer releases and in
    # langchain.agents in older ones -- import whichever is available.
    try:
        from langchain_experimental.agents import create_csv_agent
    except ImportError:
        from langchain.agents import create_csv_agent

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:

            # Why the previous version produced wrong answers:
            #   - CSVLoader turns every CSV row into its own Document, and the
            #     "map_reduce" chain answers each row separately with an LLM and
            #     then merges the answers. Aggregation questions ("name every
            #     German driver", "oldest date of birth") need an exact,
            #     table-wide computation, which a lossy LLM summarization chain
            #     cannot guarantee: rows get dropped or mis-merged, so the counts
            #     and names come out wrong. "stuff" only worked on the small
            #     sample because the whole table then fit into one prompt.
            #   - The similarity retriever returned rows that were semantically
            #     closest to the query, not necessarily *all* relevant rows, so
            #     relevant rows could be silently excluded before map/reduce even
            #     started.
            #   - Embedding every row plus one LLM call per row (map) plus the
            #     reduce step, per query per file, is extremely expensive.
            #
            # Fix: use the CSV agent. The LLM generates pandas code that is
            # executed over the entire DataFrame, so aggregation questions are
            # answered exactly, and it needs no embeddings and only a handful of
            # LLM calls per query, which removes the high API cost.
            agent = create_csv_agent(
                OpenAI(temperature=0),
                file,
                verbose=True,
                handle_parsing_errors=True,
            )

            for query in queries:
                print(query)
                result = agent.run(query)
                results.append(result)

                print(result)

            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German"
    ]
    res = run_agent(queries)
