import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_core.tools import tool
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # Fixed: the parameter is `max_results` (plural). The previous `max_result`
    # keyword was silently ignored, so the search fell back to its default
    # number of results instead of the intended 2.
    simple_search = TavilySearch(max_results=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key
    )

    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory
    )

    # The checkpointer plus this thread_id is ALL that LangGraph needs to keep
    # the conversation history for this thread across invocations.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, memory, config, user_input):
    # Fixed: do NOT manually rebuild the conversation history.
    #
    # `memory.get(config)` returns the checkpointer's low-level (serialized)
    # checkpoint, not the graph state, so `state["channel_values"]["messages"]`
    # is not a usable list of chat messages. Worse, re-feeding the stored
    # history as graph input conflicts with the state the checkpointer already
    # restored for this thread_id, duplicating/garbling the message list so the
    # model never sees a coherent prior conversation.
    #
    # Because a checkpointer is attached and every call streams with the same
    # thread_id, LangGraph automatically reloads the thread's accumulated
    # `messages` state and appends the new user message through the
    # `add_messages` reducer. Therefore we only pass the NEW user message:
    inputs = {"messages": [HumanMessage(user_input)]}

    final_message = None
    for step in agent_executor.stream(
        inputs,
        config,
        stream_mode="values"
    ):
        final_message = step["messages"][-1]

    return final_message.content if final_message is not None else ""


def main():
    agent_executor, memory, config = initialize_agent()
    response1 = run_agent(agent_executor, memory, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, memory, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
