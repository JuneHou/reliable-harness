import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # FIX: the correct parameter name is "max_results" (not "max_result"),
    # otherwise the intended result limit of 2 is never applied.
    simple_search = TavilySearch(max_results=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key
    )

    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory
    )

    # A stable thread_id identifies this conversation. The checkpointer
    # (MemorySaver) persists the full message history for this thread
    # automatically across invocations.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, memory, config, user_input):
    # FIX: Do NOT manually rebuild the conversation history from the raw
    # checkpoint (memory.get(config)["channel_values"]["messages"]) and re-feed
    # it as graph input. The agent's state uses the `add_messages` reducer and
    # the checkpointer already maintains the full history for the thread_id,
    # so re-supplying old messages as input conflicts with that mechanism and
    # prevents the agent from recalling earlier turns.
    #
    # Correct usage: pass ONLY the new user message. LangGraph loads the
    # checkpointed state for this thread, appends the new message via the
    # add_messages reducer, and the model sees the whole conversation.
    final_message = None

    for step in agent_executor.stream(
        {"messages": [HumanMessage(content=user_input)]},
        config,
        stream_mode="values"
    ):
        final_message = step["messages"][-1]

    if final_message is None:
        return ""
    return final_message.content


def main():
    agent_executor, memory, config = initialize_agent()
    response1 = run_agent(agent_executor, memory, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, memory, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
