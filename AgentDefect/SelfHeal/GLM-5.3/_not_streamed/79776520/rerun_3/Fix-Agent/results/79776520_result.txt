import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # NOTE: the correct parameter name is "max_results" (with an "s").
    # "max_result" is not a valid field and gets silently ignored.
    simple_search = TavilySearch(max_results=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key
    )

    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory
    )

    # A stable thread_id is what ties all turns of one conversation together.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, memory, config, user_input):
    """
    Send a single user message to the agent.

    The checkpointer (MemorySaver) together with the thread_id in `config`
    already stores and automatically restores the full conversation history
    for this thread on every invocation: LangGraph loads the saved
    `messages` state and appends the new input to it via the `add_messages`
    reducer.

    Therefore we must pass ONLY the new user message. Manually reading the
    raw checkpoint (memory.get(config) -> "channel_values") and re-feeding
    the history is wrong: `channel_values` holds *serialized* message
    payloads (not BaseMessage objects), which produces a corrupted input
    state and makes the agent "forget" earlier turns. (If you ever need to
    inspect the history, use `agent_executor.get_state(config)`, whose
    `.values["messages"]` are real BaseMessage objects.)
    """
    final_message = None

    for step in agent_executor.stream(
        {"messages": [HumanMessage(user_input)]},  # only the NEW message
        config,                                    # same thread_id
        stream_mode="values",
    ):
        final_message = step["messages"][-1]

    if final_message is None:
        return "No response generated."
    return final_message.content


def main():
    agent_executor, memory, config = initialize_agent()
    response1 = run_agent(agent_executor, memory, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, memory, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
