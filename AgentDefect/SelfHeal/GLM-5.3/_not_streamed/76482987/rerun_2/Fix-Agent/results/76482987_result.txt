        print("number of documents should be 2")
        return
    chroma_client = chromadb.Client()
    collection = chroma_client.create_collection(name="my_collection")
    collection.add(
        embeddings=[[1.2, 2.3, 4.5], [6.7, 8.2, 9.2]],
        documents=documents,
        metadatas=[{"source": "my_source"}, {"source": "my_source"}],
        ids=["id1", "id2"]
    )

    # By default, collection.get() only includes "metadatas" and "documents"

    # embeddings were explicitly added. To retrieve them, request them
    # explicitly via the `include` parameter.
    return collection.get(include=["embeddings", "documents", "metadatas"])
