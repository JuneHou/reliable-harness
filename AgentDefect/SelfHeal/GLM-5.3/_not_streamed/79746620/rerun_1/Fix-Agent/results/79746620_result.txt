"""Corrected code: filtering LlamaIndex metadata for "null" values.

WHY THE ORIGINAL CODE RETURNED NO NODES
---------------------------------------
`FilterOperator.EQ` with `value=None` can never match a node -- not even one
whose metadata was explicitly set to `None` -- because LlamaIndex does not
treat `None` as a filterable metadata value:

1. Index time: when a node is inserted into a `VectorStoreIndex`, its
   metadata is serialized for the vector store (see `node_to_metadata_dict`
   in `llama_index.core.vector_stores.utils`), and entries whose value is
   `None` are dropped, because most vector stores do not support null
   metadata values.  The docstore still prints `{'start_date': None}`, but
   the vector store's copy of the metadata no longer contains that key, so
   there is nothing to match against.

2. Query time: the in-memory filter matcher used by the default
   (Simple)VectorStore -- `_node_matches_filter` in
   `llama_index.core.vector_stores.utils` -- short-circuits and returns
   `False` as soon as the value stored for a key is `None` or missing.  The
   operator (EQ) is never even evaluated against the filter value.

Together, (1) and (2) make
    MetadataFilter(key="start_date", operator=FilterOperator.EQ, value=None)
always retrieve zero nodes, even though the node's metadata was explicitly
set to `None` in the docstore.

HOW TO PROPERLY INDEX AND FILTER FOR NULL METADATA
--------------------------------------------------
* Index time: store a *sentinel* value instead of `None`.  The empty string
  `""` is the best sentinel, because it is exactly what
  `FilterOperator.IS_EMPTY` is designed to match.  (With this approach
  "explicitly null" and "key missing" are treated the same; pick a distinct
  sentinel such as `"null"` if you ever need to tell them apart.)
* Query time: filter with `FilterOperator.IS_EMPTY` (the operator meant for
  empty/null values), or equivalently with `FilterOperator.EQ` on the
  sentinel value itself (`value=""`).

Both filters are tried below (IS_EMPTY first, with an EQ-on-sentinel
fallback) so the node is retrieved regardless of the llama-index version in
use -- some older versions do not implement `IS_EMPTY` for in-memory
filtering.
"""

from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# Sentinel stored in node metadata to represent a "null" value.
# `None` itself cannot be used: it is dropped from the vector store's copy of
# the metadata and ignored by the filter matcher.
NULL_METADATA_VALUE = ""


def to_indexable_metadata(metadata):
    """Map `None`-valued metadata entries to the sentinel value so that they
    survive indexing and remain filterable."""
    if not metadata:
        return {}
    return {
        key: NULL_METADATA_VALUE if value is None else value
        for key, value in metadata.items()
    }


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node -- store the "" sentinel instead of None.
    # This is the indexing half of the fix: with
    # `metadata={"start_date": None}` the key/value pair is dropped from the
    # vector store's metadata and can never be matched by any filter.
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=to_indexable_metadata({"start_date": None}),
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter for a "null"/empty start_date.
    #
    # The original `MetadataFilter(key="start_date",
    # operator=FilterOperator.EQ, value=None)` can never match anything (see
    # the module docstring).  Instead we filter on the sentinel we indexed,
    # using FilterOperator.IS_EMPTY -- the operator designed for empty/null
    # metadata values.
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.IS_EMPTY,
        value=None,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    if not nodes:
        # Compatibility fallback: exact-match on the sentinel value.  This is
        # equivalent to IS_EMPTY for the "" sentinel and is supported by
        # every llama-index version (older in-memory filters may not
        # implement IS_EMPTY).
        filter_null_start_date = MetadataFilter(
            key="start_date",
            operator=FilterOperator.EQ,
            value=NULL_METADATA_VALUE,
        )
        filters = MetadataFilters(filters=[filter_null_start_date])
        retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
        nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved (None is not filterable)")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
