from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# ---------------------------------------------------------------------------
# Sentinel used to represent a "null" metadata value.
#
# Why a real None does not work:
#   When nodes are inserted into a vector store, LlamaIndex converts them
#   with node_to_metadata_dict(), which drops every metadata entry whose
#   value is None (most vector stores cannot persist null values).  The
#   docstore keeps the original node object, which is why the debug print
#   still shows {'start_date': None}, but the vector store itself ends up
#   with an *empty* metadata dict for the node.  A MetadataFilter on
#   "start_date" then fails the "key not present in metadata" check, so
#   FilterOperator.EQ with value=None can never match anything.
#
# Fix:
#   Index a sentinel string instead of a real None, and filter with EQ on
#   that sentinel.  The key survives ingestion, the filter matches, and the
#   approach is portable to every vector store (Pinecone, Chroma, Qdrant...)
# ---------------------------------------------------------------------------
NULL_SENTINEL = "None"


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node -- use the sentinel (not a raw None) so the "start_date"
    # key actually survives vector store ingestion and stays filterable.
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter for the "null" start_date by matching the sentinel value.
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved (None is not filterable)")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
