from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# ---------------------------------------------------------------------------
# Sentinel used to represent "null" in node metadata.
#
# Root cause of the original bug:
# LlamaIndex metadata filters cannot match against `None`. A MetadataFilter
# built with `FilterOperator.EQ` and `value=None` is NOT an "IS NULL" query
# -- a metadata field whose value is None is treated as missing/empty (the
# default in-memory SimpleVectorStore explicitly does not support null
# filter values), so the equality check never matches and the retriever
# returns no nodes even though the node clearly sits in the docstore.
#
# Proper way to *index* a null metadata value: store an explicit sentinel
# string instead of `None`. Every vector store can index and match a string
# with a plain EQ filter. (Pick a sentinel that cannot collide with real
# values in your data.)
# ---------------------------------------------------------------------------
NULL_VALUE = "null"


def to_sentinel(metadata):
    """Replace None metadata values with the NULL_VALUE sentinel so that
    "null" fields are actually stored and stay filterable."""
    return {
        key: (NULL_VALUE if value is None else value)
        for key, value in metadata.items()
    }


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node with a null start_date.
    # FIX (indexing): `None` is not a filterable metadata value, so the null
    # is indexed as the sentinel string "null" instead of a raw None.
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=to_sentinel({"start_date": None}),  # -> {"start_date": "null"}
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # FIX (filtering): filter for the null start_date by matching the
    # sentinel value. `EQ` with `value=None` never matches (it is not an
    # "IS NULL" query); `EQ` on the sentinel string works on every vector
    # store.
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_VALUE,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()
    # (Map the sentinel back to None downstream if you need the raw value:
    #     value = node.metadata["start_date"]
    #     start_date = None if value == NULL_VALUE else value)

    # -----------------------------------------------------------------------
    # Built-in alternative: FilterOperator.IS_EMPTY.
    #
    # IS_EMPTY matches nodes where the key is missing, None, or empty -- it
    # is LlamaIndex's supported replacement for the unsupported
    # `EQ` + `value=None` pattern. Caveats: it cannot distinguish
    # "explicitly set to None" from "never set at all", and operator
    # support varies between vector stores and llama-index versions (which
    # is why the sentinel approach above is the portable one). This
    # demonstration is therefore guarded so it never breaks the script.
    # -----------------------------------------------------------------------
    try:
        node_none = TextNode(
            text="This document has None in the metadata",
            id_="node_none",
            metadata={"start_date": None},
        )
        none_index = VectorStoreIndex([node_none])
        is_empty_filters = MetadataFilters(
            filters=[
                MetadataFilter(
                    key="start_date",
                    operator=FilterOperator.IS_EMPTY,
                )
            ]
        )
        none_retriever = none_index.as_retriever(
            filters=is_empty_filters, similarity_top_k=1
        )
        none_nodes = none_retriever.retrieve(query)
        print("Retrieved nodes (IS_EMPTY on raw None metadata):")
        print([(node.node_id, node.metadata) for node in none_nodes])
        print()
    except Exception as exc:
        print(
            "IS_EMPTY alternative skipped (not supported by this vector "
            f"store / llama-index version: {exc})"
        )
        print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
