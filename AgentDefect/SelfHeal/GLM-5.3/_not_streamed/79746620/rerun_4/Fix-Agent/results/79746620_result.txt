from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# ---------------------------------------------------------------------------
# WHY THE ORIGINAL CODE FAILED:
#
#   A metadata value of ``None`` is NOT filterable in LlamaIndex.  When the
#   node is added to the vector store, a ``None`` metadata value is treated
#   as "no value": it never compares equal to anything, so
#   ``MetadataFilter(key=..., operator=FilterOperator.EQ, value=None)``
#   matches zero nodes -- even though the docstore clearly shows
#   ``{'start_date': None}`` in the debug print.  Vector store backends also
#   generally cannot perform null-equality filters.
#
# HOW TO PROPERLY INDEX AND FILTER FOR NULL METADATA:
#
#   1. INDEX:  never store a raw ``None``.  Store an explicit, non-falsy
#              sentinel string instead (here: "null").
#   2. FILTER: filter on that sentinel with ``FilterOperator.EQ``.
#   3. READ:   map the sentinel back to ``None`` when consuming results.
#
#   (``FilterOperator.IS_EMPTY`` can also be used to find nodes whose field
#   is empty/missing where the store supports it, but the sentinel approach
#   below is portable across every vector store and distinguishes an
#   explicitly-null field from an absent one.)
# ---------------------------------------------------------------------------

NULL_SENTINEL = "null"


def metadata_to_storable(metadata):
    """Indexing helper: replace ``None`` values with the filterable sentinel
    so the value actually lands in the vector store's metadata."""
    return {
        key: NULL_SENTINEL if value is None else value
        for key, value in (metadata or {}).items()
    }


def metadata_from_storable(metadata):
    """Read helper: map the sentinel back to ``None`` for display/consumption."""
    return {
        key: None if value == NULL_SENTINEL else value
        for key, value in (metadata or {}).items()
    }


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node -- store the sentinel ("null") instead of a raw None so the
    # metadata value is actually stored and filterable.
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=metadata_to_storable({"start_date": None}),
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter by the sentinel value (NOT by None) -- this now matches.
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results (map the sentinel back to None for readability)
    print("Retrieved nodes:")
    print(
        [
            (node.node_id, metadata_from_storable(node.metadata))
            for node in nodes
        ]
    )
    print()

    return nodes
