"""
LlamaIndex metadata filtering for "null" metadata values.

Root cause of the original bug:
    The default (in-memory) vector store evaluates metadata filters by
    interpolating the filter value into a comparison expression, e.g.
        node.metadata.get('start_date') == '<value>'
    With FilterOperator.EQ and value=None, the stored Python None is
    compared against the *string* "None", which never matches. The
    retriever therefore returns no nodes even though the docstore clearly
    shows {'start_date': None}. MetadataFilter values are expected to be
    comparable scalars (str/int/float), not None.

Fix (Change Parameter Value -- align data/filter with what the framework
actually supports):
    Do not store a raw Python None in node metadata. Index an explicit
    sentinel string (here: "null") that marks the field as "not set", and
    filter with FilterOperator.EQ on that same sentinel string. The filter
    comparison then happens string-to-string and the node is retrieved as
    expected. When consuming results, the sentinel can simply be mapped
    back to None (see NULL_SENTINEL usage below).
"""

from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# Sentinel used to represent a null / missing metadata value.
# This is what actually gets indexed and what the EQ filter compares against.
NULL_SENTINEL = "null"


def run_agent(query):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node -- store the sentinel string instead of a raw Python None
    # so the value survives the vector store's (string-based) filter
    # comparison at query time.
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter by the "null" sentinel value instead of Python None.
    # EQ now compares "null" == "null", which matches.
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results (mapping the sentinel back to None for readability)
    print("Retrieved nodes:")
    print(
        [
            (
                node.node_id,
                {
                    k: (None if v == NULL_SENTINEL else v)
                    for k, v in node.metadata.items()
                },
            )
            for node in nodes
        ]
    )
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)

    response = run_agent(query)

    if not response:
        print("\nResult: No nodes retrieved")
    else:
        print(f"\nResult: Found {len(response)} node(s)")
