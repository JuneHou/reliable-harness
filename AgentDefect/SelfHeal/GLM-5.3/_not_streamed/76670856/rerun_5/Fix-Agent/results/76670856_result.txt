from dotenv import load_dotenv
from pathlib import Path
import json
import os
import sys

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.document_loaders import JSONLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.prompts import PromptTemplate
from langchain.vectorstores import Chroma

load_dotenv()

# Custom QA prompt: the default "stuff" prompt gives the model no instruction
# about counting documents, so it guessed. Tell it that each context block is
# one review and that it must answer strictly from the provided context.
QA_PROMPT = PromptTemplate(
    input_variables=["context", "question"],
    template=(
        "You are answering questions about a collection of hotel reviews. "
        "Each block of text in the context below is exactly one review. "
        "Base your answer ONLY on the context below. "
        "If the question asks how many reviews there are, count the review "
        "texts shown in the context and report exactly that number. "
        "If the question asks to list or describe reviews, use only the "
        "reviews that actually appear in the context. "
        "If the answer is not contained in the context, say \"I don't know\".\n\n"
        "Context:\n{context}\n\n"
        "Question: {question}\n"
        "Helpful Answer:"
    ),
)


def run_agent(queries):

    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)

    review_file = data_dir / "review.json"

    if not review_file.exists():
        payload = {
            "reviews": [
                {"text": "Great hotel, excellent service and comfortable rooms."},
                {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
                {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
                {"text": "Average hotel. The room was okay, but nothing special."},
                {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
                {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
                {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
                {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
                {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
                {"text": "Not a bad hotel, but there are better options available in the area."}
            ]
        }
        review_file.write_text(json.dumps(payload, indent=2))

    # Enable to save to disk & reuse the model (for repeated queries on the same data)
    PERSIST = False

    embeddings = OpenAIEmbeddings()

    if PERSIST and os.path.exists("persist"):
        print("Reusing index...\n")
        vectorstore = Chroma(persist_directory="persist", embedding_function=embeddings)
        # How many reviews are actually stored (so we can size the retriever).
        total_docs = len(vectorstore.get()["ids"])
    else:
        # The loader is correct: jq_schema=".reviews[]" iterates every review
        # object and content_key="text" extracts the review text, producing
        # exactly one Document per review (10 in total for this file).
        loader = JSONLoader(
            file_path=str(review_file),
            jq_schema=".reviews[]",
            content_key="text",
        )
        documents = loader.load()
        total_docs = len(documents)

        # Build the vector store directly (instead of VectorstoreIndexCreator)
        # so we know the exact corpus size to use as the retriever's k.
        if PERSIST:
            vectorstore = Chroma.from_documents(
                documents, embedding=embeddings, persist_directory="persist"
            )
            vectorstore.persist()
        else:
            vectorstore = Chroma.from_documents(documents, embedding=embeddings)

    # >>> The actual bug <<<
    # index.vectorstore.as_retriever() defaults to k=4, so the chain only ever
    # saw 4 of the 10 reviews. Any question like "how many reviews are there?"
    # or "what do the reviews say?" was answered from an arbitrary subset,
    # producing wrong counts and wrong/incomplete reviews.
    # Fix: retrieve the whole (small) corpus so the LLM sees every review.
    retriever = vectorstore.as_retriever(
        search_type="similarity",
        search_kwargs={"k": max(total_docs, 1)},
    )

    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model="gpt-3.5-turbo", temperature=0),
        retriever=retriever,
        chain_type="stuff",
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
        return_source_documents=True,
    )

    chat_history = []
    results = []
    for query in queries:
        if query in ['quit', 'q', 'exit']:
            sys.exit()
        result = chain({"question": query, "chat_history": chat_history})
        print(result['answer'])
        # Show which reviews were actually retrieved so the answer can be verified.
        for doc in result.get('source_documents', []):
            print(f"  - retrieved: {doc.page_content}")
        results.append(result['answer'])
        chat_history.append((query, result['answer']))
    return results
