from dotenv import load_dotenv
from pathlib import Path
import json
import os
import sys

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.document_loaders import JSONLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.indexes import VectorstoreIndexCreator
from langchain.indexes.vectorstore import VectorStoreIndexWrapper
from langchain.prompts import PromptTemplate
from langchain.vectorstores import Chroma

load_dotenv()

# Number of documents the retriever must fetch per question.
# LangChain's default is k=4, which meant the LLM only ever saw 4 of the 10
# reviews -> wrong review counts and wrong/missing reviews.
# 10 covers every review in data/review.json.
DEFAULT_K = 10

# Strict, grounded QA prompt: forces the model to answer only from the
# retrieved reviews and to actually count them instead of guessing.
QA_PROMPT = PromptTemplate(
    input_variables=["context", "question"],
    template=(
        "You are an assistant that answers questions about hotel reviews.\n"
        "Every numbered document in the context below is exactly one review.\n"
        "Answer the question at the end using ONLY the context below.\n"
        "If the question asks how many reviews match something, carefully "
        "count the matching reviews in the context. Do not guess.\n"
        "If the answer is not contained in the context, say you don't know.\n"
        "Never invent reviews, numbers or facts that are not in the context.\n\n"
        "Context:\n{context}\n\n"
        "Question: {question}\n"
        "Helpful Answer:"
    ),
)


def run_agent(queries):

  data_dir = Path("data")
  data_dir.mkdir(exist_ok=True)

  review_file = data_dir / "review.json"

  if not review_file.exists():
      payload = {
          "reviews": [
              {"text": "Great hotel, excellent service and comfortable rooms."},
              {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
              {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
              {"text": "Average hotel. The room was okay, but nothing special."},
              {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
              {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
              {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
              {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
              {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
              {"text": "Not a bad hotel, but there are better options available in the area."}
          ]
      }
      review_file.write_text(json.dumps(payload, indent=2))

  # Enable to save to disk & reuse the model (for repeated queries on the same data)
  PERSIST = False

  # How many reviews live in the JSON file.  The JSONLoader below produces
  # exactly one Document per element of .reviews[], so this equals the number
  # of documents that end up in the vector store.
  try:
      n_reviews = len(json.loads(review_file.read_text())["reviews"])
  except Exception:
      n_reviews = DEFAULT_K

  if PERSIST and os.path.exists("persist"):
    print("Reusing index...\n")
    vectorstore = Chroma(persist_directory="persist", embedding_function=OpenAIEmbeddings())
    index = VectorStoreIndexWrapper(vectorstore=vectorstore)
  else:

    # jq_schema=".reviews[]" iterates over every review object and
    # content_key="text" extracts its text, so each review becomes its own
    # Document (with source + seq_num metadata) and all 10 get indexed.
    loader = JSONLoader(
        file_path=str(review_file),
        jq_schema=".reviews[]",
        content_key="text",
    )

    if PERSIST:
      index = VectorstoreIndexCreator(vectorstore_kwargs={"persist_directory": "persist"}).from_loaders([loader])
    else:
      index = VectorstoreIndexCreator().from_loaders([loader])

  # >>> THE FIX <<<
  # The default retriever returns only the top k=4 documents, so the LLM saw
  # just 4 of the 10 reviews and answered count/list questions from an
  # incomplete context.  Retrieve every indexed review instead.
  retriever = index.vectorstore.as_retriever(
      search_type="similarity",
      search_kwargs={"k": max(DEFAULT_K, n_reviews)},
  )

  # temperature=0 keeps the answers deterministic and reproducible.
  llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)

  try:
    chain = ConversationalRetrievalChain.from_llm(
        llm=llm,
        retriever=retriever,
        return_source_documents=True,
        combine_docs_chain_kwargs={"prompt": QA_PROMPT},
    )
  except TypeError:
    # Older LangChain releases name this parameter `chain_type_kwargs`.
    chain = ConversationalRetrievalChain.from_llm(
        llm=llm,
        retriever=retriever,
        return_source_documents=True,
        chain_type_kwargs={"prompt": QA_PROMPT},
    )

  chat_history = []
  results = []
  for question in queries:
    if question in ['quit', 'q', 'exit']:
      sys.exit()
    result = chain({"question": question, "chat_history": chat_history})
    print(result['answer'])

    # Echo which reviews the answer was based on (seq_num metadata comes from
    # JSONLoader), so "wrong review" answers are easy to spot and verify.
    source_docs = result.get('source_documents') or []
    if source_docs:
        ids = ", ".join(
            "#{}".format(d.metadata.get('seq_num', i + 1))
            for i, d in enumerate(source_docs)
        )
        print("[based on {} retrieved review(s): {}]".format(len(source_docs), ids))

    results.append(result['answer'])
    chat_history.append((question, result['answer']))
  return results
