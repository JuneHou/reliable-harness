from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_debug
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# In LangChain 0.0.345, `set_verbose(True)` (and `verbose=True`) no longer shows
# intermediate steps for chains built with LCEL (the `|` operator creates a
# RunnableSequence, and verbose mode only logs chain start/end events for the
# legacy `Chain` subclasses). To see the inputs and outputs of EVERY step of the
# chain (prompt formatting, LLM call, output parsing), enable the global debug
# flag instead. It makes every component with callback support print its raw
# inputs and outputs:
set_debug(True)


def run_agent(topic: str) -> str:
    model = ChatOpenAI(openai_api_key=api_key)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)
    chain = prompt | model | output_parser
    # With set_debug(True), invoking the chain now logs every intermediate step.
    # (Alternatively, a per-call handler can be used instead of the global flag:
    #   from langchain.callbacks import ConsoleCallbackHandler
    #   chain.invoke({"topic": topic}, config={"callbacks": [ConsoleCallbackHandler()]})
    # )
    return chain.invoke({"topic": topic})


if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
