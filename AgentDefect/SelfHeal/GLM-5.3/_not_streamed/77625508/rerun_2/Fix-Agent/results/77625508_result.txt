from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_debug
from langchain.callbacks import StdOutCallbackHandler
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# set_verbose(True) / verbose=True do NOT produce logging for LCEL chains
# (prompt | model | parser). For LCEL, enable global debug logging instead,
# which prints every intermediate step of the RunnableSequence.
set_debug(True)


def run_agent(topic: str) -> str:
    model = ChatOpenAI(openai_api_key=api_key)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)
    chain = prompt | model | output_parser

    # Per-invocation (request-level) callbacks: pass the StdOutCallbackHandler
    # through the `config` argument so the chain's intermediate steps
    # (prompt render, LLM call, output parsing) are printed to stdout.
    return chain.invoke(
        {"topic": topic},
        config={"callbacks": [StdOutCallbackHandler()]},
    )


if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
