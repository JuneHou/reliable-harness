from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer
import os

MODEL_NAME = "BAAI/bge-m3"
CACHE_DIR = "./embeddings"


def build_tokenizer():
    """Load the exact tokenizer that the BGE-M3 embedding model uses."""
    os.makedirs(CACHE_DIR, exist_ok=True)
    return AutoTokenizer.from_pretrained(MODEL_NAME, cache_folder=CACHE_DIR)


def make_tokenizer_callable(hf_tokenizer):
    """
    Wrap a HuggingFace tokenizer into the shape LlamaIndex's
    TokenCountingHandler expects: a callable that takes a str and returns a
    list (the handler applies len() to the result).

    The raw HF tokenizer object cannot be handed over directly:
    tokenizer(text) returns a BatchEncoding (a dict), so len() would return
    the number of dict keys instead of the number of tokens.
    """

    def _tokenize(text: str) -> list:
        return hf_tokenizer(text)["input_ids"]

    return _tokenize


def replicate_embedding_preprocessing(text: str) -> str:
    """
    Reproduce the pre-processing LlamaIndex applies to a text before the
    TokenCountingHandler counts its embedding tokens:

      1. BaseEmbedding.get_text_embedding() emits a CBEventType.EMBEDDING
         event with the payload {EventPayload.CHUNKS: [text]} -- i.e. the
         text is carried inside a *list*.
      2. TokenCountingHandler.on_embedding_event_end() then counts the tokens
         of str(payload.get(EventPayload.CHUNKS)) -- the Python repr of that
         list, which wraps the text in "['" ... "']".

    Consequently a standalone tokenizer only produces the same number when it
    tokenizes this very string (brackets and quotes included).
    """
    return str([text])  # identical to str(payload.get(EventPayload.CHUNKS))


def create_embedding_model(_chunk_size=None, callback_manager=None):
    print('loading embeddings...')
    os.makedirs(CACHE_DIR, exist_ok=True)
    _embed_model = HuggingFaceEmbedding(
        model_name=MODEL_NAME,
        max_length=_chunk_size,
        cache_folder=CACHE_DIR,
        # Attach the callback manager (holding the TokenCountingHandler)
        # directly to the embedding model so that the EMBEDDING events it
        # emits are actually seen by the token counter. In the original code
        # the model was built before Settings.callback_manager was replaced,
        # so it kept the default callback manager and the counter could miss
        # the embedding events entirely.
        callback_manager=callback_manager or Settings.callback_manager,
    )
    print('embeddings loaded')
    return _embed_model


def run_agent(text: str):
    chunk_size = 512

    tokenizer = build_tokenizer()
    tokenize = make_tokenizer_callable(tokenizer)

    # Fix #1: TokenCountingHandler counts with whatever tokenizer callable it
    # is given. By default it falls back to the *global* tokenizer, which
    # defaults to tiktoken's cl100k (gpt-3.5-turbo) -- a completely different
    # vocabulary than BGE-M3. That was one of the two reasons the original
    # numbers never matched. Give it the BGE-M3 tokenizer instead.
    # (An equivalent alternative would be set_global_tokenizer(tokenizer.encode).)
    token_counter = TokenCountingHandler(tokenizer=tokenize)
    callback_manager = CallbackManager([token_counter])

    # Fix #2: register the callback manager BEFORE creating any component, so
    # that everything constructed afterwards picks it up.
    Settings.callback_manager = callback_manager

    embed_model = create_embedding_model(
        _chunk_size=chunk_size, callback_manager=callback_manager
    )
    Settings.embed_model = embed_model

    # Cheap path (this is the whole point): predict the reported token count
    # WITHOUT running the expensive embedding -- the same tokenizer, and the
    # same pre-processing (the list-repr string) that the handler applies to
    # the text before counting.
    chunk_str = replicate_embedding_preprocessing(text)
    token_count = len(tokenize(chunk_str))
    print(f"(the string that actually gets tokenized is: {chunk_str})")

    # Expensive path, kept only to verify that the prediction is correct.
    embedding_vector = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    return embedding_tokens, token_count


if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
    print(f"The tokenizer broke the text into {token_count} tokens")
    assert embedding_tokens == token_count, (
        f"Token counts still do not match: {embedding_tokens} != {token_count}"
    )
    print("Success: both token counts are identical.")
