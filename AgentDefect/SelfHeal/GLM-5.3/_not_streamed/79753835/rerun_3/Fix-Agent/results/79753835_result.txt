import os
from typing import Any, Dict, List, Optional

from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from transformers import AutoTokenizer

MODEL_NAME = "BAAI/bge-m3"
CACHE_DIR = "./embeddings"


def create_embedding_model(_chunk_size: Optional[int] = None) -> HuggingFaceEmbedding:
    print("loading embeddings...")
    os.makedirs(CACHE_DIR, exist_ok=True)
    _embed_model = HuggingFaceEmbedding(
        model_name=MODEL_NAME,
        max_length=_chunk_size,
        cache_folder=CACHE_DIR,
    )
    print("embeddings loaded")
    return _embed_model


def get_tokenizer() -> Any:
    """Load the exact tokenizer HuggingFaceEmbedding uses for BGE-M3."""
    return AutoTokenizer.from_pretrained(MODEL_NAME, cache_folder=CACHE_DIR)


def _tokenize_like_embedding_model(
    tokenizer: Any,
    text: str,
    max_length: Optional[int] = None,
) -> List[int]:
    """
    Tokenize exactly the way HuggingFaceEmbedding feeds text to BGE-M3.

    BGE-M3 applies no hidden pre-processing of its own.  The only things that
    happen to the raw string are what the tokenizer itself does:

        tokenizer([text], padding=True, max_length=max_length,
                  truncation=True, return_tensors="pt")

      * the model's own special tokens (<s> ... </s>) that the tokenizer adds
        by default (add_special_tokens=True), and
      * truncation to max_length.

    Padding only pads a batch up to its longest member, so it never changes
    the token count of a single string.  (An instruction prefix would only be
    prepended for queries if query_instruction/text_instruction were set --
    they are empty by default for this model.)
    """
    encode_kwargs: Dict[str, Any] = {"add_special_tokens": True}
    if max_length is not None:
        encode_kwargs["max_length"] = max_length
        encode_kwargs["truncation"] = True
    return tokenizer(text, **encode_kwargs)["input_ids"]


def count_tokens_without_embedding(
    tokenizer: Any,
    text: str,
    max_length: Optional[int] = None,
) -> int:
    """Number of tokens BGE-M3 will see for `text` -- without embedding it."""
    return len(_tokenize_like_embedding_model(tokenizer, text, max_length))


def run_agent(text: str):
    chunk_size = 512

    tokenizer = get_tokenizer()

    # ------------------------------------------------------------------
    # Why the two numbers did not match before:
    #
    # 1) TokenCountingHandler does NOT use the embedding model's tokenizer.
    #    When no tokenizer is supplied it silently falls back to llama_index's
    #    default, i.e. tiktoken's "gpt-3.5-turbo" (cl100k_base) encoding,
    #    which tokenizes text completely differently from BGE-M3's
    #    XLM-RoBERTa tokenizer.  The handler was therefore counting OpenAI
    #    tokens, not BGE-M3 tokens -- there is no special pre-processing
    #    inside BGE-M3 to reproduce.
    #
    # 2) The callback manager holding the token counter was never attached to
    #    the embedding model.  The model captures its callback manager when it
    #    is constructed, so assigning Settings.callback_manager *after* the
    #    model was built left the model dispatching the EMBEDDING event to an
    #    empty default manager, and the counter never saw anything.
    #
    # The fix: hand the handler the very same tokenizer the embedding model
    # uses (a tiktoken-style callable: string in -> list of token ids out),
    # and make sure the embedding model dispatches its events to the manager
    # that contains the handler.
    # ------------------------------------------------------------------
    def bge_m3_token_counter(value: str) -> List[int]:
        # NOTE: the list of token ids must be returned, not the BatchEncoding
        # itself -- len() of a BatchEncoding counts its dict keys (always 2),
        # not the tokens.
        return _tokenize_like_embedding_model(tokenizer, value, chunk_size)

    token_counter = TokenCountingHandler(tokenizer=bge_m3_token_counter)
    callback_manager = CallbackManager([token_counter])

    # Register the callback manager *before* the model is built ...
    Settings.callback_manager = callback_manager

    embed_model = create_embedding_model(_chunk_size=chunk_size)
    # ... and attach it to the embedding model itself, so the EMBEDDING
    # event actually reaches the token counter.
    embed_model.callback_manager = callback_manager
    Settings.embed_model = embed_model

    embedding_vector = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Stand-alone count that replicates the model's tokenization, obtained
    # without running the expensive embedding.
    token_count = count_tokens_without_embedding(tokenizer, text, chunk_size)

    return embedding_tokens, token_count


if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
    print(f"The tokenizer broke the text into {token_count} tokens")
    print(f"The counts match: {embedding_tokens == token_count}")
