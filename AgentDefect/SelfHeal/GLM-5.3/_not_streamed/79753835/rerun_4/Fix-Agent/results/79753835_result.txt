import os
from typing import Callable, List

from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from transformers import AutoTokenizer

MODEL_NAME = "BAAI/bge-m3"
CACHE_DIR = "./embeddings"


def create_embedding_model(_chunk_size=None):
    """Load the BGE-M3 embedding model (from the local cache when available)."""
    print("loading embeddings...")
    os.makedirs(CACHE_DIR, exist_ok=True)
    _embed_model = HuggingFaceEmbedding(
        model_name=MODEL_NAME,
        max_length=_chunk_size,
        cache_folder=CACHE_DIR,
    )
    print("embeddings loaded")
    return _embed_model


def build_model_tokenizer(max_length: int) -> Callable[[str], List[int]]:
    """Create a callable that reproduces the pre-processing BGE-M3 applies to a string.

    HuggingFaceEmbedding runs the text through sentence-transformers, which:

      1. tokenizes it with the model's own XLM-RoBERTa tokenizer,
      2. adds the special tokens <s> ... </s> (add_special_tokens=True),
      3. truncates the result to the model's max_length / max_seq_length
         (the `max_length` handed to HuggingFaceEmbedding).

    Calling this callable on a string therefore yields exactly the token ids
    (and thus the token count) the embedding model works with, WITHOUT having
    to run the computationally expensive embedding itself.
    """
    # Reuse the locally cached tokenizer files (./embeddings/models--BAAI--bge-m3)
    tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME, cache_folder=CACHE_DIR)

    def tokenize(text: str) -> List[int]:
        return tokenizer(
            text,
            add_special_tokens=True,  # <s> ... </s> are added by the model
            truncation=True,          # input is cut off at max_length
            max_length=max_length,
        )["input_ids"]

    return tokenize


def run_agent(text: str):
    chunk_size = 512
    embed_model = create_embedding_model(_chunk_size=chunk_size)

    # A tokenizer that mimics the pre-processing of the embedding model.
    model_tokenizer = build_model_tokenizer(max_length=chunk_size)

    # ------------------------------------------------------------------
    # FIX for the mismatch:
    #
    # TokenCountingHandler does NOT count tokens with the tokenizer of the
    # embedding model.  When no tokenizer is given it falls back to
    # LlamaIndex's *global* tokenizer, whose default is tiktoken's
    # cl100k_base (gpt-3.5-turbo) encoding.  That is why
    # `total_embedding_token_count` disagreed with the BGE-M3 tokenizer.
    #
    # Hand the handler the BGE-M3 tokenizer (as a callable returning the
    # token ids, including the special tokens) so that both numbers describe
    # exactly the same tokens the embedding model actually processes.
    # ------------------------------------------------------------------
    token_counter = TokenCountingHandler(tokenizer=model_tokenizer)
    callback_manager = CallbackManager([token_counter])

    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager
    # Explicitly route the embedding events to the handler as well.
    embed_model.callback_manager = callback_manager

    embedding_vector = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Standalone count -- no embedding model required, just the tokenizer with
    # the same pre-processing (special tokens + truncation to max_length).
    token_count = len(model_tokenizer(text))

    return embedding_tokens, token_count


if __name__ == "__main__":
