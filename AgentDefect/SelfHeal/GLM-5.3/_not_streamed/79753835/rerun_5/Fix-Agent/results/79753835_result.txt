from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer
import os

MODEL_NAME = "BAAI/bge-m3"
CACHE_DIR = "./embeddings"


def create_embedding_model(_chunk_size=None):
    """Load the (expensive) BGE-M3 embedding model."""
    print('loading embeddings...')
    os.makedirs(CACHE_DIR, exist_ok=True)
    _embed_model = HuggingFaceEmbedding(
        model_name=MODEL_NAME,
        max_length=_chunk_size,
        cache_folder=CACHE_DIR,
    )
    print('embeddings loaded')
    return _embed_model


def create_tokenizer():
    """Load the tokenizer that BGE-M3 actually uses (XLM-RoBERTa based).

    The tokenizer is cheap to load, which lets us count tokens WITHOUT
    running the expensive embedding model.
    """
    return AutoTokenizer.from_pretrained(MODEL_NAME, cache_folder=CACHE_DIR)


def model_token_count(tokenizer, text: str, max_length: int = 512) -> int:
    """Cheaply reproduce the number of tokens BGE-M3 consumes for `text`.

    The "pre-processing" the model applies before it sees the text is just
    the tokenization that llama-index's HuggingFaceEmbedding performs:

      * the model's own (XLM-R) tokenizer -- NOT tiktoken's "cl100k_base",
        which is only llama-index's *default* token-counting tokenizer,
      * the special tokens <s> ... </s> the tokenizer adds automatically
        (this is the "+2" that made the naive comparison look off),
      * truncation to the model's `max_length`.

    No instruction/query prefix is prepended by default, so replicating the
    points above is all that is needed to match the reported count.
    """
    return len(
        tokenizer(text, max_length=max_length, truncation=True)["input_ids"]
    )


def run_agent(text: str):
    chunk_size = 512
    embed_model = create_embedding_model(_chunk_size=chunk_size)

    # Cheap tokenizer, loaded from the very same model (and cache folder)
    # that the embedding model uses internally.
    tokenizer = create_tokenizer()

    # ------------------------------------------------------------------------
    # FIX: TokenCountingHandler() does NOT count with the embedding model's
    # tokenizer. By default it uses llama-index's *global* tokenizer, which
    # is tiktoken's "cl100k_base" (the gpt-3.5-turbo tokenizer). That is why
    # `total_embedding_token_count` never matched the BGE-M3 tokenizer count.
    #
    # Give the handler the embedding model's own tokenizer. The handler
    # computes `len(tokenizer(text))`, so we wrap the HF tokenizer to return
    # the list of input ids (a raw BatchEncoding would not len() correctly).
    # ------------------------------------------------------------------------
    def counting_tokenizer(t: str):
        # Mirror the exact preprocessing of HuggingFaceEmbedding._embed():
        # truncation to max_length is applied at tokenization time.
        return tokenizer(t, max_length=chunk_size, truncation=True)["input_ids"]

    token_counter = TokenCountingHandler(tokenizer=counting_tokenizer)
    callback_manager = CallbackManager([token_counter])

    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager
    # Make sure the embedding model actually fires its events into our
    # handler, independent of how the installed llama-index version
    # propagates Settings.callback_manager.
    embed_model.callback_manager = callback_manager

    token_counter.reset_counts()
    embedding_vector = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Standalone token count: same tokenizer + same preprocessing as the
    # embedding model, computed WITHOUT running the expensive embedding.
    token_count = model_token_count(tokenizer, text, max_length=chunk_size)

    return embedding_tokens, token_count


if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
    print(f"The tokenizer broke the text into {token_count} tokens")
    if embedding_tokens == token_count:
        print("Success: both token counts match.")
    else:
        print("Warning: token counts still differ.")
