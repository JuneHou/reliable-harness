from tqdm import tqdm
from langchain.docstore.document import Document
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from dotenv import load_dotenv
load_dotenv()

def run_agent():

    embeddings = OpenAIEmbeddings()

    for db_collection_name in tqdm(["class1-sub2-chap3", "class2-sub3-chap4"]):
        documents = []
        doc_ids = []

        for doc_index in range(3):
            cl, sub, chap = db_collection_name.split("-")
            content = f"This is {db_collection_name}-doc{doc_index}"
            doc = Document(page_content=content, metadata={"chunk_num": doc_index, "chapter":chap, "class":cl, "subject":sub})
            documents.append(doc)
            doc_ids.append(str(doc_index))


        db = Chroma.from_documents(
            collection_name=db_collection_name,
            documents=documents, ids=doc_ids,
            embedding=embeddings, 
            persist_directory="./data")
        
        db.persist()
        
    initial_content = "This is an initial document content"
    document_id = "doc1"


    original_doc = Document(page_content=initial_content, metadata={"page": "0"})

    
    new_db = Chroma.from_documents(
        collection_name="test_collection",
        documents=[original_doc],
        embedding=embeddings,  
        ids=[document_id],
    )

    # ------------------------------------------------------------------
    # FIX:
    # Chroma's `collection.get()` does NOT return the stored embeddings
    # by default. Its default `include` value is ["documents", "metadatas"],
    # which is why `embeddings` previously came back as `None` even though
    # the vectors really are computed and persisted on disk.
    #
    # To view/retrieve the actual embeddings, they must be explicitly
    # requested through the `include` parameter:
    #
    #     collection.get(include=["embeddings", "documents", "metadatas"])
    #
    # The same applies to `collection.query()`, where the default include
    # is ["documents", "metadatas", "distances"].
    # ------------------------------------------------------------------
    result = db._collection.get(include=["embeddings", "documents", "metadatas"])

    # The same technique works for any other collection, e.g. the newly
    # created "test_collection":
    test_result = new_db._collection.get(include=["embeddings", "documents", "metadatas"])

    # And it also works when reloading a persisted collection from disk,
    # proving the embeddings were in fact stored there:
    reloaded_db = Chroma(
        collection_name=db._collection.name,
        embedding_function=embeddings,
        persist_directory="./data",
    )
    reloaded_result = reloaded_db._collection.get(
        include=["embeddings", "documents", "metadatas"]
    )

    # Show that real (non-None) embedding vectors are now returned.
    print(f"\nEmbeddings retrieved for '{db._collection.name}': "
          f"{result.get('embeddings') is not None} "
          f"({len(result.get('embeddings') or [])} vectors)")
    print(f"Embeddings retrieved for 'test_collection': "
          f"{test_result.get('embeddings') is not None}")
    print(f"Embeddings retrieved after reloading from './data': "
          f"{reloaded_result.get('embeddings') is not None}\n")

    return result


if __name__ == "__main__":
    result = run_agent()
    print(result)
