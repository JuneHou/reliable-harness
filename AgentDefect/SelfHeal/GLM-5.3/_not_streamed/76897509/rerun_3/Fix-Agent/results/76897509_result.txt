import os
import random
import warnings
from typing import Any, Dict, List

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.vectorstores.base import VectorStoreRetriever
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache
from langchain.schema import Document

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]


def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


class FilterConversationalRetrievalChain(ConversationalRetrievalChain):
    """
    ConversationalRetrievalChain that supports a per-query metadata filter.

    The stock ConversationalRetrievalChain only consumes `question` and
    `chat_history` from its inputs; any other key (e.g. `filter`) is silently
    ignored, so the retriever searches the whole vector store and the answer
    may be built from the wrong document.

    This subclass reads the optional `filter` key from the chain inputs and
    injects it into the retriever's `search_kwargs` before retrieval. The
    VectorStoreRetriever forwards `search_kwargs` to the vector store's
    similarity_search(query, k=..., filter=...) call, which Chroma turns into
    a `where` clause. This way the metadata filter is applied dynamically at
    query time, without creating a separate chain/retriever per filter.
    """

    def _get_docs(self, inputs: Dict[str, Any]) -> List[Document]:
        # Pull the filter out of the chain inputs (it is not a declared input
        # key, so the base chain would otherwise just ignore it).
        metadata_filter = inputs.pop("filter", None)

        # Apply the filter to the retriever used by this chain.
        search_kwargs = getattr(self.retriever, "search_kwargs", None)
        if isinstance(search_kwargs, dict):
            # Drop any filter left over from a previous query, then apply the
            # one for this query (no filter => unfiltered search).
            search_kwargs.pop("filter", None)
            if metadata_filter:
                search_kwargs["filter"] = metadata_filter
        elif metadata_filter:
            warnings.warn(
                "A 'filter' was passed but the retriever does not expose "
                "'search_kwargs', so the query-time filter cannot be applied."
            )

        return super()._get_docs(inputs)


def run_agent(query, pdf_paths):
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            text.metadata["source"] = pdf_path

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    qa = FilterConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        VectorStoreRetriever(vectorstore=db),
        verbose=True,
        return_source_documents=True)

    filter = {'source': 'paper2.pdf'}  # it means the information from paper2.pdf should be used to answer the query.
    # if it answers from paper2.pdf, then the filtering is working correctly.

    result = qa({"question": query, "chat_history": [], "filter": filter})

    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    result = run_agent(query, pdf_paths)

    print(result['answer'])
