import os
import random
from typing import Any, Dict, Optional

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.vectorstores.base import VectorStoreRetriever
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache
from langchain.callbacks.manager import CallbackManagerForChainRun

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]


def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


class CustomConversationalRetrievalChain(ConversationalRetrievalChain):
    """A ConversationalRetrievalChain that supports a metadata filter at query time.

    Root cause of the original bug:
    The stock ConversationalRetrievalChain only consumes the "question" and
    "chat_history" input keys. Any other key in the input dict (such as
    "filter") is silently ignored: the chain calls
    ``retriever.get_relevant_documents(question)`` with no filter, so
    VectorStoreRetriever ran ``Chroma.similarity_search(query)`` over ALL
    ingested chunks (paper1.pdf AND paper2.pdf). The answer was therefore not
    restricted to paper2.pdf.

    Fix:
    This subclass reads the optional "filter" key from the chain inputs and
    forwards it into the retriever's ``search_kwargs``. VectorStoreRetriever
    passes ``**search_kwargs`` straight through to
    ``Chroma.similarity_search(query, filter=...)``, so only documents whose
    metadata matches the filter are retrieved and used to build the answer.
    Because the filter is taken from the inputs on every call, a single chain
    instance can be reused for any number of document filters (no need to
    rebuild the chain per filter).
    """

    def _call(
        self,
        inputs: Dict[str, Any],
        run_manager: Optional[CallbackManagerForChainRun] = None,
    ) -> Dict[str, Any]:
        # 1) Pull the per-query metadata filter out of the chain inputs.
        metadata_filter = inputs.get("filter")

        # Strip the filter so it is never fed to the question generator /
        # combine-docs chains (they only expect "question" and "chat_history").
        chain_inputs = {k: v for k, v in inputs.items() if k != "filter"}

        # 2) Apply the filter to the retriever before retrieval happens.
        if hasattr(self.retriever, "search_kwargs"):
            if metadata_filter is not None:
                # Applied per call: only documents whose metadata matches the
                # filter are retrieved from the vector store.
                self.retriever.search_kwargs["filter"] = metadata_filter
            else:
                # No filter supplied for this query -> drop any stale filter.
                self.retriever.search_kwargs.pop("filter", None)

        # 3) Run the chain as usual; retrieval is now restricted to the
        #    filtered documents.
        return super()._call(chain_inputs, run_manager)


def run_agent(query, pdf_paths):
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            text.metadata["source"] = pdf_path

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    qa = CustomConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        VectorStoreRetriever(vectorstore=db),
        verbose=True,
        return_source_documents=True)

    filter = {'source': 'paper2.pdf'}  # it means the information from paper2.pdf should be used to answer the query.
    # if it answers from paper2.pdf, then the filtering is working correctly.

    # The "filter" key is now honored at query time: only chunks whose
    # metadata source == "paper2.pdf" are retrieved and used to answer.
    result = qa({"question": query, "chat_history": [], "filter": filter})

    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    result = run_agent(query, pdf_paths)

    print(result['answer'])
