import os
import random
from typing import Any, Dict, List, Optional

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.schema import Document
from langchain.vectorstores.base import VectorStoreRetriever
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]

def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


class FilteredRetriever(VectorStoreRetriever):
    """A VectorStoreRetriever that applies a metadata filter at query time.

    ConversationalRetrievalChain only accepts "question" and "chat_history" as
    input keys, so any "filter" key passed in the chain input dict is silently
    ignored. Instead, the filter must be applied by the retriever itself.
    The `filter` attribute is mutable, so the SAME chain instance can be
    reused with a different document filter for every query.
    """

    filter: Optional[Dict[str, Any]] = None

    def _get_relevant_documents(self, query: str, run_manager=None) -> List[Document]:
        # Push the filter down to the vector store so only matching documents
        # are retrieved (e.g. only chunks whose metadata source == "paper2.pdf").
        search_kwargs = dict(self.search_kwargs or {})
        k = search_kwargs.pop("k", 4)
        if self.filter:
            search_kwargs["filter"] = self.filter

        docs = self.vectorstore.similarity_search(query, k=k, **search_kwargs)

        # Safety net: drop any document that does not satisfy the filter.
        if self.filter:
            docs = [
                doc
                for doc in docs
                if all(
                    doc.metadata.get(key) == value
                    for key, value in self.filter.items()
                )
            ]
        return docs


def run_agent(query, pdf_paths):
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            text.metadata["source"] = pdf_path

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    # Use the custom retriever so the filter can be changed per query
    # without rebuilding the chain or the vector store.
    retriever = FilteredRetriever(vectorstore=db, search_kwargs={"k": 4})

    qa = ConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever,
        verbose=True,
        return_source_documents=True)

    # The filter is applied at query time. Setting this attribute before each
    # call lets the same chain be reused for different documents.
    # It means the information from paper2.pdf should be used to answer the query.
    retriever.filter = {'source': 'paper2.pdf'}

    # NOTE: "filter" is NOT passed inside the chain input dict anymore;
    # ConversationalRetrievalChain would simply ignore it.
    result = qa({"question": query, "chat_history": []})

    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    result = run_agent(query, pdf_paths)

    print(result['answer'])

    # Verify the filtering is working correctly: if the answer is drawn from
    # paper2.pdf, every source document below must be "paper2.pdf".
    sources = {doc.metadata.get('source') for doc in result.get('source_documents', [])}
    print("Sources used:", sources)
