import json
import os


class CBEventType:
    """Callback event types (mirrors llama_index.core.callbacks.base.CBEventType)."""
    LLM = "llm"
    EMBEDDING = "embedding"


class EventPayload:
    """Payload keys for callback events (mirrors llama_index.core.callbacks.base.EventPayload)."""
    PROMPT = "prompt"
    COMPLETION = "completion"


class OpenAIFinetuneExample:
    """A single fine-tuning example: a user prompt paired with the assistant completion."""

    def __init__(self, prompt, completion):
        self.messages = [
            {"role": "user", "content": prompt},
            {"role": "assistant", "content": completion},
        ]

    def to_dict(self):
        return {"messages": self.messages}


class OpenAIFineTuningHandler:
    """Callback handler that collects LLM prompt/completion pairs and can persist them as JSONL."""

    def __init__(self):
        self._finetuning_events = []

    def on_event_start(self, cb_event_type, payload, **kwargs):
        # Only LLM end-events carry the prompt/completion pair we need.
        pass

    def on_event_end(self, cb_event_type, payload, **kwargs):
        if cb_event_type != CBEventType.LLM:
            return
        if payload is None:
            return
        prompt = payload.get(EventPayload.PROMPT)
        completion = payload.get(EventPayload.COMPLETION)
        # Precondition: only record an example when both sides of the pair exist.
        if not prompt or not completion:
            return
        self._finetuning_events.append(
            OpenAIFinetuneExample(prompt=prompt, completion=completion)
        )

    def save_finetuning_events(self, path):
        """Write all captured fine-tuning events to a JSONL file and report the real count."""
        with open(path, "w", encoding="utf-8") as f:
            for event in self._finetuning_events:
                f.write(json.dumps(event.to_dict()) + "\n")
        print(f"Wrote {len(self._finetuning_events)} examples to {path}")
        if len(self._finetuning_events) == 0:
            print(
                "Warning: no LLM events were captured. Verify the handler is attached "
                "to the LLM's callback manager and that LLM queries were executed."
            )


class CallbackManager:
    """Dispatches callback events to all registered handlers."""

    def __init__(self, handlers=None):
        self.handlers = handlers or []

    def on_event_start(self, cb_event_type, payload=None):
        for handler in self.handlers:
            if hasattr(handler, "on_event_start"):
                handler.on_event_start(cb_event_type, payload or {})

    def on_event_end(self, cb_event_type, payload=None):
        payload = payload or {}
        for handler in self.handlers:
            if hasattr(handler, "on_event_end"):
                handler.on_event_end(cb_event_type, payload)


class MockLLM:
    """Minimal LLM stand-in whose callback_manager fires LLM events."""

    def __init__(self, callback_manager):
        self.callback_manager = callback_manager

    def complete(self, prompt):
        completion = f"Answer to: {prompt}"
        self.callback_manager.on_event_start(CBEventType.LLM, {EventPayload.PROMPT: prompt})
        self.callback_manager.on_event_end(
            CBEventType.LLM,
            {EventPayload.PROMPT: prompt, EventPayload.COMPLETION: completion},
        )
        return completion


class QueryEngine:
    def __init__(self, similarity_top_k, llm):
        self.similarity_top_k = similarity_top_k
        self.llm = llm

    def query(self, question):
        # A real query engine invokes the LLM, which triggers LLM callback events
        # carrying PROMPT/COMPLETION payloads that the finetuning handler captures.
        return self.llm.complete(question)


class VectorStoreIndex:
    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex()

    def as_query_engine(self, similarity_top_k, llm):
        return QueryEngine(similarity_top_k=similarity_top_k, llm=llm)


class FineTuningExample:
    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate JSONL file for fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        self.llm = MockLLM(callback_manager=callback_manager)

        questions_path = f'{self.dataset_path}/train_questions.txt'
        # Precondition: the questions file must exist before we try to read it.
        if not os.path.exists(questions_path):
            raise FileNotFoundError(f"Questions file not found: {questions_path}")

        questions = []
        with open(questions_path, "r", encoding='utf-8') as f:
            for line in f:
                question = line.strip()
                if question:
                    questions.append(question)

        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                response = query_engine.query(question)
                print(f"Q: {question} -> {response}")
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            # Save whatever was captured, even if an error occurred mid-run.
            finetuning_handler.save_finetuning_events(
                f'{self.dataset_path}/finetuning_events.jsonl'
            )


def prepare_test_data(questions_path):
    os.makedirs(os.path.dirname(questions_path), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path):
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
