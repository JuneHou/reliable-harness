import json
import os


# --- Lightweight stand-ins for llama_index callback machinery -------------

class CBEventType:
    """Callback event types (only the one we need here)."""
    LLM = "llm"


class EventPayload:
    """Keys used in the payload dictionaries passed to handlers."""
    MESSAGES = "messages"


class OpenAIFineTuningHandler:
    """Callback handler that captures LLM interactions as fine-tuning events.

    The handler is registered with a CallbackManager and records every
    CBEventType.LLM interaction (the user query plus the LLM response).
    The JSONL file is written from the *collected* events, so the reported
    count reflects how many examples were actually captured.
    """

    def __init__(self):
        # Where captured events are accumulated (was missing entirely before).
        self._finetuning_events = []
        self._cur_event_msgs = None

    @property
    def finetuning_events(self):
        return self._finetuning_events

    def on_event_start(self, event_type, payload=None, **kwargs):
        """Remember the messages of the LLM call that is starting."""
        if event_type == CBEventType.LLM:
            self._cur_event_msgs = (payload or {}).get(EventPayload.MESSAGES)

    def on_event_end(self, event_type, payload=None, **kwargs):
        """Store the completed messages (query + response) as one example."""
        if event_type != CBEventType.LLM:
            return
        payload = payload or {}
        messages = payload.get(EventPayload.MESSAGES, self._cur_event_msgs)
        if messages:
            self._finetuning_events.append(list(messages))
        self._cur_event_msgs = None

    def save_finetuning_events(self, path):
        """Serialize every captured event to the given JSONL file."""
        with open(path, "w", encoding="utf-8") as f:
            for messages in self._finetuning_events:
                f.write(
                    json.dumps({"messages": messages}, ensure_ascii=False) + "\n"
                )
        print(f"Wrote {len(self._finetuning_events)} examples to {path}")


class CallbackManager:
    """Dispatches start/end events to every registered handler."""

    def __init__(self, handlers=None):
        self.handlers = list(handlers) if handlers else []

    def add_handler(self, handler):
        if handler not in self.handlers:
            self.handlers.append(handler)

    def on_event_start(self, event_type, payload=None):
        for handler in self.handlers:
            handler.on_event_start(event_type, payload)

    def on_event_end(self, event_type, payload=None):
        for handler in self.handlers:
            handler.on_event_end(event_type, payload)


class LLM:
    """Mock LLM whose calls are traced through a callback manager.

    A real LLM dispatches CBEventType.LLM events around each completion;
    the mock must do the same so the finetuning handler observes them.
    """

    def __init__(self, callback_manager):
        self.callback_manager = callback_manager

    def complete(self, question):
        payload = {EventPayload.MESSAGES: [{"role": "user", "content": question}]}
        self.callback_manager.on_event_start(CBEventType.LLM, payload)

        response = f"Answer to: {question}"

        payload = {
            EventPayload.MESSAGES: payload[EventPayload.MESSAGES]
            + [{"role": "assistant", "content": response}]
        }
        self.callback_manager.on_event_end(CBEventType.LLM, payload)
        return response


class QueryEngine:
    """Query engine that runs each question through the LLM."""

    def __init__(self, index, similarity_top_k, llm):
        self.index = index
        self.similarity_top_k = similarity_top_k
        self.llm = llm

    def query(self, question):
        return self.llm.complete(question)


class VectorStoreIndex:
    def __init__(self, documents=None):
        self.documents = documents or []

    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex(docs)

    def as_query_engine(self, similarity_top_k, llm):
        return QueryEngine(self, similarity_top_k, llm)


class FineTuningExample:

    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate JSONL file for fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        self.llm = LLM(callback_manager)

        questions = []
        with open(f'{self.dataset_path}/train_questions.txt', "r", encoding='utf-8') as f:
            for line in f:
                line = line.strip()
                if line:
                    questions.append(line)

        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                response = query_engine.query(question)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                f'{self.dataset_path}/finetuning_events.jsonl'
            )


def prepare_test_data(questions_path):
    os.makedirs(os.path.dirname(questions_path), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path):
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
