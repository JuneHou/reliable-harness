import json
import os


class CBEventType:
    """Event types dispatched through the callback manager (mirrors LlamaIndex)."""
    LLM = "llm"


class EventPayload:
    """Payload keys carried by callback events (mirrors LlamaIndex)."""
    MESSAGES = "messages"
    RESPONSE = "response"


class OpenAIFineTuningHandler:
    """Handler that captures LLM interactions as fine-tuning events.

    Fine-tuning events are recorded automatically inside on_event_end whenever
    an LLM event completes; save_finetuning_events then writes every collected
    event to the JSONL file and reports the real number of examples.
    """

    def __init__(self):
        self.finetuning_events = []

    def start_trace(self, trace_type=None):
        pass

    def on_event_start(self, event_type, payload=None):
        return None

    def on_event_end(self, event_type, payload=None):
        # OpenAIFineTuningHandler only captures LLM events
        if event_type != CBEventType.LLM:
            return
        payload = payload or {}
        messages = payload.get(EventPayload.MESSAGES)
        response = payload.get(EventPayload.RESPONSE)
        if not messages or response is None:
            return
        # Normalize messages into the OpenAI fine-tuning format
        normalized_messages = [
            {"role": message["role"], "content": message["content"]}
            for message in messages
        ]
        self.finetuning_events.append({"messages": normalized_messages})

    def add_finetuning_event(self, event):
        """Manually add a fine-tuning event."""
        self.finetuning_events.append(event)

    def get_finetuning_events(self):
        return self.finetuning_events

    def save_finetuning_events(self, path):
        """Write every collected fine-tuning event to a JSONL file."""
        count = 0
        with open(path, "w", encoding="utf-8") as f:
            for event in self.finetuning_events:
                f.write(json.dumps(event) + "\n")
                count += 1
        print(f"Wrote {count} examples to {path}")


class CallbackManager:
    """Dispatches callback events to all registered handlers."""

    def __init__(self, handlers):
        self.handlers = handlers

    def on_event_start(self, event_type, payload=None):
        for handler in self.handlers:
            handler.on_event_start(event_type, payload=payload)

    def on_event_end(self, event_type, payload=None):
        for handler in self.handlers:
            handler.on_event_end(event_type, payload=payload)


class LLM:
    """LLM whose calls are traced through the callback manager."""

    def __init__(self, callback_manager):
        self.callback_manager = callback_manager

    def complete(self, messages):
        # Emit start event
        self.callback_manager.on_event_start(
            CBEventType.LLM, payload={EventPayload.MESSAGES: messages}
        )
        # Produce a response for the given messages
        response = f"Answer to: {messages[-1]['content']}"
        # Emit end event carrying both messages and response so that
        # OpenAIFineTuningHandler.on_event_end records the interaction
        self.callback_manager.on_event_end(
            CBEventType.LLM,
            payload={
                EventPayload.MESSAGES: messages,
                EventPayload.RESPONSE: response,
            },
        )
        return response


class QueryEngine:
    """Query engine that runs each question through the LLM."""

    def __init__(self, index, similarity_top_k, llm):
        self.index = index
        self.similarity_top_k = similarity_top_k
        self.llm = llm

    def query(self, question):
        messages = [
            {"role": "system", "content": "You are a helpful assistant."},
            {"role": "user", "content": question},
        ]
        return self.llm.complete(messages)


class VectorStoreIndex:

    def __init__(self, documents=None, llm=None):
        self.documents = documents or []
        self.llm = llm

    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex(documents=docs)

    def as_query_engine(self, similarity_top_k, llm):
        # The LLM (with its callback manager) must be attached to the query
        # engine so LLM events reach the fine-tuning handler during queries.
        engine_llm = llm if llm is not None else self.llm
        return QueryEngine(index=self, similarity_top_k=similarity_top_k, llm=engine_llm)


class FineTuningExample:

    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate JSONL file for fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        self.llm = LLM(callback_manager)

        questions = []
        with open(f'{self.dataset_path}/train_questions.txt', "r", encoding='utf-8') as f:
            for line in f:
                line = line.strip()
                if line:
                    questions.append(line)

        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                response = query_engine.query(question)
                print(f"Q: {question} -> A: {response}")
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                f'{self.dataset_path}/finetuning_events.jsonl'
            )


def prepare_test_data(questions_path):
    os.makedirs(os.path.dirname(questions_path), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path):
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
