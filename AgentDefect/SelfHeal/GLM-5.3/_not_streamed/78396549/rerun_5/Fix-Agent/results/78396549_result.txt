import os

from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

load_dotenv()


def run_agent(prompt: str) -> str:
    api_key = os.getenv("OPENAI_API_KEY")
    llm = ChatOpenAI(openai_api_key=api_key)

    # NOTE: ChatOpenAI.invoke() returns an AIMessage object, NOT a plain
    # string. Returning the raw object makes the printed output look like:
    #     AIMessage(content='...', additional_kwargs={...}, ...)
    # instead of the expected answer text. The model's answer is stored in
    # the message's `content` attribute, so extract it before returning.
    response = llm.invoke(prompt)

    return response.content  # the actual text of the LLM's answer


if __name__ == "__main__":
    user_prompt = "How can LangSmith help with testing?"
    answer = run_agent(user_prompt)
    print("LLM response:", answer)
