import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

load_dotenv()


def run_agent(prompt: str) -> str:
    api_key = os.getenv("OPENAI_API_KEY")
    # Use the current `api_key` keyword and set the model explicitly.
    llm = ChatOpenAI(model="gpt-4o-mini", api_key=api_key)

    # `llm.invoke()` returns an AIMessage object, not a plain string.
    # Extract the actual text of the response via the `.content` attribute,
    # otherwise printing shows the whole AIMessage repr instead of the answer.
    response = llm.invoke(prompt)
    return response.content


if __name__ == "__main__":
    user_prompt = "How can LangSmith help with testing?"
    answer = run_agent(user_prompt)
    print("LLM response:", answer)
