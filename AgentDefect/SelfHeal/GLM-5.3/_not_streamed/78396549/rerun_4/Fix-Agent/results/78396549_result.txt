import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

load_dotenv()


def run_agent(prompt: str) -> str:
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise ValueError(
            "OPENAI_API_KEY not found. Make sure a .env file exists in the "
            "project root containing OPENAI_API_KEY=your-key-here, and that "
            "python-dotenv is installed."
        )

    llm = ChatOpenAI(openai_api_key=api_key)

    # NOTE: llm.invoke() returns an AIMessage object, NOT a plain string.
    response = llm.invoke(prompt)

    # Extract the actual model output (the plain-text answer) from the
    # AIMessage object, matching what the getting-started guide displays.
    return response.content


if __name__ == "__main__":
    user_prompt = "How can LangSmith help with testing?"
    answer = run_agent(user_prompt)
    print("LLM response:", answer)
