import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

load_dotenv()


def run_agent(prompt: str) -> str:
    """Invoke a ChatOpenAI model and return the assistant's answer as plain text.

    Note: ``llm.invoke(prompt)`` in modern LangChain returns an ``AIMessage``
    object, not a string. Printing that object dumps a large metadata-laden
    repr (``AIMessage(content=..., additional_kwargs={'response_headers': ...},
    response_metadata=..., usage_metadata=..., id='run-...')``), which makes
    the actual answer effectively invisible to the user. To see the response,
    extract the text with ``response.content`` (LangChain's equivalent of the
    raw SDK's ``choices[0].message.content``).
    """
    api_key = os.getenv("OPENAI_API_KEY")
    llm = ChatOpenAI(openai_api_key=api_key)

    response = llm.invoke(prompt)

    # Correctly access the message payload instead of the AIMessage object.
    # response.content is the plain-text answer; response.content is always
    # present on chat-model output messages.
    return response.content


if __name__ == "__main__":
    user_prompt = "How can LangSmith help with testing?"
    answer = run_agent(user_prompt)
    # `answer` is now a plain string, so this prints the actual model output.
    print("LLM response:", answer)
