import os
import uuid
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv
dotenv.load_dotenv()

NUM_UNIQUE_DOCS = 6


def run_agent(query):
    loader = UnstructuredFileLoader(
        'paper2.pdf'
    )
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    # FIX: Chroma.from_documents() uses a shared chroma client and a default
    # collection name. Re-running this function therefore kept *adding* the
    # same chunks to the same collection, so the retriever returned duplicate
    # copies of the same document instead of unique ones. Using a unique
    # collection name for every run guarantees the vector store is built
    # fresh each time and contains no previously accumulated documents.
    collection_name = f"paper2_{uuid.uuid4().hex}"

    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
        collection_name=collection_name,
    )

    retriever = vectorstore.as_retriever(
        search_type="similarity",
        # Fetch more candidates than needed so that, after de-duplication,
        # we can still return exactly NUM_UNIQUE_DOCS unique documents.
        search_kwargs={"k": NUM_UNIQUE_DOCS * 2},
    )

    retrieved_docs = retriever.get_relevant_documents(
        query
    )

    # FIX: drop duplicate chunks (identical page_content) and keep the top
    # NUM_UNIQUE_DOCS unique documents.
    seen = set()
    unique_docs = []
    for doc in retrieved_docs:
        if doc.page_content not in seen:
            seen.add(doc.page_content)
            unique_docs.append(doc)
        if len(unique_docs) >= NUM_UNIQUE_DOCS:
            break

    # Clean up the ephemeral collection so nothing accumulates between runs.
    try:
        vectorstore.delete_collection()
    except Exception:
        pass

    return unique_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
