import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv
dotenv.load_dotenv()

def run_agent(query):
    loader = UnstructuredFileLoader(
        'paper2.pdf'
    )
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    # ------------------------------------------------------------------
    # Fix 1: Do not index the same chunk twice in the first place.
    # The PDF loader / splitter can emit identical chunks; once those are
    # embedded into Chroma, every copy ranks equally for the query and
    # plain similarity search returns them all -> duplicate documents.
    # ------------------------------------------------------------------
    deduped_splits = []
    seen_splits = set()
    for split in all_splits:
        if split.page_content not in seen_splits:
            seen_splits.add(split.page_content)
            deduped_splits.append(split)

    vectorstore = Chroma.from_documents(
        documents=deduped_splits, embedding=OpenAIEmbeddings()
    )

    # ------------------------------------------------------------------
    # Fix 2: Use Maximal Marginal Relevance (MMR) instead of plain
    # "similarity" search.  MMR first fetches a larger candidate pool
    # (fetch_k) and then selects results that are relevant *and* diverse,
    # explicitly penalising redundant / duplicate chunks, so the retriever
    # returns 6 distinct documents instead of 6 copies of the same one.
    # ------------------------------------------------------------------
    retriever = vectorstore.as_retriever(
        search_type="mmr",
        search_kwargs={"k": 6, "fetch_k": 24, "lambda_mult": 0.5},
    )

    retrieved_docs = retriever.get_relevant_documents(query)

    # ------------------------------------------------------------------
    # Fix 3: Guarantee uniqueness of the final result.  Even with MMR,
    # byte-identical chunks can still slip through, so drop any document
    # whose text has already been seen.
    # ------------------------------------------------------------------
    unique_docs = []
    seen_content = set()
    for doc in retrieved_docs:
        if doc.page_content not in seen_content:
            seen_content.add(doc.page_content)
            unique_docs.append(doc)

    # If fewer than 6 unique documents were returned, top the result up
    # from a wider similarity search until we have exactly 6 unique docs
    # (or run out of distinct chunks in the corpus).
    if len(unique_docs) < 6:
        wider_pool = vectorstore.similarity_search(
            query, k=min(len(deduped_splits), 50)
        )
        for doc in wider_pool:
            if len(unique_docs) >= 6:
                break
            if doc.page_content not in seen_content:
                seen_content.add(doc.page_content)
                unique_docs.append(doc)

    return unique_docs[:6]

if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
