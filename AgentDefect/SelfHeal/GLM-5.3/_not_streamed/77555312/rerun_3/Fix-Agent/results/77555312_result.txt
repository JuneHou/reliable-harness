import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv
dotenv.load_dotenv()

NUM_UNIQUE_DOCS = 6  # the number of *unique* documents the user wants back


def deduplicate_documents(docs):
    """
    Remove duplicate documents based on their page_content.

    The first occurrence of each unique text is kept and the original order
    (i.e. relevance order for retrieval results) is preserved.
    """
    seen = set()
    unique_docs = []
    for doc in docs:
        # Whitespace-normalized content is used as the dedup key so that
        # chunks that only differ in formatting are treated as duplicates.
        key = " ".join(doc.page_content.split())
        if key not in seen:
            seen.add(key)
            unique_docs.append(doc)
    return unique_docs


def retrieve_unique_documents(retriever, query, k=NUM_UNIQUE_DOCS, total_docs=None):
    """
    Retrieve k *unique* documents.

    The retriever's `k` counts raw vector-store records, so duplicate copies
    of the same chunk can occupy several of the top-k slots. To still end up
    with k unique documents we progressively fetch more results and discard
    any document whose content we have already seen, until we have collected
    k unique documents (or the collection is exhausted).
    """
    fetch_k = k
    while True:
        retriever.search_kwargs["k"] = fetch_k
        results = retriever.get_relevant_documents(query)
        unique_docs = deduplicate_documents(results)

        if len(unique_docs) >= k:
            return unique_docs[:k]

        # Collection exhausted (fewer results returned than requested, or we
        # already asked for every document that exists) -> return what we have.
        if len(results) < fetch_k or (total_docs is not None and fetch_k >= total_docs):
            return unique_docs

        fetch_k *= 2


def run_agent(query):
    loader = UnstructuredFileLoader(
        'paper2.pdf'
    )
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    # 1) Don't index the same text twice in the first place.
    #    chunk_overlap can produce chunks with identical page_content (only
    #    their start_index metadata differs); without this the vector store
    #    holds multiple copies of the same text and the retriever later
    #    returns them as "duplicates".
    all_splits = deduplicate_documents(all_splits)

    # 2) Give the store its own collection name so repeated runs of this
    #    script don't keep appending the same chunks to the shared default
    #    ("langchain") collection, which also creates duplicate records.
    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
        collection_name="paper2_unique_docs",
    )
    retriever = vectorstore.as_retriever(
        search_type="similarity", search_kwargs={"k": NUM_UNIQUE_DOCS}
    )

    # 3) Return 6 *unique* documents instead of 6 raw records:
    #    duplicates are discarded and additional documents are fetched as
    #    needed to fill the quota of 6 unique documents.
    retrieved_docs = retrieve_unique_documents(
        retriever, query, k=NUM_UNIQUE_DOCS, total_docs=len(all_splits)
    )

    return retrieved_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(f"Retrieved {len(result)} unique documents for: {query!r}\n")
    for i, doc in enumerate(result, start=1):
        print(f"--- Unique document {i} ---")
        print(doc.page_content)
        print()
