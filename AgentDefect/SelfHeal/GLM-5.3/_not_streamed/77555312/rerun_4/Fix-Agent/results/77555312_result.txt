import uuid

import chromadb
import dotenv
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma

dotenv.load_dotenv()


def _fresh_chroma_client():
    """Return a brand-new, in-memory (ephemeral) Chroma client.

    This is the key fix. Previously ``Chroma.from_documents()`` was called
    with neither a dedicated client nor a collection name, so Chroma reused
    its *default* collection. That collection keeps its data between calls,
    so every execution of ``run_agent()`` appended the very same chunks of
    ``paper2.pdf`` on top of the ones that were already stored. A top-k
    similarity search then matched several copies of the same chunk, which
    is why the retriever returned duplicate documents instead of 6 unique
    documents.
    """
    try:
        # chromadb >= 0.4.x
        return chromadb.EphemeralClient()
    except AttributeError:
        # Older chromadb versions: a client created without a persist
        # directory is ephemeral (in-memory) as well.
        return chromadb.Client()


def run_agent(query, k=6):
    loader = UnstructuredFileLoader('paper2.pdf')
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    if not all_splits:
        return []

    # A throw-away client + a unique collection name guarantee that the
    # vector store contains every chunk exactly once, no matter how many
    # times this script is executed (nothing is appended to a stale,
    # previously persisted default collection anymore).
    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
        client=_fresh_chroma_client(),
        collection_name=f"paper2-{uuid.uuid4().hex}",
    )

    # Retrieve k documents; de-duplicate by content and (only if the source
    # document itself contains identical chunks) widen the search window so
    # that we always hand back k *unique* documents.
    total = len(all_splits)
    unique_docs, seen = [], set()
    fetch_k = min(k, total)
    while True:
        retriever = vectorstore.as_retriever(
            search_type="similarity", search_kwargs={"k": fetch_k}
        )
        for doc in retriever.get_relevant_documents(query):
            if doc.page_content not in seen:
                seen.add(doc.page_content)
                unique_docs.append(doc)
        if len(unique_docs) >= k or fetch_k >= total:
            break
        fetch_k = min(fetch_k + k, total)

    return unique_docs[:k]


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
