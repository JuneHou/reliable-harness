import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv
dotenv.load_dotenv()


def unique_by_content(documents):
    """Keep only the first occurrence of every distinct piece of text."""
    unique_docs = []
    seen_contents = set()
    for doc in documents:
        content = doc.page_content.strip()
        if content and content not in seen_contents:
            seen_contents.add(content)
            unique_docs.append(doc)
    return unique_docs


def run_agent(query, k=6):
    loader = UnstructuredFileLoader(
        'paper2.pdf'
    )
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    # (1) Never index the same chunk twice.
    #
    # The PDF loader and the overlapping text splitter can emit identical
    # chunks, and every copy that is written into the vector store can be
    # handed back by the retriever later on. Keeping only the first
    # occurrence of each distinct chunk removes those exact duplicates
    # before they ever reach Chroma.
    unique_splits = unique_by_content(all_splits)

    vectorstore = Chroma.from_documents(
        documents=unique_splits, embedding=OpenAIEmbeddings()
    )

    # (2) Retrieve with maximal marginal relevance (MMR) instead of plain
    # similarity search.
    #
    # Plain "similarity" search simply returns the k nearest chunks.
    # Because the splitter uses chunk_overlap=200, neighbouring chunks
    # share (or even repeat) text, so their embeddings are almost
    # identical and several copies of the same passage end up in the
    # top-k results. MMR first fetches a wider candidate set (fetch_k)
    # and then keeps the k documents that are both relevant and mutually
    # diverse, which drops the redundant copies.
    retriever = vectorstore.as_retriever(
        search_type="mmr",
        search_kwargs={
            "k": k,
            "fetch_k": max(20, 2 * k),
            "lambda_mult": 0.5,
        },
    )

    retrieved_docs = unique_by_content(
        retriever.get_relevant_documents(query)
    )

    # (3) Guarantee k unique documents.
    #
    # If a duplicate still slipped through, we would be short of k
    # documents, so keep asking for a larger diverse candidate pool until
    # k unique documents have been collected (or the vector store is
    # exhausted).
    fetch_k = k
    while len(retrieved_docs) < k and fetch_k < 200:
        fetch_k *= 2
        try:
            candidates = vectorstore.max_marginal_relevance_search(
                query,
                k=fetch_k,
                fetch_k=3 * fetch_k,
                lambda_mult=0.5,
            )
        except Exception:
            break
        retrieved_docs = unique_by_content(candidates)
        if len(candidates) < fetch_k:
            break  # the vector store has nothing more to offer

    return retrieved_docs[:k]


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
    print(f"\n{len(result)} unique documents retrieved")
