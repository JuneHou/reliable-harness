"""
Fix: LangChain + ChatOllama returning the identical response every time
for the same prompt despite temperature / top_p / repeat_penalty being set.

Root cause
----------
1. A fixed `seed` (either set explicitly in the Modelfile / request options or
   applied by the server) forces Ollama to reproduce the *same* tokens for the
   same prompt. Any temperature > 0 is then useless, because the sampler is fed
   a deterministic RNG state -> identical output every single run.
2. Ollama also layers server-side and Modelfile-level parameters on top of the
   request options. A Modelfile that pins `seed`, `temperature 0`, or a very
   small `top_k` (e.g. top_k 1 == greedy decoding) silently overrides what you
   pass to ChatOllama, so sampling parameters appear "ignored".
3. Note: `num_ctx` is only the context-window size -- it has no effect on
   sampling/variety, and `timeout` is only a network timeout. Neither of these
   parameters influences output variety.

Fix
---
* Pass the correct, current sampling options directly to ChatOllama and make
  sure they actually vary the sampling:
    - temperature > 0        -> sampling instead of greedy decoding
    - top_p / top_k          -> widen the candidate token pool
    - repeat_penalty         -> discourage repetition
    - seed=None              -> a NEW random seed per request (no fixed seed!)
* If a fixed seed was set server-side (OLLAMA_SEED) or in the model's
  Modelfile, clear it: the request-level `seed=None` shown here overrides it,
  but a Modelfile with `PARAMETER top_k 1` must be re-saved without it.
* Include the recommended manual sanity checks (commented) that verify the
  fix is genuinely taking effect, plus a demonstration loop that calls the
  model several times to show the responses now differ.
"""

from langchain_ollama import ChatOllama

# --- Sanity checks to run first (manual verification of the fix) ----------
# 1) Make sure no fixed seed is forced globally on the server side:
#       unset OLLAMA_SEED
#    and restart `ollama serve`.
# 2) Verify the model's Modelfile is not pinning determinism:
#       ollama show --modelfile qwen2.5:14b
#    Remove lines such as `PARAMETER seed ...`, `PARAMETER temperature 0`
#    or `PARAMETER top_k 1`, then re-create the model from the Modelfile.
# 3) Verify sampling is actually varying by hitting the raw API:
#       curl http://localhost:11434/api/generate -d '{
#           "model": "qwen2.5:14b",
#           "prompt": "Tell me a lighthearted joke",
#           "options": {"temperature": 0.9, "top_p": 0.95, "num_predict": 128},
#           "stream": false}'
#    Run it twice; if the outputs still match, the Modelfile (step 2) is
#    overriding your options.


def build_llm() -> ChatOllama:
    """Build a ChatOllama instance with sampling options that actually vary.

    Key points:
    - `seed=None` -> each request gets a fresh random seed, so identical
      prompts no longer produce identical text.
    - temperature/top_p/top_k/repeat_penalty are sent as request options,
      which override Modelfile defaults.
    - `num_ctx` only sizes the context window (no sampling effect) and
      `timeout` is a network timeout only.
    """
    return ChatOllama(
        model="qwen2.5:14b",
        # --- sampling options that control output variety ---
        temperature=0.9,        # > 0 for creative sampling (0 = greedy/deterministic)
        top_p=0.95,             # nucleus sampling
        top_k=50,               # keep a wider candidate pool per token
        repeat_penalty=1.1,     # discourage token repetition
        num_predict=256,        # cap the reply length (keeps the demo snappy)
        seed=None,              # crucial: no fixed seed -> varied outputs
        # --- non-sampling options (kept, but irrelevant to variety) ---
        num_ctx=2048,           # context window size only
        timeout=120,            # network timeout only
    )


def run_agent(input_text: str) -> str:
    """Invoke the LLM once and return the response content."""
    llm = build_llm()
    response = llm.invoke(input_text)
    return response.content


if __name__ == "__main__":
    prompt = "Please tell me a lighthearted joke"

    # Call the model several times with the same prompt to demonstrate that
    # the responses now vary instead of being identical every time.
    outputs = [run_agent(prompt) for _ in range(3)]

    for i, out in enumerate(outputs, start=1):
        print(f"--- Response {i} ---")
        print(out)
        print()

    unique_outputs = {out.strip() for out in outputs}
    if len(unique_outputs) == 1 and len(outputs) > 1:
        print("WARNING: responses are still identical.")
        print("A fixed seed is most likely still active server-side "
              "(OLLAMA_SEED) or inside the model's Modelfile -- see the "
              "sanity checks at the top of this file.")
    else:
        print(f"OK: got {len(unique_outputs)} distinct response(s) for the "
              f"same prompt -- sampling is working.")
