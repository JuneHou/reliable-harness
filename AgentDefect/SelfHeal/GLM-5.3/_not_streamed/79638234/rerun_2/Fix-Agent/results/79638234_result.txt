import random

from langchain_ollama import ChatOllama


def run_agent(input_text):
    # Why the outputs were identical:
    #   Ollama's random seed defaults to a fixed value (0). With a fixed
    #   seed, the sampler draws from the same pseudo-random number stream
    #   on every request, so the same prompt returns the exact same tokens
    #   every time -- even with temperature=0.7 and top_p=0.95.
    #
    # Fix:
    #   Generate a fresh random seed for every request so each call samples
    #   from a different random number stream and produces varied outputs.
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,                    # > 0 enables sampling (0 = greedy/deterministic)
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        seed=random.randint(0, 2**31 - 1),  # random seed per call -> varied responses
        keep_alive="5m",                    # keep the model loaded between calls
        timeout=120,
    )
    response = llm.invoke(input_text)
    return response.content


if __name__ == '__main__':
    # Same prompt, multiple calls: the responses now differ instead of
    # repeating the identical answer every time.
    for i in range(3):
        print(f"--- Run {i + 1} ---")
        print(run_agent("Please tell me a lighthearted joke"))
