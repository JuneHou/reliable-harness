import random

from langchain_ollama import ChatOllama


def run_agent(input_text):
    # Build the client fresh on every call and, crucially, give Ollama a NEW
    # random seed for each request. Without a varying seed, Ollama samples
    # deterministically for the same prompt + options (and can even replay a
    # cached result), which is why the response was byte-for-byte identical
    # every time even with temperature/top_p/repeat_penalty set. Those
    # parameters shape the token distribution; the seed is what makes each
    # request actually draw a different sample from it.
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        # Fresh random seed per call -> varied, non-identical outputs.
        seed=random.randint(0, 2**31 - 1),
        timeout=120,
    )
    response = llm.invoke(input_text)
    return response.content


if __name__ == '__main__':
    prompt = "Please tell me a lighthearted joke"

    # Run the same prompt a few times to demonstrate that the outputs now vary.
    for i in range(3):
        print(f"--- Response {i + 1} ---")
        print(run_agent(prompt))
        print()
