import random

from langchain_ollama import ChatOllama


def run_agent(input_text):
    # Why the original code returned identical output every run:
    # Ollama (llama.cpp) samples deterministically when the RNG seed is fixed.
    # If the model's Modelfile pins `PARAMETER seed <n>` (or the server defaults
    # to a fixed seed), then the SAME prompt + SAME seed + SAME sampling params
    # always produce a bit-identical response -- temperature/top_p only shape the
    # sampling distribution, they don't inject randomness when the seed is pinned.
    #
    # Fix: explicitly send a fresh random seed with every request to override any
    # fixed seed baked into the model, and loosen the sampling parameters
    # (slightly higher temperature, lower repeat_penalty, add top_k) so the
    # varied sampling distribution actually shows up in the outputs.
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.9,                 # was 0.7 -- raised for more variety
        top_p=0.95,
        top_k=40,                        # widen the candidate pool
        num_ctx=2048,
        repeat_penalty=1.1,              # was 1.2 -- high values flatten variety
        seed=random.randint(0, 2**31 - 1),  # NEW: random seed per call => non-deterministic output
        timeout=120,
    )
    response = llm.invoke(input_text)
    return response.content


if __name__ == '__main__':
    print(run_agent("Please tell me a lighthearted joke"))
