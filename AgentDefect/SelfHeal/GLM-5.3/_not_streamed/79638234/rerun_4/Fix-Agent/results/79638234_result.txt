import random

from langchain_ollama import ChatOllama


def run_agent(input_text: str) -> str:
    """Send a prompt to the local Ollama model and return its reply.

    Why the original code always returned the identical answer:

    Ollama seeds its sampling RNG per request.  When the seed is left at
    the server default (or fixed anywhere, e.g. in the model's Modelfile),
    an identical prompt replays the exact same token-selection path every
    time, which makes `temperature`, `top_p` and `repeat_penalty` appear
    to have no effect whatsoever.

    Drawing a *fresh random seed* for every request makes the sampler
    branch differently on each call -- that is what actually produces
    varied output (temperature > 0 then governs how widely it varies).

    The `ChatOllama` instance is built inside this function on purpose,
    so every call carries its own seed.
    """
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,  # must stay > 0 so the sampler is non-greedy
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        # FIX: a different seed on every call -> a different sampling
        # path, so identical prompts no longer return identical text.
        seed=random.randint(0, 2**31 - 1),
        timeout=120,
    )
    response = llm.invoke(input_text)
    return response.content


if __name__ == "__main__":
    # Ask the same prompt several times to confirm the answers now vary.
    prompt = "Please tell me a lighthearted joke"
    for i in range(3):
        print(f"--- answer {i + 1} ---")
        print(run_agent(prompt))
        print()
