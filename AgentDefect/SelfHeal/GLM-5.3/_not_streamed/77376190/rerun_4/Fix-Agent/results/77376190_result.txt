import os
from dotenv import load_dotenv, find_dotenv
import openai
from llama_index.core import (
    SimpleDirectoryReader,
    VectorStoreIndex,
    Settings,
    Prompt,
)
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())

os.environ['REQUESTS_CA_BUNDLE'] = ''  # keep user's SSL/proxy workaround


def run_agent(user_input, docs_path='docs'):

    openai_api_key = os.getenv('OPENAI_API_KEY')

    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key

    def createVectorIndex(path):
        tokens = 1500  # a complete NDA draft needs more than 512 tokens
        chunk_size = 600
        max_chunk_overlap = 0.1

        Settings.llm = OpenAI(temperature=0.3, model='gpt-3.5-turbo', max_tokens=tokens)
        Settings.chunk_size = chunk_size
        Settings.chunk_overlap = int(chunk_size * max_chunk_overlap)

        # load data
        docs = SimpleDirectoryReader(path).load_data()

        # create vector index
        vectorIndex = VectorStoreIndex.from_documents(docs)

        # persist_dir must be a directory, not a '.json' filename
        vectorIndex.storage_context.persist(persist_dir='./vectorIndexStorage')
        return vectorIndex

    vectorIndex = createVectorIndex(docs_path)

    # ------------------------------------------------------------------
    # FIX (Change Prompt):
    # LlamaIndex's DEFAULT QA prompt instructs the model to
    # "answer the query given the context information and NOT prior
    # knowledge". For a generation task ("draft an NDA") that restriction
    # makes GPT-3.5 refuse with "I cannot generate a draft".
    # We override 'response_synthesizer:text_qa_template' with an explicit
    # drafting prompt so the retrieved example NDAs are used as a style
    # reference and a full draft is produced.
    # ------------------------------------------------------------------
    template = (
        "You are an experienced legal assistant specialized in drafting "
        "non-disclosure agreements (NDAs).\n"
        "Below is context information from example NDA documents. Use it as "
        "a reference for structure, style and legal language.\n"
        "---------------------\n"
        "{context_str}\n"
        "---------------------\n"
        "Given this information, answer the query. If the query asks for a "
        "non-disclosure agreement, generate a complete, professional NDA "
        "draft that includes sections for the parties, definition of "
        "confidential information, obligations of the receiving party, "
        "permitted disclosures, exclusions, term and termination, remedies, "
        "and general provisions. You may draw on your legal drafting "
        "knowledge to fill any gaps in the context. Never refuse and never "
        "say that you cannot generate a draft.\n"
        "Query: {query_str}\n"
        "Answer: "
    )
    qa_template = Prompt(template)

    def chatbot(user_query):
        # similarity_top_k=5 pulls in enough example-NDA context to imitate
        query_engine = vectorIndex.as_query_engine(similarity_top_k=5)

        # inject the custom QA prompt into the response synthesizer
        query_engine.update_prompts(
            {"response_synthesizer:text_qa_template": qa_template}
        )

        response = query_engine.query(user_query)
        return response.response if response.response else str(response)

    result = chatbot(user_input)

    return result


if __name__ == "__main__":
    # The user's goal is NDA *generation*, so the demo query asks for a draft
    user_input = "Generate a non-disclosure agreement draft between two parties."
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result}")
