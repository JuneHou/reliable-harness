import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import (
    SimpleDirectoryReader,
    GPTListIndex,
    GPTVectorStoreIndex,
    PromptHelper,
    load_index_from_storage,
    Settings,
    PromptTemplate,
)
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())


os.environ['REQUESTS_CA_BUNDLE'] = ''

# Custom prompt that instructs the model to DRAFT a new document
# based on the provided example documents, instead of refusing.
# The default text_qa_prompt tells the model to only answer questions
# from the context ("I cannot generate a draft"), which caused the mismatch.
qa_prompt_tmpl = (
    "You are an expert legal document assistant.\n"
    "Below are example non-disclosure agreement (NDA) documents that serve as reference context.\n"
    "Your task is to use these examples as a template and style guide.\n"
    "\n"
    "---------------------\n"
    "{context_str}\n"
    "---------------------\n"
    "\n"
    "Given the reference examples above, fulfill the following request.\n"
    "If the user asks a question about the documents, answer it using the context.\n"
    "If the user asks you to draft, write, or generate a document (for example, "
    "a non-disclosure agreement), DO NOT refuse. Instead, generate a complete, "
    "well-structured draft modeled on the example documents above. "
    "Never respond with phrases like 'I cannot generate a draft'.\n"
    "\n"
    "Request: {query_str}\n"
)
text_qa_template = PromptTemplate(qa_prompt_tmpl)


def run_agent(user_input, docs_path='docs'):

    openai_api_key = os.getenv('OPENAI_API_KEY')

    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key

    def createVecorIndex(path):
        max_input = 4096
        tokens = 512
        chunk_size = 600
        max_chunk_overlap = 0.1

        Settings.llm = OpenAI(temperature=0.3, model='gpt-3.5-turbo', max_tokens=tokens)
        Settings.chunk_size = chunk_size
        Settings.chunk_overlap = int(chunk_size * max_chunk_overlap)

        #load data
        docs = SimpleDirectoryReader(path).load_data()
        #create vector index
        vectorIndex = GPTVectorStoreIndex.from_documents(docs)
        vectorIndex.storage_context.persist(persist_dir='vectorIndex.json')
        return vectorIndex

    vectorIndex = createVecorIndex('docs')

    def chatbot(input_index):
        # Build the query engine with the custom generation-friendly prompt
        # and enough similarity_top_k to retrieve relevant NDA example chunks.
        query_engine = vectorIndex.as_query_engine(
            text_qa_template=text_qa_template,
            similarity_top_k=3,
        )
        response = query_engine.query(input_index)
        return str(response)

    result = chatbot(user_input)

    return result


if __name__ == "__main__":

    # Ask the chatbot to actually generate an NDA draft,
    # as the user expects, instead of only asking about the documents.
    user_input = "Generate a non-disclosure agreement draft based on the example documents."
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result}")
