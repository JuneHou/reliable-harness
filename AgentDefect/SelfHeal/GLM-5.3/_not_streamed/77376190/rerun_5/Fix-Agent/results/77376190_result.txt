import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import (
    SimpleDirectoryReader,
    GPTListIndex,
    GPTVectorStoreIndex,
    PromptHelper,
    PromptTemplate,
    load_index_from_storage,
    Settings,
)
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())


os.environ['REQUESTS_CA_BUNDLE'] = ''

# ---------------------------------------------------------------------------
# Custom query prompt (text QA template).
#
# The default LlamaIndex query prompt instructs the model to answer *only*
# from the retrieved context and to say "I don't know" otherwise. Because of
# that, a request such as "generate an NDA draft" was repeatedly refused with
# "I cannot generate a draft" instead of producing the agreement.
#
# This replacement prompt closes that gap: it explicitly tells the model that
# the retrieved example NDA documents are templates to draft from, and that it
# must always produce a full draft when asked, never refuse.
# ---------------------------------------------------------------------------
NDA_DRAFT_PROMPT = (
    "You are a legal assistant specialized in drafting non-disclosure "
    "agreements (NDAs).\n"
    "Below are example NDA documents retrieved from a knowledge base. Use them "
    "as reference for the structure, standard clauses, wording and tone.\n"
    "If the request asks you to draft or generate an NDA, always produce a "
    "complete, well-structured non-disclosure agreement based on those example "
    "documents, using placeholders such as [PARTY A], [PARTY B] and "
    "[EFFECTIVE DATE] for details that are not supplied.\n"
    "If the request is a question, answer it using the example documents.\n"
    "Never refuse the request and never reply that you cannot generate a "
    "draft.\n"
    "---------------------\n"
    "{context_str}\n"
    "---------------------\n"
    "Given the example NDA documents above, fulfil the following request.\n"
    "Request: {query_str}\n"
    "Answer: "
)


def run_agent(user_input, docs_path='docs'):

    openai_api_key = os.getenv('OPENAI_API_KEY')

    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key

    def createVectorIndex(path):
        max_input = 4096
        tokens = 2048          # enough tokens to emit a full NDA draft
        chunk_size = 600
        max_chunk_overlap = 0.1

        Settings.llm = OpenAI(temperature=0.3, model='gpt-3.5-turbo', max_tokens=tokens)
        Settings.chunk_size = chunk_size
        Settings.chunk_overlap = int(chunk_size * max_chunk_overlap)

        #load data
        docs = SimpleDirectoryReader(path).load_data()
        #create vector index
        vectorIndex = GPTVectorStoreIndex.from_documents(docs)
        vectorIndex.storage_context.persist(persist_dir='vectorIndex')
        return vectorIndex

    vectorIndex = createVectorIndex(docs_path)

    def chatbot(input_index):
        # Attach the custom drafting prompt so the engine no longer falls back
        # to the strict "context-only Q&A / say you don't know" default prompt
        # that caused the "I cannot generate a draft" refusals.
        query_engine = vectorIndex.as_query_engine(
            text_qa_template=PromptTemplate(NDA_DRAFT_PROMPT),
            similarity_top_k=3,   # retrieve enough example clauses to draft from
        )
        response = query_engine.query(input_index)
        return response.response

    result = chatbot(user_input)

    return result

if __name__ == "__main__":

    # Ask for what the user actually wants: a draft NDA based on the example
    # documents (instead of a generic "main topic" summary question).
    user_input = (
        "Based on the example NDA documents, generate a "
        "non-disclosure agreement draft."
    )
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result}")
