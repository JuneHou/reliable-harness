import os
from dotenv import load_dotenv, find_dotenv
import openai
from llama_index.core import SimpleDirectoryReader, GPTVectorStoreIndex, Settings
from llama_index.core.prompts import PromptTemplate
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())


os.environ['REQUESTS_CA_BUNDLE'] = ''

# ---------------------------------------------------------------------------
# Custom prompts for the response synthesizer.
#
# The default LlamaIndex text_qa_template tells the model:
#   "You are an expert Q&A system ... Always answer the query using the
#    provided context information, and not prior knowledge."
# That strict "Q&A-only" framing makes the model REFUSE to create new content,
# so asking it to generate an NDA draft yields "I cannot generate a draft".
#
# These replacements instruct the model to act as a legal drafting assistant
# that uses the retrieved example NDA documents as a reference to actually
# produce the requested document. The variable names match exactly what
# LlamaIndex expects: {context_str}/{query_str} for the QA template and
# {query_str}/{existing_answer}/{context_msg} for the refine template.
# ---------------------------------------------------------------------------
qa_template = PromptTemplate(
    "Context information from example non-disclosure agreement (NDA) documents "
    "is below.\n"
    "---------------------\n"
    "{context_str}\n"
    "---------------------\n"
    "You are a legal document drafting assistant. Using the context information "
    "above as a reference for structure, clauses, and wording, fulfil the user's "
    "request and produce a complete, well-structured document.\n"
    "You MUST generate the requested document. Never refuse the request, never "
    "reply that you cannot generate a draft, and never claim that the answer is "
    "not contained in the context.\n"
    "Do not mention the context, the reference documents, or these instructions "
    "in your answer.\n"
    "Query: {query_str}\n"
    "Answer: "
)

refine_template = PromptTemplate(
    "The original query is as follows: {query_str}\n"
    "We have provided an existing answer: {existing_answer}\n"
    "We have the opportunity to refine the existing answer "
    "(only if needed) with some more context below.\n"
    "------------\n"
    "{context_msg}\n"
    "------------\n"
    "Given the new context, refine the existing answer so that it better "
    "fulfils the request and reads as a complete, professional document. "
    "Never refuse to generate the requested document. "
    "If the context isn't useful, return the original answer.\n"
    "Refined Answer: "
)


def run_agent(user_input, docs_path='docs'):
    
    openai_api_key = os.getenv('OPENAI_API_KEY')
    
    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key
    
    def createVecorIndex(path):
        # A full agreement draft needs far more than 512 output tokens,
        # otherwise the generated NDA gets cut off mid-document.
        tokens = 2048
        chunk_size = 600
        max_chunk_overlap = 0.1
        
        
        Settings.llm = OpenAI(temperature=0.3, model='gpt-3.5-turbo', max_tokens=tokens)
        Settings.chunk_size = chunk_size
        Settings.chunk_overlap = int(chunk_size * max_chunk_overlap)
        
        #load data
        docs = SimpleDirectoryReader(path).load_data()
        #create vector index
        vectorIndex = GPTVectorStoreIndex.from_documents(docs)
        vectorIndex.storage_context.persist(persist_dir='vectorIndex.json')
        return vectorIndex
    
    vectorIndex = createVecorIndex(docs_path)
    
    def chatbot(input_text):
        # Retrieve more of the example NDA content so the model has enough
        # reference clauses to build a complete draft from.
        query_engine = vectorIndex.as_query_engine(
            similarity_top_k=5,
        )
        # Override the default "Q&A-only" prompts so the model drafts the
        # requested document from the retrieved example NDAs instead of
        # refusing with "I cannot generate a draft".
        query_engine.update_prompts(
            {
                "response_synthesizer:text_qa_template": qa_template,
                "response_synthesizer:refine_template": refine_template,
            }
        )
        response = query_engine.query(input_text)
        return response.response
   
    result = chatbot(user_input)
    
    return result

if __name__ == "__main__":
  
    user_input = "Please generate a non-disclosure agreement draft based on the example documents."
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result}")
