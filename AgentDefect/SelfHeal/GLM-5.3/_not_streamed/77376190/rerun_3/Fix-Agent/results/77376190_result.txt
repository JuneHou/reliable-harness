import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import (
    SimpleDirectoryReader,
    GPTListIndex,
    GPTVectorStoreIndex,
    PromptHelper,
    load_index_from_storage,
    Settings,
    PromptTemplate,
)
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())


os.environ['REQUESTS_CA_BUNDLE'] = ''  

def run_agent(user_input, docs_path='docs'):
    
    openai_api_key = os.getenv('OPENAI_API_KEY')
    
    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key
    
    def createVecorIndex(path):
        max_input = 4096
        tokens = 512
        chunk_size = 600
        max_chunk_overlap_ratio = 0.1
        
        
        Settings.llm = OpenAI(temperature=0.3, model='gpt-3.5-turbo', max_tokens=tokens)
        Settings.context_window = max_input
        Settings.chunk_size = chunk_size
        Settings.chunk_overlap = int(chunk_size * max_chunk_overlap_ratio)
        
        #load data
        docs = SimpleDirectoryReader(path).load_data()
        #create vector index
        vectorIndex = GPTVectorStoreIndex.from_documents(docs)
        vectorIndex.storage_context.persist(persist_dir='./vectorIndexStore')
        return vectorIndex
    
    vectorIndex = createVecorIndex('docs')
    
    def chatbot(input_index):
        # FIX: the default LlamaIndex text-QA prompt tells the model it is a
        # Q&A system that must only answer questions from the context and must
        # not use prior knowledge. Asked to *draft* a document, the model
        # therefore refuses ("I cannot generate a draft").
        # We replace it with a custom text_qa_template that instructs the model
        # to act as a lawyer's drafting assistant and to USE the retrieved
        # example documents as templates to generate the requested draft.
        # The template must use the expected variables: context_str + query_str.
        qa_template = PromptTemplate(
            "You are an experienced lawyer's drafting assistant. "
            "You are given example documents below which you must use as "
            "templates and guides. Do NOT refuse and do NOT say that you "
            "cannot generate a draft. Your job is to generate a complete first "
            "draft of the document requested by the user, adapting the "
            "structure, terminology and clauses from the example documents to "
            "the user's request. If specific details are missing, fill in "
            "reasonable placeholders such as [Party Name], [Date] or [Address].\n"
            "---------------------\n"
            "Example documents:\n"
            "{context_str}\n"
            "---------------------\n"
            "Request: {query_str}\n"
            "Draft:"
        )

        # Matching refine template so multi-chunk answers keep the drafting
        # behaviour instead of falling back to the default Q&A refine prompt.
        refine_template = PromptTemplate(
            "You are an experienced lawyer's drafting assistant. "
            "Do NOT refuse the request and do NOT say you cannot generate a "
            "draft.\n"
            "The original draft based on existing example documents is below.\n"
            "---------------------\n"
            "{context_msg}\n"
            "---------------------\n"
            "Using the new example document information above, refine the "
            "original draft to better match the requested document. "
            "If the new information is not useful, return the original draft.\n"
            "Original draft: {existing_answer}\n"
            "Refined draft:\n"
        )

        # Pass the custom prompts (and retrieve more of the example NDAs) so
        # the engine generates a draft instead of refusing.
        query_engine = vectorIndex.as_query_engine(
            text_qa_template=qa_template,
            refine_template=refine_template,
            similarity_top_k=3,
        )
        response = query_engine.query(input_index)
        return response.response
   
    result = chatbot(user_input)
    
    return result

if __name__ == "__main__":
  
    # FIX: exercise the actual failing scenario - request a document draft

                  "between Mike LLC and Fantasty World.")
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Request: {user_input}")
    print(f"Answer:\n{result}")
