from dotenv import load_dotenv
import os
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough, RunnableLambda

load_dotenv()


def format_docs(docs):
    """Convert the list of retrieved Documents into clean, readable prose.

    Without this step, the retriever's output (a list of Document objects) is
    stringified directly by the prompt template, producing something like
    [Document(metadata={...}, page_content='...'), Document(...)].
    The model then mimics that fragmented structure and answers in
    fragments instead of complete sentences.
    """
    return "\n\n".join(doc.page_content for doc in docs)


def run_agent(query):

    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    products = pd.read_csv("./data/products.csv")

    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )

    retriever = vectorstore.as_retriever()

    ANSWER_TEMPLATE = """Answer the question based only on the following context:

{context}

Question: {question}

Instructions: Respond with a single, complete, conversational answer written in full, grammatically correct sentences. Do not quote or repeat the context verbatim, do not output fragments, lists, or raw document text, and do not mention that you were given context or documents.
"""

    prompt = ChatPromptTemplate.from_template(ANSWER_TEMPLATE)

    chain = (
        {
            "context": retriever | RunnableLambda(format_docs),
            "question": RunnablePassthrough()
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)

    return result


if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
