import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.vectorstores.base import VectorStoreRetriever
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]

# Default metadata filter: only paper2.pdf should be used to answer queries.
DEFAULT_FILTER = {"source": "paper2.pdf"}


def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


def run_agent(query, pdf_paths, filter=DEFAULT_FILTER):
    """
    Build the vector store / QA chain and answer `query`.

    `filter` is a Chroma metadata ("where") filter, e.g. {'source': 'paper2.pdf'}.

    FIX: ConversationalRetrievalChain only accepts the input keys "question"
    and "chat_history". Passing a "filter" key into the chain call (as the
    buggy code did) is silently ignored, so retrieval ran over ALL documents
    and the answer could come from the wrong PDF.

    Instead, the metadata filter must be attached to the retriever via its
    search_kwargs. The Chroma retriever forwards `filter` as the `where`
    clause of the underlying vector search, so only the matching documents
    (here: those whose metadata "source" == "paper2.pdf") are retrieved.
    """
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            text.metadata["source"] = pdf_path

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    # Attach the metadata filter to the retriever (NOT to the chain input).
    # This makes the filter effective: only chunks whose metadata matches
    # the filter are returned by get_relevant_documents().
    search_kwargs = {"k": 4}
    if filter:
        search_kwargs["filter"] = filter

    retriever = db.as_retriever(search_kwargs=search_kwargs)

    qa = ConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever,
        verbose=True,
        return_source_documents=True)

    # FIX: do NOT pass "filter" here -- it is not a valid input key of
    # ConversationalRetrievalChain and would be silently discarded.
    result = qa({"question": query, "chat_history": []})

    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]

    # it means the information from paper2.pdf should be used to answer the query.
    # if it answers from paper2.pdf, then the filtering is working correctly.
    filter = {'source': 'paper2.pdf'}

    result = run_agent(query, pdf_paths, filter)

    print(result['answer'])
