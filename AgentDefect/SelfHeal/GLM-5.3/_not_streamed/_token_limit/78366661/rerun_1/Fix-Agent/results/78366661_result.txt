import os
import time

from dotenv import load_dotenv

from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_mongodb.vectorstores import MongoDBAtlasVectorSearch
from pymongo import MongoClient
from pymongo.operations import SearchIndexModel


def _get_atlas_collection(db_name="langchain_db", collection_name="pdf_embeddings"):
    """Atlas Vector Search requires a real PyMongo collection object.

    A plain Python list (or any non-collection object) cannot be written to
    Atlas, so from_documents silently "succeeds" locally while nothing is
    ever indexed, and every similarity_search comes back empty.
    """
    mongodb_uri = os.getenv("MONGODB_URI")
    if mongodb_uri is None:
        raise ValueError(
            "MONGODB_URI not found. Atlas Vector Search requires a real "
            "MongoDB Atlas cluster connection string."
        )
    client = MongoClient(mongodb_uri)
    return client[db_name][collection_name]


def _ensure_vector_search_index(collection, index_name, dimensions):
    """Create the vector search index and wait until it is READY.

    Atlas Vector Search returns an empty array (not an error) whenever the
    named index does not exist, is still BUILDING, or is defined over a
    different field path than where the embeddings are stored.
    """
    indices = list(collection.list_search_indexes(index_name))

    if not indices:
        search_index_model = SearchIndexModel(
            definition={
                "fields": [
                    {
                        "type": "vector",
                        "path": "embedding",  # field used by MongoDBAtlasVectorSearch
                        "numDimensions": dimensions,
                        "similarity": "cosine",
                    }
                ]
            },
            name=index_name,
            type="vectorSearch",
        )
        collection.create_search_index(search_index_model)

    # Block until the index is queryable, otherwise searches return [].
    while True:
        indices = list(collection.list_search_indexes(index_name))
        if indices and indices[0].get("status") == "READY":
            break
        time.sleep(5)


def _similarity_search_with_retry(vector_search, query, k=4, attempts=6, delay=5):
    """Newly inserted documents are not instantly visible to the index."""
    results = []
    for _ in range(attempts):
        results = vector_search.similarity_search(query, k=k)
        if results:
            break
        time.sleep(delay)
    return results


def run_agent(pdf_url):
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key is None:
        raise ValueError("OPENAI_API_KEY not found")

    # Use a real Atlas collection instead of an empty Python list.
    atlas_collection = _get_atlas_collection()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100,
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()

    docs = text_splitter.split_documents(data)
    if not docs:
        raise ValueError("No documents were produced from the PDF")

    embeddings = OpenAIEmbeddings(
        openai_api_key=openai_api_key,
        disallowed_special=(),
    )

    vector_search_index = "vector_index"

    # Derive the true embedding dimension so the index definition always
    # matches the model actually in use.
    dimensions = len(embeddings.embed_query("dimension probe"))

    # Make sure the vector search index exists and is READY before searching.
    _ensure_vector_search_index(atlas_collection, vector_search_index, dimensions)

    # Use the real MongoDBAtlasVectorSearch from langchain_mongodb.
    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=embeddings,
        collection=atlas_collection,
        index_name=vector_search_index,
    )

    query = "What were the compute requirements for training GPT 4"

    results = _similarity_search_with_retry(vector_search, query, k=4)

    if not results:
        raise RuntimeError(
            "similarity_search returned no results. Verify that the vector "
            "search index is READY, its name matches index_name, and its "
            "'path' points at the embedding field."
        )

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"

    results = run_agent(url)

    print("Result:", results)
