from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.prompts import PromptTemplate
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


# Define a custom prompt template instead of relying on the library's
# hardcoded default STUFF_PROMPT ("Given the following extracted parts
# of a long document and a question, create a final answer with
# references (\"SOURCES\")...").
#
# The template MUST declare the input variables the "stuff" chain expects:
#   - "summaries": the concatenated retrieved document contents
#   - "question": the user's question
CUSTOM_TEMPLATE = """You are a helpful assistant that answers questions using the provided context.

Context:
{summaries}

Question: {question}

Helpful Answer (cite the sources you used as "SOURCES: ..."):
"""

CUSTOM_PROMPT = PromptTemplate(
    input_variables=["summaries", "question"],
    template=CUSTOM_TEMPLATE,
)


def run_agent():
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        # Pass the custom prompt through chain_type_kwargs so the internal
        # LLM chain is built with OUR template rather than the hardcoded
        # default one. Printing the chain will now show this template.
        chain_type_kwargs={"prompt": CUSTOM_PROMPT},
    )

    return chain


if __name__ == "__main__":
    print(run_agent())
