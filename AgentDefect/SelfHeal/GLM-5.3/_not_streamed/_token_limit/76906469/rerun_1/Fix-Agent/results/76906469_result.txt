from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key
)

tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# NOTE ON THE FIX:
# The original code used AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION.
# The zero-shot ReAct agents (ZERO_SHOT_REACT_DESCRIPTION and
# CHAT_ZERO_SHOT_REACT_DESCRIPTION) are *single-turn* agents: their prompt
# templates contain no chat-history placeholder, so even though a
# ConversationBufferMemory was passed to initialize_agent(), the stored
# history was never inserted into the prompt. That is why the agent could
# not recall previous questions/answers ("what is the last question I
# asked?").
#
# To actually use chat memory you must use a *conversational* agent type,
# whose prompt template includes a "chat_history" placeholder. For chat
# models (like ChatOpenAI) the correct choice is
# CHAT_CONVERSATIONAL_REACT_DESCRIPTION.
#
# The memory must also be configured so its key matches the placeholder in
# the agent's prompt ("chat_history") and so it returns message objects
# (return_messages=True), which the chat-model agent expects.

memory = ConversationBufferMemory(
    memory_key="chat_history",
    return_messages=True
)

agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False
)

def run_agent(prompt: str) -> str:
    # .run() keeps the conversational memory in `memory` across calls,
    # so follow-up questions like "what is the last question I asked?"
    # now have access to the prior turns.
    return agent_test.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    result2 = run_agent("what is the last question I asked?")
    print(result2)
