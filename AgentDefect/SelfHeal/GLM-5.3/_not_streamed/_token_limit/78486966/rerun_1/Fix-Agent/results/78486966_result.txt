    # BUG FIX:
    # CharacterTextSplitter's default separator is "\n\n" (paragraph breaks).
    # The input text only contains single newlines, so the text was never
    # actually split and was returned as one oversized chunk.
    # Setting separator=" " makes the splitter break on spaces, so every
    # individual split is smaller than chunk_size and the merger can build
    # real 30-character chunks with 5-character overlap.

from langchain_text_splitters import CharacterTextSplitter


def run_agent(text_raw):
    # BUG FIX: CharacterTextSplitter only splits on the given `separator`,
    # which defaults to "\n\n" (double newline). The input text is separated
    # by single newlines, so with the default separator no split points are
    # found and the whole text is returned as a single chunk.
    # Setting separator="\n" matches the actual structure of the text so it
    # gets split into ~chunk_size chunks with chunk_overlap characters of
    # overlap between consecutive chunks.
    text_splitter = CharacterTextSplitter(
        chunk_size=30,
        chunk_overlap=5,
        separator="\n",
        is_separator_regex=False,
    )
    chunks = text_splitter.split_text(text_raw.strip())
    return chunks

    - Plain text file: human-readable, one chunk per section.
    - JSON file: structured, easy to reload programmatically.
    """
    with open(file_path, "w", encoding="utf-8") as f:
        for i, chunk in enumerate(chunks, start=1):
            f.write(f"--- Chunk {i} ---\n{chunk}\n\n")

    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(chunks, f, indent=2, ensure_ascii=False)

    return file_path, json_path
