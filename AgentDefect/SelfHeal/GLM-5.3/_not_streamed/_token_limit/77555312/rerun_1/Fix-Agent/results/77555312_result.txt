import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv
dotenv.load_dotenv()


def get_unique_documents(docs):
    """Remove duplicate copies of the same document.

    Two documents are considered duplicates when they carry the same
    page_content. The first (most relevant) occurrence is kept and the
    original ordering is preserved.
    """
    seen = set()
    unique_docs = []
    for doc in docs:
        if doc.page_content not in seen:
            seen.add(doc.page_content)
            unique_docs.append(doc)
    return unique_docs


def run_agent(query, k=6):
    loader = UnstructuredFileLoader(
        'paper2.pdf'
    )
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    # 1) Don't embed and store duplicate chunks in the first place.
    #    Identical chunks would produce identical vectors that all show up
    #    among the top-k nearest neighbours.
    all_splits = get_unique_documents(all_splits)

    vectorstore = Chroma.from_documents(documents=all_splits, embedding=OpenAIEmbeddings())

    # 2) Over-fetch candidates so that after removing duplicate copies we
    #    still have at least k unique documents to return.
    fetch_k = min(len(all_splits), k * 3) if all_splits else k
    retriever = vectorstore.as_retriever(
        search_type="similarity", search_kwargs={"k": fetch_k}
    )

    retrieved_docs = retriever.get_relevant_documents(
        query
    )

    # 3) Drop duplicate copies of the same document and return only the
    #    k most similar unique documents.
    unique_docs = get_unique_documents(retrieved_docs)

    return unique_docs[:k]


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
