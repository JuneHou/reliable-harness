import os
from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import LocalFileStore, EncoderBackedStore
from langchain.schema import Document
import nltk


def _serialize_document(doc: Document) -> bytes:
    """Turn a Document into bytes so it can live in a ByteStore (LocalFileStore)."""
    if hasattr(doc, "model_dump_json"):  # pydantic v2
        return doc.model_dump_json().encode("utf-8")
    return doc.json().encode("utf-8")


def _deserialize_document(raw: bytes) -> Document:
    """Rebuild a Document from the bytes stored on disk."""
    text = raw.decode("utf-8")
    if hasattr(Document, "model_validate_json"):  # pydantic v2
        return Document.model_validate_json(text)
    return Document.parse_raw(text)


def build_persistent_docstore(store_location: str) -> EncoderBackedStore:
    """
    Build a docstore that persists parent documents to disk.

    InMemoryStore is ephemeral: each new session starts with an empty
    docstore, so ParentDocumentRetriever can no longer resolve child
    chunks back to their parent documents, and documents must be
    re-added every run. LocalFileStore keeps every parent document on
    disk; it is wrapped in an EncoderBackedStore because LocalFileStore
    itself only stores raw bytes, while the retriever needs
    Document objects keyed by string IDs.
    """
    return EncoderBackedStore(
        store=LocalFileStore(store_location),
        key_encoder=lambda key: key.encode("utf-8"),
        value_serializer=_serialize_document,
        value_deserializer=_deserialize_document,
    )

    nltk.download("punkt")

    hf_embed_path = "./hf_cache"

    docs = [
        Document(page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."),

        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining.")
    ]

    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path
    )

    child_splitter = NLTKTextSplitter(chunk_size=400)

    # The child chunks are already persisted thanks to persist_directory,
    # so the same Chroma collection is reloaded on every session.
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child"
    )

    # Persistent docstore for the parent documents (replaces the
    # ephemeral InMemoryStore).
    store = build_persistent_docstore(store_location)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=store,
        child_splitter=child_splitter
    )

    # Only ingest documents on the very first run. On later runs both the
    # child chunks (Chroma) and the parent documents (LocalFileStore) are
    # loaded from disk, so we must NOT re-add the documents.
    if not list(store.yield_keys()):
        retriever.add_documents(docs)
        # Older langchain versions require an explicit flush of Chroma.
        if hasattr(vectorstore, "persist"):
            try:
                vectorstore.persist()
            except Exception:
                pass

    # Searching the vector store alone would only return child chunks.
    # Querying the retriever returns the full parent documents, resolved
    # from the persisted docstore, and this works across sessions.
    try:
        matching_doc = retriever.invoke(query)
    except AttributeError:
        # Fallback for older langchain versions.
        matching_doc = retriever.get_relevant_documents(query)

    return matching_doc
