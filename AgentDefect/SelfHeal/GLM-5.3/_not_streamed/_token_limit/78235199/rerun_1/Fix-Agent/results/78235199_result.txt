import json
import os


class FinetuningEvent:
    """A single recorded (query, response) pair suitable for OpenAI fine-tuning."""

    def __init__(self, query, response):
        self.query = query
        self.response = response

    def to_dict(self):
        return {
            "messages": [
                {"role": "user", "content": self.query},
                {"role": "assistant", "content": self.response},
            ]
        }


class OpenAIFineTuningHandler:
    """Records LLM inputs/outputs and saves them as fine-tuning events.

    The handler only collects events if it is registered with the callback
    manager of the LLM that actually serves the queries. It must implement the
    callback hooks (on_llm_end) used by the query pipeline, and
    save_finetuning_events must serialize every recorded event.
    """

    def __init__(self):
        self.finetuning_events = []

    # --- callback hooks invoked by the callback manager ---
    def start_trace(self, trace_id=None):
        pass

    def end_trace(self, trace_id=None):
        pass

    def on_event_start(self, event_type, payload=None, **kwargs):
        pass

    def on_llm_end(self, payload):
        """Record the LLM input/output pair emitted at the end of an LLM call."""
        query = payload.get("query", "")
        response = payload.get("response", "")
        self.finetuning_events.append(FinetuningEvent(query, response))

    def save_finetuning_events(self, path):
        """Persist every recorded event to `path` in JSONL format."""
        with open(path, "w", encoding="utf-8") as f:
            for event in self.finetuning_events:
                f.write(json.dumps(event.to_dict()) + "\n")
        print(f"Wrote {len(self.finetuning_events)} examples to {path}")


class CallbackManager:
    """Dispatches callback events to all registered handlers."""

    def __init__(self, handlers):
        self.handlers = handlers

    def on_llm_end(self, payload):
        for handler in self.handlers:
            handler.on_llm_end(payload)


class LLM:
    """Simulated LLM whose calls are traced through its callback manager."""

    def __init__(self, callback_manager=None):
        # The callback manager (holding the fine-tuning handler) must be
        # attached to the LLM that serves the queries so events get recorded.
        self.callback_manager = callback_manager

    def complete(self, query):
        response = f"Answer to: {query}"
        if self.callback_manager is not None:
            self.callback_manager.on_llm_end({"query": query, "response": response})
        return response


class QueryEngine:
    """Runs a query through the LLM so the callbacks attached to the LLM's
    callback manager (e.g. the fine-tuning handler) record the exchange."""

    def __init__(self, llm, similarity_top_k):
        self.llm = llm
        self.similarity_top_k = similarity_top_k

    def query(self, question):
        return self.llm.complete(question)


class VectorStoreIndex:
    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex()

    def as_query_engine(self, similarity_top_k, llm):
        # Return a real query engine that routes queries through the given LLM
        # (and therefore through its callback manager), instead of the index
        # itself whose query() dropped the LLM entirely.
        return QueryEngine(llm=llm, similarity_top_k=similarity_top_k)


class FineTuningExample:

    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate JSONL file for fine-tuning events."""
        os.makedirs(self.dataset_path, exist_ok=True)

        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        # Attach the callback manager (which holds the fine-tuning handler) to
        # the LLM used by the query engine, so every query/response pair is
        # recorded as a fine-tuning event.
        self.llm = LLM(callback_manager=callback_manager)

        questions = []
        with open(f'{self.dataset_path}/train_questions.txt', "r", encoding='utf-8') as f:
            for line in f:
                questions.append(line.strip())

        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                response = query_engine.query(question)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                f'{self.dataset_path}/finetuning_events.jsonl'
            )


def prepare_test_data(questions_path):
    os.makedirs(os.path.dirname(questions_path), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path):
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
