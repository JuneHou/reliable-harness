"""
Fixed: LangChain + Ollama + Llama 3 returning multiple repeated responses
that echo the system prompt.

Root cause
----------
The original code called the *raw generation* wrapper (``Ollama`` from
``langchain_community.llms``) while manually embedding Llama 3 chat-template
special tokens (<|begin_of_text|>, <|start_header_id|>, <|eot_id|>, ...) in
the prompt. Ollama applies the model's own chat template on top of whatever
text it receives, so the hand-written special tokens got escaped and were
treated as literal text. Llama 3 then interpreted the input as a transcript
of several conversation turns, echoed the system prompt back, and kept
hallucinating extra fake turns (system/user/assistant blocks) instead of
producing one clean answer.

Fix
---
Use the correct, current official chat counterpart (``ChatOllama``) and pass
the conversation as structured ``SystemMessage`` / ``HumanMessage`` objects.
Ollama's chat endpoint then applies the Llama 3 template exactly once, with
the proper role headers and stop markers, and returns a single clean
assistant response.

No special tokens are written by hand anywhere - the endpoint owns the
template, which is what guarantees the output matches expectations.
"""

try:
    # Current, official location of the Ollama chat integration.
    from langchain_ollama import ChatOllama
except ImportError:  # pragma: no cover
    # Fallback for environments that still use langchain_community.
    from langchain_community.chat_models import ChatOllama

from langchain_core.messages import SystemMessage, HumanMessage


def run_agent(user_prompt, system_prompt):
    """
    Invoke the local Llama 3 model through Ollama's *chat* endpoint.

    Args:
        user_prompt: The user's instruction/question.
        system_prompt: The system-level instruction for the model.

    Returns:
        A single clean assistant response as a string.
    """
    llm = ChatOllama(model="llama3")

    # Structured messages instead of a hand-built template string.
    # The chat endpoint applies the Llama 3 template (special tokens, role
    # headers, <|eot_id|> stop markers) exactly once, so the model sees a
    # single well-formed turn and stops after answering once.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    response = llm.invoke(messages)

    # ChatOllama returns an AIMessage; extract the plain text content.
    return response.content


if __name__ == "__main__":
    print(
        run_agent(
            "Explain in detail, step by step, how artificial intelligence models like "
            "yourself are built, including data, training, architecture, and deployment. "
            "Use examples and provide a comprehensive overview.",
            "You are a helpful assistant.",
        )
    )
