from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000

# The conversation history must contain BOTH sides of the dialogue.
# A system message is kept at index 0 while old turns get trimmed.
messages = [
    SystemMessage(content="You are a helpful chat assistant. Respond to the user's latest message.")
]

# temperature lowered from 1.1 to 0.8 (the Ollama CLI default) to reduce
# rambling/repetition, and num_ctx raised so the full history actually
# reaches the model instead of being silently truncated.
llm = ChatOllama(
    model=MODEL,
    temperature=0.8,
    num_ctx=8192,
    stream=True
)


def get_number_of_tokens(message_history):
    """Count tokens across every message in the history."""
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        return sum(len(encoding.encode(str(msg.content))) for msg in message_history)
    except Exception:
        return sum(len(str(msg.content)) for msg in message_history) // 4


@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    # 1) Record the user's turn as a proper HumanMessage object
    #    (raw ("user", str) tuples were passed to ChatOllama before).
    messages.append(HumanMessage(content=request.form['text']))

    # 2) Trim the OLDEST user/assistant pairs, keeping the system message.
    #    The token count is recomputed on every iteration: the old loop used
    #    a stale value and deleted messages until it raised an IndexError.
    while len(messages) > 3 and get_number_of_tokens(messages) >= MODEL_MAX_TOKENS:
        del messages[1:3]

    # 3) Send the complete, properly alternating history to the model.
    ai_msg = llm.invoke(messages)

    # 4) THE KEY FIX: store the model's reply in the history.
    #    Previously only user messages were kept, so the model saw a wall of
    #    consecutive "user" prompts with no assistant turns in between. That
    #    made llama3.2 treat the entire history as unresolved instructions,
    #    causing it to repeat the conversation and cling to the first prompt.
    #    Saving the AIMessage (like `ollama run` does in the CLI) restores
    #    the expected user/assistant alternation.
    messages.append(AIMessage(content=ai_msg.content))

    return ai_msg.content


@app.route('/')
def index():
    return render_template('index.html')


if __name__ == '__main__':
    app.run(debug=True, port=5000)
