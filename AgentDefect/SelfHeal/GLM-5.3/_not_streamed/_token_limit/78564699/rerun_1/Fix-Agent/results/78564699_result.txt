import json
import os
from llama_index.llms.llama_cpp import LlamaCPP
from llama_index.llms.llama_utils import (
    messages_to_prompt,
    completion_to_prompt,
)


def run_agent(query):
    MODEL_URL = "https://huggingface.co/TheBloke/Llama-2-13B-chat-GGUF/resolve/main/llama-2-13b-chat.Q4_0.gguf"
    MODEL_PATH = None

    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if "llama-2-13b-chat" in paths:
                MODEL_URL = None
                MODEL_PATH = paths["llama-2-13b-chat"]

    llm = LlamaCPP(
        model_url=MODEL_URL,
        model_path=MODEL_PATH,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        # Since the model being used is llama-2-chat, we must pass in the
        # proper prompt formatting helpers so the raw query is wrapped in the
        # correct [INST] <<SYS>> ... [/INST] chat template. Without these, the
        # model receives an un-formatted prompt, never emits its end-of-turn
        # token properly, and appears to hang / generate indefinitely on CPU.
        messages_to_prompt=messages_to_prompt,
        completion_to_prompt=completion_to_prompt,
        model_kwargs={"n_gpu_layers": 0},
        verbose=True,
    )

    response = llm.complete(query)

    return response


if __name__ == "__main__":
    query = "What is the capital of France?"
    response = run_agent(query)
    print(response)
