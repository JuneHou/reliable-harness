import uuid
import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# Shared artifact store.
#
# The root cause of the mismatch: an AgentExecutor only ever returns the LLM's
# final *textual* answer, and every tool observation is stringified before it
# reaches the model. A DataFrame object returned from a @tool therefore never
# survives to the caller.
#
# Fix: keep the textual channel for the agent, but hand the real DataFrame to
# the main script out-of-band via this shared store. The tool returns only a
# short textual summary (handle, shape, columns) to the LLM; the object itself
# is stored here and re-attached to the invoke() response.
# ---------------------------------------------------------------------------
dataframe_store = {}          # handle -> actual pd.DataFrame object
LAST_HANDLE_KEY = "df"        # deterministic key mirroring the requested name


def get_dataframe(name: str):
    """Retrieve an actual DataFrame produced during agent tool execution."""
    return dataframe_store.get(name)


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a transformed DataFrame.

    The DataFrame is stored in the shared dataframe_store under the handle
    'df' (and a unique handle), and only a short textual summary is returned
    to the agent. The real object can be retrieved in the main script via
    dataframe_store['df'] or the helper get_dataframe('df').
    """
    df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
    df = df.select_dtypes(include=['object'])

    # --- side channel: persist the REAL DataFrame object ---
    unique_handle = f"df_{uuid.uuid4().hex[:8]}"
    dataframe_store[unique_handle] = df
    dataframe_store[LAST_HANDLE_KEY] = df  # mirror the requested variable name

    # --- textual channel: only a compact summary goes back to the LLM ---
    return (
        f"DataFrame created from '{filepath}' and stored with handle "
        f"'{unique_handle}' (also available as '{LAST_HANDLE_KEY}'). "
        f"shape={df.shape}, columns={list(df.columns)}. "
        f"Use get_dataframe('{LAST_HANDLE_KEY}') in the main script to "
        f"retrieve the actual object."
    )


@tool
def get_dataframe_handle(name: str) -> str:
    """Report metadata (shape/columns) about a stored DataFrame handle."""
    df = dataframe_store.get(name)
    if df is None:
        return f"No DataFrame stored under handle '{name}'."
    return f"handle='{name}', shape={df.shape}, columns={list(df.columns)}"


def run_agent(user_prompt, file_name="data.json"):
    llm = ChatOpenAI(model='gpt-4o-mini')
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    # Optional: expose a lookup tool so the agent can reference the artifact
    tools_list = [load_excel_response, get_dataframe_handle, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(
        agent=shell_agent,
        tools=tools_list,
        verbose=True,
        handle_parsing_errors=True,
    )

    response = agent_executor.invoke({"input": user_prompt})

    # Attach the actual DataFrame objects created during tool execution so
    # the caller receives real objects, not just the agent's textual summary.
    response["dataframes"] = dict(dataframe_store)
    response["dataframe"] = dataframe_store.get(LAST_HANDLE_KEY)
    return response


if __name__ == "__main__":
    path = 'excel_file.xlsx'
    user_prompt = (
        f"Load the Excel file located at {path} and assign the result to a "
        f"variable named df."
    )
    result = run_agent(user_prompt)

    # The agent's textual answer (what the LLM says):
    print("Agent output:\n", result["output"])

    # The ACTUAL DataFrame object created during tool execution:
    df = result.get("dataframe") or get_dataframe(LAST_HANDLE_KEY)
    if df is not None:
        print("\nRetrieved real DataFrame object:", type(df))
        print(df.head())
    else:
        print("\nNo DataFrame was stored during this run.")
