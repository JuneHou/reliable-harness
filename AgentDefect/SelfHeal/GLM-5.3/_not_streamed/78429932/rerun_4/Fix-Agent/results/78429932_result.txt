from langchain_ollama import ChatOllama
from langchain_core.messages import SystemMessage, HumanMessage


def run_agent(user_prompt, system_prompt):
    # Use the chat-style wrapper (ChatOllama) instead of the raw completion
    # wrapper. ChatOllama takes structured messages and lets Ollama's built-in
    # Llama 3 chat template insert the special tokens (<|start_header_id|>,
    # <|eot_id|>, etc.) itself. Manually injecting those tokens into the text
    # caused them to be treated as literal content, so the model echoed the
    # system prompt and generated multiple repeated responses.
    llm = ChatOllama(
        model="llama3",
        # num_predict keeps the model from rambling on with extra turns;
        # adjust or remove if longer outputs are needed.
        num_predict=1024,
    )

    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    response = llm.invoke(messages)

    # ChatOllama returns a message object; extract the text content
    return response.content


if __name__ == "__main__":

            "You are a helpful assistant.",
