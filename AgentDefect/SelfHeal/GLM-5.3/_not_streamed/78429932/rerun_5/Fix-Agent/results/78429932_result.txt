from langchain_ollama import ChatOllama
from langchain_core.messages import SystemMessage, HumanMessage


def run_agent(user_prompt, system_prompt):
    # Use the chat model class (ChatOllama) instead of the raw completion
    # class (Ollama). ChatOllama builds the conversation using properly
    # typed messages, so Ollama applies the model's built-in Llama 3 chat
    # template itself. Manually injecting special tokens like
    # <|begin_of_text|> / <|start_header_id|> / <|eot_id|> is what caused
    # the model to echo the system prompt and generate multiple repeated
    # responses, because those tokens were treated as literal text and the
    # model never recognized a proper end of turn.
    llm = ChatOllama(model="llama3")

    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    response = llm.invoke(messages)

    # ChatOllama returns an AIMessage object; extract the text content so
    # the caller receives a single clean response string.
    return response.content


if __name__ == "__main__":

            "models like yourself are built, including data, training, "
            "architecture, and deployment. Use examples and provide a "
            "comprehensive overview.",
