from langchain_community.chat_models import ChatOllama
from langchain_core.messages import SystemMessage, HumanMessage


def run_agent(user_prompt, system_prompt):
    # Use the chat model wrapper (ChatOllama) instead of the raw-text
    # Ollama completion class. ChatOllama routes the request through
    # Ollama's /api/chat endpoint, which applies Llama 3's native chat
    # template (the <|begin_of_text|> / <|start_header_id|> / <|eot_id|>
    # tokens) exactly once and correctly honors the end-of-turn token.
    llm = ChatOllama(model="llama3")

    # Pass system and user inputs as structured chat messages with the
    # proper roles instead of hand-writing special tokens into a raw
    # prompt string, which caused duplicated headers, an echoed system
    # prompt, and the model generating multiple "responses".
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    response = llm.invoke(messages)

    # ChatOllama returns an AIMessage; extract the text content so the
    # caller receives the single, clean response string.
    return response.content


if __name__ == "__main__":
