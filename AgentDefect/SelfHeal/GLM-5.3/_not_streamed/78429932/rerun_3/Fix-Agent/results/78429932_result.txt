from langchain_core.messages import HumanMessage, SystemMessage

# Prefer the dedicated langchain-ollama package; fall back to the
# community integration for older LangChain installs.
try:
    from langchain_ollama import ChatOllama
except ImportError:  # pragma: no cover - fallback for legacy installs
    from langchain_community.chat_models import ChatOllama


def run_agent(user_prompt: str, system_prompt: str) -> str:
    """
    Send a single, correctly structured chat conversation to Llama 3 via
    Ollama and return exactly one clean response.

    Why this fixes the bug:
    - We no longer hand-craft Llama 3 special tokens (<|begin_of_text|>,
      <|start_header_id|>, <|eot_id|>) inside the raw prompt. Ollama already
      applies the llama3 chat template itself, so embedding them manually
      caused the template to be applied twice, producing malformed/nested
      tokens that confused the model.
    - Using ChatOllama with SystemMessage/HumanMessage lets Ollama format
      each turn correctly, including terminating the assistant turn with
      <|eot_id|>. That gives the model a clean stop point, so it returns a
      single response instead of generating endless extra system/user/
      assistant turns that echoed the system prompt.
    """
    llm = ChatOllama(model="llama3")

    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    response = llm.invoke(messages)
    return response.content
