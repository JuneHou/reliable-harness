from langchain_core.messages import SystemMessage, HumanMessage

try:
    # Preferred: current, dedicated Ollama chat-model package
    from langchain_ollama import ChatOllama
except ImportError:
    # Fallback for older LangChain setups where the chat model lives
    # in langchain_community
    from langchain_community.chat_models import ChatOllama


def run_agent(user_prompt, system_prompt):
    """
    Invoke Llama 3 via Ollama using proper chat message roles.

    The fix: previously, Llama 3 special tokens (<|begin_of_text|>,
    <|start_header_id|>, <|eot_id|>) were manually pasted into a raw
    completion prompt. Ollama already applies the model's own chat
    template, so those tokens were treated as plain text inside the
    user message. The model then imitated the markup and produced
    repeated responses that echoed the system prompt.

    Using ChatOllama with SystemMessage/HumanMessage lets Ollama build
    the correctly formatted prompt itself, yielding a single clean
    assistant response.
    """
    llm = ChatOllama(model="llama3")

    messages = [
        SystemMessage(content=system_prompt.strip()),
        HumanMessage(content=user_prompt.strip()),
    ]

    response = llm.invoke(messages)

    # ChatOllama returns a message object; extract the text content
    return response.content


if __name__ == "__main__":
