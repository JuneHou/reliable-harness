from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000

# Proper conversation history using LangChain message objects:
# one SystemMessage for persistent behaviour instructions,
# followed by alternating HumanMessage / AIMessage turns.
messages = [
    SystemMessage(content="You are a helpful assistant. Respond only to the "
                          "user's latest message. If the user asks you to drop "
                          "or change an earlier instruction, follow the newest "
                          "instruction instead of repeating previous topics.")
]

llm = ChatOllama(
    model=MODEL,
    temperature=0.7,   # >1.0 invites repetition loops; keep it sane
    stream=True
)

def get_number_of_tokens(messages):
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total_tokens = 0
        for message in messages:
            total_tokens += len(encoding.encode(str(message.content)))
        return total_tokens
    except Exception:
        total_chars = sum(len(str(message.content)) for message in messages)
        return total_chars // 4

@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    messages.append(HumanMessage(content=request.form['text']))

    # Trim oldest turns (never the system prompt) so Ollama receives a
    # bounded, correctly-interleaved dialogue window.
    number_of_tokens = get_number_of_tokens(messages)
    while number_of_tokens >= MODEL_MAX_TOKENS and len(messages) > 1:
        # Drop one full turn (human + its assistant reply) after the system message
        del messages[1]
        if len(messages) > 1:
            del messages[1]
        number_of_tokens = get_number_of_tokens(messages)  # recompute to avoid infinite loop

    ai_msg = llm.invoke(messages)

    # KEY FIX: store the model's reply back into the history as an
    # assistant turn. Without this, every request looks like one giant
    # ever-growing user monologue, which makes the Ollama server-side
    # context repeat the conversation history and cling to the first
    # prompt's instructions.
    messages.append(AIMessage(content=ai_msg.content))

    return ai_msg.content

@app.route('/')
def index():
    return render_template('index.html')

if __name__ == '__main__':
    app.run(debug=True, port=5000)
