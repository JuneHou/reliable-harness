from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000
# Conversation history. Must alternate: user, assistant, user, assistant...
# Storing only user turns (the old bug) produces consecutive user messages,
# which Llama's chat template does not expect, so the model echoes the
# history and keeps following the first prompt's instructions.
messages = []

llm = ChatOllama(
    model=MODEL,
    temperature=0.7,  # 1.1 amplified repetitive/erratic output
)


def get_number_of_tokens(history):
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total_tokens = 0
        for msg in history:
            total_tokens += len(encoding.encode(str(msg.content)))
        return total_tokens
    except Exception:
        total_chars = sum(len(str(msg.content)) for msg in history)
        return total_chars // 4


def trim_history():
    # Recompute the token count after every removal so the loop terminates
    # (the old version computed it once and looped forever / crashed with IndexError).
    number_of_tokens = get_number_of_tokens(messages)
    while number_of_tokens >= MODEL_MAX_TOKENS and len(messages) > 2:
        # Remove the oldest (user, assistant) pair to keep the history alternating.
        del messages[:2]
        number_of_tokens = get_number_of_tokens(messages)


@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_input = request.form['text']
    messages.append(HumanMessage(content=user_input))

    trim_history()

    ai_msg = llm.invoke(messages)

    # Record the model's reply so the next request contains proper
    # user/assistant alternation and the model knows what it already said.
    messages.append(AIMessage(content=ai_msg.content))

    return ai_msg.content


@app.route('/')
def index():
    return render_template('index.html')


if __name__ == '__main__':
    app.run(debug=True, port=5000)
