from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000

# Start with a system message and build a properly ALTERNATING
# user/assistant history. The original code only ever appended
# ("user", text) tuples and never recorded the model's own replies,
# so the model received a malformed transcript (consecutive user
# turns, no assistant turns). Llama3.2 responds to that by replaying
# the history and stubbornly continuing the FIRST instruction.
# Keeping an explicit system message also lets later user turns
# override earlier instructions.
SYSTEM_PROMPT = (
    "You are a helpful assistant. Follow the user's MOST RECENT "
    "instruction. When the user changes or cancels a previous "
    "instruction, abandon it and answer the new request. Never "
    "repeat or echo the conversation history in your reply."
)

messages = [SystemMessage(content=SYSTEM_PROMPT)]

# Llama 3.2's recommended sampling range is ~0.6-0.8; the original
# temperature=1.1 exceeds it and amplifies repetitive output.
llm = ChatOllama(
    model=MODEL,
    temperature=0.7,
    stream=True
)

def get_number_of_tokens(message_list):
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total_tokens = 0
        for msg in message_list:
            total_tokens += len(encoding.encode(str(msg.content)))
        return total_tokens
    except Exception:
        total_chars = sum(len(str(msg.content)) for msg in message_list)
        return total_chars // 4

@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_text = request.form['text']
    messages.append(HumanMessage(content=user_text))

    # Recompute the token count on every iteration (the original code
    # computed it once before the loop, so the condition never changed
    # and the loop deleted items forever / crashed with IndexError).
    # Keep the system message at index 0 while trimming oldest turns.
    while (get_number_of_tokens(messages) >= MODEL_MAX_TOKENS
           and len(messages) > 2):
        del messages[1]

    ai_msg = llm.invoke(messages)

    # THE KEY FIX: record the assistant's reply in the history so the
    # next request contains a real, alternating user/assistant
    # conversation instead of a stack of bare user prompts.
    messages.append(AIMessage(content=ai_msg.content))

    return ai_msg.content

@app.route('/')
def index():
    return render_template('index.html')

if __name__ == '__main__':
    app.run(debug=True, port=5000)
