from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000
messages = []

llm = ChatOllama(
    model=MODEL,
    temperature=1.1,
    stream=True
)

def get_number_of_tokens(messages):
    """Estimate the token count of the conversation history.

    The history is a list of LangChain message objects (HumanMessage /
    AIMessage), so each entry's text lives in `message.content`.
    """
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total_tokens = 0
        for message in messages:
            total_tokens += len(encoding.encode(str(message.content)))
        return total_tokens
    except Exception:
        total_chars = sum(len(str(message.content)) for message in messages)
        return total_chars // 4

@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    # Store the user input as a proper LangChain HumanMessage object
    # (instead of a raw ("user", text) tuple).
    messages.append(HumanMessage(content=request.form['text']))

    # Trim the history (always keeping the very first message) until we are
    # back under the model's context limit. Tokens are re-counted after every
    # removal so the loop terminates safely.
    while len(messages) > 1 and get_number_of_tokens(messages) >= MODEL_MAX_TOKENS:
        del messages[1]

    # Invoke the model with a proper list of BaseMessage objects.
    ai_msg = llm.invoke(messages)

    # Record the model's own reply as an AIMessage so that subsequent requests
    # contain an alternating user/assistant transcript, exactly like
    # `ollama run` does. Without this the model only ever saw consecutive
    # user messages, which made it repeat the conversation history and keep
    # following the first prompt's instructions.
    messages.append(AIMessage(content=ai_msg.content))

    return ai_msg.content

@app.route('/')
def index():
    return render_template('index.html')

if __name__ == '__main__':
    app.run(debug=True, port=5000)
