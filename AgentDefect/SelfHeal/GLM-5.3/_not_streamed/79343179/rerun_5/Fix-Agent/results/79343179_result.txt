import pandas as pd
import uuid
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# GLOBAL ARTIFACT STORE
# The LLM agent's output is always a *string* (textual summary / final answer),
# so a live DataFrame object can never travel through `agent_executor.invoke()`.
# Instead, the tool stashes the real DataFrame here and returns only a stable
# handle (session_id) to the agent. The main script then retrieves the actual
# object from the store using that handle.
# ---------------------------------------------------------------------------
_RESULTS_STORE = {}  # session_id -> {'df': DataFrame, 'path': str}


@tool
def load_excel_response(session_id: str, filepath: str):
    """Load an Excel file, transform it into a DataFrame, and store the real
    DataFrame object in the results store under the given session_id.
    Returns a short textual handle/summary (the object itself is stored, not
    returned, because agent tool outputs must be strings)."""
    df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
    df = df.select_dtypes(include=['object'])

    # Store the ACTUAL DataFrame object (not a serialization of it)
    _RESULTS_STORE[session_id] = {'df': df, 'path': filepath}

    # Return only a compact, serializable summary the agent can read/reason about
    return (
        f"Stored DataFrame with session_id='{session_id}'. "
        f"shape={df.shape}, columns={list(df.columns)}. "
        f"Use get_dataframe_summary('{session_id}') to inspect it."
    )


@tool
def get_dataframe_summary(session_id: str):
    """Return a textual summary of the DataFrame stored under the given session_id."""
    entry = _RESULTS_STORE.get(session_id)
    if entry is None:
        return f"No DataFrame stored for session_id='{session_id}'."
    df = entry['df']
    return (
        f"DataFrame for session_id='{session_id}': shape={df.shape}, "
        f"columns={list(df.columns)}, dtypes={dict(df.dtypes.astype(str))}, "
        f"head:\n{df.head().to_string()}"
    )


def run_agent(user_prompt, file_name="data.json"):
    session_id = str(uuid.uuid4())  # unique handle for this run

    llm = ChatOpenAI(model='gpt-4o-mini')
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    # Pre-bind the session_id so the agent only needs to supply the filepath
    load_tool = Tool(
        name="load_excel_response",
        func=lambda filepath: load_excel_response.invoke(
            {"session_id": session_id, "filepath": filepath}
        ),
        description="Load an Excel file, transform it, and store the resulting "
                    "DataFrame object for later retrieval by session_id.",
    )

    tools_list = [load_tool, get_dataframe_summary, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # Run the agent; note the agent can only ever see/return the textual summary
    response = agent_executor.invoke({"input": user_prompt})

    # Retrieve the REAL DataFrame object created during tool execution
    df = _RESULTS_STORE.get(session_id, {}).get('df', None)

    return {'response': response, 'df': df, 'session_id': session_id}


if __name__ == "__main__":
    path = 'excel_file.xlsx'
    user_prompt = (
        f"Load the Excel file located at {path} and assign the result to a "
        f"variable named df."
    )
    result = run_agent(user_prompt)

    agent_response = result['response']
    df = result['df']

    print("=== Agent textual response ===")
    print(agent_response)

    print("\n=== Actual DataFrame retrieved as a local variable ===")
    print(type(df))            # <class 'pandas.core.frame.DataFrame'>
    print(df.head())           # the real object, not a textual summary
