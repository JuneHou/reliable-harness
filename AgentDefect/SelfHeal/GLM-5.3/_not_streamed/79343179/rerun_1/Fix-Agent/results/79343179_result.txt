import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# Shared, module-level state.
#
# The ReAct agent loop only passes *strings* between the LLM and the tools,
# and `agent_executor.invoke()` returns a dict whose 'output' value is just
# the final textual summary. A DataFrame object created inside a tool's
# execution scope is therefore lost when the tool returns -- it never
# reaches the caller as an actual object.
#
# To retrieve the *real* DataFrame object in the main script, the tool stores
# it in this shared state and returns only a compact textual preview for the
# LLM. After the agent run, the caller reads the object back from here.
# ---------------------------------------------------------------------------
SHARED_STATE = {}


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a transformed DataFrame."""
    try:
        df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
        df = df.select_dtypes(include=['object'])
    except Exception as e:
        return f"Error loading '{filepath}': {e}"

    # Keep the real object so the main script can retrieve it later.
    SHARED_STATE['df'] = df

    # The agent loop only understands strings: give the LLM a compact
    # preview of the DataFrame instead of the raw object.
    return (
        f"DataFrame loaded: {len(df)} rows x {df.shape[1]} columns. "
        f"Preview: {df.head().to_json(orient='records')}"
    )


def run_agent(user_prompt, file_name="data.json"):
    llm = ChatOpenAI(model='gpt-4o-mini')
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    # Keep a handle on the REPL's globals so that a DataFrame created
    # through python_repl (e.g. `df = ...`) can also be recovered.
    repl_globals = getattr(python_repl, "_globals", None)

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(
        agent=shell_agent,
        tools=tools_list,
        verbose=True,
        handle_parsing_errors=True,
    )

    response = agent_executor.invoke({"input": user_prompt})

    # Attach the *actual* DataFrame object created during tool execution
    # to the response, so the caller receives the real object and not just
    # the textual summary in response['output'].
    df = SHARED_STATE.get('df')
    if df is None and isinstance(repl_globals, dict):
        # Fallback: the agent may have created `df` via the python_repl tool.
        df = repl_globals.get('df')
    response['df'] = df

    return response


if __name__ == "__main__":
    path = 'excel_file.xlsx'
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."
    result = run_agent(user_prompt)

    # Textual summary returned by the agent (original behaviour)
    print(result)

    # The actual DataFrame object, now available as a local variable
    df = result.get('df')
    if df is not None:
        print(f"\nRetrieved object type: {type(df)}")
        print(df.head())
    else:
        print("\nNo DataFrame was created during the agent run.")
