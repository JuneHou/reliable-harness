import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# Shared object store (module-level scope)
#
# LangChain agents can only pass *strings* between the LLM, the tools, and
# agent_executor.invoke(): every tool result is coerced to text before the LLM
# sees it, and invoke() only ever returns {"input": ..., "output": <text>}.
# A DataFrame returned by a tool therefore can never travel back through the
# agent's textual return value -- it gets stringified into a summary.
#
# Fix: keep a reference to the real DataFrame in a store living at module
# scope, have the tool return only a short textual confirmation to the LLM,
# and read the actual object back out of the store in the main script after
# the agent has finished running.
# ---------------------------------------------------------------------------
_SHARED_OBJECTS: dict = {}


def store_object(name: str, obj) -> None:
    """Keep a reference to a real Python object so it survives the agent run."""
    _SHARED_OBJECTS[name] = obj


def get_stored_object(name: str):
    """Retrieve a previously stored object (e.g. the DataFrame named 'df')."""
    return _SHARED_OBJECTS.get(name)


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file, transform it, and store the DataFrame as variable 'df'."""
    try:
        df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
        df = df.select_dtypes(include=['object'])
    except Exception as exc:
        # Errors must also be reported to the LLM as plain text.
        return f"Error while loading '{filepath}': {exc}"

    # Store the ACTUAL DataFrame object (by reference) instead of letting the
    # agent framework stringify it. The LLM only ever sees the short message,
    # which is both token-cheap and parseable.
    store_object("df", df)

    return (
        f"Loaded '{filepath}' and stored the transformed DataFrame as the "
        f"variable 'df' ({len(df)} rows, {len(df.columns)} column(s))."
    )


def run_agent(user_prompt, file_name="data.json"):
    llm = ChatOpenAI(model='gpt-4o-mini')
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description=(
            "Run Python commands in a REPL environment. Variables created "
            "here persist in the REPL session and can be inspected after the run."
        ),
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    response = agent_executor.invoke({"input": user_prompt})

    # Fallback: if the agent assigned 'df' via the python_repl tool instead of
    # the loader tool, the object is still reachable through the REPL's own
    # persistent namespace dicts (no stringification happens there either).
    if get_stored_object("df") is None:
        for scope in ("globals", "locals"):
            scope_dict = getattr(python_repl, scope, None) or {}
            df_from_repl = scope_dict.get("df")
            if df_from_repl is not None:
                store_object("df", df_from_repl)
                break

    # Hand back BOTH the agent's textual answer AND the real objects produced
    # during execution (added outside the LLM loop, so it is never stringified).
    response["shared_objects"] = dict(_SHARED_OBJECTS)
    return response


if __name__ == "__main__":
    path = 'excel_file.xlsx'
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."

    result = run_agent(user_prompt)

    # --- the agent's textual summary (all the LLM can ever produce) ---
    print("Agent output:", result["output"])

    # --- the actual DataFrame object, now a local variable in this script ---
    df = result["shared_objects"].get("df")

    if df is not None:
        print("Retrieved object type:", type(df))
        print(df.head())
    else:
        print("No DataFrame named 'df' was produced during the agent run.")

import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# Side-channel store for real Python objects.
#
# LangChain agents (ReAct / AgentExecutor) operate on a purely TEXTUAL loop:
# every tool observation is serialized to a string before it is fed back to
# the LLM, so agent_executor.invoke() can never return a live DataFrame
# object directly.  The actual DataFrame must therefore be smuggled out of
# the tool through a shared, module-level store that the main script reads
# AFTER the agent has finished running.
# ---------------------------------------------------------------------------
DATAFRAME_STORE = {}


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a transformed DataFrame.

    The DataFrame object itself is stored in the module-level
    DATAFRAME_STORE under the key 'df' so that the calling script can
    retrieve the real pandas object after the agent run completes.
    """
    df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
    df = df.select_dtypes(include=['object'])

    # Store the REAL DataFrame object in the shared side-channel store.
    DATAFRAME_STORE['df'] = df

    # Tool observations must be plain strings for the agent loop, so return
    # a compact textual summary instead of the (unserializable) DataFrame.
    return (
        f"Successfully loaded DataFrame with shape {df.shape} from '{filepath}'. "
        f"The actual DataFrame object is available in DATAFRAME_STORE['df']. "
        f"Preview:\n{df.head().to_string()}"
    )


def run_agent(user_prompt, file_name="data.json"):
    llm = ChatOpenAI(model='gpt-4o-mini')
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # Start a fresh agent run so stale DataFrames never leak between calls.
    DATAFRAME_STORE.clear()

    response = agent_executor.invoke({"input": user_prompt})

    # Attach the actual DataFrame object (not just the textual summary) to
    # the agent's response dictionary so the caller receives a real object.
    response['dataframe'] = DATAFRAME_STORE.get('df')

    return response


if __name__ == "__main__":
    path = 'excel_file.xlsx'
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."
    result = run_agent(user_prompt)

    # `df` below is the genuine pandas DataFrame object that was created
    # inside the tool during agent execution.
    df = result['dataframe']

    print("Agent textual output:")
    print(result.get('output'))
    print("\nRetrieved object type:", type(df))
    if df is not None:
        print("\nDataFrame head:")
        print(df.head())
    else:
        print("\nNo DataFrame was produced during the agent run.")
