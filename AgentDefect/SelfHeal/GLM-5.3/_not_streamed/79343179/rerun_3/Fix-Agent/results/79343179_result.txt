import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# Shared in-memory store for DataFrame objects created during tool execution.
#
# Why this is needed:
#   AgentExecutor.invoke() only returns the LLM's final *textual* answer.
#   Any non-string value returned by a tool is serialized to a string before
#   the agent sees it, so returning a raw DataFrame from the tool loses the
#   actual object. The real object must therefore be parked somewhere the
#   calling script can reach (this module-level dict), while the tool returns
#   only a short textual summary for the LLM to reason over.
# ---------------------------------------------------------------------------
DATAFRAME_STORE = {}


@tool
def load_excel_response(filepath: str, var_name: str = "df") -> str:
    """Load an Excel file into a transformed DataFrame and store it as a variable.

    The resulting DataFrame object is saved in a shared store under `var_name`
    so the calling script can retrieve it after the agent finishes.

    Args:
        filepath: Path to the Excel file to load.
        var_name: Name of the variable to store the DataFrame under (default 'df').

    Returns:
        A short textual summary of the DataFrame (shape/columns) for the agent.
    """
    try:
        df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
        df = df.select_dtypes(include=['object'])

        # Store the ACTUAL DataFrame object, not a string representation of it.
        DATAFRAME_STORE[var_name] = df

        # Return only a compact text summary: the agent/LLM works with strings.
        return (
            f"Successfully loaded '{filepath}' into a DataFrame stored as variable "
            f"'{var_name}'. Shape: {df.shape[0]} rows x {df.shape[1]} columns. "
            f"Columns: {list(df.columns)}. "
            f"The DataFrame object is available in the shared store under key '{var_name}'."
        )
    except Exception as e:
        return f"Error loading Excel file '{filepath}': {e}"


def run_agent(user_prompt, file_name="data.json"):
    llm = ChatOpenAI(model='gpt-4o-mini')
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # Start from a clean store for this run.
    DATAFRAME_STORE.clear()

    response = agent_executor.invoke({"input": user_prompt})

    # Attach the real DataFrame objects created during execution to the
    # response so callers can access them directly alongside the text output.
    response["dataframes"] = dict(DATAFRAME_STORE)
    return response


if __name__ == "__main__":
    path = 'excel_file.xlsx'
    user_prompt = (
        f"Load the Excel file located at {path} and assign the result to a variable named df."
    )
    result = run_agent(user_prompt)

    # Retrieve the actual DataFrame object created during tool execution.
    df = result.get("dataframes", {}).get("df")
    if df is not None:
        print(f"Retrieved object type: {type(df)}")  # <class 'pandas.core.frame.DataFrame'>
        print("First rows of the DataFrame:")
        print(df.head())
    else:
        print("No DataFrame was produced during agent execution.")

    # The agent's textual answer is still available as usual.
    print("Agent output:", result.get("output"))
