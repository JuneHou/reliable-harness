import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# Shared store for real Python objects created during tool execution.
#
# Why this is needed: the ReAct agent loop serializes EVERY tool output to a
# string before it is shown back to the LLM (as the "Observation" of the
# Thought/Action/Observation steps).  Therefore a DataFrame object returned
# from a tool can never survive agent_executor.invoke() -- only a textual
# rendering of it does.  The only reliable way to hand the real object back
# to the calling script is for the tool to stash it in a shared location
# that exists outside the agent loop, and for the caller to read it from
# there after the agent finishes.
# ---------------------------------------------------------------------------
DATAFRAME_STORE = {}


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file, store the transformed DataFrame in the shared store, and return a short textual summary."""
    df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
    df = df.select_dtypes(include=['object'])

    # Persist the REAL DataFrame object outside the agent loop so the
    # calling script can retrieve it after agent_executor.invoke() returns.
    DATAFRAME_STORE['df'] = df

    # The LLM only ever sees strings, so return a compact summary instead of
    # the DataFrame itself (avoids dumping the whole table into the prompt).
    return (
        f"Loaded DataFrame 'df' from {filepath} with shape {df.shape} "
        f"and columns {list(df.columns)}. The DataFrame object is stored in "
        f"the shared store under key 'df'."
    )


def run_agent(user_prompt, file_name="data.json"):
    llm = ChatOpenAI(model='gpt-4o-mini')
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # Clear any stale result from a previous run so we always retrieve the
    # DataFrame produced by THIS invocation.
    DATAFRAME_STORE.pop('df', None)

    # The invoke() call returns only the LLM's textual answer:
    # {"input": ..., "output": "..."} -- never the DataFrame itself.
    response = agent_executor.invoke({"input": user_prompt})

    # Retrieve the actual DataFrame object created during tool execution.
    df = DATAFRAME_STORE.get('df', None)

    # Fallback: if the agent assigned the DataFrame via the python_repl tool
    # instead (e.g. it ran `df = load_excel_response(...)` itself), the real
    # object lives in the REPL's namespace, which we can also reach directly.
    if df is None:
        repl_globals = getattr(python_repl, "globals", None)
        if isinstance(repl_globals, dict):
            candidate = repl_globals.get('df', None)
            if isinstance(candidate, pd.DataFrame):
                df = candidate

    # Return BOTH the agent's textual response AND the real DataFrame object,
    # so the caller can bind it to a local variable in the main script.
    return response, df


if __name__ == "__main__":
    path = 'excel_file.xlsx'
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."
    result, df = run_agent(user_prompt)

    # `result` is only the agent's textual summary; `df` is the real
    # DataFrame object created inside the tool, now a local variable here.
    print(result)
    if df is not None:
        print(f"Retrieved DataFrame with shape: {df.shape}")
        print(df.head())
    else:
        print("No DataFrame was produced during agent execution.")
