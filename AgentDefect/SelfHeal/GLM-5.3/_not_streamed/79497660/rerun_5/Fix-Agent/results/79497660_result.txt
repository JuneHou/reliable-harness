Root cause
----------

generates internally (``_get_data_model``). That model only knows the
``id`` / ``source_id`` / ``text`` / ``metadata_`` / ``embedding`` columns, and

    """A ``PGVectorStore`` that keeps ``external_id`` in its own table column."""

    # ------------------------------------------------------------------ #
    # 1) Schema: give the generated table a real `external_id` column    #
    # ------------------------------------------------------------------ #

    def _get_data_model(self, base: Any) -> Any:
        """
        Reuse the SQLAlchemy model that ``PGVectorStore`` generates and extend
        it with a dedicated ``external_id`` column.
        """
        try:
            data_model = super()._get_data_model(base)
        except Exception:
            data_model = self._build_fallback_data_model(base)

        table = data_model.__table__ if hasattr(data_model, "__table__") else data_model
        if self.external_id_key not in table.c:
            table.append_column(Column(self.external_id_key, Text, nullable=True))
        return data_model

    def _build_fallback_data_model(self, base: Any) -> Any:
        """Model mirroring the default PGVectorStore schema."""
        from pgvector.sqlalchemy import Vector
        from sqlalchemy.dialects.postgresql import JSON, JSONB
        from sqlalchemy.orm import DeclarativeBase

        metadata_type = JSONB if self.use_jsonb else JSON
        embedding_type = Vector(self.embed_dim) if self.embed_dim else Vector()
        table_name = self.table_name or "llamaindex"

        def _define(parent: Any) -> Any:
            class Data(parent):  # type: ignore[misc, valid-type]
                __tablename__ = table_name
                id = Column(Text, primary_key=True)
                source_id = Column(Text, nullable=True)
                text = Column(Text)
                metadata_ = Column(metadata_type, default=dict, index=True)
                embedding = Column(embedding_type)

            return Data

        try:
            return _define(base)
        except Exception:
            # `base` is not a declarative base *class* (or the table is
            # already registered on it) -> build the model on our own base.
            class OwnBase(DeclarativeBase):
                pass

            return _define(OwnBase)

    def _create_tables(self) -> None:
        """Create the tables (including the external_id column)."""
        # Make sure the extended table exists before the parent method adds
        # its HNSW/JSONB indexes on top of it.
        table = getattr(self, "_table", None)
        engine = getattr(self, "_engine", None)

            table_obj = table.__table__ if hasattr(table, "__table__") else table
            try:
                table_obj.create(engine, checkfirst=True)
            except Exception:
                logger.debug("Table creation handled by the parent setup.")

        super()._create_tables()
        # Safety net for tables that were created without the column before.
        self.ensure_external_id_column()

                )
            except Exception:
                pass
        return table_obj

    def ensure_external_id_column(self) -> None:
        """Guarantee the external_id column physically exists in Postgres."""
        from sqlalchemy import text as sql_text

        table = self._get_table()
        engine = getattr(self, "_engine", None)
        if table is None or engine is None:
            return

        qualified_name = f"{schema_prefix}{table.name}"
        column = self.external_id_key

        try:

                        f"ADD COLUMN IF NOT EXISTS {column} TEXT"

                )

                        f"CREATE INDEX IF NOT EXISTS "
                        f"{qualified_name}_{column}_idx "
                        f"ON {qualified_name} ({column})"
                    )
                )
        except Exception as exc:
            logger.warning("Could not ensure the %s column: %s", column, exc)

    # 2) Writing: promote the metadata key into the dedicated column     #
    # ------------------------------------------------------------------ #

    def _extract_external_ids(self, nodes: List[BaseNode]) -> Dict[str, Any]:
        """
        Temporarily remove the external id from the nodes' metadata (so that
        it is *not* serialized into the ``metadata_`` JSON column) and return
        the mapping node_id -> external_id.
        """
        external_ids: Dict[str, Any] = {}
        for node in nodes:
            metadata = getattr(node, "metadata", None)
            if metadata and self.external_id_key in metadata:
                external_ids[node.node_id] = metadata.pop(self.external_id_key)
        return external_ids

    def _restore_external_ids(
        self, nodes: List[BaseNode], external_ids: Dict[str, Any]
    ) -> None:
        """Put the external ids back onto the in-memory node objects."""
        for node in nodes:
            if node.node_id in external_ids and node.metadata is not None:
                node.metadata[self.external_id_key] = external_ids[node.node_id]

    def _write_external_ids(self, external_ids: Dict[str, Any]) -> None:
        """Upsert the external ids into the dedicated table column."""
        from sqlalchemy.dialects.postgresql import insert as postgres_insert

        table = self._get_table()
        engine = getattr(self, "_engine", None)
        if table is None or engine is None:
            logger.warning(
                "Vector store is not initialized; %s values were not persisted.",
                self.external_id_key,
            )
            return

        primary_key = [column.name for column in table.primary_key.columns] or ["id"]
        rows = [
            {primary_key[0]: node_id, self.external_id_key: value}
            for node_id, value in external_ids.items()
            if value is not None
        ]
        if not rows:
            return

        try:
            statement = postgres_insert(table).values(rows)
            statement = statement.on_conflict_do_update(
                index_elements=primary_key,
                set_={self.external_id_key: statement.excluded[self.external_id_key]},
            )
            with engine.begin() as connection:
                connection.execute(statement)
        except Exception as exc:
            logger.warning("Could not persist %s values: %s", self.external_id_key, exc)

    def add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        """Insert nodes, storing external_id in its own column."""
        external_ids = self._extract_external_ids(nodes)
        try:
            node_ids = super().add(nodes, **add_kwargs)
        finally:
            self._restore_external_ids(nodes, external_ids)
        self._write_external_ids(external_ids)
        return node_ids

    async def async_add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        """Async version of ``add``."""
        external_ids = self._extract_external_ids(nodes)
        try:
            node_ids = await super().async_add(nodes, **add_kwargs)
        finally:
            self._restore_external_ids(nodes, external_ids)
        self._write_external_ids(external_ids)
        return node_ids

    # ------------------------------------------------------------------ #
    # 3) Reading: surface the column value on retrieved nodes            #
    # ------------------------------------------------------------------ #

    def _fetch_external_ids(self, node_ids: List[str]) -> Dict[str, Any]:
        """Read external_id values for the given node ids from the column."""
        from sqlalchemy import select

        table = self._get_table()
        engine = getattr(self, "_engine", None)
        if table is None or engine is None or not node_ids:
            return {}

        primary_key = [column.name for column in table.primary_key.columns] or ["id"]
        id_column = table.c[primary_key[0]]

        try:
            with engine.connect() as connection:
                rows = connection.execute(
                    select(id_column, table.c[self.external_id_key]).where(
                        id_column.in_(node_ids)
                    )
                ).fetchall()
            return {row[0]: row[1] for row in rows if row[1] is not None}
        except Exception as exc:
            logger.warning("Could not read %s values: %s", self.external_id_key, exc)
            return {}

    def _attach_external_ids(
        self, result: Optional[VectorStoreQueryResult]
    ) -> Optional[VectorStoreQueryResult]:
        """Copy external_id from the column back onto the result nodes."""
        if result is None or not getattr(result, "nodes", None):
            return result

        external_ids = self._fetch_external_ids(
            [node.node_id for node in result.nodes]
        )
        for node in result.nodes:
            external_id = external_ids.get(node.node_id)
            if external_id is not None:
                if node.metadata is None:
                    node.metadata = {}
                node.metadata[self.external_id_key] = external_id
        return result

    def query(self, query: VectorStoreQuery, **kwargs: Any) -> VectorStoreQueryResult:
        """Query the store and re-attach external_id to the returned nodes."""
        return self._attach_external_ids(super().query(query, **kwargs))

    async def async_query(
        self, query: VectorStoreQuery, **kwargs: Any
    ) -> VectorStoreQueryResult:
        """Async version of ``query``."""
        return self._attach_external_ids(await super().async_query(query, **kwargs))

        except Exception as exc:
            logger.warning("Could not ensure the %s column: %s", column, exc)

    # ------------------------------------------------------------------ #
    # 2) Writing: promote the metadata key into the dedicated column     #
    # ------------------------------------------------------------------ #

    def _extract_external_ids(self, nodes: List[BaseNode]) -> Dict[str, Any]:
        """
        Temporarily remove the external id from the nodes' metadata (so that
        it is *not* serialized into the ``metadata_`` JSON column) and return
        the mapping node_id -> external_id.
        """
        external_ids: Dict[str, Any] = {}
        for node in nodes:
            metadata = getattr(node, "metadata", None)
            if metadata and self.external_id_key in metadata:
                external_ids[node.node_id] = metadata.pop(self.external_id_key)
        return external_ids

    def _restore_external_ids(
        self, nodes: List[BaseNode], external_ids: Dict[str, Any]
    ) -> None:
        """Put the external ids back onto the in-memory node objects."""
        for node in nodes:
            if node.node_id in external_ids and node.metadata is not None:
                node.metadata[self.external_id_key] = external_ids[node.node_id]

    def _write_external_ids(self, external_ids: Dict[str, Any]) -> None:
        """Upsert the external ids into the dedicated table column."""
        from sqlalchemy.dialects.postgresql import insert as postgres_insert

        table = self._get_table()
        engine = getattr(self, "_engine", None)
        if table is None or engine is None:
            logger.warning(
                "Vector store is not initialized; %s values were not persisted.",
                self.external_id_key,
            )
            return

        primary_key = [column.name for column in table.primary_key.columns] or ["id"]
        rows = [
            {primary_key[0]: node_id, self.external_id_key: value}
            for node_id, value in external_ids.items()
            if value is not None
        ]
        if not rows:
            return

        try:
            statement = postgres_insert(table).values(rows)
            statement = statement.on_conflict_do_update(
                index_elements=primary_key,
                set_={self.external_id_key: statement.excluded[self.external_id_key]},
            )
            with engine.begin() as connection:
                connection.execute(statement)
        except Exception as exc:
            logger.warning("Could not persist %s values: %s", self.external_id_key, exc)

    def add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        """Insert nodes, storing external_id in its own column."""
        external_ids = self._extract_external_ids(nodes)
        try:
            node_ids = super().add(nodes, **add_kwargs)
        finally:
            self._restore_external_ids(nodes, external_ids)
        self._write_external_ids(external_ids)
        return node_ids

    async def async_add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        """Async version of ``add``."""
        external_ids = self._extract_external_ids(nodes)
        try:
            node_ids = await super().async_add(nodes, **add_kwargs)
        finally:
            self._restore_external_ids(nodes, external_ids)
        self._write_external_ids(external_ids)
        return node_ids

    # ------------------------------------------------------------------ #
    # 3) Reading: surface the column value on retrieved nodes            #
    # ------------------------------------------------------------------ #

    def _fetch_external_ids(self, node_ids: List[str]) -> Dict[str, Any]:
        """Read external_id values for the given node ids from the column."""
        from sqlalchemy import select

        table = self._get_table()
        engine = getattr(self, "_engine", None)
        if table is None or engine is None or not node_ids:
            return {}

        primary_key = [column.name for column in table.primary_key.columns] or ["id"]
        id_column = table.c[primary_key[0]]

        try:
            with engine.connect() as connection:
                rows = connection.execute(
                    select(id_column, table.c[self.external_id_key]).where(
                        id_column.in_(node_ids)
                    )
                ).fetchall()
            return {row[0]: row[1] for row in rows if row[1] is not None}
        except Exception as exc:
            logger.warning("Could not read %s values: %s", self.external_id_key, exc)
            return {}

    def _attach_external_ids(
        self, result: Optional[VectorStoreQueryResult]
    ) -> Optional[VectorStoreQueryResult]:
        """Copy external_id from the column back onto the result nodes."""
        if result is None or not getattr(result, "nodes", None):
            return result

        external_ids = self._fetch_external_ids(
            [node.node_id for node in result.nodes]
        )
        for node in result.nodes:
            external_id = external_ids.get(node.node_id)
            if external_id is not None:
                if node.metadata is None:
                    node.metadata = {}
                node.metadata[self.external_id_key] = external_id
        return result

    def query(self, query: VectorStoreQuery, **kwargs: Any) -> VectorStoreQueryResult:
        """Query the store and re-attach external_id to the returned nodes."""
        return self._attach_external_ids(super().query(query, **kwargs))

    async def async_query(
        self, query: VectorStoreQuery, **kwargs: Any
    ) -> VectorStoreQueryResult:
        """Async version of ``query``."""
        return self._attach_external_ids(await super().async_query(query, **kwargs))


def create_vector_store(request):
    """Create a PGVectorStore whose table has a dedicated external_id column."""
    vector_store = CustomPGVectorStore.from_params(
        database=request.tenant,
        host=PG_HOST,
        password=PG_PASSWORD,
        port=PG_PORT,
        user=PG_USER,
        table_name="vectorstore",
        embed_dim=1536,
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
    )

    # Safety net: adds the column to tables created by earlier runs that
    # stored external_id inside the metadata JSON column.
    vector_store.ensure_external_id_column()

    return vector_store


def run_agent(request):
    """Build the vector store, index the documents and return the index."""
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )

    return index


def verify_table_schema(request) -> None:
    """
    Connect with plain psycopg2 and print the real schema of the generated
    table, proving that `external_id` is a first-class column and that it is
    no longer hidden inside the `metadata_` JSON column.
    """
    connection = psycopg2.connect(
        host=PG_HOST,
        port=PG_PORT,
        user=PG_USER,
        password=PG_PASSWORD,
        dbname=request.tenant,
    )
    try:
        with connection.cursor() as cursor:
            cursor.execute(
                """
                SELECT table_schema, table_name
                FROM information_schema.tables
                WHERE table_name LIKE %s
                ORDER BY table_schema, table_name;
                """,
                ("%vectorstore%",),
            )
            matches = cursor.fetchall()

            if not matches:
                print("No vector store table found - nothing to verify.")
                return

            for table_schema, table_name in matches:
                qualified = f'"{table_schema}"."{table_name}"'

                cursor.execute(
                    """
                    SELECT column_name, data_type
                    FROM information_schema.columns
                    WHERE table_schema = %s AND table_name = %s
                    ORDER BY ordinal_position;
                    """,
                    (table_schema, table_name),
                )
                columns = cursor.fetchall()

                print(f"Schema of table {qualified}:")
                for column_name, data_type in columns:
                    print(f"  - {column_name} ({data_type})")

                column_names = [column_name for column_name, _ in columns]

                if "external_id" not in column_names:
                    print("FAILURE: 'external_id' is NOT a column of this table.")
                    continue

                print("SUCCESS: 'external_id' is a dedicated column of the table.")

                try:
                    cursor.execute(
                        f"SELECT id, external_id, text FROM {qualified} ORDER BY id;"
                    )
                    print(f"  Rows stored in {qualified}:")
                    for row_id, external_id, row_text in cursor.fetchall():
                        print(
                            f"    id={row_id} "
                            f"external_id={external_id} "
                            f"text={row_text[:32]!r}"
                        )
                except psycopg2.Error:
                    connection.rollback()

                try:
                    cursor.execute(
                        f"SELECT id, metadata_ FROM {qualified} ORDER BY id;"
                    )
                    for row_id, metadata_value in cursor.fetchall():
                        serialized = str(metadata_value)
                        print(
                            f"    row {row_id}: 'external_id' stored inside the "
                            f"metadata JSON column: {'external_id' in serialized}"
                        )
                except psycopg2.Error:
                    connection.rollback()
    finally:
        connection.close()
