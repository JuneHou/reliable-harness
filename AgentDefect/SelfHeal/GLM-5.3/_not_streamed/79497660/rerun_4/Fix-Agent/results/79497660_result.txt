        },

def add_custom_columns(request, table_name="vectorstore"):
    """
    Add the external_id as a real column on the Postgres table so that it
    can be queried/indexed directly instead of living inside the metadata_
    JSON column.
    """
    conn = psycopg2.connect(
        host="localhost",
        password="password",
        port=5432,
        user="postgres",
        dbname=request.tenant,
    )
    conn.autocommit = True
    cursor = conn.cursor()
    
    # Only add the column if it does not already exist (idempotent ALTER)
    cursor.execute(
        """
        SELECT column_name
        FROM information_schema.columns
        WHERE table_name = %s
          AND column_name = 'external_id';
        """,
        (table_name,),
    )
    exists = cursor.fetchone() is not None
    
    if not exists:
        cursor.execute(
            f"ALTER TABLE {table_name} ADD COLUMN external_id TEXT;"
        )
        # Optional: index for fast lookups by external_id
        cursor.execute(
            f"CREATE INDEX IF NOT EXISTS {table_name}_external_id_idx "
            f"ON {table_name} (external_id);"
        )
    
    cursor.close()
    conn.close()


def run_agent(request):
    # 1) Create the vector store (which creates the default table schema)
    vector_store = create_vector_store(request)
    
    # 2) Add the custom column(s) directly to the table schema
    add_custom_columns(request)
    
    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]
    
    # 3) Attach the external_id directly onto the node id field so that it
    #    lands in a dedicated column-friendly location and is also
    #    written into the new external_id column below.
    for doc in documents:
        doc.id_ = doc.metadata["external_id"]
    
    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store
    )
    
    # 4) Backfill/keep the external_id column in sync with the metadata
    #    after indexing (PGVectorStore does not natively map arbitrary
    #    metadata keys to table columns, so we populate it ourselves).
    sync_external_id_column(request)
    
    return index


def sync_external_id_column(request, table_name="vectorstore"):
    """
    Populate the external_id column from the metadata_ JSONB column so that
    external_id exists as a separate, queryable column in the table.
    """
    conn = psycopg2.connect(
        host="localhost",
        password="password",
        port=5432,
        user="postgres",
        dbname=request.tenant,
    )
    conn.autocommit = True
    cursor = conn.cursor()
    
    cursor.execute(
        f"""
        UPDATE {table_name}
        SET external_id = metadata_->>'external_id'
        WHERE metadata_ ? 'external_id'
          AND (external_id IS NULL OR external_id = '');
        """
    )
    
    updated = cursor.rowcount
    cursor.close()
    conn.close()
    return updated

CUSTOM_COLUMNS = {

def _connect(tenant):
    """Open a raw psycopg2 connection to the tenant database."""
    return psycopg2.connect(
        database=tenant,
        host=HOST,
        password=PASSWORD,
        port=PORT,
        user=USER,
    )


def add_custom_columns(request):
    """Add custom columns (e.g. external_id) to the PGVectorStore table.

    PGVectorStore has already created the table with its fixed columns
    (id, node_id, text, metadata_, embedding); we extend that schema with
    our custom columns so they can be filtered / joined on like any other
    column.
    """

    try:
        conn.autocommit = True
        with conn.cursor() as cursor:
            for column, col_type in CUSTOM_COLUMNS.items():
                cursor.execute(
                    f'ALTER TABLE {QUALIFIED_TABLE} '
                    f'ADD COLUMN IF NOT EXISTS "{column}" {col_type};'
                )
                # make lookups / filters on the custom column fast
                cursor.execute(
                    f'CREATE INDEX IF NOT EXISTS "idx_{TABLE_NAME}_{column}" '
                    f'ON {QUALIFIED_TABLE} ("{column}");'
                )
    finally:
        conn.close()

def promote_metadata_to_columns(request):
    """Copy custom metadata keys out of the metadata_ JSON column into the
    dedicated table columns (and optionally drop them from the JSON)."""
    if not CUSTOM_COLUMNS:
        return 0

    conn = _connect(request.tenant)
    updated = 0
    try:
        conn.autocommit = True
        with conn.cursor() as cursor:
            # e.g. "external_id" = metadata_ ->> 'external_id'
            set_sql = ", ".join(
                f'"{c}" = metadata_ ->> \'{c}\'' for c in CUSTOM_COLUMNS
            )
            # rows that still carry the key(s) inside the JSON ...
            has_key_sql = " OR ".join(
                f"metadata_ ->> '{c}' IS NOT NULL" for c in CUSTOM_COLUMNS
            )
            # ... and whose dedicated column is still empty

                f'"{c}" IS NULL' for c in CUSTOM_COLUMNS
            )
            cursor.execute(
                f"UPDATE {QUALIFIED_TABLE} SET {set_sql} "
                f"WHERE ({has_key_sql}) AND ({needs_value_sql});"
            )
            updated = cursor.rowcount

            if REMOVE_FROM_METADATA_JSON:
                keys = ", ".join(f"'{c}'" for c in CUSTOM_COLUMNS)
                cursor.execute(
                    f"UPDATE {QUALIFIED_TABLE} "
                    f"SET metadata_ = metadata_ - ARRAY[{keys}] "
                    f"WHERE metadata_ ?| ARRAY[{keys}];"
                )
    finally:
        conn.close()
    return updated
