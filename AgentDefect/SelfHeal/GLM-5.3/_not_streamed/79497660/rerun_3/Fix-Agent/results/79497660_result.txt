def _resolve_physical_table(database: str, logical_table: str) -> Tuple[str, str]:
    """Return ``(schema, physical_table)`` of the table LlamaIndex created.

    Older releases of ``llama-index-vector-stores-postgres`` prefixed the
    physical table name with ``data_``; newer ones use the name as given, so
    both candidates are checked.
    """
    candidates = [logical_table, f"data_{logical_table}"]
    with closing(_connect(database)) as conn:
        with conn:
            with conn.cursor() as cur:
                for candidate in candidates:
                    cur.execute(
                        """
                        SELECT table_schema
                          FROM information_schema.tables
                         WHERE table_name = %s
                         ORDER BY table_schema
                        """,
                        (candidate,),
                    )
                    schemas = [row[0] for row in cur.fetchall()]
                    if schemas:
                        schema = "public" if "public" in schemas else schemas[0]
                        return schema, candidate
    return "public", candidates[0]

class CustomSchemaPGVectorStore(PGVectorStore):
    """A ``PGVectorStore`` whose table also carries custom, typed columns.

    LlamaIndex generates the base table itself and has no parameter for extra
    columns, so the DDL it runs is extended right after the base table has
    been created.  Everything else (inserts, similarity queries, HNSW index,
    hybrid search, ...) keeps working unchanged.
    """

    extra_columns: ClassVar[Sequence[Tuple[str, str]]] = CUSTOM_COLUMNS

    # -- table naming helpers -------------------------------------------------
    @property
    def _table_sql_name(self) -> str:
        """Physical table name as it is stored in Postgres."""
        for attr in ("_table_name", "table_name"):
            value = getattr(self, attr, None)
            if isinstance(value, str) and value:
                return value
        return TABLE_NAME

    @property
    def _schema_sql_name(self) -> str:
        for attr in ("_schema_name", "schema_name"):
            value = getattr(self, attr, None)
            if isinstance(value, str) and value:
                return value
        return "public"

    @property
    def _qualified_table(self) -> str:
        return f'"{self._schema_sql_name}"."{self._table_sql_name}"'

    # -- DDL -------------------------------------------------------------------
    def _create_tables_if_not_exists(self, connection: Any) -> None:
        """Create the base table (LlamaIndex) and then add the custom columns."""
        parent_create = getattr(super(), "_create_tables_if_not_exists", None)
        if callable(parent_create):
            parent_create(connection)

        for column_name, column_type in self.extra_columns:
            connection.execute(
                text(
                    f"ALTER TABLE {self._qualified_table} "
                    f'ADD COLUMN IF NOT EXISTS "{column_name}" {column_type}'
                )
            )
            connection.execute(
                text(
                    f'CREATE INDEX IF NOT EXISTS "{self._table_sql_name}_{column_name}_idx" '
                    f'ON {self._qualified_table} ("{column_name}")'
                )
            )

def ensure_custom_columns(
    database: str,
    logical_table: str = TABLE_NAME,
    columns: Sequence[Tuple[str, str]] = CUSTOM_COLUMNS,
) -> None:
    """Idempotently guarantee that the custom columns exist on the table.

    ``ADD COLUMN IF NOT EXISTS`` makes this safe to re-run and also works for
    tables that LlamaIndex already created in a previous run.
    """
    schema, table = _resolve_physical_table(database, logical_table)
    qualified = f'"{schema}"."{table}"'

    with closing(_connect(database)) as conn:
        with conn:
            with conn.cursor() as cur:
                for column_name, column_type in columns:
                    cur.execute(
                        f'ALTER TABLE {qualified} '
                        f'ADD COLUMN IF NOT EXISTS "{column_name}" {column_type}'
                    )
                    cur.execute(
                        f'CREATE INDEX IF NOT EXISTS "{table}_{column_name}_idx" '
                        f'ON {qualified} ("{column_name}")'
                    )
    logger.info("custom columns %s ensured on %s", [c for c, _ in columns], qualified)


def sync_metadata_to_custom_columns(
    database: str,
    logical_table: str = TABLE_NAME,
    columns: Sequence[Tuple[str, str]] = CUSTOM_COLUMNS,
    remove_from_metadata: bool = MOVE_OUT_OF_METADATA,
) -> Dict[str, int]:
    """Promote custom keys out of the ``metadata_`` JSON into their own columns.

    ``PGVectorStore.add()`` only knows its fixed columns, so anything placed in
    ``Document.metadata`` ends up JSON-serialised inside ``metadata_``.  After
    indexing we therefore copy each custom key into its dedicated column and
    (matching the requirement "instead of") delete it from the JSON blob.
    """
    schema, table = _resolve_physical_table(database, logical_table)
    qualified = f'"{schema}"."{table}"'
    moved: Dict[str, int] = {}
    column_names = [column for column, _ in columns]

    with closing(_connect(database)) as conn:
        with conn:
            with conn.cursor() as cur:
                # `->>` works for both `json` and `jsonb` columns.
                for column_name in column_names:
                    cur.execute(
                        f"""
                        UPDATE {qualified}
                           SET "{column_name}" = metadata_ ->> %s
                         WHERE "{column_name}" IS NULL
                           AND metadata_ ->> %s IS NOT NULL
                        """,
                        (column_name, column_name),
                    )
                    moved[column_name] = cur.rowcount

                if remove_from_metadata:
                    cur.execute(f"SELECT id, metadata_ FROM {qualified}")
                    rows = cur.fetchall()
                    for row_id, metadata in rows:
                        if isinstance(metadata, str):
                            try:
                                metadata = json.loads(metadata)
                            except (TypeError, ValueError):
                                continue
                        if not isinstance(metadata, dict):
                            continue
                        if any(column in metadata for column in column_names):
                            for column in column_names:
                                metadata.pop(column, None)
                            cur.execute(
                                f"UPDATE {qualified} SET metadata_ = %s WHERE id = %s",
                                (psycopg2.extras.Json(metadata), row_id),
                            )
    return moved

                    SELECT column_name, data_type
                      FROM information_schema.columns
                     WHERE table_schema = %s AND table_name = %s
                     ORDER BY ordinal_position
                    """,
                    (schema, table),
                )
                return [dict(row) for row in cur.fetchall()]


def fetch_rows(database: str, logical_table: str = TABLE_NAME, limit: int = 10):

def lookup_external_ids(database: str, node_ids: Sequence[str]) -> Dict[str, str]:
    """Read ``external_id`` from the dedicated column for the given node ids."""
    if not node_ids:
        return {}
    schema, table = _resolve_physical_table(database, TABLE_NAME)
    with closing(_connect(database)) as conn:
        with conn:
            with conn.cursor() as cur:
                cur.execute(
                    f'SELECT node_id, external_id FROM "{schema}"."{table}" '
                    f"WHERE node_id = ANY(%s)",
                    (list(node_ids),),
                )
                return {row[0]: row[1] for row in cur.fetchall()}

                        if isinstance(metadata, str):
                            try:
                                metadata = json.loads(metadata)
                            except (TypeError, ValueError):
                                continue
                        if not isinstance(metadata, dict):
                            continue
                        if any(column in metadata for column in column_names):
                            for column in column_names:
                                metadata.pop(column, None)
                            cur.execute(
                                f"UPDATE {qualified} SET metadata_ = %s WHERE id = %s",
                                (psycopg2.extras.Json(metadata), row_id),
                            )
    return moved

# Verification helpers

def describe_table(database: str, logical_table: str = TABLE_NAME) -> List[Dict[str, str]]:
    """Return the column list of the vector table."""
    schema, table = _resolve_physical_table(database, logical_table)
    with closing(_connect(database)) as conn:
        with conn:
            with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
                cur.execute(
                    """
                    SELECT column_name, data_type
                      FROM information_schema.columns
                     WHERE table_schema = %s AND table_name = %s
                     ORDER BY ordinal_position
                    """,
                    (schema, table),
                )
                return [dict(row) for row in cur.fetchall()]


def fetch_rows(database: str, logical_table: str = TABLE_NAME, limit: int = 10):
    """Fetch rows including the dedicated ``external_id`` column."""
    schema, table = _resolve_physical_table(database, logical_table)
    with closing(_connect(database)) as conn:
        with conn:
            with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
                cur.execute(
                    f'SELECT id, node_id, external_id, text, metadata_ '
                    f'FROM "{schema}"."{table}" LIMIT %s',
                    (limit,),
                )
                return [dict(row) for row in cur.fetchall()]


def lookup_external_ids(database: str, node_ids: Sequence[str]) -> Dict[str, str]:
    """Read ``external_id`` from the dedicated column for the given node ids."""
    if not node_ids:
        return {}
    schema, table = _resolve_physical_table(database, TABLE_NAME)
    with closing(_connect(database)) as conn:
        with conn:
            with conn.cursor() as cur:
                cur.execute(
                    f'SELECT node_id, external_id FROM "{schema}"."{table}" '
                    f"WHERE node_id = ANY(%s)",
                    (list(node_ids),),
                )
                return {row[0]: row[1] for row in cur.fetchall()}

def create_vector_store(request):
    """Create the vector store whose table owns a real ``external_id`` column."""
    database = request.tenant

    vector_store = CustomSchemaPGVectorStore.from_params(
        database=database,
        host=PG_HOST,
        password=PG_PASSWORD,
        port=PG_PORT,
        user=PG_USER,
        table_name=TABLE_NAME,
        embed_dim=EMBED_DIM,
        hnsw_kwargs=HNSW_KWARGS,
    )
    if not isinstance(vector_store, CustomSchemaPGVectorStore):
        # Some releases hard-code the base class inside ``from_params``;
        # build the subclass directly in that case.
        vector_store = CustomSchemaPGVectorStore(
            connection_string=_connection_string(database),
            table_name=TABLE_NAME,
            embed_dim=EMBED_DIM,
            hnsw_kwargs=HNSW_KWARGS,
        )

    # Authoritative (and idempotent) guarantee that the custom columns exist,
    # independently of how the table was created above.
    ensure_custom_columns(database)

    return vector_store


def run_agent(request):
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    # ------------------------------------------------------------------
    # FIX 1: hand the vector store to the index through a StorageContext.
    # ``from_documents`` has no ``vector_store`` argument -- the keyword is
    # swallowed by ``**kwargs`` and ignored, which is why nothing ever reached
    # Postgres before.
    # ------------------------------------------------------------------
    storage_context = StorageContext.from_defaults(vector_store=vector_store)
    index = VectorStoreIndex.from_documents(
        documents,
        storage_context=storage_context,
    )

    # ------------------------------------------------------------------
    # FIX 2: promote ``external_id`` out of the ``metadata_`` JSON column into
    # the dedicated column the table now has.
    # ------------------------------------------------------------------
    moved = sync_metadata_to_custom_columns(request.tenant)
    for column_name, count in moved.items():
        logger.info("'%s' populated for %d row(s)", column_name, count)

    return index
