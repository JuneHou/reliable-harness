from llama_index.core import VectorStoreIndex, Document
from llama_index.vector_stores.postgres import PGVectorStore
from sqlalchemy import Column, Text
import psycopg2
import os
from dotenv import load_dotenv

load_dotenv()


def create_vector_store(request):
    host = "localhost"
    password = "password"
    port = 5432
    user = "postgres"

    # FIX: By default, PGVectorStore stores ALL document metadata inside a
    # JSONB "metadata_" column. To make `external_id` a real, dedicated column
    # in the Postgres table schema, we must explicitly pass `schema_fields`.
    # `schema_fields` maps metadata keys to SQLAlchemy Column definitions.
    # PGVectorStore will then:
    #   1. include these columns in the CREATE TABLE statement for "vectorstore"
    #   2. populate them directly from the node metadata on insert
    #   3. allow filtering on them via MetadataFilters / where_clauses
    schema_fields = {
        "external_id": Column("external_id", Text),
    }

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name="vectorstore",
        embed_dim=1536,
        schema_fields=schema_fields,
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
    )

    return vector_store


def run_agent(request):
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store
    )

    return index


def verify_external_id_column(request):
    """Verify that `external_id` now exists as a dedicated column in the table."""
    conn = psycopg2.connect(
        dbname=request.tenant,
        host="localhost",
        port=5432,
        user="postgres",
        password="password",
    )
    try:
        with conn.cursor() as cur:
            cur.execute(
                """
                SELECT column_name
                FROM information_schema.columns
                WHERE table_name = 'vectorstore'
                """
            )
            columns = [row[0] for row in cur.fetchall()]
            if "external_id" in columns:
                print("Verified: 'external_id' is a dedicated column in the table")
            else:
                print(f"Column 'external_id' not found. Columns: {columns}")
    finally:
        conn.close()
