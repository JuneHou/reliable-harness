from llama_index.core import VectorStoreIndex, Document
from llama_index.core.vector_stores import VectorStoreQuery
from llama_index.vector_stores.postgres import PGVectorStore
from sqlalchemy import Column, Text, text as sql_text
import psycopg2
import os
from dotenv import load_dotenv

load_dotenv()

DB_CONFIG = {
    "host": "localhost",
    "password": "password",
    "port": 5432,
    "user": "postgres",
}

# PGVectorStore stores rows in a physical table named "data_" + table_name,
# so table_name="vectorstore" maps to the physical table "data_vectorstore".
TABLE_NAME = "vectorstore"
PHYSICAL_TABLE = f"data_{TABLE_NAME}"
SCHEMA_NAME = "public"
CUSTOM_COLUMN = "external_id"


def ensure_external_id_column(request):
    """
    KEY FIX (part 1): PGVectorStore only creates `metadata_cols` columns when
    it *creates* the table. If the table already exists from a previous run
    (created without metadata_cols), the new column would never be added and
    external_id would keep living only inside the metadata JSON column.

    This helper migrates an already-existing table in place so that its
    schema actually contains the external_id column (preserving any rows).
    """
    conn = psycopg2.connect(dbname=request.tenant, **DB_CONFIG)
    try:
        cur = conn.cursor()

        # Does the physical table exist yet?
        cur.execute(
            """
            SELECT 1
            FROM information_schema.tables
            WHERE table_schema = %s AND table_name = %s
            """,
            (SCHEMA_NAME, PHYSICAL_TABLE),
        )
        table_exists = cur.fetchone() is not None

        if table_exists:
            # Does it already have the custom column?
            cur.execute(
                """
                SELECT 1
                FROM information_schema.columns
                WHERE table_schema = %s AND table_name = %s AND column_name = %s
                """,
                (SCHEMA_NAME, PHYSICAL_TABLE, CUSTOM_COLUMN),
            )
            column_exists = cur.fetchone() is not None

            if not column_exists:
                cur.execute(
                    f'ALTER TABLE {SCHEMA_NAME}."{PHYSICAL_TABLE}" '
                    f'ADD COLUMN IF NOT EXISTS "{CUSTOM_COLUMN}" TEXT'
                )
                conn.commit()

        cur.close()
    finally:
        conn.close()


def create_vector_store(request):
    host = "localhost"
    password = "password"
    port = 5432
    user = "postgres"

    # KEY FIX (part 2): declare external_id as a real SQLAlchemy column that
    # PGVectorStore will create directly on the table, instead of letting it
    # get buried inside the metadata JSON column.
    external_id_col = Column(
        CUSTOM_COLUMN,
        Text,
        nullable=True,
    )

    def customize_query_fn(query_base, query: VectorStoreQuery):
        # KEY FIX (part 3): surface the extra column on every SELECT
        # statement so that it is returned during retrieval in
        # node.metadata["custom_fields"].
        return query_base.add_columns(sql_text(CUSTOM_COLUMN))

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name=TABLE_NAME,
        embed_dim=1536,
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
        # This makes PGVectorStore create a dedicated `external_id` column
        # on the data_<table_name> table. Its value is populated from
        # node.metadata["external_id"] on insert.
        metadata_cols={CUSTOM_COLUMN: external_id_col},
        # Retrieve the custom column back in query results.
        customize_query_fn=customize_query_fn,
    )

    return vector_store


def run_agent(request):
    # Make sure an already-created table actually has the new column.
    ensure_external_id_column(request)

    vector_store = create_vector_store(request)

    # Note: external_id must also be present in the node metadata, because
    # PGVectorStore populates the custom column from the matching metadata
    # key on insert.
    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )

    return index


def verify_table_columns(request):
    """Confirm that external_id now exists as its own table column."""
    conn = psycopg2.connect(dbname=request.tenant, **DB_CONFIG)
    try:
        cur = conn.cursor()
        cur.execute(
            """
            SELECT column_name
            FROM information_schema.columns
            WHERE table_schema = %s AND table_name = %s
            ORDER BY ordinal_position
            """,
            (SCHEMA_NAME, PHYSICAL_TABLE),
        )
        columns = [row[0] for row in cur.fetchall()]
        cur.close()
    finally:
        conn.close()
    return columns


def verify_external_id_values(request):
    """Confirm that external_id values were written into the new column."""
    conn = psycopg2.connect(dbname=request.tenant, **DB_CONFIG)
    try:
        cur = conn.cursor()
        cur.execute(
            f'SELECT "{CUSTOM_COLUMN}" FROM {SCHEMA_NAME}."{PHYSICAL_TABLE}" '
            f'WHERE "{CUSTOM_COLUMN}" IS NOT NULL'
        )
        values = [row[0] for row in cur.fetchall()]
        cur.close()
    finally:
        conn.close()
    return values
