import os
from typing import Any, List, Mapping, Optional
from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class wrapper(LLM):
    llm: OpenAI
    log: str

    @property
    def _llm_type(self) -> str:
        return "custom"

    def _call(self, prompt: str, stop: Optional[List[str]] = None) -> str:
        # FIX: forward the agent's stop sequences to the wrapped LLM.
        #
        # The "conversational-react-description" agent (ConversationalAgent /
        # ReAct-style agents) calls the LLM with stop=["\nObservation:"] so the
        # model halts right after "Action Input:" instead of hallucinating an
        # "Observation:" line itself. The previous implementation dropped the
        # `stop` argument (it called self.llm(prompt) without it), so the inner
        # OpenAI model kept generating and produced an extra "Observation:" line
        # that the original (unwrapped) LLM never produced.
        reply = self.llm(prompt, stop=stop)

        # Defensive belt-and-braces: if the underlying model still emitted a
        # stop sequence (some backends ignore `stop`), truncate the reply there
        # so the agent sees exactly what it would see with the original LLM.
        if stop:
            for s in stop:
                if s:
                    idx = reply.find(s)
                    if idx != -1:
                        reply = reply[:idx]

        f = open(self.log, 'a')
        f.write('\n{\n' + prompt + '\n}{\n' + reply + '\n}\n' + '.' * 100)
        f.close()
        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        """Get the identifying parameters."""
        return {"llm": self.llm}


def setup_components():
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0
    )

    wllm = wrapper(llm=llm, log='output.log')

    tools = [PythonAstREPLTool()]

    memory = ConversationBufferMemory(memory_key="chat_history")

    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory
    )
    return agent


def run_agent(prompt):

    try:
        wllm, tools, memory = setup_components()

    except Exception as e:
        print(f"Error setting up components: {e}")
        return None

    try:
        agent = create_agent(wllm, tools, memory)

    except Exception as e:
        print(f"Error creating agent: {e}")
        return None

    try:
        response = agent(prompt)
        return response
    except Exception as e:
        print(f"Error running agent: {e}")
        return None


if __name__ == "__main__":

    prompt = "What is 5-7?"

    response = run_agent(prompt)

    print(response)
    print(type(response), response["output"])
