from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()
def run_agent(prompt):
    
    embeddings = OpenAIEmbeddings()

    csvloader = CSVLoader("movies.csv")
    csvdocuments = csvloader.load()

    search = Chroma.from_documents(csvdocuments, embeddings)


    # FIX: the previous retriever used search_kwargs={"k": 1}, which restricted
    # retrieval to only a single document (one CSV row = one movie/author).
    # With k=1 the chain could only ever see the single most similar row, so it
    # answered with just one author (A) even when several authors had matching
    # films. By increasing k, enough source rows are retrieved for the LLM to
    # aggregate every matching author from the file.
    chain = ConversationalRetrievalChain.from_llm(ChatOpenAI(temperature=0.3), 
                                    retriever=search.as_retriever(search_kwargs={"k": 10}),
                                    return_source_documents=True)
    # chat_history is expected to be a list of (question, answer) tuples
    result = chain({"question": prompt, 'chat_history': []}, return_only_outputs=True)
    return result["answer"]

if __name__ == '__main__':
    print(run_agent("which authors have a film in 1999"))
