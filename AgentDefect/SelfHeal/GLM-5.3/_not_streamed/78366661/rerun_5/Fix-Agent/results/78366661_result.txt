from dotenv import load_dotenv
import os

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from pymongo import MongoClient

# Use the REAL MongoDB Atlas vector store implementation instead of the local
# dummy stub that always returned an empty list from similarity_search.
try:
    from langchain_mongodb import MongoDBAtlasVectorSearch
except ImportError:
    from langchain_community.vectorstores import MongoDBAtlasVectorSearch


def run_agent(pdf_url):
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key is None:
        raise ValueError("OPENAI_API_KEY not found")

    mongodb_atlas_cluster_uri = os.getenv("MONGODB_ATLAS_CLUSTER_URI")
    if mongodb_atlas_cluster_uri is None:
        raise ValueError("MONGODB_ATLAS_CLUSTER_URI not found")

    db_name = os.getenv("MONGODB_DB_NAME", "langchain_db")
    collection_name = os.getenv("MONGODB_COLLECTION_NAME", "test")

    # Connect to a real MongoDB Atlas collection instead of an empty Python
    # list, so embeddings are actually stored in and queried from Atlas.
    client = MongoClient(mongodb_atlas_cluster_uri)
    db = client[db_name]
    atlas_collection = db[collection_name]

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()

    docs = text_splitter.split_documents(data)

    vector_search_index = "vector_index"

    embeddings = OpenAIEmbeddings(disallowed_special=())

    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=embeddings,
        collection=atlas_collection,
        index_name=vector_search_index
    )

    # A missing / not-yet-built Atlas Vector Search index is the usual reason
    # similarity_search silently returns [] even though the documents were
    # successfully saved. Check the existing search indexes and create the
    # index (with the correct embedding dimensions) if it does not exist.
    existing_search_indexes = set()
    try:
        for index in atlas_collection.list_search_indexes():
            existing_search_indexes.add(index.get("name"))
    except Exception:
        # list_search_indexes requires Atlas-level privileges;
        # fall back to a best-effort creation attempt below.
        pass

    embedding_dimension = len(embeddings.embed_query("dimension probe"))

    if vector_search_index not in existing_search_indexes and hasattr(
        MongoDBAtlasVectorSearch, "create_vector_search_index"
    ):
        MongoDBAtlasVectorSearch.create_vector_search_index(
            collection=atlas_collection,
            index_name=vector_search_index,
            dimensions=embedding_dimension,
            path="embedding",
            similarity="cosine",
            wait_until_complete=60
        )

    query = "What were the compute requirements for training GPT 4"

    results = vector_search.similarity_search(query=query, k=10)

    client.close()

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"

    results = run_agent(url)

    print("Result:", results)
