import os
import time

from dotenv import load_dotenv

from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_mongodb import MongoDBAtlasVectorSearch

from pymongo import MongoClient
from pymongo.collection import Collection


def ensure_vector_search_index(collection: Collection, index_name: str, num_dimensions: int = 1536) -> None:
    """Make sure the Atlas Vector Search index exists so similarity_search can return documents.

    If the index is missing, documents can be saved in the collection but
    similarity_search will always return an empty array.
    """
    index_definition = {
        "fields": [
            {
                "type": "vector",
                "path": "embedding",
                "numDimensions": num_dimensions,
                "similarity": "cosine",
            },
            {"type": "filter", "path": "metadata.source"},
        ]
    }

    try:
        existing_index_names = [idx["name"] for idx in collection.list_search_indexes()]
        if index_name not in existing_index_names:
            collection.create_search_index(
                {"name": index_name, "type": "vectorSearch", "definition": index_definition}
            )
            # Wait (bounded) until the index is queryable, otherwise the first
            # searches may still come back empty while the index is building.
            deadline = time.time() + 120
            while time.time() < deadline:
                statuses = {
                    idx["name"]: idx.get("queryable", False)
                    for idx in collection.list_search_indexes()
                }
                if statuses.get(index_name, False):
                    break
                time.sleep(2)
    except Exception as exc:
        # On some clusters/versions the index must be created in the Atlas UI.
        print(
            f"Warning: could not verify/create the '{index_name}' search index "
            f"({exc}). Create it in the Atlas UI if searches return empty results."
        )


def run_agent(pdf_url):
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key is None:
        raise ValueError("OPENAI_API_KEY not found")

    mongodb_atlas_uri = os.getenv("MONGODB_ATLAS_URI")
    if mongodb_atlas_uri is None:
        raise ValueError("MONGODB_ATLAS_URI not found")

    # Use a real MongoDB Atlas collection, not a plain Python list.
    client = MongoClient(mongodb_atlas_uri)
    db = client["langchain_db"]
    atlas_collection = db["pdf_embeddings"]

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()

    docs = text_splitter.split_documents(data)

    vector_search_index = "vector_index"

    # The search index must exist and be queryable before similarity_search is called.
    ensure_vector_search_index(atlas_collection, vector_search_index)

    # Use the real MongoDBAtlasVectorSearch from langchain_mongodb, not a local mock.
    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(disallowed_special=()),
        collection=atlas_collection,
        index_name=vector_search_index
    )

    query = "What were the compute requirements for training GPT 4"

    results = vector_search.similarity_search(query)

    client.close()

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"

    results = run_agent(url)

    print("Result:", results)
