from dotenv import load_dotenv
import os

from pymongo import MongoClient
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_mongodb import MongoDBAtlasVectorSearch


def run_agent(pdf_url):
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if openai_api_key is None:
        raise ValueError("OPENAI_API_KEY not found")

    mongodb_uri = os.getenv("MONGODB_URI")
    if mongodb_uri is None:
        raise ValueError("MONGODB_URI not found")

    # Connect to a real MongoDB Atlas collection instead of using a mock
    # class backed by a plain Python list. The collection object is what
    # the real MongoDBAtlasVectorSearch uses to insert embeddings and run
    # $vectorSearch aggregations.
    client = MongoClient(mongodb_uri)
    db_name = os.getenv("MONGODB_DATABASE", "langchain_db")
    collection_name = os.getenv("MONGODB_COLLECTION", "vector_store")
    atlas_collection = client[db_name][collection_name]

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=100
    )

    loader = PyPDFLoader(pdf_url)
    data = loader.load()

    docs = text_splitter.split_documents(data)

    vector_search_index = "vector_index"

    # Use the real MongoDBAtlasVectorSearch from langchain_mongodb.
    # This embeds each document with OpenAIEmbeddings and inserts the
    # documents + embedding vectors into the Atlas collection.
    #
    # NOTE: For similarity_search to return results, a Vector Search index
    # named "vector_index" must exist in Atlas on this collection with:
    #   - type: "vector"
    #   - path: "embedding"
    #   - numDimensions: 1536 (for the default OpenAI embedding model)
    #   - similarity: "cosine"
    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(disallowed_special=()),
        collection=atlas_collection,
        index_name=vector_search_index
    )

    query = "What were the compute requirements for training GPT 4"

    # The real class performs an actual $vectorSearch query against Atlas
    # using the specified index, returning the nearest documents.
    results = vector_search.similarity_search(query)

    return results


if __name__ == "__main__":
    url = "https://arxiv.org/pdf/2303.08774.pdf"

    results = run_agent(url)

    print("Result:", results)
