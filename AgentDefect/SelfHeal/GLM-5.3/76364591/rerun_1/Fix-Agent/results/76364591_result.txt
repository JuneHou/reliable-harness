import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool


load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    name = "calculator"
    description = "Performs basic arithmetic"
    # KEY FIX: return_direct=True tells the AgentExecutor to stop iterating
    # immediately after this tool runs and return the tool's raw output
    # directly to the caller, instead of feeding it back to the LLM for
    # another "reformatting" iteration.
    return_direct: bool = True

    def _run(self, query: str) -> str:
        return str(eval(query))

    async def _arun(self, query: str) -> str:
        raise NotImplementedError()


def create_agent():
    llm_gpt4 = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(
        memory_key="chat_history",
        return_messages=True,
        output_key="output",
    )
    return initialize_agent(
        agent="chat-conversational-react-description",
        # Pass return_direct explicitly as well so the tool halts the agent
        # loop and its raw output is what gets returned to the client.
        tools=[CalculatorTool(return_direct=True)],
        llm=llm_gpt4,
        verbose=True,
        max_iterations=2,
        early_stopping_method="generate",
        memory=memory,
    )


def run_agent(prompt: str) -> str:
    agent = create_agent()
    return agent.run(prompt)
