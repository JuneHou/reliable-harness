import os
import uuid
from datetime import datetime
from dotenv import load_dotenv
import chromadb
from chromadb.utils import embedding_functions
from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import CharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings


load_dotenv()


config = {
    "OPENAI_API_KEY": os.getenv("OPENAI_API_KEY")
}

class configs:
    EMBEDDINGS_MODEL = "text-embedding-3-small"


class MockPDFFile:
    def __init__(self, filepath, filename):
        self.filepath = filepath
        self.filename = filename


def read_docs(pdf_file):
    pdf_loader = PyPDFLoader(pdf_file.filepath)
    pdf_documents = pdf_loader.load()

    text_splitter = CharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
    documents = text_splitter.split_documents(pdf_documents)
    
    return documents


def generate_and_store_embeddings(documents, pdf_file, user_id):
    client = chromadb.PersistentClient(path="./trained_db")
    collection = client.get_or_create_collection(
        "PDF_Embeddings",
        embedding_function=embedding_functions.OpenAIEmbeddingFunction(
            api_key=config["OPENAI_API_KEY"],
            model_name=configs.EMBEDDINGS_MODEL
        )
    )
    now = datetime.now()

    metadata = {"source": pdf_file.filename, "user": str(user_id), 'created_at':
                now.strftime("%d/%m/%Y %H:%M:%S")}
    ids = [str(uuid.uuid4()) for _ in range(len(documents))]

    # ----------------------------------------------------------------------
    # FIX (root cause of the wrong output):
    # Previously the file/user metadata was only passed as
    # `collection_metadata`, i.e. metadata of the *collection itself*. It was
    # never attached to the individual documents that get stored in Chroma,
    # so every stored embedding kept only PyPDFLoader's metadata (where
    # "source" is the FULL file path, e.g. "docs/foo.pdf"). Because of that,
    # filtering with `where={"source": filename}` (filename = "foo.pdf")
    # matched nothing and the user got an empty result instead of the
    # ids/metadata of that specific PDF.
    # The metadata must be attached to every single document so Chroma stores
    # it together with each embedding, which makes the `where` filter return
    # exactly the embeddings that belong to this PDF file.
    # ----------------------------------------------------------------------
    for document in documents:
        # keep the loader metadata (e.g. "page") but override "source" with
        # the file name and add "user" / "created_at" so the filter matches
        document.metadata.update(metadata)

    try:
        vectordb = Chroma.from_documents(
                    documents,         
                    embedding=OpenAIEmbeddings(openai_api_key=config["OPENAI_API_KEY"], 
                    model=configs.EMBEDDINGS_MODEL),
                    persist_directory='./trained_db',
                    collection_name=collection.name, 
                    client=client,
                    ids=ids,
                )
        # `persist()` was deprecated/removed in recent langchain versions
        # (persistence is automatic with a PersistentClient), so guard it.
        try:
            vectordb.persist()
        except Exception:
            pass
        
    except Exception as err:
        print(f"An error occured: {err=}, {type(err)=}")
        return {"answer": "An error occured while generating embeddings. Please check terminal for more details."}
    return vectordb


def get_pdf_embeddings(vectordb, filename, user_id=None):
    """
    Retrieve ONLY the ids and metadata of the embeddings that belong to a
    specific PDF file (optionally restricted to one user) instead of
    fetching every id/metadata of the whole collection.
    """
    if user_id is not None:
        # Chroma requires `$and` when filtering on more than one field
        where_filter = {"$and": [
            {"source": {"$eq": filename}},
            {"user": {"$eq": str(user_id)}},
        ]}
    else:
        where_filter = {"source": {"$eq": filename}}

    result = vectordb.get(where=where_filter, include=["metadatas"])
    return result["ids"], result["metadatas"]


def delete_pdf_embeddings(vectordb, filename, user_id=None):
    """Delete the embeddings that belong to a specific PDF file."""
    ids, _ = get_pdf_embeddings(vectordb, filename, user_id)
    if ids:
        vectordb.delete(ids=ids)
    return len(ids)


def run_agent(pdf_path, user_id):
   
    pdf_file = MockPDFFile(pdf_path, os.path.basename(pdf_path))
    
   
    documents = read_docs(pdf_file) 
    print(f"    ✓ Loaded {len(documents)} document chunks")
    
   
    vectordb = generate_and_store_embeddings(documents, pdf_file, user_id)
    
    if isinstance(vectordb, dict):
       
        return None, pdf_file.filename
    
    
    return vectordb, pdf_file.filename

    print(f"Embeddings stored for '{filename}': {len(result['ids'])}")

for document in documents:
