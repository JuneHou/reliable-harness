import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import (
    SimpleDirectoryReader,
    GPTListIndex,
    GPTVectorStoreIndex,

    Settings,
    PromptTemplate,
)
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())


os.environ['REQUESTS_CA_BUNDLE'] = ''

# ---------------------------------------------------------------------------
# WHY THE BOT KEPT SAYING IT "CANNOT GENERATE A DRAFT":
#   * The default LlamaIndex QA prompt says
#       "Given the context information and not prior knowledge, answer the query"
#     so when the user asks the bot to *generate* a document (e.g. an NDA)
#     instead of asking a factual question, the model correctly refuses.
#   * The default response mode ("compact") answers a question from the top-k
#     retrieved chunks instead of synthesizing/merging the example documents.
#   * similarity_top_k defaults to 2 chunks (~1,200 tokens) - not enough
#     context to model a whole example document.
#   * max_tokens=512 truncates any generated document mid-way.
#
# FIX: use a prompt that explicitly instructs the model to DRAFT the requested
# document from the example documents, combine it with the "tree_summarize"
# response mode (which merges ALL retrieved text, like the original
# GPTListIndex behaviour), retrieve more chunks, and allow a longer answer.
# The template contains both {context_str} and {query_str}, so it works as the
# QA template (used by compact/refine modes) and as the summary template
# (used by tree_summarize).
# ---------------------------------------------------------------------------
DOCUMENT_PROMPT_TMPL = (
    "You are a document-drafting assistant. "
    "The following example documents were retrieved from the knowledge base:\n"
    "---------------------\n"
    "{context_str}\n"
    "---------------------\n"
    "Instructions:\n"
    "1. If the user request asks you to create, write or generate a document "
    "(for example a non-disclosure agreement), write the COMPLETE document, "
    "using the example documents above as a template for their structure, "
    "sections, wording and tone. Fill in any missing details with clear "
    "placeholders such as [PARTY NAME], [DATE] or [ADDRESS]. NEVER refuse and "
    "never say that you cannot generate a draft or that the information is "
    "not available.\n"
    "2. Otherwise, answer the user request using the example documents above.\n"
    "User request: {query_str}\n"
    "Response:\n"
)


def run_agent(user_input, docs_path='docs'):
    
    openai_api_key = os.getenv('OPENAI_API_KEY')
    
    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key
    
    def createVecorIndex(path):
        max_input = 4096
        tokens = 2000   # was 512 -> full document drafts were being cut off
        chunk_size = 600
        max_chunk_overlap = 0.1
        
        
        Settings.llm = OpenAI(temperature=0.3, model='gpt-3.5-turbo', max_tokens=tokens)
        Settings.chunk_size = chunk_size
        Settings.chunk_overlap = int(chunk_size * max_chunk_overlap)
        
        #load data
        docs = SimpleDirectoryReader(path).load_data()
        #create vector index
        vectorIndex = GPTVectorStoreIndex.from_documents(docs)
        # persist_dir must be a directory, not a file name
        vectorIndex.storage_context.persist(persist_dir='./vectorIndexStore')
        return vectorIndex
    
    vectorIndex = createVecorIndex(docs_path)
    
    def chatbot(input_index):
        query_engine = vectorIndex.as_query_engine(
            # "tree_summarize" recursively merges ALL retrieved text into one
            # synthesized answer (the original GPTListIndex behaviour) instead
            # of answering a question from a single retrieved chunk.
            response_mode='tree_summarize',
            # default top_k is only 2 chunks - far too little context to model
            # a whole example document, so retrieve more chunks.
            similarity_top_k=6,
        )

        # Replace the default "answer-only-from-context" QA/summary prompts
        # with the document-generation prompt so the model actually drafts the
        # requested document instead of refusing.
        draft_prompt = PromptTemplate(DOCUMENT_PROMPT_TMPL)
        try:
            query_engine.update_prompts(
                {
                    "response_synthesizer:text_qa_template": draft_prompt,
                    "response_synthesizer:summary_template": draft_prompt,
                }
            )
        except Exception:
            # Older LlamaIndex versions: pass the templates directly instead.
            query_engine = vectorIndex.as_query_engine(
                response_mode='tree_summarize',
                similarity_top_k=6,
                text_qa_template=draft_prompt,
                summary_template=draft_prompt,
            )

        response = query_engine.query(input_index)

        if response is None:
            return ""
        if response.response is None:
            return str(response)
        return response.response
   
    result = chatbot(user_input)
    
    return result

if __name__ == "__main__":
  
    # Exercise the document-GENERATION behaviour the user actually wants
    # (the old demo only asked a factual question, which never triggered - or
    # tested - the drafting path).
    user_input = ("Generate a non-disclosure agreement (NDA) between Acme Corp "
                  "and Beta LLC covering a joint software development project, "
                  "using the example documents in the index as a template.")
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result}")
