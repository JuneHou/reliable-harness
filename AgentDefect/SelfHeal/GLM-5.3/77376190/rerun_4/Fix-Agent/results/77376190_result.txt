import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import SimpleDirectoryReader, GPTListIndex, GPTVectorStoreIndex, PromptHelper, load_index_from_storage, Settings, PromptTemplate
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())


os.environ['REQUESTS_CA_BUNDLE'] = ''  

# ---------------------------------------------------------------------------
# Custom query prompt.
#
# The default LlamaIndex text QA prompt tells the model to strictly "answer
# the query using the context and not prior knowledge", which is a pure
# question-answering framing. That is why the bot kept replying that it
# "cannot generate a draft" whenever the user asked it to write a document
# (e.g., an NDA) based on the indexed example documents.
#
# This prompt re-frames the task: the retrieved example documents are used as
# reference templates, and the model is explicitly instructed to produce the
# full drafted document instead of refusing or merely summarizing.
# ---------------------------------------------------------------------------
qa_prompt = PromptTemplate(
    "You are a professional document drafting assistant with access to a collection "
    "of example documents stored in a vector index.\n"
    "---------------------\n"
    "{context_str}\n"
    "---------------------\n"
    "Using the example documents above as reference templates, generate a complete, "
    "professional document draft that fulfills the user's request (for example, a "
    "non-disclosure agreement). Mirror the structure, sections, tone, and legal/business "
    "language of the example documents, and fill in reasonable, standard content wherever "
    "the examples do not provide specifics. Use placeholders such as [Party Name] or "
    "[Date] for details that must be provided by the user.\n"
    "Do NOT refuse the request, and do NOT simply summarize the example documents. "
    "Always output the full drafted document text.\n"
    "Query: {query_str}\n"
    "Drafted Document:"
)


def run_agent(user_input, docs_path='docs'):
    
    openai_api_key = os.getenv('OPENAI_API_KEY')
    
    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key
    
    def createVecorIndex(path):
        max_input = 4096
        # A full document draft (e.g., an NDA) needs more room than 512 tokens,
        # otherwise the generated draft gets cut off mid-document.
        tokens = 2048
        chunk_size = 600
        max_chunk_overlap = 0.1
        
        
        Settings.llm = OpenAI(temperature=0.3, model='gpt-3.5-turbo', max_tokens=tokens)
        Settings.chunk_size = chunk_size
        Settings.chunk_overlap = int(chunk_size * max_chunk_overlap)
        
        #load data
        docs = SimpleDirectoryReader(path).load_data()
        #create vector index
        vectorIndex = GPTVectorStoreIndex.from_documents(docs)
        vectorIndex.storage_context.persist(persist_dir='vectorIndex.json')
        return vectorIndex
    
    vectorIndex = createVecorIndex('docs')
    
    def chatbot(input_index):
        # Attach the custom drafting prompt so the engine behaves as a
        # document generator instead of a strict Q&A bot.
        query_engine = input_index.as_query_engine(text_qa_template=qa_prompt)
        response = query_engine.query(input_index)
        return response.response
   
    result = chatbot(user_input)
    
    return result

if __name__ == "__main__":
  
    user_input = "Please generate a draft non-disclosure agreement (NDA) based on the example documents."
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result}")
