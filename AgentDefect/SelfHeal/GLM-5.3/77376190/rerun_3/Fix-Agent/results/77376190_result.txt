import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import (
    SimpleDirectoryReader,
    GPTVectorStoreIndex,
    Settings,
    PromptTemplate,
    StorageContext,
    load_index_from_storage,
)
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())


os.environ['REQUESTS_CA_BUNDLE'] = ''

# Persisted index directory (a directory, not a .json file path)
PERSIST_DIR = './vectorIndex'

# ---------------------------------------------------------------------------
# FIX: the default LlamaIndex QA prompt ("You are an expert Q&A system ...
# only use the provided context ... say you don't know") makes the model treat
# every request as retrieval-based Q&A, so it refuses with "I cannot generate
# a draft". We replace it with an explicit document-drafting prompt that
# instructs the model to produce full document drafts modelled on the example
# documents retrieved from the vector index.
# ---------------------------------------------------------------------------
DOCUMENT_DRAFT_PROMPT = PromptTemplate(
    "Context information from the user's example documents is below.\n"
    "---------------------\n"
    "{context_str}\n"
    "---------------------\n"
    "You are a professional document drafting assistant. Given the context "
    "information above (example documents):\n"
    "1. If the user asks you to generate or draft a document (for example, a "
    "non-disclosure agreement), produce the COMPLETE document draft, following "
    "the structure, style, and terminology of the example documents.\n"
    "2. Replace case-specific details that are not provided in the context with "
    "clearly marked placeholders such as [PARTY A], [PARTY B], and [DATE].\n"
    "3. Never refuse and never say you cannot generate a draft - the example "
    "documents are provided precisely so you can imitate them.\n"
    "4. Output the document text directly, without explanations or apologies.\n"
    "5. If the user simply asks a question, answer it using the context.\n\n"
    "User request: {query_str}\n"
    "Answer:\n"
)


def run_agent(user_input, docs_path='docs'):

    openai_api_key = os.getenv('OPENAI_API_KEY')

    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key

    def createVectorIndex(path):
        max_input = 4096
        # FIX: 512 output tokens truncated any generated document; a full draft
        # needs a much larger completion budget.
        tokens = 2048
        chunk_size = 600
        max_chunk_overlap = 0.1

        Settings.llm = OpenAI(
            temperature=0.3,
            model='gpt-3.5-turbo',
            max_tokens=tokens,
            # FIX: reinforce the drafting behaviour at the system level so the
            # model does not fall back into "assistant cannot generate" refusals.
            system_prompt=(
                "You are a document drafting assistant. When asked to draft or "
                "generate a document, you always produce the full document text "
                "based on the retrieved example documents; you never refuse."
            ),
        )
        Settings.chunk_size = chunk_size
        Settings.chunk_overlap = int(chunk_size * max_chunk_overlap)

        # Reuse the persisted index when it exists instead of rebuilding it.
        if os.path.exists(PERSIST_DIR) and os.path.exists(
            os.path.join(PERSIST_DIR, 'docstore.json')
        ):
            storage_context = StorageContext.from_defaults(persist_dir=PERSIST_DIR)
            return load_index_from_storage(storage_context)

        # load data
        docs = SimpleDirectoryReader(path).load_data()
        # create vector index
        vectorIndex = GPTVectorStoreIndex.from_documents(docs)
        vectorIndex.storage_context.persist(persist_dir=PERSIST_DIR)
        return vectorIndex

    vectorIndex = createVectorIndex('docs')

    def chatbot(input_index):
        query_engine = vectorIndex.as_query_engine(
            # FIX: use the custom document-drafting prompt instead of the
            # default retrieval-QA prompt that caused the refusals.
            text_qa_template=DOCUMENT_DRAFT_PROMPT,
            # FIX: retrieve enough of the example documents to serve as a
            # template (the default top_k of 2 was too small).
            similarity_top_k=5,
            # FIX: combine the retrieved chunks into one prompt so the model
            # sees the example document as a whole.
            response_mode='compact',
        )
        response = query_engine.query(input_index)
        return str(response)

    result = chatbot(user_input)

    return result

if __name__ == "__main__":

    # The user's actual intent: generate a document draft from the indexed
    # example documents (previously the bot answered "I cannot generate a draft").
    user_input = "Generate a non-disclosure agreement draft based on the example documents."
    result = run_agent(user_input)
    print("\n=== User Query Result ===")
    print(f"Question: {user_input}")
    print(f"Answer: {result}")
