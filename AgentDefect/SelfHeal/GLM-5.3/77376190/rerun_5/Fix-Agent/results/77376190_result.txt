import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import SimpleDirectoryReader, GPTListIndex, GPTVectorStoreIndex, PromptHelper, load_index_from_storage, Settings
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())


os.environ['REQUESTS_CA_BUNDLE'] = ''  

def run_agent(user_input, docs_path='docs'):
    
    openai_api_key = os.getenv('OPENAI_API_KEY')
    
    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key
    
    def createVecorIndex(path):
        max_input = 4096
        # Raise the output token budget so a full document draft is not truncated
        tokens = 2048
        chunk_size = 600
        max_chunk_overlap = 0.1
        
        
        Settings.llm = OpenAI(temperature=0.3, model='gpt-3.5-turbo', max_tokens=tokens)
        Settings.chunk_size = chunk_size
        Settings.chunk_overlap = int(chunk_size * max_chunk_overlap)
        
        #load data
        docs = SimpleDirectoryReader(path).load_data()
        #create vector index
        vectorIndex = GPTVectorStoreIndex.from_documents(docs)
        vectorIndex.storage_context.persist(persist_dir='vectorIndex.json')
        return vectorIndex
    
    vectorIndex = createVecorIndex('docs')
    
    def chatbot(input_index):
        query_engine = vectorIndex.as_query_engine()
        
        # FIX: Previously the raw user request was passed straight to the query
        # engine, whose default prompt tells the model to *answer a question*
        # based on the context. When asked to draft a document (e.g. an NDA),
        # the model treated the request as out of scope and replied that it
        # cannot generate a draft.
        # Instead, wrap the user's request in an explicit drafting prompt that
        # instructs the model to act as a document drafting assistant and to
        # generate a complete document modeled on the example documents
        # retrieved from the vector index.
        drafting_prompt = (
            "You are a document drafting assistant. "
            "Using the example documents provided in the context below, "
            "generate a complete, well-structured draft of the requested document. "
            "Follow the style, structure, tone, and key clauses of the example documents in the context, "
            "and include all standard sections typical of that type of document. "
            "Never refuse the request and never say you cannot generate a draft; "
            "always output the full text of the document itself. "
            f"Requested document: {input_index}"
        )
        
        response = query_engine.query(drafting_prompt)
        return response.response
   
    result = chatbot(user_input)
    
    return result

if __name__ == "__main__":
  
    # Ask the assistant to generate a document draft from the example
    # documents in the index (e.g. an NDA) instead of a plain question.
    user_input = "Generate a draft of a non-disclosure agreement (NDA) based on the example documents."
    result = run_agent(user_input)
    print("\n=== Generated Document Draft ===")
    print(f"Request: {user_input}")
    print(f"Draft:\n{result}")
