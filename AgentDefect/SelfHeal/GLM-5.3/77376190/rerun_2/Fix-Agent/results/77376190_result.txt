import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import (
    SimpleDirectoryReader,
    GPTListIndex,
    GPTVectorStoreIndex,
    PromptHelper,
    load_index_from_storage,
    Settings,
    Prompt,
)
from llama_index.llms.openai import OpenAI

_ = load_dotenv(find_dotenv())


os.environ['REQUESTS_CA_BUNDLE'] = ''  # original workaround, kept as-is


def run_agent(user_input, docs_path='docs'):

    openai_api_key = os.getenv('OPENAI_API_KEY')

    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key

    def createVecorIndex(path):
        max_input = 4096
        # FIX: 512 tokens is too small to output a full document draft
        tokens = 2048
        chunk_size = 600
        max_chunk_overlap = 0.1

        Settings.llm = OpenAI(
            temperature=0.3,
            model='gpt-3.5-turbo',
            max_tokens=tokens,
        )
        Settings.chunk_size = chunk_size
        Settings.chunk_overlap = int(chunk_size * max_chunk_overlap)

        # load data
        docs = SimpleDirectoryReader(path).load_data()
        # create vector index
        vectorIndex = GPTVectorStoreIndex.from_documents(docs)
        vectorIndex.storage_context.persist(persist_dir='vectorIndex.json')
        return vectorIndex

    vectorIndex = createVecorIndex('docs')

    # FIX: the query engine's DEFAULT prompt tells the LLM it is a strict
    # Q&A system that must ONLY answer questions using the context and must
    # say it does not know otherwise. That is why the bot replied
    # "I cannot generate a draft". Replace it with a custom text QA prompt
    # that instructs the model to use the retrieved example documents as a
    # template and to actually WRITE the requested document draft.
    qa_prompt_tmpl = (
        "You are a professional document drafting assistant.\n"
        "Below are example documents retrieved from the user's knowledge base.\n"
        "---------------------\n"
        "{context_str}\n"
        "---------------------\n"
        "Use the example documents above as a template: follow their structure, "
        "tone, clause organization and wording.\n"
        "Task: {query_str}\n"
        "Generate a complete, ready-to-use draft document that fulfils the task "
        "above. Do not refuse, do not ask clarifying questions, and never reply "
        "that you cannot generate a document. Output only the drafted document."
    )
    qa_template = Prompt(qa_prompt_tmpl)

    def chatbot(input_index):
        query_engine = vectorIndex.as_query_engine(
            text_qa_template=qa_template,   # use the generation-oriented prompt
            similarity_top_k=3,             # pull in more of the example document
        )
        response = query_engine.query(input_index)
        return response.response

    result = chatbot(user_input)

    return result


if __name__ == "__main__":

    # FIX: ask for the actual use case (generating a draft from examples)

    user_input = (
        "Please generate a non-disclosure agreement (NDA) draft based on "

    )
    result = run_agent(user_input)
