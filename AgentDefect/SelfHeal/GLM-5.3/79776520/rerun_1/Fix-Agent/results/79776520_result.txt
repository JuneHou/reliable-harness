import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_core.tools import tool
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # NOTE: the parameter name is `max_results` (plural), not `max_result`
    simple_search = TavilySearch(max_results=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key
    )

    # The MemorySaver checkpointer is what gives the agent memory: every
    # invocation that uses the same `thread_id` automatically resumes the
    # conversation state that was saved for that thread.
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory
    )

    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, memory, config, user_input):
    # Do NOT manually read the saved history out of the checkpointer and
    # re-send it. The checkpointer is not a state-reading API (its `.get()`
    # is deprecated/removed and never returned a plain dict exposing
    # "channel_values"), and it is not needed here: because a checkpointer
    # is attached and we always pass the same `thread_id` in `config`,
    # LangGraph automatically reloads the thread's saved messages and merges
    # the new message into them via the `add_messages` reducer. Therefore we
    # only ever pass the NEW user message to the agent.
    input_message = {"messages": [HumanMessage(user_input)]}

    final_message = None
    for step in agent_executor.stream(
        input_message,
        config,
        stream_mode="values"
    ):
        final_message = step["messages"][-1]

    return final_message.content


def main():
    agent_executor, memory, config = initialize_agent()
    # Both calls use the same thread_id ("agent003"), so the second call
    # resumes the history saved by the first one and the agent remembers
    # that the user's name is Bob.
    response1 = run_agent(agent_executor, memory, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, memory, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
