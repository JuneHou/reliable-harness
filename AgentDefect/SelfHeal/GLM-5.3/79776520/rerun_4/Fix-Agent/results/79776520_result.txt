import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # NOTE: the parameter is "max_results" (plural), not "max_result",
    # otherwise the limit is silently ignored and the default is used.
    simple_search = TavilySearch(max_results=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key
    )

    # The checkpointer is what gives the agent memory: every invocation that
    # uses the same "thread_id" automatically resumes the same conversation.
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory
    )

    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, memory, config, user_input):
    # Do NOT rebuild the conversation history by hand.
    #
    # `memory.get(config)` is a low-level (deprecated) checkpointer API that
    # returns the raw serialized checkpoint, not a usable list of chat
    # messages. Concatenating that with a new HumanMessage feeds
    # invalid/duplicated entries into the graph's "messages" channel and
    # corrupts the conversation, which is why the agent appeared to forget
    # between invocations.
    #
    # The agent's state is persisted by the checkpointer, and its "messages"
    # channel is merged with the add_messages reducer. So the correct usage is
    # to send ONLY the new user message: LangGraph automatically loads the
    # previously saved messages for this thread_id and appends the new one.
    new_input = {"messages": [HumanMessage(user_input)]}

    final_message = None
    for step in agent_executor.stream(
        new_input,
        config,
        stream_mode="values"
    ):
        final_message = step["messages"][-1]

    return final_message.content if final_message is not None else ""


def main():
    agent_executor, memory, config = initialize_agent()
    response1 = run_agent(agent_executor, memory, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, memory, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
