import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_core.tools import tool
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # NOTE: the parameter is `max_results` (plural). `max_result` is not a
    # recognized argument and was silently ignored, so Tavily fell back to
    # its default number of results instead of the intended 2.
    simple_search = TavilySearch(max_results=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key
    )

    # MemorySaver persists the agent state (including all messages) per
    # thread_id, so the agent is stateful across invocations.
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory
    )

    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, memory, config, user_input):
    # Send ONLY the new user message.
    #
    # Because the agent was compiled with the MemorySaver checkpointer and is
    # invoked with the same thread_id in `config`, LangGraph automatically
    # restores the previously saved messages for this thread at the start of
    # each invocation, and the `messages` channel uses the `add_messages`
    # reducer, which appends the new input message to that restored history.
    #
    # Manually reading the checkpoint (memory.get) and re-sending the old
    # history together with the new message caused the previous turns to be
    # merged into the already-restored state again, corrupting the
    # conversation context so the agent's replies did not match what was
    # expected. The checkpointer is the single source of truth for history.
    input_message = {"messages": [HumanMessage(user_input)]}

    final_message = None
    for step in agent_executor.stream(
        input_message,
        config,
        stream_mode="values"
    ):
        final_message = step["messages"][-1]

    return final_message.content if final_message is not None else ""


def main():
    agent_executor, memory, config = initialize_agent()
    response1 = run_agent(agent_executor, memory, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, memory, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
