import os
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv
dotenv.load_dotenv()

PDF_PATH = 'paper2.pdf'
PERSIST_DIR = './chroma_db'
COLLECTION_NAME = 'paper2_collection'


def build_vectorstore():
    """
    Build (or load) the Chroma vector store EXACTLY ONCE.

    The original bug: Chroma.from_documents() was called inside run_agent(),
    so every query re-inserted all the document chunks into the persisted
    collection. The collection therefore contained N copies of each chunk,
    and the retriever returned duplicate identical documents.

    Fix:
      - Use an explicit persist_directory + collection_name so the store is
        stable across runs.
      - Only ingest the documents if the collection is empty.
      - Do the (expensive, side-effectful) ingestion once at module scope,
        not on every query.
    """
    embeddings = OpenAIEmbeddings()

    vectorstore = Chroma(
        collection_name=COLLECTION_NAME,
        embedding_function=embeddings,
        persist_directory=PERSIST_DIR,
    )

    # Guard: only load/split/embed the PDF if nothing has been ingested yet.
    # (This is the precondition check that prevents duplicate insertion.)
    try:
        existing_count = vectorstore._collection.count()
    except Exception:
        existing_count = 0

    if existing_count == 0:
        loader = UnstructuredFileLoader(PDF_PATH)
        docs = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=1000, chunk_overlap=200, add_start_index=True
        )
        all_splits = text_splitter.split_documents(docs)

        vectorstore.add_documents(all_splits)
        try:
            vectorstore.persist()
        except AttributeError:
            # Newer langchain/chroma versions persist automatically.
            pass

    return vectorstore


# Build the vector store once, at module scope (moved out of run_agent).
vectorstore = build_vectorstore()
retriever = vectorstore.as_retriever(
    search_type="similarity", search_kwargs={"k": 6}
)


def run_agent(query):
    """
    Only retrieval happens per query now — no re-ingestion of documents,
    so no duplicate documents are created on repeated runs.
    """
    retrieved_docs = retriever.get_relevant_documents(query)

    # Defensive deduplication: keep only unique documents.
    unique_docs = []
    seen = set()
    for doc in retrieved_docs:
        key = (
            doc.page_content,
            doc.metadata.get('source'),
            doc.metadata.get('start_index'),
        )
        if key not in seen:
            seen.add(key)
            unique_docs.append(doc)

    return unique_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
