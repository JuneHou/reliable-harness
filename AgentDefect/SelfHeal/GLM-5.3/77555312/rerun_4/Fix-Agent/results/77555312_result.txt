import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv
dotenv.load_dotenv()

PERSIST_DIRECTORY = "chroma_db"
COLLECTION_NAME = "paper2"


def get_unique_documents(documents):
    """Return only documents whose text (page_content) has not been seen yet."""
    seen = set()
    unique_docs = []
    for doc in documents:
        content = doc.page_content
        if content not in seen:
            seen.add(content)
            unique_docs.append(doc)
    return unique_docs


def get_or_create_vectorstore(persist_directory=PERSIST_DIRECTORY,
                              collection_name=COLLECTION_NAME):
    """
    Reuse the persisted Chroma collection instead of re-inserting the
    documents on every run.

    Previously, Chroma.from_documents() was executed on every run of the
    script (inside run_agent), which inserted the same PDF chunks into the
    persistent collection again and again.  Because each chunk existed
    multiple times in the collection, the top-k similarity retriever kept
    returning duplicate identical documents.
    """
    # Explicitly pass collection_name, embedding_function and
    # persist_directory so we always attach to the SAME persisted collection
    # instead of creating a new one on each run.
    vectorstore = Chroma(
        collection_name=collection_name,
        embedding_function=OpenAIEmbeddings(),
        persist_directory=persist_directory,
    )

    # Precondition check: only load/split/insert the PDF the first time,
    # i.e. when the collection is still empty.
    if vectorstore._collection.count() == 0:
        loader = UnstructuredFileLoader(
            'paper2.pdf'
        )
        docs = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=1000, chunk_overlap=200, add_start_index=True
        )
        all_splits = text_splitter.split_documents(docs)

        # Safety net: never store two chunks with identical text.
        unique_splits = get_unique_documents(all_splits)

        vectorstore.add_documents(unique_splits)

        # Older langchain versions require an explicit persist() call;
        # newer versions persist automatically, so ignore any failure.
        try:
            vectorstore.persist()
        except Exception:
            pass

    return vectorstore


def run_agent(query):
    # Reuse the existing (already indexed) vector store; documents are only
    # ingested once, so nothing gets duplicated on repeated runs.
    vectorstore = get_or_create_vectorstore()

    retriever = vectorstore.as_retriever(
        search_type="similarity", search_kwargs={"k": 6}
    )

    retrieved_docs = retriever.get_relevant_documents(query)

    # Guarantee unique documents in the returned result (also cleans up
    # collections that were polluted with duplicates by earlier runs).
    return get_unique_documents(retrieved_docs)


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
