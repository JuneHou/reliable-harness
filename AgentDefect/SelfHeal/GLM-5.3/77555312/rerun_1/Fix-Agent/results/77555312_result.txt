import os
import hashlib

from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()

# Use an explicit, dedicated persist directory and collection name.
# Previously no directory/collection was supplied, so every run of the
# script re-inserted the same chunks into the (default, shared, persisted)
# collection. Chroma assigns fresh random UUIDs on every insert, so it never
# noticed the documents were already there, and the retriever kept returning
# duplicate identical documents.
PERSIST_DIRECTORY = "./chroma_db"
COLLECTION_NAME = "paper2_unique"


def deterministic_id(doc) -> str:
    """Build a stable ID from a chunk's source, start index and content.

    With deterministic IDs, adding the same chunk again overwrites
    (upserts) the existing record instead of storing a second copy of it.
    """
    raw = "{}|{}|{}".format(
        doc.metadata.get("source", ""),
        doc.metadata.get("start_index", ""),
        doc.page_content,
    )
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()


def load_and_split():
    """Load paper2.pdf and split it into chunks (unchanged logic)."""
    loader = UnstructuredFileLoader("paper2.pdf")
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    return text_splitter.split_documents(docs)


def run_agent(query, force_rebuild=False):
    # 1) Connect to (or create) the persistent store WITHOUT inserting
    #    anything yet. Unlike Chroma.from_documents(), the constructor
    #    never writes documents, so we are in control of ingestion.
    embeddings = OpenAIEmbeddings()
    vectorstore = Chroma(
        collection_name=COLLECTION_NAME,
        embedding_function=embeddings,
        persist_directory=PERSIST_DIRECTORY,
    )

    # Optional: rebuild from scratch (e.g. if the source PDF changed).
    if force_rebuild and len(vectorstore) > 0:
        vectorstore.delete_collection()
        vectorstore = Chroma(
            collection_name=COLLECTION_NAME,
            embedding_function=embeddings,
            persist_directory=PERSIST_DIRECTORY,
        )

    # 2) Ingest the document ONLY if the collection is still empty.
    #    Chroma.__len__ returns the number of documents in the collection,
    #    so this guard makes ingestion a one-time operation instead of
    #    re-inserting the same chunks on every run (the source of the
    #    duplicates).
    if len(vectorstore) == 0:
        all_splits = load_and_split()
        # Deterministic IDs make the insertion idempotent: re-adding the
        # same chunk overwrites it instead of duplicating it.
        doc_ids = [deterministic_id(doc) for doc in all_splits]
        vectorstore.add_documents(all_splits, ids=doc_ids)

    retriever = vectorstore.as_retriever(
        search_type="similarity", search_kwargs={"k": 6}
    )

    retrieved_docs = retriever.get_relevant_documents(query)

    # 3) Safety net: make sure only unique documents are returned.
    unique_docs = []
    seen = set()
    for doc in retrieved_docs:
        key = (doc.page_content, str(doc.metadata))
        if key not in seen:
            seen.add(key)
            unique_docs.append(doc)

    return unique_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
