import os
import hashlib
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv
dotenv.load_dotenv()


def _stable_id(doc):
    """Deterministic ID derived from the chunk's content and metadata.

    Passing these ids to Chroma.from_documents makes insertion behave as an
    upsert: re-running the script on the same PDF overwrites the existing
    entries instead of inserting duplicate copies of the same chunk.
    """
    payload = doc.page_content + str(sorted(doc.metadata.items()))
    return hashlib.sha1(payload.encode("utf-8")).hexdigest()


def run_agent(query):
    loader = UnstructuredFileLoader(
        'paper2.pdf'
    )
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    # Content-derived, stable ids: the same chunk always maps to the same id,
    # so identical documents are never re-inserted as new entries.
    doc_ids = [_stable_id(d) for d in all_splits]

    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
        ids=doc_ids,
    )
    retriever = vectorstore.as_retriever(search_type="similarity", search_kwargs={"k": 6})

    retrieved_docs = retriever.get_relevant_documents(
        query
    )

    # Defensive deduplication: keep only the first occurrence of each unique
    # chunk so the retriever never returns duplicate identical documents.
    seen = set()
    unique_docs = []
    for d in retrieved_docs:
        key = d.page_content
        if key not in seen:
            seen.add(key)
            unique_docs.append(d)

    return unique_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
