from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()


def run_agent(prompt):
    embeddings = OpenAIEmbeddings()

    csvloader = CSVLoader("movies.csv")
    csvdocuments = csvloader.load()

    search = Chroma.from_documents(csvdocuments, embeddings)

    # FIX: the original code used search_kwargs={"k": 1}, which made the
    # retriever return only the single most similar CSV row. Since CSVLoader
    # turns every CSV row into its own document, the LLM only ever saw one
    # row (e.g. only author A's film from 1999) and therefore could only
    # report that one row. Raising k retrieves all the matching rows so the
    # model can answer with every author/movie that satisfies the question.
    retriever = search.as_retriever(search_kwargs={"k": 10})

    chain = ConversationalRetrievalChain.from_llm(
        ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    result = chain(
        {"question": prompt, "chat_history": []},
        return_only_outputs=True,
    )
    return result["answer"]


if __name__ == '__main__':
    print(run_agent("which movies are action"))
