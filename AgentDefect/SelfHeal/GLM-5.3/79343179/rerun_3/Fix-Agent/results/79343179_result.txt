import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
dotenv.load_dotenv()

# Shared, in-process store for objects created while the agent runs.
# agent_executor.invoke() only returns the agent's final *text* answer,
# so real objects (like DataFrames) must be passed back through shared state.
SHARED_STORE = {}


def _repl_namespace(python_repl):
    """Return the persistent namespace dict of a PythonREPL, handling both
    the 'locals' and '_locals' attribute spellings across versions."""
    for attr in ("locals", "_locals", "globals", "_globals"):
        ns = getattr(python_repl, attr, None)
        if isinstance(ns, dict):
            return ns
    return {}


@tool
def load_excel_response(filepath: str):
    """Load an Excel file and return a transformed DataFrame.

    The real DataFrame is cached in the shared store under the key 'df' so the
    calling script can retrieve the actual object after the agent finishes.
    Only a compact serialized summary is returned to the LLM, because tool
    observations passed back into the prompt must be plain strings.
    """
    df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
    df = df.select_dtypes(include=['object'])

    # Keep a handle to the real DataFrame for the main script.
    SHARED_STORE['df'] = df

    # Return a serializable summary (string) instead of the raw DataFrame.
    return df.to_json()


def run_agent(user_prompt, file_name="data.json"):
    # Reset any object left over from a previous run.
    SHARED_STORE.pop('df', None)

    llm = ChatOpenAI(model='gpt-4o-mini')
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    # Expose the loader to the REPL so the agent can also do
    # `df = load_excel_response('excel_file.xlsx')` inside python_repl.
    _repl_namespace(python_repl)['load_excel_response'] = load_excel_response

    repl_tool = Tool(
        name="python_repl",
        description=(
            "Run Python commands in a REPL environment. "
            "The function load_excel_response(filepath) is available in this REPL."
        ),
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(
        agent=shell_agent,
        tools=tools_list,
        verbose=True,
        handle_parsing_errors=True,
        return_intermediate_steps=True,
    )

    response = agent_executor.invoke({"input": user_prompt})

    # 1) Object created by the tool, retrieved from the shared store.
    df = SHARED_STORE.get('df')
    # 2) Fallback: the variable the agent assigned inside the Python REPL,
    #    whose namespace persists in python_repl.locals after the run.
    if df is None:
        df = _repl_namespace(python_repl).get('df')

    return response, df


if __name__ == "__main__":
    path = 'excel_file.xlsx'
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."
    result, df = run_agent(user_prompt)

    # result['output'] is only the agent's textual summary;
    # df is the actual DataFrame built during execution.
    print("Agent output:", result.get('output'))

    if df is not None:
        print("Retrieved object:", type(df), "shape:", getattr(df, 'shape', None))
        print(df.head())
    else:
        print("No DataFrame was created during the agent run.")
