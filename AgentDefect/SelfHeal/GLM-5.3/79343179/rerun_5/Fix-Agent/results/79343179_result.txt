import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# Shared result store: this is the bridge between the agent's tool calls and
# the calling script's local scope. Objects placed here survive the agent run.
# ---------------------------------------------------------------------------
RESULT_STORE = {}

# ---------------------------------------------------------------------------
# Shared REPL globals: passing the same globals dict to PythonREPL means any
# variable the agent assigns via python_repl (e.g. df = ...) is written into
# this dict, so it is retrievable from the main script after invoke().
# ---------------------------------------------------------------------------
SHARED_GLOBALS = {}


@tool
def load_excel_response(filepath: str):
    """Load an Excel file and return a transformed DataFrame."""
    df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
    df = df.select_dtypes(include=['object'])

    # Persist the real DataFrame object in the shared store so the calling
    # script can retrieve it after agent_executor.invoke() finishes. Without
    # this, the LLM only sees a string summary and the object is lost.
    RESULT_STORE['df'] = df
    # Also make it visible to the REPL in case the agent manipulates it later.
    SHARED_GLOBALS['df'] = df

    # Keep the observation returned to the LLM compact -- a full DataFrame
    # repr would blow up the context. The real object is in RESULT_STORE.
    return (
        f"Loaded Excel file '{filepath}' into a DataFrame named 'df' "
        f"with shape {df.shape}. The DataFrame object has been stored and is "
        f"available to the caller under the key 'df'."
    )


def run_agent(user_prompt, file_name="data.json"):
    llm = ChatOpenAI(model='gpt-4o-mini')

    # Give the REPL a shared globals dict: variables the agent creates at
    # runtime (e.g. "df = ...") land in SHARED_GLOBALS instead of a throwaway
    # isolated namespace, making them retrievable from the main script.
    python_repl = PythonREPL(_globals=SHARED_GLOBALS)
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description=(
            "Run Python commands in a REPL environment. Variables you define "
            "here (e.g. df) persist across calls and are shared with the "
            "calling script."
        ),
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    # return_intermediate_steps=True keeps the (action, observation) pairs,
    # which is how the DataFrame observation can also be surfaced alongside
    # the final answer dict instead of being silently discarded.
    agent_executor = AgentExecutor(
        agent=shell_agent,
        tools=tools_list,
        verbose=True,
        return_intermediate_steps=True,
        handle_parsing_errors=True,
    )

    response = agent_executor.invoke({"input": user_prompt})

    # Attach the real DataFrame (and anything the agent assigned in the REPL)
    # to the response so the caller gets the object, not just a summary dict.
    response['df'] = RESULT_STORE.get('df')
    response['repl_variables'] = {k: v for k, v in SHARED_GLOBALS.items() if k != '__builtins__'}
    return response


if __name__ == "__main__":
    path = 'excel_file.xlsx'
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."

    result = run_agent(user_prompt)
    print(result)

    # The actual DataFrame created during agent execution, now accessible as
    # a local variable in the main script:
    df = result.get('df')
    if df is not None:
        print("\nRetrieved DataFrame from agent execution:")
        print(df.head())
    else:
        print("\nDataFrame was not produced by the agent; available REPL variables:",
              list(result.get('repl_variables', {}).keys()))
