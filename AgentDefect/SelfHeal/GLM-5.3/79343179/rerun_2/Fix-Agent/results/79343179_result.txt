import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# Shared namespace.
#
# Why this exists: a ReAct agent only exchanges *strings* with the LLM, so
# agent_executor.invoke() can never return the real DataFrame object -- it
# only returns {'input': ..., 'output': <text summary>}.  To get the actual
# DataFrame back in the main script we use a dict that both the loader tool
# and the agent's PythonREPL write into, and read it out after the run.
# ---------------------------------------------------------------------------
_SHARED_NAMESPACE = {}


@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file, transform it, and store the resulting DataFrame in
    the shared namespace under the variable name 'df' so it can be used from
    the python_repl tool. Returns a short text summary of the result."""
    df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
    df = df.select_dtypes(include=['object'])

    # Store the real object (NOT its string representation) so it survives
    # outside of the agent's text-only conversation loop.
    _SHARED_NAMESPACE['df'] = df

    # Tools must return a short string for the LLM. Returning the raw
    # DataFrame would just dump a huge table into the model's context and
    # would not give the caller the actual object back.
    return (
        f"Loaded '{filepath}' into a DataFrame named 'df' "
        f"({len(df)} rows, {df.shape[1]} column(s)). "
        f"It is now available as the variable 'df' in the python_repl environment."
    )


def run_agent(user_prompt, file_name="data.json"):
    llm = ChatOpenAI(model='gpt-4o-mini')
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    def repl_run(command: str) -> str:
        # Expose objects created by the tools (e.g. 'df') to the agent's REPL
        # session before executing each command, so the agent can work with
        # e.g. df.head() inside python_repl.
        python_repl._globals.update(_SHARED_NAMESPACE)
        return python_repl.run(command)

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=repl_run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(
        agent=shell_agent,
        tools=tools_list,
        verbose=True,
        handle_parsing_errors=True,
    )

    response = agent_executor.invoke({"input": user_prompt})

    # Sync anything the agent itself created inside the REPL (e.g. a
    # "df = ..." assignment typed into python_repl) back into the shared
    # namespace. PythonREPL.run() calls exec(command, self._globals, ...),
    # so all REPL variables live in python_repl._globals.
    _SHARED_NAMESPACE.update(python_repl._globals)

    # Return the REPL instance as well so the caller can reach the real
    # Python objects created during the agent run.
    return response, python_repl


if __name__ == "__main__":
    path = 'excel_file.xlsx'
    user_prompt = (
        f"Load the Excel file located at {path} and assign the result to "
        f"a variable named df."
    )

    result, python_repl = run_agent(user_prompt)
    print(result)

    # Retrieve the *actual* DataFrame object created during the agent run.
    # NOTE: use explicit None checks -- never `or` -- because calling bool()
    # on a DataFrame raises "truth value of a DataFrame is ambiguous".
    df = _SHARED_NAMESPACE.get('df')
    if df is None:
        df = python_repl._globals.get('df')

    if isinstance(df, pd.DataFrame):
        print("\nSuccessfully retrieved the DataFrame as a local variable:")
        print(f"shape: {df.shape}")
        print(df.head())
    else:
        print("\nNo DataFrame named 'df' was created during the agent run.")
