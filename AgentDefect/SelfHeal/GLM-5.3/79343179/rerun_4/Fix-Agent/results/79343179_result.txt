import dotenv
from pathlib import Path

import pandas as pd
from langchain import hub
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_experimental.utilities import PythonREPL
from langchain_openai import ChatOpenAI

dotenv.load_dotenv()


class AgentWorkspace:
    """Shared, mutable state that bridges the agent's tools and the main script.

    `agent_executor.invoke()` only returns a textual summary dictionary; any
    object created inside a tool's scope (or inside PythonREPL) is normally
    lost when execution finishes. By writing the DataFrame into this
    workspace, the main script can retrieve the *actual* DataFrame object.
    """

    def __init__(self) -> None:
        # New attributes holding the real objects produced during execution.
        self.df = None                # the actual pandas DataFrame
        self.repl_namespace = {}      # objects shared with the python REPL

    def store_dataframe(self, df: pd.DataFrame, name: str = "df") -> None:
        self.df = df
        self.repl_namespace[name] = df  # makes `df` usable inside python_repl

    def reset(self) -> None:
        self.df = None
        self.repl_namespace = {}


def _repl_globals(python_repl: PythonREPL):
    """Return the REPL's global namespace dict, across langchain versions."""
    for attr in ("globals", "_globals"):
        ns = getattr(python_repl, attr, None)
        if isinstance(ns, dict):
            return ns
    return None


def _expose_to_repl(python_repl: PythonREPL, mapping: dict) -> None:
    """Inject shared objects (e.g. `df`) into the REPL's global namespace."""
    ns = _repl_globals(python_repl)
    if ns is not None:
        ns.update(mapping)


def _capture_from_repl(python_repl: PythonREPL, workspace: "AgentWorkspace") -> None:
    """If the agent created/overwrote `df` inside the REPL, capture it too."""
    if workspace.df is not None:
        return
    ns = _repl_globals(python_repl)
    if ns is not None:
        candidate = ns.get("df")
        if isinstance(candidate, pd.DataFrame):
            workspace.store_dataframe(candidate, "df")


def make_load_excel_tool(workspace: AgentWorkspace):
    @tool
    def load_excel_response(filepath: str) -> str:
        """Load an Excel file, transform it into a pandas DataFrame, and store it
        as the variable `df` in the shared workspace (also available as a global
        `df` inside python_repl). Returns a short text summary of the DataFrame."""
        df = pd.read_excel(filepath, skiprows=2, names=['responses']).reset_index(drop=True)
        df = df.select_dtypes(include=['object']).reset_index(drop=True)
        # Store the actual object in shared state, return only a string to the LLM.
        workspace.store_dataframe(df, "df")
        return (
            f"Loaded {filepath} into the DataFrame variable `df` "
            f"({len(df)} rows x {len(df.columns)} columns: {list(df.columns)}). "
            "You can now use `df` directly with the python_repl tool."
        )

    return load_excel_response


def run_agent(user_prompt, file_name="data.json"):
    llm = ChatOpenAI(model='gpt-4o-mini')
    workspace = AgentWorkspace()

    python_repl = PythonREPL()
    shell_tool = ShellTool()

    def repl_run(code: str) -> str:
        # Make sure the shared `df` is visible inside the REPL before running.
        _expose_to_repl(python_repl, workspace.repl_namespace)
        return python_repl.run(code)

    repl_tool = Tool(
        name="python_repl",
        description=(
            "Run Python commands in a REPL environment. A pandas DataFrame named "
            "`df` is available as a global variable once it has been loaded with "
            "the load_excel_response tool."
        ),
        func=repl_run,
    )

    load_tool = make_load_excel_tool(workspace)

    tools_list = [load_tool, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(
        agent=shell_agent,
        tools=tools_list,
        verbose=True,
        handle_parsing_errors=True,
        return_intermediate_steps=True,
    )

    # invoke() returns only a textual summary dict; the DataFrame lives in `workspace`.
    response = agent_executor.invoke({"input": user_prompt})

    # Pick up a `df` the agent may have created/overwritten inside the REPL.
    _capture_from_repl(python_repl, workspace)

    # Optionally persist the DataFrame to disk (mirrors the original df.to_json() idea).
    if workspace.df is not None and file_name:
        Path(file_name).write_text(
            workspace.df.to_json(orient='records', force_ascii=False, indent=2),
            encoding='utf-8',
        )

    # Return BOTH the agent's textual response and the actual DataFrame object.
    return response, workspace.df


if __name__ == "__main__":
    path = 'excel_file.xlsx'
    user_prompt = (
        f"Load the Excel file located at {path} and assign the result to a variable named df."
    )

    result, df = run_agent(user_prompt)

    print("\n=== Agent textual response ===")
    print(result.get("output", result))

    print("\n=== Actual DataFrame retrieved from the shared workspace ===")
    if df is not None:
        print(f"type:  {type(df)}")
        print(f"shape: {df.shape}")
        print(df.head())
    else:
        print("No DataFrame was produced by the agent.")
