"""

actual DataFrame created during agent execution as a variable in my main script?"

ROOT CAUSE
----------
1. AgentExecutor.invoke() returns only {"input": ..., "output": <text>}. The
   "output" is the LLM's final ANSWER STRING; Python objects created by tools
   never cross the agent boundary as objects.
2. Tool return values are stringified before being fed back into the LLM
   context, so returning a raw DataFrame from a tool only pollutes the prompt
   with a giant table dump - the object itself is lost.
3. langchain_experimental's PythonREPL.run() executes code in its OWN private
   globals dict, returns only captured stdout, and discards both the value of
   the last expression and the namespace. Any `df` created there is unreachable
   from __main__.

FIX
---
- A module-level shared store (AGENT_STORE) acts as the bridge between the
  agent's execution scope and the calling script. Tools store REAL objects
  (DataFrames) in it instead of (only) returning them as text.
- load_excel_response stores the DataFrame under the requested variable name
  (input "path as df") and returns a compact, LLM-friendly summary instead of
  the raw DataFrame.
- A self-contained python_repl tool runs code in a SHARED, PERSISTENT namespace
  that is readable from the main script, captures stdout, and returns the value
  of the last expression (exec() alone discards it), so `df = ...` inside the
  agent makes `df` available outside.
- run_agent() returns both the agent's textual response AND the actual
  DataFrame, so the main block can bind it to a local variable `df`.

from langchain import hub

# ---------------------------------------------------------------------------
# Shared state: the bridge between the agent's execution scope and this script.
# Objects created during agent execution are stashed here so they can be
# retrieved as real Python objects after agent_executor.invoke() returns.
# ---------------------------------------------------------------------------
AGENT_STORE = {
    "variables": {},      # variable name -> real Python object (e.g. the df)
    "last_result": None,  # value of the last expression executed by the REPL
}

# One namespace shared by every python_repl call AND readable from __main__,
# so variables the agent creates (like df) survive past invoke().
REPL_NAMESPACE = {"pd": pd}
AGENT_STORE["repl_namespace"] = REPL_NAMESPACE


def _reset_runtime() -> None:
    """Start a clean run: drop variables from previous agent invocations."""
    AGENT_STORE["variables"].clear()
    AGENT_STORE["last_result"] = None
    REPL_NAMESPACE.clear()
    REPL_NAMESPACE["pd"] = pd


# ---------------------------------------------------------------------------
# Tool 1: load the Excel file and STORE the DataFrame (not just describe it)
# ---------------------------------------------------------------------------
@tool
def load_excel_response(command: str) -> str:
    """Load an Excel file and store the transformed DataFrame for later use.

    Input is the file path, optionally followed by ' as <variable_name>' to
    choose the name the DataFrame is stored under. Examples:
        excel_file.xlsx
        excel_file.xlsx as df
    Returns a compact summary (shape, columns, dtypes, first rows). The
    DataFrame itself is stored in the session's shared variables, where it can
    be inspected with get_variable or python_repl.
    """
    command = command.strip()

    # Parse "<path> as <name>" (mirrors "assign the result to a variable named df").
    match = re.search(r"\s+as\s+(?P<name>[A-Za-z_]\w*)\s*$", command, flags=re.IGNORECASE)
    if match:
        var_name = match.group("name")
        path = command[: match.start()]
    else:
        var_name = "df"
        path = command
    path = path.strip().strip("'\"").strip()

    # Precondition check + tolerant path recovery if the LLM added extra words.
    if not path or not os.path.isfile(path):
        found = re.search(r"[^\s'\"]+\.(?:xlsx|xls|xlsm|csv)", command, flags=re.IGNORECASE)
        if found and os.path.isfile(found.group(0)):
            path = found.group(0)
        else:
            available = os.listdir(".") if os.path.isdir(".") else []
            hint = ", ".join(f for f in available if f.lower().endswith((".xlsx", ".xls", ".csv")))[:300]
            return (
                f"ERROR: file not found: {path or '(empty)'!r}. "
                f"Pass an existing Excel path. Files in the working directory: {hint or 'none'}"
            )

    try:
        df = pd.read_excel(path, skiprows=2, names=["responses"]).reset_index(drop=True)
        df = df.select_dtypes(include=["object"]).reset_index(drop=True)
    except Exception as exc:  # never crash the agent loop on a bad file
        return f"ERROR: failed to load {path!r}: {type(exc).__name__}: {exc}"

    # >>> Store the REAL object (not a string) so the caller can retrieve it.
    AGENT_STORE["variables"][var_name] = df
    AGENT_STORE["last_result"] = df
    REPL_NAMESPACE.setdefault(var_name, df)

    # Return only a compact, LLM-friendly summary (a raw DataFrame dump would
    # bloat the prompt and still not give the caller the object).
    return (
        f"Loaded {path!r} into variable {var_name!r}: "
        f"{len(df)} rows x {df.shape[1]} columns; "
        f"columns/dtypes: {dict(df.dtypes.astype(str))}; "
        f"first rows: {df.head(3).to_dict(orient='records')}"
    )


# ---------------------------------------------------------------------------
# Tool 2: inspect a stored variable by name
# ---------------------------------------------------------------------------
@tool
def get_variable(variable_name: str) -> str:
    """Return a compact summary of a variable stored earlier in this session
    (for example a DataFrame created by load_excel_response). Input: the
    variable name, e.g. 'df'."""
    name = variable_name.strip().strip("'\"")
    value = AGENT_STORE["variables"].get(name)
    if value is None:
        available = ", ".join(AGENT_STORE["variables"]) or "(none)"
        return f"ERROR: no stored variable named {name!r}. Stored variables: {available}"
    if isinstance(value, pd.DataFrame):
        return (
            f"{name}: DataFrame, shape={value.shape}, "
            f"columns={list(value.columns)}\n{value.head(5).to_string()}"
        )
    return f"{name}: {value!r}"


# ---------------------------------------------------------------------------
# Tool 3: python_repl with a SHARED namespace and last-expression capture
# ---------------------------------------------------------------------------
def _exec_and_capture(code: str):
    """Execute `code` in the shared REPL namespace.

    Returns (stdout_text, last_value). If execution raised, last_value is the
    exception instance. A trailing expression statement is evaluated separately
    so its VALUE is returned (plain exec() discards expression results), which
    is what makes calls like `df.head()` useful.
    """
    stdout = io.StringIO()

    try:
        tree = ast.parse(code)
    except SyntaxError as exc:
        return "", exc

    statements = list(tree.body)
    eval_code = None
    if statements and isinstance(statements[-1], ast.Expr):
        last_expr = statements.pop()
        try:
            eval_code = compile(ast.Expression(last_expr.value), "<python_repl>", "eval")
        except Exception as exc:
            return "", exc

    try:
        exec_code = compile(ast.Module(body=statements, type_ignores=[]), "<python_repl>", "exec")
    except SyntaxError as exc:
        return "", exc

    error = None
    last_value = None
    with contextlib.redirect_stdout(stdout):
        try:
            exec(exec_code, REPL_NAMESPACE)  # noqa: S102 - intentional REPL
            if eval_code is not None:
                last_value = eval(eval_code, REPL_NAMESPACE)  # noqa: S102
        except Exception as exc:
            error = exc

    return stdout.getvalue(), error if error is not None else last_value


@tool
def python_repl(code: str) -> str:
    """Execute Python code in a persistent REPL (pandas is pre-imported as pd).

    Variables persist between calls AND are shared with the calling program,
    so `df = ...` makes `df` available after the agent finishes. The value of
    the last expression, plus anything printed, is returned as text.
    """
    # Make objects stored by other tools (e.g. the loaded DataFrame) visible.
    for name, obj in AGENT_STORE["variables"].items():
        REPL_NAMESPACE.setdefault(name, obj)

    stdout, value = _exec_and_capture(code)

    if isinstance(value, BaseException):
        return f"ERROR: {type(value).__name__}: {value}"

    REPL_NAMESPACE["_"] = value
    if value is not None:
        AGENT_STORE["last_result"] = value

    # Publish non-private, non-module objects the agent created (e.g. df) so
    # the main script can fetch them by name after invoke() returns.
    for name, obj in list(REPL_NAMESPACE.items()):
        if name.startswith("_") or isinstance(obj, types.ModuleType):
            continue
        AGENT_STORE["variables"][name] = obj

    parts = []
    if stdout.strip():
        parts.append(stdout.strip())
    if value is not None:
        if isinstance(value, pd.DataFrame):
            parts.append(f"[{value.shape[0]} rows x {value.shape[1]} cols]\n{value.head(5).to_string()}")
        else:
            parts.append(repr(value))
    return "\n".join(parts) if parts else "OK (no output)"

# Agent setup
# ---------------------------------------------------------------------------
def run_agent(user_prompt: str, variable_name: str = "df"):
    """Run the ReAct agent and return BOTH the textual answer and the real
    DataFrame created during execution.

    Returns a dict:
        {
          "response":  <the AgentExecutor.invoke() result, i.e. {'input', 'output'}>,
          "df":        <the actual pandas DataFrame, or None>,
          "variables": <all Python objects stored during the run>,
        }
    """
    _reset_runtime()

    llm = ChatOpenAI(model="gpt-4o-mini")
    shell_tool = ShellTool()  # note: subprocess commands do NOT share variables

    tools_list = [load_excel_response, python_repl, get_variable, shell_tool]
    prompt = hub.pull("hwchase17/react")

    agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(
        agent=agent,
        tools=tools_list,
        verbose=True,
        handle_parsing_errors=True,  # tolerate malformed ReAct output from the LLM
        max_iterations=15,
    )

    # invoke() still returns only {'input': ..., 'output': <text>} ...
    response = agent_executor.invoke({"input": user_prompt})

    # ... so the REAL DataFrame is recovered from the shared store instead.
    df = AGENT_STORE["variables"].get(variable_name)
    if not isinstance(df, pd.DataFrame):
        # Fallbacks: the last DataFrame produced, or any stored DataFrame.
        if isinstance(AGENT_STORE["last_result"], pd.DataFrame):
            df = AGENT_STORE["last_result"]
        else:
            for name, obj in AGENT_STORE["variables"].items():
                if isinstance(obj, pd.DataFrame):
                    df = obj
                    break

    return {
        "response": response,
        "df": df,
        "variables": dict(AGENT_STORE["variables"]),
    }
