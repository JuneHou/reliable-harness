2.  The retriever was left at its default k=4 with no score threshold, so the

Request: {question}

You are given a precomputed list of control pairs.  Each pair already has a
similarity score computed from text embeddings; your job is to present the

List every pair you are given, in the same order, one pair per numbered

  <number>. [<DUPLICATE|SIMILAR>] <control A id> <-> <control B id> (similarity score: <score>)

# ---------------------------------------------------------------------------
# Data loading: one document per control (chunking removed on purpose)
# ---------------------------------------------------------------------------
def load_controls(csv_path=CSV_PATH):
    """Load every CSV row as one atomic control (Name/ID + Description)."""
    if not os.path.exists(csv_path):
        raise FileNotFoundError(f"Control dataset not found: {csv_path}")

    with open(csv_path, newline="", encoding="utf-8-sig") as handle:
        reader = csv.DictReader(handle)
        rows = [row for row in reader if any((v or "").strip() for v in row.values())]

    if not rows:
        raise ValueError(f"No usable rows found in {csv_path}")

    header = list(rows[0].keys())
    text_col = next(
        (c for c in (TEXT_COLUMN, "Control Description", "Description", "Control", "Name")
         if c in header),
        header[0],
    )
    desc_col = next(
        (c for c in ("Description", "Description of Control", "Control Description")
         if c in header and c != text_col),
        None,
    )
    id_col = next(
        (c for c in (ID_COLUMN, "Control ID", "ID", "Id") if c in header),
        None,
    )

    controls = []
    for idx, row in enumerate(rows):
        text = (row.get(text_col) or "").strip()
        if desc_col:
            desc = (row.get(desc_col) or "").strip()
            if desc:
                text = f"{text} - {desc}" if text else desc
        if not text:  # fall back to the whole row so nothing is silently dropped
            text = " | ".join(f"{k}: {v}" for k, v in row.items() if (v or "").strip())
        control_id = (
            ((row.get(id_col) or "").strip() or f"row_{idx + 1}") if id_col else f"row_{idx + 1}"
        )
        controls.append({"id": control_id, "text": text})
    return controls


# ---------------------------------------------------------------------------
# Vector store: cosine space so scores are true cosine similarities
# ---------------------------------------------------------------------------
def build_vectorstore(controls):
    """Index every control as one document in a cosine-space Chroma store."""
    documents = [
        Document(
            page_content=c["text"],
            metadata={"control_id": c["id"], "idx": i},
        )
        for i, c in enumerate(controls)
    ]
    embeddings = OpenAIEmbeddings(model=EMBEDDING_MODEL)
    return Chroma.from_documents(
        documents=documents,
        embedding=embeddings,
        collection_name="controls",
        collection_metadata={"hnsw:space": "cosine"},  # score = 1 - cosine distance
    )


# ---------------------------------------------------------------------------
# Pair detection with real similarity scores
# ---------------------------------------------------------------------------
def find_similar_pairs(vectorstore, controls,
                       similar_threshold=SIMILAR_THRESHOLD,
                       duplicate_threshold=DUPLICATE_THRESHOLD,
                       top_k=TOP_K):
    """Return every control pair whose similarity exceeds the threshold."""
    best = {}
    k = max(1, min(top_k, len(controls)))

    for i, control in enumerate(controls):
        hits = vectorstore.similarity_search_with_score(control["text"], k=k)
        for doc, distance in hits:
            j = doc.metadata.get("idx")
            if j is None or j == i:  # skip self-matches
                continue
            similarity = 1.0 - float(distance)  # cosine distance -> similarity
            if similarity < similar_threshold:
                continue
            key = (min(i, j), max(i, j))  # (A, B) == (B, A)
            if similarity > best.get(key, -1.0):
                best[key] = similarity
        if (i + 1) % 500 == 0 or i + 1 == len(controls):
            print(f"  ... scored {i + 1}/{len(controls)} controls")

    pairs = []
    for (i, j), score in best.items():
        a, b = controls[i], controls[j]
        pairs.append({
            "control_a": a["id"], "text_a": a["text"],
            "control_b": b["id"], "text_b": b["text"],
            "score": round(score, 4),
            "label": "duplicate" if score >= duplicate_threshold else "similar",
        })
    pairs.sort(key=lambda p: (-p["score"], p["control_a"], p["control_b"]))
    return pairs


def extract_threshold_from_query(query, default):
    """Let the query raise the bar, e.g. 'pairs with score above 0.9'."""
    match = re.search(
        r"(?:above|over|exceed(?:ing|s)?|greater than|at least|>=|>)\s*([01](?:\.\d+)?)",
        query or "",
        re.IGNORECASE,
    )
    if not match:
        match = re.search(r"\b(0\.\d+)\b", query or "")
    return float(match.group(1)) if match else default


def save_pairs(pairs, path=OUTPUT_CSV):
    """Persist the full result so nothing is lost to context-window limits."""
    with open(path, "w", newline="", encoding="utf-8") as handle:
        writer = csv.writer(handle)
        writer.writerow(["label", "score", "control_a", "control_b", "text_a", "text_b"])
        for p in pairs:
            writer.writerow([p["label"], p["score"], p["control_a"],
                             p["control_b"], p["text_a"], p["text_b"]])


def format_pairs_for_prompt(pairs):
    lines = []
    for n, p in enumerate(pairs, start=1):
        lines.append(
            f"{n}. {p['label'].upper()} | score={p['score']} | "
            f"{p['control_a']} <-> {p['control_b']}\n"
            f"   A: {p['text_a']}\n   B: {p['text_b']}"
        )
    return "\n".join(lines)


# ---------------------------------------------------------------------------
# Agent
# ---------------------------------------------------------------------------
def run_agent(query,
              csv_path=CSV_PATH,
              similar_threshold=SIMILAR_THRESHOLD,
              duplicate_threshold=DUPLICATE_THRESHOLD,
              top_k=TOP_K):
    """
    Answer `query` by generating and listing the similar / duplicate control
    pairs whose similarity scores exceed the specified thresholds.
    """
    # A threshold stated in the query overrides the configured default.
    query_threshold = extract_threshold_from_query(query, None)
    if query_threshold is not None:
        similar_threshold = query_threshold
    duplicate_threshold = max(duplicate_threshold, similar_threshold)

    print(f"Loading controls from {csv_path} ...")
    controls = load_controls(csv_path)
    print(f"Indexed {len(controls)} controls "
          f"(similar >= {similar_threshold}, duplicate >= {duplicate_threshold})")

    vectorstore = build_vectorstore(controls)
    pairs = find_similar_pairs(vectorstore, controls,
                               similar_threshold, duplicate_threshold, top_k)
    save_pairs(pairs)

    duplicates = [p for p in pairs if p["label"] == "duplicate"]
    similar = [p for p in pairs if p["label"] == "similar"]
    summary = (f"Analysed {len(controls)} controls.  Found {len(pairs)} pairs above the "
               f"similarity threshold {similar_threshold}:  {len(duplicates)} duplicate "
               f"pairs (>= {duplicate_threshold}) and {len(similar)} similar pairs.  "
               f"Full list saved to {OUTPUT_CSV}.")
    print(summary)

    # Deterministic listing: always printed so the output matches the request.
    for n, p in enumerate(pairs, start=1):
        print(f"{n}. [{p['label'].upper()}] {p['control_a']} <-> {p['control_b']} "
              f"(score: {p['score']})")

    if not pairs:
        return summary

    # LLM step: format the precomputed pairs into the requested listing.
    # Scores come from the embeddings, the model only presents them.
    shown = pairs[:MAX_PAIRS_FOR_LLM]
    try:
        prompt = ChatPromptTemplate.from_template(REPORT_TEMPLATE)
        llm = ChatOpenAI(temperature=0, model=LLM_MODEL)
        rag_chain = (
            {
                "pairs": format_pairs_for_prompt(shown),
                "duplicate_threshold": duplicate_threshold,
                "question": RunnablePassthrough(),
            }
            | prompt
            | llm
            | StrOutputParser()
        )
        output = rag_chain.invoke(query)
        if len(pairs) > len(shown):
            output += (f"\n\n(Note: {len(pairs) - len(shown)} additional pairs were found and "
                       f"written to {OUTPUT_CSV}; only the top {len(shown)} are listed above.)")
        return output
    except Exception as exc:
        # The pair list above is authoritative even if the model call fails.
        return (f"LLM formatting failed ({exc}).  The deterministic pair list "
                f"printed above (and saved to {OUTPUT_CSV}) is the result.")


if __name__ == "__main__":

    duplicate_threshold = max(duplicate_threshold, similar_threshold)

    print(f"Loading controls from {csv_path} ...")
    controls = load_controls(csv_path)
    print(f"Indexed {len(controls)} controls "
          f"(similar >= {similar_threshold}, duplicate >= {duplicate_threshold})")

    vectorstore = build_vectorstore(controls)
    pairs = find_similar_pairs(vectorstore, controls,
                               similar_threshold, duplicate_threshold, top_k)
    save_pairs(pairs)

    duplicates = [p for p in pairs if p["label"] == "duplicate"]
    similar = [p for p in pairs if p["label"] == "similar"]
    summary = (f"Analysed {len(controls)} controls.  Found {len(pairs)} pairs above the "
               f"similarity threshold {similar_threshold}:  {len(duplicates)} duplicate "
               f"pairs (>= {duplicate_threshold}) and {len(similar)} similar pairs.  "
               f"Full list saved to {OUTPUT_CSV}.")
    print(summary)

    # Deterministic listing: always printed so the output matches the request.
    for n, p in enumerate(pairs, start=1):
        print(f"{n}. [{p['label'].upper()}] {p['control_a']} <-> {p['control_b']} "
              f"(score: {p['score']})")

    if not pairs:
        return summary

    # LLM step: format the precomputed pairs into the requested listing.
    # Scores come from the embeddings, the model only presents them.
    shown = pairs[:MAX_PAIRS_FOR_LLM]
    try:
        prompt = ChatPromptTemplate.from_template(REPORT_TEMPLATE)
        llm = ChatOpenAI(temperature=0, model=LLM_MODEL)
        rag_chain = (
            {
                "pairs": format_pairs_for_prompt(shown),
                "duplicate_threshold": duplicate_threshold,
                "question": RunnablePassthrough(),
            }
            | prompt
            | llm
            | StrOutputParser()
        )
        output = rag_chain.invoke(query)
        if len(pairs) > len(shown):
            output += (f"\n\n(Note: {len(pairs) - len(shown)} additional pairs were found and "
                       f"written to {OUTPUT_CSV}; only the top {len(shown)} are listed above.)")
        return output
    except Exception as exc:
        # The pair list above is authoritative even if the model call fails.
        return (f"LLM formatting failed ({exc}).  The deterministic pair list "
                f"printed above (and saved to {OUTPUT_CSV}) is the result.")


if __name__ == "__main__":

    result = run_agent(query)
