import inspect
import os
import random
from typing import Any, Dict, Optional

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.vectorstores.base import VectorStoreRetriever
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]


def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


class MetadataFilterConversationalRetrievalChain(ConversationalRetrievalChain):
    """``ConversationalRetrievalChain`` that honours a metadata ``filter`` passed
    in together with the query, so the filter is applied dynamically at query
    time and the very same chain instance can be reused for different filter
    values (no need to build a separate chain per filter).

    Why the original code produced the wrong output:
    The stock ``ConversationalRetrievalChain`` only consumes the ``question``
    and ``chat_history`` input keys.  The extra ``filter`` key passed in
    ``qa({... "filter": filter})`` is silently dropped on the floor, so the
    retriever searches the *whole* Chroma collection (paper1.pdf AND
    paper2.pdf) and the answer can be based on documents that were supposed to
    be filtered out.

    How this fixes it:
    This subclass grabs the ``filter`` for the current query out of the raw
    chain inputs and injects it into the retriever's ``search_kwargs`` before
    the parent chain performs retrieval.  Chroma turns
    ``search_kwargs["filter"]`` into a metadata ``where`` clause, so only the
    documents whose metadata matches (here: chunks coming from paper2.pdf) are
    retrieved and used to answer.
    """

    def _apply_query_time_filter(self, filter: Optional[Dict[str, Any]]) -> None:
        """Attach (or clear) the metadata filter on the retriever."""
        retriever = getattr(self, "retriever", None)
        search_kwargs = getattr(retriever, "search_kwargs", None)
        if not isinstance(search_kwargs, dict):
            return
        if filter:
            search_kwargs["filter"] = filter
        else:
            # No filter for this query -> clear any filter left over from a
            # previous call so the same chain can also be reused unfiltered.
            search_kwargs.pop("filter", None)

    def _call(self, inputs: Dict[str, Any], run_manager=None, **kwargs: Any) -> Dict[str, Any]:
        # Apply the metadata filter that came in with this query *before* the
        # chain performs retrieval.
        self._apply_query_time_filter(inputs.get("filter"))

        # Delegate to the parent implementation.  ``_call`` gained a
        # ``run_manager`` argument in newer langchain versions, so inspect the
        # parent signature to stay compatible with both the old and new API.
        parent_call = super(MetadataFilterConversationalRetrievalChain, self)._call
        try:
            accepts_run_manager = "run_manager" in inspect.signature(parent_call).parameters
        except (TypeError, ValueError):
            accepts_run_manager = True
        if accepts_run_manager:
            return parent_call(inputs, run_manager=run_manager, **kwargs)
        return parent_call(inputs, **kwargs)


def run_agent(query, pdf_paths, filter=None):
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            text.metadata["source"] = pdf_path

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    # Metadata filter applied dynamically at query time.
    # It means the information from paper2.pdf should be used to answer the query.
    # If it answers from paper2.pdf, then the filtering is working correctly.
    if filter is None:
        filter = {'source': 'paper2.pdf'}

    # The filter travels with the query through the retriever's search_kwargs
    # (Chroma metadata filter), so retrieval is restricted to matching documents.
    retriever = VectorStoreRetriever(
        vectorstore=db,
        search_kwargs={"filter": filter} if filter else {},
    )

    qa = MetadataFilterConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever,
        verbose=True,
        return_source_documents=True)

    # The filter can be passed in with the inputs and changed on every call on
    # the very same chain instance - there is no need to rebuild the chain for
    # each filter value, e.g.:
    #   qa({"question": q, "chat_history": [], "filter": {"source": "paper1.pdf"}})
    #   qa({"question": q, "chat_history": [], "filter": {"source": "paper2.pdf"}})
    #   qa({"question": q, "chat_history": []})  # no filter -> whole store
    result = qa({"question": query, "chat_history": [], "filter": filter})

    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    result = run_agent(query, pdf_paths)

    print(result['answer'])
