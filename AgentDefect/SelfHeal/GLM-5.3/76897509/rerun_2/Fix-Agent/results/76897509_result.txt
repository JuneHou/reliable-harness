import os
import random
from typing import List

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.schema import BaseRetriever, Document
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]


def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


class DynamicFilterRetriever(BaseRetriever):
    """Retriever that lets the caller pass a Chroma `where` metadata filter
    at query time through the chain's inputs.

    ConversationalRetrievalChain ignores an extra "filter" input by default,
    so we capture the filter from the chain inputs in `_get_relevant_documents`
    and apply it directly to the underlying Chroma collection.
    """

    vectorstore: Chroma
    search_kwargs: dict = {}

    def _get_relevant_documents(self, query: str, *, run_manager=None) -> List[Document]:
        filter = self.search_kwargs.get("filter")
        return self.vectorstore.similarity_search(
            query,
            k=self.search_kwargs.get("k", 4),
            filter=filter,
        )

    async def _aget_relevant_documents(self, query: str, *, run_manager=None) -> List[Document]:
        filter = self.search_kwargs.get("filter")
        return await self.vectorstore.asimilarity_search(
            query,
            k=self.search_kwargs.get("k", 4),
            filter=filter,
        )


def _apply_filter_at_runtime(retriever: DynamicFilterRetriever, filter: dict) -> None:
    """Inject the per-query filter into the retriever before the chain runs."""
    retriever.search_kwargs["filter"] = filter


def run_agent(query, pdf_paths, filter=None):
    # Nothing to build if no PDFs are provided
    if not pdf_paths:
        raise ValueError("pdf_paths must contain at least one PDF path")

    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        # Guard against paths that don't exist so the failure is explicit
        if not os.path.exists(pdf_path):
            raise FileNotFoundError(f"PDF not found: {pdf_path}")

        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            # Use just the file name so it matches the filter values users pass in
            text.metadata["source"] = os.path.basename(pdf_path)

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    retriever = DynamicFilterRetriever(vectorstore=db, search_kwargs={"k": 4})

    qa = ConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever,
        verbose=True,
        return_source_documents=True)

    # Apply the metadata filter dynamically at query time instead of
    # hard-coding it inside the chain/retriever construction.
    filter = {'source': 'paper2.pdf'}  # it means the information from paper2.pdf should be used to answer the query.
    _apply_filter_at_runtime(retriever, filter)

    result = qa({"question": query, "chat_history": []})

    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    result = run_agent(query, pdf_paths)

    print(result['answer'])

    if result.get("source_documents"):
        print("\nSources used:")
        for doc in result["source_documents"]:
            print(" -", doc.metadata.get("source"))
