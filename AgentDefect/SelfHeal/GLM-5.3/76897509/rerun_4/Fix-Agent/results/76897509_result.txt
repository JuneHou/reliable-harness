import os
import random
from typing import Optional
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.vectorstores.base import VectorStoreRetriever
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]

def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


def run_agent(query, pdf_paths, filter=None):
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))
    
    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()
        
        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)
        
        for text in texts:
            text.metadata["source"] = pdf_path
        
        all_texts.extend(texts)
    
    db = Chroma.from_documents(all_texts, embeddings)
    
    # Build the retriever once. Its search_kwargs are forwarded to
    # Chroma's similarity_search(...) on every query, which natively accepts a
    # `filter` argument. That is the correct place to inject a metadata filter.
    retriever = VectorStoreRetriever(
        vectorstore=db,
        search_kwargs={"k": 4},
    )
    
    qa = ConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever,
        verbose=True,
        return_source_documents=True)

    # NOTE: ConversationalRetrievalChain only accepts the input keys
    # "question" and "chat_history". A "filter" key passed inside the qa(...)
    # input dict is silently ignored (that was the bug), so the retrieval ran
    # over all documents instead of just the filtered source.
    #
    # To apply the metadata filter dynamically at query time WITHOUT creating
    # a new chain instance for each filter value, we update the retriever's
    # search_kwargs right before invoking the (already built) chain. The next
    # query can simply set a different (or no) filter the same way.
    if filter is not None:
        retriever.search_kwargs["filter"] = filter

    result = qa({"question": query, "chat_history": []})
    
    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    filter = {'source': 'paper2.pdf'}  # the information from paper2.pdf should be used to answer the query.
    
    result = run_agent(query, pdf_paths, filter=filter)
    
    print(result['answer'])
    
    # Verify the filter actually worked: every retrieved source document should
    # come from paper2.pdf only.
    sources = {doc.metadata.get('source') for doc in result.get('source_documents', [])}
    print("\nSources used:", sources)
