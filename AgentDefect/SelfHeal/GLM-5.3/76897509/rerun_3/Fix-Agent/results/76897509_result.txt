import inspect
import os
import random
from typing import Any, Dict, List, Optional

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.vectorstores.base import VectorStoreRetriever
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache
from langchain.docstore.document import Document

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]


def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


class MetadataFilteredConversationalRetrievalChain(ConversationalRetrievalChain):
    """ConversationalRetrievalChain that applies a metadata filter at query time.

    The stock chain only consumes the ``question`` and ``chat_history`` input
    keys, so a ``filter`` key placed in the chain inputs is silently dropped:
    the retriever searches the whole vector store (both PDFs) and the answer
    therefore does not respect the requested filter.

    This subclass picks the ``filter`` key out of the chain inputs and hands
    it to the Chroma vector store when the documents are retrieved:

        chain({
            "question": "What is Theia",
            "chat_history": [],
            "filter": {"source": "paper2.pdf"},
        })

    Because the filter travels with the query, one single chain instance can
    serve every filter value -- no need to build one chain per filter.
    """

    def _call(self, inputs: Dict[str, Any], run_manager: Any = None,
              **kwargs: Any) -> Dict[str, Any]:
        """Remember the per-call filter, then run the chain as usual."""
        # Stash the filter so _get_docs() can use it even on LangChain
        # versions where the base chain only forwards
        # {"question": ..., "chat_history": ...} to _get_docs().
        # (Pydantic-backed chains can be picky about assigning private
        # attributes, so set it at the raw object level.)
        object.__setattr__(self, "_query_filter", inputs.get("filter"))

        # The base chain only understands "question" and "chat_history";
        # strip the extra key so it is not forwarded anywhere else.
        chain_inputs = {k: v for k, v in inputs.items() if k != "filter"}

        base_call = super()._call
        if "run_manager" in inspect.signature(base_call).parameters:
            # Newer LangChain versions pass a run manager to _call.
            return base_call(chain_inputs, run_manager=run_manager, **kwargs)
        # Older LangChain versions only accept the inputs.
        return base_call(chain_inputs, **kwargs)

    def _get_docs(self, inputs: Dict[str, Any], run_manager: Any = None,
                  **kwargs: Any) -> List[Document]:
        """Retrieve documents while honouring the metadata filter of this call."""
        # The filter either arrives with the inputs (older versions forward
        # the raw input dict) or was stashed away by _call() (newer versions
        # build a fresh dict without the extra keys).
        metadata_filter = inputs.get("filter")
        if metadata_filter is None:
            metadata_filter = getattr(self, "_query_filter", None)

        # "question" is already the standalone question for this turn.
        question = inputs["question"]

        retriever = self.retriever
        search_kwargs = dict(getattr(retriever, "search_kwargs", None) or {})
        if metadata_filter is not None:
            search_kwargs["filter"] = metadata_filter

        vectorstore = getattr(retriever, "vectorstore", None)
        if vectorstore is not None:
            # Query the vector store directly: this guarantees the metadata
            # filter is applied no matter how the retriever forwards its
            # arguments (e.g. Chroma's similarity_search(query, k, filter)).
            return vectorstore.similarity_search(question, **search_kwargs)
        return retriever.get_relevant_documents(question, **search_kwargs)


def run_agent(query, pdf_paths):
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            text.metadata["source"] = pdf_path

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    qa = MetadataFilteredConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        VectorStoreRetriever(vectorstore=db),
        verbose=True,
        return_source_documents=True)

    filter = {'source': 'paper2.pdf'}  # it means the information from paper2.pdf should be used to answer the query.
    # if it answers from paper2.pdf, then the filtering is working correctly.

    # The "filter" key is now honoured at query time (see the chain subclass
    # above), so only documents from paper2.pdf are retrieved and used.
    result = qa({"question": query, "chat_history": [], "filter": filter})

    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    result = run_agent(query, pdf_paths)

    print(result['answer'])
