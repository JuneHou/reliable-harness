import os
import random
from typing import Any, Dict, Optional

from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.vectorstores.base import VectorStoreRetriever
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

OPENAI_API_KEY_POOL = [
    os.getenv('OPENAI_API_KEY'),
]


def get_random_key(key_pool):
    return random.choice([k for k in key_pool if k is not None])


def apply_metadata_filter(retriever: VectorStoreRetriever,
                          metadata_filter: Optional[Dict[str, Any]] = None) -> None:
    """
    Set (or clear) the Chroma metadata filter of an existing retriever at query time.

    ConversationalRetrievalChain only reads the "question" and "chat_history"
    keys from its input dictionary: any other key (e.g. "filter") is silently
    ignored, so a filter placed in the chain inputs never reaches the vector
    store. The filter has to be handed to the *retriever* through its
    `search_kwargs` instead (Chroma's `similarity_search` accepts a `filter`
    argument). Because `search_kwargs` is just a plain dict on the retriever
    that the chain already holds, it can be swapped for every query without
    creating a new chain instance.
    """
    if metadata_filter:
        retriever.search_kwargs["filter"] = metadata_filter
    else:
        retriever.search_kwargs.pop("filter", None)


def run_agent(query, pdf_paths, metadata_filter=None):
    all_texts = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        texts = text_splitter.split_documents(documents)

        for text in texts:
            text.metadata["source"] = pdf_path

        all_texts.extend(texts)

    db = Chroma.from_documents(all_texts, embeddings)

    retriever = db.as_retriever(search_kwargs={"k": 4})

    qa = ConversationalRetrievalChain.from_llm(
        OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever,
        verbose=True,
        return_source_documents=True)

    # Apply the metadata filter dynamically at query time on the very same
    # chain/retriever.  Note: no "filter" key is put into the chain inputs,
    # because ConversationalRetrievalChain would simply drop it.
    apply_metadata_filter(qa.retriever, metadata_filter)

    result = qa({"question": query, "chat_history": []})

    return result


if __name__ == "__main__":
    query = "What is Theia"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]

    # It means the information from paper2.pdf should be used to answer the query.
    # If it answers from paper2.pdf, then the filtering is working correctly.
    metadata_filter = {'source': 'paper2.pdf'}

    result = run_agent(query, pdf_paths, metadata_filter=metadata_filter)

    print(result['answer'])

    # Sanity check that only chunks coming from paper2.pdf were retrieved.
    sources = sorted({doc.metadata.get('source')
                      for doc in result.get('source_documents', [])})
    print('Sources used for the answer:', sources)
