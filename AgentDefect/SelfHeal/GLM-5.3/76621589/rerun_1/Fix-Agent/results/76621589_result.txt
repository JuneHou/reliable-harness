import asyncio

def run_agent(texts):
    
    import os
    from time import time

    import openai
    from dotenv import load_dotenv, find_dotenv
    from langchain.chains import LLMChain
    from langchain.chat_models import ChatOpenAI
    from langchain.prompts import ChatPromptTemplate

    _ = load_dotenv(find_dotenv())
    openai.api_key = os.getenv('OPENAI_API_KEY')

    llm = ChatOpenAI(temperature=0)

    prompt = ChatPromptTemplate.from_template(
        'Classify the text based on the Common European Framework of Reference '
        'for Languages (CEFR). Give a single value: {text}',
    )
    chain = LLMChain(llm=llm, prompt=prompt)

    start = time()
    res_a = chain.apply(texts)
    print(res_a)
    print(f"apply time taken: {time() - start:.2f} seconds")
    print()

    # FIX: aapply() is a coroutine function. Calling it directly returns an
    # unawaited coroutine object instead of the results. Drive it with an
    # event loop via asyncio.run() so the coroutine is actually awaited and
    # returns the list of classification results (matching `apply`'s output).
    start = time()
    res_aa = asyncio.run(chain.aapply(texts))
    print(res_aa)
    print(f"aapply time taken: {time() - start:.2f} seconds")
    
    return res_a



if __name__ == "__main__":
    texts = [
        {'text': 'Hallo, ich bin 25 Jahre alt.'},

        {'text': 'In meiner Freizeit, spiele ich gerne Fussball.'}
    ]
    
    result = run_agent(texts)
    print("\nReturned result:", result)

import asyncio
import os
from time import time

import openai
from dotenv import load_dotenv, find_dotenv
from langchain.chains import LLMChain
from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate

_ = load_dotenv(find_dotenv())
openai.api_key = os.getenv('OPENAI_API_KEY')

llm = ChatOpenAI(temperature=0)

prompt = ChatPromptTemplate.from_template(
    'Classify the text based on the Common European Framework of Reference '
    'for Languages (CEFR). Give a single value: {text}',
)
chain = LLMChain(llm=llm, prompt=prompt)


async def run_chain_async(texts):
    """Run the chain asynchronously and await its completion."""
    # aapply is a coroutine function: it must be awaited (or gathered).
    # Awaiting directly on aapply lets LangChain handle concurrent
    # execution of the input list internally.
    return await chain.aapply(texts)


def run_agent(texts):
    # --- Synchronous baseline ---
    start = time()
    res_a = chain.apply(texts)
    print(res_a)
    print(f"apply time taken: {time() - start:.2f} seconds")
    print()

    # --- Asynchronous path ---
    # The bug: calling chain.aapply(texts) directly returns an un-awaited
    # coroutine object (printed as <coroutine object ...>) instead of results.
    # The fix: execute it inside an event loop via asyncio.run(), which
    # schedules the coroutine and returns its actual result.
    start = time()
    try:
        res_aa = asyncio.run(run_chain_async(texts))
    except RuntimeError:
        # If an event loop is already running in this environment
        # (e.g. inside a Jupyter notebook or an outer async app),
        # asyncio.run() cannot create a nested loop; use the existing
        # loop's task machinery instead via a new task + run_until_complete
        # on a fresh loop is not possible, so fall back to awaiting via
        # nest-style scheduling through the running loop.
        try:
            loop = asyncio.get_event_loop()
        except RuntimeError:
            loop = asyncio.new_event_loop()
            asyncio.set_event_loop(loop)
        res_aa = loop.run_until_complete(run_chain_async(texts))

    print(res_aa)
    print(f"aapply time taken: {time() - start:.2f} seconds")

    return res_a


if __name__ == "__main__":
    texts = [
        {'text': 'Hallo, ich bin 25 Jahre alt.'},
        {'text': 'Wie geht es dir?'},
        {'text': 'In meiner Freizeit, spiele ich gerne Fussball.'}
    ]

    result = run_agent(texts)
    print("\nReturned result:", result)
