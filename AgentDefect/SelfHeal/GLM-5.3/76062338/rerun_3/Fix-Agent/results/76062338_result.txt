import os
from typing import Any, List, Mapping, Optional
from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool
from langchain.callbacks.manager import CallbackManagerForLLMRun

load_dotenv()


class wrapper(LLM):
    llm: OpenAI
    log: str

    @property
    def _llm_type(self) -> str:
        return "custom"

    def _call(
        self,
        prompt: str,
        stop: Optional[List[str]] = None,
        run_manager: Optional[CallbackManagerForLLMRun] = None,
    ) -> str:
        """Call the wrapped LLM and return its (truncated) reply.

        CRITICAL FIX: the agent's `stop` sequences (e.g. "\nObservation:") MUST be
        forwarded to the underlying LLM. ReAct agents rely on these stop sequences
        to cut off the model's generation right before "Observation:". The original
        code called `self.llm(prompt)` without `stop`, so the wrapped LLM kept
        generating and hallucinated an extra "Observation: ..." line, which made
        the agent's output differ from the output produced with the original LLM.
        """
        # Forward the stop sequences so generation is truncated exactly like it
        # is when the agent talks directly to the original OpenAI LLM.
        reply = self.llm(prompt, stop=stop)

        # Safety net: emulate the OpenAI `stop` semantics (returned text excludes
        # the stop sequence and everything after it) in case the wrapped provider
        # still emits one of the stop sequences. This guarantees the agent sees
        # exactly the same text it would get from the original LLM.
        if stop:
            cut = len(reply)
            for s in stop:
                if s:
                    idx = reply.find(s)
                    if idx != -1 and idx < cut:
                        cut = idx
            reply = reply[:cut]

        # Log prompt/reply (the same reply the agent actually receives).
        # Use a context manager so the file is always flushed and closed.
        with open(self.log, "a") as f:
            f.write('\n{\n' + prompt + '\n}{\n' + reply + '\n}\n' + '.' * 100)

        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        """Get the identifying parameters."""
        return {"llm": self.llm}


def setup_components():
  
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0
    )

    wllm = wrapper(llm=llm, log='output.log')

    tools = [PythonAstREPLTool()]
    
    memory = ConversationBufferMemory(memory_key="chat_history")
    
    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory
    )
    return agent


def run_agent(prompt):
    
    try:
        wllm, tools, memory = setup_components()
        
    except Exception as e:
        print(f"Error during setup: {e}")
        return None
    
    try:
        agent = create_agent(wllm, tools, memory)
        
    except Exception as e:
        print(f"Error creating agent: {e}")
        return None
    
    try:
        response = agent(prompt)
        return response
    except Exception as e:
        print(f"Error running agent: {e}")
        return None


if __name__ == "__main__":

    prompt = "What is 5-7?"
    
    response = run_agent(prompt)
    
    print(response)
    if response is not None:
        print(type(response), response["output"])
