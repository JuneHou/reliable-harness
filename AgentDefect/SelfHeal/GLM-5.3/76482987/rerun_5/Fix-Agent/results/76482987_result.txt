    collection = chroma_client.create_collection(name="my_collection")
    collection.add(
        embeddings=[[1.2, 2.3, 4.5], [6.7, 8.2, 9.2]],
        documents=documents,
        metadatas=[{"source": "my_source"}, {"source": "my_source"}],
        ids=["id1", "id2"]
    )   

    # ChromaDB's get() excludes embeddings by default (include defaults to
    # ["metadatas", "documents"]), which is why "embeddings" came back as None.
    # Explicitly request embeddings (along with documents and metadatas) so the
    # returned result contains the embeddings that were stored via add().
