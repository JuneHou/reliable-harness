from dotenv import load_dotenv
from pathlib import Path
import json
import os
import sys

import openai
from langchain.chains import ConversationalRetrievalChain, RetrievalQA
from langchain.chat_models import ChatOpenAI
from langchain.document_loaders import DirectoryLoader, TextLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.indexes import VectorstoreIndexCreator
from langchain.indexes.vectorstore import VectorStoreIndexWrapper
from langchain.llms import OpenAI
from langchain.prompts import PromptTemplate
from langchain.vectorstores import Chroma
from langchain.document_loaders import JSONLoader

load_dotenv()

# Grounded QA prompt: force the model to answer ONLY from the retrieved reviews
# (no outside knowledge, no made-up answers) and to consider every review in
# the context when counting / summarizing.
QA_PROMPT_TEMPLATE = """Use the following pieces of context (hotel reviews) to answer the question at the end.
Answer strictly and only based on the reviews provided in the context. Do not use any outside knowledge and do not make up information. When counting or summarizing, carefully consider every review in the context.

{context}

Question: {question}
Helpful Answer:"""


def run_agent(queries):

  data_dir = Path("data")
  data_dir.mkdir(exist_ok=True)

  review_file = data_dir / "review.json"

  if not review_file.exists():
      payload = {
          "reviews": [
              {"text": "Great hotel, excellent service and comfortable rooms."},
              {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
              {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
              {"text": "Average hotel. The room was okay, but nothing special."},
              {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
              {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
              {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
              {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
              {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
              {"text": "Not a bad hotel, but there are better options available in the area."}
          ]
      }
      review_file.write_text(json.dumps(payload, indent=2))



  # Enable to save to disk & reuse the model (for repeated queries on the same data)
  PERSIST = False

  query = None
  if len(sys.argv) > 1:
    query = sys.argv[1]


  # Correct loader configuration:
  #   jq_schema=".reviews[]" iterates over EVERY review object in the "reviews"
  #   array, and content_key="text" extracts each review's text, so each review
  #   becomes its own Document (10 separate documents, not 1 blob of raw JSON).
  loader = JSONLoader(
      file_path=str(review_file),
      jq_schema=".reviews[]",
      content_key="text",
  )

  if PERSIST and os.path.exists("persist"):
    print("Reusing index...\n")
    vectorstore = Chroma(persist_directory="persist", embedding_function=OpenAIEmbeddings())
    index = VectorStoreIndexWrapper(vectorstore=vectorstore)
  else:
    if PERSIST:
      index = VectorstoreIndexCreator(vectorstore_kwargs={"persist_directory": "persist"}).from_loaders([loader])
    else:
      index = VectorstoreIndexCreator().from_loaders([loader])

  # Figure out how many review documents are actually stored in the vectorstore
  # so that we can retrieve ALL of them for every question.
  try:
      n_docs = index.vectorstore._collection.count()
  except Exception:
      n_docs = len(loader.load())

  # ---------------------------------------------------------------------
  # THE FIX:
  # `as_retriever()` defaults to a similarity search with k=4, i.e. only the
  # 4 most similar reviews were passed to the LLM for every question. For
  # questions about the whole review set ("how many reviews...", "what is the
  # overall sentiment...", "did anyone complain about...") the model only ever
  # saw a fraction of the data, which produced incorrect/incomplete answers.
  # Retrieve every review document instead so the model always sees the full
  # dataset, and the loader/jq_schema above guarantees each review is indexed
  # as its own retrievable document.
  # ---------------------------------------------------------------------
  retriever = index.vectorstore.as_retriever(
      search_type="similarity",
      search_kwargs={"k": n_docs},
  )

  chain = ConversationalRetrievalChain.from_llm(
    llm=ChatOpenAI(model="gpt-3.5-turbo", temperature=0),  # temperature=0 for accurate, deterministic answers
    retriever=retriever,
    chain_type="stuff",
    chain_type_kwargs={"prompt": PromptTemplate.from_template(QA_PROMPT_TEMPLATE)},
  )

  chat_history = []
  results = []
  for query in queries:
    if query in ['quit', 'q', 'exit']:
      sys.exit()
    result = chain({"question": query, "chat_history": chat_history})
    print(result['answer'])
    results.append(result['answer'])
    chat_history.append((query, result['answer']))
    query = None
  return results
