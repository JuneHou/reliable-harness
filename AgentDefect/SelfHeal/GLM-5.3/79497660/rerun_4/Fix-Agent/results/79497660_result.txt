"""
Fix: persist metadata such as ``external_id`` in a *dedicated column* of the
PostgreSQL table used by LlamaIndex's ``PGVectorStore``.

Why the original code did not match the expected output
-------------------------------------------------------
``PGVectorStore`` hard-codes its table layout (``id``, ``name``, ``text``,
``metadata``, ``embeddings``) and offers no constructor parameter for extra
columns.  So ``Document(..., metadata={"external_id": "ext_001"})`` could only
end up buried inside the ``metadata`` JSON column -- it never became a real,
directly queryable column of the table.

The fix
-------
All DDL and insert logic of ``PGVectorStore`` lives in small, overridable
methods, so the supported way to extend the schema is to subclass the store
and extend its hooks:

1. ``_create_tables_if_not_exists`` -- runs when the store is initialized.
   We let the base class build its usual table and then run
   ``ALTER TABLE ... ADD COLUMN IF NOT EXISTS external_id VARCHAR`` (plus an
   index) so that ``external_id`` becomes part of the real table schema.

2. ``_get_node_with_embedding`` / ``_aget_node_with_embedding`` -- the
   (a)sync INSERT paths used while indexing.  After the base class stores a
   node we write the custom metadata keys into their own columns.

The resulting table can be used like any normal PostgreSQL table, e.g.::

    SELECT id, external_id, text FROM public.vectorstore
    WHERE external_id = 'ext_001';

The value is mirrored in the ``metadata`` JSON column as well, which keeps
LlamaIndex retrieval and metadata filters working unchanged.
"""

# Every metadata key listed here is persisted in its own column of the
# vector-store table (in addition to the metadata JSON).

class CustomPGVectorStore(PGVectorStore):
    """A ``PGVectorStore`` that stores selected metadata keys (such as
    ``external_id``) in real table columns instead of only inside the
    ``metadata`` JSON column."""

    # ------------------------------------------------------------------ #
    # 1. Schema creation: add the custom columns to the table            #
    # ------------------------------------------------------------------ #
    def _create_tables_if_not_exists(self) -> None:
        """Create the stock table first, then append the custom columns."""
        # Base implementation creates `id / name / text / metadata /
        # embeddings`, enables the pgvector extension and builds the HNSW
        # index.
        super()._create_tables_if_not_exists()
        self.ensure_custom_columns()

    def ensure_custom_columns(self) -> None:
        """Idempotently add the custom columns (and their indexes) to the
        table.  Safe to call at any time, also on tables that already
        contain rows."""
        engine = getattr(self, "_sync_engine", None) or getattr(self, "_engine", None)
        if engine is None:
            return

        with engine.connect() as connection:
            for column_name, column_type in CUSTOM_COLUMNS.items():
                # Idempotent: works for brand new *and* pre-existing tables.
                connection.execute(
                    text(
                        f"ALTER TABLE {self.schema_name}.{self.table_name} "
                        f"ADD COLUMN IF NOT EXISTS {column_name} {column_type}"
                    )
                )
                # An index makes direct lookups / deletes by the column fast.
                connection.execute(
                    text(
                        f"CREATE INDEX IF NOT EXISTS "
                        f"{self.table_name}_{column_name}_idx "
                        f"ON {self.schema_name}.{self.table_name} ({column_name})"
                    )
                )
            connection.commit()

    # ------------------------------------------------------------------ #
    # 2. Indexing: populate the custom columns                            #
    # ------------------------------------------------------------------ #
    def _get_node_with_embedding(
        self,
        embedding_results: List[Tuple[str, Any]],
        **llm_kwargs: Any,
    ) -> List[str]:
        """Index nodes, writing custom metadata keys into their own columns."""
        # Let the library perform its regular INSERT (id, name, text,
        # metadata, embeddings).  Re-using the base implementation keeps this
        # subclass compatible with every llama-index-vector-stores-postgres
        # release.
        ids = super()._get_node_with_embedding(embedding_results, **llm_kwargs)

        # Now write the custom columns for the rows that were just inserted.
        if CUSTOM_COLUMNS and embedding_results:
            update_stmt = self._custom_columns_update_stmt()
            with self._sync_engine.connect() as conn:
                for node_id, node in embedding_results:
                    conn.execute(
                        update_stmt, self._custom_column_params(node_id, node)
                    )
                conn.commit()

        return ids

    async def _aget_node_with_embedding(
        self,
        embedding_results: List[Tuple[str, Any]],
        **llm_kwargs: Any,
    ) -> List[str]:
        """Async variant of ``_get_node_with_embedding``."""
        parent_impl = getattr(super(), "_aget_node_with_embedding", None)
        if parent_impl is None:
            # Library version without async support -> fall back to sync.
            return self._get_node_with_embedding(embedding_results, **llm_kwargs)

        ids = await parent_impl(embedding_results, **llm_kwargs)

        if CUSTOM_COLUMNS and embedding_results:
            update_stmt = self._custom_columns_update_stmt()
            async with self._async_engine.connect() as conn:
                for node_id, node in embedding_results:
                    await conn.execute(
                        update_stmt, self._custom_column_params(node_id, node)
                    )
                await conn.commit()

        return ids

    # ------------------------------------------------------------------ #
    # Helpers                                                             #
    # ------------------------------------------------------------------ #
    def _custom_columns_update_stmt(self):
        """Build ``UPDATE <table> SET <custom cols...> WHERE id = :node_id``."""
        set_clause = ", ".join(f"{column} = :{column}" for column in CUSTOM_COLUMNS)
        return text(
            f"UPDATE {self.schema_name}.{self.table_name} "
            f"SET {set_clause} "
            f"WHERE id = :node_id"
        )

    @staticmethod
    def _custom_column_params(node_id: str, node: Any) -> Dict[str, Any]:
        """Bind parameters for one node's custom columns."""
        params: Dict[str, Any] = {"node_id": node_id}
        for column in CUSTOM_COLUMNS:
            value = node.metadata.get(column)
            params[column] = None if value is None else str(value)
        return params

    # added, PGVectorStore moves its value out of the metadata JSON and into

        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )
