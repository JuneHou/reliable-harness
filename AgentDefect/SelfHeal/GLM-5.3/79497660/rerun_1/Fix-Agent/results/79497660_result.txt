from llama_index.core import VectorStoreIndex, Document
from llama_index.vector_stores.postgres import PGVectorStore
from sqlalchemy import Column, String, create_engine, inspect, text

# psycopg2 is the driver used by SQLAlchemy's "postgresql+psycopg2://" URLs
import psycopg2  # noqa: F401
import os
from dotenv import load_dotenv

load_dotenv()

# ---------------------------------------------------------------------------
# PostgreSQL connection details
# ---------------------------------------------------------------------------
DB_CONFIG = {
    "host": "localhost",
    "password": "password",
    "port": 5432,
    "user": "postgres",
}
TABLE_NAME = "vectorstore"
FULL_TABLE_NAME = f"data_{TABLE_NAME}"

# ---------------------------------------------------------------------------
# THE FIX: PGVectorStore.from_params() accepts a `metadata_cols` argument
# (a list of SQLAlchemy Column objects).  Each listed column is created as a
# REAL column of the `data_vectorstore` table.  When a Document's metadata
# contains a key matching one of these column names, PGVectorStore pops that
# key out of the metadata JSON blob and writes the value into its own
# dedicated column instead.  Without this parameter every metadata key
# (including external_id) is only nested inside the `metadata` JSON column.
# ---------------------------------------------------------------------------
METADATA_COLS = [
    Column("external_id", String, nullable=True),
]


def _engine_for(tenant: str):
    return create_engine(
        "postgresql+psycopg2://{user}:{password}@{host}:{port}/{db}".format(
            db=tenant, **DB_CONFIG
        )
    )


def _ensure_custom_columns_in_schema(tenant: str) -> None:
    """PGVectorStore only creates the table if it does not exist yet.

    If an earlier run already created `data_vectorstore` WITHOUT the
    external_id column, PGVectorStore never performs an ALTER TABLE, so the
    new custom column would never be added and inserts would fail.  In that
    case the stale table is dropped so it is recreated with the new schema.
    """
    engine = _engine_for(tenant)
    try:
        inspector = inspect(engine)
        if inspector.has_table(FULL_TABLE_NAME):
            existing_cols = {
                col["name"] for col in inspector.get_columns(FULL_TABLE_NAME)
            }
            needed_cols = {col.name for col in METADATA_COLS}
            if not needed_cols.issubset(existing_cols):
                with engine.begin() as conn:
                    conn.execute(
                        text(f'DROP TABLE IF EXISTS "{FULL_TABLE_NAME}" CASCADE')
                    )
    finally:
        engine.dispose()


def create_vector_store(request):
    # Make sure a table created by the old (buggy) code does not prevent the
    # external_id column from being added to the schema.
    _ensure_custom_columns_in_schema(request.tenant)

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=DB_CONFIG["host"],
        password=DB_CONFIG["password"],
        port=DB_CONFIG["port"],
        user=DB_CONFIG["user"],
        table_name=TABLE_NAME,
        embed_dim=1536,
        metadata_cols=METADATA_COLS,  # <-- external_id becomes a separate column
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
    )

    return vector_store


def run_agent(request):
    vector_store = create_vector_store(request)

    # external_id is still set in the Document metadata; PGVectorStore moves
    # it into the dedicated `external_id` column declared via metadata_cols.
    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )

    return index


def verify_external_id_column(tenant: str):
    """Verify that external_id exists as a real column and holds the values."""
    engine = _engine_for(tenant)
    try:
        with engine.connect() as conn:
            columns = conn.execute(
                text(
                    "SELECT column_name, data_type FROM information_schema.columns "
                    "WHERE table_name = :t ORDER BY ordinal_position"
                ),
                {"t": FULL_TABLE_NAME},
            ).fetchall()

            rows = conn.execute(
                text(
                    f'SELECT external_id, text, metadata FROM "{FULL_TABLE_NAME}" '
                    "ORDER BY external_id"
                )
            ).fetchall()
    finally:
        engine.dispose()
    return columns, rows


if __name__ == "__main__":
    class Request:
        tenant = "test_db"

    request = Request()
    index = run_agent(request)

    columns, rows = verify_external_id_column(request.tenant)

    print("Vector store created and documents indexed")
    print(f"Schema of {FULL_TABLE_NAME}:")
    for name, dtype in columns:
        print(f"  - {name} ({dtype})")

    assert any(name == "external_id" for name, _ in columns), (
        "external_id is missing as a separate column!"
    )

    print("\nexternal_id values stored in the dedicated column:")
    for external_id, doc_text, metadata in rows:
        print(
            f"  external_id={external_id!r} | text={doc_text!r} | "
            f"metadata={metadata!r}"
        )
