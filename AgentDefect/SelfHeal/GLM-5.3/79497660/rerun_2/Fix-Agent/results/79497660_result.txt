    Args:
        external_id_column: name of the SQL column added to the table.
        external_id_key: key inside `node.metadata` holding the value that is
            written to that column.
        keep_in_metadata: if True the value is ALSO kept inside the `metadata_`
            JSON column (default False -> the value lives in its own column
            instead of the metadata JSON).
        pg_connection_kwargs: explicit psycopg2 connection parameters used for
            the ALTER/UPDATE/SELECT statements; falls back to the store's own
            connection string when omitted.
    """

    def __init__(
        self,
        *args: Any,
        external_id_column: str = "external_id",
        external_id_key: str = "external_id",
        keep_in_metadata: bool = False,
        pg_connection_kwargs: Optional[Dict[str, Any]] = None,
        **kwargs: Any,
    ) -> None:
        # The parent constructor creates the engine/session and the data table.
        super().__init__(*args, **kwargs)

        self._external_id_column = external_id_column
        self._external_id_key = external_id_key
        self._keep_in_metadata = keep_in_metadata
        self._pg_connection_kwargs = pg_connection_kwargs or {}
        self._external_id_column_ready = False

        # Add the custom column directly to the PostgreSQL table schema.
        self._ensure_external_id_column()

    # ------------------------------------------------------------------ helpers

    @property
    def _table_schema(self) -> str:
        return getattr(self, "schema_name", None) or "public"

    @property
    def _table(self) -> str:
        return self.table_name

    def _pg_connect(self):
        """Open a raw psycopg2 connection to the store's database."""
        if self._pg_connection_kwargs:
            return psycopg2.connect(**self._pg_connection_kwargs)

        connection_string = getattr(self, "connection_string", None)
        if not connection_string:
            raise ValueError(
                "ExternalIdPGVectorStore needs either `pg_connection_kwargs` "
                "or a `connection_string` to talk to PostgreSQL directly."
            )
        return psycopg2.connect(connection_string)

    def _ensure_external_id_column(self) -> None:
        """ALTER TABLE ... ADD COLUMN IF NOT EXISTS external_id TEXT (+ index)."""
        if self._external_id_column_ready:
            return

        conn = self._pg_connect()
        try:
            with conn.cursor() as cur:
                # Add the custom column directly to the table schema.
                cur.execute(
                    pg_sql.SQL(
                        "ALTER TABLE {}.{} ADD COLUMN IF NOT EXISTS {} TEXT"
                    ).format(
                        pg_sql.Identifier(self._table_schema),
                        pg_sql.Identifier(self._table),
                        pg_sql.Identifier(self._external_id_column),
                    )
                )
                # Index it so the column can be used for fast lookups / joins.
                cur.execute(
                    pg_sql.SQL("CREATE INDEX IF NOT EXISTS {} ON {}.{} ({})").format(
                        pg_sql.Identifier(
                            f"{self._table}_{self._external_id_column}_idx"
                        ),
                        pg_sql.Identifier(self._table_schema),
                        pg_sql.Identifier(self._table),
                        pg_sql.Identifier(self._external_id_column),
                    )
                )
            conn.commit()
            self._external_id_column_ready = True
        except psycopg2.errors.UndefinedTable:
            # The data table has not been created yet (some versions create it
            # lazily); the column is then added just before the first write.
            conn.rollback()
        finally:
            conn.close()

    def _update_external_ids(self, external_ids: Dict[str, str]) -> None:
        """Write the external_id values into their dedicated column."""
        conn = self._pg_connect()
        try:
            with conn.cursor() as cur:
                stmt = pg_sql.SQL("UPDATE {}.{} SET {} = %s WHERE id = %s").format(
                    pg_sql.Identifier(self._table_schema),
                    pg_sql.Identifier(self._table),
                    pg_sql.Identifier(self._external_id_column),
                )
                for node_id, external_id in external_ids.items():
                    cur.execute(stmt, (external_id, node_id))
            conn.commit()
        except Exception:
            conn.rollback()
            raise
        finally:
            conn.close()

    def _fetch_external_ids(self, node_ids: List[str]) -> Dict[str, str]:
        """Read external_id values back from the dedicated column."""
        if not node_ids:
            return {}

        conn = self._pg_connect()
        try:
            with conn.cursor() as cur:
                cur.execute(
                    pg_sql.SQL("SELECT id, {} FROM {}.{} WHERE id = ANY(%s)").format(
                        pg_sql.Identifier(self._external_id_column),
                        pg_sql.Identifier(self._table_schema),
                        pg_sql.Identifier(self._table),
                    ),
                    (list(set(node_ids)),),
                )
                return {row[0]: row[1] for row in cur.fetchall() if row[1] is not None}
        finally:
            conn.close()

    def _attach_external_ids(self, nodes: List[BaseNode]) -> None:
        """Re-attach external_id (read from its own column) onto nodes."""
        if not nodes:
            return

        external_ids = self._fetch_external_ids([node.node_id for node in nodes])
        for node in nodes:
            external_id = external_ids.get(node.node_id)
            if external_id is None:
                continue
            if node.metadata is None:
                node.metadata = {}
            node.metadata[self._external_id_key] = external_id

    def get_node_ids_by_external_id(self, external_id: str) -> List[str]:
        """Look node ids up directly through the dedicated external_id column."""
        conn = self._pg_connect()
        try:
            with conn.cursor() as cur:
                cur.execute(
                    pg_sql.SQL("SELECT id FROM {}.{} WHERE {} = %s").format(
                        pg_sql.Identifier(self._table_schema),
                        pg_sql.Identifier(self._table),
                        pg_sql.Identifier(self._external_id_column),
                    ),
                    (external_id,),
                )
                return [row[0] for row in cur.fetchall()]
        finally:
            conn.close()

    # ---------------------------------------------------- VectorStore interface

    def add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        """Index nodes and store their external_id in its own table column."""
        # Collect every node's external_id and temporarily take it off the
        # metadata so the base class does NOT write it into the `metadata_`
        # JSON column.
        external_ids: Dict[str, str] = {}
        for node in nodes:
            external_id = getattr(node, "external_id", None)
            if external_id is None and node.metadata:
                external_id = node.metadata.get(self._external_id_key)
            if external_id is not None:
                external_ids[node.node_id] = str(external_id)
                if not self._keep_in_metadata and node.metadata:
                    node.metadata.pop(self._external_id_key, None)

        try:
            # Base implementation inserts the id / hash / text / node_info /
            # metadata_ / embedding columns.
            ids = super().add(nodes, **add_kwargs)
        finally:
            # Restore external_id on the in-memory nodes so retrievers,
            # docstores, ... keep working exactly as before.
            if not self._keep_in_metadata:
                for node in nodes:
                    if node.node_id in external_ids:
                        if node.metadata is None:
                            node.metadata = {}
                        node.metadata[self._external_id_key] = external_ids[
                            node.node_id
                        ]

        if external_ids:
            # Make sure the column exists (table may have been created lazily)
            # and fill it in.
            self._ensure_external_id_column()
            self._update_external_ids(external_ids)

        return ids

    async def async_add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        return self.add(nodes, **add_kwargs)

    def get(self, text_id: str, *args: Any, **kwargs: Any) -> BaseNode:
        node = super().get(text_id, *args, **kwargs)
        if node is not None:
            self._attach_external_ids([node])
        return node

    def get_nodes(
        self,
        node_ids: Optional[List[str]] = None,
        filters: Optional[MetadataFilters] = None,
    ) -> List[BaseNode]:
        nodes = super().get_nodes(node_ids, filters)
        if nodes:
            self._attach_external_ids(nodes)
        return nodes

    def query(
        self, query: VectorStoreQuery, **kwargs: Any
    ) -> VectorStoreQueryResult:
        result = super().query(query, **kwargs)
        if result is not None and result.nodes:
            self._attach_external_ids(result.nodes)
        return result

    async def async_query(
        self, query: VectorStoreQuery, **kwargs: Any
    ) -> VectorStoreQueryResult:
        return self.query(query, **kwargs)

                stmt = pg_sql.SQL("UPDATE {}.{} SET {} = %s WHERE id = %s").format(
                    pg_sql.Identifier(self._table_schema),
                    pg_sql.Identifier(self._table),
                    pg_sql.Identifier(self._external_id_column),
                )
                for node_id, external_id in external_ids.items():
                    cur.execute(stmt, (external_id, node_id))
            conn.commit()
        except Exception:
            conn.rollback()
            raise
        finally:
            conn.close()

    def _fetch_external_ids(self, node_ids: List[str]) -> Dict[str, str]:
        """Read external_id values back from the dedicated column."""
        if not node_ids:
            return {}

        conn = self._pg_connect()
        try:
            with conn.cursor() as cur:
                cur.execute(
                    pg_sql.SQL("SELECT id, {} FROM {}.{} WHERE id = ANY(%s)").format(
                        pg_sql.Identifier(self._external_id_column),
                        pg_sql.Identifier(self._table_schema),
                        pg_sql.Identifier(self._table),
                    ),
                    (list(set(node_ids)),),
                )
                return {row[0]: row[1] for row in cur.fetchall() if row[1] is not None}
        finally:
            conn.close()

    def _attach_external_ids(self, nodes: List[BaseNode]) -> None:
        """Re-attach external_id (read from its own column) onto nodes."""
        if not nodes:
            return

        external_ids = self._fetch_external_ids([node.node_id for node in nodes])
        for node in nodes:
            external_id = external_ids.get(node.node_id)
            if external_id is None:
                continue
            if node.metadata is None:
                node.metadata = {}
            node.metadata[self._external_id_key] = external_id

    def get_node_ids_by_external_id(self, external_id: str) -> List[str]:
        """Look node ids up directly through the dedicated external_id column."""
        conn = self._pg_connect()
        try:
            with conn.cursor() as cur:
                cur.execute(
                    pg_sql.SQL("SELECT id FROM {}.{} WHERE {} = %s").format(
                        pg_sql.Identifier(self._table_schema),
                        pg_sql.Identifier(self._table),
                        pg_sql.Identifier(self._external_id_column),
                    ),
                    (external_id,),
                )
                return [row[0] for row in cur.fetchall()]
        finally:
            conn.close()

    # ---------------------------------------------------- VectorStore interface

    def add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        """Index nodes and store their external_id in its own table column."""
        # Collect every node's external_id and temporarily take it off the
        # metadata so the base class does NOT write it into the `metadata_`
        # JSON column.
        external_ids: Dict[str, str] = {}
        for node in nodes:
            external_id = getattr(node, "external_id", None)
            if external_id is None and node.metadata:
                external_id = node.metadata.get(self._external_id_key)
            if external_id is not None:
                external_ids[node.node_id] = str(external_id)
                if not self._keep_in_metadata and node.metadata:
                    node.metadata.pop(self._external_id_key, None)

        try:
            # Base implementation inserts the id / hash / text / node_info /
            # metadata_ / embedding columns.
            ids = super().add(nodes, **add_kwargs)
        finally:
            # Restore external_id on the in-memory nodes so retrievers,
            # docstores, ... keep working exactly as before.
            if not self._keep_in_metadata:
                for node in nodes:
                    if node.node_id in external_ids:
                        if node.metadata is None:
                            node.metadata = {}
                        node.metadata[self._external_id_key] = external_ids[
                            node.node_id
                        ]

        if external_ids:
            # Make sure the column exists (table may have been created lazily)
            # and fill it in.
            self._ensure_external_id_column()
            self._update_external_ids(external_ids)

        return ids

    async def async_add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        return self.add(nodes, **add_kwargs)

    def get(self, text_id: str, *args: Any, **kwargs: Any) -> BaseNode:
        node = super().get(text_id, *args, **kwargs)
        if node is not None:
            self._attach_external_ids([node])
        return node

    def get_nodes(
        self,
        node_ids: Optional[List[str]] = None,
        filters: Optional[MetadataFilters] = None,
    ) -> List[BaseNode]:
        nodes = super().get_nodes(node_ids, filters)
        if nodes:
            self._attach_external_ids(nodes)
        return nodes

    def query(
        self, query: VectorStoreQuery, **kwargs: Any
    ) -> VectorStoreQueryResult:
        result = super().query(query, **kwargs)
        if result is not None and result.nodes:
            self._attach_external_ids(result.nodes)
        return result

    async def async_query(
        self, query: VectorStoreQuery, **kwargs: Any
    ) -> VectorStoreQueryResult:
        return self.query(query, **kwargs)


def create_vector_store(request) -> ExternalIdPGVectorStore:
    """Create a vector store whose table has a dedicated external_id column."""
    connection_string = (
        f"postgresql://{DB_USER}:{DB_PASSWORD}@{DB_HOST}:{DB_PORT}/{request.tenant}"
    )

    vector_store = ExternalIdPGVectorStore(
        connection_string=connection_string,
        table_name=TABLE_NAME,
        embed_dim=1536,
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
        # custom-column handling
        external_id_column="external_id",
        external_id_key="external_id",
        keep_in_metadata=False,  # store it in the column *instead of* the JSON
        pg_connection_kwargs={
            "host": DB_HOST,
            "port": DB_PORT,
            "user": DB_USER,
            "password": DB_PASSWORD,
            "dbname": request.tenant,
        },
    )

    return vector_store


def run_agent(request):
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )

    return index


def verify_external_id_column(request) -> None:
    """Print the table schema and rows, proving external_id is a real column."""
    conn = psycopg2.connect(
        host=DB_HOST,
        port=DB_PORT,
        user=DB_USER,
        password=DB_PASSWORD,
        dbname=request.tenant,
    )
    try:
        with conn.cursor() as cur:
            # 1) The table schema now contains external_id as a real column.
            cur.execute(
                """
                SELECT column_name, data_type
                FROM information_schema.columns
                WHERE table_schema = %s AND table_name = %s
                ORDER BY ordinal_position
                """,
                (SCHEMA_NAME, TABLE_NAME),
            )
            columns = cur.fetchall()

            print(f"Table schema of {SCHEMA_NAME}.{TABLE_NAME}:")
            for name, dtype in columns:
                marker = "   <-- custom column (fixed)" if name == "external_id" else ""
                print(f"  - {name} ({dtype}){marker}")

            has_external_id_column = any(name == "external_id" for name, _ in columns)
            print(
                "\nexternal_id is a separate column of the table schema: "
                f"{has_external_id_column}"
            )

            # 2) The rows carry external_id in the column, not in the JSON.
            cur.execute(
                pg_sql.SQL(
                    "SELECT id, external_id, "
                    "metadata_->>'external_id' AS external_id_in_metadata_json "
                    "FROM {}.{} ORDER BY external_id"
                ).format(
                    pg_sql.Identifier(SCHEMA_NAME),
                    pg_sql.Identifier(TABLE_NAME),
                )
            )
            print("\nRows (id | external_id | external_id inside metadata_ JSON):")
            for row_id, external_id, in_json in cur.fetchall():
                print(f"  {row_id} | {external_id} | {in_json!r}")
    finally:
        conn.close()
