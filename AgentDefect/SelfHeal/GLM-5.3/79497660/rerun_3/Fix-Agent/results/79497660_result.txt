"""

Original problem (output mismatch):
    The script indexed documents and reported success, but `external_id` only
    ended up serialized inside the `metadata_` JSON(B) column of the
    `vectorstore` table.  The user expected `external_id` to be a real column
    of the PostgreSQL table schema.

Why it happens:
    `PGVectorStore` builds its SQLAlchemy table model *dynamically* inside
    `_create_table_class()` (standard columns: id, node_id, source_node_id,
    node_content, embedding, metadata_).  `from_params()` offers no parameter
    for custom columns, so nothing in the original code could ever create one.

How this fix works:
    1. `ExternalIdPGVectorStore` subclasses `PGVectorStore` and overrides
       `_create_table_class()`: it takes the model produced by the parent and
       appends an `external_id` column to it, so `CREATE TABLE` emits the
       extra column and Core INSERT statements can populate it.
    2. `add()` is overridden so that every inserted row writes
       `node.metadata["external_id"]` into the dedicated `external_id` column.
       The value is still kept inside `metadata_` so LlamaIndex retrieval and
       metadata filters keep working, but it now also exists as a first-class,
       indexable, SQL-queryable column.
    3. `CREATE TABLE` only runs when the table does not exist yet, so
       `ensure_external_id_column()` patches a table created by an earlier
       (buggy) run with `ALTER TABLE ... ADD COLUMN external_id TEXT`.
    4. psycopg2 is used to inspect `information_schema.columns` and the stored
       rows, verifying that the output now matches the expectation:
       `external_id` exists as a separate column and contains ext_001/ext_002.

class ExternalIdPGVectorStore(PGVectorStore):
    """A PGVectorStore whose table has a dedicated ``external_id`` column."""

    def _create_table_class(self) -> type:
        """Build the standard PGVectorStore table model, then attach the custom column.

        PGVectorStore generates the SQLAlchemy model for ``table_name``
        dynamically (which is why custom columns cannot simply be passed to
        ``from_params``), so we extend the generated model here.
        """
        model = super()._create_table_class()

        if not hasattr(model, EXTERNAL_ID_COLUMN):
            external_id_column = Column(EXTERNAL_ID_COLUMN, Text, nullable=True)

            # (1) register the column on the physical table -> it is emitted by
            #     CREATE TABLE and usable in the Core INSERT/SELECT below.
            model.__table__.append_column(external_id_column)

            # (2) register it on the ORM mapper so ``Model.external_id``
            #     attribute access works as well.
            try:
                class_mapper(model, configure=False).add_property(
                    EXTERNAL_ID_COLUMN, external_id_column
                )
            except Exception:
                # Core inserts/queries still work through the table itself.
                setattr(model, EXTERNAL_ID_COLUMN, external_id_column)

        return model

    def add(self, nodes: List[BaseNode], **add_kwargs: Any) -> List[str]:
        """Insert nodes, writing ``metadata['external_id']`` into its own column."""
        chunk_size = getattr(self, "insert_chunk_size", 500) or 500
        ids: List[str] = []

        with self._session() as session:
            try:
                for start in range(0, len(nodes), chunk_size):
                    chunk = nodes[start : start + chunk_size]
                    rows = []
                    for node in chunk:
                        embedding = node.get_embedding()
                        if embedding is not None:
                            try:
                                embedding = np.array(embedding)
                            except Exception:
                                pass  # pgvectormaybe also accepts plain lists

                        rows.append(
                            {
                                "id": node.id,
                                "node_id": node.node_id,
                                "source_node_id": node.source_node_id,
                                "embedding": embedding,
                                "metadata_": node.metadata,
                                "node_content": node.json(),
                                # >>> the actual fix: dedicated column instead
                                #     of only the metadata JSON blob <<<
                                EXTERNAL_ID_COLUMN: (node.metadata or {}).get(
                                    EXTERNAL_ID_COLUMN
                                ),
                            }
                        )

                    stmt = postgres_insert(self._table_class).values(rows)
                    session.execute(stmt)
                    session.commit()
                    ids.extend([node.node_id for node in chunk])
            except Exception:
                session.rollback()
                raise

        return ids

    def get_node_ids_by_external_id(self, external_id: str) -> List[str]:
        """Query the new column directly with SQL."""
        table = self._table_class.__table__
        with self._session() as session:
            result = session.execute(
                select(table.c.node_id).where(
                    table.c[EXTERNAL_ID_COLUMN] == external_id
                )
            )
            return [row[0] for row in result.fetchall()]

In every version of `llama-index-vector-stores-postgres` that supports `hnsw_kwargs` (which this code uses), the method is documented "to allow for easy overloading" with the signature:

The fix overrides it as `def _create_table_class(self) -> type` (zero parameters) and calls `super()._create_table_class()` with no arguments.
