from llama_index.core import VectorStoreIndex, Document, StorageContext
from llama_index.core.vector_stores import MetadataFilters, ExactMatchFilter
from llama_index.vector_stores.postgres import PGVectorStore
from sqlalchemy import Column, String
import psycopg2
import os
from dotenv import load_dotenv

load_dotenv()

# Connection configuration (env vars with fallbacks to the original hard-coded values)
DB_CONFIG = {
    "host": os.getenv("PG_HOST", "localhost"),
    "password": os.getenv("PG_PASSWORD", "password"),
    "port": int(os.getenv("PG_PORT", "5432")),
    "user": os.getenv("PG_USER", "postgres"),
}


def create_vector_store(request):
    """
    Create a PGVectorStore that stores `external_id` as a REAL column in the
    PostgreSQL table instead of only inside the metadata JSON(B) blob.

    FIX: PGVectorStore.from_params() accepts `metadata_cols`, a list of
    SQLAlchemy Column objects. Each column declared there is created in the
    underlying table, and values are pulled from
    Document.metadata[col_name] on insert. The buggy code never passed
    `metadata_cols`, so `external_id` could only land in the metadata JSON.
    """
    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=DB_CONFIG["host"],
        password=DB_CONFIG["password"],
        port=DB_CONFIG["port"],
        user=DB_CONFIG["user"],
        table_name="vectorstore",
        embed_dim=1536,
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
        # FIX: declare custom table columns.
        # This makes `external_id` a first-class column of the
        # "vectorstore" table (created automatically if missing, and
        # ALTER TABLEd into an existing table in recent versions).
        metadata_cols=[
            Column("external_id", String, nullable=True),
        ],
    )

    return vector_store


def run_agent(request):
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    # FIX: the documented way to attach a vector store when building an index
    # from documents is through a StorageContext (passing `vector_store=`
    # directly is deprecated/unsupported in several LlamaIndex versions).
    storage_context = StorageContext.from_defaults(vector_store=vector_store)

    index = VectorStoreIndex.from_documents(
        documents,
        storage_context=storage_context,
        show_progress=False,
    )

    return index


def drop_stale_table(request):
    """
    If the table was created by the buggy code (without the `external_id`
    column) and the installed store version does not auto-migrate existing
    tables, recreate it so the new schema is guaranteed.
    Only needed once when switching an existing table to metadata_cols.
    """
    conn = psycopg2.connect(
        dbname=request.tenant,
        user=DB_CONFIG["user"],
        password=DB_CONFIG["password"],
        host=DB_CONFIG["host"],
        port=DB_CONFIG["port"],
    )
    try:
        with conn.cursor() as cur:
            cur.execute("DROP TABLE IF EXISTS vectorstore")
        conn.commit()
    finally:
        conn.close()


def verify_table_schema(request):
    """Prove that `external_id` is now a real column of the table."""
    conn = psycopg2.connect(
        dbname=request.tenant,
        user=DB_CONFIG["user"],
        password=DB_CONFIG["password"],
        host=DB_CONFIG["host"],
        port=DB_CONFIG["port"],
    )
    try:
        with conn.cursor() as cur:
            cur.execute(
                """
                SELECT column_name, data_type
                FROM information_schema.columns
                WHERE table_name = 'vectorstore'
                ORDER BY ordinal_position;
                """
            )
            columns = cur.fetchall()

            cur.execute(
                "SELECT external_id, left(text, 40) FROM vectorstore ORDER BY external_id;"
            )
            rows = cur.fetchall()
    finally:
        conn.close()

    print("Columns in the 'vectorstore' table:")
    for name, dtype in columns:
        print(f"  - {name} ({dtype})")

    assert any(name == "external_id" for name, _ in columns), (
        "'external_id' is NOT a real column in the table!"
    )

    print("\nOK: 'external_id' now exists as a dedicated column (not just metadata JSON).")
    print("Sample rows:")
    for external_id, text in rows:
        print(f"  external_id={external_id!r} | text={text!r}")


def query_by_external_id(request, external_id="ext_001", query_str="Sample document"):
    """
    Retrieve nodes filtered on the dedicated `external_id` column using
    LlamaIndex metadata filters (the store maps the filter key onto the
    custom column).
    """
    vector_store = create_vector_store(request)
    index = VectorStoreIndex.from_vector_store(vector_store)

    retriever = index.as_retriever(
        similarity_top_k=5,
        filters=MetadataFilters(
            filters=[ExactMatchFilter(key="external_id", value=external_id)]
        ),
    )
    nodes = retriever.retrieve(query_str)

    print(f"\nRetrieved {len(nodes)} node(s) filtered on external_id={external_id!r}:")
    for n in nodes:
        print(f"  - {n.get_content()!r} (external_id={n.metadata.get('external_id')!r})")
    return nodes
