from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv


def run_agent(query: str):
    """Run a simple retrieval example that demonstrates how to filter on a
    ``null``‑like metadata value.

    The original problem stemmed from using ``None`` as the metadata value.
    Most vector‑store back‑ends (including the default in‑memory store used by
    ``VectorStoreIndex``) drop ``None`` values during serialization – the key is
    simply omitted. Consequently a filter that checks ``start_date == None`` never
    matches because the stored document does not contain the ``start_date`` key.

    The workaround is to store a sentinel string (e.g. ``"<NULL>"``) instead of
    ``None`` and filter on that sentinel. This keeps the metadata explicit and
    works with any back‑end.
    """
    # Load environment variables (e.g. OPENAI_API_KEY) from a .env file.
    load_dotenv()

    # Configure the embedding model for the index.
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Use a sentinel value for "null" metadata – a plain string works with all
    # vector stores because it survives JSON / pickle round‑tripping.
    NULL_SENTINEL = "<NULL>"

    # Create a node whose metadata deliberately contains the sentinel.
    node_01 = TextNode(
        text="This document has a NULL sentinel in the metadata",
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},
    )

    # Build the index from the node.
    doc_index = VectorStoreIndex([node_01])

    # Debug output – show exactly what is stored inside the docstore.
    print("Index nodes metadata:")
    for stored_node in doc_index.docstore.docs.values():
        print(stored_node.metadata)
    print()

    # Construct a filter that looks for the sentinel string.
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])

    # Retrieve with the filter applied.
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show retrieval results.
    print("Retrieved nodes:")
    for node in nodes:
        print(f"Node ID: {node.node_id}, metadata: {node.metadata}")
    print()

    return nodes

    """
