    """Replace ``None`` values with a sentinel that can be stored and filtered.

    whose values are ``None`` because they are not JSON‑serialisable.  By mapping
    ``None``→``NULL_SENTINEL`` we keep the information and can later query it
    using a normal equality filter.
    """
    return {k: (v if v is not None else NULL_SENTINEL) for k, v in metadata.items()}

# NOTE:
# LlamaIndex metadata filters do not support ``None`` as a filter value.
# Most vector‑store back‑ends (Chroma, Pinecone, etc.) treat a missing field
# as the absence of a key rather than a stored ``null`` value.  Consequently
# a filter like ``key == None`` never matches.
#
# A common workaround is to normalise ``None`` values to a sentinel string
# (e.g. "__NULL__") when indexing and then filter on that sentinel.
# The code below demonstrates this pattern.

# Sentinel used to represent a ``None``/null value in metadata.
NULL_SENTINEL = "__NULL__"


def run_agent(query: str):

    (represented by ``NULL_SENTINEL``) and retrieve it using a metadata filter.

    Args:
        query: The text query used for similarity search.
    """
    # Load environment variables (e.g. ``OPENAI_API_KEY``)
    load_dotenv()

    # Configure the embedding model – this will read the key from the env.
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # ---------------------------------------------------------------------
    # 1️⃣ Create a node.  If the logical value of ``start_date`` is ``None``
    #    we store the sentinel instead so that the underlying vector store has
    #    an explicit value to filter on.
    # ---------------------------------------------------------------------

        text="This document has None (represented by a sentinel) in the metadata",
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},

    # Build the index from the single node.
    doc_index = VectorStoreIndex([node_01])

    # Debug: show what is actually persisted in the docstore.
    print("Index nodes metadata:")
    for node in doc_index.docstore.docs.values():
        print(node.metadata)
    print()

    # ---------------------------------------------------------------------
    # 2️⃣ Create a metadata filter that looks for the sentinel value.
    # ---------------------------------------------------------------------
    filter_null_start_date = MetadataFilter(
        key="start_date",

        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])

    # Retrieve the most similar node that also satisfies the metadata filter.

    print()

    return retrieved_nodes
