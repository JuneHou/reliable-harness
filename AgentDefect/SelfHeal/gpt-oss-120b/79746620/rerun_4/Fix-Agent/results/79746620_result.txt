"""Demonstration of how to filter on "null" (missing) metadata values in LlamaIndex.

LlamaIndex does not currently support filtering for a literal ``None`` value.
During indexing many vector‑store back‑ends drop keys whose values are ``None``
(e.g. they are omitted from the stored metadata JSON).  Consequently a
``MetadataFilter`` with ``value=None`` will never match because the key is not
present in the persisted metadata.

A common workaround is to store a sentinel value (e.g. the string "__NULL__")
instead of ``None`` and use that sentinel in the filter.  The example below
creates a single node with a ``start_date`` field set to ``"__NULL__"`` and
retrieves it via a filter.

If you later need a real ``None`` in your Python objects you can translate the
sentinel back to ``None`` after retrieval.
"""

from __future__ import annotations

import os
from dotenv import load_dotenv

from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding

# ---------------------------------------------------------------------------
# Helper: convert a sentinel back to ``None`` after retrieval (optional).
# ---------------------------------------------------------------------------
NULL_SENTINEL = "__NULL__"

def _decode_null(value: str | None) -> None | str:
    """Replace the sentinel with ``None``.

    The vector store always returns strings, so we only need to handle the
    sentinel case.
    """
    return None if value == NULL_SENTINEL else value

# ---------------------------------------------------------------------------
# Main demo function
# ---------------------------------------------------------------------------
def run_demo(query: str) -> list[TextNode]:
    # Load environment variables (e.g. ``OPENAI_API_KEY``).
    load_dotenv()

    # Configure the embedding model that the index will use.
    # (Any embedding model works – the focus here is metadata filtering.)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # ---------------------------------------------------------------------
    # Create a node where ``start_date`` is "null".  We store a sentinel string
    # because ``None`` values are stripped by the underlying vector store.
    # ---------------------------------------------------------------------
    node = TextNode(
        text="This document has a null start_date in its metadata.",
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},
    )

    # Build the index from the single node.
    index = VectorStoreIndex([node])

    # Debug: show what actually landed in the docstore.
    print("--- Stored metadata ---")
    for doc_id, stored_node in index.docstore.docs.items():
        print(f"{doc_id}: {stored_node.metadata}")
    print()

    # ---------------------------------------------------------------------
    # Construct a filter that looks for the sentinel value.
    # ``FilterOperator.EQ`` works with normal strings.
    # ---------------------------------------------------------------------
    null_filter = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[null_filter])

    # Retrieve with the filter applied.
    retriever = index.as_retriever(filters=filters, similarity_top_k=1)
    retrieved = retriever.retrieve(query)

    # Optional: decode sentinel back to ``None`` for downstream code.
    for node in retrieved:
        if "start_date" in node.metadata:
            node.metadata["start_date"] = _decode_null(node.metadata["start_date"])

    # Show results.
    print("--- Retrieved nodes ---")
    for n in retrieved:
        print(f"ID: {n.node_id}, metadata: {n.metadata}")
    print()

    return retrieved

if __name__ == "__main__":
