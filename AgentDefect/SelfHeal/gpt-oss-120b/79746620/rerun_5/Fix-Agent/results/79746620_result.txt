from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv


def run_agent(query: str):
    """Run a simple example demonstrating how to filter on a *null*‑like metadata value.

    LlamaIndex does not support filtering directly on Python ``None`` because the
    underlying vector store (e.g., Chroma, Pinecone, etc.) typically expects JSON‑
    serialisable values.  ``None`` is dropped or converted to ``null`` in JSON,
    which breaks equality filters.  The common workaround is to store a sentinel
    string (e.g., "__NULL__") and filter on that sentinel.
    """
    # Load environment variables from .env file (e.g., OPENAI_API_KEY)
    load_dotenv()

    # Set up the embedding model – the OpenAI key is read from the environment.
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # ---------------------------------------------------------------------
    # 1️⃣  Create a node where ``start_date`` is *effectively* null.
    # ---------------------------------------------------------------------
    NULL_SENTINEL = "__NULL__"  # any unique string that will not appear in real data
    node_01 = TextNode(
        text="This document has a null‑like value in the metadata",
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},
    )

    # ---------------------------------------------------------------------
    # 2️⃣  Build the index.
    # ---------------------------------------------------------------------
    doc_index = VectorStoreIndex([node_01])

    # Debug: show what is actually stored in the docstore.
    print("Index nodes metadata:")
    for node in doc_index.docstore.docs.values():
        print(node.metadata)
    print()

    # ---------------------------------------------------------------------
    # 3️⃣  Filter for the sentinel value – this mimics "start_date IS NULL".
    # ---------------------------------------------------------------------
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show retrieval results.
    print("Retrieved nodes:")
    for node in nodes:
        # Convert the sentinel back to ``None`` for a cleaner user view.
        display_meta = {
            k: (None if v == NULL_SENTINEL else v) for k, v in node.metadata.items()
        }
        print(f"{node.node_id}: {display_meta}")
    print()

    return nodes


if __name__ == "__main__":

# many vector‑store back‑ends drop ``None`` (or treat it as missing) when persisting
