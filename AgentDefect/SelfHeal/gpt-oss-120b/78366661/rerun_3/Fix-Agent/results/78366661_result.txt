from dotenv import load_dotenv
import os
from pymongo import MongoClient

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import MongoDBAtlasVectorSearch


def run_agent(pdf_url: str):
    """Load a PDF, split it into chunks, store the chunks in MongoDB Atlas
    vector search, and perform a similarity query.

    The function expects two environment variables:
        * ``OPENAI_API_KEY`` – API key for the OpenAI embeddings model.
        * ``MONGODB_ATLAS_URI`` – Connection string for a MongoDB Atlas cluster.

    The Atlas cluster must have a collection with a *vector* search index
    (named ``vector_index`` by default). The index should be configured to store
    embeddings generated by the same model used here (OpenAI embeddings).
    """

    # ------------------------------------------------------------
    # Load secrets
    # ------------------------------------------------------------
    load_dotenv()
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment")

    mongo_uri = os.getenv("MONGODB_ATLAS_URI")
    if not mongo_uri:
        raise ValueError("MONGODB_ATLAS_URI not found in environment")

    # ------------------------------------------------------------
    # Initialise MongoDB client and obtain the collection that will store
    # the documents.  Replace ``my_documents`` with the name of your collection.
    # ------------------------------------------------------------
    client = MongoClient(mongo_uri)
    db = client.get_default_database()
    collection = db["my_documents"]  # <- change to your collection name

    # ------------------------------------------------------------
    # Load and chunk the PDF
    # ------------------------------------------------------------
    loader = PyPDFLoader(pdf_url)
    raw_documents = loader.load()

    splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = splitter.split_documents(raw_documents)

    # ------------------------------------------------------------
    # Create (or retrieve) the Atlas vector store
    # ------------------------------------------------------------
    vector_store = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(disallowed_special=()),
        collection=collection,
        index_name="vector_index",  # must match the index you created in Atlas
    )

    # ------------------------------------------------------------
    # Perform a similarity search
    # ------------------------------------------------------------
    query = "What were the compute requirements for training GPT 4"
    # ``k`` is the number of nearest neighbours to return; adjust as needed.
    results = vector_store.similarity_search(query, k=4)

    return results


if __name__ == "__main__":
    # Example PDF from arXiv; you can replace it with any publicly accessible PDF.
    pdf_url = "https://arxiv.org/pdf/2303.08774.pdf"
    found_docs = run_agent(pdf_url)

    for i, doc in enumerate(found_docs, start=1):
        print(f"--- Result {i} ---")
        # Show a snippet of the retrieved page content
        print(doc.page_content[:500])
        print()
