import os
from dotenv import load_dotenv

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import MongoDBAtlasVectorSearch
import pymongo


def run_agent(pdf_url: str):
    """Load a PDF, split it into chunks, store those chunks in an Atlas Vector
    Search collection and perform a similarity search.

    The function expects the following environment variables to be set:

    * ``OPENAI_API_KEY`` – OpenAI API key for the embedding model.
    * ``MONGODB_ATLAS_URI`` – Full MongoDB Atlas connection string (e.g.
      ``mongodb+srv://<user>:<pwd>@cluster0.xxx.mongodb.net``).
    * ``MONGODB_DATABASE`` – Name of the database that contains the collection.
    * ``MONGODB_COLLECTION`` – Name of the collection that will hold the vectors.
    * ``ATLAS_VECTOR_INDEX`` – Name of the Atlas Search index that is
      configured for vector search (default ``vector_index``).
    """
    load_dotenv()

    # ---------------------------------------------------------------------
    # Verify required environment variables
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment")

    mongo_uri = os.getenv("MONGODB_ATLAS_URI")
    if not mongo_uri:
        raise ValueError("MONGODB_ATLAS_URI not found in environment")

    db_name = os.getenv("MONGODB_DATABASE")
    if not db_name:
        raise ValueError("MONGODB_DATABASE not found in environment")

    coll_name = os.getenv("MONGODB_COLLECTION")
    if not coll_name:
        raise ValueError("MONGODB_COLLECTION not found in environment")

    index_name = os.getenv("ATLAS_VECTOR_INDEX", "vector_index")

    # ---------------------------------------------------------------------
    # Load and chunk the PDF
    # ---------------------------------------------------------------------
    loader = PyPDFLoader(pdf_url)
    raw_documents = loader.load()

    splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = splitter.split_documents(raw_documents)

    # ---------------------------------------------------------------------
    # Create a MongoDB client and get the collection object
    # ---------------------------------------------------------------------
    client = pymongo.MongoClient(mongo_uri)
    db = client[db_name]
    collection = db[coll_name]

    # ---------------------------------------------------------------------
    # Create (or retrieve) the Atlas Vector Search index via LangChain
    # ---------------------------------------------------------------------
    vector_store = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(disallowed_special=()),
        collection=collection,
        index_name=index_name,
    )

    # ---------------------------------------------------------------------
    # Perform a similarity search
    # ---------------------------------------------------------------------
    query = "What were the compute requirements for training GPT 4"
    results = vector_store.similarity_search(query)

    # Clean up the client – not strictly required but good practice
    client.close()

    return results


if __name__ == "__main__":
    # Example PDF – replace with any accessible URL or local file path

            print(f"{i}. {snippet}\n")
    except Exception as e:
