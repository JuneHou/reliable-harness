from dotenv import load_dotenv
import os

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import MongoDBAtlasVectorSearch
from pymongo import MongoClient


def run_agent(pdf_url: str):
    """Load a PDF, split it into chunks, index the chunks in MongoDB Atlas
    vector search and perform a similarity search.

    Args:
        pdf_url: Direct link to a PDF document.

    Returns:
        A list of ``Document`` objects matching the query.
    """
    # Load environment variables (API keys, MongoDB connection string, etc.)
    load_dotenv()

    # ---------------------------------------------------------------------
    # Validate required secrets
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment")

    mongodb_uri = os.getenv("MONGODB_ATLAS_URI")
    if not mongodb_uri:
        raise ValueError("MONGODB_ATLAS_URI not found in environment")

    # ---------------------------------------------------------------------
    # Initialise MongoDB client and get a collection reference
    # ---------------------------------------------------------------------
    client = MongoClient(mongodb_uri)
    # The database and collection names can also be supplied via env vars –
    # here we use defaults that the user can change.
    db_name = os.getenv("MONGODB_DB_NAME", "langchain")
    collection_name = os.getenv("MONGODB_COLLECTION_NAME", "documents")
    db = client[db_name]
    collection = db[collection_name]

    # ---------------------------------------------------------------------
    # Load and chunk the PDF
    # ---------------------------------------------------------------------
    loader = PyPDFLoader(pdf_url)
    raw_documents = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = text_splitter.split_documents(raw_documents)

    # ---------------------------------------------------------------------
    # Create the vector store (this will upsert the embeddings into the
    # MongoDB Atlas collection).  The ``index_name`` must match the name of a
    # vector search index that exists in the Atlas UI.
    # ---------------------------------------------------------------------
    embedding = OpenAIEmbeddings(disallowed_special=())
    vector_search_index = os.getenv("ATLAS_VECTOR_SEARCH_INDEX", "vector_index")

    vector_store = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=embedding,
        collection=collection,
        index_name=vector_search_index,
    )

    # ---------------------------------------------------------------------
    # Perform a similarity search
    # ---------------------------------------------------------------------
    query = "What were the compute requirements for training GPT 4"
    results = vector_store.similarity_search(query)

    return results


if __name__ == "__main__":

    url = "https://arxiv.org/pdf/2303.08774.pdf"
