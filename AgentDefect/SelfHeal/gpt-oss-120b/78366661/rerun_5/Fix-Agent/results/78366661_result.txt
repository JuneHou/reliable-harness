import os
from dotenv import load_dotenv

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import MongoDBAtlasVectorSearch
from pymongo import MongoClient


def run_agent(pdf_url: str):
    """Load a PDF, split it into chunks and perform a similarity search using
    MongoDB Atlas Vector Search.

    The function expects the following environment variables to be set:
        * ``OPENAI_API_KEY`` – API key for OpenAI embeddings.
        * ``MONGODB_URI`` – Full connection string to the Atlas cluster.
        * ``MONGODB_DATABASE`` – Name of the database that holds the collection.
        * ``MONGODB_COLLECTION`` – Name of the collection that will store the
          vectorised documents.
        * ``ATLAS_VECTOR_SEARCH_INDEX`` – Name of the Atlas Vector Search index
          (must already exist in the collection).
    """
    load_dotenv()

    # ---------------------------------------------------------------------
    # Validate required environment variables
    # ---------------------------------------------------------------------
    required_vars = [
        "OPENAI_API_KEY",
        "MONGODB_URI",
        "MONGODB_DATABASE",
        "MONGODB_COLLECTION",
        "ATLAS_VECTOR_SEARCH_INDEX",
    ]
    missing = [var for var in required_vars if os.getenv(var) is None]
    if missing:
        raise ValueError(f"Missing required environment variables: {', '.join(missing)}")

    # ---------------------------------------------------------------------
    # Initialise the PDF loader and split the document into manageable chunks
    # ---------------------------------------------------------------------
    loader = PyPDFLoader(pdf_url)
    raw_docs = loader.load()
    splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = splitter.split_documents(raw_docs)

    # ---------------------------------------------------------------------
    # Set up the MongoDB Atlas Vector Search store
    # ---------------------------------------------------------------------
    mongodb_uri = os.getenv("MONGODB_URI")
    database_name = os.getenv("MONGODB_DATABASE")
    collection_name = os.getenv("MONGODB_COLLECTION")
    index_name = os.getenv("ATLAS_VECTOR_SEARCH_INDEX")

    client = MongoClient(mongodb_uri)
    collection = client[database_name][collection_name]

    # Create (or reuse) the vector store – this will upsert the documents into the
    # collection and make them searchable via the specified Atlas Vector Search index.
    vector_store = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(),
        collection=collection,
        index_name=index_name,
    )

    # ---------------------------------------------------------------------
    # Perform a similarity search
    # ---------------------------------------------------------------------
    query = "What were the compute requirements for training GPT 4"
    results = vector_store.similarity_search(query)
    return results
