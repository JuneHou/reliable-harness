import os
from dotenv import load_dotenv

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import MongoDBAtlasVectorSearch
from pymongo import MongoClient


def run_agent(pdf_url: str):
    """Load a PDF, split it into chunks, store the chunks in a MongoDB Atlas
    vector search collection and perform a similarity search.

    Environment variables expected:
        - OPENAI_API_KEY: OpenAI API key for embeddings
        - MONGODB_URI: Full MongoDB Atlas connection string (including user/password)
        - MONGODB_DB_NAME: Name of the database to use
        - MONGODB_COLLECTION_NAME: Name of the collection that will hold the vectors
        - MONGODB_VECTOR_INDEX_NAME (optional): Name of the Atlas vector index
    """
    load_dotenv()

    # ---------------------------------------------------------------------
    # Validate required environment variables
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment")

    mongodb_uri = os.getenv("MONGODB_URI")
    if not mongodb_uri:
        raise ValueError("MONGODB_URI not found in environment")

    db_name = os.getenv("MONGODB_DB_NAME", "vector_search_db")
    collection_name = os.getenv("MONGODB_COLLECTION_NAME", "documents")
    vector_index_name = os.getenv("MONGODB_VECTOR_INDEX_NAME", "vector_index")

    # ---------------------------------------------------------------------
    # Initialise the MongoDB client and obtain a collection handle
    # ---------------------------------------------------------------------
    client = MongoClient(mongodb_uri)
    db = client[db_name]
    collection = db[collection_name]

    # ---------------------------------------------------------------------
    # Load and split the PDF into manageable chunks for embedding
    # ---------------------------------------------------------------------
    loader = PyPDFLoader(pdf_url)
    raw_docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = text_splitter.split_documents(raw_docs)

    # ---------------------------------------------------------------------
    # Create (or load) the Atlas vector store from the documents
    # ---------------------------------------------------------------------
    embeddings = OpenAIEmbeddings(disallowed_special=())

    vector_store = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=embeddings,
        collection=collection,
        index_name=vector_index_name,
    )

    # ---------------------------------------------------------------------
    # Perform a similarity search
    # ---------------------------------------------------------------------
    query = "What were the compute requirements for training GPT 4"
    results = vector_store.similarity_search(query)

    return results

from pymongo import MongoClient


def get_atlas_collection() -> "pymongo.collection.Collection":
    """Create a pymongo collection object for the Atlas DB.

    The function expects the following environment variables to be set:
        * ATLAS_URI – the full MongoDB connection string (including username/password)
        * ATLAS_DB_NAME – name of the database that holds the vector collection
        * ATLAS_COLLECTION_NAME – name of the collection that will store the vectors
    """
    atlas_uri = os.getenv("ATLAS_URI")
    db_name = os.getenv("ATLAS_DB_NAME")
    coll_name = os.getenv("ATLAS_COLLECTION_NAME")

    if not all([atlas_uri, db_name, coll_name]):
        raise ValueError(
            "Missing one of ATLAS_URI, ATLAS_DB_NAME, ATLAS_COLLECTION_NAME environment variables"
        )

    client = MongoClient(atlas_uri)
    db = client[db_name]
    return db[coll_name]


def run_agent(pdf_url: str):
    """Load a PDF, split it into chunks, embed the chunks, store them in
    MongoDB Atlas Vector Search and finally perform a similarity search.

    Returns a list of ``Document`` objects that are most similar to the query.
    """
    # Load .env variables (API keys, Atlas connection string, …)
    load_dotenv()

    # Validate OpenAI key – required by ``OpenAIEmbeddings``

        raise ValueError("OPENAI_API_KEY not found in environment variables")

    # ---------------------------------------------------------------------
    # 1️⃣ Load the PDF and split it into manageable chunks
    # ---------------------------------------------------------------------
    loader = PyPDFLoader(pdf_url)
    raw_documents = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = text_splitter.split_documents(raw_documents)

    # ---------------------------------------------------------------------
    # 2️⃣ Create (or get) the Atlas collection that will store the vectors
    # ---------------------------------------------------------------------
    collection = get_atlas_collection()

    # ---------------------------------------------------------------------
    # 3️⃣ Embed the documents and push them to Atlas Vector Search

    embeddings = OpenAIEmbeddings(disallowed_special=())
    vector_search_index_name = os.getenv("ATLAS_VECTOR_SEARCH_INDEX", "vector_index")

    # ``from_documents`` will:
