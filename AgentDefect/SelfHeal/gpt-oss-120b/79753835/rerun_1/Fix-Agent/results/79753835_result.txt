import os
from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer


def create_embedding_model(chunk_size: int | None = None) -> HuggingFaceEmbedding:
    """Create a BGE‑M3 embedding model.
    If the model is already cached under ``./embeddings`` we reuse it, otherwise we download it.
    ``chunk_size`` is passed as ``max_length`` to the underlying ``HuggingFaceEmbedding`` – this
    is the same value that the ``TokenCountingHandler`` uses when it tokenises the text.
    """
    model_name = "BAAI/bge-m3"
    cache_dir = "./embeddings"
    os.makedirs(cache_dir, exist_ok=True)
    # ``max_length`` is the truncation length used by the embedding model.
    embed_model = HuggingFaceEmbedding(
        model_name=model_name,
        max_length=chunk_size,
        cache_folder=cache_dir,
    )
    return embed_model


def run_agent(text: str) -> tuple[int, int]:
    """Return the token count reported by the embedding model and the raw tokenizer.

    The ``TokenCountingHandler`` reports the number of tokens *after* the same preprocessing
    (truncation, removal of special tokens, etc.) that the embedding model applies.  To obtain a
    comparable raw token count we must use the exact same tokenizer configuration.
    """
    chunk_size = 512
    embed_model = create_embedding_model(chunk_size)

    # Set up the LlamaIndex callback manager with a token‑counting handler.
    token_counter = TokenCountingHandler()
    Settings.callback_manager = CallbackManager([token_counter])
    Settings.embed_model = embed_model

    # This call triggers the token counting inside ``TokenCountingHandler``.
    _ = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Use the *same* tokenizer and preprocessing steps as the embedding model.
    # ``embed_model`` stores the tokenizer it created internally as ``_tokenizer``.
    tokenizer: AutoTokenizer = embed_model._tokenizer
    tokenized = tokenizer(
        text,
        truncation=True,
        max_length=chunk_size,
        add_special_tokens=False,
        return_attention_mask=False,
        return_token_type_ids=False,
    )
    raw_token_count = len(tokenized["input_ids"])

    return embedding_tokens, raw_token_count


if __name__ == "__main__":
    sample_text = "Random words. This is a test! A very exciting test, indeed."
    embed_cnt, raw_cnt = run_agent(sample_text)
    print(f"The embedding model reported: {embed_cnt} tokens")
    print(f"The raw tokenizer (with identical settings) reports: {raw_cnt} tokens")
