import os
from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer


def create_embedding_model(chunk_size: int | None = None) -> HuggingFaceEmbedding:
    """Create a HuggingFaceEmbedding for the BGE‑M3 model.

    The ``max_length`` argument is forwarded to the underlying tokenizer and
    should therefore be used consistently when we count tokens manually.
    """
    model_name = "BAAI/bge-m3"
    cache_dir = "./embeddings"
    os.makedirs(cache_dir, exist_ok=True)
    # ``max_length`` is the chunk size that the embedding model will use when it
    # falls back to chunking the input text.
    embed_model = HuggingFaceEmbedding(
        model_name=model_name,
        max_length=chunk_size,
        cache_folder=cache_dir,
        # make sure the tokenizer inside the embedding model truncates the same
        # way we will do it manually.
        tokenizer_kwargs={"truncation": True, "max_length": chunk_size},
    )
    return embed_model


def count_tokens_like_embedding(text: str, chunk_size: int) -> int:
    """Return the number of tokens the BGE‑M3 model would see.

    The embedding implementation tokenises the text *once* (it does not split on
    whitespace) and, if ``max_length`` is set, it truncates to that length while
    still keeping the special tokens added by the tokenizer.  Replicating this
    behaviour is as simple as calling the model's tokenizer with the same
    arguments.
    """
    tokenizer = AutoTokenizer.from_pretrained("BAAI/bge-m3")
    tokens = tokenizer(
        text,
        truncation=True,
        max_length=chunk_size,
        add_special_tokens=True,
    )
    # ``tokens['input_ids']`` is a list of token ids.
    return len(tokens["input_ids"])


def run_agent(text: str):
    chunk_size = 512
    embed_model = create_embedding_model(chunk_size)

    # Set up the token‑counting callback – it will be filled when we request an
    # embedding.
    token_counter = TokenCountingHandler()
    callback_manager = CallbackManager([token_counter])

    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    # Trigger an embedding so that ``token_counter`` records the amount of tokens
    # used by the model.
    _ = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Manual token count that mirrors the model's preprocessing.
    manual_token_count = count_tokens_like_embedding(text, chunk_size)

    return embedding_tokens, manual_token_count


if __name__ == "__main__":
    sample_text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, manual_token_count = run_agent(sample_text)
    print(f"The embedding model reported: {embedding_tokens} tokens")
    print(f"Manual tokenisation (mirroring the model) reports: {manual_token_count} tokens")
