from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer
import os

    """Load the BGE‑M3 embedding model.

    The function checks whether the model has already been cached locally
    (``./embeddings/models--BAAI--bge-m3``). If it is not present the directory is
    created and the model is downloaded into it. ``_chunk_size`` is forwarded to
    the ``max_length`` argument of :class:`HuggingFaceEmbedding` so that the same
    token limit is used during embedding and when we manually count tokens.
    """
    print("loading embeddings…")

    # for the cache folder directly.
    if not os.path.isdir(cache_dir):
        os.makedirs(cache_dir, exist_ok=True)
    embed_model = HuggingFaceEmbedding(
        model_name=model_name,
        max_length=_chunk_size,
        cache_folder=cache_dir,
    )
    print("embeddings loaded")
    return embed_model


def run_agent(text: str):
    """Run a minimal agent that embeds *text* and reports two token counts.

    * ``embedding_tokens`` – the number of tokens that the embedding pipeline
      actually fed into the model (reported by ``TokenCountingHandler``).
    * ``token_count`` – the number of tokens obtained by tokenising the same
      text with the *exact* same settings (truncation and max‑length) that the
      embedding model uses.
    """
    chunk_size = 512  # the max length used by the BGE‑M3 model
    embed_model = create_embedding_model(_chunk_size=chunk_size)

    # Callback that records how many tokens are passed to the model during the
    # ``get_text_embedding`` call.
    token_counter = TokenCountingHandler()
    callback_manager = CallbackManager([token_counter])

    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    # Perform the embedding – the callback updates ``total_embedding_token_count``.
    _ = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Re‑create the tokenizer with the same model name.  The BGE‑M3 tokenizer
    # expects ``truncation=True`` and ``max_length`` equal to the model's
    # ``max_length`` (512 by default).  Using the same arguments guarantees that
    # the token count matches what the embedding pipeline used.
    tokenizer = AutoTokenizer.from_pretrained("BAAI/bge-m3")
    tokenized = tokenizer(
        text,
        truncation=True,
        max_length=chunk_size,
        add_special_tokens=True,
    )
    token_count = len(tokenized["input_ids"])
    return embedding_tokens, token_count
