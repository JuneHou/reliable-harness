def run_agent(text: str):
    """Run the embedding pipeline and return two token counts:
    1. The number of tokens the embedding model actually processed (as recorded by
       ``TokenCountingHandler``).
    2. The number of tokens produced by a raw call to the model's tokenizer when
       the same truncation settings are applied.  This mirrors the behaviour of
       ``HuggingFaceEmbedding`` so the two counts should match.
    """
    chunk_size = 512
    embed_model = create_embedding_model(_chunk_size=chunk_size)

    # Set up LlamaIndex globals – they are required for the embedding call to
    # trigger the token‑count callback.
    token_counter = TokenCountingHandler()
    callback_manager = CallbackManager([token_counter])
    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    # Obtain the embedding – this internally tokenises, truncates (to ``max_length``)
    # and therefore updates ``token_counter``.
    _ = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Replicate the exact tokenisation that ``HuggingFaceEmbedding`` performs.
    # The class uses ``tokenizer(text, truncation=True, max_length=self.max_length)``.
    tokenizer = AutoTokenizer.from_pretrained("BAAI/bge-m3")
    tokenized = tokenizer(
        text,
        truncation=True,
        max_length=chunk_size,
        add_special_tokens=True,
        return_attention_mask=False,
        return_token_type_ids=False,
    )
    token_count = len(tokenized["input_ids"])

    return embedding_tokens, token_count
