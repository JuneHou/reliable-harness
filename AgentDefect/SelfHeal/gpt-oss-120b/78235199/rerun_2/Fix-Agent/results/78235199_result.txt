import os
import json

class OpenAIFineTuningHandler:
    """Handler that collects fine‑tuning examples and writes them to a JSONL file."""

    def __init__(self):
        # Store a list of dicts: {"prompt": ..., "completion": ...}
        self.events = []

    def record_event(self, prompt: str, completion: str):
        """Add a fine‑tuning example to the internal list.

        Args:
            prompt: The input question or instruction.
            completion: The model's response.
        """
        self.events.append({"prompt": prompt, "completion": completion})

    def save_finetuning_events(self, path: str):
        """Write all recorded events to *path* as a JSONL file.

        If no events have been recorded the file will be empty, but we still
        create it so the caller can rely on its existence.
        """
        os.makedirs(os.path.dirname(path), exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            for ev in self.events:
                f.write(json.dumps(ev, ensure_ascii=False) + "\n")
        print(f"Wrote {len(self.events)} example{'s' if len(self.events) != 1 else ''} to {path}")


class CallbackManager:
    """Very small callback manager that simply stores a list of handlers.

    In a real implementation the manager would dispatch events (e.g.,
    ``on_llm_end``) to each handler. For this minimal example we only need the
    reference to the ``OpenAIFineTuningHandler`` instance.
    """

    def __init__(self, handlers):
        self.handlers = handlers

    # Helper for the simplified flow used below
    def record(self, prompt, completion):
        for h in self.handlers:
            if hasattr(h, "record_event"):
                h.record_event(prompt, completion)


class VectorStoreIndex:
    """Placeholder for the vector‑store index used by the agent.

    The real library would perform similarity search; here we just return the
    query unchanged because the purpose of the demo is to produce fine‑tuning
    examples, not to perform retrieval.
    """

    @staticmethod
    def from_documents(docs):
        # ``docs`` are not used in this mock implementation.
        return VectorStoreIndex()

    def as_query_engine(self, similarity_top_k, llm):
        # Return *self* as a mock query engine.
        self.llm = llm
        return self

    def query(self, question):
        """Simulate a response from an LLM.

        In a real setting this would call the language model. For the purpose of
        the unit test we produce a deterministic dummy answer.
        """
        # Create a fake response
        answer = f"Answer to: {question}"
        # Record the example via the callback manager attached to the LLM.
        if hasattr(self, "llm") and hasattr(self.llm, "callback_manager"):
            self.llm.callback_manager.record(question, answer)
        return answer


class FineTuningExample:
    """Orchestrates generation of a JSONL file containing prompt/completion pairs.

    The class reads questions from ``train_questions.txt`` and pretends to query
    a vector store / LLM. Each (question, answer) pair is handed to the
    ``OpenAIFineTuningHandler`` which later writes them to disk.
    """

    def __init__(self, dataset_path: str, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate the JSONL file for fine‑tuning events.

        Steps:
        1. Ensure the dataset directory exists.
        2. Initialise the handler and callback manager.
        3. Load questions from ``train_questions.txt``.
        4. Run each question through a mock query engine, recording the
           resulting (prompt, completion) pair.
        5. Finally, ask the handler to write everything to
           ``finetuning_events.jsonl``.
        """
        os.makedirs(self.dataset_path, exist_ok=True)

        # Initialise handler and callback infrastructure
        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        # Mock LLM object that only carries the callback manager
        self.llm = type("LLM", (), {"callback_manager": callback_manager})()

        # Load questions
        questions_path = os.path.join(self.dataset_path, "train_questions.txt")
        questions = []
        with open(questions_path, "r", encoding="utf-8") as f:
            for line in f:
                stripped = line.strip()
                if stripped:
                    questions.append(stripped)

        # Run queries and let the callback manager record events
        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                _ = query_engine.query(question)  # The answer is recorded via callbacks
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                os.path.join(self.dataset_path, "finetuning_events.jsonl")
            )


def prepare_test_data(questions_path: str):
    """Create a minimal ``train_questions.txt`` file for demonstration purposes."""
    os.makedirs(os.path.dirname(questions_path), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path: str):
    """Entry‑point used by the ``__main__`` block.

    It builds a ``FineTuningExample`` pointing at the directory that contains the
    ``train_questions.txt`` file and triggers JSONL generation.
    """
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
