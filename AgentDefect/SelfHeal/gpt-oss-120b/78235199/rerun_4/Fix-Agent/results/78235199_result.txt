import os
import json

class OpenAIFineTuningHandler:
    """Collects fine‑tuning examples and writes them to a JSONL file.

    The original implementation always wrote an empty file, which caused
    `save_finetuning_events` to report *0 examples*.  This version stores each
    (prompt, completion) pair in ``self.events`` and serialises them to JSONL
    when ``save_finetuning_events`` is called.
    """

    def __init__(self):
        self.events = []

    def record_event(self, prompt: str, completion: str):
        """Add a new fine‑tuning example.

        Args:
            prompt: The user question or prompt.
            completion: The model's response.
        """
        self.events.append({"prompt": prompt, "completion": completion})

    def save_finetuning_events(self, path: str):
        """Write all recorded events to ``path`` as a JSONL file.

        If no events have been recorded, an empty file is still created but the
        message correctly reflects that *0 examples* were written.
        """
        os.makedirs(os.path.dirname(path), exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            for event in self.events:
                json.dump(event, f)
                f.write("\n")
        print(f"Wrote {len(self.events)} examples to {path}")


class CallbackManager:
    def __init__(self, handlers):
        self.handlers = handlers

    # handlers.  For this simplified example we expose the first handler directly.

        """Generate a JSONL file containing fine‑tuning examples.

        # Initialise the handler and callback manager.

        # Attach the callback manager to a dummy LLM object.
        self.llm = type("LLM", (), {"callback_manager": callback_manager})()

        # Load questions.
        questions_path = os.path.join(self.dataset_path, "train_questions.txt")
        with open(questions_path, "r", encoding="utf-8") as f:
            questions = [line.strip() for line in f if line.strip()]

        # Generate responses and record events.
        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                response = query_engine.query(question)
                # Record the example using the handler.
                finetuning_handler.record_event(question, response)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                os.path.join(self.dataset_path, "finetuning_events.jsonl")
            )
