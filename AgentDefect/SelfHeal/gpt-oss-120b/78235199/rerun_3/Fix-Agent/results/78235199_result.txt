import os
import json

class OpenAIFineTuningHandler:
    """Collects fine‑tuning examples and writes them to a JSONL file.

    The original implementation only created an empty file, resulting in
    "0 examples" being written.  This version stores events internally and
    serialises them correctly.
    """

    def __init__(self):
        self.events = []  # List of dicts ``{"prompt": ..., "completion": ...}``

    def add_event(self, prompt: str, completion: str) -> None:
        """Add a single fine‑tuning example.

        Args:
            prompt: The input that will be sent to the model.
            completion: The model's response that should be used as the target.
        """
        self.events.append({"prompt": prompt, "completion": completion})

    def save_finetuning_events(self, path: str) -> None:
        """Write the collected events to *path* as a JSONL file.

        If no events have been added the file will be empty, but the function
        will still report how many examples were written.
        """
        os.makedirs(os.path.dirname(path), exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            for event in self.events:
                json_line = json.dumps(event, ensure_ascii=False)
                f.write(json_line + "\n")
        print(f"Wrote {len(self.events)} examples to {path}")


class CallbackManager:
    """Simple container for callback handlers.

    The real library passes the manager to the LLM so that events can be
    recorded.  For this minimal example we only need to store the handlers.
    """

    def __init__(self, handlers):
        self.handlers = handlers

    def record_event(self, prompt: str, completion: str) -> None:
        """Forward the event to every handler that implements *add_event*.
        """
        for handler in self.handlers:
            if hasattr(handler, "add_event"):
                handler.add_event(prompt, completion)


class VectorStoreIndex:
    """Placeholder index that pretends to answer queries.

    In a real implementation the index would search over embedded documents.
    Here we simply echo the question so that we have deterministic output for
    the test suite.
    """

    @staticmethod
    def from_documents(docs):
        # ``docs`` are not used in this stub implementation.
        return VectorStoreIndex()

    def as_query_engine(self, similarity_top_k, llm):
        # Store the provided LLM so we can call back to it later.
        self.llm = llm
        return self

    def query(self, question: str):
        """Return a deterministic answer and record the interaction.
        """
        # Produce a mock answer.
        answer = f"Answer to '{question}'"
        # Notify the LLM's callback manager about the event.
        if hasattr(self, "llm") and hasattr(self.llm, "callback_manager"):
            self.llm.callback_manager.record_event(question, answer)
        return answer


class FineTuningExample:
    """Generates a JSONL file containing (prompt, completion) pairs.
    """

    def __init__(self, dataset_path: str, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate JSONL file for fine‑tuning events.

        The method reads ``train_questions.txt`` from ``self.dataset_path``,
        runs each question through a mock ``VectorStoreIndex`` and records the
        resulting (question, answer) pairs via the ``OpenAIFineTuningHandler``.
        """
        # Ensure the output directory exists.
        os.makedirs(self.dataset_path, exist_ok=True)

        # Initialise the handler and callback manager.
        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        # Create a minimal LLM stand‑in that carries the callback manager.
        self.llm = type("LLM", (), {"callback_manager": callback_manager})()

        # Load questions from the training file.
        questions_path = os.path.join(self.dataset_path, "train_questions.txt")
        with open(questions_path, "r", encoding="utf-8") as f:
            questions = [line.strip() for line in f if line.strip()]

        # Run each question through the (mock) index and let the callback manager
        # record the events.
        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                _ = query_engine.query(question)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            # Persist the collected events.
            finetuning_handler.save_finetuning_events(
                os.path.join(self.dataset_path, "finetuning_events.jsonl")
            )


def prepare_test_data(questions_path: str) -> None:
    """Create a minimal ``train_questions.txt`` file for testing purposes."""
    os.makedirs(os.path.dirname(questions_path), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path: str) -> None:
    """Execute the fine‑tuning example generation workflow."""
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":

    prepare_test_data(questions_path)
    run_agent(questions_path)
