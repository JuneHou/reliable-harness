import os
import json

class OpenAIFineTuningHandler:
    """Handler responsible for persisting fine‑tuning events.

    The original implementation always wrote an empty file, which caused the
    user to see "0 examples" regardless of the actual number of generated
    events.  The fixed version accepts a list of events (each a ``dict`` with a
    ``prompt`` and a ``completion``) and writes them to a JSONL file – one JSON
    object per line.  It also prints a helpful summary.
    """

    def save_finetuning_events(self, path: str, events=None) -> None:
        """Write *events* to *path* in JSONL format.

        Args:
            path: Destination file path.
            events: An iterable of dictionaries.  If ``None`` or empty a file
                containing no lines is created.
        """
        # Ensure the directory exists.
        os.makedirs(os.path.dirname(path), exist_ok=True)
        # Normalise to a list for counting.
        events = list(events or [])
        with open(path, "w", encoding="utf-8") as f:
            for ev in events:
                # ``json.dumps`` guarantees proper escaping.
                f.write(json.dumps(ev, ensure_ascii=False) + "\n")
        print(f"Wrote {len(events)} examples to {path}")


class CallbackManager:
    """A minimal callback manager used only to illustrate integration.

    In the original code the manager simply stored handlers but never invoked
    them.  For the purpose of this example we keep the class but do not add any
    extra functionality.
    """

    def __init__(self, handlers):
        self.handlers = handlers


class VectorStoreIndex:
    """A stub index that pretends to retrieve information.

    The real library would perform similarity search; for the unit‑test we
    return a deterministic placeholder response so that the fine‑tuning JSONL
    contains meaningful data.
    """

    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex()

    def as_query_engine(self, similarity_top_k, llm):
        # The method simply returns *self* because the mock ``query`` method is
        # defined on the same class.
        return self

    def query(self, question):
        # Produce a deterministic fake answer for demonstration purposes.
        return f"Answer to: {question}"


class FineTuningExample:
    """Generate a JSONL file containing fine‑tuning examples.

    The class reads a list of training questions, runs each through a (mock)
    query engine and records the (question, answer) pairs.  The resulting pairs
    are written to ``finetuning_events.jsonl`` using ``OpenAIFineTuningHandler``.
    """

    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate the JSONL file for fine‑tuning events.

        The function creates the output directory, loads training questions,
        runs them through a mock LLM, and finally persists the (prompt,
        completion) pairs.
        """
        # Ensure the dataset directory exists.
        os.makedirs(self.dataset_path, exist_ok=True)

        # Initialise handler & mock LLM with a callback manager (kept for API
        # compatibility with the original code).
        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        self.llm = type("LLM", (), {"callback_manager": callback_manager})()

        # Load questions from the training file.
        questions_file = os.path.join(self.dataset_path, "train_questions.txt")
        questions = []
        with open(questions_file, "r", encoding="utf-8") as f:
            for line in f:
                stripped = line.strip()
                if stripped:
                    questions.append(stripped)

        # Collect (prompt, completion) pairs.
        events = []
        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                response = query_engine.query(question)
                events.append({"prompt": question, "completion": response})
        except Exception as e:
            print(f"An error occurred while generating events: {e}")
        finally:
            # Persist the gathered events – even if the list is empty we still
            # create a file so downstream code can rely on its existence.
            finetuning_handler.save_finetuning_events(
                os.path.join(self.dataset_path, "finetuning_events.jsonl"),
                events,
            )


def prepare_test_data(questions_path: str) -> None:
    """Create a simple training‑question file used by the demo.

    Args:
        questions_path: Full path where ``train_questions.txt`` will be written.
    """
    os.makedirs(os.path.dirname(questions_path), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path: str) -> None:
    """Entry point that mirrors the original script's behaviour.

    It constructs a ``FineTuningExample`` instance and triggers JSONL creation.
    """
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
