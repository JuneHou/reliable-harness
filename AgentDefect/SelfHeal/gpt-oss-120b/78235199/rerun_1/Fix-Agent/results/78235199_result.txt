import os
import json

class OpenAIFineTuningHandler:
    """Collects fine‑tuning examples and writes them to a JSONL file.
    In a real implementation the handler would be called by the LLM callback
    system. For this simplified example we expose an ``add_event`` method that
    the calling code can use directly.
    """

    def __init__(self):
        self.events = []

    def add_event(self, prompt: str, completion: str):
        """Store a single fine‑tuning example.

        Args:
            prompt: The user input / question.
            completion: The model's response.
        """
        self.events.append({"prompt": prompt, "completion": completion})

    def save_finetuning_events(self, path: str):
        """Write all collected events to ``path`` as JSONL.

        Each line in the file is a valid JSON object containing ``prompt`` and
        ``completion`` keys.
        """
        # Ensure the directory exists.
        os.makedirs(os.path.dirname(path), exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            for event in self.events:
                json_line = json.dumps(event, ensure_ascii=False)
                f.write(json_line + "\n")
        print(f"Wrote {len(self.events)} examples to {path}")


class CallbackManager:
    """Placeholder for compatibility with the original code.
    The real library would forward callbacks to the handlers. Here we only
    store the provided handlers so that external code could call methods on
    them if needed.
    """

    def __init__(self, handlers):
        self.handlers = handlers


class VectorStoreIndex:
    """A minimal stub that mimics the behaviour of the original class.
    For the purpose of this example ``query`` simply returns a dummy answer.
    """

    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex()

    def as_query_engine(self, similarity_top_k, llm):
        return self

    def query(self, question):
        # Return a deterministic dummy answer so that we have something to
        # record in the fine‑tuning file.
        return f"Answer to: {question}"


class FineTuningExample:
    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate a JSONL file containing fine‑tuning events.

        The method reads the list of questions from ``train_questions.txt``,
        queries the (stub) index for each question, records the prompt/completion
        pair via ``OpenAIFineTuningHandler`` and finally writes the file.
        """
        os.makedirs(self.dataset_path, exist_ok=True)

        # Initialise the handler and a mock LLM that holds a reference to the
        # handler through a ``callback_manager`` attribute – this mirrors the
        # original design where the LLM would automatically call the handler.
        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        self.llm = type("LLM", (), {"callback_manager": callback_manager})()

        # Load questions.
        questions_file = os.path.join(self.dataset_path, "train_questions.txt")
        with open(questions_file, "r", encoding="utf-8") as f:
            questions = [line.strip() for line in f if line.strip()]

        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                response = query_engine.query(question)
                # Record the example using the handler.
                finetuning_handler.add_event(prompt=question, completion=response)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                os.path.join(self.dataset_path, "finetuning_events.jsonl")
            )


def prepare_test_data(questions_path):

        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path):

    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
