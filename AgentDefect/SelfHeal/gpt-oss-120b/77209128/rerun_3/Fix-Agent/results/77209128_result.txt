metadata_json = json.dumps(metadata_dict)


class SqlQueryEngine:
    """Wraps the creation of the SQL table, LlamaIndex objects and the query engine.

    The important fix is that we now propagate the column ``comment`` (our custom
    JSON metadata) into the ``SQLTableSchema.context_str`` attribute.  LlamaIndex
    uses ``context_str`` when it builds the vector index, so the LLM can retrieve
    the description of the field during a natural‑language query.
    """

    def __init__(self) -> None:
        self.schema: list[SQLTableSchema] = []

    def _create_snapshot_table(self, engine) -> Table:
        """Create (or reuse) the ``snapshots`` table with a JSON ``data`` column.

        The column comment stores the JSON metadata string.
        """
        metadata_obj = MetaData()
        snapshots_table = Table(
            "snapshots",
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
        )
        metadata_obj.create_all(engine)
        return snapshots_table

    def _build_schema_from_metadata(self, table: Table) -> None:
        """Create a :class:`SQLTableSchema` for *table*.

        ``SQLTableSchema`` accepts a ``context_str`` argument that we fill with the
        column comment containing our JSON metadata.  If more than one column has a
        comment we simply concatenate them (separated by new‑lines).
        """
        # Gather any column comments – in our case there is only one, but we keep it
        # generic.
        comments = []
        for col in table.columns:
            if col.comment:
                comments.append(col.comment)
        context_str = "\n".join(comments) if comments else None

        schema_obj = SQLTableSchema(
            table_name=table.name,
            # ``context_str`` is what LlamaIndex later vectorises so the LLM can
            # retrieve it.
            context_str=context_str,
        )
        self.schema.append(schema_obj)

    def query_engine(self) -> SQLTableRetrieverQueryEngine:
        # -------------------------------------------------------------------
        # 1. Initialise the DB and create the snapshots table
        # -------------------------------------------------------------------
        engine = create_engine(pg_uri)
        snapshots_table = self._create_snapshot_table(engine)

        # -------------------------------------------------------------------
        # 2. Build LlamaIndex objects (SQLDatabase, node mapping, schema list)
        # -------------------------------------------------------------------
        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        # Populate ``self.schema`` – this is the list that will be turned into a
        # vector store index.  We must include the metadata as ``context_str``.
        self._build_schema_from_metadata(snapshots_table)

        # -------------------------------------------------------------------
        # 3. Create a vector index from the schema objects
        # -------------------------------------------------------------------
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        # -------------------------------------------------------------------
        # 4. Initialise the LLM (OpenAI) and attach it to LlamaIndex Settings
        # -------------------------------------------------------------------
        llm = OpenAI(temperature=0.9, model="gpt-3.5-turbo")
        Settings.llm = llm

        # -------------------------------------------------------------------
        # 5. Build the query engine that will retrieve the relevant schema based
        #    on similarity and then run the SQL query.
        # -------------------------------------------------------------------
        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )
        return query_engine


def run_agent(query: str):

    sql_engine = SqlQueryEngine()

    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response
