# Custom metadata that we want LlamaIndex to be aware of. It will be stored as a

    "page_call_phone_clicks_logged_in_unique": (
        "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
    )
}

class SqlQueryEngine:
    """Creates a simple table with a JSON column that carries custom metadata.

    The metadata is attached to the column comment and then copied into the
    ``SQLTableSchema.context_str`` attribute so that the LlamaIndex query engine
    can utilise it when answering natural‑language questions.
    """

    def __init__(self):
        self.schema = []
        self._meta = None  # MetaData object after reflection
        self._snapshots_table = None

    def _create_table(self, engine) -> Table:
        """Define and create the ``snapshots`` table.

        Returns the SQLAlchemy ``Table`` object.
        """
        metadata_obj = MetaData()
        table_name = "snapshots"
        metadata_json = json.dumps(metadata_dict)
        snapshots_table = Table(
            table_name,
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
        )
        metadata_obj.create_all(engine)
        return snapshots_table

    def _reflect_metadata(self, engine) -> MetaData:
        """Reflect the database schema so we can iterate over tables.
        """
        meta = MetaData()
        meta.reflect(bind=engine)
        return meta

    def query_engine(self):
        # -----------------------------------------------------------------
        # 1. Initialise the DB and create the snapshots table with metadata.
        # -----------------------------------------------------------------
        engine = create_engine(pg_uri)
        self._snapshots_table = self._create_table(engine)
        self._meta = self._reflect_metadata(engine)

        # -----------------------------------------------------------------
        # 2. Build LlamaIndex structures.
        # -----------------------------------------------------------------
        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        # Build a list of SQLTableSchema objects – one per reflected table.
        self.schema = []
        for table_name in self._meta.tables.keys():
            schema_obj = SQLTableSchema(table_name=table_name)
            # Inject column comment for the ``snapshots`` table.
            if table_name == self._snapshots_table.name:
                col_comment = self._snapshots_table.c["data"].comment
                # Attach it to the schema object; LlamaIndex will treat this as
                # supplemental context when retrieving the table.
                setattr(schema_obj, "context_str", col_comment)
            self.schema.append(schema_obj)

        # -----------------------------------------------------------------
        # 3. Create a vector index over the schema objects.
        # -----------------------------------------------------------------
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        # -----------------------------------------------------------------
        # 4. Initialise LLM and the query engine.
        # -----------------------------------------------------------------
        llm = OpenAI(temperature=0.9, model="gpt-3.5-turbo")
        Settings.llm = llm

        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )
        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")
    return sql_engine, response
