import os
import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer

from llama_index.core import SQLDatabase, VectorStoreIndex, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

load_dotenv()

OPENAI_API_KEY = os.environ.get("OPENAI_API_KEY")
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

pg_uri = os.environ.get("DATABASE_URL", "sqlite:///test.db")

# Custom metadata we want LlamaIndex to see during query processing
metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
}

class SqlQueryEngine:
    """Wraps the creation of a SQLAlchemy table with custom column comments and
    exposes a LlamaIndex SQLTableRetrieverQueryEngine that can make use of those
    comments as context for LLM‑driven query answering.
    """

    def __init__(self):
        # Will hold SQLTableSchema objects – one per table – that include the
        # column comment as ``context_str`` so the vector index can embed it.
        self.schema: list[SQLTableSchema] = []

    def _create_snapshot_table(self, engine) -> Table:
        """Create (or reflect) the ``snapshots`` table and attach the JSON
        description as a column comment.  The comment is stored on the ``data``
        column because SQLite – the default fallback – does not support native
        comments, but SQLAlchemy still keeps the information in the column
        object, which LlamaIndex can later read.
        """
        metadata_obj = MetaData()
        # Convert the metadata dict to a JSON string – this will be the comment.
        metadata_json = json.dumps(metadata_dict)
        snapshots_table = Table(
            "snapshots",
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
        )
        metadata_obj.create_all(engine)
        return snapshots_table

    def _build_schema_from_metadata(self, tables: list[Table]):
        """Populate ``self.schema`` with ``SQLTableSchema`` objects.  The
        ``context_str`` field receives the column comment (if any) so that the
        resulting vector store contains the custom metadata.
        """
        self.schema.clear()
        for table in tables:
            # Grab the comment from the first column that has one – in our case
            # the ``data`` column.
            comment = None
            for col in table.columns:
                if col.comment:
                    comment = col.comment
                    break
            # ``SQLTableSchema`` expects ``table_name`` and optionally
            # ``context_str`` (the free‑form text used for embedding).
            schema_item = SQLTableSchema(table_name=table.name, context_str=comment)
            self.schema.append(schema_item)

    def query_engine(self) -> SQLTableRetrieverQueryEngine:
        # -----------------------------------------------------------------
        # 1️⃣ Initialise the DB engine and create the snapshot table with a
        #    comment that holds our custom JSON metadata.
        # -----------------------------------------------------------------
        engine = create_engine(pg_uri)
        snapshots_table = self._create_snapshot_table(engine)

        # -----------------------------------------------------------------
        # 2️⃣ Build a LlamaIndex ``SQLDatabase`` wrapper and the node mapping.
        # -----------------------------------------------------------------
        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        # -----------------------------------------------------------------
        # 3️⃣ Build the vector index, embedding the custom metadata stored in
        #    ``context_str`` of each ``SQLTableSchema`` instance.
        # -----------------------------------------------------------------
        self._build_schema_from_metadata([snapshots_table])
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        # -----------------------------------------------------------------
        # 4️⃣ Initialise the LLM (OpenAI) and attach it to the global Settings.
        # -----------------------------------------------------------------
        llm = OpenAI(temperature=0.9, model="gpt-3.5-turbo")
        Settings.llm = llm

        # -----------------------------------------------------------------
        # 5️⃣ Assemble the final query engine that will retrieve the most
        #    relevant table description (including our JSON comment) before
        #    executing the SQL query.
        # -----------------------------------------------------------------
        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )
        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine…")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response


if __name__ == "__main__":
    # Example query that should surface the custom JSON description.

    sql_engine, response = run_agent(query)

            print(f"  context_str: {schema_item.context_str}")
            print("  Status: Metadata context is SET")

            print("  context_str: None")
