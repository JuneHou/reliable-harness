import os

from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer
from llama_index.core import SQLDatabase, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

# Load environment variables
load_dotenv()

OPENAI_API_KEY = os.environ.get("OPENAI_API_KEY")
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

# Database connection string – default to a local SQLite file for testing
pg_uri = os.environ.get("DATABASE_URL", "sqlite:///test.db")

# The custom metadata we want to attach to the table (as a JSON string)
metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
}

class SqlQueryEngine:
    """Creates a SQL query engine that can surface custom column‑level metadata.

    The key fix is that we now populate ``SQLTableSchema.context_str`` with the
    comment attached to the ``data`` column.  ``SQLTableRetrieverQueryEngine``
    will use that context when answering natural‑language queries.
    """

    def __init__(self) -> None:
        self.schema: list[SQLTableSchema] = []

    def _create_snapshot_table(self, engine) -> MetaData:
        """Creates a ``snapshots`` table with a JSON column whose *comment* holds
        the custom metadata. The function returns the ``MetaData`` instance so
        that callers can inspect the table definition.
        """
        metadata_obj = MetaData()
        metadata_json = json.dumps(metadata_dict)
        Table(
            "snapshots",
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
        )
        metadata_obj.create_all(engine)
        return metadata_obj

    def _build_schema_from_metadata(self, metadata_obj: MetaData) -> None:
        """Convert SQLAlchemy table objects into ``SQLTableSchema`` instances.
        The ``context_str`` for each schema is populated with the first column
        comment found (in our case the comment on the ``data`` column).
        """
        for table in metadata_obj.sorted_tables:
            # Retrieve the comment – we only expect one column with a comment.
            comment: str | None = None
            for col in table.columns:
                if col.comment:
                    comment = col.comment
                    break
            # Build the schema object, attaching the comment as context.
            schema_item = SQLTableSchema(table_name=table.name, context_str=comment)
            self.schema.append(schema_item)

    def query_engine(self) -> SQLTableRetrieverQueryEngine:
        # ------------------------------------------------------------------
        # 1. Initialise the DB and create the snapshot table with metadata.
        # ------------------------------------------------------------------
        engine = create_engine(pg_uri)
        metadata_obj = self._create_snapshot_table(engine)

        # ------------------------------------------------------------------
        # 2. Build the LlamaIndex objects that map SQL tables to nodes.
        # ------------------------------------------------------------------
        sql_database = SQLDatabase(engine)
        node_mapping = SQLTableNodeMapping(sql_database)

        # Populate ``self.schema`` with context so the vector index knows about it.
        self._build_schema_from_metadata(metadata_obj)

        # ------------------------------------------------------------------
        # 3. Create a vector store index from the schema objects.
        # ------------------------------------------------------------------
        index = ObjectIndex.from_objects(self.schema, node_mapping, VectorStoreIndex)

        # ------------------------------------------------------------------
        # 4. Set up the LLM and the query engine.
        # ------------------------------------------------------------------
        llm = OpenAI(temperature=0.9, model="gpt-3.5-turbo")
        Settings.llm = llm

        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )
        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!\n")

    print(f"Executing query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}\n")
    return sql_engine, response
