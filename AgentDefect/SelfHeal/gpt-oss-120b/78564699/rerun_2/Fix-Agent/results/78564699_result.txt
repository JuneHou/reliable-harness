import json
import os
from llama_index.llms.llama_cpp import LlamaCPP


def run_agent(query: str) -> str:
    """Run a simple LlamaCPP inference.

    The function expects a *local* GGUF model file.  If a ``paths.json`` file
    exists and contains a path for ``llama-2-13b-chat`` the model will be loaded
    from that location; otherwise the user is expected to download the model
    manually and point ``MODEL_PATH`` to it.

    Using ``model_url`` together with ``model_path`` can cause the library to
    attempt a download every time, which on a CPU‑only system appears to "freeze"
    for a long time.  By providing only ``model_path`` we avoid the hidden
    download step and get deterministic behaviour.
    """

    # ---------------------------------------------------------------------
    # Resolve the model location.
    # ---------------------------------------------------------------------
    # Default path – the user must have placed the model here.
    default_model_path = "./models/llama-2-13b-chat.Q4_0.gguf"
    model_path = default_model_path

    # Allow an override via a small JSON file.
    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r", encoding="utf-8") as f:
            paths = json.load(f)
        # ``paths`` can map a friendly name to an absolute path on the host.
        if "llama-2-13b-chat" in paths:
            model_path = paths["llama-2-13b-chat"]

    # ---------------------------------------------------------------------
    # Initialise the LlamaCPP wrapper.
    # ---------------------------------------------------------------------
    llm = LlamaCPP(
        # ``model_path`` is the only argument we pass – ``model_url`` is left as
        # ``None`` to avoid any automatic download that could block the process.
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=64,          # keep the generation short for CPU tests
        context_window=3900,
        # ``n_gpu_layers`` = 0 forces a pure‑CPU run.  Adding ``n_threads``
        # speeds up the inference on multi‑core CPUs.
        model_kwargs={"n_gpu_layers": 0, "n_threads": max(1, os.cpu_count() - 1)},
        verbose=False,             # silence internal logging unless debugging
    )

    # ``complete`` returns a ``CompletionResponse``; the generated text lives in
    # the ``text`` attribute.
    response = llm.complete(query)
    return response.text


if __name__ == "__main__":
    # Simple sanity‑check query.
    question = "What is the capital of France?"
    answer = run_agent(question)
    print(answer)

    """Download the GGUF model if it is not already present locally.

    Returns
    -------
    str
        The absolute path to the downloaded model file.
    """
    # Directory where we keep downloaded models
    models_dir = Path("./models")
    models_dir.mkdir(parents=True, exist_ok=True)

    # Expected filename (same as on HF hub)
    filename = "llama-2-13b-chat.Q4_0.gguf"
    model_path = models_dir / filename

    if not model_path.is_file():
        # Download from Hugging Face Hub (the repo contains the GGUF file)
        # The repo URL is the same as used originally.
        repo_id = "TheBloke/Llama-2-13B-chat-GGUF"
        # The `hf_hub_download` function will handle caching & resumable download.
        model_path = Path(
            hf_hub_download(repo_id=repo_id, filename=filename, cache_dir=str(models_dir))
        )
    return str(model_path.resolve())


def load_paths_override() -> str | None:
    """Optionally read a JSON file that maps model identifiers to local paths.

    The JSON should have a structure like:
    {
        "llama-2-13b-chat": "/my/custom/path/llama-2-13b-chat.Q4_0.gguf"
    }
    """
    paths_file = Path("./paths.json")
    if paths_file.is_file():
        with paths_file.open("r", encoding="utf-8") as f:
            try:
                paths = json.load(f)
                return paths.get("llama-2-13b-chat")
            except json.JSONDecodeError:
                # Corrupt JSON – ignore and fall back to normal download
                return None
    return None

    if overridden_path:
        model_path = overridden_path

    # Initialise the LlamaCPP wrapper – we only pass a local path; model_url is left as None.

        model_path=model_path,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={"n_gpu_layers": 0, "n_threads": os.cpu_count()},
        verbose=True,
    )

    # LlamaCPP's `complete` method returns a CompletionResponse; the actual text is in the
    # `text` attribute.  Returning that string matches the expectation of a simple answer.
    response = llm.complete(query)
    return response.text if hasattr(response, "text") else str(response)
