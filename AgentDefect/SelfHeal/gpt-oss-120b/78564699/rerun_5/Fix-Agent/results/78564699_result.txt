import json
import os
from llama_index.llms.llama_cpp import LlamaCPP


def run_agent(query: str) -> str:
    """Run a simple LlamaCPP inference on a local GGUF model.

    The function first tries to read a ``paths.json`` file that may contain a
    pre‑downloaded model path. If the path is present we use it, otherwise we
    fall back to a remote URL. In the latter case the model will be downloaded
    the first time the wrapper is instantiated – for large models (e.g. 13B)
    this can take a long time and may look like the program is *frozen*.

    To avoid that surprise we prefer a local model and explicitly disable the
    remote download when a path is supplied.
    """

    # Remote URL – used only when no local path is available.
    remote_url = (
        "https://huggingface.co/TheBloke/Llama-2-13B-chat-GGUF/resolve/main/"
        "llama-2-13b-chat.Q4_0.gguf"
    )
    model_path: str | None = None

    # ``paths.json`` can be generated by the user to point to a locally stored
    # GGUF model, e.g. ``{"llama-2-13b-chat": "/home/user/models/llama-2-13b-chat.Q4_0.gguf"}``
    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        try:
            with open(paths_file, "r", encoding="utf-8") as f:
                paths = json.load(f)
                # If we have a stored path we use it and *do not* set a URL.
                if "llama-2-13b-chat" in paths:
                    model_path = paths["llama-2-13b-chat"]
        except Exception as e:
            # If the JSON is malformed we fall back to the remote URL.
            print(f"Failed to read paths.json ({e}), falling back to remote download.")

    # Initialise the LlamaCPP wrapper.  When ``model_path`` is provided we set
    # ``model_url`` to ``None`` to prevent an automatic (and very large) download.
    llm = LlamaCPP(
        model_path=model_path,
        model_url=None if model_path else remote_url,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        # ``n_gpu_layers`` = 0 forces CPU‑only execution. ``n_threads`` caps the
        # number of CPU threads to avoid oversubscription which can make the
        # process appear hung.
        model_kwargs={"n_gpu_layers": 0, "n_threads": 4},
        verbose=True,
    )

    # ``complete`` returns a ``CompletionResponse`` – we only need the text.
    response = llm.complete(query)
    return response.text

import json
import os
from pathlib import Path
from llama_index.llms.llama_cpp import LlamaCPP

def run_agent(query: str) -> str:
    """Run a prompt through a locally‑stored GGUF model using LlamaCPP.

    The function prefers a pre‑downloaded model located at ``paths.json``.
    If the model is not found, it raises a clear error instead of silently
    trying to download a 13 B‑parameter file (which can appear to "freeze").
    """

    # ---------------------------------------------------------------------
    # Resolve the path to the GGUF model.
    # ---------------------------------------------------------------------
    # Expected format of ``paths.json``:
    # {
    #   "llama-2-13b-chat": "/absolute/or/relative/path/to/llama-2-13b-chat.Q4_0.gguf"
    # }
    # ---------------------------------------------------------------------
    default_key = "llama-2-13b-chat"
    paths_file = Path("./paths.json")

    if not paths_file.is_file():
        raise FileNotFoundError(
            "paths.json not found. Please create the file and point it to a local GGUF model."
        )

    with paths_file.open("r", encoding="utf-8") as f:
        paths = json.load(f)

    model_path_str = paths.get(default_key)
    if not model_path_str:
        raise ValueError(
            f"'{default_key}' entry missing in paths.json. Provide the absolute path to the GGUF file."
        )

    model_path = Path(model_path_str).expanduser().resolve()
    if not model_path.is_file():
        raise FileNotFoundError(f"GGUF model not found at {model_path}")

    # ---------------------------------------------------------------------
    # Initialise the LlamaCPP LLM.  All GPU‑related layers are disabled (CPU‑only).
    # ``n_threads`` is set to the number of logical CPUs to keep the inference
    # responsive. ``verbose=True`` prints loading progress.
    # ---------------------------------------------------------------------
    llm = LlamaCPP(
        model_path=str(model_path),
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={
            "n_gpu_layers": 0,            # CPU‑only inference
            "n_threads": os.cpu_count(),  # utilise all cores
            "use_mmap": True,
            "use_mlock": False,
        },
        verbose=True,
    )

    # ``complete`` returns a ``CompletionResponse``; the generated text is in ``.text``.
    response = llm.complete(query)
    return response.text
