import json
import os
from pathlib import Path
from llama_index.llms.llama_cpp import LlamaCPP


def run_agent(query: str) -> str:
    """Run a LlamaCPP inference on the given query.

    The function first tries to resolve a local path for the model from a
    ``paths.json`` file. If the path is not present, it falls back to the
    HuggingFace URL and lets ``llama_index`` download the model automatically.
    Only one of ``model_path`` or ``model_url`` is passed to ``LlamaCPP`` –
    providing both can cause the library to hang on CPU inference.
    """

    # Default remote model URL (used only if we cannot find a local file).
    default_model_url = (
        "https://huggingface.co/TheBloke/Llama-2-13B-chat-GGUF/resolve/main/"
        "llama-2-13b-chat.Q4_0.gguf"
    )
    model_url: str | None = default_model_url
    model_path: str | None = None

    # Look for a JSON file that maps model names to local paths.
    paths_file = Path("./paths.json")
    if paths_file.is_file():
        with paths_file.open("r", encoding="utf-8") as f:
            paths = json.load(f)
            # Expected key in the JSON – adjust if you use a different identifier.
            local_path = paths.get("llama-2-13b-chat")
            if isinstance(local_path, str) and Path(local_path).is_file():
                model_path = local_path
                model_url = None  # Ensure we do NOT pass both arguments.

    # Initialise the LlamaCPP LLM.  ``model_path`` takes precedence when provided.
    llm = LlamaCPP(
        model_path=model_path,
        model_url=model_url,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        # ``n_gpu_layers=0`` forces CPU‑only execution.
        model_kwargs={"n_gpu_layers": 0},
        verbose=True,
    )

    # ``complete`` returns a ``CompletionResponse``; ``.text`` holds the generated string.
    completion = llm.complete(query)
    return completion.text


if __name__ == "__main__":
