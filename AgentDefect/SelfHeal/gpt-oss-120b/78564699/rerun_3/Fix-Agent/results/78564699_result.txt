import os
import json
from llama_index.llms.llama_cpp import LlamaCPP
from huggingface_hub import hf_hub_download


def _resolve_model_path() -> str:
    """Return a local path to the GGUF model.

    * If ``paths.json`` exists and contains a mapping for ``"llama-2-13b-chat"``
      we honour that entry.
    * Otherwise we download the model from HuggingFace (cached under ``./models``)
      and return the downloaded file path.
    """
    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if "llama-2-13b-chat" in paths:
                return paths["llama-2-13b-chat"]

    # Default repository / filename for the 13B Chat GGUF model
    repo_id = "TheBloke/Llama-2-13B-chat-GGUF"
    filename = "llama-2-13b-chat.Q4_0.gguf"
    # ``hf_hub_download`` caches the file locally (default cache dir is
    # ``~/.cache/huggingface``).  We explicitly point to ``./models`` to keep the
    # repository tidy.
    return hf_hub_download(repo_id=repo_id, filename=filename, cache_dir="./models")


def run_agent(query: str) -> str:
    """Run a simple completion against the 13B Chat model using LlamaCPP.

    The function loads the model in CPU‑only mode (``n_gpu_layers=0``) and
    returns the generated text.  ``max_new_tokens`` limits the output length,
    while ``temperature`` keeps the answer deterministic enough for testing.
    """
    model_path = _resolve_model_path()

    # Initialise the LlamaCPP wrapper – only ``model_path`` is required for
    # local inference. Supplying ``model_url`` together with ``model_path`` can
    # cause the loader to hang while trying to resolve both sources.
    llm = LlamaCPP(
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={"n_gpu_layers": 0, "n_threads": 8},
        verbose=True,
    )

    # ``complete`` returns an ``LLMResponse`` object; the generated text is in
    # the ``text`` attribute.
    response = llm.complete(query)
    return getattr(response, "text", str(response))


if __name__ == "__main__":
    print(run_agent("What is the capital of France?"))
