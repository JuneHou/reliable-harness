import os
import json
from huggingface_hub import hf_hub_download
from llama_index.llms.llama_cpp import LlamaCPP


def get_model_path():
    """Return the local path to the GGUF model, downloading it if necessary."""
    # Path to a JSON file that can store custom model locations
    paths_file = "./paths.json"
    model_name_key = "llama-2-13b-chat"
    # If the user has stored a custom path, use it
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if model_name_key in paths:
                return paths[model_name_key]
    # Otherwise download the model from HuggingFace (once) and return the local path
    repo_id = "TheBloke/Llama-2-13B-chat-GGUF"
    filename = "llama-2-13b-chat.Q4_0.gguf"
    # `hf_hub_download` caches the file under ~/.cache/huggingface/hub
    return hf_hub_download(repo_id=repo_id, filename=filename)


def run_agent(query: str) -> str:
    """Run a simple completion using LlamaCPP on CPU.

    The function loads the GGUF model from a local path (downloading it on the
    first run) and returns the generated text.
    """
    model_path = get_model_path()

    # Initialise the LlamaCPP wrapper.  Only `model_path` is supplied – the
    # wrapper does not download the model itself.  All inference runs on the CPU
    # (`n_gpu_layers=0`).  `verbose=False` avoids excessive logging that can look
    # like a hang.
    llm = LlamaCPP(
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={"n_gpu_layers": 0, "n_threads": max(1, os.cpu_count() - 1)},
        verbose=False,
    )

    # `complete` returns a Generation object; the generated text is stored in
    # the `text` attribute.
    response = llm.complete(query)
    return response.text if hasattr(response, "text") else str(response)


if __name__ == "__main__":
    query = "What is the capital of France?"
    answer = run_agent(query)
    print(answer)
