import os
from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer


def create_embedding_model(chunk_size: int | None = None) -> HuggingFaceEmbedding:
    """Load the BGE‑M3 embedding model.

    The function mirrors the original logic but simplifies the path handling – the
    model will be cached in ``./embeddings`` regardless of whether it already
    exists.
    """
    print("loading embeddings…")
    model_name = "BAAI/bge-m3"
    # ``max_length`` is the chunk size that the embedding model will use when
    # it needs to split long texts.  Passing ``None`` disables truncation.
    embed_model = HuggingFaceEmbedding(
        model_name=model_name,
        max_length=chunk_size,
        cache_folder="./embeddings",
    )
    print("embeddings loaded")
    return embed_model


def run_agent(text: str) -> tuple[int, int]:
    """Return the token count reported by the embedding handler and the exact
    token count that the BGE‑M3 tokenizer would see.

    The previous version used a fresh ``AutoTokenizer`` which does not share the
    same preprocessing settings (e.g., special‑token handling) as the tokenizer
    internally used by ``HuggingFaceEmbedding``.  This caused a mismatch between
    ``embedding_tokens`` and ``token_count``.

    The fix obtains the tokenizer directly from the embedding model and applies
    the same parameters (no padding, no truncation) that the embedding pipeline
    uses.  The token count is therefore identical to the count recorded by the
    ``TokenCountingHandler``.
    """
    chunk_size = 512
    embed_model = create_embedding_model(chunk_size)

    # Set up the token‑counting callback so that the embedding pipeline reports
    # how many tokens it has processed.
    token_counter = TokenCountingHandler()
    callback_manager = CallbackManager([token_counter])
    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    # Perform embedding – the callback updates ``token_counter``.
    _ = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Use the *same* tokenizer that the embedding model uses.  The embedding
    # implementation tokenises with ``add_special_tokens=True`` (the default) and
    # does **not** truncate, so we mimic that here.
    tokenizer = embed_model.tokenizer  # type: ignore[attr-defined]
    tokenized = tokenizer(text, add_special_tokens=True, truncation=False)
    token_count = len(tokenized["input_ids"])

    return embedding_tokens, token_count


if __name__ == "__main__":
    sample_text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(sample_text)
    print(f"The embedding model processed: {embedding_tokens} tokens")
    print(f"The tokenizer sees: {token_count} tokens")
