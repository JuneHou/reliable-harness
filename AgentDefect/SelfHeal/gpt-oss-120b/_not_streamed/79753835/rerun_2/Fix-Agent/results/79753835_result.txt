from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer
import os


def create_embedding_model(_chunk_size=None):
    """Load the BGE‑M3 embedding model.
    The function keeps the original logic but makes the intent clearer.
    """
    print("loading embeddings...")
    cache_dir = "./embeddings"
    os.makedirs(cache_dir, exist_ok=True)
    _emb_model_name = "BAAI/bge-m3"
    # ``max_length`` is passed to the underlying HuggingFace tokenizer – it will be used for truncation.
    _embed_model = HuggingFaceEmbedding(
        model_name=_emb_model_name,
        max_length=_chunk_size,
        cache_folder=cache_dir,
    )
    print("embeddings loaded")
    return _embed_model


def run_agent(text: str):
    # The chunk size used by the embedding model – this must be the same when we count tokens manually.
    chunk_size = 512
    embed_model = create_embedding_model(_chunk_size=chunk_size)

    # Set up the token counting callback – it will record how many tokens the embedding pipeline actually sees.
    token_counter = TokenCountingHandler()
    callback_manager = CallbackManager([token_counter])
    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    # Trigger the embedding (the callback records the token count).
    _ = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # To verify the count we must replicate *exactly* the tokenizer configuration used by
    # ``HuggingFaceEmbedding``.  The class creates a tokenizer with ``truncation=True``,
    # ``add_special_tokens=False`` and honours the ``max_length`` argument.
    model_name = "BAAI/bge-m3"
    tokenizer = AutoTokenizer.from_pretrained(model_name)
    tokenized = tokenizer(
        text,
        truncation=True,
        max_length=chunk_size,
        add_special_tokens=False,
        return_attention_mask=False,
    )
    token_count = len(tokenized["input_ids"])

    return embedding_tokens, token_count


if __name__ == "__main__":
    text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(text)
    print(f"The embedding model broke the text into: {embedding_tokens} tokens")
    print(f"The tokenizer broke the text into {token_count} tokens")
