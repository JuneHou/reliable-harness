import os
from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer


def create_embedding_model(_chunk_size=None):
    """Load the BGE‑M3 embedding model.

    The function checks for a cached model directory and loads the model
    from the HuggingFace hub if it is not present. ``_chunk_size`` is passed
    as the ``max_length`` argument so that the embedding model truncates input
    sequences to the same length that we later use for token counting.
    """
    print('loading embeddings...')
    model_name = "BAAI/bge-m3"
    cache_dir = './embeddings'
    os.makedirs(cache_dir, exist_ok=True)
    embed_model = HuggingFaceEmbedding(
        model_name=model_name,
        max_length=_chunk_size,
        cache_folder=cache_dir,
    )
    print('embeddings loaded')
    return embed_model


def run_agent(text: str):
    """Return the token count used by the embedding model and the raw tokenizer.

    The function:
    1. Instantiates the embedding model with a ``max_length`` (chunk size).
    2. Attaches a ``TokenCountingHandler`` to capture the number of tokens the
       embedding model actually consumes (including any special tokens and
       truncation).
    3. Calls ``get_text_embedding`` – this triggers the callback and therefore
       populates ``token_counter.total_embedding_token_count``.
    4. Tokenises the same text *exactly* as the embedding model would, i.e. with
       the same ``max_length`` and ``truncation`` settings, and counts the
       resulting token ids.
    The two numbers should now match.
    """
    chunk_size = 512
    embed_model = create_embedding_model(_chunk_size=chunk_size)

    # Set up token‑counting callback
    token_counter = TokenCountingHandler()
    callback_manager = CallbackManager([token_counter])
    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    # Run the embedding – this updates the token counter
    _ = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Tokenise with the identical configuration used inside the embedding model
    tokenizer = AutoTokenizer.from_pretrained("BAAI/bge-m3")
    tokenized = tokenizer(
        text,
        truncation=True,
        max_length=chunk_size,
        add_special_tokens=True,
    )
    token_count = len(tokenized["input_ids"])

    return embedding_tokens, token_count


if __name__ == "__main__":
    sample_text = "Random words. This is a test! A very exciting test, indeed."
    embed_tokens, tokenizer_tokens = run_agent(sample_text)
    print(f"The embedding model processed: {embed_tokens} tokens")
    print(f"The tokenizer (same settings) produced: {tokenizer_tokens} tokens")
