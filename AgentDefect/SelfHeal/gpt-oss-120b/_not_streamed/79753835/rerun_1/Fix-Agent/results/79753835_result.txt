def run_agent(text: str):
    # BGE‑M3 expects a maximum context length of 512 tokens.  This is the value
    # we also pass to ``HuggingFaceEmbedding`` so that the model truncates inputs
    # in exactly the same way that LlamaIndex does when it creates an embedding.
    chunk_size = 512
    embed_model = create_embedding_model(chunk_size)

    # Set up LlamaIndex settings – this is required only if you later want to
    # generate embeddings through ``Settings.embed_model``.
    token_counter = TokenCountingHandler()
    callback_manager = CallbackManager([token_counter])
    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    # Obtain the embedding – this will also update the token counter.
    _ = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # Compute the *exact* token count that the BGE‑M3 model uses.
    # The HuggingFaceEmbedding class internally uses ``AutoTokenizer`` with the
    # following arguments: ``add_special_tokens=True``, ``truncation=True`` and

    # gives us a token count that matches the embedding behaviour without needing
    # to perform the actual embedding inference.
    tokenizer = AutoTokenizer.from_pretrained("BAAI/bge-m3")

        text,
        truncation=True,
        max_length=chunk_size,
        padding=False,
        add_special_tokens=True,
        return_attention_mask=False,
    )
    token_count = len(tokenized["input_ids"])

    return embedding_tokens, token_count
