from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
import os


def create_embedding_model(chunk_size=None):
    """Load the BGE‑M3 embedding model.

    Args:
        chunk_size (int, optional): Maximum number of tokens the model will
            accept. This value is passed to ``HuggingFaceEmbedding`` as
            ``max_length`` so that the internal tokenizer respects the same
            truncation behaviour as the embedding call.
    """
    print("loading embeddings...")
    # Use a deterministic cache location – the folder is created if it does not exist.
    cache_dir = "./embeddings"
    os.makedirs(cache_dir, exist_ok=True)
    model_name = "BAAI/bge-m3"
    embed_model = HuggingFaceEmbedding(
        model_name=model_name,
        max_length=chunk_size,
        cache_folder=cache_dir,
    )
    print("embeddings loaded")
    return embed_model


def run_agent(text: str):
    # BGE‑M3 recommends a context window of 512 tokens for embedding.
    chunk_size = 512
    embed_model = create_embedding_model(chunk_size)

    # Set up token counting – the handler will be triggered from the embedding call.
    token_counter = TokenCountingHandler()
    callback_manager = CallbackManager([token_counter])
    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    # Perform the embedding – the token counter records the number of tokens that
    # the model actually processes (including any special tokens and truncation).
    _ = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # ---------------------------------------------------------------------
    # Compute the token count using the *same* tokenizer configuration that the
    # embedding model uses. This mirrors the preprocessing performed inside the
    # HuggingFaceEmbedding implementation (adding special tokens, truncating to
    # ``max_length`` and applying any model‑specific prefix).
    # ---------------------------------------------------------------------
    tokenizer = embed_model._tokenizer if hasattr(embed_model, "_tokenizer") else embed_model.tokenizer
    tokenized = tokenizer(
        text,
        truncation=True,
        max_length=chunk_size,
        add_special_tokens=True,
        return_attention_mask=False,
        return_token_type_ids=False,
    )
    token_count = len(tokenized["input_ids"])

    return embedding_tokens, token_count


if __name__ == "__main__":
    sample_text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, token_count = run_agent(sample_text)
    print(f"The embedding model processed: {embedding_tokens} tokens")
    print(f"The tokenizer (same config) reports: {token_count} tokens")
