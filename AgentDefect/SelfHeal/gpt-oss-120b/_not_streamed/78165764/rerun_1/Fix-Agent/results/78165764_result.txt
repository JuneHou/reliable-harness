from dotenv import load_dotenv
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

    title="LangChain Server",
    version="1.0",
    description="Spin up a simple api server using LangChain's Runnable interfaces",

vectorstore = FAISS.from_texts(
    products["text"].astype(str).tolist(), embedding=OpenAIEmbeddings()
)

# Prompt template that explicitly asks the model to answer in full sentences
ANSWER_TEMPLATE = """Answer the question based on the following context. Provide a clear, complete‑sentence answer.

Context:
{context}

Question: {question}
"""

prompt = ChatPromptTemplate.from_messages(
    [
        ("system", "You are a helpful assistant. Always respond with well‑formed, complete sentences."),
        ("human", ANSWER_TEMPLATE),
    ]
)

# Initialise the model – the correct argument name is `model_name`
model = ChatOpenAI(
    temperature=0.1,
    model_name="gpt-3.5-turbo",
    openai_api_key=os.getenv("OPENAI_API_KEY"),
)

chain = (
    {
        "context": retriever,
        "question": RunnablePassthrough(),
    }
    | prompt
    | model
    | StrOutputParser()
)


def run_agent(query: str) -> str:
    """Run a single query through the RAG chain and return the answer.

    The function is deliberately lightweight – all heavy objects (vectorstore, model, chain)
    are instantiated at module import time so that the API can serve many requests
    without rebuilding resources on each call.
    """
    return chain.invoke(query)


if __name__ == "__main__":
