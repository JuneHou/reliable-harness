import os
from fastapi import FastAPI
from dotenv import load_dotenv
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

# Initialise the FastAPI app once (outside the function) – it is not recreated on every request
app = FastAPI(
    title="LangChain Server",
    version="1.0",
    description="Spin up a simple api server using LangChain's Runnable interfaces",
)

# Load product data and build the vector store once – this avoids re‑building it for every query
products = pd.read_csv("./data/products.csv")
vectorstore = FAISS.from_texts(
    products["text"].astype(str).tolist(),
    embedding=OpenAIEmbeddings(),
)
retriever = vectorstore.as_retriever()

# Prompt that explicitly asks the model to answer in full sentences
ANSWER_TEMPLATE = (
    "Answer the question based on the following context. "
    "Provide the answer in complete, grammatically correct sentences.\n\n"
    "Context:\n{context}\n\n"
    "Question: {question}\n"
    "Answer:"
)
prompt = ChatPromptTemplate.from_template(ANSWER_TEMPLATE)

# Initialise the LLM – use the correct argument name (model_name) and a higher temperature for more natural prose
model = ChatOpenAI(
    model_name="gpt-3.5-turbo",
    temperature=0.7,
    openai_api_key=os.getenv("OPENAI_API_KEY"),
)

# Build the chain once – it can be reused for every query
chain = (
    {
        "context": retriever,
        "question": RunnablePassthrough(),
    }
    | prompt
    | model
    | StrOutputParser()
)


def run_agent(query: str) -> str:
    """Run the RAG chain for a single user query and return the answer as a string."""
    return chain.invoke(query)

if __name__ == "__main__":
    test_query = "What is the highest budget movie?"
    print(run_agent(test_query))
