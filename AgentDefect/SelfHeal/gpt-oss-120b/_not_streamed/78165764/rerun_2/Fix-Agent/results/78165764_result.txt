import os
from fastapi import FastAPI
from dotenv import load_dotenv
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

# Helper to turn a list of Documents into a single string for the prompt
def format_docs(docs):
    """Join document contents into a single string.

    LangChain's retriever returns a list of `Document` objects. The LLM expects a plain
    string, so we concatenate the `page_content` of each document separated by newlines.
    """
    return "\n".join([doc.page_content for doc in docs])


def run_agent(query: str) -> str:
    """Run a simple RAG chain that returns a complete‑sentence answer.

    The prompt now explicitly asks the model to answer in a full sentence, and the
    retriever output is correctly formatted before being injected into the prompt.
    """
    # ---------------------------------------------------------------------
    # Model configuration – use the correct parameter name ``model_name``
    # ---------------------------------------------------------------------
    model = ChatOpenAI(
        temperature=0.1,
        model_name="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    # ---------------------------------------------------------------------
    # Load data & build vector store
    # ---------------------------------------------------------------------
    products = pd.read_csv("./data/products.csv")
    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(), embedding=OpenAIEmbeddings()
    )
    retriever = vectorstore.as_retriever()

    # ---------------------------------------------------------------------
    # Prompt – ask for a complete‑sentence answer
    # ---------------------------------------------------------------------
    ANSWER_TEMPLATE = (
        "Answer the following question using the provided context. "
        "Please respond in a complete sentence.\n\n"
        "Context:\n{context}\n\n"
        "Question: {question}\n"
    )
    prompt = ChatPromptTemplate.from_template(ANSWER_TEMPLATE)

    # ---------------------------------------------------------------------
    # Build the chain: retrieve → format docs → inject into prompt → LLM → parse
    # ---------------------------------------------------------------------
    chain = (
        {"context": retriever | format_docs, "question": RunnablePassthrough()}
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)
    return result


if __name__ == "__main__":
    test_query = "What is the highest budget movie?"
    print(run_agent(test_query))
