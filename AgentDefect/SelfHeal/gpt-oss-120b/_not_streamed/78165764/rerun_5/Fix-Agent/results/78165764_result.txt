import os
from dotenv import load_dotenv
import pandas as pd
from fastapi import FastAPI
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough, RunnableLambda

load_dotenv()

def run_agent(query: str) -> str:
    """Run a simple RAG chain that returns a full‑sentence answer.

    The original implementation passed the raw list of `Document` objects
    directly to the prompt, causing the model to see a Python representation
    (e.g. ``[Document(page_content='...'), ...]``).  This confused the LLM and
    resulted in terse or fragmented output.  The fix joins the retrieved
    documents into a plain text context and adds an explicit instruction for
    the model to answer in complete sentences.
    """

    # ---------------------------------------------------------------------
    # Model configuration – keep temperature low for deterministic answers.
    # ---------------------------------------------------------------------
    model = ChatOpenAI(
        temperature=0.1,
        model_name="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    # ---------------------------------------------------------------------
    # Load data and build the vector store.
    # ---------------------------------------------------------------------
    products = pd.read_csv("./data/products.csv")
    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings(),
    )
    retriever = vectorstore.as_retriever()

    # ---------------------------------------------------------------------
    # Helper runnable that converts a list of Document objects into a single
    # string that can be interpolated safely into the prompt.
    # ---------------------------------------------------------------------
    def docs_to_string(docs):
        # Each `doc` is a `Document` with a `page_content` attribute.
        return "\n\n".join(doc.page_content for doc in docs)

    combine_docs = RunnableLambda(docs_to_string)

    # ---------------------------------------------------------------------
    # Prompt – explicit instruction to answer in full sentences.
    # ---------------------------------------------------------------------
    ANSWER_TEMPLATE = (
        "Answer the question based on the following context and respond in a "
        "complete sentence (or sentences).\n\n"
        "Context:\n{context}\n\n"
        "Question: {question}\n"
        "Answer:"  # the model will fill this in
    )
    prompt = ChatPromptTemplate.from_template(ANSWER_TEMPLATE)

    # ---------------------------------------------------------------------
    # Build the chain: retrieve → combine docs → prompt → model → parse.
    # ---------------------------------------------------------------------
    chain = (
        {
            "context": retriever | combine_docs,
            "question": RunnablePassthrough(),
        }
        | prompt
        | model
        | StrOutputParser()
    )

    # Execute the chain with the user query.
    result = chain.invoke(query)
    return result

# -------------------------------------------------------------------------
# Example execution (only when run as a script).
# -------------------------------------------------------------------------
if __name__ == "__main__":
    example_query = "What is the highest budget movie?"
    print(run_agent(example_query))
