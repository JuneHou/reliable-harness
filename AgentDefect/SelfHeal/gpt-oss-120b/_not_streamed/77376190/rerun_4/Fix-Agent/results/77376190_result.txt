import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr

# Load environment variables from a .env file (if present)
load_dotenv(find_dotenv())

def run_agent(user_input: str) -> str:
    """Generate a Non‑Disclosure Agreement (NDA) draft using OpenAI's chat model.

    The function sets up a system prompt that frames the model as a legal assistant and then
    sends the user's request. The response is returned as plain text.
    """
    # ---------------------------------------------------------------------
    # 1. Retrieve and validate the OpenAI API key
    # ---------------------------------------------------------------------
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise ValueError("OPENAI_API_KEY environment variable is not set")
    openai.api_key = api_key

    # ---------------------------------------------------------------------
    # 2. Define a clear system prompt that encourages the model to output an NDA
    # ---------------------------------------------------------------------
    system_prompt = (
        "You are a helpful legal assistant. Draft a concise, standard Non‑Disclosure "
        "Agreement (NDA) based on the user's request. The draft should contain the "
        "following sections: Parties, Definition of Confidential Information, Obligations, "
        "Term, and Signature blocks. Do not add any disclaimer about legal advice or ask "
        "for clarification unless the request is ambiguous."
    )

    # ---------------------------------------------------------------------
    # 3. Call the OpenAI ChatCompletion endpoint
    # ---------------------------------------------------------------------
    messages = [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": user_input},
    ]

    response = openai.ChatCompletion.create(
        model="gpt-3.5-turbo",
        messages=messages,
        temperature=0.3,
        max_tokens=1200,
        n=1,
        stop=None,
    )

    # ---------------------------------------------------------------------
    # 4. Extract and return the assistant's reply
    # ---------------------------------------------------------------------
    reply = response["choices"][0]["message"]["content"].strip()
    return reply

    # Build the vector index

    storage_dir = os.path.join(path, "_vector_index")
    vector_index.storage_context.persist(persist_dir=storage_dir)
