import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import SimpleDirectoryReader, GPTVectorStoreIndex, Settings
from llama_index.llms.openai import OpenAI

# Load environment variables
_ = load_dotenv(find_dotenv())

# Ensure requests can verify SSL (use default bundle)
os.environ.pop('REQUESTS_CA_BUNDLE', None)


def _init_openai():
    """Initialise OpenAI API key and the LlamaIndex LLM settings."""
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise RuntimeError("OPENAI_API_KEY not found in environment variables")
    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key
    # Configure LlamaIndex LLM (used for the vector‑store query path)
    Settings.llm = OpenAI(temperature=0.3, model="gpt-3.5-turbo", max_tokens=512)
    Settings.chunk_size = 600
    Settings.chunk_overlap = int(600 * 0.1)


def _create_vector_index(path: str) -> GPTVectorStoreIndex:
    """Create (or load) a vector index from a directory of documents.

    The index is persisted to ``vector_index.json`` so that subsequent calls
    can reuse the already‑built index, dramatically reducing latency.
    """
    # Persisted index location
    persist_dir = "vector_index"
    # If an index already exists, load it instead of rebuilding each time
    if os.path.isdir(persist_dir):
        try:
            from llama_index.core import load_index_from_storage
            return load_index_from_storage(persist_dir)
        except Exception:
            # Fall back to rebuilding if loading fails
            pass

    docs = SimpleDirectoryReader(path).load_data()
    vector_index = GPTVectorStoreIndex.from_documents(docs)
    # Persist for future runs
    vector_index.storage_context.persist(persist_dir=persist_dir)
    return vector_index


def _generate_nda_draft() -> str:
    """Ask OpenAI to produce a basic Non‑Disclosure Agreement draft.

    The prompt is carefully worded to stay within policy – it requests a
    *generic* template without asking for any party‑specific legal advice.
    """
    prompt = (
        "You are a helpful assistant. Produce a concise, generic template for a "
        "Non‑Disclosure Agreement (NDA) that could be used between two parties. "
        "Do not include any jurisdiction‑specific clauses or advise on legal "
        "strategy. Provide the sections in clear markdown format."
    )
    response = openai.ChatCompletion.create(
        model="gpt-3.5-turbo",
        messages=[{"role": "user", "content": prompt}],
        temperature=0.3,
        max_tokens=1000,
    )
    return response["choices"][0]["message"]["content"].strip()


def run_agent(user_input: str, docs_path: str = "docs") -> str:
    """Main entry point.

    - If the user explicitly asks for an NDA draft, we generate it directly via the
      OpenAI API.
    - Otherwise we fall back to the vector‑store query approach.
    """
    _init_openai()

    # Simple heuristic to decide which path to take
    lower_input = user_input.lower()
    if any(keyword in lower_input for keyword in ["nda", "non‑disclosure", "non disclosure", "draft"]):
        # Direct generation path
        return _generate_nda_draft()

    # Query path – build or load the vector index once per process
    vector_index = _create_vector_index(docs_path)
    query_engine = vector_index.as_query_engine()
    response = query_engine.query(user_input)
    return response.response


if __name__ == "__main__":

def _chat_completion(prompt: str, model: str = "gpt-3.5-turbo", temperature: float = 0.0, max_tokens: int = 1024) -> str:
    """Send *prompt* to the OpenAI chat API and return the assistant's reply.

    The function isolates the API call so we can reuse it for both the demo UI
    and any future extensions.
    """
    response = openai.ChatCompletion.create(
        model=model,

            {"role": "system", "content": "You are a helpful assistant that drafts legal documents. Do not refuse to provide a draft of a non‑disclosure agreement when asked. Keep the language clear, concise, and generic (no jurisdiction‑specific clauses)."},
            {"role": "user", "content": prompt},
        ],
        temperature=temperature,
        max_tokens=max_tokens,
        n=1,
        stop=None,
    )
    return response.choices[0].message["content"].strip()
