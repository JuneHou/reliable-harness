# -----------------------------------------------------------------------------
# Helper: Generate a Non‑Disclosure Agreement (NDA) draft using the OpenAI API.
# -----------------------------------------------------------------------------

def generate_nda_draft(user_request: str) -> str:
    """Ask the model to produce an NDA draft.

    The function builds a clear system prompt that tells the model it is allowed
    to draft legal‑style documents and that the content is for *illustrative*
    purposes only.  By using a low temperature and a reasonably high token
    limit we obtain a deterministic, complete draft.
    """
    # System prompt – clarifies the role and the permissibility of the request.
    system_prompt = (
        "You are an assistant that helps users draft non‑disclosure agreements (NDAs). "
        "All responses must be a complete, well‑formatted NDA draft suitable for "
        "illustrative purposes only. Do not refuse the request."
    )
    # User prompt – the actual request from the caller.
    user_prompt = user_request

    response = openai.ChatCompletion.create(
        model="gpt-3.5-turbo-16k",  # larger context window to hold a full draft
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_prompt},
        ],
        temperature=0.0,            # deterministic output
        max_tokens=1500,            # enough for a typical NDA
        n=1,
        stop=None,
    )
    # The content field contains the generated draft.
    return response.choices[0].message.content.strip()

from llama_index.core import SimpleDirectoryReader, GPTVectorStoreIndex, Settings
from llama_index.llms.openai import OpenAI


def create_vector_index(path: str) -> GPTVectorStoreIndex:
    """Create (or load) a vector index from the documents in *path*.

    The index is persisted to ``vector_index.json`` so that subsequent calls can
    reuse it without rebuilding.  Settings are tuned for the 3.5‑turbo model.
    """
    max_input = 4096
    tokens = 512
    chunk_size = 600
    max_chunk_overlap = 0.1

    # Configure the LLM used by llama_index.
    Settings.llm = OpenAI(
        temperature=0.3,
        model="gpt-3.5-turbo",
        max_tokens=tokens,
    )
    Settings.chunk_size = chunk_size
    Settings.chunk_overlap = int(chunk_size * max_chunk_overlap)

    # Load documents and build the index.
    docs = SimpleDirectoryReader(path).load_data()
    vector_index = GPTVectorStoreIndex.from_documents(docs)
    # Persist the index for future runs.
    vector_index.storage_context.persist(persist_dir="vector_index.json")
    return vector_index


# -----------------------------------------------------------------------------
# Main entry point – decides whether to generate an NDA or to answer via the
# vector store.
# -----------------------------------------------------------------------------
def run_agent(user_input: str, docs_path: str = "docs") -> str:
    """Route *user_input* to the appropriate handler.

    If the request looks like a request for an NDA draft, we bypass the vector
    store and call :func:`generate_nda_draft`. Otherwise we fall back to the
    document query engine.
    """
    # Simple heuristic – if the user explicitly mentions "NDA" or "non‑disclosure"
    # we treat it as a drafting request.
    lowered = user_input.lower()
    if "nda" in lowered or "non‑disclosure" in lowered or "non disclosure" in lowered:
        return generate_nda_draft(user_input)

    # Otherwise use the vector‑store query engine.
    vector_index = create_vector_index(docs_path)
    query_engine = vector_index.as_query_engine()
    response = query_engine.query(user_input)
    return response.response

# ---------------------------------------------------------------------------
# Helper: generate a Non‑Disclosure Agreement (NDA) draft using the OpenAI
# ChatCompletion endpoint.  The function does **not** rely on a document
# collection – it simply asks the model to produce a draft based on the
# user's request.  The system prompt is crafted to keep the model within the
# allowed content policy while still providing a useful legal‑style template.
# ---------------------------------------------------------------------------

def generate_nda(user_prompt: str, model: str = "gpt-3.5-turbo", temperature: float = 0.2) -> str:
    """Generate an NDA draft.

    Parameters
    ----------
    user_prompt: str
        The request from the user (e.g. "Create an NDA for a software project").
    model: str, optional
        The OpenAI model to use. Defaults to ``gpt-3.5-turbo``.
    temperature: float, optional
        Sampling temperature; a lower value makes the output more deterministic.

    Returns
    -------
    str
        The draft returned by the model.
    """
    # System message that guides the model to produce a standard NDA template.
    system_message = (
        "You are a helpful assistant. Draft a concise, standard non‑disclosure "
        "agreement (NDA) based on the user's request. Do not include any "
        "disallowed content and keep the language professional and neutral. "
        "If the user provides specifics (parties, jurisdiction, duration, etc.), "
        "incorporate them; otherwise, use generic placeholders."
    )

    messages = [
        {"role": "system", "content": system_message},
        {"role": "user", "content": user_prompt},
    ]

    try:
        response = openai.ChatCompletion.create(
            model=model,
            messages=messages,
            temperature=temperature,
            max_tokens=1000,
            n=1,
        )
        # Extract the assistant's reply
        return response.choices[0].message["content"].strip()
    except openai.error.OpenAIError as e:
        # Return a friendly error instead of letting the exception bubble up.
        return f"Error generating NDA: {e}"
