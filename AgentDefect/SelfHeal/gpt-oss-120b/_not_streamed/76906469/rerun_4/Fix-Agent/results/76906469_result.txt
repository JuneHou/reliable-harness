from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Initialise the LLM – we use a chat model (gpt‑4) which works with the
# *structured* chat‑react agent.  The structured version is required because
# the legacy ``CHAT_ZERO_SHOT_REACT_DESCRIPTION`` prompt does not contain a
# ``{chat_history}`` placeholder, so the agent cannot make use of a memory
# object.  The structured variant adds the ``{history}`` variable to the prompt
# and therefore automatically incorporates the supplied ``ConversationBufferMemory``.
llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

# Load the tools that the agent can invoke.
# ``llm-math`` and ``wikipedia`` are standard examples.
tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Create a conversation buffer memory.  The default ``memory_key`` for the
# structured chat‑react agent is ``"history"`` – we therefore keep the default
# name (or explicitly set it to match the placeholder used by the agent).
memory = ConversationBufferMemory(memory_key="history")

# Initialise the agent.  Using the *structured* chat‑react agent type enables
# the memory to be passed into the prompt.  No extra ``agent_kwargs`` are needed
# for basic usage.
agent = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.STRUCTURED_CHAT_ZERO_SHOT_REACT_DESCRIPTION,
    memory=memory,
    handle_parsing_errors=True,
    verbose=False,
)

def run_agent(prompt: str) -> str:
    """Run the agent with the supplied prompt and return its response.

    The agent now has access to the conversation history stored in ``memory``.
    """
    return agent.run(prompt)

if __name__ == "__main__":
    result1 = run_agent("What is the capital of France?")
    print(result1)
    # The second query can refer to the previous interaction because the
    # memory is now part of the prompt.
    result2 = run_agent("What was the last question I asked?")
    print(result2)
