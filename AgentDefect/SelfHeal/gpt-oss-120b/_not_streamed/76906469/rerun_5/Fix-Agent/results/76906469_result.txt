from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Initialise the LLM (GPT‑4) – temperature 0 for deterministic output
llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

# Load the tools we want the agent to have access to
tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Set up conversation memory.  The key "chat_history" is the default name used by
# the chat‑based agents to retrieve past dialogue.
memory = ConversationBufferMemory(memory_key="chat_history")

# NOTE:
#   The zero‑shot REACT agent (`CHAT_ZERO_SHOT_REACT_DESCRIPTION`) does **not**
#   read from memory – it makes decisions based solely on the current user
#   query and tool descriptions.  To enable the agent to consider prior turns we
#   must use the *chat* variant of the REACT agent (`CHAT_REACT_DESCRIPTION`).
#   This agent prefixes the LLM prompt with the stored ``chat_history``.
agent = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_REACT_DESCRIPTION,  # <-- memory‑aware agent
    handle_parsing_errors=True,
    memory=memory,
    verbose=False,
)


def run_agent(prompt: str) -> str:
    """Run the agent with the supplied user prompt.

    The function returns the agent's final answer.  The conversation buffer
    automatically records the user input and the agent's response, making the
    history available for subsequent calls.
    """
    return agent.run(prompt)


if __name__ == "__main__":
    # First query – the agent should answer using the tools.
    result1 = run_agent("what is the capital of France?")
    print(result1)

    # Second query – the agent now has access to the previous turn via memory.
    result2 = run_agent("what is the last question I asked?")
    print(result2)
