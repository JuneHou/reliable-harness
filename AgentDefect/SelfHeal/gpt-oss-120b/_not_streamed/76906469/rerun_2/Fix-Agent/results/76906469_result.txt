from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Initialize the LLM (Chat model)
llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

# Load the desired tools. The tools can be used by the agent during its reasoning steps.
tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Set up conversation memory. The key `chat_history` matches the placeholder used in the
# conversational react prompt template.
memory = ConversationBufferMemory(memory_key="chat_history")

# NOTE: For the agent to actually make use of the conversation memory you need a
# *conversational* ReAct agent. The `CHAT_ZERO_SHOT_REACT_DESCRIPTION` variant does not
# read from or write to the supplied memory. Switching to `CONVERSATIONAL_REACT_DESCRIPTION`
# enables the memory integration.
agent = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False,
)


def run_agent(prompt: str) -> str:
    """Run the agent with a user prompt and return its response."""
    return agent.run(prompt)


if __name__ == "__main__":
    # First query – the agent answers using the tool(s).
    result1 = run_agent("what is the capital of France?")
    print(result1)

    # Second query – the agent should recall the previous question from memory.
    result2 = run_agent("what is the last question I asked?")
    print(result2)
