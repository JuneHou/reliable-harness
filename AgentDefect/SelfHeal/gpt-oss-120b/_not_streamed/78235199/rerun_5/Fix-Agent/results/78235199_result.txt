import os
import json

class OpenAIFineTuningHandler:
    """Collects fine‑tuning examples and writes them to a JSONL file.

    The original implementation only wrote an empty string, which resulted in
    "Wrote 0 examples" and an empty file.  This version stores each example in an
    internal list and writes the list to disk in the expected JSONL format.
    """

    def __init__(self):
        self.examples = []  # each element is a dict with "prompt" and "completion"

    def add_example(self, prompt: str, completion: str):
        """Add a single fine‑tuning example.

        Args:
            prompt: The user input / question.
            completion: The model's response.
        """
        self.examples.append({"prompt": prompt, "completion": completion})

    def save_finetuning_events(self, path: str):
        """Write all collected examples to *path* as a JSONL file.

        The function creates the directory if it does not exist, writes one JSON
        object per line and reports the number of written examples.
        """
        os.makedirs(os.path.dirname(path), exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            for ex in self.examples:
                f.write(json.dumps(ex, ensure_ascii=False) + "\n")
        print(f"Wrote {len(self.examples)} examples to {path}")


class CallbackManager:
    """Simple container for callbacks used by the mock LLM.

    In a real implementation this would forward events to each handler.  For the
    purpose of this demonstration we only store the handlers – the LLM will call
    ``add_example`` directly on the ``OpenAIFineTuningHandler`` instance.
    """

    def __init__(self, handlers):
        self.handlers = handlers

    def get_finetuning_handler(self) -> OpenAIFineTuningHandler:
        for h in self.handlers:
            if isinstance(h, OpenAIFineTuningHandler):
                return h
        raise RuntimeError("OpenAIFineTuningHandler not found in handlers")


class VectorStoreIndex:
    """A stub index that simply returns a dummy answer for any query.

    The real library would perform similarity search over ``self.documents``.
    For the unit test we only need deterministic output.
    """

    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex()

    def as_query_engine(self, similarity_top_k, llm):
        # Store reference to the LLM so that we can forward responses to the
        # fine‑tuning handler later.
        self.llm = llm
        return self

    def query(self, question: str):
        # Produce a deterministic dummy response.
        response = f"Answer to: {question}"
        # Record the event using the handler attached to the LLM.
        try:
            handler = self.llm.callback_manager.get_finetuning_handler()
            handler.add_example(prompt=question, completion=response)
        except Exception:
            # If the handler cannot be found we simply ignore the recording – the
            # primary goal is not to crash.
            pass
        return response


class FineTuningExample:
    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate a JSONL file of fine‑tuning events.

        The function reads questions from ``train_questions.txt``, queries a mock
        index, records each (question, answer) pair via ``OpenAIFineTuningHandler``
        and finally writes the collected events to ``finetuning_events.jsonl``.
        """
        os.makedirs(self.dataset_path, exist_ok=True)

        # Initialise the handler and callback manager.
        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        # Create a lightweight mock LLM that carries the callback manager.
        self.llm = type("LLM", (), {"callback_manager": callback_manager})()

        # Load questions.
        questions_file = os.path.join(self.dataset_path, "train_questions.txt")
        questions = []
        with open(questions_file, "r", encoding="utf-8") as f:
            for line in f:
                stripped = line.strip()
                if stripped:
                    questions.append(stripped)

        # Perform queries and let the ``VectorStoreIndex`` record events.
        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                _ = query_engine.query(question)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            # Write out the collected examples.
            finetuning_handler.save_finetuning_events(
                os.path.join(self.dataset_path, "finetuning_events.jsonl")
            )


def prepare_test_data(questions_path):

def run_agent(questions_path):
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
