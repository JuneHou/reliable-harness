import os
import json

class OpenAIFineTuningHandler:
    """Collects fine‑tuning examples and writes them to a JSONL file.
    
    The original implementation always wrote an empty file.  This class now
    stores each (prompt, completion) pair that is recorded via ``record_event``
    and persists them correctly.
    """
    def __init__(self):
        self.events = []  # List of dicts with ``prompt`` and ``completion``

    def record_event(self, prompt: str, completion: str) -> None:
        """Save a single fine‑tuning example in memory.
        
        Args:
            prompt: The input question or prompt.
            completion: The model's response.
        """
        self.events.append({"prompt": prompt, "completion": completion})

    def save_finetuning_events(self, path: str) -> None:
        """Write all recorded events to *path* as JSONL.
        
        If no events have been recorded the file will be empty, but the function
        will still report the correct (zero) count.
        """
        # Ensure the directory exists.
        os.makedirs(os.path.dirname(path), exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            for event in self.events:
                f.write(json.dumps(event, ensure_ascii=False) + "\n")
        print(f"Wrote {len(self.events)} examples to {path}")


class CallbackManager:
    """A very small stand‑in for the real callback manager.
    
    It simply stores a list of handlers.  In this trimmed‑down example the
    handlers are not used directly, but the attribute is kept so that the code
    mirrors the expected interface of the original library.
    """
    def __init__(self, handlers):
        self.handlers = handlers


class VectorStoreIndex:
    """Placeholder for the vector‑store index used in the original code.
    
    The real implementation would perform similarity search; for the purpose
    of this demonstration we only need a dummy ``query`` method that returns a
    canned response.
    """
    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex()
    
    def as_query_engine(self, similarity_top_k, llm):
        # Return ``self``; ``query`` will be called on the returned object.
        self.llm = llm
        return self
    
    def query(self, question):
        # In a real scenario this would use the vector store and the LLM.
        # Here we simply echo the question with a fake answer.
        return f"Answer to '{question}'"


class FineTuningExample:
    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None
    
    def jsonl_generation(self):
        """Generate a JSONL file containing fine‑tuning events.
        
        The function reads questions from ``train_questions.txt``, runs each
        through a mock query engine, records the (question, answer) pair via the
        ``OpenAIFineTuningHandler`` and finally persists the collected events
        to ``finetuning_events.jsonl``.
        """
        os.makedirs(self.dataset_path, exist_ok=True)
        
        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        # Simulate an LLM object that holds the callback manager.
        self.llm = type("LLM", (), {"callback_manager": callback_manager})()
        
        # Load questions.
        questions_file = os.path.join(self.dataset_path, "train_questions.txt")
        questions = []
        with open(questions_file, "r", encoding="utf-8") as f:
            for line in f:
                stripped = line.strip()
                if stripped:
                    questions.append(stripped)
        
        # Execute the (mock) query for each question and record the event.
        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                response = query_engine.query(question)
                # Record the (prompt, completion) pair.
                finetuning_handler.record_event(prompt=question, completion=response)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            # Persist all recorded events.
            finetuning_handler.save_finetuning_events(
                os.path.join(self.dataset_path, "finetuning_events.jsonl")
            )


def prepare_test_data(questions_path):
    """Create a small ``train_questions.txt`` file for demonstration purposes."""
    os.makedirs(os.path.dirname(questions_path), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path):
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
    questions_path = "./dataset_data/train_questions.txt"
    prepare_test_data(questions_path)
    run_agent(questions_path)
