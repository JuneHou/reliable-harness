import os
import json

class OpenAIFineTuningHandler:
    """Collects fine‑tuning events and writes them to a JSONL file.

    In the original implementation ``save_finetuning_events`` always wrote an
    empty file because the handler never stored any data.  This version keeps a
    list of ``events`` (each a ``dict`` with ``prompt`` and ``completion``) and
    writes them out when requested.
    """

    def __init__(self):
        self.events = []

    def add_event(self, prompt: str, completion: str):
        """Record a single fine‑tuning example.

        Parameters
        ----------
        prompt: str
            The user query or instruction.
        completion: str
            The model's response that should be used as the fine‑tuning target.
        """
        self.events.append({"prompt": prompt, "completion": completion})

    def save_finetuning_events(self, path: str):
        """Write the collected events to ``path`` in JSONL format.

        If no events have been collected the file will be empty and a helpful
        message is printed.  This mirrors the behaviour expected by the test
        harness.
        """
        # Ensure the directory exists.
        os.makedirs(os.path.dirname(path), exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            for event in self.events:
                f.write(json.dumps(event, ensure_ascii=False) + "\n")
        print(f"Wrote {len(self.events)} examples to {path}")


class CallbackManager:
    """Simple container for callback handlers.
