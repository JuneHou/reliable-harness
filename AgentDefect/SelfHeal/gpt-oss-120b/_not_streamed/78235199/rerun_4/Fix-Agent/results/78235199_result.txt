import os
import json

class OpenAIFineTuningHandler:
    """Collects fine‑tuning events during the run and writes them to a JSONL file.
    
    Each event is stored as a dict with a ``prompt`` (the question) and a
    ``completion`` (the model's response).  The ``save_finetuning_events`` method
    writes one JSON object per line, which is the format expected by OpenAI's
    fine‑tuning endpoint.
    """
    def __init__(self):
        self.events = []

    def record_event(self, prompt: str, completion):
        """Record a single fine‑tuning example.
        
        Args:
            prompt: The input query.
            completion: The model's response (converted to string).
        """
        self.events.append({
            "prompt": prompt,
            "completion": str(completion)
        })

    def save_finetuning_events(self, path: str):
        """Write all recorded events to *path* as JSONL.
        
        If no events have been recorded an empty file is still created but a
        helpful message is printed.
        """
        # Ensure the directory exists.
        os.makedirs(os.path.dirname(path), exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            for event in self.events:
                f.write(json.dumps(event, ensure_ascii=False) + "\n")
        print(f"Wrote {len(self.events)} examples to {path}")


class CallbackManager:
    def __init__(self, handlers):
        self.handlers = handlers

    # The real library forwards events to handlers; here we provide a tiny
    # helper that forwards a ``record_event`` call to each handler that supports
    # it.
    def record_event(self, prompt, completion):
        for h in self.handlers:
            if hasattr(h, "record_event"):
                h.record_event(prompt, completion)


class VectorStoreIndex:
    """A stub implementation that pretends to retrieve a response.
    
    In the original code ``query_engine`` was the same object, and ``query``
    returned ``None`` – causing no useful data to be recorded.  For the purpose
    of this example we simply echo back a fabricated answer.
    """
    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex()
    
    def as_query_engine(self, similarity_top_k, llm):
        # Store a reference to the LLM so we can forward events via its
        # ``callback_manager``.
        self.llm = llm
        return self
    
    def query(self, question):
        # Generate a dummy response.
        response = f"Answer to '{question}'"
        # Forward the event to the callback manager attached to the LLM.
        if hasattr(self, "llm") and hasattr(self.llm, "callback_manager"):
            self.llm.callback_manager.record_event(question, response)
        return response


class FineTuningExample:
    def __init__(self, dataset_path, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None
    
    def jsonl_generation(self):
        """Generate a JSONL file containing fine‑tuning events.
        
        The function reads questions from ``train_questions.txt`` and, for each
        one, queries a mock ``VectorStoreIndex``.  The responses are captured by
        ``OpenAIFineTuningHandler`` via the callback manager and finally saved
        to ``finetuning_events.jsonl``.
        """
        os.makedirs(self.dataset_path, exist_ok=True)
        
        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        # Create a lightweight LLM stand‑in that holds the callback manager.
        self.llm = type("LLM", (), {"callback_manager": callback_manager})()
        
        questions = []
        questions_file = os.path.join(self.dataset_path, "train_questions.txt")
        with open(questions_file, "r", encoding="utf-8") as f:
            for line in f:
                stripped = line.strip()
                if stripped:
                    questions.append(stripped)
        
        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                _ = query_engine.query(question)  # Result is recorded via callbacks.
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            finetuning_handler.save_finetuning_events(
                os.path.join(self.dataset_path, "finetuning_events.jsonl")
            )
