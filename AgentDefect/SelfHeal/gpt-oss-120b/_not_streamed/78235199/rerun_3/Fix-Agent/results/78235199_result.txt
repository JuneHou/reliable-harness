import os
import json

class OpenAIFineTuningHandler:
    """Handler that collects fine‑tuning events and writes them to a JSONL file."""

    def __init__(self):
        self.events = []

    def add_event(self, prompt: str, completion: str) -> None:
        """Store a single fine‑tuning example.

        Args:
            prompt: The user question or prompt.
            completion: The model's response that should be used as the completion.
        """
        self.events.append({"prompt": prompt, "completion": completion})

    def save_finetuning_events(self, path: str) -> None:
        """Write all collected events to *path* in JSONL format.

        If no events have been collected the file will be empty but the
        function will still report the number of examples written.
        """
        os.makedirs(os.path.dirname(path), exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            for ev in self.events:
                json.dump(ev, f)
                f.write("\n")
        print(f"Wrote {len(self.events)} examples to {path}")


class CallbackManager:
    """Simple container for callbacks – kept for compatibility with the original code."""

    def __init__(self, handlers):
        self.handlers = handlers


class VectorStoreIndex:
    """Placeholder implementation – the real library would perform similarity search.

    For the purpose of this example we only need the ``query`` method to return a
    deterministic string so that a fine‑tuning event can be recorded.
    """

    @staticmethod
    def from_documents(docs):
        return VectorStoreIndex()

    def as_query_engine(self, similarity_top_k, llm):
        return self

    def query(self, question):
        # Simulated response – in a real scenario this would call an LLM.
        return f"Generated response for: {question}"


class FineTuningExample:
    """Orchestrates reading questions, generating responses and persisting them.

    The class now correctly writes a JSONL file containing one object per
    question/response pair.
    """

    def __init__(self, dataset_path: str, documents):
        self.dataset_path = dataset_path
        self.documents = documents
        self.llm = None

    def jsonl_generation(self):
        """Generate a JSONL file for fine‑tuning events.

        Steps:
        1. Ensure the dataset directory exists.
        2. Load the list of questions.
        3. Initialise the mock LLM and callback infrastructure.
        4. For each question, obtain a response and record the event.
        5. Persist all recorded events to ``finetuning_events.jsonl``.
        """
        os.makedirs(self.dataset_path, exist_ok=True)

        finetuning_handler = OpenAIFineTuningHandler()
        callback_manager = CallbackManager([finetuning_handler])
        # A very lightweight mock LLM exposing the callback manager.
        self.llm = type("LLM", (), {"callback_manager": callback_manager})()

        # Load questions from the provided file.
        questions_file = os.path.join(self.dataset_path, "train_questions.txt")
        questions = []
        with open(questions_file, "r", encoding="utf-8") as f:
            for line in f:
                stripped = line.strip()
                if stripped:
                    questions.append(stripped)

        # Generate responses and store events.
        try:
            index = VectorStoreIndex.from_documents(self.documents)
            query_engine = index.as_query_engine(similarity_top_k=2, llm=self.llm)
            for question in questions:
                response = query_engine.query(question)
                # Record the prompt/completion pair.
                finetuning_handler.add_event(prompt=question, completion=response)
        except Exception as e:
            print(f"An error occurred: {e}")
        finally:
            # Persist the collected events.
            finetuning_handler.save_finetuning_events(
                os.path.join(self.dataset_path, "finetuning_events.jsonl")
            )


def prepare_test_data(questions_path: str) -> None:
    """Create a small test dataset containing a couple of questions."""
    os.makedirs(os.path.dirname(questions_path), exist_ok=True)
    with open(questions_path, "w", encoding="utf-8") as f:
        f.write("Question 1\nQuestion 2\n")


def run_agent(questions_path: str) -> None:
    """Run the fine‑tuning example generation pipeline on the test data."""
    dataset_path = os.path.dirname(questions_path)
    ft = FineTuningExample(dataset_path=dataset_path, documents=[])
    ft.jsonl_generation()


if __name__ == "__main__":
