from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_verbose
from langchain.callbacks import ConsoleCallbackHandler
import logging
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Enable verbose mode globally
set_verbose(True)
# Configure root logger to show INFO level messages (used by LangChain)
logging.basicConfig(level=logging.INFO)

def run_agent(topic: str) -> str:
    """Generate a joke about the given topic while printing the intermediate
    prompts, LLM calls, and parsed output.

    Args:
        topic: The subject for the joke.

    Returns:
        The final string output produced by the chain.
    """
    # Attach a console callback handler so that each step in the LCEL chain is
    # printed to the console.
    callbacks = [ConsoleCallbackHandler()]
    model = ChatOpenAI(openai_api_key=api_key, callbacks=callbacks)
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)
    output_parser = StrOutputParser()
    # Build the LCEL chain (prompt -> model -> parser)
    chain = prompt | model | output_parser
    return chain.invoke({"topic": topic})

if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
