from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_verbose
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Enable global verbose logging
set_verbose(True)

def run_agent(topic: str) -> str:
    # Pass verbose=True to the model so that it logs its prompts and responses
    model = ChatOpenAI(openai_api_key=api_key, verbose=True)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)
    # The chain itself can also be set to verbose via a config when invoked
    chain = prompt | model | output_parser
    # Invoke with the "verbose" flag to see intermediate steps
    return chain.invoke({"topic": topic}, config={"verbose": True})

if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
