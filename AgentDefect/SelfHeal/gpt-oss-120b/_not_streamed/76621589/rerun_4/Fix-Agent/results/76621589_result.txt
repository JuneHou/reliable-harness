from time import time

import openai

from langchain.chains import LLMChain
from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate

_ = load_dotenv(find_dotenv())
openai.api_key = os.getenv("OPENAI_API_KEY")

# Initialise the LLM and prompt template
llm = ChatOpenAI(temperature=0)
prompt = ChatPromptTemplate.from_template(
    "Classify the text based on the Common European Framework of Reference "
    "for Languages (CEFR). Give a single value: {text}"
)
chain = LLMChain(llm=llm, prompt=prompt)


async def run_agent(texts):
    """Run the chain synchronously with ``apply`` and asynchronously with ``aapply``.

    Returns the result of the synchronous ``apply`` call for backward compatibility.
    """
    # Synchronous call – this blocks until the LLM responds for each input.
    start = time()
    res_sync = chain.apply(texts)
    print(res_sync)
    print(f"apply time taken: {time() - start:.2f} seconds\n")

    # Asynchronous call – we must ``await`` the coroutine returned by ``aapply``.
    start = time()
    res_async = await chain.aapply(texts)
    print(res_async)
    print(f"aapply time taken: {time() - start:.2f} seconds")

    # Return the synchronous result (or you could return the async one if preferred).
    return res_sync

    ]

    # ``run_agent`` is asynchronous, so we execute it via ``asyncio.run``.
    result = asyncio.run(run_agent(texts))
    print("\nReturned result:", result)

from dotenv import load_dotenv, find_dotenv

from langchain.prompts import ChatPromptTemplate

def run_agent(texts):
    """Run the LLM chain synchronously and also demonstrate the async version.

    The function:
    1. Loads the OpenAI API key from a .env file.
    2. Builds a ``ChatOpenAI`` model with deterministic output.
    3. Creates a ``ChatPromptTemplate`` that asks the model to classify a given
       text according to the CEFR scale.
    4. Executes ``chain.apply`` (the synchronous API) and prints the result.
    5. Executes ``chain.aapply`` (the asynchronous API) **awaiting** it via
       ``asyncio.run`` so that the coroutine is properly executed.
    6. Returns the synchronous result (mirroring the original behaviour).
    """
    # Load environment variables and set OpenAI key
    load_dotenv(find_dotenv())
    openai.api_key = os.getenv("OPENAI_API_KEY")

    # Initialise the LLM and the prompt template
    llm = ChatOpenAI(temperature=0)
    prompt = ChatPromptTemplate.from_template(
        "Classify the text based on the Common European Framework of Reference "
        "for Languages (CEFR). Give a single value: {text}"
    )
    chain = LLMChain(llm=llm, prompt=prompt)

    # Synchronous call
    start = time()
    res_a = chain.apply(texts)
    print(res_a)
    print(f"apply time taken: {time() - start:.2f} seconds\n")

    # Asynchronous call – we must *await* the coroutine. ``asyncio.run`` creates a

    start = time()
    res_aa = asyncio.run(chain.aapply(texts))
    print(res_aa)
    print(f"aapply time taken: {time() - start:.2f} seconds")
