import os
import asyncio
from time import time

import openai
from dotenv import load_dotenv, find_dotenv
from langchain.chains import LLMChain
from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate

# Load environment variables (e.g., OPENAI_API_KEY)
_ = load_dotenv(find_dotenv())
openai.api_key = os.getenv("OPENAI_API_KEY")

# Initialise the LLM and prompt template once – they can be reused for both sync and async calls
llm = ChatOpenAI(temperature=0)
prompt = ChatPromptTemplate.from_template(
    "Classify the text based on the Common European Framework of Reference "
    "for Languages (CEFR). Give a single value: {text}"
)
chain = LLMChain(llm=llm, prompt=prompt)

async def run_agent(texts):
    """Run the chain synchronously with ``apply`` and asynchronously with ``aapply``.

    The function is **async** so that we can ``await`` the asynchronous ``aapply`` call.
    It returns the result of the synchronous ``apply`` call (mirroring the original
    behaviour) while printing both results and their timings.
    """

    # ----------- Synchronous call (apply) -----------
    start = time()
    res_sync = chain.apply(texts)
    print(res_sync)
    print(f"apply time taken: {time() - start:.2f} seconds\n")

    # ----------- Asynchronous call (aapply) -----------
    start = time()
    res_async = await chain.aapply(texts)  # Properly await the coroutine
    print(res_async)
    print(f"aapply time taken: {time() - start:.2f} seconds\n")

    return res_sync

if __name__ == "__main__":

    # Run the async ``run_agent`` function using asyncio.run
    result = asyncio.run(run_agent(texts))
