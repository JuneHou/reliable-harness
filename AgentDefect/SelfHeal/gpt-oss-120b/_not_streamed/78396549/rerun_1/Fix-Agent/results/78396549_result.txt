import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.schema import HumanMessage

load_dotenv()

def run_agent(prompt: str) -> str:
    """Send a user prompt to the OpenAI chat model and return the response text.

    Args:
        prompt: The user query to be sent to the model.

    Returns:
        The model's reply as a plain string.
    """
    # Retrieve the API key from environment variables.  ``load_dotenv``
    # populates ``os.getenv`` from a ``.env`` file if present.
    api_key = os.getenv("OPENAI_API_KEY")
    llm = ChatOpenAI(openai_api_key=api_key)

    # ``ChatOpenAI`` expects a list of ``Message`` objects.  The simplest
    # way to pass a single user prompt is to wrap it in a ``HumanMessage``.
    response_message = llm.invoke([HumanMessage(content=prompt)])

    # ``invoke`` returns an ``AIMessage`` (or similar) which contains the
    # generated text in the ``content`` attribute.
    return response_message.content

if __name__ == "__main__":
