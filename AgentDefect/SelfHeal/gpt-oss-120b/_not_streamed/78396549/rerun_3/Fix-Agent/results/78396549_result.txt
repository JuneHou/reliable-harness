load_dotenv()


def run_agent(prompt: str) -> str:
    """Send *prompt* to the OpenAI chat model and return the textual response.

    LangChain's ``ChatOpenAI`` expects a list of ``BaseMessage`` objects (e.g., a
    ``HumanMessage``) rather than a raw string. The original code passed the
    string directly, which resulted in no visible output. Here we wrap the
    prompt in ``HumanMessage`` and extract the ``content`` field from the
    ``AIMessage`` returned by ``invoke``.
    """
    # ``ChatOpenAI`` will automatically read ``OPENAI_API_KEY`` from the
    # environment, so we don't need to pass it explicitly.
    llm = ChatOpenAI()

    response_message = llm.invoke([HumanMessage(content=prompt)])

    return response_message.content
