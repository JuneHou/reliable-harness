import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.schema import HumanMessage

load_dotenv()


def run_agent(prompt: str) -> str:
    """Send a user prompt to the OpenAI chat model and return the response text.

    Args:
        prompt: The user message you want the model to answer.

    Returns:
        The textual content of the model's reply.
    """
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise ValueError("OPENAI_API_KEY environment variable not set")

    # Initialise the chat model. You can customize model name, temperature, etc.
    llm = ChatOpenAI(openai_api_key=api_key)

    # ChatOpenAI expects a list of Message objects. Wrap the plain string
    # in a HumanMessage so the model knows it is a user prompt.
    response = llm.invoke([HumanMessage(content=prompt)])

    # The returned object is an AIMessage; its textual content is stored in
    # the ``content`` attribute.
    return response.content


if __name__ == "__main__":
    user_prompt = "How can LangSmith help with testing?"
    answer = run_agent(user_prompt)
    print("LLM response:", answer)
