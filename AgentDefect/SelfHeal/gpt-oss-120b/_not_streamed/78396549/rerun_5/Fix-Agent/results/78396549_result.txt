import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.schema import HumanMessage

load_dotenv()


def run_agent(prompt: str) -> str:
    """Send a user prompt to the OpenAI chat model and return the text response.

    Args:
        prompt: The user query to send to the LLM.

    Returns:
        The content of the model's reply as a plain string.
    """
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise ValueError("OPENAI_API_KEY not found in environment variables")

    llm = ChatOpenAI(openai_api_key=api_key)
    # ChatOpenAI expects a list of message objects; we provide a single HumanMessage.
    response_message = llm.invoke([HumanMessage(content=prompt)])
    # The invoke method returns a BaseMessage; the actual text is in its 'content' attribute.
    return response_message.content


if __name__ == "__main__":
