import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.llms import OpenAI
from langchain.chains import CSVQAChain
from langchain.callbacks import get_openai_callback

warnings.filterwarnings("ignore")

# Load environment variables (including OPENAI_API_KEY)
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def run_agent(csv_files, queries):
    """Run a set of natural‑language queries against one or more CSV files.

    This implementation uses LangChain's built‑in ``CSVQAChain`` which
    loads the CSV into a pandas ``DataFrame`` and lets the LLM operate on the
    tabular data directly.  It avoids the expensive *retrieval* step and the
    generic ``map_reduce`` chain, both of which struggle with aggregation
    queries (e.g. counts, max/min, look‑ups) on CSV rows.

    Parameters
    ----------
    csv_files: list[str]
        Paths to CSV files that should be queried.  The function processes each
        file separately and returns a list of results in the order the queries
        were executed.
    queries: list[str]
        Natural‑language questions about the CSV data.

    Returns
    -------
    list[str]
        Results from the LLM for each query/file combination.
    """
    results = []
    for file in csv_files:
        print("=" * 40)
        print(f"Processing {file}")
        print("=" * 40)
        # Use a callback to track token usage and cost – helpful for debugging
        with get_openai_callback() as cb:
            # Initialise the LLM (no temperature for deterministic output)
            llm = OpenAI(temperature=0, batch_size=20)
            # Build a CSV‑QA chain for the current file
            chain = CSVQAChain.from_llm(llm, csv_path=file)
            for query in queries:
                print(f"Query: {query}")
                result = chain.run(query)
                results.append(result)
                print(f"Result: {result}\n")
        # Show token usage for the whole file processing
        print(cb)
    return results
