import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.callbacks import get_openai_callback
from langchain.llms import OpenAI
from langchain.agents import create_csv_agent

# Suppress noisy warnings
warnings.filterwarnings("ignore")

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def run_agent(queries):
    """Run a CSV‑agent against each CSV file and collect the answers.

    The original implementation used a RetrievalQA with a map‑reduce chain,
    which is designed for summarisation and not for exact data look‑ups.  It
    embedded every row of the CSV, causing hallucinations and high token
    usage.  LangChain provides a specialised CSV agent (`create_csv_agent`) that
    translates natural‑language questions into pandas‑style operations.  This
    yields accurate results at a fraction of the cost.
    """
    results = []
    files = ["drivers.csv", "drivers_full.csv"]

    for file in files:
        print("=" * 36)
        print(file)
        print("=" * 36)

        # Use the OpenAI LLM with deterministic output (temperature=0)
        # The agent will load the CSV lazily and execute the query directly.
        llm = OpenAI(temperature=0)
        agent = create_csv_agent(llm, file_path=file, verbose=True)

        for query in queries:
            print(f"Query: {query}")
            # Track token usage via the callback context
            with get_openai_callback() as cb:
                result = agent.run(query)
                results.append(result)
                print(f"Result: {result}\n")
                # Optional: print token usage for debugging / cost monitoring
                print(cb)
    return results


if __name__ == "__main__":
    sample_queries = [
        "Tell me the name of every driver who is German",
        "How many drivers are German?",
        "Which driver has the number 14?",
        "What is the oldest birthdate in the file?",
    ]
    outputs = run_agent(sample_queries)
    # Print a summary of all results
    print("--- Summary ---")
    for out in outputs:
        print(out)
