import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.prompts import PromptTemplate

warnings.filterwarnings("ignore")

# Load environment variables (API key, etc.)
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def _build_qa_chain(retriever, chain_type="stuff"):
    """Create a RetrievalQA chain.

    * ``stuff`` – concatenates *all* retrieved docs and asks the LLM to answer in one go.  It is the most accurate for exact‑lookup
      questions (counts, specific rows, dates) because the model sees the full context.
    * ``map_reduce`` – summarises each doc individually before a final reduction, which works for open‑ended questions but loses
      fine‑grained facts.
    """
    # A simple prompt that tells the model the docs are CSV rows and asks for a direct answer.
    # Using ``stuff`` we can keep the prompt short while still giving the model the whole table.
    prompt = PromptTemplate(
        input_variables=["context", "question"],
        template=(
            "You are given several rows from a CSV file. Each row is a comma‑separated list of values."
            " Answer the question using ONLY the information present in the rows."
            " If the answer is a count, give the exact integer."
            " If the answer is a name or a date, provide it exactly as it appears."
            "\n\nContext:\n{context}\n\nQuestion: {question}\nAnswer:"),
    )

    return RetrievalQA.from_chain_type(
        llm=OpenAI(temperature=0, batch_size=20),
        chain_type=chain_type,
        retriever=retriever,
        combine_documents_chain_kwargs={"prompt": prompt},
        verbose=False,
    )


def run_agent(queries, files=("drivers.csv", "drivers_full.csv")):
    """Run the provided *queries* against each CSV file.

    The function now:
    1. Loads every CSV with ``CSVLoader`` – each row becomes a ``Document``.
    2. Creates a Chroma vector store from those documents.
    3. Retrieves *all* rows (``k`` is set to the number of documents) so that the LLM can compute exact aggregates.
    4. Uses the ``stuff`` chain which concatenates the rows, giving the model the full table for deterministic answers.
    5. Returns a list of results in the same order as the queries.
    """
    results = []

    for file in files:
        print("=" * 40)
        print(file)
        print("=" * 40)
        with get_openai_callback() as cb:
            # 1️⃣ Load CSV
            loader = CSVLoader(file_path=file, encoding="utf-8")
            docs = loader.load()

            # 2️⃣ Embed + index
            embeddings = OpenAIEmbeddings()
            db = Chroma.from_documents(docs, embeddings)

            # 3️⃣ Retrieve *all* rows – we want the LLM to see the complete dataset
            retriever = db.as_retriever(search_type="similarity", search_kwargs={"k": len(docs)})

            # 4️⃣ Build a ``stuff`` QA chain (most reliable for exact look‑ups)
            qa_chain = _build_qa_chain(retriever, chain_type="stuff")

            for query in queries:
                print(f"\nQuery: {query}")
                answer = qa_chain.run(query)
                results.append(answer)
                print(f"Answer: {answer}")

            # Show token usage / cost for this file
            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German.",
        "How many German drivers are there?",
        "What is the name of the driver with number 14?",
        "What is the oldest birthdate among all drivers?",
    ]
    res = run_agent(queries)

def build_vectorstore(csv_files):
    """Load all CSV files, turn each row into a Document, and build a single
    Chroma vector store that can be used by a retriever.

    Args:
        csv_files (list[str]): Paths to CSV files.

    Returns:
        Chroma: A vector store ready to be used as a retriever.
    """
    all_docs = []
    for file in csv_files:
        loader = CSVLoader(file_path=file, encoding="utf-8")
        docs = loader.load()
        # The loader yields one Document per row, with the raw CSV row as page_content.
        # Adding the file name as metadata makes debugging easier.
        for doc in docs:
            doc.metadata["source"] = file
        all_docs.extend(docs)

    embeddings = OpenAIEmbeddings()
    # Chroma automatically creates a persistent collection; using a temporary in‑memory location.
    db = Chroma.from_documents(all_docs, embeddings, collection_name="drivers")
    return db
