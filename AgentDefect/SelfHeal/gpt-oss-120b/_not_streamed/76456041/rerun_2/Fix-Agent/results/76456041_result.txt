import os
import warnings
from dotenv import load_dotenv, find_dotenv

from langchain.llms import OpenAI
from langchain.agents import create_pandas_dataframe_agent
import pandas as pd

warnings.filterwarnings("ignore")

# Load environment variables (API key)
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def run_agent(queries, files):
    """Run a Pandas‑DataFrame‑based agent on the supplied CSV files.

    The agent uses the schema of the DataFrame rather than retrieving raw
    text chunks, which avoids the similarity‑search bottleneck of the original
    map‑reduce chain and dramatically reduces token usage.
    """
    results = []
    for file in files:
        print("=" * 40)
        print(file)
        print("=" * 40)
        # Load the CSV into a DataFrame – this is cheap and gives us column types.
        df = pd.read_csv(file, encoding="utf-8")

        # Create an agent that can answer natural‑language questions about the DataFrame.
        # The agent internally translates the question to a pandas expression, so no
        # retrieval step (and no large similarity index) is required.
        agent = create_pandas_dataframe_agent(
            llm=OpenAI(temperature=0, model_name="gpt-3.5-turbo"),
            df=df,
            verbose=True,
        )

        for query in queries:
            print(f"Query: {query}")
            result = agent.run(query)
            results.append(result)
            print("Result:", result)
    return results


if __name__ == "__main__":
    # Example queries that previously gave wrong results with the map‑reduce chain.
    queries = [
        "Tell me the name of every driver who is German",
        "How many drivers are German?",
        "Which driver has the number 14?",
        "What is the oldest birthdate in the file?",
    ]
    files = ["drivers.csv", "drivers_full.csv"]
    run_agent(queries, files)
