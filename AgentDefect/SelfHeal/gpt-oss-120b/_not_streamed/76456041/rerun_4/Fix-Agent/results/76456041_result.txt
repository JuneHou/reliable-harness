import os
import warnings
from dotenv import load_dotenv, find_dotenv
import pandas as pd
from langchain.llms import OpenAI
from langchain.agents import create_pandas_dataframe_agent
from langchain.callbacks import get_openai_callback

# Silence warnings that are not relevant for the demo
warnings.filterwarnings("ignore")

# Load environment variables (especially OPENAI_API_KEY)
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def run_agent(queries):
    """Run a set of natural‑language queries against each CSV file.

    The function uses LangChain's Pandas DataFrame agent, which translates the
    user question into a pandas expression and executes it locally.  This avoids
    the inefficient RetrievalQA + map‑reduce approach that relied on embeddings
    and multiple LLM calls, dramatically lowering token usage and cost while
    producing exact answers for structured data.
    """
    results = []
    files = ["drivers.csv", "drivers_full.csv"]

    for file in files:
        print("=" * 40)
        print(file)
        print("=" * 40)

        # Load the CSV into a DataFrame – pandas handles all data‑type inference.
        df = pd.read_csv(file)

        # Create a single‑turn LLM agent that works with the DataFrame.
        llm = OpenAI(temperature=0)
        agent = create_pandas_dataframe_agent(llm, df, verbose=True)

        for query in queries:
            with get_openai_callback() as cb:
                print(query)
                result = agent.run(query)
                results.append(result)
                print(result)
                # The callback prints token usage & cost for the call.
                print(cb)
    return results


if __name__ == "__main__":
    # Example queries that previously produced wrong or incomplete answers.
    queries = [
        "Tell me the name of every driver who is German",
        "How many drivers are German?",
        "What is the name of the driver with number 14?",
        "What is the oldest birthdate among the drivers?",
    ]
    run_agent(queries)
