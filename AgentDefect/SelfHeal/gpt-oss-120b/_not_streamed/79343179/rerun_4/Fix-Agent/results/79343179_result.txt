import json
import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv

dotenv.load_dotenv()

# -----------------------------------------------------------------------------
# Tool definition
# -----------------------------------------------------------------------------
@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a *JSON* representation of the DataFrame.

    The LangChain tool interface expects the return value to be JSON‑serialisable
    (i.e., a string).  Returning a raw ``pandas.DataFrame`` causes LangChain to
    coerce it to a string representation, which loses the actual data.  By
    serialising the DataFrame to JSON (using the ``split`` orientation) we can
    reconstruct it later without loss.
    """
    df = (
        pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    )
    # Keep only object‑type columns as in the original implementation
    df = df.select_dtypes(include=["object"])
    # Serialise to JSON so that the agent can pass the data back as a string
    return df.to_json(orient="split")

# -----------------------------------------------------------------------------
# Agent runner
# -----------------------------------------------------------------------------
def run_agent(user_prompt: str, file_name: str = "data.json"):
    """Run the ReAct agent and return the *reconstructed* DataFrame (if any).

    The function executes the agent with the supplied ``user_prompt``.  The
    agent will call ``load_excel_response`` which returns a JSON string.  After
    the agent finishes we inspect the final tool output; if it looks like a
    DataFrame JSON we convert it back to a ``pd.DataFrame`` and return it alongside
    the raw agent response.
    """
    # LLM and tools
    llm = ChatOpenAI(model="gpt-4o-mini")
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    # Build the ReAct agent
    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # Execute
    response = agent_executor.invoke({"input": user_prompt})

    # The agent's final textual answer is under the ``output`` key.  If the
    # answer is a JSON string that can be parsed into a DataFrame we return it.
    df = None
    try:
        candidate = response.get("output", "")
        # Some agents wrap the JSON in backticks or code fences – strip them.
        candidate = candidate.strip().strip('`')
        # Attempt to parse JSON; if successful and it contains the expected keys
        # ("data", "columns", "index"), we treat it as the DataFrame payload.
        parsed = json.loads(candidate)
        if isinstance(parsed, dict) and {"data", "columns", "index"}.issubset(parsed.keys()):
            df = pd.read_json(json.dumps(parsed), orient="split")
    except Exception:
        # If parsing fails we simply keep df = None and return the raw response.
        pass

    # Return a tuple so the caller can decide what to do with the DataFrame.
    return response, df

# -----------------------------------------------------------------------------
# Example usage
# -----------------------------------------------------------------------------
if __name__ == "__main__":
    path = "excel_file.xlsx"
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."
    raw_response, df = run_agent(user_prompt)
    print("--- Agent raw response ---")
    print(raw_response)
    if df is not None:
        print("\n--- Reconstructed DataFrame ---")
        print(df)
    else:
        print("\nNo DataFrame could be reconstructed from the agent output.")
