# ---------------------------------------------------------------------------
# Tool definition

@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a JSON string that can be reconstituted
    into a pandas DataFrame.

    The agent can only return plain text, so we serialise the DataFrame as a
    JSON string using the ``split`` orientation, which preserves the column
    order, index and data values. The calling code can then deserialize the
    string back into a DataFrame.
    """
    # Load the Excel file – the caller controls the path.
    df = pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    # Keep only object (string) columns, mirroring the original behaviour.
    df = df.select_dtypes(include=["object"])
    # Serialise to JSON for transport via the LLM.
    return df.to_json(orient="split")

# ---------------------------------------------------------------------------
# Agent helper
# ---------------------------------------------------------------------------
def _extract_dataframe_from_response(response: dict) -> pd.DataFrame | None:
    """Attempt to extract a DataFrame from the agent's textual output.

    The ReAct agent returns a dictionary like ``{"output": "..."}``.  If the
    output looks like JSON produced by ``load_excel_response`` we deserialize it
    into a DataFrame and return it; otherwise we return ``None``.
    """
    output = response.get("output", "")
    # The LLM often wraps the JSON in back‑ticks or adds extra whitespace –
    # strip those characters before attempting to parse.
    candidate = output.strip().strip('`')
    try:
        # Verify that the string is valid JSON.
        _ = json.loads(candidate)
        # If valid, let pandas read it using the same orientation.
        return pd.read_json(candidate, orient="split")
    except Exception:
        # Not valid JSON – the agent likely returned a textual summary.
        return None

def run_agent(user_prompt: str, file_name: str = "data.json") -> pd.DataFrame | dict:
    """Run a ReAct agent that can load an Excel file and return the DataFrame.

    If the agent succeeds in invoking the ``load_excel_response`` tool, the
    function returns the reconstructed pandas DataFrame.  If the agent fails
    or produces a non‑JSON response, the raw response dictionary is returned so
    the caller can inspect the textual answer.
    """
    llm = ChatOpenAI(model="gpt-4o-mini")
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    response = agent_executor.invoke({"input": user_prompt})

    # Try to turn the textual output into a DataFrame.
    df = _extract_dataframe_from_response(response)
    if df is not None:
        return df
    # Fallback – return the full agent response for debugging.
    return response
