import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv

dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# Tool that loads an Excel file and returns a JSON representation of the DataFrame.
# Returning a JSON string is safe for LangChain's serialization layer while still
# allowing the REPL tool to reconstruct the original DataFrame.
# ---------------------------------------------------------------------------
@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a JSON representation of the transformed DataFrame.

    The ReAct agent will typically assign the result to a variable (e.g. ``df``) in
    the Python REPL tool.  By returning a JSON string we avoid serialization errors
    that occur when returning a raw ``DataFrame`` object.
    """
    df = (
        pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    )
    df = df.select_dtypes(include=["object"])
    # Convert to JSON so the tool output can be safely transmitted.
    return df.to_json(orient="records")


def run_agent(user_prompt: str, file_name: str = "data.json"):
    """Execute a ReAct agent and return the DataFrame created by the agent.

    The function returns a tuple ``(agent_output, df)`` where ``agent_output`` is the
    raw response from the ``AgentExecutor`` and ``df`` is the actual pandas DataFrame
    that the agent stored in the Python REPL's global namespace.  If the agent never
    created a variable named ``df``, ``df`` will be ``None``.
    """
    # Initialise the LLM and the REPL tool.
    llm = ChatOpenAI(model="gpt-4o-mini")
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    # Register the tools that the agent can use.
    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    # Build the ReAct agent.
    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # Run the agent.
    response = agent_executor.invoke({"input": user_prompt})

    # After execution the PythonREPL instance keeps its globals in the ``_globals``
    # attribute.  If the agent assigned the loaded data to a variable called ``df``
    # we can retrieve the actual DataFrame object here.
    df = python_repl._globals.get("df") if hasattr(python_repl, "_globals") else None

    return response, df


if __name__ == "__main__":
    path = "excel_file.xlsx"
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."
    agent_output, df = run_agent(user_prompt)
    print("Agent response:")
    print(agent_output)
    print("\nDataFrame retrieved from REPL globals:")
    print(df)
