import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv

dotenv.load_dotenv()

@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return its contents as a JSON string.

    The ReAct agent can later deserialize this JSON back into a pandas DataFrame
    inside the Python REPL environment.
    """
    df = (
        pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    )
    # Ensure only object columns are kept (as in the original code)
    df = df.select_dtypes(include=["object"])
    # Return a JSON representation that the REPL can easily read back
    return df.to_json(orient="records")

def run_agent(user_prompt: str, file_name: str = "data.json"):
    """Run a ReAct agent and return any DataFrame objects created in the REPL.

    The function executes the agent with three tools:
    1. ``load_excel_response`` – returns a JSON string of the Excel data.
    2. ``python_repl`` – a REPL that can import pandas and reconstruct the DataFrame.
    3. ``shell_tool`` – for any needed shell interaction.

    After the agent finishes, we inspect the Python REPL's global namespace for
    ``pandas.DataFrame`` instances (e.g., a variable named ``df``) and return them
    alongside the textual response.
    """
    # Initialise LLM and tools
    llm = ChatOpenAI(model="gpt-4o-mini")
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run arbitrary Python code in a stateful REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    # Build and run the ReAct agent
    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    response = agent_executor.invoke({"input": user_prompt})

    # ---------------------------------------------------------------------
    # Extract DataFrame objects from the REPL's globals.  The REPL stores its
    # namespace in the ``globals`` attribute (a dict).  We look for any values
    # that are instances of ``pd.DataFrame`` and collect them.
    # ---------------------------------------------------------------------
    dfs = {
        var_name: value
        for var_name, value in getattr(python_repl, "globals", {}).items()
        if isinstance(value, pd.DataFrame)
    }

    # If a DataFrame named ``df`` exists, return it directly; otherwise return
    # the full mapping of found DataFrames.
    result = {
        "agent_output": response,
        "dataframes": dfs,
    }
    return result

if __name__ == "__main__":
    path = "excel_file.xlsx"
    user_prompt = (
        f"Load the Excel file located at {path} using the provided tool, "
