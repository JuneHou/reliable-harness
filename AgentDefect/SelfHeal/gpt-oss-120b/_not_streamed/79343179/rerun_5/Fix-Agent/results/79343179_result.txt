import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv

dotenv.load_dotenv()

@tool
def load_excel_response(filepath: str):
    """Load an Excel file and return a transformed DataFrame.

    The tool returns the DataFrame object; the ReAct agent can then assign it
    to a variable inside the Python REPL (e.g., ``df``).  The function itself
    simply returns the DataFrame – LangChain will convert the return value to a
    string for the agent, but the actual object will also be available in the
    REPL's global namespace after the agent executes the corresponding Python
    code.
    """
    df = pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    df = df.select_dtypes(include=["object"])
    return df

def run_agent(user_prompt: str, file_name: str = "data.json"):
    """Run a ReAct agent that can load an Excel file and manipulate data.

    The function returns the ``df`` variable created by the agent (if any).
    If the variable is not created, the raw agent response is returned.
    """
    # Initialise the LLM and the Python REPL with a shared globals dict.
    llm = ChatOpenAI(model="gpt-4o-mini")
    shared_globals: dict = {}
    python_repl = PythonREPL(_globals=shared_globals)
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    # List of available tools for the agent.
    tools_list = [load_excel_response, repl_tool, shell_tool]

    # Use the standard ReAct prompt from the LangChain hub.
    prompt = hub.pull("hwchase17/react")

    # Build the agent and executor.
    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # Invoke the agent.
    response = agent_executor.invoke({"input": user_prompt})

    # If the agent stored a DataFrame in the REPL globals under the name "df",
    # return that object; otherwise return the textual response.
    df = shared_globals.get("df")
    return df if df is not None else response

if __name__ == "__main__":
    path = "excel_file.xlsx"
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."
    result = run_agent(user_prompt)
    # ``result`` will be a pandas DataFrame if the agent created one.
    if isinstance(result, pd.DataFrame):
        print("DataFrame retrieved from agent:")
        print(result)
    else:
        print("Agent textual response:")
        print(result)

@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a JSON representation of the transformed DataFrame.

    The LangChain tool system expects return values to be JSON‑serialisable.  We
    therefore convert the DataFrame to a JSON string (using the "split" orient so
    that it can be re‑hydrated with ``pd.read_json`` later).
    """
    df = (
        pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    )
    df = df.select_dtypes(include=["object"]).copy()
    # Return a JSON string that can be reconstructed into a DataFrame.
    return df.to_json(orient="split")

def run_agent(user_prompt: str, file_name: str = "data.json"):
    """Run the ReAct agent and return the DataFrame produced by ``load_excel_response``.

    The agent returns a dictionary whose ``output`` field contains the JSON string
    emitted by the ``load_excel_response`` tool.  We convert that JSON back into a
    ``pandas.DataFrame`` before returning it to the caller.
    """
    llm = ChatOpenAI(model="gpt-4o-mini")
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    response = agent_executor.invoke({"input": user_prompt})

    # The tool's JSON payload is stored under the "output" key of the response.
    json_payload = response.get("output") or response.get("result") or ""
    if not json_payload:
        raise ValueError("Agent did not return a DataFrame payload.")

        df = pd.read_json(json_payload, orient="split")
