from pathlib import Path
from langchain_text_splitters import CharacterTextSplitter


def run_agent(text_raw: str, output_dir: str = "chunks"):
    """Split the provided text into 30‑character chunks with a 5‑character overlap
    and write each chunk to a separate file in *output_dir*.

    Args:
        text_raw: The raw input string.
        output_dir: Directory where the chunk files will be stored.
    """
    # Normalise the input – strip leading/trailing whitespace and keep internal newlines.
    text_raw = text_raw.strip()

    # CharacterTextSplitter uses "\n\n" as the default separator, which can cause
    # chunk sizes to deviate from the requested *chunk_size*.  Setting ``separator=""``
    # forces a strict character‑wise split so each chunk is exactly ``chunk_size``
    # characters long (except possibly the final chunk) with the desired overlap.
    text_splitter = CharacterTextSplitter(chunk_size=30, chunk_overlap=5, separator="")
    chunks = text_splitter.split_text(text_raw)

    # Ensure the output directory exists.
    out_path = Path(output_dir)
    out_path.mkdir(parents=True, exist_ok=True)

    # Write each chunk to its own file.
    for i, chunk in enumerate(chunks):
        file_path = out_path / f"chunk_{i + 1}.txt"
        file_path.write_text(chunk, encoding="utf-8")

    return chunks
