import os
from langchain_text_splitters import CharacterTextSplitter


def run_agent(text_raw: str, output_dir: str = "chunks") -> list[str]:
    """Split *text_raw* into 30‑character chunks with a 5‑character overlap.

    The function also saves each chunk to a separate file inside ``output_dir``.
    Returns the list of chunks.
    """
    # Remove leading/trailing newlines and spaces that could affect chunk sizes
    cleaned_text = text_raw.strip()

    # ``separator=""`` forces the splitter to work on raw characters rather than
    # trying to split on whitespace or new‑lines, ensuring exact chunk sizes.
    text_splitter = CharacterTextSplitter(
        separator="",
        chunk_size=30,
        chunk_overlap=5,
    )

    chunks = text_splitter.split_text(cleaned_text)

    # Ensure the output directory exists
    os.makedirs(output_dir, exist_ok=True)

    # Save each chunk to a separate file for later use
    for idx, chunk in enumerate(chunks):
        filename = os.path.join(output_dir, f"chunk_{idx + 1}.txt")
        with open(filename, "w", encoding="utf-8") as f:
            f.write(chunk)

    return chunks
