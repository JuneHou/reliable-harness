    """Split *text_raw* into 30‑character chunks with a 5‑character overlap.

    The function also saves the resulting chunks to *output_path*, one chunk per line.
    """
    # Strip leading/trailing whitespace (including the leading newline in the
    # triple‑quoted string) so that the splitter works on the intended content.
    text = text_raw.strip()

    # Use an empty string as the separator so the splitter works on a pure

    text_splitter = CharacterTextSplitter(separator="", chunk_size=30, chunk_overlap=5)
    chunks = text_splitter.split_text(text)

    # Persist the chunks locally.
    with open(output_path, "w", encoding="utf-8") as f:
