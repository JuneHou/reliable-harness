"""
Corrected script that creates a PostgreSQL pgvector table with a custom column
`external_id` (as a plain TEXT column) instead of storing it inside the JSON
metadata column.  The `Column` class from ``llama_index.vector_stores.postgres``
is used to describe the extra column, and ``ignore_metadata_columns`` tells the
store not to include ``external_id`` in the JSON metadata field.
"""

from llama_index.core import VectorStoreIndex, Document
from llama_index.vector_stores.postgres import PGVectorStore, Column
import os
from dotenv import load_dotenv

load_dotenv()


def create_vector_store(request):
    """Create a PGVectorStore with a custom ``external_id`` column.

    Args:
        request: An object with a ``tenant`` attribute specifying the target
            PostgreSQL database name.
    """
    host = "localhost"
    password = "password"
    port = 5432
    user = "postgres"

    # Define a custom column that will hold the external identifier.
    external_id_column = Column(name="external_id", type_="TEXT")

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name="vectorstore",
        embed_dim=1536,
        # Standard HNSW settings
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
        # Tell the store to create the extra column and *not* put it into the
        # JSON metadata column.
        metadata_columns=[external_id_column],
        ignore_metadata_columns=["external_id"],
    )
    return vector_store


def run_agent(request):
    """Index a couple of sample documents using the custom vector store."""
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )
    return index


if __name__ == "__main__":
    class Request:
        tenant = "test_db"

    request = Request()
    index = run_agent(request)
    print("Vector store created and documents indexed with custom 'external_id' column.")
