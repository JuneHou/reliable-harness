from llama_index.core import VectorStoreIndex, Document
from llama_index.vector_stores.postgres import PGVectorStore, Column
import os
from dotenv import load_dotenv

load_dotenv()

def create_vector_store(request):
    """Create a PGVectorStore where ``external_id`` is stored in its own column.
    
    The ``metadata_columns`` argument tells ``PGVectorStore`` to create extra
    (non‑JSON) columns.  By declaring a ``Column`` named ``external_id`` the
    value stored in the ``metadata`` dict under the same key will be written to
    that column instead of being nested inside the ``metadata`` JSON column.
    """
    host = "localhost"
    password = "password"
    port = 5432
    user = "postgres"

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name="vectorstore",
        embed_dim=1536,
        # HNSW index configuration
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
        # Declare a custom column for the external identifier.
        # The column type must be a valid PostgreSQL type (e.g., TEXT).
        metadata_columns=[Column(name="external_id", type="TEXT")],
        # Keep the regular JSON column for any other metadata.
        metadata_json_column="metadata",
    )
    return vector_store

def run_agent(request):

    ]

    index = VectorStoreIndex.from_documents(

    )
    return index
