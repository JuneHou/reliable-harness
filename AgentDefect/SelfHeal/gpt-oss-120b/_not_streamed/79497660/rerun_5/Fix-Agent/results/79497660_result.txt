def create_vector_store(request):
    """Create a PGVectorStore that stores an ``external_id`` column separately.

    The default PGVectorStore stores all metadata inside a JSONB column called
    ``metadata``.  By specifying ``metadata_columns`` we can tell the store to map
    specific metadata keys to their own typed columns – in this case a
    ``VARCHAR`` column called ``external_id``.  Setting ``metadata_json_column``
    to ``False`` disables the fallback JSON column, ensuring the data lives only
    in the dedicated column.
    """
    host = "localhost"
    password = "password"
    port = 5432
    user = "postgres"

    # Define a typed column for the external_id field
    external_id_col = Column(name="external_id", dtype="VARCHAR")

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name="vectorstore",
        embed_dim=1536,
        # HNSW index parameters (optional but kept from original code)
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
        # -----------------------------------------------------------------
        # Custom schema: add a dedicated column for the external_id metadata.
        # -----------------------------------------------------------------
        metadata_columns=[external_id_col],
        # Disable the generic JSON metadata column – we only need the typed one.
