from llama_index.core import VectorStoreIndex, Document
from llama_index.vector_stores.postgres import PGVectorStore, Column

# Load environment variables (e.g., DB credentials) if you keep them in a .env file
load_dotenv()


def create_vector_store(request):
    """Create a PGVectorStore with a custom column for ``external_id``.

    The default PGVectorStore stores all metadata in a JSON column named
    ``metadata``.  By supplying a ``metadata_columns`` list we can tell the store
    to write selected metadata fields into dedicated, non‑JSON columns.  In this
    example we want the ``external_id`` value to be persisted in its own column
    called ``external_id`` (type ``TEXT``).
    """

    host = os.getenv("PGHOST", "localhost")
    password = os.getenv("PGPASSWORD", "password")
    port = int(os.getenv("PGPORT", "5432"))
    user = os.getenv("PGUSER", "postgres")

    # Define the extra column that will hold the external identifier.
    # ``Column`` takes the column name and the PostgreSQL type as string.
    extra_metadata_columns = [Column(name="external_id", pg_type="TEXT")]

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name="vectorstore",
        embed_dim=1536,
        # HNSW parameters – keep the same configuration as before
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
        # Tell PGVectorStore to store ``external_id`` in a dedicated column
        metadata_columns=extra_metadata_columns,
        # Disable the generic JSON metadata column – we only need the custom one
        store_metadata=False,
        metadata_json_column=None,
    )

    return vector_store


def run_agent(request):
    """Create the vector store and index a couple of sample documents.

    ]

    index = VectorStoreIndex.from_documents(

        vector_store=vector_store,
    )

    return index
