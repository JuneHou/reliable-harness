from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv


def run_agent(query: str):
    """Run a simple retrieval demo.

    LlamaIndex does not store metadata fields whose value is ``None`` – the key is simply omitted.
    Consequently, a filter like ``start_date == None`` will never match anything.

    To make "null" values searchable we store a sentinel string (e.g., "<NULL>") and filter on that
    sentinel. This keeps the metadata JSON‑serialisable and works across all vector store backends.
    """
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Use a sentinel for "null" metadata values; ``None`` would be dropped silently.
    NULL_SENTINEL = "<NULL>"

    # Create node with a "null" start_date
    node_01 = TextNode(
        text="This document has a null start_date in its metadata",
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},
    )

    # Create the index
    doc_index = VectorStoreIndex([node_01])

    # Debug: show stored metadata
    print("Index nodes metadata:")
    for nid, node in doc_index.docstore.docs.items():
        print(f"  {nid}: {node.metadata}")
    print()

    # Build a filter that matches the sentinel value
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])

    # Retrieve
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    retrieved = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    for r in retrieved:
        print(f"  node_id={r.node_id}, metadata={r.metadata}")
    print()

    return retrieved


if __name__ == "__main__":

    response = run_agent(query)
    if not response:
        print("\nResult: No nodes retrieved (null sentinel not matched)")
    else:
        print(f"\nResult: Found {len(response)} node(s)")

    """
    return {k: (NULL_SENTINEL if v is None else v) for k, v in metadata.items()}
