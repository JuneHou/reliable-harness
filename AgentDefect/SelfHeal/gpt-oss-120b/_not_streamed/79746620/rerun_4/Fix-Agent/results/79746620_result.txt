from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# Helper to convert ``None`` metadata values into a sentinel string.
# LlamaIndex's filter engine does not support equality checks against ``None``
# because the underlying vector store (e.g., Pinecone, Milvus, FAISS, etc.)
# cannot represent a ``null`` value in its metadata schema.  By converting the
# ``None`` into a known sentinel we can both store the information and later
# filter on it.
NULL_SENTINEL = "<NULL>"

def normalize_metadata(metadata: dict) -> dict:
    """Replace ``None`` values with ``NULL_SENTINEL``.

    Args:
        metadata: Original metadata dictionary.

    Returns:
        A new dictionary with ``None`` values replaced.
    """
    return {k: (NULL_SENTINEL if v is None else v) for k, v in metadata.items()}


def run_agent(query: str):
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node with None in metadata (converted to sentinel)
    raw_metadata = {"start_date": None}
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=normalize_metadata(raw_metadata),
    )

    # Create index
    doc_index = VectorStoreIndex([node_01])

    # Debug: Check what's actually stored
    print("Index nodes (stored metadata):")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter by the sentinel value representing None
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes


if __name__ == "__main__":
