from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv


def run_agent(query: str):
    """Run a simple retrieval example demonstrating how to filter on "null"
    metadata values.

    LlamaIndex does not support filtering on Python ``None`` because most
    vector‑store back‑ends cannot store a true null value.  The typical workaround
    is to store a sentinel value (e.g., the string ``"null"``) and filter on that
    sentinel.
    """
    # Load environment variables from .env file
    load_dotenv()

    # Configure the embedding model (requires ``OPENAI_API_KEY`` in the .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create a node whose ``start_date`` is intentionally set to a sentinel that
    # represents "null".  Using the literal ``None`` would be dropped by the
    # vector‑store and consequently could not be filtered on.
    SENTINEL_NULL = "null"  # any unique string works
    node_01 = TextNode(
        text="This document has a null‑like value in the metadata",
        id_="node_01",
        metadata={"start_date": SENTINEL_NULL},
    )

    # Build the index from the node list
    doc_index = VectorStoreIndex([node_01])

    # Debug: show what is actually stored in the docstore
    print("Index nodes metadata:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Construct a filter that matches the sentinel value
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=SENTINEL_NULL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])

    # Retrieve using the filter
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show the retrieved nodes
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes

# ---------------------------------------------------------------------------
# NOTE:
# LlamaIndex does **not** support filtering on ``None`` (JSON ``null``) values.
# When metadata is serialized to JSON (the format used by most vector stores),
# fields with a ``None`` value are omitted.  Consequently a filter like
# ``MetadataFilter(key="start_date", operator=EQ, value=None)`` never matches
# anything – the key simply does not exist in the stored metadata.
#
# The common workaround is to store a sentinel value (e.g. a special string)
# that represents “null”.  The sentinel is then used both when inserting the
# node and when constructing the filter.  This example uses ``"__NULL__"`` as
# the sentinel, but any unique value that will not clash with real data works.
# ---------------------------------------------------------------------------

# Sentinel value used to represent a missing/NULL metadata field
NULL_SENTINEL = "__NULL__"


def _null_safe_metadata(metadata: dict) -> dict:
    """Replace ``None`` values with ``NULL_SENTINEL`` so they survive JSON
    serialisation and can be filtered on.
    """
    return {k: (NULL_SENTINEL if v is None else v) for k, v in metadata.items()}


def run_agent(query: str):

        text="This document has None (represented by a sentinel) in the metadata",
        id_="node_01",
        metadata=safe_metadata,

    # Build the index
