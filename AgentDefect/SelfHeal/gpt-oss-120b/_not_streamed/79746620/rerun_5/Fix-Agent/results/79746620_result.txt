from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# Sentinel value to represent a "null" metadata entry
NULL_SENTINEL = "__NULL__"


def run_agent(query: str):
    """Create an index with a node that has a "null" metadata field and
    retrieve it using a filter.

    LlamaIndex stores metadata as JSON – ``None`` values are omitted during
    serialization, which means they cannot be used in a ``MetadataFilter``.
    To work around this we store a sentinel string (``NULL_SENTINEL``) and
    filter on that value.
    """
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create node with a sentinel representing a null start_date
    node_01 = TextNode(
        text="This document has a placeholder for a null start_date",
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},
    )

    # Create the index from the node list
    doc_index = VectorStoreIndex([node_01])

    # Debug: show what is stored in the docstore
    print("Index nodes metadata:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Build a filter that matches the sentinel value
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])

    # Retrieve with the filter (similarity_top_k=1 for brevity)
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes
