from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv


def run_agent(query: str):
    """Run a simple retrieval demo showing how to filter on "null" metadata.

    LlamaIndex does **not** support filtering on Python ``None`` values – the
    underlying vector store drops ``None`` during serialization, so the field is
    absent when the filter is evaluated.  To simulate a *null* value we store a
    sentinel (an empty string) and use the ``IS_EMPTY`` operator which checks for
    empty‑or‑missing values.
    """
    # Load environment variables (e.g. ``OPENAI_API_KEY``)
    load_dotenv()

    # Configure the embedding model – this uses the key from the ``.env`` file.
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # ---------------------------------------------------------------------
    # 1️⃣ Create a node where the ``start_date`` metadata is intentionally "null".
    #    We store an empty string as a sentinel because ``None`` cannot be
    #    persisted by the vector store.
    # ---------------------------------------------------------------------
    node_01 = TextNode(
        text="This document has a null start_date in its metadata",
        id_="node_01",
        metadata={"start_date": ""},  # empty string = null sentinel
    )

    # Build the index from the single node.
    doc_index = VectorStoreIndex([node_01])

    # Debug: Show what actually ended up in the docstore.
    print("Index nodes (raw metadata):")
    for nid, node in doc_index.docstore.docs.items():
        print(f"{nid}: {node.metadata}")
    print()

    # ---------------------------------------------------------------------
    # 2️⃣ Filter for the "null" sentinel.
    #    ``FilterOperator.IS_EMPTY`` matches fields that are empty *or* missing.
    # ---------------------------------------------------------------------
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.IS_EMPTY,
        value="",  # value is ignored for IS_EMPTY but required by the ctor
    )
    filters = MetadataFilters(filters=[filter_null_start_date])

    # Retrieve the most similar node that also satisfies the filter.
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show the retrieval result.
    print("Retrieved nodes:")
    for node in nodes:
        print(f"{node.node_id} -> {node.metadata}")
    print()

    return nodes


if __name__ == "__main__":
    query = "this"
    print(f"Query: '{query}'")
    print("-" * 50)
    response = run_agent(query)
    if not response:
        print("Result: No nodes retrieved (filter did not match any document)")
    else:
        print(f"Result: Found {len(response)} node(s)")
