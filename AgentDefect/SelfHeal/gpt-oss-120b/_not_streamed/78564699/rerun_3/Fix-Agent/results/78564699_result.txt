import json
import os
from pathlib import Path

from huggingface_hub import hf_hub_download
from llama_index.llms.llama_cpp import LlamaCPP


def get_model_path() -> str:
    """Return a local path to the Llama-2-13B chat GGUF model.

    The function first checks for a user‑provided mapping in ``paths.json``.
    If the mapping is missing, the model is downloaded from the HuggingFace
    repository and the local path to the downloaded file is returned.
    """
    paths_file = Path("./paths.json")
    if paths_file.is_file():
        try:
            with paths_file.open("r", encoding="utf-8") as f:
                paths = json.load(f)
                if "llama-2-13b-chat" in paths:
                    return paths["llama-2-13b-chat"]
        except Exception as e:
            # If the JSON is malformed we fall back to downloading the model.
            print(f"Warning: could not read paths.json ({e}), downloading model instead.")

    # Download the GGUF file from HuggingFace (cached on disk).
    # This call blocks until the file is fully downloaded, after which the
    # local cache path is returned. Using ``repo_id`` and ``filename`` ensures
    # we get the exact file we expect.
    model_path = hf_hub_download(
        repo_id="TheBloke/Llama-2-13B-chat-GGUF",
        filename="llama-2-13b-chat.Q4_0.gguf",
        # The repo is large; we keep the default cache directory.
    )
    return model_path


def run_agent(query: str) -> str:
    """Run a simple LlamaCPP inference on the provided *query*.

    The function creates a ``LlamaCPP`` instance configured for CPU‑only
    inference (``n_gpu_layers=0``). ``max_new_tokens`` and ``temperature`` can be
    tweaked as needed.
    """
    model_path = get_model_path()

    llm = LlamaCPP(
        model_path=model_path,
        # ``model_url`` is omitted because we operate on a local file.
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={"n_gpu_layers": 0, "n_threads": max(1, os.cpu_count() - 1)},
        verbose=True,
    )

    # ``complete`` returns a ``CompletionResponse``; we extract the generated text.
    response = llm.complete(query)
    # ``response`` may be a string or a response object depending on the version.
    # ``response.text`` is a safe way to get the content if it exists.
    return getattr(response, "text", response)


if __name__ == "__main__":
    query = "What is the capital of France?"
    answer = run_agent(query)
    print(answer)
