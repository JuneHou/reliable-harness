import os
import json
import requests
from tqdm import tqdm
from llama_index.llms.llama_cpp import LlamaCPP

# -----------------------------------------------------------------------------
# Helper to download the GGUF model if it is not already present locally.
# This avoids the "hang" that occurs when LlamaCPP is given only a remote URL
# for a large model – the library will try to download the file synchronously
# during initialisation and appears to block forever on a CPU‑only machine.
# -----------------------------------------------------------------------------

def download_model(url: str, dst_path: str) -> None:
    """Download a file with a progress bar.

    The function streams the HTTP response to avoid loading the whole model
    into memory.  It is safe to call repeatedly – if the file already exists the
    download is skipped.
    """
    if os.path.exists(dst_path):
        return
    response = requests.get(url, stream=True, timeout=30)
    response.raise_for_status()
    total = int(response.headers.get("content-length", 0))
    os.makedirs(os.path.dirname(dst_path), exist_ok=True)
    with open(dst_path, "wb") as f, tqdm(
        total=total,
        unit="iB",
        unit_scale=True,
        unit_divisor=1024,
        desc=os.path.basename(dst_path),
    ) as bar:
        for chunk in response.iter_content(chunk_size=8192):
            if chunk:  # filter out keep‑alive chunks
                f.write(chunk)
                bar.update(len(chunk))


def get_model_path() -> str:
    """Return a local path to the GGUF model, downloading it if necessary.

    The function first checks a ``paths.json`` file (used by the original code)
    for a user‑provided location.  If that file does not contain a valid entry,
    it falls back to downloading the model into ``./models``.
    """
    # 1️⃣ Check for a user‑supplied path in ``paths.json``
    paths_file = "./paths.json"
    if os.path.isfile(paths_file):
        with open(paths_file, "r", encoding="utf-8") as f:
            try:
                paths = json.load(f)
                user_path = paths.get("llama-2-13b-chat")
                if user_path and os.path.isfile(user_path):
                    return user_path
            except json.JSONDecodeError:
                pass  # ignore malformed JSON – we will download instead

    # 2️⃣ Otherwise download the model to a deterministic location
    model_url = (
        "https://huggingface.co/TheBloke/Llama-2-13B-chat-GGUF/resolve/main/"
        "llama-2-13b-chat.Q4_0.gguf"
    )
    model_dir = "./models"
    os.makedirs(model_dir, exist_ok=True)
    model_path = os.path.join(model_dir, "llama-2-13b-chat.Q4_0.gguf")
    download_model(model_url, model_path)
    return model_path


def run_agent(query: str) -> str:
    """Run a single LLaMA‑2‑13B‑chat inference on CPU.

    The previous implementation passed ``model_url`` which forces LlamaCPP to
    download the model lazily during initialisation – on a CPU‑only setup this
    can appear as a hang.  By supplying a concrete ``model_path`` and limiting the
    number of threads we obtain deterministic, fast inference.
    """
    model_path = get_model_path()

    llm = LlamaCPP(
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        # ``n_gpu_layers`` is 0 for pure‑CPU, ``n_threads`` limits CPU usage.
        model_kwargs={"n_gpu_layers": 0, "n_threads": 8},
        verbose=False,
    )

    # ``complete`` returns a ``CompletionResponse``; the actual generated text
    # lives in the ``text`` attribute.
    response = llm.complete(query)
    if hasattr(response, "text"):
        return response.text
    return str(response)


if __name__ == "__main__":
    print(run_agent("What is the capital of France?"))
