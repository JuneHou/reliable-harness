import json
import os
from pathlib import Path

from huggingface_hub import hf_hub_download
from llama_index.llms.llama_cpp import LlamaCPP


def get_model_path() -> str:
    """Return a local path to the Llama‑2‑13B‑chat model.
    The function first checks a user‑provided ``paths.json`` for a custom location.
    If no entry is found, it downloads the model from HuggingFace into a cache
    directory (``~/.cache/llama_models``) and returns that path.
    """
    # Default location for custom overrides
    paths_file = Path("./paths.json")
    if paths_file.is_file():
        with paths_file.open("r", encoding="utf-8") as f:
            paths = json.load(f)
            if "llama-2-13b-chat" in paths:
                return paths["llama-2-13b-chat"]

    # Fallback: download the model from HF Hub (cached for future runs)
    repo_id = "TheBloke/Llama-2-13B-chat-GGUF"
    filename = "llama-2-13b-chat.Q4_0.gguf"
    cache_dir = Path.home() / ".cache" / "llama_models"
    cache_dir.mkdir(parents=True, exist_ok=True)
    return hf_hub_download(
        repo_id=repo_id,
        filename=filename,
        cache_dir=str(cache_dir),
        resume_download=True,
    )


def run_agent(query: str) -> str:
    """Run a simple inference using LlamaCPP.

    The function loads the model onto the CPU (no GPU layers) and returns the
    generated text for the supplied *query*.
    """
    model_path = get_model_path()

    llm = LlamaCPP(
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        # Ensure we stay on CPU – ``n_gpu_layers=0`` – and use a reasonable
        # number of threads for faster inference.
        model_kwargs={"n_gpu_layers": 0, "n_threads": 8},
        verbose=True,
    )

    # ``complete`` returns a ``CompletionResponse``; the generated text is in
    # the ``text`` attribute.
    response = llm.complete(query)
    return response.text


if __name__ == "__main__":
    query = "What is the capital of France?"
    answer = run_agent(query)
    print(answer)
