import os
import json
from pathlib import Path
from huggingface_hub import snapshot_download
from llama_index.llms.llama_cpp import LlamaCPP


def get_model_path():
    """Download the GGUF model if not already present and return its local path.
    This avoids using the `model_url` argument which can cause the library to
    attempt a streamed download and hang on some systems.
    """
    # Resolve custom path mapping if a ``paths.json`` file is present
    paths_file = Path("./paths.json")
    if paths_file.is_file():
        with paths_file.open("r", encoding="utf-8") as f:
            paths = json.load(f)
        if "llama-2-13b-chat" in paths:
            return Path(paths["llama-2-13b-chat"]).expanduser().resolve()

    # Default location inside a ``models`` directory next to this script
    model_dir = Path(__file__).parent / "models"
    model_dir.mkdir(parents=True, exist_ok=True)
    model_name = "TheBloke/Llama-2-13B-chat-GGUF"
    file_name = "llama-2-13b-chat.Q4_0.gguf"
    # ``snapshot_download`` will only download the file once and then reuse it
    local_path = snapshot_download(
        repo_id=model_name,
        revision="main",
        cache_dir=str(model_dir),
        allow_patterns=[file_name],
        local_files_only=False,
    )
    return Path(local_path).joinpath(file_name)


def run_agent(query: str) -> str:
    """Run a simple inference query using LlamaCPP on CPU.

    The function returns the raw generated text, not the full response object.
    """
    model_path = get_model_path()

    # Initialise LlamaCPP – we explicitly set ``n_gpu_layers`` to 0 for pure‑CPU
    # inference and limit the thread count to a sensible default (e.g. number of
    # physical cores minus one). Adjust ``n_threads`` according to your hardware.
    llm = LlamaCPP(
        model_path=str(model_path),
        model_url=None,          # avoid the internal downloader which can hang
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={
            "n_gpu_layers": 0,
            "n_threads": max(1, os.cpu_count() - 1),
        },
        verbose=True,
    )

    # ``complete`` returns a ``CompletionResponse``; extract the text attribute.
    response = llm.complete(query)
    return response.text


if __name__ == "__main__":
    query = "What is the capital of France?"
    answer = run_agent(query)
    print(answer)

from llama_index.llms.llama_cpp import LlamaCPP

def _ensure_model(model_url: str, cache_dir: str = "~/.cache/llama_cpp") -> str:
    """Download the GGUF model if it is not already present.

    Args:
        model_url: Direct URL to the GGUF file.
        cache_dir: Directory where the model will be stored.

    Returns:
        The absolute path to the downloaded model file.
    """
    cache_dir = os.path.expanduser(cache_dir)
    os.makedirs(cache_dir, exist_ok=True)
    filename = Path(model_url).name
    model_path = os.path.join(cache_dir, filename)
    if not os.path.isfile(model_path):
        print(f"Downloading model to {model_path} …")
        urllib.request.urlretrieve(model_url, model_path)
        print("Download completed.")
    return model_path
