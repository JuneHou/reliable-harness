from langchain_community.chat_models import ChatOllama
from langchain.schema import SystemMessage, HumanMessage

def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Generate a single, clean assistant reply using Ollama's Llama 3.

    Args:
        user_prompt: The question or instruction from the user.
        system_prompt: The system‑level instruction (role definition).

    Returns:
        The assistant's response as plain text.
    """
    # Initialise the chat model. ``ChatOllama`` handles the token formatting and
    # stop‑token logic internally, ensuring only the assistant's content is
    # returned without the surrounding header tags.
    chat = ChatOllama(model="llama3")

    # Build the message list using LangChain's schema – this mirrors the
    # typical ChatML layout that Llama 3 expects.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # ``invoke`` returns a ``ChatMessage`` (or similar) object; the actual text
    # is stored in its ``content`` attribute.
    response = chat.invoke(messages)
    return response.content
