from langchain_community.chat_models import ChatOllama
from langchain.schema import SystemMessage, HumanMessage


def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Generate a single, clean response from the Llama 3 model via Ollama.

    Args:
        user_prompt: The user's question or instruction.
        system_prompt: The system instruction that sets the behavior of the assistant.

    Returns:
        The assistant's answer as a plain string without extra tokens or duplicated
        content.
    """
    # Initialise the chat model. ``stop`` tells the model to stop generating once it
    # outputs the end‑of‑turn token used by Llama 3, preventing trailing or duplicated
    # text.
    llm = ChatOllama(model="llama3", stop=["<|eot_id|>"])

    # Build the chat history using the proper schema. LangChain will format the
    # messages with the required <|start_header_id|> / <|eot_id|> markers internally.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # Invoke the model and extract the content of the generated assistant message.
    response = llm.invoke(messages)
    # ``response`` is an ``AIMessage``; its ``content`` attribute holds the clean text.
    return response.content
