from langchain_community.chat_models import ChatOllama
from langchain_core.messages import SystemMessage, HumanMessage


def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Generate a single, clean response from Ollama's Llama 3 model.

    Args:
        user_prompt: The query or instruction from the user.
        system_prompt: The system message that sets the behavior of the assistant.

    Returns:
        The assistant's response as a plain string.
    """
    # Initialise the chat model – Ollama will handle the chat‑style formatting internally.
    chat = ChatOllama(model="llama3")

    # Build the message list using LangChain's message types.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # Invoke the model and extract the textual content.
    ai_message = chat.invoke(messages)
    return ai_message.content


if __name__ == "__main__":

    )
    system_prompt = "You are a helpful assistant."
    print(run_agent(user_prompt, system_prompt))
