from langchain_community.chat_models import ChatOllama
from langchain.schema import HumanMessage, SystemMessage


def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Send a single user prompt to Ollama's Llama 3 model and return a clean assistant response.

    This implementation uses ``ChatOllama`` which handles the required
    system/user/assistant message formatting internally, ensuring that only one
    assistant reply is produced and that the model knows where to stop.
    """
    # Initialise the chat model. ``ChatOllama`` automatically uses the ChatML
    # style of messages expected by Llama 3.
    chat = ChatOllama(model="llama3")

    # Build the message sequence – first the system prompt, then the user prompt.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # Invoke the model and extract the textual content from the returned AIMessage.
    ai_message = chat.invoke(messages)
    return ai_message.content
