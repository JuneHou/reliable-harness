from langchain_community.chat_models import ChatOllama
from langchain.schema import SystemMessage, HumanMessage


def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Send a single, clean response from Ollama's Llama 3 model.

    The original implementation manually built a prompt with special
    ``<|...|>`` tokens, which caused the model to emit the ``assistant`` header
    and potentially duplicate or continue output.  Using LangChain's
    ``ChatOllama`` wrapper lets the library handle the chat‑formatting and
    stop‑token logic, guaranteeing a single assistant reply.
    """
    # Initialise the chat model.  ``ChatOllama`` knows how to format messages
    # for Ollama's chat endpoint and will automatically stop after the assistant
    # message is complete.
    chat = ChatOllama(model="llama3")

    # Build the message list using LangChain's schema objects.  The system
    # prompt sets the assistant’s behavior, while the human message contains the
    # user query.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # ``invoke`` returns a ``ChatResult`` where ``content`` holds the assistant
    # reply.  We return the plain string to the caller.
    result = chat.invoke(messages)
    return result.content


if __name__ == "__main__":

    system_prompt = "You are a helpful assistant."
