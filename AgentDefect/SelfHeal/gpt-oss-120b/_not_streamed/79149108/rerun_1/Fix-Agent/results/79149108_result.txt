from flask import Flask, render_template, request
from langchain_ollama import ChatOllama
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
import tiktoken

app = Flask(__name__)

# -------------------------------------------------------------
# Configuration
# -------------------------------------------------------------
MODEL = "llama3.2"
# Max tokens that the model can accept (adjusted for the chosen model)
MODEL_MAX_TOKENS = 128_000

# Store the conversation as LangChain message objects
# We start with an optional system prompt – you can customise it as needed.
messages = []

# Initialise the Ollama chat wrapper. ``stream=False`` because we want the
# complete response returned synchronously; if you need streaming you would
# have to iterate over the generator and concatenate the parts yourself.
llm = ChatOllama(model=MODEL, temperature=1.1, stream=False)

# -------------------------------------------------------------
# Helper – token counter
# -------------------------------------------------------------
def get_number_of_tokens(msgs):
    """Return an approximate token count for a list of LangChain messages.
    LangChain messages expose a ``content`` attribute that contains the text.
    For simplicity we reuse the ``tiktoken`` encoder for a comparable model.
    If the encoder cannot be loaded we fall back to a rough estimate of
    4 characters per token.
    """
    try:
        # ``gpt-3.5-turbo`` is a good proxy for many LLaMA‑style models.
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total = 0
        for m in msgs:
            total += len(encoding.encode(str(m.content)))
        return total
    except Exception:
        # Very rough fallback – 1 token ≈ 4 characters.
        total_chars = sum(len(str(m.content)) for m in msgs)
        return total_chars // 4

# -------------------------------------------------------------
# Routes
# -------------------------------------------------------------
@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    # Append the new user message to the conversation history.
    user_text = request.form['text']
    messages.append(HumanMessage(content=user_text))

    # Ensure we stay under the model token limit.  We keep removing the
    # *oldest* non‑system messages until the token count fits.
    while get_number_of_tokens(messages) > MODEL_MAX_TOKENS:
        # Preserve a possible leading SystemMessage (index 0).
        # If the first message is a SystemMessage we never delete it.
        if isinstance(messages[0], SystemMessage) and len(messages) > 1:
            del messages[1]
        else:
            del messages[0]

    # Query the model with the current message list.
    ai_response = llm.invoke(messages)

    # ``invoke`` returns a ``BaseMessage`` (AIMessage).  Store it for context.
    if isinstance(ai_response, AIMessage):
        messages.append(ai_response)
        reply = ai_response.content
    else:
        # In the unlikely event Ollama returns a plain string.
        reply = str(ai_response)
        messages.append(AIMessage(content=reply))

    return reply

@app.route('/')
def index():
    return render_template('index.html')

if __name__ == '__main__':
    # Turn off Flask's reloader when debugging to avoid duplicate
    # initialisation of the LLM which can cause odd behaviour.
    app.run(debug=True, port=5000, use_reloader=False)
