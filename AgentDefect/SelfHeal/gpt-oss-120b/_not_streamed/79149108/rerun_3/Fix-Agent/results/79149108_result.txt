from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128_000
# Keep a list of LangChain Message objects (HumanMessage / AIMessage)
conversation = []

# Initialise the Ollama chat model – streaming disabled for simplicity
llm = ChatOllama(
    model=MODEL,
    temperature=1.1,
    stream=False,
)

def _num_tokens_from_messages(messages: list) -> int:
    """Return an approximate token count for a list of LangChain messages.
    Uses tiktoken's GPT‑3.5 tokenizer (the best‑available approximation for
    most Llama models). If the encoder cannot be loaded we fall back to a
    simple character‑based estimate.
    """
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total = 0
        for msg in messages:
            # Both HumanMessage and AIMessage expose a ``content`` attribute
            total += len(encoding.encode(str(msg.content)))
        return total
    except Exception:
        # Rough estimate: 1 token ≈ 4 characters
        total_chars = sum(len(str(msg.content)) for msg in messages)
        return total_chars // 4

@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    # Append the new user message to the conversation
    user_msg = HumanMessage(content=request.form['text'])
    conversation.append(user_msg)

    # Ensure the conversation stays within the model's context window
    while _num_tokens_from_messages(conversation) > MODEL_MAX_TOKENS:
        # Remove the oldest *pair* (user + assistant) if possible; otherwise just the first message
        if len(conversation) > 2:
            del conversation[0:2]
        else:
            del conversation[0]

    # Generate the assistant response
    ai_msg: AIMessage = llm.invoke(conversation)
    # Store the assistant reply so that subsequent calls have context
    conversation.append(ai_msg)

    return ai_msg.content

@app.route('/')
def index():
    return render_template('index.html')

if __name__ == '__main__':
    app.run(debug=True, port=5000)
