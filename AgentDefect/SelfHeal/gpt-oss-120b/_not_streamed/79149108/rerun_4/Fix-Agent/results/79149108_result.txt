from flask import Flask, render_template, request, jsonify
from langchain_core.messages import HumanMessage, AIMessage, BaseMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
# Approximate token limit for the model (adjust if needed)
MODEL_MAX_TOKENS = 128_000

# Conversation history – a list of LangChain Message objects
messages: list[BaseMessage] = []

# Initialise the Ollama chat model
llm = ChatOllama(
    model=MODEL,
    temperature=1.1,
    stream=False,  # streaming not needed for Flask response
)

def _num_tokens_from_messages(msgs: list[BaseMessage]) -> int:
    """Return an estimated token count for a list of LangChain messages.
    The function uses the same encoding that OpenAI models use – this is a
    reasonable proxy for most LLMs and keeps the implementation simple.
    """
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
    except Exception:
        # Fallback – rough character based estimate
        return sum(len(str(m.content)) for m in msgs) // 4
    total = 0
    for m in msgs:
        total += len(encoding.encode(str(m.content)))
    return total

def _trim_messages(msgs: list[BaseMessage]):
    """Trim the oldest user/assistant pairs until we are under the token limit.
    The first system/initial prompt is kept (index 0) while older turns are
    discarded from the front of the list.
    """
    while _num_tokens_from_messages(msgs) > MODEL_MAX_TOKENS and len(msgs) > 2:
        # Remove the earliest *pair* (user + assistant) to preserve context.
        # Index 0 may be a system prompt – keep it.
        del msgs[1:3]
    return msgs

@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    # Append the newest user message
    user_text = request.form.get('text', '')
    messages.append(HumanMessage(content=user_text))

    # Ensure the conversation fits within the token budget
    _trim_messages(messages)

    # Get the model's reply
    response = llm.invoke(messages)  # returns an AIMessage
    # Store assistant reply for future turns
    messages.append(AIMessage(content=response.content))

    # Return just the text content (JSON makes it easier for the front‑end)
    return jsonify({"reply": response.content})

@app.route('/')
def index():
    return render_template('index.html')

if __name__ == '__main__':
    # Debug mode is useful during development but should be disabled in prod.
    app.run(debug=True, port=5000)
