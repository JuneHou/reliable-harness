from flask import Flask, render_template, request
from langchain_core.messages import AIMessage, HumanMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128_000
# Store chat history as LangChain Message objects
messages = []

llm = ChatOllama(
    model=MODEL,
    temperature=1.1,
    stream=False,  # stream=False simplifies response handling for Flask
)

def get_number_of_tokens(msgs):
    """Estimate token usage for a list of LangChain Message objects.
    Falls back to a rough character‑based estimate if the model tokenizer is not
    available.
    """
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total = 0
        for m in msgs:
            total += len(encoding.encode(str(m.content)))
        return total
    except Exception:
        # Approximate 4 characters per token
        total_chars = sum(len(str(m.content)) for m in msgs)
        return total_chars // 4

@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    # Append the new user message to the history
    user_msg = HumanMessage(content=request.form["text"])
    messages.append(user_msg)

    # Trim history if it exceeds the token limit
    while get_number_of_tokens(messages) > MODEL_MAX_TOKENS:
        # Remove the oldest message (index 0)
        if messages:
            messages.pop(0)
        else:
            break

    # Get the model's reply and store it in the history
    ai_msg = llm.invoke(messages)
    # `invoke` returns a LangChain Message; ensure we work with its content
    if isinstance(ai_msg, str):
        # Safety net – if the provider returns a plain string, wrap it
        ai_msg_obj = AIMessage(content=ai_msg)
    else:
        # Assume it's already an AIMessage‑like object
        ai_msg_obj = AIMessage(content=ai_msg.content)
    messages.append(ai_msg_obj)

    return ai_msg_obj.content

@app.route('/')
def index():
    return render_template('index.html')

if __name__ == '__main__':
    app.run(debug=True, port=5000)
