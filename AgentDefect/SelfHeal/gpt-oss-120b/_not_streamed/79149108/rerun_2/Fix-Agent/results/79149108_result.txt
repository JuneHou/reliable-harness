from flask import Flask, render_template, request
from langchain_ollama import ChatOllama
from langchain_core.messages import HumanMessage, AIMessage
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128_000

# Store the conversation as LangChain message objects
messages = []

llm = ChatOllama(
    model=MODEL,
    temperature=1.1,
    stream=False,  # ``invoke`` returns a full response; set ``stream=False`` for simplicity
)


def get_number_of_tokens(message_objs):
    """Return an approximate token count for a list of LangChain message objects.

    The function falls back to a char‑based heuristic if the model‑specific encoder
    cannot be loaded.  It only counts the ``content`` of each message because the
    role token overhead is negligible for the size limits we are dealing with.
    """
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total = 0
        for msg in message_objs:
            total += len(encoding.encode(str(msg.content)))
        return total
    except Exception:
        # Approximate: 1 token ≈ 4 characters
        total_chars = sum(len(str(msg.content)) for msg in message_objs)
        return total_chars // 4


@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_text = request.form["text"].strip()

    # Optional: allow the user to reset/stop the conversation
    if user_text.lower() in {"stop", "quit", "exit", "end"}:
        messages.clear()
        return "Conversation ended. Feel free to start a new one!"

    # Add the new user message to the history
    messages.append(HumanMessage(content=user_text))

    # Enforce the token limit – remove the oldest *pairs* (user + assistant)
    while get_number_of_tokens(messages) > MODEL_MAX_TOKENS and len(messages) > 2:
        # Delete the first two messages (oldest user‑assistant exchange)
        del messages[0:2]

    # Get the model response
    ai_response = llm.invoke(messages)
    # ``invoke`` returns a ``ChatResult`` whose ``content`` attribute holds the reply.
    # Depending on the library version it may be a plain string; we handle both.
    response_text = getattr(ai_response, "content", str(ai_response))

    # Store the model's answer so the next turn has full context
    messages.append(AIMessage(content=response_text))

    return response_text


@app.route('/')
def index():
    return render_template('index.html')


if __name__ == '__main__':
    app.run(debug=True, port=5000)
