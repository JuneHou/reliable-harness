import os
from dotenv import load_dotenv

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.embeddings import OpenAIEmbeddings
from langchain_community.vectorstores import MongoDBAtlasVectorSearch
from pymongo import MongoClient


def run_agent(pdf_url: str):
    """Load a PDF, split it into chunks, embed the chunks with OpenAI,
    store them in a MongoDB Atlas Vector Search collection, and then
    perform a similarity search.

    Parameters
    ----------
    pdf_url: str
        Direct URL to the PDF document.

    Returns
    -------
    list
        A list of `Document` objects that are most similar to the query.
    """
    # Load environment variables (API keys, MongoDB connection string, etc.)
    load_dotenv()

    # ---------------------------------------------------------------------
    # Validate required environment variables
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment")

    mongo_uri = os.getenv("MONGODB_ATLAS_URI")
    if not mongo_uri:
        raise ValueError("MONGODB_ATLAS_URI not found in environment")

    mongo_db_name = os.getenv("MONGODB_ATLAS_DB") or "myDatabase"
    mongo_collection_name = os.getenv("MONGODB_ATLAS_COLLECTION") or "myCollection"
    index_name = os.getenv("MONGODB_ATLAS_VECTOR_SEARCH_INDEX") or "vector_index"

    # ---------------------------------------------------------------------
    # Connect to MongoDB Atlas
    # ---------------------------------------------------------------------
    client = MongoClient(mongo_uri)
    db = client[mongo_db_name]
    collection = db[mongo_collection_name]

    # ---------------------------------------------------------------------
    # Load and split the PDF into manageable text chunks
    # ---------------------------------------------------------------------
    loader = PyPDFLoader(pdf_url)
    raw_documents = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = text_splitter.split_documents(raw_documents)

    # ---------------------------------------------------------------------
    # Create the vector store (or retrieve an existing one)
    # ---------------------------------------------------------------------
    # If the collection already contains an Atlas Vector Search index, the
    # `from_documents` call will upsert the embeddings; otherwise it will create
    # the index automatically.
    embeddings = OpenAIEmbeddings(disallowed_special=())
    vector_store = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=embeddings,
        collection=collection,
        index_name=index_name,
    )

    # ---------------------------------------------------------------------
    # Perform a similarity search
    # ---------------------------------------------------------------------
    query = "What were the compute requirements for training GPT 4"
    # Adjust `k` to control the number of returned results
    results = vector_store.similarity_search(query, k=4)

    return results
