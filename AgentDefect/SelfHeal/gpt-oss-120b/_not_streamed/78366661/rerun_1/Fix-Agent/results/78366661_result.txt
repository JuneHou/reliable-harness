    * ``MONGODB_ATLAS_URI`` – the full MongoDB connection string (including
      username/password and the ``?retryWrites=true&w=majority`` suffix).

        raise ValueError(

        )

    client = MongoClient(mongo_uri)
    collection = client[db_name][coll_name]

    raw_documents = loader.load()
    text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = text_splitter.split_documents(raw_documents)

    # Create the vector store – this will upsert the embeddings into the
    # specified Atlas collection and create (or reuse) the named index.
    embeddings = OpenAIEmbeddings(disallowed_special=())
    vector_store = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=embeddings,
        collection=collection,
        index_name="vector_index",
    )

    # Perform the similarity search.
    query = "What were the compute requirements for training GPT 4"
    # ``k`` determines how many closest chunks are returned.
    results = vector_store.similarity_search(query, k=4)

    return results

import os
from dotenv import load_dotenv

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import MongoDBAtlasVectorSearch
from pymongo import MongoClient


def run_agent(pdf_url: str):
    """Load a PDF, split it into chunks, store the chunks in an Atlas Vector Search
    collection and perform a similarity search.

    Returns a list of ``Document`` objects that are most similar to the query.
    """
    load_dotenv()

    # ---------------------------------------------------------------------
    # 1. Retrieve required configuration values from the environment.
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment")

    atlas_uri = os.getenv("MONGODB_ATLAS_URI")
    db_name = os.getenv("MONGODB_DATABASE_NAME")
    collection_name = os.getenv("MONGODB_COLLECTION_NAME")
    index_name = os.getenv("VECTOR_SEARCH_INDEX_NAME", "vector_index")

    if not all([atlas_uri, db_name, collection_name]):
        raise ValueError(
            "MongoDB Atlas connection details (MONGODB_ATLAS_URI, MONGODB_DATABASE_NAME, MONGODB_COLLECTION_NAME) must be set"
        )

    # ---------------------------------------------------------------------
    # 2. Initialise the embedding model.  LangChain will automatically read the
    #    ``OPENAI_API_KEY`` environment variable, but passing it explicitly makes
    #    the code clearer.
    # ---------------------------------------------------------------------
    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key, disallowed_special=())

    # ---------------------------------------------------------------------
    # 3. Load and split the PDF into manageable text chunks.
    # ---------------------------------------------------------------------
    loader = PyPDFLoader(pdf_url)
    raw_docs = loader.load()
    splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = splitter.split_documents(raw_docs)

    # ---------------------------------------------------------------------
    # 4. Connect to the Atlas collection.  ``MongoClient`` handles TLS and
    #    authentication automatically when given the full connection string.
    # ---------------------------------------------------------------------
    client = MongoClient(atlas_uri)
    collection = client[db_name][collection_name]

    # ---------------------------------------------------------------------
    # 5. Create (or retrieve) the Atlas Vector Search index and ingest the
    #    documents.  ``from_documents`` will up‑sert the documents into the
    #    collection and create the index if it does not already exist.
    # ---------------------------------------------------------------------
    vector_store = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=embeddings,
        collection=collection,
        index_name=index_name,
    )

    # ---------------------------------------------------------------------
    # 6. Perform the similarity search.
    # ---------------------------------------------------------------------
    query = "What were the compute requirements for training GPT 4"
    results = vector_store.similarity_search(query)

    return results
