import os

from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from pymongo import MongoClient
from langchain_community.vectorstores import MongoDBAtlasVectorSearch


def run_agent(pdf_url: str):
    """Load a PDF, store its chunks in MongoDB Atlas Vector Search and run a similarity query.

    The function expects the following environment variables to be defined:

    * ``OPENAI_API_KEY`` – API key for the OpenAI embedding model.
    * ``MONGODB_URI`` – Full MongoDB connection string (e.g. ``mongodb+srv://user:pwd@cluster0.mongodb.net``).
    * ``MONGODB_DATABASE`` – Name of the database that contains the collection.
    * ``MONGODB_COLLECTION`` – Name of the collection that will hold the vector index.
    """
    load_dotenv()

    # ---------------------------------------------------------------------
    # Validation of required environment variables
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment")

    mongo_uri = os.getenv("MONGODB_URI")
    if not mongo_uri:
        raise ValueError("MONGODB_URI not found in environment")
    mongo_db_name = os.getenv("MONGODB_DATABASE")
    if not mongo_db_name:
        raise ValueError("MONGODB_DATABASE not found in environment")
    mongo_coll_name = os.getenv("MONGODB_COLLECTION")
    if not mongo_coll_name:
        raise ValueError("MONGODB_COLLECTION not found in environment")

    # ---------------------------------------------------------------------
    # Initialise MongoDB client and obtain a reference to the collection
    # ---------------------------------------------------------------------
    client = MongoClient(mongo_uri)
    db = client[mongo_db_name]
    collection = db[mongo_coll_name]

    # ---------------------------------------------------------------------
    # Load and chunk the PDF
    # ---------------------------------------------------------------------
    loader = PyPDFLoader(pdf_url)
    raw_documents = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = text_splitter.split_documents(raw_documents)

    # ---------------------------------------------------------------------
    # Create (or retrieve) the Atlas Vector Search index and store the chunks
    # ---------------------------------------------------------------------
    # ``OpenAIEmbeddings`` will use the ``OPENAI_API_KEY`` from the env.
    embedding = OpenAIEmbeddings(disallowed_special=())

    # ``MongoDBAtlasVectorSearch.from_documents`` will up‑sert the documents into
    # the collection and automatically create the vector index if it does not
    # already exist.  The ``index_name`` must match the name you defined in the
    # Atlas UI (or can be created on‑the‑fly).
    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=embedding,
        collection=collection,
        index_name="vector_index",  # adjust if you used a different name
    )

    # ---------------------------------------------------------------------
    # Perform a similarity search
    # ---------------------------------------------------------------------
    query = "What were the compute requirements for training GPT 4"
    results = vector_search.similarity_search(query)

    # ``results`` is a list of ``Document`` objects.  For a quick console view we
    # return the `page_content` of each hit.
    return [doc.page_content for doc in results]
