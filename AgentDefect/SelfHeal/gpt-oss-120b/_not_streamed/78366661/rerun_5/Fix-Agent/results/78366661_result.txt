import os
from dotenv import load_dotenv

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import MongoDBAtlasVectorSearch
from pymongo import MongoClient


def run_agent(pdf_url: str):
    """Load a PDF, split it into chunks, index the chunks in a MongoDB Atlas
    vector search collection and perform a similarity search.

    The function expects the following environment variables to be set:
        * ``OPENAI_API_KEY`` – your OpenAI key (used by ``OpenAIEmbeddings``)
        * ``MONGODB_URI`` – the full connection string to your Atlas cluster
        * ``MONGODB_DB`` – the database name that holds the collection
        * ``MONGODB_COLLECTION`` – the collection name that will store the vectors

    Returns the list of ``Document`` objects that are most similar to the query.
    """
    # ---------------------------------------------------------------------
    # Load configuration & validate
    # ---------------------------------------------------------------------
    load_dotenv()

    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment")

    mongodb_uri = os.getenv("MONGODB_URI")
    mongodb_db = os.getenv("MONGODB_DB")
    mongodb_collection = os.getenv("MONGODB_COLLECTION")
    if not all([mongodb_uri, mongodb_db, mongodb_collection]):
        raise ValueError(
            "One or more MongoDB environment variables are missing: "
            "MONGODB_URI, MONGODB_DB, MONGODB_COLLECTION"
        )

    # ---------------------------------------------------------------------
    # Prepare the PDF text
    # ---------------------------------------------------------------------
    loader = PyPDFLoader(pdf_url)
    raw_documents = loader.load()

    splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = splitter.split_documents(raw_documents)

    # ---------------------------------------------------------------------
    # Connect to MongoDB Atlas
    # ---------------------------------------------------------------------
    client = MongoClient(mongodb_uri)
    collection = client[mongodb_db][mongodb_collection]

    # ---------------------------------------------------------------------
    # Build (or retrieve) the Atlas vector search index
    # ---------------------------------------------------------------------
    # The name of the Atlas search index that has a vector search definition.
    # It must already exist in the Atlas UI (or be created via the API).
    vector_index_name = "vector_index"

    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=OpenAIEmbeddings(disallowed_special=()),
        collection=collection,
        index_name=vector_index_name,
    )

    # ---------------------------------------------------------------------
    # Perform a similarity search
    # ---------------------------------------------------------------------
    query = "What were the compute requirements for training GPT‑4?"
    results = vector_search.similarity_search(query)
    return results
