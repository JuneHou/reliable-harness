import logging
from langchain_community.document_loaders import UnstructuredPDFLoader
from langchain_community.embeddings import OllamaEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain.prompts import ChatPromptTemplate, PromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_community.chat_models import ChatOllama
from langchain_core.runnables import RunnablePassthrough
from langchain.retrievers.multi_query import MultiQueryRetriever

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s - %(levelname)s - %(message)s",
)


def _format_docs(docs):
    """Utility to turn a list of Document objects into a single string.

    LangChain retrievers return a list of `Document` objects. The RAG prompt
    expects a plain text context, so we concatenate the content of each document.
    """
    return "\n\n".join(doc.page_content for doc in docs)


def run_agent(question: str):
    """Runs a simple RAG pipeline using an Ollama embedding model and an
    Ollama chat model.

    The function loads a local PDF, splits it, creates a Chroma vector store,
    builds a Multi‑Query retriever, and finally answers the supplied *question*.
    """
    local_path = "./paper1.pdf"
    try:
        # ------------------------------------------------------------------
        # Load PDF
        # ------------------------------------------------------------------
        if not local_path:
            raise ValueError("No PDF file path supplied")
        loader = UnstructuredPDFLoader(file_path=local_path)
        docs = loader.load()
        logging.info("PDF loaded – %d page(s) read", len(docs))
        logging.info("First page preview: %s...", docs[0].page_content[:500])

        # ------------------------------------------------------------------
        # Chunk the documents
        # ------------------------------------------------------------------
        splitter = RecursiveCharacterTextSplitter(chunk_size=7500, chunk_overlap=100)
        chunks = splitter.split_documents(docs)
        logging.info("Created %d text chunks", len(chunks))

        # ------------------------------------------------------------------
        # Create the vector store (embeddings are generated lazily by Chroma)
        # ------------------------------------------------------------------
        embedding_model = OllamaEmbeddings(model="nomic-embed-text")
        vector_db = Chroma.from_documents(
            documents=chunks,
            embedding=embedding_model,
            collection_name="local-rag",
        )
        logging.info("Chroma vector store created")

        # ------------------------------------------------------------------
        # Initialise the LLM (Ollama chat model)
        # ------------------------------------------------------------------
        llm = ChatOllama(model="llama3")
        logging.info("Chat model loaded: %s", llm.model)

        # ------------------------------------------------------------------
        # Prompt for generating multiple queries
        # ------------------------------------------------------------------
        multi_query_prompt = PromptTemplate(
            input_variables=["question"],
            template=(
                "You are an AI assistant. Generate FIVE different versions of the"
                " following user question. Return each version on a separate line.\n"
                "Original question: {question}"
            ),
        )
        logging.info("Multi‑query prompt prepared")

        # ------------------------------------------------------------------
        # Build a MultiQueryRetriever – it will ask the LLM for alternative
        # questions, then run each through the underlying vector store retriever.
        # ------------------------------------------------------------------
        base_retriever = vector_db.as_retriever(search_kwargs={"k": 4})
        retriever = MultiQueryRetriever.from_llm(
            base_retriever,
            llm,
            prompt=multi_query_prompt,
        )
        logging.info("Multi‑query retriever ready")

        # ------------------------------------------------------------------
        # RAG answer prompt – we only provide the retrieved context.
        # ------------------------------------------------------------------
        rag_prompt = ChatPromptTemplate.from_template(
            """Answer the question using ONLY the provided context.\n\n"
            "Context:\n{context}\n\n"
            "Question: {question}\n"""
        )
        logging.info("RAG prompt created")

        # ------------------------------------------------------------------
        # Assemble the LangChain runnable: retrieve docs → format → prompt → LLM
        # ------------------------------------------------------------------
        chain = (
            {"context": retriever | _format_docs, "question": RunnablePassthrough()}
            | rag_prompt
            | llm
            | StrOutputParser()
        )
        logging.info("Runnable chain assembled")

        # ------------------------------------------------------------------
        # Execute
        # ------------------------------------------------------------------
        answer = chain.invoke(question)
        logging.info("Chain execution finished")
        logging.info("Answer: %s", answer)
        return answer

    except Exception as exc:
        logging.exception("An error occurred while running the RAG pipeline")
        return None
