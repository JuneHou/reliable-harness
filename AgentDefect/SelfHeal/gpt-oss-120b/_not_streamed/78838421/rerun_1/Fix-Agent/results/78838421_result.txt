def run_agent(question: str) -> str | None:
    """Run a simple RAG pipeline using Ollama for both embedding and LLM.

    The function:
    1. Loads a local PDF.
    2. Splits it into reasonably‑sized chunks.
    3. Creates a Chroma vector store using the ``nomic-embed-text`` model.
    4. Builds a ``MultiQueryRetriever`` that expands the user question.
    5. Retrieves relevant passages and answers the question with ``llama3``.

    Args:
        question: The user query.

    Returns:
        The generated answer or ``None`` if an error occurs.
    """
    pdf_path = Path("./paper1.pdf")
    if not pdf_path.is_file():
        logging.error("PDF file not found at %s", pdf_path)
        return None

    try:
        # -------------------------------------------------------------------
        # 1. Load PDF
        # -------------------------------------------------------------------
        loader = UnstructuredPDFLoader(file_path=str(pdf_path))
        documents = loader.load()
        logging.info("PDF loaded – %d pages extracted", len(documents))
        logging.info(
            "First page preview: %s...",
            documents[0].page_content[:200].replace("\n", " "),
        )

        # -------------------------------------------------------------------
        # 2. Split into chunks (smaller chunk size prevents the embedding model
        #    from hitting token limits and dramatically speeds up indexing).
        # -------------------------------------------------------------------
        splitter = RecursiveCharacterTextSplitter(chunk_size=2000, chunk_overlap=200)
        chunks = splitter.split_documents(documents)
        logging.info("Created %d text chunks", len(chunks))

        # -------------------------------------------------------------------
        # 3. Create vector store – use a persistent directory so that the store
        #    can be reused across runs (prevents re‑embedding on every call).
        # -------------------------------------------------------------------
        embedding = OllamaEmbeddings(model="nomic-embed-text")  # show_progress is optional and can cause hangs in non‑tty environments
        vectordb = Chroma.from_documents(
            documents=chunks,
            embedding=embedding,
            persist_directory="./chroma_local",
            collection_name="local-rag",
        )
        logging.info("Vector store created and persisted")

        # -------------------------------------------------------------------
        # 4. Prepare the LLM – ``ChatOllama`` talks to the local Ollama server.
        # -------------------------------------------------------------------
        llm = ChatOllama(model="llama3", temperature=0.0)
        logging.info("LLM model loaded: %s", llm.model)

        # -------------------------------------------------------------------
        # 5. Build the multi‑query retriever.  The prompt is a simple *PromptTemplate*
        #    because ``MultiQueryRetriever`` expects a standard LangChain ``PromptTemplate``.
        # -------------------------------------------------------------------
        query_prompt = PromptTemplate(
            input_variables=["question"],
            template=(
                "You are an AI assistant. Generate **five** different ways of asking the following "
                "question so that a vector store can retrieve the most relevant documents."
                "\nOriginal question: {question}\n"
                "\nReturn each variation on its own line."
            ),
        )
        retriever = MultiQueryRetriever.from_llm(
            vectorstore=vectordb.as_retriever(search_kwargs={"k": 5}),
            llm=llm,
            prompt=query_prompt,
        )
        logging.info("Multi‑query retriever created")

        # -------------------------------------------------------------------
        # 6. Build the final RAG chain.
        # -------------------------------------------------------------------
        rag_template = (
            "Answer the question **using only** the provided context.\n\n"
            "Context:\n{context}\n\n"
            "Question: {question}\n"
        )
        rag_prompt = ChatPromptTemplate.from_template(rag_template)

        chain = (
            {"context": retriever, "question": RunnablePassthrough()}
            | rag_prompt
            | llm
            | StrOutputParser()
        )
        logging.info("RAG chain assembled – invoking")

        answer = chain.invoke(question)
        logging.info("RAG chain completed")
        return answer

    except Exception as exc:
        logging.exception("An error occurred while running the RAG agent")
        return None
