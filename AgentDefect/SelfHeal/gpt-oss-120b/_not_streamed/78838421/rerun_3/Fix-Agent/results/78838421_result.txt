def run_agent(question: str):

    documents = loader.load()

        documents[0].page_content[:500].replace("\n", " "),
    )

    #    between retrieval quality and embedding speed for most models.
    # ---------------------------------------------------------------------

    # 3. Create the embedding model.  ``show_progress`` can cause the script to
    #    appear to hang in headless environments, so we disable it.

        collection_name="local-rag",

    answer = chain.invoke(question)

    return chunks
