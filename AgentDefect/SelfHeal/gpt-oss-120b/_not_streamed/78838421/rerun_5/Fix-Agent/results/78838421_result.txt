from pathlib import Path

from langchain_community.document_loaders import UnstructuredPDFLoader

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s - %(levelname)s - %(message)s",
)

PDF_PATH = Path("./paper1.pdf")
CHROMA_DIR = Path("./chroma_db")
CHUNK_SIZE = 1500          # smaller chunks avoid over‑length embeddings
CHUNK_OVERLAP = 200
MAX_RETRIEVE_K = 4          # how many docs to pull per generated query

# -----------------------------------------------------------------------------
# Helper functions
# -----------------------------------------------------------------------------

def load_pdf(path: Path):
    """Load a PDF – raises a clear error if the file is missing."""
    if not path.is_file():
        raise FileNotFoundError(f"PDF not found at {path.resolve()}")
    loader = UnstructuredPDFLoader(file_path=str(path))
    docs = loader.load()
    logging.info("PDF loaded – %d page(s) found", len(docs))
    logging.info("First page preview: %s", docs[0].page_content[:200].replace("\n", " "))
    return docs


def chunk_documents(docs):
    splitter = RecursiveCharacterTextSplitter(chunk_size=CHUNK_SIZE, chunk_overlap=CHUNK_OVERLAP)
    chunks = splitter.split_documents(docs)
    logging.info("Document split into %d chunk(s)", len(chunks))
    return chunks


def build_vectorstore(chunks):
    """Create (or load) a Chroma DB from the supplied chunks.

    The function is deliberately split out so that any problem with the embedding
    model is isolated and logged.
    """
    # Ollama can be a little chatty – silence progress to avoid unnecessary I/O.
    embedder = OllamaEmbeddings(model="nomic-embed-text", show_progress=False)
    logging.info("Embedding model instantiated: %s", embedder)

    # Ensure the persistence directory exists – Chroma will create it if missing.
    CHROMA_DIR.mkdir(parents=True, exist_ok=True)

    vector_store = Chroma.from_documents(
        documents=chunks,
        embedding_function=embedder,
        persist_directory=str(CHROMA_DIR),
        collection_name="local-rag",
    )
    logging.info("Vector store created – %d vectors stored", vector_store._client._collection.count())
    return vector_store


def create_multi_query_retriever(vector_store):
    # Prompt used to generate alternate questions for retrieval.
    query_prompt = PromptTemplate(
        input_variables=["question"],
        template=(
            "You are an AI assistant. Generate five distinct variations of the following "
            "question that could be used to retrieve relevant documents from a vector "
            "database. Return each variation on a separate line.\n"
            "Original question: {question}"
        ),
    )

    # Load the LLM – we keep the default temperature (0) for deterministic output.
    llm = ChatOllama(model="llama3")
    logging.info("LLM instantiated (model=llama3)")

    retriever = MultiQueryRetriever.from_llm(
        vector_store.as_retriever(search_kwargs={"k": MAX_RETRIEVE_K}),
        llm,
        prompt=query_prompt,
    )
    logging.info("Multi‑query retriever built")
    return retriever, llm


def build_rag_chain(retriever, llm):
    rag_prompt = ChatPromptTemplate.from_template(
        """Answer the question using **only** the provided context.\n\n"
        "Context:\n{context}\n\n"
        "Question: {question}\n"""
    )
    chain = (
        {"context": retriever, "question": RunnablePassthrough()}
        | rag_prompt
        | llm
        | StrOutputParser()
    )
    logging.info("RAG chain assembled")
    return chain


def run_agent(question: str):
