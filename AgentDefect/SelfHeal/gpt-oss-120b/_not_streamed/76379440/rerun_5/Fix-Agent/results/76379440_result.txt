import os
from tqdm import tqdm
from langchain.docstore.document import Document
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from dotenv import load_dotenv

load_dotenv()


def _store_collection(collection_name: str, documents: list[Document], ids: list[str], embedding) -> None:
    """Create (or replace) a Chroma collection and persist it to disk.

    Args:
        collection_name: Name of the Chroma collection.
        documents: List of ``Document`` objects to store.
        ids: List of string IDs – must be the same length as ``documents``.
        embedding: An instantiated LangChain ``Embeddings`` object.
    """
    # ``from_documents`` will create the collection if it does not exist or
    # replace it if it already exists.  We explicitly pass ``persist_directory``
    # so the collection is written to disk.
    db = Chroma.from_documents(
        collection_name=collection_name,
        documents=documents,
        ids=ids,
        embedding=embedding,
        persist_directory="./data",
    )
    # ``persist`` flushes the in‑memory state to the on‑disk store.
    db.persist()


def _load_collection(collection_name: str, embedding) -> Chroma:
    """Load an already‑persisted collection.

    The wrapper ``Chroma`` can be instantiated directly with the
    ``persist_directory`` and ``collection_name`` – it will *not* recompute the
    embeddings, it just loads the existing index.
    """
    return Chroma(
        persist_directory="./data",
        collection_name=collection_name,
        embedding_function=embedding,
    )


def run_agent() -> dict:
    """Create a few test collections, persist them, and return their stored
    embeddings.

    The function returns a mapping of collection name → the raw dictionary
    returned by ``Chromadb``'s ``get`` method, which includes ``ids`` and the
    corresponding ``embeddings``.
    """
    embeddings = OpenAIEmbeddings()

    # ---------------------------------------------------------------------
    # 1. Create and persist the collections.
    # ---------------------------------------------------------------------
    collection_names = ["class1-sub2-chap3", "class2-sub3-chap4"]
    for db_collection_name in tqdm(collection_names):
        documents: list[Document] = []
        doc_ids: list[str] = []
        for doc_index in range(3):
            cl, sub, chap = db_collection_name.split("-")
            content = f"This is {db_collection_name}-doc{doc_index}"
            doc = Document(
                page_content=content,
                metadata={
                    "chunk_num": doc_index,
                    "chapter": chap,
                    "class": cl,
                    "subject": sub,
                },
            )
            documents.append(doc)
            doc_ids.append(str(doc_index))
        _store_collection(db_collection_name, documents, doc_ids, embeddings)

    # ---------------------------------------------------------------------
    # 2. Load the collections and pull out the stored embeddings.
    # ---------------------------------------------------------------------
    results: dict = {}
    for name in collection_names:
        db = _load_collection(name, embeddings)
        # ``include`` tells Chroma which fields to return.  "embeddings" is the
        # vector representation we are interested in.
        collection_data = db._collection.get(include=["ids", "embeddings"])
        results[name] = collection_data

    # Demonstrate adding a single document to a new collection and retrieving it.
    initial_content = "This is an initial document content"
    document_id = "doc1"
    original_doc = Document(page_content=initial_content, metadata={"page": "0"})

    # Store the new collection.
    _store_collection(
        collection_name="test_collection",
        documents=[original_doc],
        ids=[document_id],
        embedding=embeddings,
    )
    # Load and fetch its embeddings.
    test_db = _load_collection("test_collection", embeddings)
    results["test_collection"] = test_db._collection.get(include=["ids", "embeddings"])

    return results


if __name__ == "__main__":
    result = run_agent()
    for coll, data in result.items():
        print(f"Collection: {coll}")
        print(data)
        print("---")
