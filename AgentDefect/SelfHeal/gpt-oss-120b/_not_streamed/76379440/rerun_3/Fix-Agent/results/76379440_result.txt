from tqdm import tqdm
from langchain.docstore.document import Document
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from dotenv import load_dotenv

load_dotenv()


def run_agent():
    # Initialise the embedding model once
    embeddings = OpenAIEmbeddings()

    # Persist a few collections – each collection gets its own Chroma DB on disk
    for db_collection_name in tqdm(["class1-sub2-chap3", "class2-sub3-chap4"]):
        documents = []
        doc_ids = []
        cl, sub, chap = db_collection_name.split("-")
        for doc_index in range(3):
            content = f"This is {db_collection_name}-doc{doc_index}"
            doc = Document(
                page_content=content,
                metadata={
                    "chunk_num": doc_index,
                    "chapter": chap,
                    "class": cl,
                    "subject": sub,
                },
            )
            documents.append(doc)
            doc_ids.append(str(doc_index))

        # Create (or overwrite) the collection and persist it to disk
        db = Chroma.from_documents(
            collection_name=db_collection_name,
            documents=documents,
            ids=doc_ids,
            embedding=embeddings,
            persist_directory="./data",
        )
        db.persist()

    # ---------------------------------------------------------------------
    # Create a separate collection that we will query later
    # ---------------------------------------------------------------------
    initial_content = "This is an initial document content"
    document_id = "doc1"
    original_doc = Document(page_content=initial_content, metadata={"page": "0"})

    # Store the document in its own collection
    test_collection_name = "test_collection"
    test_db = Chroma.from_documents(
        collection_name=test_collection_name,
        documents=[original_doc],
        ids=[document_id],
        embedding=embeddings,
        persist_directory="./data",
    )
    test_db.persist()

    # ---------------------------------------------------------------------
    # Load the collection back from disk and retrieve the stored embeddings
    # ---------------------------------------------------------------------
    loaded_db = Chroma(
        collection_name=test_collection_name,
        embedding_function=embeddings,
        persist_directory="./data",
    )

    # `get` returns a dictionary with keys: ids, documents, metadatas, embeddings
    result = loaded_db.get()
    return result


if __name__ == "__main__":

    print(result)
