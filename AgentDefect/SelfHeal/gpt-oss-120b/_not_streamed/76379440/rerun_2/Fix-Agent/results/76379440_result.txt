from tqdm import tqdm
from langchain.docstore.document import Document
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from dotenv import load_dotenv

load_dotenv()


def run_agent():
    """Create a few collections in a persistent Chroma DB and return the stored
    embeddings (together with the documents and metadata) for each collection.
    """
    embeddings = OpenAIEmbeddings()
    persist_dir = "./data"
    # store results for every collection so the caller can inspect them
    all_results = {}

    for db_collection_name in tqdm(["class1-sub2-chap3", "class2-sub3-chap4"]):
        documents = []
        doc_ids = []
        # split once – we need these values for the metadata of every doc
        cl, sub, chap = db_collection_name.split("-")
        for doc_index in range(3):
            content = f"This is {db_collection_name}-doc{doc_index}"
            doc = Document(
                page_content=content,
                metadata={
                    "chunk_num": doc_index,
                    "chapter": chap,
                    "class": cl,
                    "subject": sub,
                },
            )
            documents.append(doc)
            doc_ids.append(str(doc_index))

        # Build (or load) the collection and persist it to disk
        db = Chroma.from_documents(
            collection_name=db_collection_name,
            documents=documents,
            ids=doc_ids,
            embedding=embeddings,
            persist_directory=persist_dir,
        )
        db.persist()

        # Retrieve the stored data – we explicitly ask for embeddings
        result = db.get(ids=doc_ids, include=["documents", "metadatas", "embeddings"])
        all_results[db_collection_name] = result

    # ---------------------------------------------------------------------
    # Example: add a single document to a new collection and fetch its embedding
    # ---------------------------------------------------------------------
    initial_content = "This is an initial document content"
    document_id = "doc1"
    original_doc = Document(page_content=initial_content, metadata={"page": "0"})

    new_db = Chroma.from_documents(
        collection_name="test_collection",
        documents=[original_doc],
        ids=[document_id],
        embedding=embeddings,
        persist_directory=persist_dir,
    )
    new_db.persist()
    all_results["test_collection"] = new_db.get(
        ids=[document_id], include=["documents", "metadatas", "embeddings"]
    )

    return all_results
