import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool

# Load environment variables containing the OpenAI API key
load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    """Simple calculator tool that evaluates a Python arithmetic expression."""

    name = "calculator"
    description = "Performs basic arithmetic. Input should be a valid Python expression, e.g., '2 + 2'."

    def _run(self, query: str) -> str:
        # Directly return the raw result of the evaluation as a string.
        return str(eval(query))

    async def _arun(self, query: str) -> str:
        raise NotImplementedError()


def create_agent():
    """Create a LangChain conversational React agent that stops immediately after a tool call.

    By limiting ``max_iterations`` to ``1`` and using ``early_stopping_method="force"``
    we guarantee that the agent terminates right after the first tool execution. The
    result returned to the user will therefore be the raw output of the tool (the
    calculator) without any additional LLM re‑formatting.
    """
    llm_gpt4 = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    return initialize_agent(
        agent="chat-conversational-react-description",
        tools=[CalculatorTool()],
        llm=llm_gpt4,
        verbose=True,
        max_iterations=1,               # only one iteration (the tool call)
        early_stopping_method="force",  # force stop after the iteration
        memory=memory,
    )


def run_agent(prompt: str) -> str:
    """Run the agent on the given prompt and return the raw tool output."""
    agent = create_agent()
    return agent.run(prompt)
