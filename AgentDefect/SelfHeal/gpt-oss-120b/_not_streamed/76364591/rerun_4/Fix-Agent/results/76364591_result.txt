import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool

load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    """A simple calculator tool that evaluates a Python expression.

    The tool returns the raw result of ``eval`` as a string.  It deliberately
    raises ``ValueError`` when the expression cannot be evaluated – this allows
    the surrounding ``run_agent`` helper to fall back to the conversational
    agent when the input is not a pure arithmetic expression.
    """

    name = "calculator"
    description = "Performs basic arithmetic"

    def _run(self, query: str) -> str:
        # ``eval`` is used only for demonstration purposes; in production you
        # should replace it with a safe arithmetic parser.
        try:
            return str(eval(query, {"__builtins__": {}}))
        except Exception as exc:
            raise ValueError(f"CalculatorTool could not evaluate expression: {exc}")

    async def _arun(self, query: str) -> str:
        raise NotImplementedError()


def create_agent():
    """Create a conversational React‑style agent.

    The agent is configured with ``max_iterations=2`` and ``early_stopping_method="force"``
    so that it will stop after the tool has been executed – the tool's return value
    is passed straight back to the caller without an extra LLM‑generated summary.
    """
    llm_gpt4 = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    return initialize_agent(
        agent="chat-conversational-react-description",
        tools=[CalculatorTool()],
        llm=llm_gpt4,
        verbose=False,
        max_iterations=2,
        early_stopping_method="force",  # stop after the tool run
        memory=memory,
    )


def run_agent(prompt: str) -> str:
    """Run the agent.

    If the prompt is a valid arithmetic expression the calculator tool is used
    directly and its raw output is returned.  Otherwise the full conversational
    agent is invoked.
    """
    # Try to handle pure arithmetic without involving the LLM.
    calc_tool = CalculatorTool()
    try:
        # ``calc_tool._run`` will raise ``ValueError`` for non‑arithmetic inputs.
        return calc_tool._run(prompt)
    except Exception:
        # Fallback to the full agent for any other kind of request.
        agent = create_agent()
        return agent.run(prompt)
