import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool

load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    """Simple calculator tool that evaluates a Python arithmetic expression."""

    name = "calculator"
    description = "Performs basic arithmetic. Input should be a valid Python arithmetic expression."

    def _run(self, query: str) -> str:
        # Using eval is acceptable here because the tool is sandboxed and only arithmetic is expected.
        # Convert the result to string so the agent can forward it unchanged.
        return str(eval(query))

    async def _arun(self, query: str) -> str:  # pragma: no cover
        raise NotImplementedError()


def create_agent():
    """Create a LangChain conversational‑react agent that returns the raw tool output.

    The key changes are:
    * ``max_iterations`` is set to ``1`` so the loop stops after the first tool call.
    * ``early_stopping_method`` is ``"force"`` which forces termination after the
      maximum iterations.
    * ``agent_kwargs`` includes ``{"return_intermediate_steps": True}`` so the
      executor returns the observation (the tool's raw output) in the result dict.
    """
    llm_gpt4 = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    return initialize_agent(
        agent="chat-conversational-react-description",
        tools=[CalculatorTool()],
        llm=llm_gpt4,
        verbose=True,
        max_iterations=1,
        early_stopping_method="force",
        memory=memory,
        agent_kwargs={"return_intermediate_steps": True},
    )


def run_agent(prompt: str) -> str:
    """Run the agent and return the raw output produced by the tool.

    Because we asked the executor to return intermediate steps, the result is a
    dictionary with the keys ``output`` and ``intermediate_steps``. The first
    element of ``intermediate_steps`` is a tuple ``(action, observation)`` where
    ``observation`` is exactly the string returned by ``CalculatorTool._run``.
    We extract and return that observation.
    """
    agent = create_agent()
    # ``agent`` expects a dict with an ``input`` key.
    result = agent({"input": prompt})

    # ``result`` is a dict when ``return_intermediate_steps`` is True.
    # Guard against unexpected shapes.
    if isinstance(result, dict) and "intermediate_steps" in result:
        # Grab the observation from the first (and only) step.
        observation = result["intermediate_steps"][0][1]
        return observation
    # Fallback – return whatever the agent decided to output.
    return result.get("output", "")
