import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent, AgentType, AgentOutputParser, AgentFinish
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool
from typing import Union

load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    """A simple calculator tool that evaluates arithmetic expressions safely.
    It returns the result as a plain string so the agent can forward it directly
    to the user without any additional LLM formatting.
    """

    name = "calculator"
    description = "Performs basic arithmetic. Input should be a valid Python arithmetic expression."

    def _run(self, query: str) -> str:
        # Using eval is acceptable for this controlled example, but in a real
        # scenario you would want a safe parser (e.g., `asteval` or `numexpr`).
        return str(eval(query))

    async def _arun(self, query: str) -> str:  # pragma: no cover
        raise NotImplementedError()


class RawToolOutputParser(AgentOutputParser):
    """Parse the LLM output and return it *unchanged*.

    The default ReAct parser expects a ``Final Answer:`` block, which forces the
    LLM to re‑format the tool result. By overriding ``parse`` we simply forward the
    raw tool output back to the caller, achieving the desired "tool‑only" response.
    """

    def parse(self, llm_output: str) -> Union[AgentFinish, dict]:  # type: ignore[override]
        # ``llm_output`` at this point already contains the tool's raw result.
        # We wrap it in ``AgentFinish`` so LangChain treats it as the final answer.
        return AgentFinish(return_values={"output": llm_output.strip()}, log=llm_output)

    async def aparse(self, llm_output: str) -> Union[AgentFinish, dict]:  # pragma: no cover
        return self.parse(llm_output)


def create_agent():
    llm_gpt4 = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    return initialize_agent(
        agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
        tools=[CalculatorTool()],
        llm=llm_gpt4,
        verbose=True,
        max_iterations=1,               # stop after the first tool execution
        early_stopping_method="force", # bypass any extra generation step
        memory=memory,
        agent_kwargs={"output_parser": RawToolOutputParser()},
    )


def run_agent(prompt: str) -> str:
    agent = create_agent()
    # ``agent.run`` returns the value stored under the ``output`` key of the
    # ``AgentFinish`` object produced by our custom parser.
    return agent.run(prompt)
