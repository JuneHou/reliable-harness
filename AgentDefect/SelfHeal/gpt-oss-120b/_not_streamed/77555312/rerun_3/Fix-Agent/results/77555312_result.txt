import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()


def run_agent(query: str):
    # Load the PDF (or any file) as a single document
    loader = UnstructuredFileLoader('paper2.pdf')
    docs = loader.load()

    # Split the document into overlapping chunks and give each chunk a unique
    # identifier (the start index) that will later help us deduplicate any
    # accidental repeats.
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=200,
        add_start_index=True,
    )
    all_splits = text_splitter.split_documents(docs)

    # Create a Chroma vector store from the chunks.
    # Using MMR (Maximum Marginal Relevance) as the search type makes the
    # retriever return more diverse results, which helps avoid near‑duplicate
    # chunks being returned.
    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
    )

    # Retrieve 6 *distinct* chunks.  `fetch_k` is larger than `k` so MMR has a
    # pool to select from, and `lambda_mult` balances relevance vs. diversity.
    retriever = vectorstore.as_retriever(
        search_type="mmr",
        search_kwargs={"k": 6, "fetch_k": 20, "lambda_mult": 0.5},
    )

    retrieved_docs = retriever.get_relevant_documents(query)

    # Final safeguard: deduplicate by the start index (or by page content if
    # start index is missing).  This ensures we never return the same chunk
    # twice even if MMR somehow picks it.
    seen = set()
    unique_docs = []
    for doc in retrieved_docs:
        # Use the start index metadata added by the splitter as a unique key.
        key = doc.metadata.get("start_index")
        if key is None:
            # Fallback to the document's text content hash.
            key = hash(doc.page_content)
        if key not in seen:
            seen.add(key)
            unique_docs.append(doc)

    return unique_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    for i, doc in enumerate(result, 1):
        print(f"--- Chunk {i} (start index: {doc.metadata.get('start_index')}) ---")
        print(doc.page_content)
        print()
