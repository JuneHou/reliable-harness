import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()

def run_agent(query: str):
    """Load a PDF, split it into overlapping chunks, index the chunks in
    a Chroma vector store and retrieve *six* **distinct** chunks that are
    most relevant to the supplied query.
    
    The original implementation used a similarity search which can return
    overlapping chunks that are essentially duplicates (because the text
    splitter creates overlapping windows).  Switching to Maximal Marginal
    Relevance (MMR) ensures that the retrieved set is diverse, and we also
    perform a light post‑retrieval de‑duplication as a safety net.
    """
    # Load the document
    loader = UnstructuredFileLoader('paper2.pdf')
    docs = loader.load()

    # Split the document into overlapping chunks
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    # Create (or reuse) a Chroma collection in memory
    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
        # Use a deterministic collection name to avoid recreating the DB on
        # every call when this function is used in a longer‑lived process.
        collection_name="paper2_chunks",
    )

    # Use MMR (maximal marginal relevance) to obtain diverse results.
    # fetch_k > k allows the algorithm to consider a larger pool before
    # selecting the final k results.
    retriever = vectorstore.as_retriever(
        search_type="mmr",
        search_kwargs={"k": 6, "fetch_k": 20, "lambda_mult": 0.5},
    )

    # Retrieve documents
    retrieved_docs = retriever.get_relevant_documents(query)

    # Post‑process: de‑duplicate any remaining exact matches (unlikely but
    # harmless).  We keep the first occurrence of each unique page_content.
    seen = set()
    unique_docs = []
    for doc in retrieved_docs:
        if doc.page_content not in seen:
            seen.add(doc.page_content)
            unique_docs.append(doc)
        if len(unique_docs) == 6:
            break

    return unique_docs

if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    for i, doc in enumerate(result, start=1):
        print(f"--- Chunk {i} ---")
        print(doc.page_content)
        print()
