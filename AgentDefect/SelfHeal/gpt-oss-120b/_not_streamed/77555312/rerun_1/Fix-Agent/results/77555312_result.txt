import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()


def _deduplicate_documents(docs, max_len=6):
    """Return up to `max_len` documents with unique page_content.

    Chroma can return the same chunk (or very similar overlapping chunks) more than
    once when the similarity scores are identical. This helper removes duplicates
    based on the raw text of the chunk while preserving the order returned by the
    retriever.
    """
    unique_docs = []
    seen_contents = set()
    for doc in docs:
        # Use the raw text as the uniqueness key. If the document has metadata
        # that can also be used, extend the key accordingly.
        content_key = doc.page_content.strip()
        if content_key and content_key not in seen_contents:
            unique_docs.append(doc)
            seen_contents.add(content_key)
        if len(unique_docs) >= max_len:
            break
    return unique_docs


def run_agent(query: str):
    # Load the PDF (or any file) and split it into overlapping chunks.
    loader = UnstructuredFileLoader("paper2.pdf")
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=200,
        add_start_index=True,
    )
    all_splits = text_splitter.split_documents(docs)

    # Create a Chroma vector store. Using a persistent directory prevents the
    # creation of a new collection on every run, which could otherwise duplicate
    # embeddings.
    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
        collection_name="paper_chunks",
        persist_directory="./chroma_db",  # optional but useful for persistence
    )

    # Retrieve the top‑k most similar chunks.
    retriever = vectorstore.as_retriever(
        search_type="similarity",
        search_kwargs={"k": 6},
    )
    retrieved_docs = retriever.get_relevant_documents(query)

    # Ensure we return six *distinct* chunks.
    unique_docs = _deduplicate_documents(retrieved_docs, max_len=6)
    return unique_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    for i, doc in enumerate(result, 1):
        print(f"--- Chunk {i} ---")
        print(doc.page_content)
        print()
