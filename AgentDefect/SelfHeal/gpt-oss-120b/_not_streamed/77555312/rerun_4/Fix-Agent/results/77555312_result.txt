import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()


def _deduplicate_documents(docs, max_num=6):
    """Return up to ``max_num`` unique documents based on their text content.
    This helper removes exact duplicates that can appear when overlapping chunks
    are very similar or when the similarity search returns the same chunk multiple
    times (e.g., when using a similarity retriever).
    """
    seen = set()
    unique_docs = []
    for doc in docs:
        # Normalise whitespace for a more robust duplicate check
        normalized = " ".join(doc.page_content.split())
        if normalized not in seen:
            seen.add(normalized)
            unique_docs.append(doc)
        if len(unique_docs) >= max_num:
            break
    return unique_docs


def run_agent(query: str):
    """Load a PDF, split it into overlapping chunks, embed the chunks with
    OpenAI, store them in a Chroma vector store and retrieve six *distinct*
    chunks that are most relevant to ``query``.
    """
    # Load the PDF (or any file supported by UnstructuredFileLoader)
    loader = UnstructuredFileLoader("paper2.pdf")
    docs = loader.load()

    # Split the document into overlapping chunks – the overlap helps preserve
    # context across chunks.
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    all_splits = text_splitter.split_documents(docs)

    # Create (or load) a Chroma vector store.  ``persist_directory`` is omitted
    # here because we only need an in‑memory store for this example.
    vectorstore = Chroma.from_documents(
        documents=all_splits, embedding=OpenAIEmbeddings()
    )

    # Use an MMR retriever to encourage diversity in the returned chunks.
    # ``lambda_mult`` controls the trade‑off between relevance and diversity.
    retriever = vectorstore.as_retriever(
        search_type="mmr",  # maximum marginal relevance
        search_kwargs={"k": 6, "lambda_mult": 0.5},
    )

    # Retrieve candidate documents.
    candidate_docs = retriever.get_relevant_documents(query)

    # Remove any exact duplicates that may still appear.
    unique_docs = _deduplicate_documents(candidate_docs, max_num=6)

    return unique_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    for i, doc in enumerate(result, 1):
        print(f"--- Chunk {i} ---\n{doc.page_content}\n")
