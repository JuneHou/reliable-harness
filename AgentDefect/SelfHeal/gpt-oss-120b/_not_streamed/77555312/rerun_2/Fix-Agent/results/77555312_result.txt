import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()


def run_agent(query: str):
    """Load a PDF, split it into overlapping chunks, store the chunks in a
    Chroma vector store and retrieve *six* distinct chunks that are most
    relevant to ``query``.

    The original implementation could return duplicate chunks because the
    default ``similarity`` search simply returns the top‑k closest vectors, which
    may be highly overlapping (especially when the chunks themselves overlap).
    Switching to a *Maximum‑Marginal‑Relevance* (MMR) search promotes diversity
    and, together with a small post‑processing deduplication step, guarantees that
    the returned list contains six unique documents.
    """

    # ---------------------------------------------------------------------
    # 1. Load the PDF and split it into overlapping chunks.
    # ---------------------------------------------------------------------
    loader = UnstructuredFileLoader("paper2.pdf")
    docs = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000, chunk_overlap=200, add_start_index=True
    )
    chunks = text_splitter.split_documents(docs)

    # ---------------------------------------------------------------------
    # 2. Create a Chroma vector store from the chunks.
    # ---------------------------------------------------------------------
    vectorstore = Chroma.from_documents(
        documents=chunks,
        embedding=OpenAIEmbeddings(),
        # Optional: persist_directory="chroma_db"  # enables caching between runs
    )

    # ---------------------------------------------------------------------
    # 3. Retrieve using MMR to reduce the chance of returning near‑duplicate
    #    chunks. ``fetch_k`` is set larger than ``k`` so that MMR has a richer
    #    candidate pool to select from.
    # ---------------------------------------------------------------------
    retriever = vectorstore.as_retriever(
        search_type="mmr",  # Max‑Marginal‑Relevance search
        search_kwargs={
            "k": 6,                # final number of results we want
            "fetch_k": 20,         # how many raw neighbours to consider
            "lambda_mult": 0.5,    # trade‑off between relevance and diversity
        },
    )

    # ---------------------------------------------------------------------
    # 4. Perform the retrieval.
    # ---------------------------------------------------------------------
    retrieved_docs = retriever.get_relevant_documents(query)

    # ---------------------------------------------------------------------
    # 5. Post‑process to guarantee uniqueness (defensive measure).
    # ---------------------------------------------------------------------
    unique_docs = []
    seen_texts = set()
    for doc in retrieved_docs:
        if doc.page_content not in seen_texts:
            unique_docs.append(doc)
            seen_texts.add(doc.page_content)

    # Ensure we always return exactly six documents – if deduplication removed
    # some, we simply return what we have (the caller can decide how to handle a
    # shorter list).
    return unique_docs[:6]


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    for i, doc in enumerate(result, start=1):
        print(f"--- Document {i} ---")
        print(doc.page_content[:500])  # print a preview of each chunk
        print()
