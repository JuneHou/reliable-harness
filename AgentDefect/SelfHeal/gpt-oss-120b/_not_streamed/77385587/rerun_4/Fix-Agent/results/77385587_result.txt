import os
from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage.file_system import LocalFileStore
from langchain.schema import Document
import nltk


def run_agent(query: str, store_location: str = "./docstore", chroma_dir: str = "./chroma_db_child"):
    """Retrieve documents relevant to *query* using a persistent docstore.

    Args:
        query: The user query.
        store_location: Directory where the docstore will be persisted.
        chroma_dir: Directory for the Chroma vector store persistence.
    """
    # Ensure NLTK tokenizer data is available.
    nltk.download("punkt", quiet=True)

    # ---------------------------------------------------------------------
    # Embeddings – cached locally to avoid repeated downloads.
    # ---------------------------------------------------------------------
    hf_embed_path = "./hf_cache"
    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path,
    )

    # ---------------------------------------------------------------------
    # Initialise a *persistent* docstore.  ``LocalFileStore`` stores arbitrary
    # key/value pairs on disk, making it suitable as a replacement for the
    # volatile ``InMemoryStore`` used in the original example.
    # ---------------------------------------------------------------------
    docstore = LocalFileStore(store_location)

    # ---------------------------------------------------------------------
    # Initialise the Chroma vector store – it is also persisted on disk via
    # ``persist_directory``.  We reuse the same collection name on every run.
    # ---------------------------------------------------------------------
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory=chroma_dir,
    )

    # ---------------------------------------------------------------------
    # Child splitter – turns parent documents into smaller chunks that are
    # stored in the vector store.
    # ---------------------------------------------------------------------
    child_splitter = NLTKTextSplitter(chunk_size=400)

    # ---------------------------------------------------------------------
    # Build the ParentDocumentRetriever with the persistent components.
    # ---------------------------------------------------------------------
    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=docstore,
        child_splitter=child_splitter,
    )

    # Only add the original documents if the docstore is empty.  This avoids
    # creating duplicate entries on subsequent executions.
    if not docstore.mget(list(docstore.keys())):  # type: ignore[arg-type]
        docs = [
            Document(
                page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."
            ),
            Document(
                page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."
            ),
            Document(
                page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."
            ),
        ]
        retriever.add_documents(docs)
        # Ensure the vector store writes its state to disk.
        vectorstore.persist()

    # ---------------------------------------------------------------------
    # Perform the similarity search using the persisted vector store.
    # ---------------------------------------------------------------------
    matching_docs = vectorstore.max_marginal_relevance_search(query, k=3)
    return matching_docs


if __name__ == "__main__":
    query = "What is RAG?"
    results = run_agent(query)
    for doc in results:
        print("---")
        print(doc.page_content)

        Document(page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."),

    )

    # ---------------------------------------------------------------------

    #    from chunk ID → parent document in ``docstore``.
    # ---------------------------------------------------------------------
    child_splitter = NLTKTextSplitter(chunk_size=400)
    retriever = ParentDocumentRetriever(
        vectorstore=child_vectorstore,
        docstore=docstore,
        child_splitter=child_splitter,
    )

    # ---------------------------------------------------------------------

    # ---------------------------------------------------------------------

    return relevant_parents
