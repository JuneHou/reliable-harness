    """Retrieve the full parent documents relevant to *query* using a persistent docstore.

    The function:

    4. Creates a ``ParentDocumentRetriever`` that stores both the parent documents and the

    # It internally queries the child chunks, then looks up the parent IDs in the

# ---------------------------------------------------------------------------

class PersistentInMemoryStore(InMemoryStore):
    """A thin wrapper around :class:`InMemoryStore` that persists its data to
    disk using ``pickle``.

    The store saves a mapping of ``doc_id`` -> :class:`Document` objects.  The
    location of the pickle file is supplied via ``persist_path``.  On
    initialization the class attempts to load an existing file; if the file
    does not exist an empty store is created.
    """

    def __init__(self, persist_path: str = "./docstore.pkl"):
        self.persist_path = Path(persist_path)
        # Load existing data if present, otherwise start with an empty dict
        if self.persist_path.is_file():
            with self.persist_path.open("rb") as f:
                data = pickle.load(f)
        else:
            data = {}
        super().__init__(data)

    def _persist(self) -> None:
        """Write the internal ``_dict`` to ``persist_path``.

        The method is called after any mutation operation (``add`` or ``delete``).
        ``InMemoryStore`` already provides ``add`` and ``delete`` methods, so we
        simply override them to add persistence.
        """
        with self.persist_path.open("wb") as f:
            pickle.dump(self._dict, f)

    # ``InMemoryStore`` implements ``add`` and ``delete``; we hook into them.
    def add(self, texts, ids=None, **kwargs):  # type: ignore[override]
        result = super().add(texts, ids=ids, **kwargs)
        self._persist()
        return result

    def delete(self, id: str) -> None:  # type: ignore[override]
        super().delete(id)
        self._persist()

    # ``InMemoryStore`` also provides ``mset`` (multiple set) used by the
    # retriever when adding documents.  Hooking into ``mset`` ensures the store
    # stays in sync.
    def mset(self, items: dict) -> None:  # type: ignore[override]
        super().mset(items)
        self._persist()
