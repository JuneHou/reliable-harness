import os
from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage.file_system import LocalFileStore
from langchain.schema import Document
import nltk


def run_agent(query: str, store_location: str = "./docstore") -> list[Document]:
    """Run a simple retrieval‑augmented generation (RAG) example.

    This function demonstrates how to keep the *parent‑document* mapping
    (the ``docstore``) persistent across runs.  Instead of using the default
    ``InMemoryStore`` we use ``LocalFileStore`` which writes its contents to the
    filesystem.  The retriever returned by :class:`ParentDocumentRetriever`
    automatically looks up the parent documents for the most relevant child
    chunks, so we can simply call ``retriever.get_relevant_documents`` to obtain
    the full parent texts.

    Args:
        query: The user query used for retrieval.
        store_location: Directory where the persistent docstore will be saved.

    Returns:
        A list of ``Document`` objects representing the parent documents that
        match the query.
    """
    # Ensure the NLTK tokenizer is available (only needs to run once).
    nltk.download("punkt", quiet=True)

    # ---------------------------------------------------------------------
    # Embeddings
    # ---------------------------------------------------------------------
    hf_embed_path = "./hf_cache"
    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path,
    )

    # ---------------------------------------------------------------------
    # Documents to index
    # ---------------------------------------------------------------------
    docs = [
        Document(
            page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."
        ),
        Document(
            page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."
        ),
        Document(
            page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."
        ),
    ]

    # ---------------------------------------------------------------------
    # Vector store for the *child* chunks (persisted with Chroma)
    # ---------------------------------------------------------------------
    child_vectorstore = Chroma(
        collection_name="child_chunks",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child",
    )

    # ---------------------------------------------------------------------
    # Persistent docstore for parent documents
    # ---------------------------------------------------------------------
    # ``LocalFileStore`` serialises the documents to the given directory so they
    # survive process termination.
    parent_docstore = LocalFileStore(path=store_location)

    # ---------------------------------------------------------------------
    # ParentDocumentRetriever – combines the two layers
    # ---------------------------------------------------------------------
    child_splitter = NLTKTextSplitter(chunk_size=400)
    retriever = ParentDocumentRetriever(
        vectorstore=child_vectorstore,
        docstore=parent_docstore,
        child_splitter=child_splitter,
    )

    # Index the documents (this populates both the vector store and the docstore)
    retriever.add_documents(docs)

    # ---------------------------------------------------------------------
    # Retrieval – ask the retriever for the most relevant parent docs
    # ---------------------------------------------------------------------
    # ``k`` controls how many parent documents are returned.
    matching_parents = retriever.get_relevant_documents(query, k=3)
    return matching_parents


if __name__ == "__main__":
    query = "What is RAG?"
    result = run_agent(query)
    for doc in result:
        print("--- Parent Document ---")
        print(doc.page_content)

    )

    docstore = LocalFileStore("./docstore")

    # Retriever that keeps a mapping between parent docs and their child chunks

        docstore=docstore,
        child_splitter=child_splitter
    )

    # Add parent documents (this also creates and stores child chunks in the docstore)
    retriever.add_documents(docs)

    # Persist the vectorstore so that it can be re‑loaded in future runs
    vectorstore.persist()

    # Perform a similarity / MMR search on the persisted vector store
    matching_docs = vectorstore.max_marginal_relevance_search(query, k=3)
    return matching_docs
