def run_agent(documents):
    """Creates a Chroma collection, adds documents with embeddings, and retrieves them.

    Args:
        documents (list[str]): A list of exactly two document strings.

    Returns:
        dict: The collection contents, including embeddings, documents, metadatas, and ids.
    """
    if len(documents) != 2:
        print("number of documents should be 2")
        return

    # Initialize the Chroma client and create a collection.
    chroma_client = chromadb.Client()
    collection = chroma_client.create_collection(name="my_collection")

    # Add the records with embeddings, metadata, and ids.
    collection.add(
        embeddings=[[1.2, 2.3, 4.5], [6.7, 8.2, 9.2]],
        documents=documents,
        metadatas=[{"source": "my_source"}, {"source": "my_source"}],
        ids=["id1", "id2"],
    )

    # Retrieve the full records, explicitly requesting the embeddings.
    return collection.get(include=["embeddings", "documents", "metadatas", "ids"])
