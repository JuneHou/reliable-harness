from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings

load_dotenv()


def run_agent(query: str, k: int = 5) -> str:
    """Return the *k* most similar rows from ``control.csv`` for the given *query*.

    The function:
    1. Loads the CSV where each row becomes a ``Document``.
    2. Embeds the documents with ``OpenAIEmbeddings``.
    3. Stores them in a Chroma vectorstore.
    4. Performs a similarity search that also returns the cosine distance scores.
    5. Formats the results as a human‑readable string.

    Args:
        query: The natural‑language query (e.g., "Find movies for family").
        k: Number of top similar rows to return.

    Returns:
        A string containing each matching row and its similarity score.
    """
    # 1. Load CSV – each row becomes a Document with the raw CSV line as its content.
    loader = CSVLoader(file_path="control.csv")
    documents = loader.load()

    # 2‑3. Create a Chroma vectorstore from the documents.
    #    ``OpenAIEmbeddings`` will call the OpenAI embedding endpoint.
    vectorstore = Chroma.from_documents(documents=documents, embedding=OpenAIEmbeddings())

    # 4. Perform similarity search *with* scores.
    #    ``similarity_search_with_score`` returns a list of (Document, float) where the
    #    float is the cosine distance (lower = more similar).
    results = vectorstore.similarity_search_with_score(query, k=k)

    # 5. Format the output.
    if not results:
        return "No similar entries were found."

    formatted_lines = []
    for doc, score in results:
        # ``score`` is a distance; convert to a similarity-like value if desired.
        similarity = 1 - score  # simple conversion to a 0‑1 similarity range
        # Clean up the document content – remove excessive whitespace.
        content = " ".join(doc.page_content.split())
        formatted_lines.append(f"• {content} (similarity: {similarity:.3f})")

    return "\n".join(formatted_lines)

    The original implementation built a RAG chain that fed the retrieved chunks to an LLM.
    That approach never exposed the similarity scores and relied on the model to format the answer,

    control pairs.  The fix below removes the unnecessary LLM step and uses ``similarity_search_with_score``

    """

    # 1. Load CSV rows as LangChain ``Document`` objects.
    loader = CSVLoader(file_path="control.csv")
    documents = loader.load()

    # 2. (Optional) Split long rows into manageable chunks – useful if any cell contains a large amount of text.
    splitter = CharacterTextSplitter(chunk_size=500, chunk_overlap=50)
    chunks = splitter.split_documents(documents)

    # 3. Create a vector store backed by Chroma.
    vectorstore = Chroma.from_documents(documents=chunks, embedding=OpenAIEmbeddings())

    # 4. Perform a similarity search *with* the score for each hit.
    #    ``similarity_search_with_score`` returns a list of ``(Document, float)`` tuples where the float is

        # The distance returned by Chroma is typically Euclidean; we transform it to a similarity score.

        similarity = 1 / (1 + distance) if distance >= 0 else 0.0
        formatted_results.append(

    # 6. Join all entries into a single string for the caller.
