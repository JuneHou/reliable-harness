def _retrieve_with_score(vectorstore: Chroma, query: str, k: int = 10):
    """Return a list of (Document, score) tuples ordered by decreasing similarity.
    ``Chroma`` exposes ``similarity_search_with_score`` which already returns the
    cosine similarity (the lower the score, the more similar).  For a user‑friendly
    display we invert the score so that higher numbers mean higher relevance.
    """
    docs_scores = vectorstore.similarity_search_with_score(query, k=k)
    # Invert the distance to a similarity-like value (1 / (1 + distance))
    # This keeps the values in a 0‑1 range and makes them easier to read.
    return [(doc, 1 / (1 + score)) for doc, score in docs_scores]


def _format_docs_with_score(docs_scores):
    """Create a human‑readable string that lists each document together with
    its similarity score.
    The CSVLoader stores the original column values in ``metadata`` – we assume
    the CSV has columns named ``Name`` and ``Description``.  If those keys are not
    present we fall back to an empty string.
    """
    lines = []
    for i, (doc, sim) in enumerate(docs_scores, start=1):
        name = doc.metadata.get("Name", "")
        description = doc.page_content.strip()
        lines.append(
            f"{i}. Name: {name}\n   Description: {description}\n   Similarity: {sim:.3f}"
        )
    return "\n".join(lines)
