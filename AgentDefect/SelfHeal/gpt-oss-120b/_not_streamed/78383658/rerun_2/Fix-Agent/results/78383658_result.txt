def run_agent(query: str, top_k: int = 10) -> str:
    """Return a formatted list of the *top_k* most similar rows from ``control.csv``
    together with their similarity scores.

    The original implementation used a generic RAG QA chain, which is suitable for
    answering open‑ended questions but does **not** expose the similarity scores that
    the user expects.  Here we bypass the LLM chain and directly call
    ``similarity_search_with_score`` on the Chroma vector store.  This returns a list
    of ``(Document, score)`` tuples where a lower score means a higher similarity.
    The function formats the results as a numbered list, showing the relevant CSV
    fields (``Name`` and ``Description`` in this example) and the distance score.
    """

    # Load CSV and split into documents (each row becomes a document)
    loader = CSVLoader(file_path="control.csv")
    data = loader.load()
    # Split large rows if needed – keep a reasonable chunk size.
    text_splitter = CharacterTextSplitter(chunk_size=500, chunk_overlap=50)
    chunks = text_splitter.split_documents(data)

    # Build the vector store.  ``OpenAIEmbeddings`` pulls the API key from the
    # environment (handled by ``load_dotenv``).
    vectorstore = Chroma.from_documents(documents=chunks, embedding=OpenAIEmbeddings())

    # Retrieve the most similar documents together with their distance scores.
    # ``similarity_search_with_score`` returns a list of (Document, float) tuples.
    results = vectorstore.similarity_search_with_score(query, k=top_k)

    if not results:
        return "No similar control pairs found."

    # Helper to extract the fields we care about – the CSV loader stores the raw
    # row in ``metadata`` under the key ``row``.
    def _format_doc(doc, score, idx):
        row = doc.metadata.get("row", {})
        # Adjust the field names to match the CSV columns you care about.
        name = row.get("Name", "<no name>")
        description = row.get("Description", "<no description>")
        # Chroma returns *distance*; converting to a similarity‑like score (1/(1+dist))
        # makes the output more intuitive.
        similarity = 1 / (1 + score) if score is not None else 0
        return f"{idx}. Name: {name}\n   Description: {description}\n   Similarity score: {similarity:.4f}"

    formatted = [
        _format_doc(doc, score, i + 1) for i, (doc, score) in enumerate(results)
    ]

    return "\n\n".join(formatted)

    """Return a list of the *top_k* most similar rows from ``control.csv`` together with
    their similarity scores.

    The function:
    1. Loads the CSV as a list of ``Document`` objects (one per row).
    2. Stores the documents in a Chroma vector store using ``OpenAIEmbeddings``.
    3. Performs a *similarity* search that returns both the document and the raw
       cosine‑similarity score.
    4. Formats the results as a concise, human‑readable list.
    """

    # 1. Load the CSV – each row becomes a Document whose ``page_content`` is a
    #    string representation of the row (e.g. "Name: Foo, Description: Bar").
    loader = CSVLoader(file_path="control.csv")
    docs = loader.load()

    # 2. (Optional) Split long rows – for most CSVs the rows are short enough,
    #    but we keep the splitter to stay close to the original pipeline.
    splitter = CharacterTextSplitter(chunk_size=500, chunk_overlap=50)
    chunks = splitter.split_documents(docs)

    # 3. Create / load a Chroma collection.
    vectorstore = Chroma.from_documents(documents=chunks, embedding=OpenAIEmbeddings())

    # 4. Perform similarity search *with* scores.
    #    ``similarity_search_with_score`` returns a list of (Document, float) tuples.
    results = vectorstore.similarity_search_with_score(query, k=top_k)

    # 5. Build a readable output.
    if not results:
        return "No similar control pairs were found."

    # Sort by descending similarity (higher score = more similar).
    results.sort(key=lambda tup: tup[1], reverse=True)

    formatted_lines = []
    for idx, (doc, score) in enumerate(results, start=1):
        # Keep the original CSV row (doc.page_content) and show the score rounded
        # to three decimal places for readability.
        formatted_lines.append(f"{idx}. {doc.page_content.strip()} (score: {score:.3f})")

    # Join everything into a single string.
    output = "Top similar control pairs:\n" + "\n".join(formatted_lines)
    return output
