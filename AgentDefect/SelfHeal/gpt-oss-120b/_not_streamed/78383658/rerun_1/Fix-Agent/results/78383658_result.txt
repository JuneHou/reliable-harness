import os
from dotenv import load_dotenv

from langchain.document_loaders import CSVLoader
from langchain.text_splitter import CharacterTextSplitter
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.runnable import RunnablePassthrough, RunnableLambda
from langchain.schema.output_parser import StrOutputParser

load_dotenv()

def _format_retrieved(docs, distances):
    """Create a human‑readable string that lists each document together with its similarity score.
    The distance returned by Chroma is a Euclidean distance; we convert it to a similarity
    score between 0 and 1 with ``score = 1 / (1 + distance)`` (a simple monotonic transform).
    """
    lines = []
    for i, (doc, dist) in enumerate(zip(docs, distances)):
        name = doc.metadata.get("Name", "<no name>")
        description = doc.page_content.strip().replace("\n", " ")
        similarity = 1 / (1 + dist)  # transform distance → similarity
        lines.append(f"{i+1}. Name: {name}\n   Description: {description}\n   Similarity: {similarity:.3f}")
    return "\n\n".join(lines)

def run_agent(query: str, top_k: int = 10):
    # 1. Load CSV – each row becomes a Document with metadata containing the column names.
    loader = CSVLoader(file_path="control.csv")
    data = loader.load()

    # 2. Split – the CSV rows are short; a simple splitter avoids breaking a row in two.
    text_splitter = CharacterTextSplitter(chunk_size=500, chunk_overlap=0)
    chunks = text_splitter.split_documents(data)

    # 3. Vector store – embed the chunks.
    vectorstore = Chroma.from_documents(documents=chunks, embedding=OpenAIEmbeddings())

    # 4. Retriever – ask for the *top_k* most similar chunks.
    retriever = vectorstore.as_retriever(search_kwargs={"k": top_k})

    # 5. Get raw documents and the underlying distances.
    # ``retriever.invoke`` returns only the Document objects, not the distances.
    # We therefore call the underlying Chroma collection directly to also fetch distances.
    docs = retriever.invoke(query)
    # Each Document created by the CSV loader stores a hidden ``_id`` in its metadata.
    # Chroma uses the same id to look up distances.
    ids = [doc.metadata.get("_id") or doc.metadata.get("id") for doc in docs]
    # Fallback: if ids are not present (some loaders do not set them), we can retrieve
    # distances by re‑querying the collection with ``include=['distances']``.
    collection = vectorstore._collection
    results = collection.get(ids=ids, include=["distances", "metadatas", "documents"])
    distances = results.get("distances", [])

    # 6. Build a prompt that simply asks the LLM to present the retrieved rows in a
    #    concise bullet‑point list.  The LLM is kept as a formatter; the heavy lifting
    #    (similarity search) is done by the vector store.
    template = """You are an assistant that formats a list of control pairs.

Provide the information exactly as supplied, preserving the order of relevance.
If the list is empty, reply with "No matching controls found."

{retrieved}
"""
    prompt = ChatPromptTemplate.from_template(template)
    llm = ChatOpenAI(temperature=0, model="gpt-3.5-turbo", verbose=False)

    # 7. Assemble the LangChain runnable chain.
    chain = (
        RunnablePassthrough()
        | RunnableLambda(lambda q: {
            "retrieved": _format_retrieved(docs, distances)
        })
        | prompt
        | llm
        | StrOutputParser()
    )

    # 8. Invoke the chain with the original query (the query itself is not used in the
    #    prompt beyond being passed through – the similarity work has already happened).
    return chain.invoke(query)

if __name__ == "__main__":
    query = "Find movies for family, combining their Name and Description."
    result = run_agent(query, top_k=10)
    print(result)
