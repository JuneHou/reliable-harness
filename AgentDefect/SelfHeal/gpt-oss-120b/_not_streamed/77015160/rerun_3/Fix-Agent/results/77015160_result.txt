from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.prompts import PromptTemplate
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def run_agent(question: str) -> str:
    """Run a retrieval‑augmented QA chain with a custom (blank) prompt.

    Args:
        question: The query to ask the model.

    Returns:
        The model's answer including source citations.
    """
    # Initialise LLM
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    # Simple in‑memory docs for the demo
    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    # Build vector store
    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # Define a custom prompt – here we use an empty template so no default
    # system message or instruction is added. Only the user question and the
    # retrieved context are passed to the LLM.
    custom_prompt = PromptTemplate(
        input_variables=["question", "context"],
        template="{question}\n\nContext:\n{context}"  # minimal template
    )

    # Build the RetrievalQAWithSourcesChain with the custom prompt
    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        chain_type_kwargs={"prompt": custom_prompt},
    )

    # Execute the chain
    result = chain({"question": question})
    # The chain returns a dict with keys like 'answer' and 'source_documents'
    return result.get("answer", "")

if __name__ == "__main__":
    # Example usage – replace with any query you like
    print(run_agent("What countries are mentioned in the documents?"))
