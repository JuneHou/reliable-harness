def run_agent(question: str) -> str:
    """Create a RetrievalQAWithSourcesChain that uses a custom (blank) prompt.

    The default LangChain prompt/template that is shown with every answer is replaced
    by a minimal template that only inserts the question and the retrieved summaries.
    This prevents the default wording (e.g., "Answer the question based on the
    following context:") from appearing in the output.

    Args:
        question: The user query to answer.

    Returns:
        The answer string generated by the LLM.
    """
    # Initialise the LLM.
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    # Sample documents and their metadata.
    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    # Create a vector store.
    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # Define a minimal/custom prompt.  The "summaries" variable will contain the
    # combined retrieved documents; "question" is the user query.
    custom_prompt = PromptTemplate(
        input_variables=["summaries", "question"],
        template="{question}\n\n{summaries}"
    )

    # Build the RetrievalQAWithSourcesChain, injecting the custom prompt.
    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        combine_documents_chain_kwargs={"prompt": custom_prompt},
    )

    # Run the chain on the supplied question and return the answer.
    return chain.run(question)
