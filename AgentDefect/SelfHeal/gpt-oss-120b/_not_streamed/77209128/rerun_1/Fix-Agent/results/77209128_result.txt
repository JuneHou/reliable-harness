import os
import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer

from llama_index.core import SQLDatabase, VectorStoreIndex, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

load_dotenv()

OPENAI_API_KEY = os.environ.get("OPENAI_API_KEY")
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

pg_uri = os.environ.get("DATABASE_URL", "sqlite:///test.db")

# Define custom field‑level metadata that we want LlamaIndex to surface during queries.
metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": (
        "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
    )
}

# Serialize the metadata once – we will embed it in the SQLTableSchema objects so the
# query engine can retrieve it as part of the context string.
metadata_json = json.dumps(metadata_dict)


class SqlQueryEngine:
    def __init__(self):
        # This will hold the SQLTableSchema objects that describe each table.
        self.schema: list[SQLTableSchema] = []

    def _create_snapshot_table(self, engine) -> None:
        """Create a simple snapshot table with a JSON column.

        The column comment *is* set to the raw JSON string; however, LlamaIndex does not
        automatically read column comments.  Instead we store the same JSON string in the
        ``context_str`` attribute of ``SQLTableSchema`` objects (see ``_build_schema``).
        """
        metadata_obj = MetaData()
        snapshots_table = Table(
            "snapshots",
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
        )
        metadata_obj.create_all(engine)
        # Keep a reference to the metadata object for later schema extraction.
        self._metadata_obj = metadata_obj

    def _build_schema(self) -> None:
        """Populate ``self.schema`` with ``SQLTableSchema`` objects.

        ``SQLTableSchema`` accepts a ``context_str`` argument which LlamaIndex uses as
        additional context for the vector store.  We pass the serialized ``metadata_dict``
        so that natural‑language queries about a field can be answered.
        """
        for table_name in self._metadata_obj.tables.keys():
            schema_item = SQLTableSchema(
                table_name=table_name,
                # The context string is free‑form text; we simply embed the JSON.
                context_str=metadata_json,
            )
            self.schema.append(schema_item)

    def query_engine(self):
        # ---------------------------------------------------------------------
        # 1️⃣ Initialise the underlying DB engine and create the snapshot table.
        # ---------------------------------------------------------------------
        engine = create_engine(pg_uri)
        self._create_snapshot_table(engine)

        # ---------------------------------------------------------------------
        # 2️⃣ Build the LlamaIndex SQL database wrapper.
        # ---------------------------------------------------------------------
        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        # ---------------------------------------------------------------------
        # 3️⃣ Build the schema objects that will be stored in the vector index.
        # ---------------------------------------------------------------------
        self._build_schema()

        # ---------------------------------------------------------------------
        # 4️⃣ Create a VectorStoreIndex from the schema objects (used as a retriever).
        # ---------------------------------------------------------------------
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        # ---------------------------------------------------------------------
        # 5️⃣ Initialise the LLM and the query engine.
        # ---------------------------------------------------------------------
        llm = OpenAI(temperature=0.9, model="gpt-3.5-turbo")
        Settings.llm = llm

        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )
        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")
    return sql_engine, response


if __name__ == "__main__":
    # Example query that asks about the custom field description.
