import os
import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer

from llama_index.core import SQLDatabase, VectorStoreIndex, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

# Load environment variables
load_dotenv()

OPENAI_API_KEY = os.environ.get("OPENAI_API_KEY")
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

# Database connection string – defaults to SQLite for local testing
pg_uri = os.environ.get("DATABASE_URL", "sqlite:///test.db")

# Example metadata that we want to expose as field‑level documentation
metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": (
        "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
    )
}


class SqlQueryEngine:
    """Wraps the creation of a LlamaIndex SQL query engine.

    The key change compared to the original snippet is that we now populate the
    ``context_str`` attribute of ``SQLTableSchema`` with the column comment stored
    in the database. LlamaIndex uses ``context_str`` as additional context when
    building the prompt for the LLM, so the description becomes available to the
    model during query time.
    """

    def __init__(self):
        # This will hold ``SQLTableSchema`` objects – one per table.
        self.schema: list[SQLTableSchema] = []

    def _create_snapshot_table(self, engine) -> None:
        """Creates the ``snapshots`` table with a JSON column that contains the
        custom metadata as a comment.
        """
        metadata_obj = MetaData()
        # Convert the metadata dictionary to a JSON string for the column comment.
        metadata_json = json.dumps(metadata_dict)
        Table(
            "snapshots",
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
        )
        # Actually create the table in the database.
        metadata_obj.create_all(engine)
        # Store the metadata object for later schema extraction.
        self._metadata_obj = metadata_obj

    def _build_schema_objects(self) -> None:
        """Build ``SQLTableSchema`` objects that include the column comment.
        ``SQLTableSchema`` accepts a ``context_str`` argument – this is where we
        inject the field‑level description so the LLM can answer questions like
        *"What does the page_call_phone_clicks_logged_in_unique field mean?"*.
        """
        for table in self._metadata_obj.tables.values():
            # Retrieve the comment attached to the ``data`` column (if present).
            column_comment = None
            if "data" in table.c:
                column_comment = table.c["data"].comment

            # Build a human‑readable context string.
            context = (
                f"Column 'data' holds JSON payloads.\n"
                f"Metadata comment: {column_comment}" if column_comment else None
            )

            self.schema.append(
                SQLTableSchema(table_name=table.name, context_str=context)
            )

    def query_engine(self) -> SQLTableRetrieverQueryEngine:
        # Initialise the DB engine.
        engine = create_engine(pg_uri)
        # Create the snapshot table with its comment.
        self._create_snapshot_table(engine)

        # Build the schema objects that include the comment as context.
        self._build_schema_objects()

        # Initialise LlamaIndex helpers.
        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        # Create a VectorStore index from the schema objects.
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        # Configure the LLM.
        llm = OpenAI(temperature=0.9, model="gpt-3.5-turbo")
        Settings.llm = llm

        # Assemble the retriever‑based query engine.
        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )
        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response
