import os
import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer

from llama_index.core import SQLDatabase, Settings, VectorStoreIndex
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

load_dotenv()

OPENAI_API_KEY = os.environ.get('OPENAI_API_KEY')
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

pg_uri = os.environ.get('DATABASE_URL', 'sqlite:///test.db')

# Dictionary that maps a field name to its human‑readable description.
# This will be stored as a JSON comment on the ``data`` column and later
# injected into the ``context_str`` of the ``SQLTableSchema`` objects so
# that LlamaIndex can surface the description during retrieval.
metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
}

class SqlQueryEngine:
    """Factory that builds a LlamaIndex SQL query engine capable of reading
    custom JSON field metadata.
    """

    def __init__(self):
        # Will hold ``SQLTableSchema`` objects enriched with our field
        # descriptions.  The list is populated in ``query_engine``.
        self.schema: list[SQLTableSchema] = []

    def _create_snapshot_table(self, engine) -> Table:
        """Create (or reflect) the ``snapshots`` table with a JSON column that
        stores the supplied ``metadata_dict`` as a comment.
        """
        metadata_obj = MetaData()
        # Convert the metadata dictionary to a JSON string for the column comment.
        metadata_json = json.dumps(metadata_dict)
        snapshots_table = Table(
            "snapshots",
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
        )
        # Create the table in the underlying DB if it does not already exist.
        metadata_obj.create_all(engine)
        return snapshots_table

    def query_engine(self) -> SQLTableRetrieverQueryEngine:
        # -----------------------------------------------------------------
        # 1️⃣ Initialise the SQLAlchemy engine and build the snapshots table.
        # -----------------------------------------------------------------
        engine = create_engine(pg_uri)
        snapshots_table = self._create_snapshot_table(engine)

        # --------------------------------------------------------------
        # 2️⃣ Wrap the SQLAlchemy engine with LlamaIndex's SQLDatabase.
        # --------------------------------------------------------------
        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        # ------------------------------------------------------------
        # 3️⃣ Build a list of ``SQLTableSchema`` objects enriched with the
        #    column comment (our custom metadata).  LlamaIndex uses the
        #    ``context_str`` attribute when constructing the vector store.
        # ------------------------------------------------------------
        self.schema = []
        for table_name, table_obj in snapshots_table.metadata.tables.items():
            # Extract the comment we stored on the ``data`` column.  If the
            # column does not have a comment we fall back to an empty string.
            comment = (
                getattr(table_obj.c.get("data"), "comment", None) or ""
            )
            schema_item = SQLTableSchema(
                table_name=table_name,
                context_str=comment,
            )
            self.schema.append(schema_item)

        # --------------------------------------------------------
        # 4️⃣ Create a vector index over the enriched ``SQLTableSchema``
        #     objects.  ``ObjectIndex.from_objects`` expects the list of
        #     schema objects, the node mapping, and the concrete index class.
        # --------------------------------------------------------
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        # --------------------------------------------------------
        # 5️⃣ Configure the LLM (OpenAI) and attach it to LlamaIndex's
        #    ``Settings`` – this is the newer API (no need for LLMPredictor).
        # --------------------------------------------------------
        llm = OpenAI(temperature=0.9, model="gpt-3.5-turbo")
        Settings.llm = llm

        # --------------------------------------------------------
        # 6️⃣ Build the ``SQLTableRetrieverQueryEngine`` which will use the
        #    vector retriever (top‑k=1) to fetch the most relevant schema
        #    description before executing the actual SQL query.
        # --------------------------------------------------------
        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )
        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response


if __name__ == "__main__":

    sql_engine, response = run_agent(query)
