class SqlQueryEngine:
    """Build a LlamaIndex SQL‑Table‑Retriever query engine that includes column comments.

    The engine creates a simple ``snapshots`` table with a JSON column. The column's
    ``comment`` attribute is populated with a JSON string that maps column names to
    human‑readable descriptions. These comments are then extracted and added to the
    ``context_str`` of each :class:`SQLTableSchema` instance so that LlamaIndex can use
    them during retrieval.
    """

    def __init__(self) -> None:
        # Holds the generated ``SQLTableSchema`` objects for later inspection
        self.schema: list[SQLTableSchema] = []

    def _build_table_and_metadata(self, engine) -> MetaData:
        """Create the ``snapshots`` table and embed the custom metadata.

        Args:
            engine: SQLAlchemy engine connected to the target database.

        Returns:
            A ``MetaData`` object containing the newly created table.
        """
        metadata_obj = MetaData()
        # Store the column‑level metadata as a JSON string in the column comment
        column_comment = json.dumps(metadata_dict)
        Table(
            "snapshots",
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=column_comment),
        )
        # Actually create the table in the database
        metadata_obj.create_all(engine)
        return metadata_obj

    def _extract_schema_with_context(self, metadata_obj: MetaData) -> list[SQLTableSchema]:
        """Convert SQLAlchemy table objects into ``SQLTableSchema`` objects that include
        a ``context_str`` containing column comments.
        """
        schemas: list[SQLTableSchema] = []
        for table in metadata_obj.tables.values():
            # Build a readable context string from column comments (if any)
            column_infos = []
            for col in table.columns:
                # ``col.comment`` may be a JSON string – try to decode for readability
                comment = col.comment
                if comment:
                    try:
                        comment_dict = json.loads(comment)
                        # If the comment is a dict, format each mapping on its own line
                        if isinstance(comment_dict, dict):
                            comment = ", ".join(f"{k}: {v}" for k, v in comment_dict.items())
                    except Exception:
                        # Not JSON – keep raw string
                        pass
                column_infos.append(f"- {col.name}: {comment or 'No comment'}")
            context_str = f"Table `{table.name}` columns:\n" + "\n".join(column_infos)
            schemas.append(SQLTableSchema(table_name=table.name, context_str=context_str))
        return schemas

    def query_engine(self) -> SQLTableRetrieverQueryEngine:
        """Create and return a LlamaIndex ``SQLTableRetrieverQueryEngine`` instance.
        """
        engine = create_engine(pg_uri)
        # 1️⃣ Create the table and embed custom column comments
        metadata_obj = self._build_table_and_metadata(engine)

        # 2️⃣ Initialise LlamaIndex helpers
        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        # 3️⃣ Build schema objects that contain the comment‑rich context
        self.schema = self._extract_schema_with_context(metadata_obj)

        # 4️⃣ Create a vector index over the schema objects – this is used for table retrieval
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        # 5️⃣ Attach the LLM (OpenAI) to the global Settings (required by LlamaIndex)
        Settings.llm = OpenAI(temperature=0.0, model="gpt-3.5-turbo")

        # 6️⃣ Assemble the final query engine
        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )
        return query_engine


def run_agent(query: str):
