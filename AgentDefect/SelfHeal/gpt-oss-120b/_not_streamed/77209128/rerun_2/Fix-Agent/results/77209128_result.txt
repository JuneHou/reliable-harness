import os
import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer, inspect
from llama_index.core import SQLDatabase, VectorStoreIndex, Settings
from llama_index.core.objects import SQLTableNodeMapping, SQLTableSchema, ObjectIndex
from llama_index.core.query_engine import SQLTableRetrieverQueryEngine
from llama_index.llms.openai import OpenAI

load_dotenv()

OPENAI_API_KEY = os.environ.get('OPENAI_API_KEY')
if not OPENAI_API_KEY:
    raise ValueError("OPENAI_API_KEY not found in .env file")

pg_uri = os.environ.get('DATABASE_URL', 'sqlite:///test.db')

# Mapping of column name -> human‑readable description. This will be stored as a comment on the
# ``data`` JSON column and later exposed to LlamaIndex through the ``context_str`` field of
# ``SQLTableSchema``.
metadata_dict = {
    "page_call_phone_clicks_logged_in_unique": "Number of people who logged into Facebook and clicked the Call Now button. Ex: 2"
}

class SqlQueryEngine:
    """Create a simple SQL table with a JSON column that carries field‑level metadata.

    The metadata is attached as a column comment (SQLAlchemy ``comment`` argument).
    LlamaIndex's ``SQLTableSchema`` objects are then populated with a ``context_str``
    containing the JSON‑encoded metadata so that the query engine can answer questions
    like *"What does the X field mean?"*.
    """

    def __init__(self) -> None:
        self.schema: list[SQLTableSchema] = []

    def _create_table(self, engine) -> str:
        """Create the ``snapshots`` table and return its name.

        The ``data`` column stores arbitrary JSON payloads.  The column comment holds a
        JSON string that maps JSON keys to human‑readable descriptions.
        """
        metadata_obj = MetaData()
        table_name = "snapshots"
        # Store the metadata dictionary as a JSON string in the comment attribute.
        column_comment = json.dumps(metadata_dict)
        Table(
            table_name,
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=column_comment),
        )
        metadata_obj.create_all(engine)
        return table_name

    def _build_schema(self, engine, table_name: str) -> None:
        """Populate ``self.schema`` with ``SQLTableSchema`` objects.

        For each table we extract the column comment (if any) and use it as the
        ``context_str``.  LlamaIndex will embed this string in the vector index and
        retrieve it during query time.
        """
        inspector = inspect(engine)
        columns = inspector.get_columns(table_name)
        # Look for a comment on the JSON column.  If found, use it; otherwise use an empty JSON.
        comment_json = "{}"
        for col in columns:
            if col.get("comment"):
                comment_json = col["comment"]
                break
        self.schema.append(SQLTableSchema(table_name=table_name, context_str=comment_json))

    def query_engine(self):
        """Build the LlamaIndex query engine that can answer metadata questions."""
        engine = create_engine(pg_uri)
        table_name = self._create_table(engine)
        self._build_schema(engine, table_name)

        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        # Create a vector index from the schema objects (each carries the metadata as context).
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        # Configure the LLM.
        llm = OpenAI(temperature=0.9, model="gpt-3.5-turbo")
        Settings.llm = llm

        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )
        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")
    response = query_engine.query(query)
    print(f"Response: {response}")

    return sql_engine, response


if __name__ == "__main__":
    query = "What does the page_call_phone_clicks_logged_in_unique field mean in the data column?"
    sql_engine, response = run_agent(query)

    for schema_item in sql_engine.schema:
        print(f"\nTable: {schema_item.table_name}")
        if getattr(schema_item, "context_str", None):
            print(f"  context_str: {schema_item.context_str}")
            print("  Status: Metadata context is SET")
        else:
            print("  context_str: None")
            print("  Status: Metadata context is NOT SET")
