import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables and set up an in‑memory LLM cache
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# ---------------------------------------------------------------------------
# Helper to randomly pick an API key from a pool (useful when you have many).
# ---------------------------------------------------------------------------
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]

def get_random_key(key_pool: list[str]) -> str:
    """Return a random non‑empty API key from *key_pool*.

    LangChain raises an error if the key is ``None``; this helper guards
    against that situation.
    """
    return random.choice([k for k in key_pool if k])

# ---------------------------------------------------------------------------
# Main agent function – now supports an optional metadata filter that limits
# the retrieved documents to a specific source (e.g. a particular PDF).
# ---------------------------------------------------------------------------
def run_agent(query: str, pdf_paths: list[str], source_filter: dict | None = None):
    """Run a conversational retrieval over a collection of PDFs.

    Parameters
    ----------
    query: str
        The user question.
    pdf_paths: list[str]
        Paths to the PDF files that should be ingested.
    source_filter: dict | None, optional
        A metadata filter that will be passed to the underlying retriever.
        Example: ``{"source": "paper2.pdf"}`` restricts the search to that
        document only.
    """

    # -------------------------------------------------------------------
    # 1️⃣ Load the PDFs, split them into chunks and attach a ``source`` field
    #    to each chunk's metadata so we can filter on it later.
    # -------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()

        splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        chunks = splitter.split_documents(documents)

        # Add the originating file name to the metadata of each chunk.
        for chunk in chunks:
            chunk.metadata["source"] = pdf_path

        all_chunks.extend(chunks)

    # -------------------------------------------------------------------
    # 2️⃣ Create a Chroma vector store from the chunks.
    # -------------------------------------------------------------------
    db = Chroma.from_documents(all_chunks, embeddings)

    # -------------------------------------------------------------------
    # 3️⃣ Build a retriever that respects the optional filter.
    #    ``as_retriever`` returns a ``VectorStoreRetriever`` under the hood.
    # -------------------------------------------------------------------
    if source_filter:
        retriever = db.as_retriever(search_kwargs={"filter": source_filter})
    else:
        retriever = db.as_retriever()

    # -------------------------------------------------------------------
    # 4️⃣ Wire the LLM and the retriever together using ConversationalRetrievalChain.
    # -------------------------------------------------------------------
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(
            openai_api_key=get_random_key(OPENAI_API_KEY_POOL),
            cache=True,
            temperature=0,
        ),
        retriever=retriever,
        verbose=True,
        return_source_documents=True,
    )

    # -------------------------------------------------------------------
    # 5️⃣ Execute the chain.  ``chat_history`` is kept empty for this example.
    # -------------------------------------------------------------------
    result = qa_chain({"question": query, "chat_history": []})
    return result


if __name__ == "__main__":
    # Example usage – only information from ``paper2.pdf`` will be considered.
    query = "What is Theia?"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    filter_meta = {"source": "paper2.pdf"}

    result = run_agent(query, pdf_paths, source_filter=filter_meta)
    print(result["answer"])
