import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables and set LLM cache
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# A pool of API keys – you can extend this with more keys if you wish
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]


def get_random_key(key_pool: list) -> str:
    """Return a random, non‑None API key from the supplied pool."""
    return random.choice([k for k in key_pool if k is not None])


def run_agent(query: str, pdf_paths: list) -> dict:
    """Run a conversational retrieval QA over the supplied PDFs.

    Args:
        query: The user question.
        pdf_paths: List of paths to PDF files to be indexed.

    Returns:
        The result dictionary returned by ``ConversationalRetrievalChain`` –
        containing ``answer`` and ``source_documents`` (when ``return_source_documents``
        is ``True``).
    """
    # ---------------------------------------------------------------------
    # 1️⃣ Load PDFs, split them into chunks and attach a ``source`` metadata field
    # ---------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        docs = loader.load()
        splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        chunks = splitter.split_documents(docs)

        # Add a ``source`` tag so we can filter later
        for chunk in chunks:
            chunk.metadata["source"] = pdf_path
        all_chunks.extend(chunks)

    # ---------------------------------------------------------------------
    # 2️⃣ Create a Chroma vector store and obtain a *retriever* that supports filtering
    # ---------------------------------------------------------------------
    db = Chroma.from_documents(all_chunks, embeddings)
    # ``as_retriever`` returns a ``VectorStoreRetriever`` that respects the
    # ``search_kwargs`` we pass (including a metadata ``filter``).
    retriever = db.as_retriever(search_kwargs={})

    # ---------------------------------------------------------------------
    # 3️⃣ Build the ConversationalRetrievalChain – we keep ``return_source_documents``
    #    so we can see which documents were actually used.
    # ---------------------------------------------------------------------
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever=retriever,
        verbose=True,
        return_source_documents=True,
    )

    # ---------------------------------------------------------------------
    # 4️⃣ Apply a *dynamic* metadata filter before the query.  This demonstrates
    #    that we can change the filter on‑the‑fly without touching LangChain's
    #    source code.
    # ---------------------------------------------------------------------
    metadata_filter = {"source": "paper2.pdf"}  # only use chunks from paper2.pdf
    qa_chain.retriever.search_kwargs["filter"] = metadata_filter

    # ---------------------------------------------------------------------
    # 5️⃣ Execute the chain.
    # ---------------------------------------------------------------------
    result = qa_chain({"question": query, "chat_history": []})
    return result


if __name__ == "__main__":
    user_query = "What is Theia?"
    pdf_files = ["paper1.pdf", "paper2.pdf"]
    answer = run_agent(user_query, pdf_files)
    print(answer["answer"])  # The generated answer
