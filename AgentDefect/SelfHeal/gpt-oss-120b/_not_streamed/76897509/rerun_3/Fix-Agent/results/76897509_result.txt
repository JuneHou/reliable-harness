import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables and set up an in‑memory LLM cache
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# If you have multiple API keys you can rotate them; here we just pick the first non‑null one
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]

def get_random_key(key_pool: list[str]) -> str:
    """Return a random non‑empty API key from the pool."""
    return random.choice([k for k in key_pool if k])


def run_agent(query: str, pdf_paths: list[str]) -> dict:
    """Run a conversational retrieval chain that **only** looks at the PDF
    specified in ``metadata_filter``.

    The original code attempted to pass a ``filter`` argument to the chain at
    call time, but ``ConversationalRetrievalChain`` does not forward that value
    to the underlying retriever. The fix is to configure the retriever *before*
    constructing the chain (or mutate its ``search_kwargs`` per request).
    """
    # ---------------------------------------------------------------------
    # 1. Load all PDFs, split them into chunks and attach a ``source`` metadata
    # ---------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        docs = loader.load()
        splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        chunks = splitter.split_documents(docs)
        for chunk in chunks:
            # Store the originating file so we can filter on it later
            chunk.metadata["source"] = pdf_path
        all_chunks.extend(chunks)

    # ---------------------------------------------------------------------
    # 2. Create the vector store (Chroma) from the chunks
    # ---------------------------------------------------------------------
    db = Chroma.from_documents(all_chunks, embeddings)

    # ---------------------------------------------------------------------
    # 3. Build a retriever that *only* returns documents from ``paper2.pdf``.
    #    The filter is supplied via ``search_kwargs`` – this is the mechanism
    #    LangChain expects for metadata filtering.
    # ---------------------------------------------------------------------
    metadata_filter = {"source": "paper2.pdf"}
    retriever = db.as_retriever(search_kwargs={"filter": metadata_filter})

    # ---------------------------------------------------------------------
    # 4. Initialise the ConversationalRetrievalChain with the filtered retriever.
    # ---------------------------------------------------------------------
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(
            openai_api_key=get_random_key(OPENAI_API_KEY_POOL),
            cache=True,
            temperature=0,
        ),
        retriever=retriever,
        verbose=True,
        return_source_documents=True,
    )

    # ---------------------------------------------------------------------
    # 5. Execute the chain.  No ``filter`` key is needed in the input because the
    #    retriever already knows which documents are allowed.
    # ---------------------------------------------------------------------
    result = qa_chain({"question": query, "chat_history": []})
    return result


if __name__ == "__main__":
    # Example usage – make sure the two PDF files exist in the working dir.
    query = "What is Theia?"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    answer = run_agent(query, pdf_paths)
    print(answer["answer"])
