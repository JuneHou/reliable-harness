import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables and set an in‑memory LLM cache for faster debugging
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# If you have multiple API keys you can rotate them; here we just pull the first one
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]

def get_random_key(key_pool: list[str]) -> str:
    """Return a random non‑null API key from the pool."""
    return random.choice([k for k in key_pool if k])


def run_agent(query: str, pdf_paths: list[str], source_filter: str | None = None):
    """Run a conversational retrieval chain over a list of PDFs.

    Args:
        query: The user question.
        pdf_paths: Paths to the PDF files.
        source_filter: Optional metadata filter – only documents whose
            ``source`` metadata matches this string will be considered.
    """
    # ---------------------------------------------------------------------
    # 1. Load, split and annotate the PDF documents
    # ---------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        docs = loader.load()
        splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        chunks = splitter.split_documents(docs)
        # Store the originating file in the metadata – this is what we will filter on
        for chunk in chunks:
            chunk.metadata["source"] = pdf_path
        all_chunks.extend(chunks)

    # ---------------------------------------------------------------------
    # 2. Create a Chroma vector store from the chunks
    # ---------------------------------------------------------------------
    db = Chroma.from_documents(all_chunks, embeddings)

    # ---------------------------------------------------------------------
    # 3. Build a retriever.  We use ``as_retriever`` so we can pass a filter
    #    via ``search_kwargs``.  The filter can be changed later before each
    #    query, allowing *dynamic* per‑query filtering without touching LangChain
    #    internals.
    # ---------------------------------------------------------------------
    retriever = db.as_retriever(search_kwargs={})
    if source_filter is not None:
        # Apply the filter at creation time – this works for a static filter
        retriever.search_kwargs["filter"] = {"source": source_filter}

    # ---------------------------------------------------------------------
    # 4. Build the conversational retrieval chain
    # ---------------------------------------------------------------------
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(
            openai_api_key=get_random_key(OPENAI_API_KEY_POOL),
            cache=True,
            temperature=0,
        ),
        retriever=retriever,
        verbose=True,
        return_source_documents=True,
    )

    # ---------------------------------------------------------------------
    # 5. If a filter is supplied *per query* we inject it right before the call.
    #    ``ConversationalRetrievalChain`` forwards any ``filter`` key in the
    #    input dictionary to the underlying retriever, so this works out‑of‑the
    #    box.
    # ---------------------------------------------------------------------
    input_dict = {"question": query, "chat_history": []}
    if source_filter is not None:
        input_dict["filter"] = {"source": source_filter}

    result = qa_chain(input_dict)
    return result


if __name__ == "__main__":
    # Example usage – only documents originating from ``paper2.pdf`` are
    # consulted when answering the question.
    query_text = "What is Theia"
    pdf_files = ["paper1.pdf", "paper2.pdf"]
    answer = run_agent(query_text, pdf_files, source_filter="paper2.pdf")
    print(answer["answer"])
