import os
import random
from dotenv import load_dotenv, find_dotenv

from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables (including OPENAI_API_KEY)
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# ---------------------------------------------------------------------------
# Helper: pick a random non‑None API key from the pool.
# ---------------------------------------------------------------------------
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]

def get_random_key(key_pool: list[str]) -> str:
    """Return a random API key that is not ``None``.

    LangChain expects a concrete string; passing ``None`` triggers a runtime error.
    """
    return random.choice([k for k in key_pool if k is not None])

# ---------------------------------------------------------------------------
# Core agent – builds a Chroma vector store from the supplied PDFs and runs a
# conversational retrieval chain that **honours a metadata filter**.
# ---------------------------------------------------------------------------
def run_agent(query: str, pdf_paths: list[str], metadata_filter: dict | None = None):
    """Answer *query* using only the documents that match *metadata_filter*.

    Parameters
    ----------
    query: str
        The question to ask.
    pdf_paths: list[str]
        Paths to the PDF files that should be indexed.
    metadata_filter: dict | None, optional
        Metadata key/value pairs that the retriever should use to limit the
        search (e.g. ``{"source": "paper2.pdf"}``).  If ``None`` all documents
        are considered.
    """

    # ---------------------------------------------------------------------
    # 1️⃣  Load, split and annotate the PDFs.
    # ---------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        docs = loader.load()
        splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        chunks = splitter.split_documents(docs)
        # Attach the source filename as metadata – this is what we will filter on.
        for chunk in chunks:
            chunk.metadata["source"] = os.path.basename(pdf_path)
        all_chunks.extend(chunks)

    # ---------------------------------------------------------------------
    # 2️⃣  Create the vector store.
    # ---------------------------------------------------------------------
    db = Chroma.from_documents(all_chunks, embeddings)

    # ---------------------------------------------------------------------
    # 3️⃣  Build a *retriever* that can accept a dynamic ``filter`` argument.
    # ---------------------------------------------------------------------
    # ``as_retriever`` returns a ``VectorStoreRetriever`` that forwards any
    # ``search_kwargs`` straight to the underlying vector store.  By default we
    # start with an empty dict; we will inject the filter right before the call.
    retriever = db.as_retriever(search_kwargs={})

    # ---------------------------------------------------------------------
    # 4️⃣  Assemble the ConversationalRetrievalChain.
    # ---------------------------------------------------------------------
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(
            openai_api_key=get_random_key(OPENAI_API_KEY_POOL),
            cache=True,
            temperature=0,
        ),
        retriever=retriever,
        verbose=True,
        return_source_documents=True,
    )

    # ---------------------------------------------------------------------
    # 5️⃣  Apply the metadata filter *per‑query*.
    # ---------------------------------------------------------------------
    # The chain does **not** accept a ``filter`` key in its input dict, so we
    # modify the underlying retriever just before invocation.
    if metadata_filter is not None:
        # ``search_kwargs`` is read by the retriever each time ``get_relevant_documents``
        # is called, therefore updating it here limits the search to the supplied
        # metadata.
        retriever.search_kwargs["filter"] = metadata_filter
    else:
        # Ensure no stale filter remains from a previous call.
        retriever.search_kwargs.pop("filter", None)

    # ---------------------------------------------------------------------
    # 6️⃣  Run the chain.
    # ---------------------------------------------------------------------
    result = qa_chain({"question": query, "chat_history": []})
    return result

# ---------------------------------------------------------------------------
# Example usage.
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    # Ask a question while forcing the chain to look only at ``paper2.pdf``.
    query = "What is Theia?"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    filter_by_source = {"source": "paper2.pdf"}

    answer = run_agent(query, pdf_paths, metadata_filter=filter_by_source)
    print("Answer:", answer["answer"])
    print("Sources used:")
    for doc in answer.get("source_documents", []):
        print("-", doc.metadata.get("source"))
