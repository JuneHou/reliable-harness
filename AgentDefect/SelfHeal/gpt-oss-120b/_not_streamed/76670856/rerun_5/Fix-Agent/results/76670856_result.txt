import os
import sys
import json
from pathlib import Path
from dotenv import load_dotenv

from langchain.chat_models import ChatOpenAI
from langchain.chains import ConversationalRetrievalChain
from langchain.document_loaders import JSONLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.indexes import VectorstoreIndexCreator
from langchain.vectorstores import Chroma

load_dotenv()


def _create_or_load_index(persist: bool) -> VectorstoreIndexCreator:
    """Create a new vectorstore index from the review JSON or load a persisted one.

    Args:
        persist: Whether to persist the index to disk.

    Returns:
        The VectorstoreIndexCreator (which holds the underlying vectorstore).
    """
    if persist and os.path.isdir("persist"):
        # Load the persisted Chroma vectorstore directly.
        vectorstore = Chroma(persist_directory="persist", embedding_function=OpenAIEmbeddings())
        return VectorstoreIndexCreator(vectorstore=vectorstore)

    # Ensure the data directory exists and the review JSON is present.
    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)
    review_file = data_dir / "review.json"

    if not review_file.exists():
        payload = {
            "reviews": [
                {"text": "Great hotel, excellent service and comfortable rooms."},
                {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
                {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
                {"text": "Average hotel. The room was okay, but nothing special."},
                {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
                {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
                {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
                {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
                {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
                {"text": "Not a bad hotel, but there are better options available in the area."},
            ]
        }
        review_file.write_text(json.dumps(payload, indent=2))

    # Load each review's text directly. Using jq to pull the *text* field avoids the need for
    # the `content_key` argument and guarantees that the page content is a plain string.
    loader = JSONLoader(
        file_path=str(review_file),
        jq_schema=".reviews[].text",  # extracts each review's text as a separate document
    )

    if persist:
        # Persist the vectorstore so subsequent runs are fast.
        index_creator = VectorstoreIndexCreator(
            vectorstore_kwargs={"persist_directory": "persist"},
            embedding=OpenAIEmbeddings(),
        )
    else:
        index_creator = VectorstoreIndexCreator(embedding=OpenAIEmbeddings())

    return index_creator.from_loaders([loader])


def run_agent(queries, persist: bool = False):
    """Run a ConversationalRetrievalChain against the review data.

    Args:
        queries: Iterable of user queries.
        persist: Whether to reuse a persisted vectorstore.

    Returns:
        List of answers produced by the chain.
    """
    # Build or load the index.
    index = _create_or_load_index(persist)

    # Configure the retriever – we limit to a small number of relevant chunks to keep the
    # LLM's context window focused and improve answer quality.
    retriever = index.vectorstore.as_retriever(search_kwargs={"k": 4})

    # Initialise the conversational chain.
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model_name="gpt-3.5-turbo"),
        retriever=retriever,
        return_source_documents=False,
    )

    chat_history = []
    results = []
    for query in queries:
        if query.lower() in {"quit", "q", "exit"}:
            sys.exit()
        response = chain({"question": query, "chat_history": chat_history})
        answer = response["answer"]
        print(answer)
        results.append(answer)
        chat_history.append((query, answer))
    return results
