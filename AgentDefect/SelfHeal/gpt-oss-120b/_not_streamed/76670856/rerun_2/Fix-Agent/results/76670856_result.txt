def _ensure_review_file(path: Path) -> Path:

    The function returns the path to the JSON file so the caller can safely use
    it afterwards.
    """

            "reviews": [
                {"text": "Great hotel, excellent service and comfortable rooms."},
                {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
                {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
                {"text": "Average hotel. The room was okay, but nothing special."},
                {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
                {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
                {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
                {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
                {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
                {"text": "Not a bad hotel, but there are better options available in the area."},
            ]
        }
        path.write_text(json.dumps(payload, indent=2))
    return path

def run_agent(queries, persist: bool = False):
    """Run a conversational retrieval chain on a set of *queries*.

    Parameters
    ----------
    queries: list[str]
        The questions that should be asked to the LLM.
    persist: bool, optional
        If ``True`` the Chroma vector store is written to ``./persist`` and reused
        across calls.  This is useful for interactive sessions where the same data
        is queried many times.
    """
    load_dotenv()  # Load OPENAI_API_KEY from a .env file if present

    # ---------------------------------------------------------------------
    # 1️⃣ Load / generate the JSON source file
    # ---------------------------------------------------------------------
    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)
    review_file = _ensure_review_file(data_dir / "review.json")

    # ---------------------------------------------------------------------
    # 2️⃣ Build (or load) the vector store
    # ---------------------------------------------------------------------
    if persist and os.path.isdir("persist"):
        # Re‑use an existing persisted Chroma DB
        vectorstore = Chroma(persist_directory="persist", embedding_function=OpenAIEmbeddings())
        # ``VectorStoreIndexWrapper`` gives us the same ``.vectorstore`` attribute as the
        # object returned by ``VectorstoreIndexCreator`` – this keeps the rest of the
        # code unchanged.
        index = VectorstoreIndexCreator(vectorstore_kwargs={"persist_directory": "persist"})
        index.vectorstore = vectorstore
    else:
        # Load the JSON – each element of ``.reviews`` becomes one Document whose
        # ``page_content`` is the value of the ``text`` field.
        loader = JSONLoader(
            file_path=str(review_file),
            jq_schema=".reviews[]",
            content_key="text",
        )
        # ``VectorstoreIndexCreator`` builds a Chroma DB from the loader.  When
        # ``persist`` is ``True`` we ask Chroma to write the DB to disk.
        creator_kwargs = {"embedding": OpenAIEmbeddings()}
        if persist:
            creator_kwargs["vectorstore_kwargs"] = {"persist_directory": "persist"}
        index = VectorstoreIndexCreator(**creator_kwargs).from_loaders([loader])
        # If we are persisting, make sure the DB is flushed to disk now.
        if persist:
            index.vectorstore.persist()

    # ---------------------------------------------------------------------
    # 3️⃣ Configure the ConversationalRetrievalChain
    # ---------------------------------------------------------------------
    # ``ConversationBufferMemory`` keeps the chat history in a format that the
    # chain expects.  By passing the memory we no longer need to manually build a
    # ``chat_history`` list for each call.
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)

    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model_name="gpt-3.5-turbo"),
        retriever=index.vectorstore.as_retriever(search_kwargs={"k": 4}),
        memory=memory,
        # Returning source documents is extremely helpful for debugging and also
        # nudges the LLM to stay grounded in the supplied context.
        return_source_documents=True,
    )

    # ---------------------------------------------------------------------
    # 4️⃣ Run the provided queries
    # ---------------------------------------------------------------------
    results = []
    for query in queries:
        if query.lower() in {"quit", "q", "exit"}:
            sys.exit(0)
        # ``ConversationalRetrievalChain`` expects a dict with a ``question`` key.
        response = chain({"question": query})
        answer = response["answer"].strip()
        print(answer)
        results.append(answer)
        # The memory is automatically updated – no need to handle ``chat_history``
        # manually.
    return results
