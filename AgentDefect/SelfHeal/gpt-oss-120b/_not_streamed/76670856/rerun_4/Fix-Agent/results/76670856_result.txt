import os
import sys
import json
from pathlib import Path
from dotenv import load_dotenv

import openai
from langchain.chat_models import ChatOpenAI
from langchain.chains import ConversationalRetrievalChain
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.document_loaders import JSONLoader
from langchain.indexes import VectorstoreIndexCreator

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()


def _create_or_load_index(persist: bool) -> VectorstoreIndexCreator:
    """Create a new vector index from the JSON reviews or load a persisted one.

    Args:
        persist: If ``True`` the index will be written to disk under the ``persist``
            folder and reused on subsequent runs.
    Returns:
        A :class:`VectorstoreIndexCreator` that already contains the loaded documents.
    """
    # Prepare the data directory and the review JSON file
    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)
    review_file = data_dir / "review.json"

    # If the file does not exist, write a small dataset of hotel reviews.
    if not review_file.exists():
        payload = {
            "reviews": [
                {"text": "Great hotel, excellent service and comfortable rooms."},
                {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
                {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
                {"text": "Average hotel. The room was okay, but nothing special."},
                {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
                {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
                {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
                {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
                {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
                {"text": "Not a bad hotel, but there are better options available in the area."},
            ]
        }
        review_file.write_text(json.dumps(payload, indent=2))

    # ---------------------------------------------------------------------
    # Load the JSON file. ``jq_schema".reviews[]"`` iterates over each element of
    # the ``reviews`` array and yields a dict that contains a ``text`` key.
    # ``content_key='text'`` tells the loader that the actual searchable content
    # lives under that key.
    # ---------------------------------------------------------------------
    loader = JSONLoader(
        file_path=str(review_file),
        jq_schema=".reviews[]",
        content_key="text",
    )

    if persist:
        # Persist the Chroma collection so that subsequent runs are fast.
        return VectorstoreIndexCreator(
            vectorstore_kwargs={"persist_directory": "persist"},
            embedding=OpenAIEmbeddings(),
        ).from_loaders([loader])
    else:
        return VectorstoreIndexCreator(
            embedding=OpenAIEmbeddings()
        ).from_loaders([loader])


def run_agent(queries, *, persist: bool = False):
    """Run a ConversationalRetrievalChain over the review JSON.

    Args:
        queries: An iterable of user questions.
        persist: Whether to reuse a persisted Chroma index.
    Returns:
        A list with the textual answers produced by the model.
    """
    # Create or load the index. ``index`` is a ``VectorStoreIndexWrapper`` which
    # provides the ``as_retriever`` helper.
    index = _create_or_load_index(persist)

    # Build the conversational chain. We explicitly ask the retriever to return
    # a small number of relevant documents (k=4) to keep the context focused.
    retriever = index.as_retriever(search_kwargs={"k": 4})
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model="gpt-3.5-turbo"),
        retriever=retriever,
        return_source_documents=False,
    )

    chat_history = []
    answers = []
    for query in queries:
        if query.lower() in {"quit", "q", "exit"}:
            sys.exit()
        result = chain({"question": query, "chat_history": chat_history})
        answer = result["answer"]
        print(answer)
        answers.append(answer)
        # Store the turn so the model can maintain context.
        chat_history.append((query, answer))
    return answers


if __name__ == "__main__":
    # Allow a single command‑line argument or interactive usage.
    if len(sys.argv) > 1:
        # When called with a CLI argument we treat it as a one‑off query.
        run_agent([sys.argv[1]], persist=False)
    else:
        # Simple interactive loop.
        print("Enter questions about the hotel reviews (type 'exit' to quit):")
        while True:
            user_q = input("\n> ")
            if not user_q:
                continue
            run_agent([user_q], persist=False)
