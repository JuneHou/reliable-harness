from dotenv import load_dotenv
from pathlib import Path
import json
import os
import sys

import openai
from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.document_loaders import JSONLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.indexes import VectorstoreIndexCreator
from langchain.indexes.vectorstore import VectorStoreIndexWrapper
from langchain.vectorstores import Chroma

load_dotenv()

def run_agent(queries):
    """Run a conversational retrieval chain over a small JSON review dataset.

    The function creates (or re‑uses) a vector store from the JSON file and then
    answers each query while keeping track of the conversation history.

    Args:
        queries (list[str]): List of questions to ask.

    Returns:
        list[str]: List of answers returned by the chain.
    """

    # ---------------------------------------------------------------------
    # Prepare the data file – if it does not exist we create a simple JSON
    # structure containing a list of reviews.
    # ---------------------------------------------------------------------
    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)
    review_file = data_dir / "review.json"

    if not review_file.exists():
        payload = {
            "reviews": [
                {"text": "Great hotel, excellent service and comfortable rooms."},
                {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
                {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
                {"text": "Average hotel. The room was okay, but nothing special."},
                {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
                {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
                {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
                {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
                {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
                {"text": "Not a bad hotel, but there are better options available in the area."},
            ]
        }
        review_file.write_text(json.dumps(payload, indent=2))

    # ---------------------------------------------------------------------
    # Settings – we optionally persist the vector store to speed up subsequent
    # runs.  For this example we keep it in memory (PERSIST = False).
    # ---------------------------------------------------------------------
    PERSIST = False

    # ---------------------------------------------------------------------
    # Load (or re‑load) the vector store.
    # ---------------------------------------------------------------------
    if PERSIST and os.path.exists("persist"):
        print("Reusing index...\n")
        vectorstore = Chroma(
            persist_directory="persist", embedding_function=OpenAIEmbeddings()
        )
        index = VectorStoreIndexWrapper(vectorstore=vectorstore)
    else:
        # NOTE: The original code used `jq_schema=".reviews[]"` together with
        # `content_key='text'`.  That yields documents whose `page_content` looks
        # like "{'text': 'Great hotel, ...'}" – the text is wrapped in a Python
        # dict representation, which harms similarity search.  By extracting the
        # *text* field directly with the JQ expression ``.reviews[].text`` we store
        # clean, plain‑text strings in the vector store, resulting in accurate
        # retrieval.
        loader = JSONLoader(
            file_path=str(review_file),
            jq_schema=".reviews[].text",
        )

        if PERSIST:
            index = VectorstoreIndexCreator(
                vectorstore_kwargs={"persist_directory": "persist"}
            ).from_loaders([loader])
        else:
            index = VectorstoreIndexCreator().from_loaders([loader])

    # ---------------------------------------------------------------------
    # Build the ConversationalRetrievalChain.  The newer `ChatOpenAI` class
    # expects the argument `model_name` rather than `model`.
    # ---------------------------------------------------------------------
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model_name="gpt-3.5-turbo"),
        retriever=index.vectorstore.as_retriever(),
        return_source_documents=False,
    )

    # ---------------------------------------------------------------------
    # Run the provided queries while maintaining chat history.
    # ---------------------------------------------------------------------
    chat_history = []
    results = []
    for query in queries:
        if query.lower() in {"quit", "q", "exit"}:
            sys.exit()
        result = chain({"question": query, "chat_history": chat_history})
        answer = result["answer"]
        print(answer)
        results.append(answer)
        chat_history.append((query, answer))

    return results
