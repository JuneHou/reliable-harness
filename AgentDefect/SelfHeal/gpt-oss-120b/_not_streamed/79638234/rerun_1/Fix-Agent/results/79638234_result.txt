from langchain.schema import HumanMessage


def run_agent(input_text: str, temperature: float = 0.7) -> str:
    """Generate a response from an Ollama model with controllable randomness.

    Args:
        input_text: The user prompt.
        temperature: Sampling temperature (higher -> more random).
    Returns:
        The model's response as a plain string.
    """
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=temperature,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        timeout=120,
    )
    # Ollama's chat models expect a list of message objects.
    response = llm.invoke([HumanMessage(content=input_text)])
    return response.content
