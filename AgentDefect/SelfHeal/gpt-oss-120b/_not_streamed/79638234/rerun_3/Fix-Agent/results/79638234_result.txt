from langchain_ollama import ChatOllama
from langchain.schema import HumanMessage


def run_agent(input_text: str) -> str:
    """Generate a response from the Ollama model with randomness.

    Args:
        input_text: The user prompt.

    Returns:
        The model's reply as a string.
    """
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,  # non‑zero temperature enables sampling
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        seed=None,  # ensure the call is not deterministic
        timeout=120,
    )
    # LangChain chat models expect a list of messages, not a raw string.
    response = llm.invoke([HumanMessage(content=input_text)])
    return response.content
