from langchain_ollama import ChatOllama
from langchain.schema import HumanMessage


def run_agent(input_text: str) -> str:
    """Generate a response from the Ollama model.

    The function creates a new ``ChatOllama`` instance for every call (so no cached
    conversation history is reused) and sends the user's prompt as a ``HumanMessage``.
    Providing the prompt as a ``HumanMessage`` is the correct way to interact with
    LangChain's chat models and allows the ``temperature`` and ``top_p`` parameters
    to affect the sampling, producing varied answers for the same prompt.
    """
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        timeout=120,
    )
    # ``invoke`` expects a list of messages (or a single HumanMessage).
    response = llm.invoke([HumanMessage(content=input_text)])
    # ``response`` is a ``BaseMessage``; its textual content is stored in ``content``.
    return response.content


if __name__ == "__main__":
