import random
from langchain_ollama import ChatOllama

def run_agent(input_text: str) -> str:
    """Generate a response from the Ollama model with non‑deterministic output.

    A random ``seed`` is supplied on each call so that the model does not produce
    the same answer every time even when ``temperature`` is set.
    """

    llm = ChatOllama(

        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        seed=random_seed,  # ensure varied output
        timeout=120,
    )
    response = llm.invoke(input_text)
    return response.content
