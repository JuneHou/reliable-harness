from langchain_ollama import ChatOllama
import random

def run_agent(input_text: str) -> str:
    """Generate a response from an Ollama model with non‑deterministic sampling.

    A random ``seed`` is provided for each request. When a fixed seed (e.g. 0) is
    used the model will produce the same output for identical prompts even if the
    ``temperature`` is greater than zero. Supplying a different seed per call
    enables the temperature setting to take effect and yields varied answers.
    """
    # Pick a new random seed for every invocation so the temperature can influence sampling
    random_seed = random.randint(0, 2**31 - 1)

    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        seed=random_seed,  # ensure nondeterministic behaviour
        timeout=120,
    )
    response = llm.invoke(input_text)
    # ``response`` is a BaseMessage; extract its content safely
    return getattr(response, "content", str(response))

if __name__ == "__main__":
    print(run_agent("Please tell me a lighthearted joke"))
