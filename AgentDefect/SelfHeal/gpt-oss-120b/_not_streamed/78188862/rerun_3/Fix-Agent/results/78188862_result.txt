import os
import uuid
from datetime import datetime
from dotenv import load_dotenv
import chromadb
from chromadb.utils import embedding_functions
from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import CharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings

# Load environment variables (e.g., OpenAI API key)
load_dotenv()

config = {
    "OPENAI_API_KEY": os.getenv("OPENAI_API_KEY")
}

class configs:
    EMBEDDINGS_MODEL = "text-embedding-3-small"


class MockPDFFile:
    """Simple container to mimic an uploaded PDF file.

    Attributes
    ----------
    filepath: str
        Full path to the PDF on disk.
    filename: str
        Base name of the file (used as metadata).
    """

    def __init__(self, filepath: str, filename: str):
        self.filepath = filepath
        self.filename = filename


def read_docs(pdf_file: MockPDFFile):
    """Load a PDF and split it into manageable text chunks.

    Parameters
    ----------
    pdf_file: MockPDFFile
        The PDF to be processed.

    Returns
    -------
    List[Document]
        A list of LangChain Document objects representing the chunks.
    """
    pdf_loader = PyPDFLoader(pdf_file.filepath)
    pdf_documents = pdf_loader.load()

    text_splitter = CharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
    documents = text_splitter.split_documents(pdf_documents)
    return documents


def generate_and_store_embeddings(documents, pdf_file: MockPDFFile, user_id: str):
    """Create embeddings for *documents* and store them in a persistent Chroma collection.

    Each chunk receives its own UUID and per‑document metadata (source file,
    user, timestamp). The function returns a ``Chroma`` wrapper that can be used
    for subsequent queries.
    """
    # Initialise (or retrieve) the persistent client and collection.
    client = chromadb.PersistentClient(path="./trained_db")
    collection = client.get_or_create_collection(
        name="PDF_Embeddings",
        embedding_function=embedding_functions.OpenAIEmbeddingFunction(
            api_key=config["OPENAI_API_KEY"], model_name=configs.EMBEDDINGS_MODEL
        ),
    )

    now = datetime.now()
    # Build a *list* of metadata dicts – one entry per document.
    per_doc_metadata = [
        {
            "source": pdf_file.filename,
            "user": str(user_id),
            "created_at": now.strftime("%d/%m/%Y %H:%M:%S"),
        }
        for _ in documents
    ]

    # Unique IDs for each chunk.
    ids = [str(uuid.uuid4()) for _ in range(len(documents))]

    try:
        vectordb = Chroma.from_documents(
            documents,
            embedding=OpenAIEmbeddings(
                openai_api_key=config["OPENAI_API_KEY"], model=configs.EMBEDDINGS_MODEL
            ),
            persist_directory="./trained_db",
            collection_name=collection.name,
            client=client,
            ids=ids,
            metadatas=per_doc_metadata,
        )
        vectordb.persist()
    except Exception as err:
        print(f"An error occured: {err=}, {type(err)=}")
        return {"answer": "An error occured while generating embeddings. Please check terminal for more details."}

    return vectordb


def run_agent(pdf_path: str, user_id: str):
    """High‑level orchestration: load PDF, create embeddings, and return the DB wrapper.
    """
    pdf_file = MockPDFFile(pdf_path, os.path.basename(pdf_path))
    documents = read_docs(pdf_file)
    print(f"    ✓ Loaded {len(documents)} document chunks")

    vectordb = generate_and_store_embeddings(documents, pdf_file, user_id)
    if isinstance(vectordb, dict):
        # An error dictionary was returned.
        return None, pdf_file.filename
    return vectordb, pdf_file.filename


if __name__ == "__main__":
    pdf_path = None
    user_id = "test_user_123"

    # Locate a PDF file inside the optional ``docs`` folder.
    if os.path.isdir("docs"):
        for file in os.listdir("docs"):
            if file.lower().endswith(".pdf"):
                pdf_path = os.path.join("docs", file)
                break

    if not pdf_path:
        print("No PDF file found in docs folder")
        exit(1)

    vectordb, filename = run_agent(pdf_path, user_id)
    if vectordb is None:
        print("Agent failed to create embeddings")
        exit(1)

    # Retrieve *only* IDs and metadata for the chunks that belong to the given PDF.
    result = vectordb.get(
        where={"source": filename},
        include=["ids", "metadatas"]
    )
    print(f"Number of matching entries: {len(result.get('ids', []))}")
    print("IDs:")
    for i in result.get("ids", []):
        print(i)
    print("Metadata samples:")
    for meta in result.get("metadatas", [])[:3]:
        print(meta)
