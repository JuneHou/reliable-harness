import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import StructuredOutputParser, ResponseSchema

load_dotenv()


def get_output_parser():
    """Return a parser that expects a JSON list of dictionaries.

    Each dictionary should contain the keys ``identifier`` and ``text``.
    The parser will output a dict with a single key ``result`` that holds
    the list, e.g. ``{"result": [{"identifier": "id1", "text": "t1"}, ...]}``.
    """
    # Define a single response schema whose description tells the model
    # to output a *list* of objects.  The list will be stored under the key
    # ``result``.
    result_schema = ResponseSchema(
        name="result",
        description=(
            "A JSON array (list) where each element is an object containing "
            "the fields 'identifier' and 'text' extracted from the error "
            "message."
        ),
    )
    return StructuredOutputParser.from_response_schemas([result_schema])


def predict_result(msg: str, PROMPT_TEMPLATE: str):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    # Build the prompt. ``ChatPromptTemplate.from_template`` expects a string
    # with ``{variable}`` placeholders that match the kwargs passed to
    # ``format_messages``.
    prompt = ChatPromptTemplate.from_template(PROMPT_TEMPLATE)
    messages = prompt.format_messages(msg=msg, format_instructions=format_instructions)

    response = model.invoke(messages)

    # ``response.content`` holds the model's raw text output.  The parser will
    # convert it into a Python dict according to the schema defined above.
    parsed = output_parser.parse(response.content)
    # Extract the list of dictionaries from the parsed result.
    return parsed.get("result", [])
