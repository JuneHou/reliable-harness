import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import StructuredOutputParser, ResponseSchema

load_dotenv()


def get_output_parser():
    """Create a StructuredOutputParser that expects a *list* of dictionaries.

    LangChain's ``StructuredOutputParser`` works with JSON. By defining a single
    ``ResponseSchema`` whose description states that the value should be an
    array of objects (each object containing ``identifier`` and ``text``), the
    model will be instructed to emit a JSON list. The parser will then return
    that list directly, e.g. ``[{"identifier": "id1", "text": "t1"}, ...]``.
    """
    # The schema name can be anything – we call it "result". The description
    # makes it clear that the value must be a list of objects with the two
    # fields we care about.
    result_schema = ResponseSchema(
        name="result",
        description=(
            "A JSON array where each element is an object with two keys: "
            "'identifier' (the id) and 'text' (the associated text)."
        ),
    )

    output_parser = StructuredOutputParser.from_response_schemas([result_schema])
    return output_parser


def predict_result(msg: str, PROMPT_TEMPLATE: str):
    """Send *msg* to the LLM using *PROMPT_TEMPLATE* and parse the response.

    The function builds the prompt, calls the model, and parses the LLM's JSON
    output into a Python list of dictionaries.
    """
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    # ``ChatPromptTemplate.from_template`` expects a *single* template string.
    # We embed the placeholders ``{msg}`` and ``{format_instructions}``.
    prompt = ChatPromptTemplate.from_template(PROMPT_TEMPLATE)
    # ``format_messages`` returns a list of ``BaseMessage`` objects that can be
    # directly passed to ``model.invoke``.
    messages = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(messages)

    # ``response`` is a ``ChatMessage``; its ``content`` attribute holds the raw
    # text returned by the LLM.
    parsed = output_parser.parse(response.content)
    # ``parsed`` is a dict with a single key "result" that holds the list.
    return parsed["result"]
