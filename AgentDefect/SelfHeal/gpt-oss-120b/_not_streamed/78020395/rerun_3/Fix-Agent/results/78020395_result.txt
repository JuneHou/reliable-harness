import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import StructuredOutputParser, ResponseSchema

load_dotenv()


def get_output_parser() -> StructuredOutputParser:
    """Return a parser that expects a *list* of ``identifier``/``text`` objects.

    The LLM is instructed to output JSON with a single top‑level key
    ``results`` whose value is an array of objects, each containing the fields
    ``identifier`` and ``text``.  This matches the user's requirement of a
    *nested* JSON structure (list of dictionaries) instead of a flat dictionary.
    """
    results_schema = ResponseSchema(
        name="results",
        description=(
            "A list of objects extracted from the error message. Each object "
            "must contain two keys: 'identifier' (the id) and 'text' (the "
            "corresponding text)."
        ),
    )
    return StructuredOutputParser.from_response_schemas([results_schema])


def predict_result(msg: str, PROMPT_TEMPLATE: str):
    """Parse *msg* using the LLM and return a list of dictionaries.

    The function builds a prompt that includes the format instructions from the
    ``StructuredOutputParser``.  After invoking the model, the raw JSON is parsed
    and the ``results`` list is returned directly for convenience.
    """
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    # Build the prompt template – we keep placeholders for the message and the
    # format instructions.
    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    # ``format_messages`` will replace ``{msg}`` and ``{format_instructions}``
    # with the actual values.
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)

    response = model.invoke(message)
    # ``response.content`` holds the LLM's textual output.
    parsed = output_parser.parse(response.content)
    # ``parsed`` is a dict like {"results": [{...}, {...}]}.  Return the list.
    return parsed.get("results", [])
