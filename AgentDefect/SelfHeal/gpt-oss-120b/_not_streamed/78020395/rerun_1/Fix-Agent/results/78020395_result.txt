import os
from dotenv import load_dotenv
from typing import List

from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import JsonOutputParser
from pydantic import BaseModel, Field

load_dotenv()

# ---------------------------------------------------------------------------
# Pydantic models that describe the desired structured output.
# We want a *list* of dictionaries, each dictionary containing an
# ``identifier`` and a ``text`` field.  The top‑level model therefore has a
# single field ``items`` which is a list of ``Item`` objects.
# ---------------------------------------------------------------------------
class Item(BaseModel):
    """A single identifier/text pair extracted from the error message."""

    identifier: str = Field(..., description="The identifier extracted from the error message.")
    text: str = Field(..., description="The text associated with the identifier.")


class OutputModel(BaseModel):
    """The complete response expected from the LLM.

    ``items`` is a list of ``Item`` objects, which will be returned by the
    ``JsonOutputParser`` as a Python ``list[dict]`` after parsing.
    """

    items: List[Item] = Field(..., description="List of identifier/text pairs.")


def get_output_parser() -> JsonOutputParser:
    """Create a JsonOutputParser that validates against ``OutputModel``.

    Using ``JsonOutputParser`` (instead of ``StructuredOutputParser``) allows us
    to define nested structures such as a list of objects.  The parser also
    automatically generates the proper *format instructions* for the LLM.
    """
    return JsonOutputParser(pydantic_object=OutputModel)


def predict_result(msg: str, PROMPT_TEMPLATE: str):
    """Run the LLM on ``msg`` using the provided prompt template.

    The function returns the parsed Python object – a list of dictionaries –
    that matches the schema defined in ``OutputModel``.
    """
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    # ``format_instructions`` is injected into the prompt, allowing the model to
    # know the exact JSON shape it must return.
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    # ``response.content`` contains the raw text generated by the model.  The
    # parser validates the JSON against ``OutputModel`` and returns a Python
    # dict (or list) that matches the schema.
    parsed = output_parser.parse(response.content)
    # ``parsed`` is an instance of ``OutputModel``; we return the underlying
    # list of dictionaries for convenience.
    return parsed.items


if __name__ == "__main__":
    test_msg = (
        "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),"
        "AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"

    --------
    Error message is {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
