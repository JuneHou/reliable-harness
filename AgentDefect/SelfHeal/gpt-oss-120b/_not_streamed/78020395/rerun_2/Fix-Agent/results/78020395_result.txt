import os
from dotenv import load_dotenv
from typing import List

from pydantic import BaseModel, Field
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import JsonOutputParser

load_dotenv()

# ---------------------------------------------------------------------------
# Pydantic models that define the desired nested JSON structure.
# ``Item`` represents a single identifier/text pair.
# ``Output`` wraps a list of those pairs under the key ``items``.
# ---------------------------------------------------------------------------
class Item(BaseModel):
    """A single identifier / text pair extracted from the error message."""

    identifier: str = Field(
        description="The identifier string extracted from the error message."
    )
    text: str = Field(
        description="The text string associated with the identifier."
    )


class Output(BaseModel):
    """The overall result returned by the LLM.

    It is a JSON object with a single field ``items`` that contains a list of
    ``Item`` objects.
    """

    items: List[Item] = Field(
        description=(
            "A list of objects, each containing an 'identifier' and its corresponding "
            "'text' extracted from the error message."
        )
    )


def get_output_parser() -> JsonOutputParser:
    """Create a JSON output parser that enforces the ``Output`` schema.

    The ``JsonOutputParser`` uses the provided Pydantic model to generate format
    instructions for the LLM and to validate/parse the response.
    """
    return JsonOutputParser(pydantic_object=Output)


def predict_result(msg: str, PROMPT_TEMPLATE: str) -> dict:
    """Send the prompt to the model and return the parsed JSON result.

    Parameters
    ----------
    msg: str
        The raw error message that needs to be parsed.
    PROMPT_TEMPLATE: str
        A template string that includes ``{msg}`` and ``{format_instructions}``.
    """
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    # ``format_messages`` builds a list of ``BaseMessage`` objects suitable for
    # ``ChatOpenAI.invoke``.
    messages = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(messages)

    # ``response.content`` holds the raw string generated by the LLM.
    parsed = output_parser.parse(response.content)
    # ``parsed`` is a Pydantic model instance; convert it to a plain dict for
    # easier downstream consumption.
    return parsed.dict()
