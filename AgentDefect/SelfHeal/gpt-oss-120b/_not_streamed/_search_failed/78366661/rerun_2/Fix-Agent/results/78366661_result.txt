import os
from dotenv import load_dotenv
from pymongo import MongoClient

from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import MongoDBAtlasVectorSearch


def run_agent(pdf_url: str):
    """Load a PDF, chunk it, embed the chunks into a MongoDB Atlas Vector Search
    collection and perform a similarity search.

    The function expects the following environment variables to be set (e.g. in a
    ``.env`` file):
        * ``OPENAI_API_KEY`` – OpenAI API key used by ``OpenAIEmbeddings``.
        * ``MONGODB_ATLAS_URI`` – Full MongoDB Atlas connection string.
        * ``MONGODB_DB_NAME`` – Name of the database that contains the collection.
        * ``MONGODB_COLLECTION_NAME`` – Name of the collection that will store the
          vector index.
    """
    # ---------------------------------------------------------------------
    # Load environment variables
    # ---------------------------------------------------------------------
    load_dotenv()
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment")

    mongodb_uri = os.getenv("MONGODB_ATLAS_URI")
    db_name = os.getenv("MONGODB_DB_NAME")
    collection_name = os.getenv("MONGODB_COLLECTION_NAME")
    if not all([mongodb_uri, db_name, collection_name]):
        raise ValueError(
            "MongoDB connection information (MONGODB_ATLAS_URI, MONGODB_DB_NAME, "
            "MONGODB_COLLECTION_NAME) must be set in the environment"
        )

    # ---------------------------------------------------------------------
    # Initialise MongoDB client and obtain the target collection
    # ---------------------------------------------------------------------
    client = MongoClient(mongodb_uri)
    db = client[db_name]
    collection = db[collection_name]

    # ---------------------------------------------------------------------
    # Load the PDF and split into manageable chunks
    # ---------------------------------------------------------------------
    loader = PyPDFLoader(pdf_url)
    raw_documents = loader.load()

    text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = text_splitter.split_documents(raw_documents)

    # ---------------------------------------------------------------------
    # Create (or reuse) a MongoDB Atlas Vector Search index and insert the docs
    # ---------------------------------------------------------------------
    embeddings = OpenAIEmbeddings(disallowed_special=())
    vector_search = MongoDBAtlasVectorSearch.from_documents(
        documents=docs,
        embedding=embeddings,
        collection=collection,
        index_name="vector_index",  # name of the Atlas Search index
    )

    # ---------------------------------------------------------------------
    # Perform a similarity search
    # ---------------------------------------------------------------------
    query = "What were the compute requirements for training GPT 4"
    # ``k`` determines how many nearest‑neighbors are returned; 5 is a reasonable default
    results = vector_search.similarity_search(query, k=5)

    return results


if __name__ == "__main__":

        print(f"Error: {e}")
