import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    """A simple calculator tool that evaluates arithmetic expressions.

    The tool returns the raw result as a string. By configuring the agent to
    stop after the first iteration we ensure this raw output is returned to the
    user without any additional LLM‑generated formatting.
    """

    name = "calculator"
    description = "Performs basic arithmetic"

    def _run(self, query: str) -> str:
        # NOTE: ``eval`` is used here for simplicity; in production code a
        # safer expression parser should be used.
        return str(eval(query))

    async def _arun(self, query: str) -> str:  # pragma: no cover
        raise NotImplementedError()


def create_agent():
    """Create a conversational React‑style agent.

    The key change is ``max_iterations=1`` together with
    ``early_stopping_method="force"``. This forces the agent to stop immediately
    after the tool call, returning the tool's raw output directly.
    """
    llm_gpt4 = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(
        memory_key="chat_history", return_messages=True
    )
    return initialize_agent(
        agent="chat-conversational-react-description",
        tools=[CalculatorTool()],
        llm=llm_gpt4,
        verbose=True,
        # One iteration = one LLM thought + optional tool call.
        # After the tool runs the loop ends, so the raw tool output is returned.
        max_iterations=1,
        early_stopping_method="force",
        memory=memory,
    )


def run_agent(prompt: str) -> str:
    """Run the agent with the supplied user prompt and return its answer."""
    agent = create_agent()
    return agent.run(prompt)
