import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool

# Load environment variables and set the OpenAI key
load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    """Simple arithmetic calculator tool used by the agent.

    The tool returns the raw result of ``eval`` as a string.  This is the
    value we ultimately want the agent to return to the user without any
    additional LLM formatting.
    """

    name = "calculator"
    description = "Performs basic arithmetic"

    def _run(self, query: str) -> str:
        # NOTE: ``eval`` is used here for simplicity; in production you would
        # replace this with a safe arithmetic parser.
        return str(eval(query))

    async def _arun(self, query: str) -> str:
        raise NotImplementedError()


def create_agent():
    """Create a Conversational REACT agent that returns intermediate steps.

    By setting ``return_intermediate_steps=True`` we can access the raw tool
    observation directly.  The caller ``run_agent`` will then return that
    observation instead of the LLM‑generated final answer.
    """
    llm = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(
        memory_key="chat_history", return_messages=True
    )
    return initialize_agent(
        tools=[CalculatorTool()],
        llm=llm,
        agent="chat-conversational-react-description",
        verbose=True,
        max_iterations=3,
        early_stopping_method="generate",
        memory=memory,
        return_intermediate_steps=True,
    )


def run_agent(prompt: str) -> str:
    """Run the agent and return the raw calculator output.

    The ``AgentExecutor.run`` call returns a tuple ``(final_output,
    intermediate_steps)`` when ``return_intermediate_steps`` is enabled.  The
    last observation in ``intermediate_steps`` corresponds to the tool's raw
    result, which we return directly.
    """
    agent = create_agent()
    result, intermediate_steps = agent.run(prompt)
    # If a tool was used, return its raw observation; otherwise fall back to
    # the agent's final answer.
    if intermediate_steps:
        # ``intermediate_steps`` is a list of (AgentAction, observation) tuples.
        # The observation is the raw output from the tool.
        return intermediate_steps[-1][1]
    return result
