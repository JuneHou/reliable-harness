from llama_index.core import Settings

import os

    model_name = "BAAI/bge-m3"
    # The cache folder is kept the same as in the original script.
    cache_dir = "./embeddings"

    embed_model = HuggingFaceEmbedding(

        cache_folder=cache_dir,
    )
    print("embeddings loaded")
    return embed_model


def run_agent(text: str) -> tuple[int, int]:
    """Run a tiny agent that returns two token counts.

    * ``embedding_tokens`` – the number of tokens that were actually fed to the
      BGE‑M3 model when the embedding was generated. This value is taken from the
      :class:`TokenCountingHandler` which records the token usage of the
      ``HuggingFaceEmbedding`` instance.
    * ``tokenizer_tokens`` – the number of tokens produced by the *same* tokenizer
      that the embedding model uses, **including** any special tokens and respecting
      the ``max_length`` truncation that the embedding model applies.
    The latter is the value the user expects when they compare against the raw
      ``AutoTokenizer`` output.
    """
    chunk_size = 512
    embed_model = create_embedding_model(chunk_size)

    # Initialise the token‑counting callback.
    token_counter = TokenCountingHandler()
    callback_manager = CallbackManager([token_counter])

    # Register the embedding model and the callback manager globally.
    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    # Generate the embedding – this will trigger the callback.
    _ = Settings.embed_model.get_text_embedding(text)
    embedding_tokens = token_counter.total_embedding_token_count

    # ---------------------------------------------------------------------
    # Compute the token count using the *exact* tokenizer configuration that
    # the embedding model uses.  ``HuggingFaceEmbedding`` exposes its tokenizer
    # via the ``tokenizer`` attribute; if for any reason that attribute does not
    # exist we fall back to constructing a fresh tokenizer with the same model
    # name.
    # ---------------------------------------------------------------------
    if hasattr(embed_model, "tokenizer"):
        tokenizer = embed_model.tokenizer
    else:
        tokenizer = AutoTokenizer.from_pretrained("BAAI/bge-m3")

    # The embedding model truncates to ``max_length`` and adds the model's
    # special tokens (CLS/EOS).  We replicate that behaviour here.
    tokenised = tokenizer(
        text,
        truncation=True,
        max_length=chunk_size,
        add_special_tokens=True,
        return_attention_mask=False,
        return_token_type_ids=False,
    )
    tokenizer_tokens = len(tokenised["input_ids"])

    return embedding_tokens, tokenizer_tokens


if __name__ == "__main__":

    The count mirrors the preprocessing performed inside ``HuggingFaceEmbedding``:
    * the same tokenizer

        text,
        truncation=True,
        max_length=max_length,
        padding=False,
        add_special_tokens=True,
