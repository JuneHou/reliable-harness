import os
from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager, TokenCountingHandler
from transformers import AutoTokenizer


def create_embedding_model(max_len: int = None) -> HuggingFaceEmbedding:
    """Load the BGE‑M3 embedding model.
    The function abstracts the caching logic and returns a ``HuggingFaceEmbedding``
    instance with the ``max_length`` argument forwarded so that the underlying
    tokenizer behaves exactly like the one we will call manually.
    """
    model_name = "BAAI/bge-m3"
    cache_dir = "./embeddings"
    os.makedirs(cache_dir, exist_ok=True)
    return HuggingFaceEmbedding(
        model_name=model_name,
        max_length=max_len,
        cache_folder=cache_dir,
    )


def run_agent(text: str) -> tuple[int, int]:
    """Return the number of tokens used by the embedding model *and* by a raw
    tokenizer call.

    The embedding model counts tokens after applying the same ``max_length``
    and ``truncation`` logic it uses internally.  To obtain a comparable token
    count we explicitly tokenize the text with the same parameters (including
    special tokens) and then take the length of ``input_ids``.
    """
    chunk_size = 512

    # ------------------------------------------------------------
    # 1. Initialise the embedding model and the token‑count handler
    # ------------------------------------------------------------
    embed_model = create_embedding_model(max_len=chunk_size)
    token_counter = TokenCountingHandler()
    callback_manager = CallbackManager([token_counter])

    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    # ------------------------------------------------------------
    # 2. Generate the embedding – this also updates ``token_counter``
    # ------------------------------------------------------------
    _ = Settings.embed_model.get_text_embedding(text)
    # ``total_embedding_token_count`` holds the number of tokens that the
    # embedding model actually processed (including any special tokens and the
    # truncation imposed by ``max_length``).
    embedding_tokens = token_counter.total_embedding_token_count

    # ------------------------------------------------------------
    # 3. Replicate the exact tokenisation performed inside the model
    # ------------------------------------------------------------
    tokenizer = AutoTokenizer.from_pretrained("BAAI/bge-m3")
    tokenized = tokenizer(
        text,
        truncation=True,
        max_length=chunk_size,
        add_special_tokens=True,
        return_attention_mask=False,
        return_token_type_ids=False,
    )
    token_count = len(tokenized["input_ids"])

    return embedding_tokens, token_count


if __name__ == "__main__":
