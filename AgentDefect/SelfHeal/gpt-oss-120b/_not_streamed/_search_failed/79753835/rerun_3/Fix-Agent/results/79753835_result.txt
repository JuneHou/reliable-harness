import os
from llama_index.embeddings.huggingface import HuggingFaceEmbedding
from llama_index.core import Settings
from llama_index.core.callbacks import CallbackManager
from transformers import AutoTokenizer


def create_embedding_model(max_length: int | None = None) -> HuggingFaceEmbedding:
    """Create a BGE‑M3 embedding model.

    The function loads the model (downloading if necessary) and returns the
    `HuggingFaceEmbedding` instance.  The model internally creates its own
    tokenizer which we later reuse to compute the exact token count that the
    embedding model will see (including any special tokens and truncation).
    """
    model_name = "BAAI/bge-m3"
    # The cache folder is created if it does not exist.
    embed_model = HuggingFaceEmbedding(
        model_name=model_name,
        max_length=max_length,
        cache_folder="./embeddings",
    )
    return embed_model


def get_embedding_token_count(embed_model: HuggingFaceEmbedding, text: str) -> int:
    """Return the number of tokens that *exactly* the embedding model will use.

    The embedding model uses its own tokenizer with `add_special_tokens=True`
    and respects the `max_length`/`truncation` settings passed at creation time.
    We replicate that logic here without performing the expensive forward pass.
    """
    tokenizer = embed_model.tokenizer  # type: ignore[attr-defined]
    # Mimic the tokenisation performed inside HuggingFaceEmbedding.
    tokenised = tokenizer(
        text,
        add_special_tokens=True,
        truncation=True,
        max_length=embed_model.max_length,  # type: ignore[attr-defined]
        return_attention_mask=False,
        return_token_type_ids=False,
    )
    return len(tokenised["input_ids"])


def run_agent(text: str):
    chunk_size = 512
    embed_model = create_embedding_model(max_length=chunk_size)

    # Attach a (no‑op) callback manager – we are not interested in counting via
    # callbacks any more, but keeping the original structure intact.
    callback_manager = CallbackManager([])
    Settings.embed_model = embed_model
    Settings.callback_manager = callback_manager

    # Get the token count *as the model would see it*.
    embedding_tokens = get_embedding_token_count(embed_model, text)

    # For comparison, we also show the raw tokenizer count (without the model's
    # special‑token handling).  This uses a fresh tokenizer instance just to
    # demonstrate the difference.
    raw_tokenizer = AutoTokenizer.from_pretrained("BAAI/bge-m3")
    raw_tokenized = raw_tokenizer(text)
    raw_token_count = len(raw_tokenized["input_ids"])

    return embedding_tokens, raw_token_count


if __name__ == "__main__":
    sample_text = "Random words. This is a test! A very exciting test, indeed."
    embedding_tokens, raw_token_count = run_agent(sample_text)
    print(f"The embedding model sees: {embedding_tokens} tokens (including special tokens and truncation)")
    print(f"A plain tokenizer (no special tokens) sees: {raw_token_count} tokens")
