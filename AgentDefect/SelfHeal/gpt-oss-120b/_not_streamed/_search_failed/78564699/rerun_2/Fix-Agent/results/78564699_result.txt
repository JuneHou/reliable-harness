import json
import os
from huggingface_hub import hf_hub_download
from llama_index.llms.llama_cpp import LlamaCPP


def run_agent(query: str) -> str:
    """Run a simple Llama‑CPP based inference on a CPU.

    The function first tries to resolve a local path for the GGUF model via a
    ``paths.json`` file (the same approach the original script used).  If the
    path is missing or the file does not exist, the model is downloaded from the
    Hugging Face hub using :func:`huggingface_hub.hf_hub_download`.  The model is
    then loaded with ``n_gpu_layers=0`` (CPU‑only) and a modest number of CPU
    threads to avoid hanging.
    """
    # ---------------------------------------------------------------------
    # Resolve the model location
    # ---------------------------------------------------------------------
    paths_file = "./paths.json"
    model_path = None

    if os.path.exists(paths_file):
        try:
            with open(paths_file, "r", encoding="utf-8") as f:
                paths = json.load(f)
                model_path = paths.get("llama-2-13b-chat")
        except Exception:
            # If the JSON is malformed we fall back to downloading the model
            model_path = None

    # If we don't have a valid local file, download it from HF hub.
    if not model_path or not os.path.isfile(model_path):
        repo_id = "TheBloke/Llama-2-13B-chat-GGUF"
        filename = "llama-2-13b-chat.Q4_0.gguf"
        model_path = hf_hub_download(repo_id=repo_id, filename=filename)

    # ---------------------------------------------------------------------
    # Initialise the LlamaCPP wrapper – CPU‑only inference
    # ---------------------------------------------------------------------
    llm = LlamaCPP(
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        # ``n_gpu_layers=0`` forces the engine to run on the CPU.
        # Setting a reasonable thread count prevents the process from
        # stalling on machines with many cores.
        model_kwargs={"n_gpu_layers": 0, "n_threads": 8},
        verbose=True,
    )

    # ``complete`` returns a ``LLMResponse``; we extract the generated text.
    response = llm.complete(query)
    return str(response)


if __name__ == "__main__":
    query = "What is the capital of France?"
    print(run_agent(query))

def download_model_if_needed(model_url: str, model_path: str) -> str:
    """Download the GGUF model from HuggingFace if it is not already present.

    Args:
        model_url: The full HuggingFace URL to the model file.
        model_path: Desired local path (including filename) where the model should be stored.

    Returns:
        The absolute path to the local model file.
    """
    # Convert possible relative path to absolute Path object
    model_path_obj = Path(model_path).expanduser().resolve()
    if model_path_obj.is_file():
        return str(model_path_obj)

    # The URL is expected to be in the form:
    #   https://huggingface.co/<repo>/resolve/main/<filename>
    # Split it to obtain the repo id and the filename.
    try:
        # Remove the leading "https://huggingface.co/"
        repo_and_file = model_url.replace("https://huggingface.co/", "")
        repo, _, file_path = repo_and_file.partition("/resolve/main/")
        # Download using huggingface_hub, which handles caching and retries.
        local_file = hf_hub_download(repo_id=repo, filename=file_path, cache_dir=str(model_path_obj.parent))
        # Ensure the file is where the user expects it.
        if str(local_file) != str(model_path_obj):
            # Move/rename to the exact location requested.
            Path(local_file).rename(model_path_obj)
        return str(model_path_obj)
    except Exception as e:
        raise RuntimeError(f"Failed to download model from {model_url}: {e}")
