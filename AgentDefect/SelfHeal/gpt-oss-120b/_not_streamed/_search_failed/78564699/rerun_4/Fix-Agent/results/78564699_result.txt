import json
from huggingface_hub import hf_hub_download
from llama_index.llms.llama_cpp import LlamaCPP

    """Download the GGUF model (if not already cached) and return the local path.

    filesystem path and can cause the library to hang when a remote URL is

    # Download to the default HF cache directory (or a custom one if you prefer)
    model_path = hf_hub_download(repo_id=repo_id, filename=filename, cache_dir="./model_cache")
    return model_path


def run_agent(query: str) -> str:
    """Run a simple LlamaCPP inference on the supplied *query*.

    The function loads the model from a local path (downloaded on first call),
    configures it for CPU inference (no GPU layers), and returns the generated
    completion.
    """
    # Allow an optional override via a small JSON file – useful for testing with
    # a different model on the same machine.
    paths_file = "./paths.json"
    if os.path.exists(paths_file):
        with open(paths_file, "r") as f:
            paths = json.load(f)
            if "llama-2-13b-chat" in paths:
                model_path = paths["llama-2-13b-chat"]
            else:
                model_path = get_model_path()
    else:
        model_path = get_model_path()

    # Initialise LlamaCPP.  Setting `n_gpu_layers` to 0 forces CPU execution.
    # `n_threads` speeds up inference on multi‑core CPUs, and `verbose=True`
    # provides useful diagnostics if something still goes wrong.
    llm = LlamaCPP(
        model_path=model_path,
        temperature=0.1,
        max_new_tokens=256,
        context_window=3900,
        model_kwargs={
            "n_gpu_layers": 0,  # CPU only
            "n_threads": os.cpu_count() or 4,
        },
        verbose=True,
    )

    # `complete` returns a simple `Response` object; we extract the text.
    response = llm.complete(query)
    return response.text if hasattr(response, "text") else str(response)


if __name__ == "__main__":

    repo_id = "TheBloke/Llama-2-13B-chat-GGUF"

    # Allow a user‑provided path via ``paths.json``

                user_path = paths["llama-2-13b-chat"]
                if os.path.isfile(user_path):

                else:
                    raise FileNotFoundError(
                        f"Model path specified in paths.json does not exist: {user_path}"

    # Fallback: download the model to a temporary cache directory.
    # ``snapshot_download`` returns the directory that contains the file.

    repo_dir = snapshot_download(
        repo_id=repo_id,
        revision="main",
        cache_dir=str(cache_dir),
        allow_patterns=filename,
        local_files_only=False,
    )
    model_path = Path(repo_dir) / filename
    if not model_path.is_file():
        raise FileNotFoundError(f"Failed to locate the model file after download: {model_path}")
    return str(model_path)
