import os
from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain_community.storage.sql import SQLStore  # Persistent docstore implementation
from langchain.schema import Document
import nltk


def run_agent(
    query: str,
    chroma_dir: str = "./chroma_db_child",
    docstore_path: str = "./docstore.db",
) -> list[Document]:
    """Retrieve parent documents relevant to *query* using a persistent docstore.

    The function:

    1. Loads the NLTK punkt tokenizer.
    2. Sets up a HuggingFace embedding model (cached locally).
    3. Creates a ``Chroma`` vectorstore that persists its collection on disk.
    4. Uses ``SQLStore`` (SQLite‑backed) as the docstore for the
       ``ParentDocumentRetriever`` – this keeps the parent‑child mapping across runs.
    5. Adds the source documents **only if** the vectorstore is empty (i.e. the
       first run).  Subsequent runs will reuse the already‑indexed data.
    6. Returns the parent documents that are most relevant to ``query``.
    """

    # Ensure the NLTK tokenizer is available.
    nltk.download("punkt", quiet=True)

    # ---------------------------------------------------------------------
    # Embeddings
    # ---------------------------------------------------------------------
    hf_embed_path = "./hf_cache"
    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path,
    )

    # ---------------------------------------------------------------------
    # Vector store (persisted)
    # ---------------------------------------------------------------------
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory=chroma_dir,
    )

    # ---------------------------------------------------------------------
    # Persistent docstore (SQLite)
    # ---------------------------------------------------------------------
    # SQLStore uses a SQLAlchemy compatible URL. ``sqlite:///`` creates/opens a
    # local SQLite file at the given path.
    docstore = SQLStore(sql_url=f"sqlite:///{docstore_path}")

    # Child splitter – determines how documents are chunked before being stored
    # as children.
    child_splitter = NLTKTextSplitter(chunk_size=400)

    # Initialise the ParentDocumentRetriever with the persistent components.
    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=docstore,
        child_splitter=child_splitter,
    )

    # ---------------------------------------------------------------------
    # Add documents only on the first execution.
    # ---------------------------------------------------------------------
    # ``Chroma`` exposes a ``get" method that returns the number of stored
    # vectors.  If the collection is empty we need to add the documents and
    # persist the vector store.
    if vectorstore._collection.count() == 0:  # type: ignore[attr-defined]
        docs = [
            Document(
                page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."
            ),
            Document(
                page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."
            ),
            Document(
                page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."
            ),
        ]
        retriever.add_documents(docs)
        # Persist the Chroma collection so that it is available on the next run.
        vectorstore.persist()
        # ``SQLStore`` writes changes immediately, so no extra persist call is
        # required.

    # ---------------------------------------------------------------------
    # Retrieve parent documents relevant to the query.
    # ---------------------------------------------------------------------
    relevant_parents = retriever.get_relevant_documents(query)
    return relevant_parents


if __name__ == "__main__":

    results = run_agent(query)
