import os
from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import LocalFileStore  # Use a persistent docstore instead of InMemoryStore
from langchain.schema import Document
import nltk


def run_agent(query, store_location="./store_location"):
    # Ensure the NLTK tokenizer is available
    nltk.download("punkt", quiet=True)

    hf_embed_path = "./hf_cache"

    # Sample documents – in a real scenario these would come from a data source
    docs = [
        Document(
            page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."
        ),
        Document(
            page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."
        ),
        Document(
            page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."
        ),
    ]

    # Embedding model – cached locally to avoid repeated downloads
    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path,
    )

    # Splitter for creating child chunks from the parent documents
    child_splitter = NLTKTextSplitter(chunk_size=400)

    # Persistent vector store for the child chunks
    vectorstore = Chroma(
        collection_name="full_documents",
        embedding_function=embedding_function,
        persist_directory="./chroma_db_child",
    )

    # Persistent docstore – this will survive process restarts
    docstore = LocalFileStore("./docstore")

    # Retriever that links parent documents to their child chunks
    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=docstore,
        child_splitter=child_splitter,
    )

    # Add documents only if the docstore is empty. This prevents duplicate indexing on subsequent runs.
    if not docstore.keys():
        retriever.add_documents(docs)

    # Perform the similarity search on the persisted vector store
    matching_doc = vectorstore.max_marginal_relevance_search(query, k=3)
    return matching_doc


if __name__ == "__main__":
    query = "What is RAG?"
    result = run_agent(query)
    print(result)
