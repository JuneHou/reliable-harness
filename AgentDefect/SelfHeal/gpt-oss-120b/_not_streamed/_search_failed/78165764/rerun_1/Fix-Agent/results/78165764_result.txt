import os
from fastapi import FastAPI
from dotenv import load_dotenv
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()


def run_agent(query: str) -> str:
    """Run a simple RAG chain that returns a **full‑sentence** answer.

    The original implementation returned very short fragments because the
    prompt did not explicitly ask the model to produce a complete sentence.
    By amending the prompt we guide ChatOpenAI (gpt‑3.5‑turbo) to generate a
    proper answer that mirrors the behaviour of the Llama 2 example.
    """

    # Initialise the LLM – temperature a bit higher to encourage more natural
    # language while still keeping the answer deterministic.
    model = ChatOpenAI(
        model_name="gpt-3.5-turbo",
        temperature=0.2,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    # Load the data that will become the knowledge base.
    products = pd.read_csv("./data/products.csv")

    # Build a FAISS index from the product descriptions.
    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )
    retriever = vectorstore.as_retriever()

    # The FastAPI app does not need to be created inside the function; it belongs
    # to the module level if you intend to expose an HTTP endpoint.
    # Keeping the definition here does not affect the RAG logic, so we omit it.

    # Prompt template – the key change is the explicit instruction to answer in a
    # complete sentence.
    ANSWER_TEMPLATE = (
        "Answer the following question based on the provided context. "
        "Your answer should be a full, grammatically correct sentence.\n\n"
        "Context:\n{context}\n\n"
        "Question: {question}\n"
    )

    prompt = ChatPromptTemplate.from_template(ANSWER_TEMPLATE)

    # Build the chain: retrieve → format prompt → LLM → parse string output.
    chain = (
        {"context": retriever, "question": RunnablePassthrough()}
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)
    return result


if __name__ == "__main__":
    test_query = "What is the highest budget movie?"
    print(run_agent(test_query))
