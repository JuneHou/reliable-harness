from llama_index.core import VectorStoreIndex, Document
from llama_index.vector_stores.postgres import PGVectorStore
import os
from dotenv import load_dotenv

load_dotenv()


def create_vector_store(request):
    """Create a PGVectorStore where `external_id` is stored in its own column.

    The key changes are:
    * ``metadata_columns`` tells PGVectorStore which fields from the Document.metadata
      should be promoted to individual columns.
    * ``metadata_json_column`` is set to ``None`` (or ``False``) so that a generic JSONB
      ``metadata`` column is not created – only the explicit columns are persisted.
    """
    host = "localhost"
    password = "password"
    port = 5432
    user = "postgres"

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name="vectorstore",
        embed_dim=1536,
        # Explicitly promote ``external_id`` to its own column
        metadata_columns=["external_id"],
        # Disable the catch‑all JSON metadata column – we only want the explicit column(s)
        metadata_json_column=None,
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
    )

    return vector_store


def run_agent(request):

    documents = [

        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(

    return index
