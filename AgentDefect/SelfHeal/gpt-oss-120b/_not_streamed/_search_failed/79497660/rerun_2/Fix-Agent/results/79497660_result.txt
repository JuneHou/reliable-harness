"""Corrected example showing how to store a custom column (external_id) in the PGVectorStore table
instead of the generic metadata JSON column.

Key changes:
- Import the ``Column`` helper from ``llama_index.vector_stores.postgres``.
- Define a custom ``metadata_columns`` list that creates a dedicated ``external_id`` column of type ``String``.
- Use ``ignore_metadata_columns`` so that ``external_id`` is not duplicated in the ``metadata`` JSON column.
- When creating ``Document`` objects, keep ``external_id`` in the metadata dict (the store will extract it).
"""

from llama_index.core import VectorStoreIndex, Document
from llama_index.vector_stores.postgres import PGVectorStore, Column  # ``Column`` is a small helper that mirrors SQLAlchemy's Column
from sqlalchemy import String
import os
from dotenv import load_dotenv

load_dotenv()


def create_vector_store(request):
    """Create a PGVectorStore that contains a dedicated ``external_id`` column.

    Args:
        request: An object with a ``tenant`` attribute that indicates the target DB name.

    Returns:
        An instantiated ``PGVectorStore`` ready for indexing.
    """
    host = "localhost"
    password = "password"
    port = 5432
    user = "postgres"

    # Define a custom column that will be added to the generated table.
    # ``Column`` takes a name and a SQLAlchemy type.
    external_id_column = Column(name="external_id", type_=String)

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name="vectorstore",
        embed_dim=1536,
        # Custom columns (outside the generic ``metadata`` JSON column)
        metadata_columns=[external_id_column],
        # Prevent the same field from being stored redundantly in the JSON metadata column
        ignore_metadata_columns=["external_id"],
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
    )

    return vector_store


def run_agent(request):
    """Create an index and store a couple of documents.

    The ``external_id`` field will appear as its own column in the ``vectorstore``
    table rather than inside the ``metadata`` JSON column.
    """
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )
    return index


if __name__ == "__main__":
    class Request:
        tenant = "test_db"

    request = Request()
    index = run_agent(request)
    print("Vector store created and documents indexed")
    print("The 'external_id' now lives in its own column rather than metadata JSON.")
