import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
import json

dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# Tool definition
# ---------------------------------------------------------------------------
@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a JSON representation of the DataFrame.

    The LangChain tool interface expects a string return value because the
    agent serialises tool outputs into the LLM prompt.  Returning the raw
    ``DataFrame`` object would raise a serialization error and, even if it
    succeeded, the agent would only be able to include a textual representation
    in its final answer.  By converting the DataFrame to JSON we preserve the
    full data while keeping the contract of the tool interface.
    """
    df = (
        pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    )
    # Keep only object columns (as the original code intended)
    df = df.select_dtypes(include=["object"])
    # Convert to JSON – ``orient='records'`` produces a list of row dicts which
    # can be easily re‑created as a DataFrame on the caller side.
    return df.to_json(orient="records")

# ---------------------------------------------------------------------------
# Agent runner
# ---------------------------------------------------------------------------

    """Execute a ReAct agent and return the DataFrame produced by the tool.

    The function returns a tuple ``(agent_output, df)`` where ``agent_output`` is
    the textual response from the LLM and ``df`` is the pandas DataFrame that
    was generated by the ``load_excel_response`` tool (or ``None`` if the agent
    did not produce a valid JSON payload).
    """
    llm = ChatOpenAI(model="gpt-4o-mini")
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # ``invoke`` returns a dictionary; the key "output" holds the final LLM answer.
    response = agent_executor.invoke({"input": user_prompt})
    textual_output = response.get("output", "")

    # Attempt to extract the JSON produced by ``load_excel_response``.  The agent
    # usually echoes the tool's return value verbatim; we therefore look for a
    # substring that can be parsed as JSON.
    df = None
    try:
        # Find the first occurrence of '[' or '{' which indicates start of JSON.

            *(pos for pos in (textual_output.find("["), textual_output.find("{")) if pos != -1),

                    df = pd.read_json(possible_json[:end])
                    break

                    continue
    except Exception:

    return textual_output, df
