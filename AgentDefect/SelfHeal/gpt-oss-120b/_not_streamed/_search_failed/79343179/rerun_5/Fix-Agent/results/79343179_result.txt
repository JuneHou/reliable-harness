from langchain.tools import tool

    This tool is kept for demonstration purposes – the ReAct agent can also
    achieve the same result directly via the ``python_repl`` tool.  Returning the
    actual ``DataFrame`` (instead of a JSON string) allows the agent to include a
    textual representation in its answer while the real object lives in the
    shared REPL globals.

    )
    df = df.select_dtypes(include=["object"])
    return df

    we can inspect that dictionary and return the variable that the LLM asked to
    create (``df`` in the example).  The function therefore returns a tuple:
    ``(agent_text_output, df_object_or_None)``.

    # LLM and tools configuration

@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a serialized DataFrame.

    The ReAct agent can only return strings from tools, so we serialize the
    DataFrame to JSON (using the "split" orientation which preserves the
    index, columns, and data). The caller can deserialize this back into a
    ``pd.DataFrame``.
    """
    df = (
        pd.read_excel(filepath, skiprows=2, names=["responses"])
        .reset_index(drop=True)
    )
    df = df.select_dtypes(include=["object"])
    # Serialize the DataFrame – ``orient='split'`` keeps it easy to rebuild.
    return df.to_json(orient="split")

def _deserialize_dataframe(json_str: str) -> pd.DataFrame:
    """Convert the JSON string produced by ``load_excel_response`` back to a DataFrame."""
    try:
        data = json.loads(json_str)
        return pd.DataFrame(**data)
    except Exception as exc:  # pragma: no cover
        raise ValueError(f"Failed to deserialize DataFrame JSON: {exc}")

def run_agent(user_prompt: str, file_name: str = "data.json"):

    The ReAct agent returns a dictionary with an ``intermediate_steps`` field

    The agent will receive the string output, but the actual ``pd.DataFrame``
    object is also stored in the REPL globals under the name ``df`` so that

    # Store the DataFrame in the REPL globals for later retrieval.
    # ``PythonREPL`` keeps a mutable ``globals`` dict that survives across
    # calls.  If the REPL has not been instantiated yet we simply ignore.

        pass

    return df.head().to_json()
