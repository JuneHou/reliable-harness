import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.callbacks import get_openai_callback
from langchain.llms import OpenAI
from langchain.agents import create_pandas_dataframe_agent
import pandas as pd

warnings.filterwarnings("ignore")

# Load environment variables (including OPENAI_API_KEY)
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def run_agent(queries):
    """Run the given list of natural‑language queries against each CSV file.

    The function loads each CSV into a pandas DataFrame and creates a lightweight
    Pandas‑DataFrame agent for it. The agent translates the query into pandas code
    and executes it, returning an exact answer (e.g., counts, filters, max dates).
    This approach avoids the expensive embedding step used by a vectorstore and
    guarantees that the entire table is considered, fixing the earlier mismatch
    where only a handful of rows were retrieved.
    """
    results = []
    files = ["drivers.csv", "drivers_full.csv"]

    for file in files:
        print("=" * 40)
        print(file)
        print("=" * 40)
        # Load the CSV into a DataFrame – this is fast and memory‑efficient for
        # the modest data sizes typical of these examples.
        df = pd.read_csv(file, encoding="utf-8")

        # Create a pandas‑agent that uses the OpenAI LLM. The agent translates
        # natural‑language queries into pandas expressions.
        agent = create_pandas_dataframe_agent(
            llm=OpenAI(temperature=0),
            df=df,
            verbose=True,
        )

        for query in queries:
            print(f"Query: {query}")
            with get_openai_callback() as cb:
                # The agent.run method returns the LLM's answer.
                answer = agent.run(query)
                results.append(answer)
                print(answer)
                # Show token usage for this call – helpful for cost monitoring.
                print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German",
        "How many drivers are German?",
        "Which driver has the number 14?",
        "What is the oldest birthdate among the drivers?",
    ]
    run_agent(queries)
