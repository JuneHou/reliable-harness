)

from dotenv import load_dotenv


def run_agent(query: str):
    """Demonstrates how to work with "null" metadata values in LlamaIndex.

    LlamaIndex does **not** support filtering on ``None`` (Python null) because the
    underlying vector stores serialize metadata to JSON – ``None`` becomes ``null``
    in JSON, which is lost when the store builds its internal filter index.  To
    filter on a "null" value you must store a sentinel that can be compared, for
    example the string ``"__null__"``.  This example shows how to store such a
    sentinel and then retrieve the node using a metadata filter.
    """
    # Load environment variables (e.g. ``OPENAI_API_KEY``) from a .env file.
    load_dotenv()

    # Set up the embedding model – this will respect the ``OPENAI_API_KEY``
    # environment variable.  If you do not have an API key you can replace the
    # embedding with a mock one for local testing.
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # ---------------------------------------------------------------------
    # 1. Create a node where the "start_date" is conceptually missing.
    # Instead of storing ``None`` we store a sentinel string that we can later
    # query for.  The sentinel must be a value that would never appear in real
    # data – ``"__null__"`` is a common choice.
    # ---------------------------------------------------------------------
    node_01 = TextNode(
        text="This document has a missing start_date.",
        id_="node_01",
        metadata={"start_date": "__null__"},
    )

    # ---------------------------------------------------------------------
    # 2. Build the index from the single node.
    # ---------------------------------------------------------------------
    doc_index = VectorStoreIndex([node_01])

    # Debug: show what metadata is actually stored in the index.
    print("Index nodes metadata:")
    for doc_id, stored_node in doc_index.docstore.docs.items():
        print(f"  {doc_id}: {stored_node.metadata}")
    print()

    # ---------------------------------------------------------------------
    # 3. Retrieve using a filter that matches the sentinel value.
    # ---------------------------------------------------------------------
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value="__null__",
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show the retrieved nodes and their metadata.
    print("Retrieved nodes:")
    for node in nodes:
        print(f"  node_id={node.node_id}, metadata={node.metadata}")
    print()

    return nodes
