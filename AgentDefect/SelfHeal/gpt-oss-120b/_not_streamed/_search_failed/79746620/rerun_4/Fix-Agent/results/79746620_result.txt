from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv

# NOTE:
# LlamaIndex (as of the current version) does **not** support filtering on
# metadata fields that are set to Python's ``None``.  The underlying vector store
# expects JSON‑serialisable values and treats ``None`` as a missing key rather
# than a value that can be compared.  To enable "null"‑like filtering we store a
# sentinel value (e.g., the string "null") in the metadata and filter on that
# sentinel.
# This pattern works across the built‑in vector stores (FAISS, Chroma, etc.)
# and mirrors the behaviour developers typically use when they need to
# represent a missing value in a searchable field.

NULL_SENTINEL = "null"  # sentinel used to represent a missing/None value


def run_agent(query: str):
    """Create a simple index with a node whose ``start_date`` metadata is
    represented by ``NULL_SENTINEL`` and retrieve it using a metadata filter.

    Args:
        query: The query string used for similarity search.
    """
    # Load environment variables from .env file (e.g., OpenAI key)
    load_dotenv()

    # Configure the embedding model – the key is pulled from the .env file.
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create a node where we want to indicate "no start date".
    # Instead of ``None`` we store the sentinel string.
    node_01 = TextNode(
        text="This document has a null‑like start_date in the metadata",
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},
    )

    # Build the index from the single node.
    doc_index = VectorStoreIndex([node_01])

    # Debug: show what actually gets persisted in the docstore.
    print("Index nodes metadata:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Build a filter that matches the sentinel value.
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])

    # Retrieve with the filter – we keep similarity_top_k=1 for brevity.
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show what was retrieved.
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes
