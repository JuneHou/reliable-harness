from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv


def run_agent(query: str):
    """Create an index with a node that has a "null" metadata value and
    retrieve it using a filter.

    LlamaIndex (and most underlying vector store back‑ends) cannot filter on
    Python ``None`` because the value is not serialisable to the vector store.
    The workaround is to store a sentinel value (e.g., the string ``"null"``)
    and filter on that sentinel.
    """
    # Load environment variables from .env file
    load_dotenv()

    # Set up the embedding model (uses OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Create a node whose ``start_date`` metadata is deliberately set to a
    # sentinel representing “null”.  Using a plain ``None`` would be dropped or
    # cause an error when the metadata is persisted.
    node_01 = TextNode(
        text="This document has a null-like metadata value",
        id_="node_01",
        metadata={"start_date": "null"},  # sentinel for null
    )

    # Build the index
    doc_index = VectorStoreIndex([node_01])

    # Debug: show what is stored in the docstore
    print("Index nodes metadata:")
    print([node.metadata for node in doc_index.docstore.docs.values()])
    print()

    # Filter for nodes where ``start_date`` equals the sentinel value
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value="null",
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)

    nodes = retriever.retrieve(query)

    # Show retrieval results
    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
    print()

    return nodes

# Helper to replace Python ``None`` with a sentinel value that can be filtered on.
# Most vector stores (e.g., Chroma, Milvus, Pinecone) serialize metadata to JSON.
# ``None`` becomes ``null`` in JSON, but the filter implementation in LlamaIndex
# currently treats ``null`` as *missing* rather than as a comparable value.
# As a workaround we store a known placeholder (e.g., the string "__NULL__").
NULL_SENTINEL = "__NULL__"

def _sanitize_metadata(metadata: dict) -> dict:
    """Replace ``None`` values with ``NULL_SENTINEL``.

    This allows us to later filter for "null" metadata using a normal EQ filter.
    """
    return {k: (v if v is not None else NULL_SENTINEL) for k, v in metadata.items()}

    # We sanitize the metadata so that ``None`` becomes a filter‑able sentinel.

        text="This document has None in the metadata",

        metadata={"start_date": None},
    )
    raw_node.metadata = _sanitize_metadata(raw_node.metadata)

    # Build the index from the single node.
