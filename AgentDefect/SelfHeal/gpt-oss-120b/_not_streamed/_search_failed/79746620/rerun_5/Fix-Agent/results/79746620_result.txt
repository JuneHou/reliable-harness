# Sentinel value to represent "null" in metadata. Most vector stores cannot store Python ``None``
# values, so we use a string (or any other JSON‑serialisable placeholder) that we can filter on.
NULL_SENTINEL = "__NULL__"

    """

    # Create a node – we store the sentinel instead of ``None`` for the
    # ``start_date`` field.
    node_01 = TextNode(
        text="This document has a null‑like start_date in the metadata",
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},
    )

    # Build an index from the node.
    doc_index = VectorStoreIndex([node_01])

    # Debug: show what is actually stored in the docstore.
    print("Index nodes (metadata):")
    print([node.metadata for node in doc_index.docstore.docs.values()])

    # Construct a metadata filter that looks for the sentinel value.
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,

    # Retrieve with the filter applied.
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)

    print("Retrieved nodes:")
    print([(node.node_id, node.metadata) for node in nodes])
