from llama_index.core import VectorStoreIndex, Settings
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding
from dotenv import load_dotenv


def run_agent(query: str):
    """Run a simple retrieval example where we want to filter on a "null"
    metadata value.

    LlamaIndex does not support filtering on Python ``None`` directly – the
    underlying vector store expects serialisable values (e.g. strings, numbers).
    To emulate a ``null`` value we store a sentinel string (``"__NULL__"``) and
    filter on that sentinel.  This pattern works for all vector‑store backends
    supported by LlamaIndex.
    """
    # Load environment variables from .env file
    load_dotenv()

    # Set up embedding model (will use OPENAI_API_KEY from .env)
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # Use a sentinel value to represent "null" in metadata
    NULL_SENTINEL = "__NULL__"

    # Create node with the sentinel in metadata
    node_01 = TextNode(
        text="This document has a null‑like metadata value",
        id_="node_01",
        metadata={"start_date": NULL_SENTINEL},
    )

    # Create the index from the node list
    doc_index = VectorStoreIndex([node_01])

    # Debug: Show what is actually stored in the docstore
    print("Index nodes metadata:")
    for node in doc_index.docstore.docs.values():
        print(node.metadata)
    print()

    # Build a filter that looks for the sentinel value
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])

    # Retrieve nodes that match the filter
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results
    print("Retrieved nodes:")
    for node in nodes:
        print(node.node_id, node.metadata)
    print()

    return nodes
