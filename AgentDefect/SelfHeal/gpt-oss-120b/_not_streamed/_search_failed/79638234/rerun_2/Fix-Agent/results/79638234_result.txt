import random
from langchain_ollama import ChatOllama


def run_agent(input_text: str) -> str:
    """Generate a response using Ollama with a non‑deterministic seed.

    By providing a fresh ``seed`` value for each call we ensure the model
    respects the ``temperature`` setting and produces varied output even for
    identical prompts.
    """
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        # A random seed on every invocation prevents deterministic behaviour.
        seed=random.randint(0, 2**31 - 1),
        timeout=120,
    )
    response = llm.invoke(input_text)
    # ``response`` is a ``ChatMessage``; the generated text lives in ``content``.
    return response.content


if __name__ == "__main__":
    print(run_agent("Please tell me a lighthearted joke"))
