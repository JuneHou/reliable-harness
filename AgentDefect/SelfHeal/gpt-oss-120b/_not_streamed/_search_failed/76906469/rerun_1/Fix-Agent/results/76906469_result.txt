from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType, load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# LLM configuration
llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

# Load the desired tools
tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Conversation memory – this will store the full chat history
memory = ConversationBufferMemory(memory_key="chat_history")

# Use the conversational‑react agent which is designed to utilise the provided memory
agent = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    memory=memory,
    handle_parsing_errors=True,
    verbose=False,
)

def run_agent(prompt: str) -> str:
    """Run the agent with the supplied prompt, returning its response."""
    return agent.run(prompt)

if __name__ == "__main__":
    # First query – adds the interaction to memory
    result1 = run_agent("What is the capital of France?")
    print(result1)

    # Second query – the agent can now refer to the previous question via memory
    result2 = run_agent("What was my previous question?")
    print(result2)
