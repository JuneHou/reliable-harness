import os
from dotenv import load_dotenv

from langchain.memory import ConversationBufferMemory
from langchain.agents import initialize_agent, AgentType, load_tools
from langchain_openai import ChatOpenAI

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Initialise the LLM – we use the chat‑model variant because the
# CHAT_ZERO_SHOT_REACT_DESCRIPTION agent expects a chat‑style LLM.
llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

# Load the tools that the agent can use.
tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# ConversationBufferMemory must return a list of ChatMessage objects so that the
# agent can correctly inject the "chat_history" variable into its prompt template.
# Setting `return_messages=True` ensures the memory provides messages in the
# format expected by the CHAT_ZERO_SHOT_REACT_DESCRIPTION agent.
memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)

# Initialise the agent with the supplied memory.
agent = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION,
    memory=memory,
    handle_parsing_errors=True,
    verbose=False,
)

def run_agent(prompt: str) -> str:
    """Send a prompt to the agent and return its answer.

    The agent keeps the conversation history in `memory`, so subsequent calls
    can refer to earlier exchanges (e.g., “what was the last question I asked?”).
    """
    return agent.run(prompt)

if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    # The second call can now access the previous question via the memory.
    result2 = run_agent("what is the last question I asked?")
    print(result2)
