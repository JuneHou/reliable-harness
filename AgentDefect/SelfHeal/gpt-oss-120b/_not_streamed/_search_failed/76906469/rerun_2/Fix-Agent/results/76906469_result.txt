from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Initialise the LLM (GPT-4) with deterministic output for reproducibility
llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

# Load the desired tools – math and Wikipedia lookup – and bind the LLM
tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Configure memory so the chat history is available to the CHAT_ZERO_SHOT_REACT_DESCRIPTION agent.
# The agent expects a variable named ``chat_history``; we therefore set ``memory_key`` accordingly.
# ``return_messages=True`` makes the memory expose the raw message objects, which the chat‑based
# agent formats correctly.
memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)

# Build the zero‑shot ReAct agent with the supplied tools, LLM and memory.
agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False,
)

def run_agent(prompt: str) -> str:
    """Run the agent on a user prompt and return the response string."""
    return agent_test.run(prompt)

if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)
    result2 = run_agent("what is the last question I asked?")
    print(result2)
