import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables and set up LLM cache
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# Pool of API keys – you can extend this list if you have multiple keys
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]


def get_random_key(key_pool: list[str]) -> str:
    """Return a random non‑None API key from the pool.
    """
    return random.choice([k for k in key_pool if k is not None])


def run_agent(query: str, pdf_paths: list[str], filter: dict | None = None):
    """Answer a question using only the documents that match ``filter``.

    Parameters
    ----------
    query: The user question.
    pdf_paths: List of PDF files that will be indexed.
    filter: Optional LangChain metadata filter (e.g. {"source": "paper2.pdf"}).
    """
    # ---------------------------------------------------------------------
    # 1. Load PDFs, split them and attach a ``source`` metadata field that
    #    contains the path of the originating PDF.
    # ---------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()
        splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        chunks = splitter.split_documents(documents)
        for chunk in chunks:
            # Ensure every chunk carries the source file name – this is the key
            # we will filter on later.
            if chunk.metadata is None:
                chunk.metadata = {}
            chunk.metadata["source"] = pdf_path
        all_chunks.extend(chunks)

    # ---------------------------------------------------------------------
    # 2. Create a Chroma vector store from the chunks.
    # ---------------------------------------------------------------------
    db = Chroma.from_documents(all_chunks, embeddings)

    # ---------------------------------------------------------------------
    # 3. Build a retriever. ``as_retriever`` returns a ``VectorStoreRetriever``
    #    that forwards any ``search_kwargs`` directly to Chroma.  We start with an
    #    empty dict and later inject the user‑provided filter if present.
    # ---------------------------------------------------------------------
    retriever = db.as_retriever(search_kwargs={})

    # ---------------------------------------------------------------------
    # 4. Build the ConversationalRetrievalChain.  We keep ``return_source_documents``
    #    True so we can inspect which documents were actually used.
    # ---------------------------------------------------------------------
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever=retriever,
        verbose=True,
        return_source_documents=True,
    )

    # ---------------------------------------------------------------------
    # 5. Apply the dynamic filter (if any) *before* the chain is invoked.  The
    #    filter is stored in ``retriever.search_kwargs`` – this is the mechanism
    #    Chroma expects for metadata filtering.
    # ---------------------------------------------------------------------
    if filter is not None:
        # Overwrite (or add) the ``filter`` entry used by Chroma.
        qa_chain.retriever.search_kwargs["filter"] = filter
    else:
        # Ensure no stale filter remains from a previous call.
        qa_chain.retriever.search_kwargs.pop("filter", None)

    # ---------------------------------------------------------------------
    # 6. Run the chain.  ``chat_history`` is an empty list for a single‑turn
    #    query, but you could pass a list of (question, answer) tuples for a
    #    multi‑turn conversation.
    # ---------------------------------------------------------------------
    result = qa_chain({"question": query, "chat_history": []})
    return result


if __name__ == "__main__":

    output = run_agent(query, pdf_paths, filter=metadata_filter)
