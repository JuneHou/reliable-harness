import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables and set up an in‑memory LLM cache
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# A pool of API keys – you can extend this if you have more than one key
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]


def get_random_key(key_pool: list[str]) -> str:
    """Return a random non‑None API key from the pool."""
    return random.choice([k for k in key_pool if k is not None])


def run_agent(query: str, pdf_paths: list[str]):
    """Create a Chroma vector store from the supplied PDFs and answer *query*.

    The function demonstrates how to apply a **metadata filter** so that only
    documents originating from a specific PDF are considered during retrieval.
    """
    # ------------------------------------------------------------------
    # 1️⃣ Load the PDFs, split them into chunks and tag each chunk with its
    #    source file name inside the ``metadata`` dictionary.
    # ------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        docs = loader.load()
        splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        chunks = splitter.split_documents(docs)
        for chunk in chunks:
            # ``metadata`` is a mutable dict on the Document object.
            chunk.metadata["source"] = pdf_path
        all_chunks.extend(chunks)

    # ------------------------------------------------------------------
    # 2️⃣ Build a Chroma vector store from the chunks.
    # ------------------------------------------------------------------
    db = Chroma.from_documents(all_chunks, embeddings)

    # ------------------------------------------------------------------
    # 3️⃣ Obtain a *retriever* that is capable of receiving a metadata filter
    #    at query‑time. ``as_retriever`` returns a ``VectorStoreRetriever`` that
    #    forwards any ``filter`` argument to Chroma's ``search`` method.
    # ------------------------------------------------------------------
    retriever = db.as_retriever()  # defaults to k=4; you can adjust via search_kwargs

    # ------------------------------------------------------------------
    # 4️⃣ Build the ConversationalRetrievalChain.
    # ------------------------------------------------------------------
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(
            openai_api_key=get_random_key(OPENAI_API_KEY_POOL),
            cache=True,
            temperature=0,
        ),
        retriever=retriever,
        verbose=True,
        return_source_documents=True,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Define a metadata filter – only documents whose ``source`` field is
    #    ``paper2.pdf`` will be considered.
    # ------------------------------------------------------------------
    metadata_filter = {"source": "paper2.pdf"}

    # ------------------------------------------------------------------
    # 6️⃣ Execute the chain. The ``filter`` key is automatically passed to the
    #    retriever's ``get_relevant_documents`` method.
    # ------------------------------------------------------------------
    result = qa_chain({"question": query, "chat_history": [], "filter": metadata_filter})
    return result


if __name__ == "__main__":
    # Example usage – replace with real PDF file names.
    query = "What is Theia?"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    answer = run_agent(query, pdf_paths)
    print(answer["answer"])
