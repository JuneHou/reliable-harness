import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables and initialise in‑memory LLM cache
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# A pool of API keys – you can extend this if you rotate keys
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]


def get_random_key(key_pool: list[str]) -> str:
    """Return a random non‑empty key from the pool.
    In production you would want a more robust selection/rotation strategy.
    """
    return random.choice([k for k in key_pool if k])


def run_agent(query: str, pdf_paths: list[str], source_filter: dict | None = None):
    """Answer *query* using the supplied PDFs.

    Parameters
    ----------
    query: str
        The natural‑language question.
    pdf_paths: list[str]
        Paths to the PDF documents that should be indexed.
    source_filter: dict | None
        Optional metadata filter – e.g. ``{"source": "paper2.pdf"}``.  When
        supplied only chunks whose ``metadata['source']`` matches the filter are
        considered during retrieval.
    """
    # ---------------------------------------------------------------------
    # 1. Load, split and annotate the PDFs
    # ---------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        docs = loader.load()
        splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        chunks = splitter.split_documents(docs)
        # Store the origin of each chunk so we can filter on it later
        for chunk in chunks:
            chunk.metadata["source"] = pdf_path
        all_chunks.extend(chunks)

    # ---------------------------------------------------------------------
    # 2. Build a vector store (Chroma) from the chunks
    # ---------------------------------------------------------------------
    db = Chroma.from_documents(all_chunks, embeddings)

    # ---------------------------------------------------------------------
    # 3. Create a retriever – we attach the filter *once* if it is static.
    #    For a dynamic filter we simply replace ``search_kwargs`` right before
    #    the call.
    # ---------------------------------------------------------------------
    retriever = db.as_retriever()
    if source_filter is not None:
        # ``search_kwargs`` are passed straight to the underlying vector store
        retriever.search_kwargs = {"filter": source_filter}

    # ---------------------------------------------------------------------
    # 4. Build the ConversationalRetrievalChain
    # ---------------------------------------------------------------------
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(
            openai_api_key=get_random_key(OPENAI_API_KEY_POOL),
            cache=True,
            temperature=0,
        ),
        retriever=retriever,
        verbose=True,
        return_source_documents=True,
    )

    # ---------------------------------------------------------------------
    # 5. Run the chain – ``chat_history`` is empty for this simple demo.
    # ---------------------------------------------------------------------
    result = qa_chain({"question": query, "chat_history": []})
    return result


if __name__ == "__main__":
    # Example usage – only ``paper2.pdf`` should be consulted for the answer.
    query_text = "What is Theia?"
    pdf_files = ["paper1.pdf", "paper2.pdf"]
    metadata_filter = {"source": "paper2.pdf"}
    answer = run_agent(query_text, pdf_files, source_filter=metadata_filter)
    print(answer["answer"]) 
    # Optional: print the source documents to verify the filter worked
    for doc in answer.get("source_documents", []):
        print("--- source:", doc.metadata.get("source"))
        print(doc.page_content[:200], "\n")
