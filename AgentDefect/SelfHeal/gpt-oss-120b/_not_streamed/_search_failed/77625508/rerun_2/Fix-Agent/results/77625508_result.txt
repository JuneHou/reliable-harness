import os
from dotenv import load_dotenv

from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_verbose
from langchain.callbacks import StdOutCallbackHandler

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Enable global verbose mode – this makes LangChain include debugging information
set_verbose(True)

def run_agent(topic: str) -> str:
    """Generate a joke about *topic* while printing intermediate LCEL steps.

    The chain is built with LCEL (prompt | model | parser) and configured to use
    ``StdOutCallbackHandler`` which prints the prompt, model inputs/outputs and
    any parsing steps to stdout.  The ``verbose`` flag is also set on the chain
    configuration so that LangChain’s internal logging respects the global
    setting.
    """
    model = ChatOpenAI(openai_api_key=api_key)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)

    # Build the LCEL chain and attach the StdOutCallbackHandler for verbose output
    chain = (prompt | model | output_parser).with_config(
        callbacks=[StdOutCallbackHandler()], verbose=True
    )

    # Invoke the chain – the handler will automatically print the intermediate steps
    return chain.invoke({"topic": topic})

if __name__ == "__main__":
    result = run_agent("ice cream")
    print("\nGenerated output:", result)
