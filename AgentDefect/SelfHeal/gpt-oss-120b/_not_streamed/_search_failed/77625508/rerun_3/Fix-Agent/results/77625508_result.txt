import os
from dotenv import load_dotenv

from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_verbose

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Enable global verbose mode (keeps legacy behaviour)
set_verbose(True)


def run_agent(topic: str) -> str:
    """Generate a joke about the supplied *topic* and log every LCEL step.

    The ``ChatOpenAI`` model and the LCEL chain are both configured with
    ``verbose=True`` so that LangChain prints the prompt sent to the model,
    the raw LLM response, and the final parsed output.
    """
    # Enable verbose logging on the model itself
    model = ChatOpenAI(openai_api_key=api_key, verbose=True)

    # Simple output parser – converts the LLM ``BaseMessage`` to a plain string
    output_parser = StrOutputParser()

    # Prompt template – ``ChatPromptTemplate`` works with ``ChatOpenAI``
    prompt = ChatPromptTemplate.from_template("tell me a joke about {topic}")

    # Build the LCEL runnable chain and explicitly turn on verbose mode for the chain
    chain = (prompt | model | output_parser).with_config({"verbose": True})

    # Invoke the chain; verbose mode will print the intermediate prompt and response
    return chain.invoke({"topic": topic})


if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
