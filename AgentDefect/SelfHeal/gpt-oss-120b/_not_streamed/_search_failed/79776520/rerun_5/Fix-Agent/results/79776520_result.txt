from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    """Create the React‑style agent with a memory‑backed checkpointer.

    The ``MemorySaver`` stores checkpoints in‑process, allowing the agent to
    recall prior turns when the same ``thread_id`` is used.
    """

    google_api_key = os.getenv("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    simple_search = TavilySearch(max_result=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key,
    )

    # ``MemorySaver`` will automatically persist the ``messages`` channel for us.
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory,
    )

    # All calls that share this ``thread_id`` will read/write from the same
    # checkpoint, giving the agent memory of earlier turns.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, config


def run_agent(agent_executor, config, user_input: str) -> str:
    """Send a single user message to the agent and return its reply.

    The memory handling is performed by ``MemorySaver`` – we only need to pass
    the new ``HumanMessage``. The checkpointer automatically fills the ``messages``
    channel with the prior conversation history.

    # ``stream`` yields intermediate states; the final step contains the answer.
    final_message = None

    if final_message is None:
        raise RuntimeError("Agent did not produce a response.")
    return final_message.content


def main():
    agent_executor, config = initialize_agent()
    response1 = run_agent(agent_executor, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, config, "What's my name?")

    print(response2)

    model = ChatGoogleGenerativeAI(

        google_api_key=google_api_key,
    )

    # MemorySaver is a lightweight in‑memory checkpoint store.

    agent_executor = create_react_agent(

        tools=tools,
        checkpointer=memory,

    # ``thread_id`` uniquely identifies the conversation; the same ID is

    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config
