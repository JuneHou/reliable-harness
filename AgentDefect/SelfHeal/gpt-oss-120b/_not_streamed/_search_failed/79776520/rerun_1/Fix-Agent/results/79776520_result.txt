import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    """Load environment variables, create the model, tools, and the React agent.

    Returns:
        tuple: (agent_executor, memory, config) where ``config`` contains the
        thread identifier used by the checkpointer to store conversational
        state.
    """
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # Initialise a simple web‑search tool.
    simple_search = TavilySearch(max_result=2)
    tools = [simple_search]

    # Gemini model used by the agent.
    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key,
    )

    # MemorySaver acts as a checkpointer that persists the thread state.
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory,
    )

    # ``thread_id`` uniquely identifies the conversation.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, memory, config, user_input: str) -> str:
    """Send a user message to the agent and return the assistant's reply.

    The ``MemorySaver`` checkpointer automatically stores the full chat history
    for the supplied ``thread_id``. By invoking the agent with just the new
    ``HumanMessage`` we let LangGraph handle context stitching, ensuring the
    assistant remembers earlier details (e.g., the user's name).
    """
    # The agent expects a dict with a ``messages`` key.
    input_data = {"messages": [HumanMessage(user_input)]}

    # ``invoke`` runs the agent to completion and internally updates the
    # checkpointer, so the next call will have access to the accumulated history.
    result = agent_executor.invoke(input_data, config)
    final_message = result["messages"][-1]
    return final_message.content


def main():
    agent_executor, memory, config = initialize_agent()
    response1 = run_agent(agent_executor, memory, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, memory, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
