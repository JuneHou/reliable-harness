from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()

def run_agent(prompt: str) -> str:
    """Run a conversational retrieval QA over the ``movies.csv`` file.

    The original implementation limited the retriever to a single nearest
    neighbour (``k=1``).  This caused the chain to return only the most
    similar document – e.g. only *author A* for movies from 1999 – even when the
    query matches multiple rows.  By increasing ``k`` (or leaving it at the
    default) the retriever returns a richer set of relevant documents, allowing
    the LLM to mention all matching authors.
    """

    # Initialise the embedding model (uses the OpenAI API key from .env)
    embeddings = OpenAIEmbeddings()

    # Load the CSV.  Each row becomes a Document with the raw row as the
    # ``page_content`` and the column names stored in ``metadata``.
    csv_loader = CSVLoader(file_path="movies.csv")
    documents = csv_loader.load()

    # Create a Chroma vector store from the documents.
    vectorstore = Chroma.from_documents(documents, embeddings)

    # Retrieve *more* than a single document.  Five is a reasonable default for
    # a small CSV; you can increase it further if you want exhaustive results.
    retriever = vectorstore.as_retriever(search_kwargs={"k": 5})

    # Build the conversational retrieval chain.
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # ``chat_history`` should be a list of (question, answer) tuples; an empty
    # list is appropriate for a single‑turn query.
    response = qa_chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return response["answer"]

if __name__ == "__main__":
    print(run_agent("which movies are action"))
