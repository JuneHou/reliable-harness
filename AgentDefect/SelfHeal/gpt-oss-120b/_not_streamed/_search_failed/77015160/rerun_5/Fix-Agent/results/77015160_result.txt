from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.prompts import PromptTemplate
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def get_chain():
    """Create a RetrievalQAWithSourcesChain that uses a custom (blank) prompt.

    By default ``RetrievalQAWithSourcesChain`` comes with a built‑in prompt that
    prefixes the answer with a header like "Here is the answer...".  The user
    wants to suppress that default template and provide either an empty prompt
    or a custom one.  We achieve this by constructing a ``PromptTemplate`` that
    only contains the question placeholder (or an empty string) and passing it
    to the underlying ``StuffDocumentsChain`` via ``combine_documents_chain_kwargs``.
    """
    # Initialise the LLM
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    # Sample documents and their metadata
    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    # Build the vector store
    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # Create a prompt that does *not* add any extra wording – only the question.
    # If you really want a completely empty prompt you could use "" instead.
    custom_prompt = PromptTemplate.from_template("{question}")

    # Build the chain, injecting the custom prompt.
    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        combine_documents_chain_kwargs={"prompt": custom_prompt},
    )
    return chain


def run_agent(query: str):
    """Run the QA chain on *query* and return the answer with sources.
    """
    chain = get_chain()
    # The chain expects a dict with the key ``question``.
    result = chain({"question": query})
    return result

if __name__ == "__main__":
    # Example usage – ask a question about the stored documents.
    answer = run_agent("What country is mentioned in the first document?")
    print(answer)
