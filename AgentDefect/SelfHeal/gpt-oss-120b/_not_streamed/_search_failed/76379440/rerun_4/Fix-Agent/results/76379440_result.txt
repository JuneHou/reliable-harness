import os
from tqdm import tqdm
from langchain.docstore.document import Document
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from dotenv import load_dotenv

load_dotenv()

def run_agent():
    """Create a few collections, persist them, then retrieve the stored embeddings.

    The function returns a dictionary containing the documents and their
    corresponding embeddings for the *last* collection that was created.  The
    retrieval uses the public ``Chroma.get`` method with ``include=["embeddings"]``
    so that the embeddings are actually returned – accessing the private
    ``_collection`` attribute (as the original code did) returns ``None`` because
    the internal ``get`` method does not expose the embeddings by default.
    """

    embeddings = OpenAIEmbeddings()
    persist_dir = "./data"
    os.makedirs(persist_dir, exist_ok=True)

    # ---------------------------------------------------------------------
    # 1️⃣  Create and persist a couple of collections.
    # ---------------------------------------------------------------------
    for db_collection_name in tqdm(["class1-sub2-chap3", "class2-sub3-chap4"]):
        documents = []
        doc_ids = []
        cl, sub, chap = db_collection_name.split("-")
        for doc_index in range(3):
            content = f"This is {db_collection_name}-doc{doc_index}"
            doc = Document(
                page_content=content,
                metadata={
                    "chunk_num": doc_index,
                    "chapter": chap,
                    "class": cl,
                    "subject": sub,
                },
            )
            documents.append(doc)
            doc_ids.append(str(doc_index))

        # ``from_documents`` both creates the collection *and* stores the vectors.
        db = Chroma.from_documents(
            collection_name=db_collection_name,
            documents=documents,
            ids=doc_ids,
            embedding=embeddings,
            persist_directory=persist_dir,
        )
        # Persist to disk so that we can reload it later.
        db.persist()

    # ---------------------------------------------------------------------
    # 2️⃣  Add a single document to a new collection ("test_collection").
    # ---------------------------------------------------------------------
    initial_content = "This is an initial document content"
    document_id = "doc1"
    original_doc = Document(page_content=initial_content, metadata={"page": "0"})

    test_db = Chroma.from_documents(
        collection_name="test_collection",
        documents=[original_doc],
        ids=[document_id],
        embedding=embeddings,
        persist_directory=persist_dir,
    )
    test_db.persist()

    # ---------------------------------------------------------------------
    # 3️⃣  Retrieve the stored embeddings.
    # ---------------------------------------------------------------------
    # Reload the collection to demonstrate a typical read‑only workflow.
    # ``embedding_function`` is required but not used for a pure read operation.
    read_back = Chroma(
        collection_name="test_collection",
        embedding_function=embeddings,
        persist_directory=persist_dir,
    )

    # ``include=["embeddings"]`` asks Chroma to return the vectors.
    result = read_back._collection.get(include=["embeddings"])
    return result


if __name__ == "__main__":

def create_and_persist_collections():
    """Create sample collections and persist them to disk.
    This function is separated from the retrieval logic so that we can
    demonstrate how to load a persisted collection later and inspect the
    stored embeddings.
    """
    embeddings = OpenAIEmbeddings()
    persist_dir = "./data"
    os.makedirs(persist_dir, exist_ok=True)

    for collection_name in tqdm(["class1-sub2-chap3", "class2-sub3-chap4"]):
        documents = []
        ids = []
        # Create three dummy documents for each collection
        for doc_index in range(3):
            cl, sub, chap = collection_name.split("-")
            content = f"This is {collection_name}-doc{doc_index}"
            doc = Document(
                page_content=content,
                metadata={
                    "chunk_num": doc_index,
                    "chapter": chap,
                    "class": cl,
                    "subject": sub,
                },
            )
            documents.append(doc)
            ids.append(str(doc_index))

        # Initialise (or overwrite) the collection and persist it
        db = Chroma.from_documents(
            collection_name=collection_name,
            documents=documents,
            ids=ids,
            embedding=embeddings,
            persist_directory=persist_dir,
        )
        db.persist()
        # Explicitly delete the instance so the next loop iteration starts fresh
        del db

    # Add a separate test collection – also persisted for later retrieval

    test_collection = "test_collection"

    test_db.persist()
    del test_db


def retrieve_embeddings(collection_name: str, ids: list[str]):
    """Load a persisted collection and fetch stored embeddings.

    Args:
        collection_name: Name of the collection to load.
        ids: List of document IDs whose embeddings we want to inspect.

    Returns:
        The dictionary returned by ``Chroma.get`` which contains the ids,
        documents, metadatas and the raw embeddings as a list of ``np.ndarray``.
    """
    embeddings = OpenAIEmbeddings()
    persist_dir = "./data"

    # Load the existing collection – note we **must** supply the same embedding
    # function that was used during creation; otherwise the collection cannot be
    # initialised correctly.
    db = Chroma(
        persist_directory=persist_dir,
        collection_name=collection_name,
        embedding_function=embeddings,
    )

    # ``db.get`` returns a dict with keys: ids, documents, metadatas, embeddings.
    # If you omit ``ids`` you get the entire collection; providing the list
    # filters to the requested documents only.
    result = db.get(ids=ids)
    return result
