import os
import sys
import json
from pathlib import Path
from dotenv import load_dotenv

import openai
from langchain.chat_models import ChatOpenAI
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.document_loaders import JSONLoader
from langchain.indexes import VectorstoreIndexCreator
from langchain.indexes.vectorstore import VectorStoreIndexWrapper
from langchain.chains import ConversationalRetrievalChain

load_dotenv()


def _create_or_load_index(persist: bool, loader: JSONLoader) -> VectorStoreIndexWrapper:
    """Create a new index from *loader* or load an existing persisted index.

    Args:
        persist: Whether the index should be persisted to disk.
        loader: The loader that provides the documents for a fresh index.
    """
    if persist and os.path.exists("persist"):
        # Re‑use an existing persisted chroma collection
        vectorstore = Chroma(persist_directory="persist", embedding_function=OpenAIEmbeddings())
        return VectorStoreIndexWrapper(vectorstore=vectorstore)

    # Build a fresh index.  When ``persist`` is True we tell the creator to store the
    # collection on disk so it can be re‑loaded on subsequent runs.
    index_creator = VectorstoreIndexCreator(
        vectorstore_kwargs={"persist_directory": "persist"} if persist else {}
    )
    return index_creator.from_loaders([loader])


def run_agent(queries):
    """Run a conversational retrieval chain over a small JSON review dataset.

    The function creates (or loads) a vector store, builds a conversational chain and
    iterates over *queries*, returning a list with the generated answers.
    """
    # ---------------------------------------------------------------------
    # Prepare the data file – create it once if it does not already exist.
    # ---------------------------------------------------------------------
    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)
    review_file = data_dir / "review.json"

    if not review_file.exists():
        payload = {
            "reviews": [
                {"text": "Great hotel, excellent service and comfortable rooms."},
                {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
                {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
                {"text": "Average hotel. The room was okay, but nothing special."},
                {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
                {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
                {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
                {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
                {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
                {"text": "Not a bad hotel, but there are better options available in the area."},
            ]
        }
        review_file.write_text(json.dumps(payload, indent=2))

    # ---------------------------------------------------------------------
    # Load the JSON reviews. ``jq_schema`` extracts each element of the array,
    # and ``content_key='text'`` tells the loader to use the value of the
    # ``text`` field as the document's content.
    # ---------------------------------------------------------------------
    loader = JSONLoader(
        file_path=str(review_file),
        jq_schema=".reviews[]",
        content_key="text",
    )

    # ---------------------------------------------------------------------
    # Whether to persist the vector store between runs.  This is useful when the
    # same data is queried many times – it avoids re‑embedding.
    # ---------------------------------------------------------------------
    PERSIST = False

    # Build or load the index.
    index = _create_or_load_index(persist=PERSIST, loader=loader)

    # ---------------------------------------------------------------------
    # Build a ConversationalRetrievalChain.  The ``ChatOpenAI`` class expects the
    # model name to be provided via ``model_name`` – using ``model`` raises a
    # ``TypeError`` in recent versions of LangChain.
    # ---------------------------------------------------------------------
    llm = ChatOpenAI(model_name="gpt-3.5-turbo")
    chain = ConversationalRetrievalChain.from_llm(
        llm=llm,
        retriever=index.vectorstore.as_retriever(),
        return_source_documents=False,
    )

    chat_history = []
    results = []

    for query in queries:
        if query.lower() in {"quit", "q", "exit"}:
            sys.exit()
        response = chain({"question": query, "chat_history": chat_history})
        answer = response["answer"]
        print(answer)
        results.append(answer)
        chat_history.append((query, answer))

    return results
