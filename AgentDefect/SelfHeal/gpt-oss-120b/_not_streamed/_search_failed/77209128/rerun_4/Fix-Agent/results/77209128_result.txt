import json
from dotenv import load_dotenv
from sqlalchemy import create_engine, MetaData, Table, Column, JSON, Integer

from llama_index.core import SQLDatabase, Settings, VectorStoreIndex

class SqlQueryEngine:
    """Encapsulates creation of a SQL‑Table‑aware query engine.

    The class builds a simple table, attaches the custom metadata as a
    *column comment*, and then creates a ``SQLTableSchema`` for each table
    that includes the comment in the ``context_str`` field. ``context_str``
    is what LlamaIndex uses to enrich the LLM prompt with table‑specific
    information, so without populating it the engine cannot answer questions
    about the meaning of fields.
    """

    def __init__(self) -> None:
        # This will hold the generated ``SQLTableSchema`` objects.
        self.schema: list[SQLTableSchema] = []

    def _create_snapshot_table(self, engine) -> MetaData:
        """Create (or reflect) the ``snapshots`` table.

        The table has a single JSON column called ``data``. The column comment
        stores the JSON‑encoded ``metadata_dict`` so it can be inspected later
        when we build the ``SQLTableSchema`` objects.
        """
        metadata_obj = MetaData()
        # Encode our metadata dictionary as a JSON string – this is the format
        # that SQLAlchemy expects for a column comment.
        metadata_json = json.dumps(metadata_dict)
        Table(
            "snapshots",
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
        )
        # Create the table in the chosen database (no‑op if it already exists).
        metadata_obj.create_all(engine)
        return metadata_obj

    def _build_schema(self, metadata_obj: MetaData) -> None:
        """Populate ``self.schema`` with ``SQLTableSchema`` objects.

        For each reflected table we pull the comment from its columns, decode
        the JSON (if possible) and place the resulting string into the
        ``context_str`` attribute of the ``SQLTableSchema``.
        """
        for table_name, table in metadata_obj.tables.items():
            # Gather column comments – we only have one column with a comment
            # but the logic works for any number of columns.
            col_comments: dict[str, str] = {}
            for col in table.columns:
                if col.comment:
                    # Attempt to load JSON; fall back to raw string on error.
                    try:
                        col_comments[col.name] = json.loads(col.comment)
                    except Exception:
                        col_comments[col.name] = col.comment
            # ``context_str`` is a JSON‑stringified version of the comments.
            context_str = json.dumps(col_comments) if col_comments else ""
            self.schema.append(SQLTableSchema(table_name=table_name, context_str=context_str))

    def query_engine(self) -> SQLTableRetrieverQueryEngine:
        # -----------------------------------------------------------------
        # 1. Initialise the DB engine and create the snapshot table.
        # -----------------------------------------------------------------
        engine = create_engine(pg_uri)
        metadata_obj = self._create_snapshot_table(engine)

        # -----------------------------------------------------------------
        # 2. Build the LlamaIndex schema objects, injecting our custom metadata.
        # -----------------------------------------------------------------
        self._build_schema(metadata_obj)

        # -----------------------------------------------------------------
        # 3. Wire the LlamaIndex components together.
        # -----------------------------------------------------------------
        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        # ``ObjectIndex`` stores the ``SQLTableSchema`` objects in a vector store
        # (here we rely on the default in‑memory store).  The retriever will later
        # fetch the most relevant schema based on the user query.
        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        # LLM configuration – temperature 0.9 matches the original example.
        llm = OpenAI(temperature=0.9, model="gpt-3.5-turbo")
        Settings.llm = llm

        # The retriever‑based query engine.
        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )
        return query_engine


def run_agent(query: str):
    print("Initializing SQL Query Engine...")
    sql_engine = SqlQueryEngine()
    query_engine = sql_engine.query_engine()
    print("Query engine created successfully!")

    print(f"\nExecuting query: {query}")

    print(f"Response: {response}")

    return sql_engine, response
