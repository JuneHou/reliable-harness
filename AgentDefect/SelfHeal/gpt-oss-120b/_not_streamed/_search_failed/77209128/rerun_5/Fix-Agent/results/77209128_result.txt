class SqlQueryEngine:
    """Wraps the creation of a SQLTableRetrieverQueryEngine that is aware of column comments.

    The key fix is that we now populate ``SQLTableSchema.context_str`` with the comment
    attached to the ``data`` JSON column.  The original code built ``SQLTableSchema``
    objects without any context, so the LLM never saw the descriptive metadata.
    """

    def __init__(self) -> None:
        self.schema: list[SQLTableSchema] = []

    def _create_table(self) -> MetaData:
        """Create the ``snapshots`` table with a JSON column that carries a comment.

        Returns the ``MetaData`` instance containing the freshly defined table.
        """
        engine = create_engine(pg_uri)
        metadata_obj = MetaData()
        Table(
            "snapshots",
            metadata_obj,
            Column("id", Integer, primary_key=True),
            Column("data", JSON, comment=metadata_json),
        )
        # Ensure the table exists in the underlying DB.
        metadata_obj.create_all(engine)
        return metadata_obj

    def _build_schema(self, metadata_obj: MetaData) -> None:
        """Populate ``self.schema`` with ``SQLTableSchema`` objects that include context.

        For each table we extract the comment attached to the ``data`` column (if present)
        and store it in ``SQLTableSchema.context_str``.  LlamaIndex's query engine looks
        at ``context_str`` to provide extra information to the LLM during query time.
        """
        for table in metadata_obj.tables.values():
            # Retrieve the comment from the ``data`` column; fallback to an empty string.
            comment: str | None = None
            if "data" in table.c:
                comment = table.c["data"].comment  # type: ignore[attr-defined]
            context = comment if comment is not None else ""
            self.schema.append(
                SQLTableSchema(table_name=table.name, context_str=context)
            )

    def query_engine(self) -> SQLTableRetrieverQueryEngine:
        """Create and return a fully‑configured ``SQLTableRetrieverQueryEngine``.

        metadata_obj = self._create_table()

        # Step 2 – build the schema objects with the desired context.
        self._build_schema(metadata_obj)

        engine = create_engine(pg_uri)
        sql_database = SQLDatabase(engine)
        nodes = SQLTableNodeMapping(sql_database)

        index = ObjectIndex.from_objects(self.schema, nodes, VectorStoreIndex)

        Settings.llm = llm

        # Final query engine.
        query_engine = SQLTableRetrieverQueryEngine(
            sql_database,
            index.as_retriever(similarity_top_k=1),
        )
        return query_engine
