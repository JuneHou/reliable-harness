import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import StructuredOutputParser, ResponseSchema

load_dotenv()


def get_output_parser():
    """Create a StructuredOutputParser that expects a list of dictionaries.

    Each dictionary should contain the keys ``identifier`` and ``text``.  The
    parser therefore defines a single top‑level field called ``results`` whose
    description makes it clear that the value must be a JSON *array* of objects
    with those two keys.  LangChain will include this description in the format
    instructions that are sent to the LLM, prompting it to return the correct
    shape.
    """
    results_schema = ResponseSchema(
        name="results",
        description=(
            "A JSON array where each element is an object with two keys: "
            "'identifier' (the missing identifier) and 'text' (the missing text)."
        ),
    )
    return StructuredOutputParser.from_response_schemas([results_schema])


def predict_result(msg: str, PROMPT_TEMPLATE: str):
    """Invoke the LLM and parse its response using the list‑output parser.

    Parameters
    ----------
    msg: str
        The error message that should be parsed.
    PROMPT_TEMPLATE: str
        The template that contains placeholders ``{msg}`` and ``{format_instructions}``.
    """

        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    # ``parse`` returns a dict matching the schema – i.e. {'results': [...]}
    parsed = output_parser.parse(response.content)
    return parsed
