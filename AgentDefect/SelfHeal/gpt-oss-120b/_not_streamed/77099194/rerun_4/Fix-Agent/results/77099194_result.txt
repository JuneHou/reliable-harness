from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()

def run_agent(prompt: str) -> str:
    """Answer a question about the CSV using a conversational retrieval chain.

    The original implementation limited the retriever to ``k=1`` which means only the
    single most‑relevant document was returned for any query.  This caused queries that
    should match multiple rows – for example *all action movies from 1999* – to return
    only one row (author A) instead of all relevant rows (author A **and** author C).

    By increasing ``k`` (or omitting it altogether, which defaults to a reasonable
    number) we let the retriever return a broader set of matching documents, allowing
    the LLM to incorporate all relevant entries in its answer.
    """
    # Initialise embeddings – the OpenAI key is pulled from the environment.
    embeddings = OpenAIEmbeddings()

    # Load the CSV file as LangChain Documents. Each row becomes a separate document.
    csv_loader = CSVLoader("movies.csv")
    csv_documents = csv_loader.load()

    # Build a vector store from the documents.
    vector_store = Chroma.from_documents(csv_documents, embeddings)

    # Use a larger ``k`` so that multiple relevant rows are retrieved.
    retriever = vector_store.as_retriever(search_kwargs={"k": 10})

    # ConversationalRetrievalChain that returns the answer and source docs.
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # ``chat_history`` is an empty list for a single‑turn query.
    result = chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return result["answer"]

if __name__ == "__main__":
    print(run_agent("which movies are action"))
