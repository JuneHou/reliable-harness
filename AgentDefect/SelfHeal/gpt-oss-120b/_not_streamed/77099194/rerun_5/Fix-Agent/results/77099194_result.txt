import os
from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()

def run_agent(prompt: str) -> str:
    """Answer a question against the ``movies.csv`` dataset.

    The original implementation retrieved only the *single* most similar
    document (``k=1``) which meant queries that should match several rows –
    e.g. *all action movies from 1999* – returned just one result (author A).
    By increasing ``k`` we surface multiple relevant documents, allowing the
    LLM to synthesise a more complete answer that includes all matching authors
    (author A, author C, …).
    """

    # Initialise the embedding model
    embeddings = OpenAIEmbeddings()

    # Load the CSV; each row becomes a Document with the row data in metadata
    csv_loader = CSVLoader(file_path="movies.csv")
    documents = csv_loader.load()

    # Build a vector store from the documents
    vectorstore = Chroma.from_documents(documents, embeddings)

    # Retrieve *multiple* relevant documents (adjust ``k`` as needed)
    retriever = vectorstore.as_retriever(search_kwargs={"k": 5})

    # Set up the conversational retrieval chain
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # ``chat_history`` should be a list of (question, answer) tuples – use an empty
    # list for a single‑turn query.
    result = qa_chain({"question": prompt, "chat_history": []})
    return result["answer"]

if __name__ == "__main__":
    # Example usage – ask for action movies (or any other query)
    print(run_agent("which movies are action"))
