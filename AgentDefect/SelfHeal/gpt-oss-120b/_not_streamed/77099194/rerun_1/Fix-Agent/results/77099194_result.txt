import os
from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()


def run_agent(prompt: str, k: int = 5) -> str:
    """Return an answer to *prompt* using the CSV data.

    Parameters
    ----------
    prompt: str
        The user's question.
    k: int, optional
        Number of most‑relevant documents to retrieve from the vector store.
        The default (5) is enough to capture multiple rows that match the query
        (e.g., all movies from 1999).  Previously the code forced ``k=1`` which
        caused only a single document – often from the first matching author –
        to be considered, leading to incomplete answers.
    """

    # Initialise the embedding model
    embeddings = OpenAIEmbeddings()

    # Load the CSV file into LangChain ``Document`` objects
    csv_loader = CSVLoader(file_path="movies.csv")
    documents = csv_loader.load()

    # Build a Chroma vector store from the documents
    vectorstore = Chroma.from_documents(documents, embeddings)

    # Configure the retriever – we now expose ``k`` so callers can decide how
    # many relevant rows should be considered.
    retriever = vectorstore.as_retriever(search_kwargs={"k": k})

    # Set up a conversational retrieval chain with a temperature that provides
    # deterministic yet fluent answers.
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # Execute the chain. ``chat_history`` is kept empty for a single‑turn query.
    response = chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return response["answer"]


if __name__ == "__main__":
    # Example: ask for all action movies (or any other query). Adjust ``k`` as needed.
    print(run_agent("which movies are action", k=10))
