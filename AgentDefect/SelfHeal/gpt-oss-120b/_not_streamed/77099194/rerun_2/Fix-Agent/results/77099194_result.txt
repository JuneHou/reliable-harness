import os
from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

# Load environment variables (e.g., OpenAI API key)
load_dotenv()

# Create embeddings and vector store once – they can be reused for every query
embeddings = OpenAIEmbeddings()

# Load the CSV – each row becomes a Document with the row content in `page_content`
# and the column names as metadata (e.g., "author", "year", "genre", ...)
csv_loader = CSVLoader(file_path="movies.csv")
documents = csv_loader.load()

# Build a Chroma vector store from the documents
vector_store = Chroma.from_documents(documents, embeddings)

# Configure the retriever to return *multiple* relevant documents.
# Using a larger `k` (e.g., 10) ensures that all matching movies are considered.
retriever = vector_store.as_retriever(search_kwargs={"k": 10})

# Build a conversational retrieval chain.  `chat_history` is expected to be a list of
# (human, AI) tuples, so we initialise it as an empty list.
qa_chain = ConversationalRetrievalChain.from_llm(
    llm=ChatOpenAI(temperature=0.3),
    retriever=retriever,
    return_source_documents=True,
)

def run_agent(prompt: str) -> str:
    """Ask a question about the CSV data and return the answer.

    Args:
        prompt: The natural‑language question.
    Returns:
        The answer generated by the LLM.
    """
    # `chat_history` is an empty list for a standalone query.
    result = qa_chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return result["answer"]

if __name__ == "__main__":
    # Example query – you can replace this with any question about the CSV.
    print(run_agent("which movies are action"))
