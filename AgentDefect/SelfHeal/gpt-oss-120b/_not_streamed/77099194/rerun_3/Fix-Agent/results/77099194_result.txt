import os
from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

# Load environment variables (e.g., OpenAI API key)
load_dotenv()

def run_agent(prompt: str) -> str:
    """Answer a question using a CSV‑based vector store.

    The original implementation returned only a single document because the
    retriever was configured with ``k=1``. This limited the answer to the most
    similar row (e.g., a single author for a given year) and omitted other
    relevant rows. The fix expands the number of retrieved documents and passes
    a proper ``chat_history`` structure expected by ``ConversationalRetrievalChain``.
    """

    # Initialize the embedding model
    embeddings = OpenAIEmbeddings()

    # Load CSV data – each row becomes a Document with ``page_content`` equal to
    # the raw CSV line and metadata containing the column names.
    csv_loader = CSVLoader(file_path="movies.csv")
    documents = csv_loader.load()

    # Create a Chroma vector store from the documents
    vectorstore = Chroma.from_documents(documents, embeddings)

    # Retrieve more than one relevant document (adjust ``k`` as needed). Using a
    # higher ``k`` ensures that all matching movies (e.g., multiple authors for
    # a given year) are considered.
    retriever = vectorstore.as_retriever(search_kwargs={"k": 10})

    # Build the conversational retrieval chain
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # ``chat_history`` should be a list of (user, assistant) tuples. An empty list
    # represents a fresh conversation.
    response = qa_chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return response["answer"]

if __name__ == "__main__":
    print(run_agent("which movies are action"))
