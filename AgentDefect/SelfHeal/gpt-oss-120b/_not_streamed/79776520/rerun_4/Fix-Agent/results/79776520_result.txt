from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    """Initialises the LangGraph React agent with a memory checkpoint.

    Returns:
        agent_executor: The agent executor that can be invoked.
        memory: The MemorySaver checkpoint object (kept for possible external use).
        config: A configuration dict containing a stable thread identifier.
    """
    load_dotenv()

    google_api_key = os.getenv("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # Simple web‑search tool used by the agent
    simple_search = TavilySearch(max_result=2)
    tools = [simple_search]

    # Gemini model
    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key,
    )

    # MemorySaver will automatically store the agent's state per thread
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory,
    )

    # All calls that share this thread_id will share the same memory
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, config, user_input: str) -> str:
    """Runs the agent for a single user turn.

    The `MemorySaver` attached to the executor automatically remembers the
    conversation history, so we only need to provide the new user message.
    """
    # The agent expects a dict with a "messages" key containing the new HumanMessage

        {"messages": [HumanMessage(content=user_input)]},
        config,
    )
    # The final AI response is the last message in the returned ``messages`` list
    return result["messages"][-1].content


def main():

    # First turn – the user introduces themselves

    # Second turn – the agent should recall the name from memory
