import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    """Load environment variables, set up the model, tools and the REACT agent.

    Returns:
        tuple: (agent_executor, config) where ``agent_executor`` is the REACT
        agent ready to be invoked/streamed and ``config`` contains the thread
        identifier used by the MemorySaver checkpoint.
    """
    load_dotenv()

    google_api_key = os.getenv("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # Initialise the search tool (it already implements the Tool interface).
    simple_search = TavilySearch(max_result=2)
    tools = [simple_search]

    # Initialise the Gemini model.
    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key,
    )

    # MemorySaver will automatically persist the thread state between calls.
    checkpointer = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=checkpointer,
    )

    # A deterministic thread ID so the same conversation history is used.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, config


def run_agent(agent_executor, config, user_input):
    """Send a single user message to the agent and return the assistant's reply.

    The ``MemorySaver`` attached to the agent automatically retrieves prior
    messages for the supplied ``thread_id`` – we only need to provide the new
    ``HumanMessage``.
    """
    # Build the input payload – only the newest human message.
    payload = {"messages": [HumanMessage(content=user_input)]}
    final_message = None
    for step in agent_executor.stream(payload, config, stream_mode="values"):
        # Each ``step`` contains the updated ``messages`` list; the last one is
        # the most recent assistant response.
        final_message = step["messages"][-1]
    return final_message.content if final_message else ""


def main():
    agent_executor, config = initialize_agent()
    # First turn – the agent learns the user's name.
    response1 = run_agent(agent_executor, config, "Hi! I am Bob!")
    # Second turn – the agent should remember the name.
    response2 = run_agent(agent_executor, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()

    # The agent (and the underlying checkpoint) will automatically prepend the
