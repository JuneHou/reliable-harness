import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    """Load environment variables, set up the model, tools and agent.

    Returns:
        tuple: (agent_executor, memory, config) where ``agent_executor`` is the
        LangGraph React agent, ``memory`` is a ``MemorySaver`` checkpoint that
        persists the conversation thread, and ``config`` contains the thread
        identifier.
    """
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # Initialise the search tool – it already implements the ``Tool`` interface.
    simple_search = TavilySearch(max_result=2)
    tools = [simple_search]

    # Initialise the Gemini model.
    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key,
    )

    # Use ``MemorySaver`` so that the agent can retain conversation history.
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory,
    )

    # Unique thread ID – this groups messages belonging to the same conversation.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, config, user_input: str) -> str:
    """Send a single user message to the agent and return its reply.

    The LangGraph ``MemorySaver`` automatically injects prior messages stored in
    the checkpoint, so we only provide the new ``HumanMessage``.
    """
    # Only the newest user message is supplied – the checkpoint supplies the
    # previous message history.
    input_payload = {"messages": [HumanMessage(user_input)]}

    final_message = None
    for step in agent_executor.stream(
        input_payload,
        config,
        stream_mode="values",
    ):
        # The latest step always contains the most recent agent message.
        final_message = step["messages"][-1]

    if final_message is None:
        raise RuntimeError("Agent did not produce a response")
    return final_message.content


def main():
    agent_executor, memory, config = initialize_agent()
    # First interaction – the agent learns the user's name.
    response1 = run_agent(agent_executor, config, "Hi! I am Bob!")
    # Second interaction – the agent should recall the name from memory.
    response2 = run_agent(agent_executor, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
