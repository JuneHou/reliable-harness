import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage, BaseMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    load_dotenv()
    google_api_key = os.getenv("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    search = TavilySearch(max_result=2)
    tools = [search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key,
    )

    checkpointer = MemorySaver()
    agent = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=checkpointer,
    )

    config = {"configurable": {"thread_id": "agent003"}}
    return agent, checkpointer, config


def run_agent(agent, checkpointer, config, user_input: str) -> str:
    # Retrieve prior conversation from the checkpoint if it exists
    state = checkpointer.get(config)
    if (
        state
        and "channel_values" in state
        and "messages" in state["channel_values"]
    ):
        history = state["channel_values"]["messages"]
    else:
        history = []

    # Append the new user message (explicitly set the content)
    messages: list[BaseMessage] = history + [HumanMessage(content=user_input)]

    # Run the agent; invoke returns the final state after the graph finishes.
    result = agent.invoke({"messages": messages}, config)

    # The assistant's reply is the last message in the returned list.
    reply_message = result["messages"][-1]
    return reply_message.content


def main():
    agent, checkpointer, config = initialize_agent()
    response1 = run_agent(agent, checkpointer, config, "Hi! I am Bob!")
    response2 = run_agent(agent, checkpointer, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
