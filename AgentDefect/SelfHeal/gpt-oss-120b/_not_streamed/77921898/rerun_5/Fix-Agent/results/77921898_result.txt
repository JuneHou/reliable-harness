import os
from dotenv import load_dotenv, find_dotenv
from glob import glob
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader

# Load environment variables (including OPENAI_API_KEY)
_ = load_dotenv(find_dotenv())


def run_agent(user_input: str, document_paths=None, persist_directory: str = "chroma_db"):
    """Create (or load) a Chroma vector store and retrieve the most relevant documents.

    Args:
        user_input: The query string.
        document_paths: Optional list of file paths to ingest. If ``None`` the function will try
            to load an existing persisted collection from ``persist_directory``.
        persist_directory: Directory where the Chroma collection is stored.

    Returns:
        List of retrieved ``Document`` objects.
    """

    # Grab the API key from the environment and make sure LangChain sees it.
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment variables")
    os.environ["OPENAI_API_KEY"] = openai_api_key

    # Initialise the embeddings model.
    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    # ---------------------------------------------------------------------
    # 1️⃣  Index creation / ingestion (only when document_paths are supplied)
    # ---------------------------------------------------------------------
    if document_paths:
        all_texts = []
        for doc_path in document_paths:
            loader = TextLoader(doc_path)
            documents = loader.load()
            # Split each document into manageable chunks.
            splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
            chunks = splitter.split_documents(documents)
            all_texts.extend(chunks)

        # Build a new Chroma collection from the chunks and persist it.
        docsearch = Chroma.from_documents(
            all_texts,
            embeddings,
            persist_directory=persist_directory,
        )
        # Ensure the vectors are written to disk for later runs.
        docsearch.persist()
    else:
        # ---------------------------------------------------------------
        # 2️⃣  Load an existing collection if no new documents are given.
        # ---------------------------------------------------------------
        if not os.path.isdir(persist_directory):
            raise FileNotFoundError(
                f"Persist directory '{persist_directory}' does not exist and no documents were provided to create one."
            )
        docsearch = Chroma(persist_directory=persist_directory, embedding_function=embeddings)

    # ---------------------------------------------------------------------
    # 3️⃣ Retrieve the most relevant chunks for the user query.
    # ---------------------------------------------------------------------
    retriever = docsearch.as_retriever(search_kwargs={"k": 5})
    docs = retriever.get_relevant_documents(user_input)
    print(f"Number of documents retrieved: {len(docs)}")
    return docs


if __name__ == "__main__":
    # Load all txt files inside the "docs" folder.
    document_paths = glob("docs/*.txt")
    query = "What is machine learning?"
    # First run will ingest the documents and persist the vector store.
    # Subsequent runs can omit ``document_paths`` to just query the persisted store.
    docs = run_agent(query, document_paths=document_paths)

    print("\n=== User Query Result ===")
    print(f"Query: {query}")
    print(f"Number of relevant documents: {len(docs)}")
    for i, doc in enumerate(docs):
