import os
from dotenv import load_dotenv, find_dotenv
from glob import glob
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader

# Load environment variables (e.g., OPENAI_API_KEY)
_ = load_dotenv(find_dotenv())


def run_agent(user_input: str, document_paths=None, persist_directory: str = "chroma_db"):
    """Create (or load) a Chroma vector store from the given documents and
    retrieve the most relevant chunks for *user_input*.

    Args:
        user_input: The query string.
        document_paths: Optional list of file paths to load into the vector store.
        persist_directory: Directory where the Chroma DB is persisted.
    """

    # Ensure the OpenAI key is available for the embedding model
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment variables")
    os.environ["OPENAI_API_KEY"] = openai_api_key

    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    # If we received new documents, (re)build the vector store and persist it.
    if document_paths:
        all_texts = []
        for doc_path in document_paths:
            loader = TextLoader(doc_path)
            documents = loader.load()
            # Split each document into manageable chunks.
            text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
            texts = text_splitter.split_documents(documents)
            all_texts.extend(texts)

        # Create (or overwrite) the persisted Chroma database with the new chunks.
        docsearch = Chroma.from_documents(
            all_texts, embeddings, persist_directory=persist_directory
        )
    else:
        # No new documents – just load the existing persisted DB.
        docsearch = Chroma(
            persist_directory=persist_directory, embedding_function=embeddings
        )

    # Retrieve the top‑k most relevant chunks.
    retriever = docsearch.as_retriever(search_kwargs={"k": 5})
    docs = retriever.get_relevant_documents(user_input)
    print(f"Number of documents retrieved: {len(docs)}")
    return docs


if __name__ == "__main__":
    # Load all *.txt files in the ./docs folder.
    document_paths = glob("docs/*.txt")
    query = "What is machine learning?"
    docs = run_agent(query, document_paths)

    print("\n=== User Query Result ===")
    print(f"Query: {query}")
    print(f"Number of relevant documents: {len(docs)}")

    for i, doc in enumerate(docs):
        print(f"\n--- Document {i+1} ---")
        print(doc.page_content)
