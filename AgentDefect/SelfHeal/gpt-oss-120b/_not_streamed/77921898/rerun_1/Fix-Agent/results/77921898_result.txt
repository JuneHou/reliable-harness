import os
from dotenv import load_dotenv, find_dotenv
from glob import glob
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader

# Load environment variables (e.g., OPENAI_API_KEY)
_ = load_dotenv(find_dotenv())


def run_agent(user_input: str, document_paths=None, persist_directory: str = "chroma_db"):
    """Create (or load) a Chroma vector store from the supplied documents and
    retrieve the most relevant chunks for *user_input*.

    Args:
        user_input: The query string.
        document_paths: Optional list of file paths to load and index. If ``None``
            the function will attempt to load an existing persisted Chroma DB.
        persist_directory: Directory where the Chroma DB is persisted.
    """
    # ---------------------------------------------------------------------
    # Initialise OpenAI embeddings – the key can be provided via the env var
    # ``OPENAI_API_KEY``. LangChain's ``OpenAIEmbeddings`` will read the env var
    # automatically, but we set it explicitly for clarity.
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY environment variable not set")
    os.environ["OPENAI_API_KEY"] = openai_api_key
    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    # ---------------------------------------------------------------
    # If document paths are supplied we (re)build the vector store.
    # Otherwise we load an existing persisted store.
    # ---------------------------------------------------------------
    if document_paths:
        # Load and split each document into manageable chunks
        all_chunks = []
        for doc_path in document_paths:
            loader = TextLoader(doc_path)
            documents = loader.load()
            splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
            chunks = splitter.split_documents(documents)
            all_chunks.extend(chunks)

        # Create a new Chroma collection from the chunks and persist it.
        docsearch = Chroma.from_documents(
            all_chunks,
            embeddings,
            persist_directory=persist_directory,
        )
        # Ensure the collection is written to disk for later runs.
        docsearch.persist()
    else:
        # Load the previously persisted collection (may raise if it does not exist).
        docsearch = Chroma(
            persist_directory=persist_directory,
            embedding_function=embeddings,
        )

    # -----------------------------------------------------------------
    # Retrieve the top‑k most relevant documents.
    # -----------------------------------------------------------------
    retriever = docsearch.as_retriever(search_kwargs={"k": 5})
    docs = retriever.get_relevant_documents(user_input)
    print(f"Number of documents retrieved: {len(docs)}")
    return docs


if __name__ == "__main__":
    # Grab all .txt files inside the "docs" folder.
    document_paths = glob("docs/*.txt")
    query = "What is machine learning?"
    # On the first run supply the paths so the DB gets built; on subsequent
    # runs you can set ``document_paths = None`` to reuse the persisted DB.
    docs = run_agent(query, document_paths=document_paths)

    print("\n=== User Query Result ===")
    print(f"Query: {query}")
    print(f"Number of relevant documents: {len(docs)}")

    for i, doc in enumerate(docs):
        print(f"\n--- Document {i+1} ---")
        print(doc.page_content)
