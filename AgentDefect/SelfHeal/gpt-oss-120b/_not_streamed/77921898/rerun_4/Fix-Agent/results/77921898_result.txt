import os
from dotenv import load_dotenv, find_dotenv
from glob import glob
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader

# Load environment variables (e.g., OPENAI_API_KEY)
_ = load_dotenv(find_dotenv())


def run_agent(user_input: str, document_paths=None, persist_directory: str = "chroma_db"):
    """Create (or load) a Chroma vector store from the supplied documents and retrieve
    the most relevant chunks for *user_input*.

    Parameters
    ----------
    user_input: str
        The query string.
    document_paths: list[str] | None
        List of file paths to load. If ``None`` the function will try to load an existing
        persisted Chroma collection from *persist_directory*.
    persist_directory: str
        Directory used by Chroma to store the vector collection.
    """

    # ---------------------------------------------------------------------
    # Initialise OpenAI embeddings – the key is read from the environment.
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment variables")
    os.environ["OPENAI_API_KEY"] = openai_api_key
    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    # ---------------------------------------------------------------------
    # Build a new Chroma collection from the supplied documents or load an
    # existing one.  The previous buggy version overwrote the freshly created
    # collection with an empty one, which caused the retriever to return zero
    # documents.
    # ---------------------------------------------------------------------
    if document_paths:
        # Load, split and embed the documents.
        all_texts = []
        for doc_path in document_paths:
            loader = TextLoader(doc_path)
            documents = loader.load()
            splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
            chunks = splitter.split_documents(documents)
            all_texts.extend(chunks)

        # Create (and persist) the vector store.
        docsearch = Chroma.from_documents(
            documents=all_texts,
            embedding=embeddings,
            persist_directory=persist_directory,
        )
        # Ensure the collection is written to disk so it can be re‑loaded later.
        docsearch.persist()
    else:
        # Load an existing collection from disk.
        docsearch = Chroma(
            persist_directory=persist_directory,
            embedding_function=embeddings,
        )

    # ---------------------------------------------------------------------
    # Retrieve the most relevant chunks.
    # ---------------------------------------------------------------------
    retriever = docsearch.as_retriever(search_kwargs={"k": 5})
    docs = retriever.get_relevant_documents(user_input)
    print(f"Number of documents retrieved: {len(docs)}")
    return docs


if __name__ == "__main__":
    # Example usage – load all *.txt files under a "docs" folder.
    document_paths = glob("docs/*.txt")
    query = "What is machine learning?"
    docs = run_agent(query, document_paths=document_paths)

    print("\n=== User Query Result ===")
    print(f"Query: {query}")
    print(f"Number of relevant documents: {len(docs)}")
    for i, doc in enumerate(docs, start=1):
        print(f"\n--- Document {i} ---")
        print(doc.page_content)
