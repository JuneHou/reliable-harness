import os
from dotenv import load_dotenv, find_dotenv
from glob import glob

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader

# Load environment variables (including OPENAI_API_KEY)
_ = load_dotenv(find_dotenv())


def run_agent(user_input: str, document_paths=None, persist_directory: str = "chroma_db"):
    """Create (or load) a Chroma vector store and retrieve the most relevant documents.

    Args:
        user_input: The query string.
        document_paths: Optional list of paths to text documents to index. If ``None`` the function
            will attempt to load an existing persisted DB from ``persist_directory``.
        persist_directory: Directory where the Chroma DB is persisted.

    Returns:
        A list of ``Document`` objects that are most relevant to ``user_input``.
    """

    # ---------------------------------------------------------------------
    # 1️⃣  Set up the OpenAI embeddings – these require an API key.
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise RuntimeError("OPENAI_API_KEY not found in environment variables")
    os.environ["OPENAI_API_KEY"] = openai_api_key
    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    # ---------------------------------------------------------------------
    # 2️⃣  Build or load the vector store.
    # ---------------------------------------------------------------------
    if document_paths:
        # -----------------------------------------------------------------
        #   a) Load and split the provided documents.
        # -----------------------------------------------------------------
        all_chunks = []
        for doc_path in document_paths:
            loader = TextLoader(doc_path)
            documents = loader.load()
            splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
            chunks = splitter.split_documents(documents)
            all_chunks.extend(chunks)

        # -----------------------------------------------------------------
        #   b) Create a new Chroma collection from the chunks and persist it.
        # -----------------------------------------------------------------
        docsearch = Chroma.from_documents(
            documents=all_chunks,
            embedding=embeddings,
            persist_directory=persist_directory,
        )
        # Explicitly write the collection to disk so that a subsequent run can load it.
        docsearch.persist()
    else:
        # -----------------------------------------------------------------
        #   Load an existing collection – raise a clear error if it does not exist.
        # -----------------------------------------------------------------
        if not os.path.isdir(persist_directory):
            raise RuntimeError(
                f"Persist directory '{persist_directory}' not found. Provide document_paths to create the index first."
            )
        docsearch = Chroma(
            persist_directory=persist_directory,
            embedding_function=embeddings,
        )

    # ---------------------------------------------------------------------
    # 3️⃣  Retrieve relevant documents.
    # ---------------------------------------------------------------------
    retriever = docsearch.as_retriever(search_kwargs={"k": 5})
    docs = retriever.get_relevant_documents(user_input)
    print(f"Number of documents retrieved: {len(docs)}")
    return docs


if __name__ == "__main__":
    # ---------------------------------------------------------------------
    # Example usage – index the *.txt files in the ./docs folder on first run.
    # ---------------------------------------------------------------------
    document_paths = glob("docs/*.txt")
    query = "What is machine learning?"

    for i, doc in enumerate(docs):
