import os
from typing import Any, List, Mapping, Optional

from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class WrapperLLM(LLM):
    """A thin wrapper around an existing OpenAI LLM.

    The wrapper logs every prompt/response pair to a file while behaving
    exactly like the wrapped LLM.  By exposing the same ``_llm_type`` as the
    underlying model and forwarding the optional ``stop`` argument we ensure
    that agents see no difference between ``WrapperLLM`` and a plain ``OpenAI``
    instance.
    """

    llm: OpenAI
    log: str

    def __init__(self, llm: OpenAI, log: str):
        super().__init__()
        self.llm = llm
        self.log = log

    # The agent uses ``_llm_type`` to decide which prompt template to use.
    # Returning the original LLM's type keeps the behaviour identical.
    @property
    def _llm_type(self) -> str:
        return self.llm._llm_type  # typically "openai"

    def _call(self, prompt: str, stop: Optional[List[str]] = None) -> str:
        # Forward the optional ``stop`` tokens to the wrapped LLM.
        reply = self.llm(prompt, stop=stop)  # type: ignore[arg-type]
        # Log the interaction for debugging purposes.
        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100)
        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        # Return a plain‑serialisable description of the wrapped LLM.
        # ``self.llm`` itself is not JSON‑serialisable, so we expose the
        # parameters that uniquely identify it.
        return self.llm._identifying_params

    # Optional – useful for caching / serialization.
    def dict(self, **kwargs: Any) -> Mapping[str, Any]:  # pragma: no cover
        return {"_type": "wrapper", "llm": self.llm.dict(**kwargs), "log": self.log}


def setup_components():
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    wllm = WrapperLLM(llm=llm, log="output.log")

    tools = [PythonAstREPLTool()]
    memory = ConversationBufferMemory(memory_key="chat_history")
    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt: str):
    try:
        wllm, tools, memory = setup_components()
    except Exception as e:
        return None

    try:
        agent = create_agent(wllm, tools, memory)
    except Exception as e:
        return None

    try:
        response = agent(prompt)
        return response
    except Exception as e:
        return None


if __name__ == "__main__":
    prompt = "What is 5-7?"
    response = run_agent(prompt)
    print(response)
    if isinstance(response, dict):
        print(type(response), response.get("output"))
