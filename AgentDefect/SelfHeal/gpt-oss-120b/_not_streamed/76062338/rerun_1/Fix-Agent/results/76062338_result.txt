import os
from typing import Any, List, Mapping, Optional
from dotenv import load_dotenv
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class Wrapper(OpenAI):
    """A thin wrapper around the OpenAI LLM that logs prompt/response pairs.
    Inheriting from ``OpenAI`` (instead of the generic ``LLM`` base class) ensures
    that all internal attributes and methods expected by LangChain agents are
    preserved, preventing the extra ``Observation`` token that appears when the
    generic ``LLM`` subclass is used.
    """

    def __init__(self, *, log: str, **openai_kwargs: Any):
        """Create a wrapper.

        Args:
            log: Path to a file where prompts and LLM replies will be appended.
            **openai_kwargs: All additional keyword arguments are forwarded to the
                underlying ``OpenAI`` constructor (e.g., ``openai_api_key``,
                ``temperature``, ``model_name``, etc.).
        """
        super().__init__(**openai_kwargs)
        self.log = log

    def _call(self, prompt: str, stop: Optional[List[str]] = None) -> str:
        """Call the underlying OpenAI model, log the interaction, and return the text.

        The ``stop`` argument is passed through to the parent implementation so
        that any stop‑sequence handling remains identical to the native LLM.
        """
        # Use the parent ``_call`` to get the raw response.
        reply = super()._call(prompt, stop=stop)
        # Log the interaction.
        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100 + "\n")
        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        """Return identifying parameters for hashing/caching.

        We include the parent's identifying params plus the log file path so that
        two wrappers with different logs are treated as distinct.
        """
        base_params = super()._identifying_params
        return {**base_params, "log": self.log}

    @property
    def _llm_type(self) -> str:
        # Keep the original type to avoid altering agent parsing logic.
        return super()._llm_type


def setup_components():
    """Instantiate the OpenAI LLM, wrap it, and create tools and memory.
    """
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    # Wrap the LLM – all behaviour of the original LLM is retained.
    wllm = Wrapper(log="output.log", openai_api_key=os.getenv("OPENAI_API_KEY"), temperature=0)

    tools = [PythonAstREPLTool()]
    memory = ConversationBufferMemory(memory_key="chat_history")
    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    """Create a conversational‑react agent using the wrapped LLM.
    """
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt: str):
    """Run the agent on a single prompt, handling any unexpected errors gracefully.
    """
    try:
        wllm, tools, memory = setup_components()
        agent = create_agent(wllm, tools, memory)
        response = agent(prompt)
        return response
    except Exception as e:
        # In a real application you'd want proper logging.
        return None


if __name__ == "__main__":
    prompt = "What is 5-7?"
    response = run_agent(prompt)
    if response is not None:
        print(response)
        # LangChain agents return a dict with an "output" key.
        print(type(response), response.get("output"))
    else:
        print("Agent failed to produce a response.")
