import os
from typing import Any, List, Mapping, Optional
from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class Wrapper(LLM):
    """A thin wrapper around an existing OpenAI LLM that logs prompts and responses.

    The wrapper mirrors the original LLM's type and identifying parameters so that
    LangChain agents treat it exactly the same as the underlying LLM. This prevents
    the agent from inserting extra ``Observation`` blocks that appear when the LLM
    type is reported as ``custom``.
    """

    llm: OpenAI
    log: str

    def __init__(self, llm: OpenAI, log: str):
        super().__init__()
        self.llm = llm
        self.log = log

    # The agent inspects ``_llm_type`` when building its prompts.  By returning the
    # wrapped LLM's type ("openai") we keep the prompt format identical to the
    # original OpenAI LLM.
    @property
    def _llm_type(self) -> str:
        return self.llm._llm_type

    # Forward the call to the underlying LLM while honouring the optional ``stop``
    # tokens.  ``self.llm._call`` is used instead of ``self.llm(prompt)`` because the
    # ``stop`` argument would otherwise be ignored.
    def _call(self, prompt: str, stop: Optional[List[str]] = None) -> str:
        reply = self.llm._call(prompt, stop=stop)
        # Log the interaction for debugging / auditing purposes.
        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100)
        return reply

    # Preserve the identifying parameters of the wrapped LLM so that caching and
    # hashing behave identically.
    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        return self.llm._identifying_params

    # Some downstream components also look for ``_model_kwargs``.  Delegating to the
    # underlying LLM keeps behaviour consistent.
    @property
    def _model_kwargs(self) -> Mapping[str, Any]:
        return getattr(self.llm, "_model_kwargs", {})


def setup_components():
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    wllm = Wrapper(llm=llm, log="output.log")

    tools = [PythonAstREPLTool()]
    memory = ConversationBufferMemory(memory_key="chat_history")
    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt: str):
    try:
        wllm, tools, memory = setup_components()
    except Exception as e:
        return None

    try:
        agent = create_agent(wllm, tools, memory)
    except Exception as e:
        return None

    try:
        response = agent(prompt)
        return response
    except Exception as e:
        return None


if __name__ == "__main__":
    prompt = "What is 5-7?"
    response = run_agent(prompt)
    print(response)
    # ``response`` is a dict when using the ``conversational-react-description``
    # agent.  It contains an ``output`` key with the final answer.
    if isinstance(response, dict):
        print(type(response), response.get("output"))
    else:
        print(type(response), response)
