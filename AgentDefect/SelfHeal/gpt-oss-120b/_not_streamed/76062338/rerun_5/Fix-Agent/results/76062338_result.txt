import os
from typing import Any, List, Mapping, Optional
from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class WrapperLLM(LLM):
    """A thin wrapper around an existing LangChain LLM.

    The wrapper records prompts and responses to a log file but otherwise behaves
    exactly like the wrapped LLM.  By returning the same ``_llm_type`` as the
    underlying model ("openai"), agents that inspect the type continue to treat the
    wrapper identically to the original LLM, preventing unexpected extra
    ``Observation`` entries in the output.
    """

    def __init__(self, llm: OpenAI, log: str):
        super().__init__()
        self.llm = llm
        self.log = log

    @property
    def _llm_type(self) -> str:
        # Return the same type as the wrapped model so agents do not apply special‑case logic.
        return "openai"

    def _call(self, prompt: str, stop: Optional[List[str]] = None) -> str:
        # Call the underlying LLM.
        reply = self.llm(prompt)
        # Log the interaction.
        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100)
        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        # Return the identifying parameters of the wrapped LLM – this mirrors the
        # behaviour of the original model and satisfies LangChain's serialization
        # expectations.
        return self.llm._identifying_params


def setup_components():
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    wllm = WrapperLLM(llm=llm, log="output.log")

    tools = [PythonAstREPLTool()]
    memory = ConversationBufferMemory(memory_key="chat_history")
    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt: str):
    try:
        wllm, tools, memory = setup_components()
    except Exception as e:
        return None

    try:
        agent = create_agent(wllm, tools, memory)
    except Exception as e:
        return None

    try:
        response = agent(prompt)
        return response
    except Exception as e:
        return None


if __name__ == "__main__":
    prompt = "What is 5-7?"
    response = run_agent(prompt)
    if response:
        print(response)
        print(type(response), response["output"])
    else:
        print("Agent failed to produce a response.")
