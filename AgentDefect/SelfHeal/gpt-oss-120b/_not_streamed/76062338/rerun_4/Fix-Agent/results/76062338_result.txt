import os
from typing import Any, List, Mapping, Optional
from dotenv import load_dotenv

from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class WrapperLLM(LLM):
    """A thin wrapper around an existing LangChain LLM.

    The wrapper logs the prompt/response pair to a file but otherwise
    behaves exactly like the wrapped LLM.  Importantly it forwards the
    ``stop`` argument to the underlying model – this is what the
    ``conversational-react-description`` agent relies on to prevent the
    model from emitting stray ``Observation`` text.
    """

    llm: OpenAI
    log: str

    def __init__(self, llm: OpenAI, log: str):
        super().__init__()
        self.llm = llm
        self.log = log

    @property
    def _llm_type(self) -> str:
        # Keep the original type so the agent does not treat this as a
        # custom model that might require different parsing.
        return self.llm._llm_type

    def _call(self, prompt: str, stop: Optional[List[str]] = None) -> str:
        """Delegate the call to the wrapped LLM, forwarding ``stop``.

        The original buggy implementation ignored the ``stop`` argument,
        causing the agent to receive an extra ``Observation`` token in the
        model's output.  By passing ``stop`` through we preserve the exact
        behaviour of the base LLM.
        """
        # Forward the stop tokens so the LLM respects the agent's output
        # parsing expectations.
        reply = self.llm(prompt, stop=stop) if stop is not None else self.llm(prompt)

        # Log the interaction safely.
        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100)
        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        # Return the identifying parameters of the wrapped LLM so caching
        # and hashing work correctly.
        return self.llm._identifying_params


def setup_components():
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    wllm = WrapperLLM(llm=llm, log="output.log")

    tools = [PythonAstREPLTool()]
    memory = ConversationBufferMemory(memory_key="chat_history")
    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt: str):
    try:
        wllm, tools, memory = setup_components()
    except Exception as e:
        print(f"Setup error: {e}")
        return None

    try:
        agent = create_agent(wllm, tools, memory)
    except Exception as e:
        print(f"Agent creation error: {e}")
        return None

    try:
        # ``agent`` is callable and returns a dict with an ``output`` field.
        response = agent(prompt)
        return response
    except Exception as e:
        print(f"Run error: {e}")
        return None


if __name__ == "__main__":
    prompt = "What is 5-7?"
    response = run_agent(prompt)
    if response:
        print(response)
        print(type(response), response["output"])
    else:
        print("No response received.")
