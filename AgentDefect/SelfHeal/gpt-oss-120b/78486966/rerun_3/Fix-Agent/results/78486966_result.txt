from pathlib import Path
from langchain_text_splitters import CharacterTextSplitter


def split_and_save(text: str, chunk_size: int = 30, chunk_overlap: int = 5, out_dir: str = "chunks") -> list[str]:
    """Split *text* into fixed‑size overlapping chunks and write each chunk to a file.

    Args:
        text: The raw text to be chunked.
        chunk_size: Desired length of each chunk (in characters).
        chunk_overlap: Number of characters that each chunk should overlap with the previous one.
        out_dir: Directory where the chunk files will be saved.

    Returns:
        A list containing the string representation of each chunk.
    """
    # Remove leading/trailing whitespace that can interfere with expected chunk boundaries.
    clean_text = text.strip()

    # Initialise the splitter – we use an empty string as the separator so the splitter works purely
    # on character count (the default for ``CharacterTextSplitter`` is already "").
    splitter = CharacterTextSplitter(chunk_size=chunk_size, chunk_overlap=chunk_overlap)
    chunks = splitter.split_text(clean_text)

    # Ensure the output directory exists.
    Path(out_dir).mkdir(parents=True, exist_ok=True)

    # Write each chunk to its own file.
    for i, chunk in enumerate(chunks):
        file_path = Path(out_dir) / f"chunk_{i}.txt"
        file_path.write_text(chunk, encoding="utf-8")

    return chunks
