from langchain_text_splitters import CharacterTextSplitter
import os

def split_and_save(text: str, output_dir: str = "chunks"):
    """Split *text* into fixed‑size overlapping chunks and write each chunk to a file.

    Args:
        text: The raw multi‑line string to split.
        output_dir: Directory where chunk files will be created. It is created if it does not exist.

    Returns:
        A list of chunk strings.
    """
    # Remove leading/trailing whitespace introduced by the triple‑quoted literal and collapse
    # any internal newline/extra spaces into a single space so the character count matches the
    # logical content of the string.
    cleaned = " ".join(text.strip().split())

    splitter = CharacterTextSplitter(chunk_size=30, chunk_overlap=5)
    chunks = splitter.split_text(cleaned)

    # Ensure the output directory exists.
    os.makedirs(output_dir, exist_ok=True)

    # Write each chunk to its own file.
    for idx, chunk in enumerate(chunks, start=1):
        file_path = os.path.join(output_dir, f"chunk_{idx}.txt")
        with open(file_path, "w", encoding="utf-8") as fp:
            fp.write(chunk)

    return chunks

from langchain_text_splitters import CharacterTextSplitter


def run_agent(text_raw: str):
    """Split *text_raw* into fixed-size overlapping character chunks.

    The default ``CharacterTextSplitter`` uses a newline ("\n") as the
    separator, which causes the splitter to break the text at each line
    boundary before applying the ``chunk_size`` and ``chunk_overlap``
    logic.  This results in chunks that are shorter than the requested size
    and breaks the intended overlapping behaviour.

    By setting ``separator=""`` we tell the splitter to treat the entire
    string as a continuous sequence of characters, so the chunk size and
    overlap are applied exactly as specified.
    In addition we normalise whitespace – stripping leading/trailing
    whitespace and collapsing internal runs of whitespace to a single
    space – to avoid unexpected newline or indentation characters affecting
    the chunk boundaries.
    """
    # Normalise whitespace: remove leading/trailing spaces and collapse any
    # sequence of whitespace characters (including newlines) into a single space.
    cleaned_text = " ".join(text_raw.split())

    text_splitter = CharacterTextSplitter(
        separator="",  # character‑level splitting, no newline separator
        chunk_size=30,
        chunk_overlap=5,
        length_function=len,
    )
    return text_splitter.split_text(cleaned_text)
