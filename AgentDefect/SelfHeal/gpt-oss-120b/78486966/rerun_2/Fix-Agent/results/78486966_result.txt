    """Return a version of *text* without leading indentation and with newlines
    replaced by single spaces.
    
    The raw string used in the example contains a leading newline and the
    indentation introduced by the triple‑quoted literal. Those characters become
    part of the split output, which makes the resulting chunks longer than the
    expected 30 characters and introduces unwanted whitespace.  By dedenting the
    string and collapsing line‑breaks we obtain a clean, continuous block of
    characters that the ``CharacterTextSplitter`` can chunk correctly.
    """
    # Remove common leading indentation (e.g. the 4 spaces that appear after the
    # opening triple‑quote in the example).
    dedented = textwrap.dedent(text)
    # Strip leading/trailing whitespace and replace any newline (or carriage
    # return) with a single space.  ``splitlines`` drops the line‑break characters
    # while preserving the content of each line.
    collapsed = " ".join(line.strip() for line in dedented.splitlines() if line.strip())
    return collapsed

from langchain_text_splitters import CharacterTextSplitter
import os

def run_agent(text_raw: str, output_dir: str = "chunks"):
    """Split the input text into overlapping character chunks and save each chunk to a file.

    Args:
        text_raw: The raw input string.
        output_dir: Directory where chunk files will be stored.
    """
    # Remove leading/trailing whitespace that can interfere with chunk boundaries
    text_raw = text_raw.strip()

    # Use an empty string as the separator so the splitter works on raw characters.
    text_splitter = CharacterTextSplitter(
        chunk_size=30,
        chunk_overlap=5,
        separator="",
    )
    chunks = text_splitter.split_text(text_raw)

    # Ensure the output directory exists
    os.makedirs(output_dir, exist_ok=True)

    # Save each chunk to a separate file named chunk_0.txt, chunk_1.txt, ...
    for idx, chunk in enumerate(chunks):
        filename = os.path.join(output_dir, f"chunk_{idx}.txt")
        with open(filename, "w", encoding="utf-8") as f:
            f.write(chunk)

    return chunks
