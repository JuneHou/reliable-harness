from langchain.text_splitter import CharacterTextSplitter

def split_into_chunks(text_raw: str, chunk_size: int = 30, chunk_overlap: int = 5) -> list[str]:
    """Split *text_raw* into fixed‑size overlapping character chunks.

    Args:
        text_raw: The input string to be split.
        chunk_size: Desired length of each chunk (default 30).
        chunk_overlap: Number of characters each chunk should overlap with the previous
            chunk (default 5).

    Returns:
        A list of string chunks that respect the requested size and overlap.
    """
    # Strip leading/trailing whitespace to avoid an initial empty chunk.
    cleaned = text_raw.strip()
    # Use an empty separator so the splitter works at the pure character level.
    splitter = CharacterTextSplitter(
        separator="",
        chunk_size=chunk_size,
        chunk_overlap=chunk_overlap,
    )
    return splitter.split_text(cleaned)

def run_agent(text_raw, output_dir: str = "chunks") -> list[str]:
    """Split *text_raw* into fixed‑size overlapping chunks and save each chunk to a file.

    Parameters
    ----------
    text_raw: str
        The raw multi‑line string to be split.
    output_dir: str, optional
        Directory where the chunk files will be stored. It is created if it does not exist.

    Returns
    -------
    list[str]
        A list containing the text of each chunk.
    """
    # Remove common leading indentation and stray new‑lines so the splitter works on the
    # intended content rather than on the literal triple‑quoted string formatting.
    cleaned_text = textwrap.dedent(text_raw).strip()

    splitter = CharacterTextSplitter(chunk_size=30, chunk_overlap=5)
    chunks = splitter.split_text(cleaned_text)

    # Write each chunk to its own file for later inspection or downstream processing.
    os.makedirs(output_dir, exist_ok=True)
    for idx, chunk in enumerate(chunks, start=1):
        file_path = os.path.join(output_dir, f"chunk_{idx}.txt")
        with open(file_path, "w", encoding="utf-8") as fp:
            fp.write(chunk)

    return chunks
