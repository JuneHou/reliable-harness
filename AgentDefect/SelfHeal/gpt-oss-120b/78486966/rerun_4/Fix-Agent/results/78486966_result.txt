from pathlib import Path
import textwrap
from langchain_text_splitters import CharacterTextSplitter


def split_and_save(text_raw: str, chunk_size: int = 30, chunk_overlap: int = 5, out_dir: str = "chunks"):
    """Split *text_raw* into overlapping character chunks and write each chunk to a file.

    Args:
        text_raw: The original text.
        chunk_size: Desired length of each chunk (including the overlap part).
        chunk_overlap: Number of characters that each chunk should share with the previous one.
        out_dir: Directory where the chunk files will be written.
    """
    # Normalise the string – remove leading/trailing whitespace and common indentation.
    # This prevents an empty first chunk caused by the leading newline in the original literal.
    cleaned_text = textwrap.dedent(text_raw).strip()

    # Initialise the splitter.
    splitter = CharacterTextSplitter(chunk_size=chunk_size, chunk_overlap=chunk_overlap)
    chunks = splitter.split_text(cleaned_text)

    # Ensure the output directory exists.
    out_path = Path(out_dir)
    out_path.mkdir(parents=True, exist_ok=True)

    # Write each chunk to its own file (e.g., chunk_0.txt, chunk_1.txt, ...).
    for i, chunk in enumerate(chunks):
        file_path = out_path / f"chunk_{i}.txt"
        file_path.write_text(chunk, encoding="utf-8")

    return chunks
