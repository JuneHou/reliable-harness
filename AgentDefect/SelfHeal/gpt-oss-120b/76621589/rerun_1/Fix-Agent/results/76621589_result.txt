import os
import asyncio
from time import time

import openai
from dotenv import load_dotenv, find_dotenv
from langchain.chains import LLMChain
from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate

# Load environment variables once at import time
_ = load_dotenv(find_dotenv())
openai.api_key = os.getenv("OPENAI_API_KEY")

# ---------------------------------------------------------------------------
# Async‑compatible helper that runs both the synchronous `apply` and the
# asynchronous `aapply` methods of a LangChain LLMChain.
# ---------------------------------------------------------------------------
async def run_agent(texts):
    """Run the LLMChain on a list of input dictionaries.

    The function demonstrates the difference between the synchronous
    ``apply`` method and the asynchronous ``aapply`` method.  ``aapply``
    returns a coroutine, so we must ``await`` it inside an ``async``
    function (or run it with ``asyncio.run``).

    Args:
        texts: A list of dictionaries matching the prompt variables.  E.g.:
            [{"text": "Hallo, ich bin 25 Jahre alt."}, ...]

    Returns:
        The result of the synchronous ``apply`` call (kept for backward
        compatibility).  The asynchronous result is printed for
        demonstration purposes.
    """
    # ---------------------------------------------------------------------
    # Set up the LLM and the prompt template.
    # ---------------------------------------------------------------------
    llm = ChatOpenAI(temperature=0)

    prompt = ChatPromptTemplate.from_template(
        "Classify the text based on the Common European Framework of Reference "
        "for Languages (CEFR). Give a single value: {text}"
    )
    chain = LLMChain(llm=llm, prompt=prompt)

    # ---------------------------------------------------------------------
    # Synchronous call – this blocks until all LLM calls are finished.
    # ---------------------------------------------------------------------
    start = time()
    res_a = chain.apply(texts)
    print(res_a)
    print(f"apply time taken: {time() - start:.2f} seconds\n")

    # ---------------------------------------------------------------------
    # Asynchronous call – we *await* the coroutine returned by ``aapply``.
    # ---------------------------------------------------------------------
    start = time()
    res_aa = await chain.aapply(texts)
    print(res_aa)
    print(f"aapply time taken: {time() - start:.2f} seconds\n")

    # Return the synchronous result so existing code that expects a return
    # value continues to work.
    return res_a


if __name__ == "__main__":
    texts = [
        {"text": "Hallo, ich bin 25 Jahre alt."},
        {"text": "Wie geht es dir?"},
        {"text": "In meiner Freizeit, spiele ich gerne Fussball."},
    ]

    # ``run_agent`` is async, so we invoke it via ``asyncio.run``.
    result = asyncio.run(run_agent(texts))
    print("\nReturned result:", result)
