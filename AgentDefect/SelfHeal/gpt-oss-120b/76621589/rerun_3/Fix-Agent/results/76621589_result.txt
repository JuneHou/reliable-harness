import time
from dotenv import load_dotenv, find_dotenv

from langchain.chains import LLMChain
from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate

# Load environment variables and set the OpenAI API key
_ = load_dotenv(find_dotenv())
openai.api_key = os.getenv("OPENAI_API_KEY")

# Initialise the LLM and prompt template once – they can be reused for both sync and async calls
llm = ChatOpenAI(temperature=0)
prompt = ChatPromptTemplate.from_template(
    "Classify the text based on the Common European Framework of Reference for Languages (CEFR). "
    "Give a single value: {text}"
)
chain = LLMChain(llm=llm, prompt=prompt)


def run_sync(texts):
    """Run the synchronous `apply` method and print timing information."""
    start = time.time()
    res = chain.apply(texts)
    print(res)
    print(f"apply time taken: {time.time() - start:.2f} seconds\n")
    return res


async def run_async(texts):
    """Run the asynchronous `aapply` method, awaiting the coroutine correctly."""
    start = time.time()
    res = await chain.aapply(texts)
    print(res)
    print(f"aapply time taken: {time.time() - start:.2f} seconds\n")
    return res


def main():
    texts = [
        {"text": "Hallo, ich bin 25 Jahre alt."},
        {"text": "Wie geht es dir?"},
        {"text": "In meiner Freizeit, spiele ich gerne Fussball."},
    ]

    # Synchronous call
    sync_result = run_sync(texts)

    # Asynchronous call – we need an event loop to await the coroutine
    async_result = asyncio.run(run_async(texts))

    print("\nReturned sync result:", sync_result)
    print("Returned async result:", async_result)


if __name__ == "__main__":
    main()
