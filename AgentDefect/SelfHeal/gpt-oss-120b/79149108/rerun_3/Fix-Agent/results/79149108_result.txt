from flask import Flask, render_template, request, jsonify
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from langchain_ollama import ChatOllama
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
# Llama 3.2 context window is 128k tokens – we keep a safe margin.
MODEL_MAX_TOKENS = 120_000

# Initialise the Ollama chat model. Temperature a bit lower for more coherent replies.
llm = ChatOllama(model=MODEL, temperature=0.8, stream=False)

# Keep a conversation history made of LangChain Message objects.
# The first entry can be a system prompt if you want the model to follow certain instructions.
messages = []

def get_number_of_tokens(msgs):
    """Return an approximate token count for a list of LangChain Message objects.
    We use the tiktoken encoder for a comparable OpenAI model; the count is only
    used to enforce a rough token budget.
    """
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total = 0
        for m in msgs:
            total += len(encoding.encode(m.content))
        return total
    except Exception:
        # Fallback – 1 token ≈ 4 characters.
        return sum(len(m.content) for m in msgs) // 4

@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_input = request.form['text']

    # Append the new user message to the conversation history.
    messages.append(HumanMessage(content=user_input))

    # Trim old messages if we exceed the token limit.
    while get_number_of_tokens(messages) > MODEL_MAX_TOKENS:
        # Preserve the first system message (if any) and drop the oldest user/assistant pair.
        # Remove the second element because the first is usually a system prompt.
        if len(messages) > 1:
            del messages[1]
        else:
            break

    # Invoke the model with the full list of Message objects.
    response = llm.invoke(messages)

    # `response` is an AIMessage when using ChatOllama.
    if isinstance(response, AIMessage):
        ai_msg = response
    else:
        # In case the library returns a plain dict-like object.
        ai_msg = AIMessage(content=response.content if hasattr(response, "content") else str(response))

    # Store the assistant reply so the next turn has context.
    messages.append(ai_msg)

    # Return just the content – the front‑end can decide how to display it.
    return jsonify({"reply": ai_msg.content})

@app.route('/')
def index():
    return render_template('index.html')

if __name__ == '__main__':
    # Turn off Flask's reloader when running inside a container to avoid duplicate LLM init.
    app.run(debug=True, port=5000, use_reloader=False)
