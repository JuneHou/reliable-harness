from flask import Flask, render_template, request
from langchain_ollama import ChatOllama
from langchain_core.messages import SystemMessage, HumanMessage, AIMessage
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
# The maximum context size of llama3.2; adjust if needed
MODEL_MAX_TOKENS = 128_000

# Initialise the chat model
llm = ChatOllama(
    model=MODEL,
    temperature=1.1,
    stream=False,  # non‑streaming for simplicity in Flask route
)

# Conversation history – start with a system prompt that tells the model how to behave.
# This message is kept for the whole lifecycle of the app.
messages = [
    SystemMessage(content="You are a helpful assistant. Answer the user's questions concisely.")
]


def get_number_of_tokens(msgs: list) -> int:
    """Estimate token count for a list of LangChain Message objects.
    Falls back to a simple character‑based heuristic if the model is unknown to tiktoken.
    """
    try:
        # Use a generic OpenAI tokenizer as a proxy – works for most LLMs.
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total = 0
        for m in msgs:
            total += len(encoding.encode(m.content))
        return total
    except Exception:
        # Rough estimate: 1 token ≈ 4 characters
        total_chars = sum(len(m.content) for m in msgs)
        return total_chars // 4


@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    # Append the new user message
    user_msg = HumanMessage(content=request.form["text"])
    messages.append(user_msg)

    # Trim old messages if we exceed the model's context window.
    # Preserve the first system message (index 0).
    while get_number_of_tokens(messages) > MODEL_MAX_TOKENS:
        # Remove the oldest *pair* of messages (user + AI) after the system prompt.
        # This keeps the dialogue coherent.
        if len(messages) > 2:
            del messages[1]  # remove the earliest user message
            if len(messages) > 2 and isinstance(messages[1], AIMessage):
                del messages[1]  # also remove its corresponding AI reply
        else:
            break

    # Get the model's response
    ai_response = llm.invoke(messages)
    # The invoke call returns an AIMessage (or a LangChain Message object).
    # Ensure we store it for future turns.
    if isinstance(ai_response, str):
        # In some configurations ChatOllama may return raw text.
        ai_msg = AIMessage(content=ai_response)
    else:
        ai_msg = ai_response
    messages.append(ai_msg)

    return ai_msg.content


@app.route('/')
def index():
    return render_template('index.html')

if __name__ == "__main__":
    app.run(debug=True, port=5000)
