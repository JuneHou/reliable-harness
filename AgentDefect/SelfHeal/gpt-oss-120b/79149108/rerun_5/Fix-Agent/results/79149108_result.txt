from flask import Flask, render_template, request, jsonify
from langchain_ollama import ChatOllama
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
import tiktoken

app = Flask(__name__)

MODEL = "llama3.2"
MODEL_MAX_TOKENS = 128000

# Optional system prompt to guide the assistant
messages = [
    SystemMessage(content="You are a helpful assistant. Answer the user's questions concisely.")
]

llm = ChatOllama(
    model=MODEL,
    temperature=0.7,
    stream=False
)

def num_tokens_from_messages(messages, model="gpt-3.5-turbo"):
    """Estimate token count for a list of LangChain message objects.
    The same encoding is used for most OpenAI‑compatible models; if the model
    name is not recognised we fall back to a generic encoding.
    """
    try:
        encoding = tiktoken.encoding_for_model(model)
    except Exception:
        encoding = tiktoken.get_encoding("cl100k_base")
    total = 0
    for msg in messages:
        total += len(encoding.encode(msg.content))
    return total

@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_text = request.form['text']
    # Append the new user message
    messages.append(HumanMessage(content=user_text))

    # Trim the conversation history if token limit is exceeded
    while num_tokens_from_messages(messages) > MODEL_MAX_TOKENS:
        # Preserve the system message at index 0; remove the oldest non‑system message
        if len(messages) > 1:
            del messages[1]
        else:
            break

    # Get the assistant's response
    ai_msg = llm.invoke(messages)
    # Store the assistant's reply for the next round
    messages.append(AIMessage(content=ai_msg.content))

    return jsonify({"response": ai_msg.content})

@app.route('/')
def index():
    return render_template('index.html')

if __name__ == '__main__':
    app.run(debug=True, port=5000)
