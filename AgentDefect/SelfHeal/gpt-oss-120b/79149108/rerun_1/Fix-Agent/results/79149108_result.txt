from flask import Flask, render_template, request
from langchain_ollama import ChatOllama
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
import tiktoken

app = Flask(__name__)

# Model configuration
MODEL = "llama3.2"
# Llama 3.2 supports up to ~128k tokens – keep a safety margin
MODEL_MAX_TOKENS = 120_000

# Keep the conversation history as LangChain message objects
# The first message can be a system prompt that you want the model to follow.
messages: list = [
    SystemMessage(
        content="You are a helpful assistant. Answer the user's questions concisely and stay on topic."
    )
]

# Initialise the Ollama LLM – streaming is disabled for the simple request/response cycle
llm = ChatOllama(
    model=MODEL,
    temperature=0.7,
    stream=False,
)


def get_number_of_tokens(msgs: list) -> int:
    """Estimate token count for a list of LangChain message objects.
    We reuse the OpenAI tiktoken encoder because the tokenisation is compatible for
    most English text. If anything fails we fall back to a rough character‑based
    estimate.
    """
    try:
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total = 0
        for m in msgs:
            total += len(encoding.encode(m.content))
        return total
    except Exception:
        # Approximate: 1 token ≈ 4 characters
        total_chars = sum(len(m.content) for m in msgs)
        return total_chars // 4


@app.route("/ask_gpt", methods=["POST"])
def ask_gpt():
    # Append the new user message to the history
    user_text = request.form["text"]
    messages.append(HumanMessage(content=user_text))

    # Ensure we stay under the model token limit – drop the oldest *user/AI*
    # messages (but keep the system prompt at index 0).
    while get_number_of_tokens(messages) >= MODEL_MAX_TOKENS:
        # Remove the second element (first after the system prompt).
        # This keeps the conversation rolling without blowing the context window.
        if len(messages) > 1:
            del messages[1]
        else:
            break

    # Call the LLM with the current conversation history
    response = llm.invoke(messages)

    # ``response`` is an ``AIMessage`` – extract its textual content.
    ai_msg: AIMessage = response
    return ai_msg.content


@app.route('/')
def index():
    return render_template('index.html')


if __name__ == "__main__":
    # Debug mode is convenient during development; turn it off in production.
    app.run(debug=True, port=5000)
