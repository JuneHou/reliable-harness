from flask import Flask, render_template, request
from langchain_ollama import ChatOllama
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
import tiktoken

app = Flask(__name__)

# ------------------------------------------------------------
# Model configuration
# ------------------------------------------------------------
MODEL = "llama3.2"
# The max context length of the model (in tokens). 128k for llama3.2.
MODEL_MAX_TOKENS = 128_000

# ------------------------------------------------------------
# Conversation state (list of LangChain Message objects)
# ------------------------------------------------------------
# We start the conversation with a system prompt that explains the
# assistant's role.  This is kept at index 0 so we never delete it
# when we trim the history to stay under the token limit.
messages = [
    SystemMessage(
        content="You are a helpful AI assistant. Answer the user's questions concisely and stay on topic."
    )
]

# ------------------------------------------------------------
# LLM client – we don't need streaming for a simple Flask endpoint.
# ------------------------------------------------------------
llm = ChatOllama(
    model=MODEL,
    temperature=1.1,
    # ``stream=False`` returns a single ``AIMessage`` object that
    # contains the full response, which is easier for HTTP APIs.
    stream=False,
)

# ------------------------------------------------------------
# Helper: approximate token count for the current message list.
# ------------------------------------------------------------
def get_number_of_tokens(messages):
    """Return an estimate of the number of tokens used by *messages*.

    The function tries to use the tiktoken library with a model that has a
    compatible tokenizer.  If that fails we fall back to a simple character‑
    based heuristic (roughly 4 characters per token).
    """
    try:
        # ``gpt-3.5-turbo`` is a good surrogate for many open‑source models.
        encoding = tiktoken.encoding_for_model("gpt-3.5-turbo")
        total = 0
        for msg in messages:
            # ``msg`` is a LangChain Message object – we only need the content.
            total += len(encoding.encode(str(msg.content)))
        return total
    except Exception:
        # Very rough fallback: 1 token ≈ 4 characters.
        total_chars = sum(len(str(msg.content)) for msg in messages)
        return total_chars // 4

# ------------------------------------------------------------
# Flask route – receive a user message, update the conversation,
# trim if necessary, and return the model's reply.
# ------------------------------------------------------------
@app.route('/ask_gpt', methods=['POST'])
def ask_gpt():
    user_text = request.form.get('text', '')
    if not user_text:
        return "", 400

    # Add the new user turn.
    messages.append(HumanMessage(content=user_text))

    # Trim oldest *non‑system* messages while staying under the token limit.
    # We keep the system message at index 0 permanently.
    while get_number_of_tokens(messages) > MODEL_MAX_TOKENS:
        # Remove the second element (first user/assistant pair after system).
        # This ensures the system prompt is never discarded.
        del messages[1]

    # Generate the assistant's response.
    ai_msg = llm.invoke(messages)  # Returns an AIMessage when stream=False

    # Store the assistant's reply so that the next turn has full context.
    messages.append(ai_msg)

    # Return only the text content to the front‑end.
    return ai_msg.content

# ------------------------------------------------------------
# Simple index page.
# ------------------------------------------------------------
@app.route('/')
def index():
    return render_template('index.html')

# ------------------------------------------------------------
# Run the app.
# ------------------------------------------------------------
if __name__ == '__main__':
    # ``debug=True`` is convenient during development but should be
    # disabled in production.
    app.run(debug=True, port=5000)
