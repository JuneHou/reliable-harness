from tqdm import tqdm
from langchain.docstore.document import Document
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from dotenv import load_dotenv

load_dotenv()

def run_agent():
    """Create several Chroma collections, persist them, and then retrieve the
    stored embeddings for a test document.

    The original code returned ``db._collection.get()`` which yields ``None``
    because ``get`` requires explicit query parameters (e.g., ``ids`` or
    ``include``).  The fixed version returns the raw entry (including the
    embedding vector) for the document we just inserted.
    """

    embeddings = OpenAIEmbeddings()

    # Create and persist a few example collections
    for db_collection_name in tqdm(["class1-sub2-chap3", "class2-sub3-chap4"]):
        documents = []
        doc_ids = []
        for doc_index in range(3):
            cl, sub, chap = db_collection_name.split("-")
            content = f"This is {db_collection_name}-doc{doc_index}"
            doc = Document(
                page_content=content,
                metadata={
                    "chunk_num": doc_index,
                    "chapter": chap,
                    "class": cl,
                    "subject": sub,
                },
            )
            documents.append(doc)
            doc_ids.append(str(doc_index))

        db = Chroma.from_documents(
            collection_name=db_collection_name,
            documents=documents,
            ids=doc_ids,
            embedding=embeddings,
            persist_directory="./data",
        )
        # Ensure the collection is written to disk
        db.persist()

    # ------------------------------------------------------------
    # Insert a single test document into its own collection
    # ------------------------------------------------------------
    initial_content = "This is an initial document content"
    document_id = "doc1"
    original_doc = Document(page_content=initial_content, metadata={"page": "0"})

    test_db = Chroma.from_documents(
        collection_name="test_collection",
        documents=[original_doc],
        ids=[document_id],
        embedding=embeddings,
        persist_directory="./data",
    )
    test_db.persist()

    # Retrieve the stored embedding (and associated metadata) for the test doc
    # ``include`` tells Chroma which fields to return.  "embeddings" will contain the
    # vector produced by ``OpenAIEmbeddings``.
    result = test_db._collection.get(
        ids=[document_id],
        include=["embeddings", "documents", "metadatas", "ids"],
    )
    return result

if __name__ == "__main__":
