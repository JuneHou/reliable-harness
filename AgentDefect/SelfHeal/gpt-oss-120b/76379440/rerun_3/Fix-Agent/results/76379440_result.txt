from tqdm import tqdm
from langchain.docstore.document import Document
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from dotenv import load_dotenv

load_dotenv()


def run_agent():
    """Create a few Chroma collections, persist them, and then retrieve the stored
    embeddings for each document.

    The original implementation attempted to call the private attribute
    ``db._collection.get()`` which returns ``None`` for the embeddings unless the
    ``include`` argument is explicitly set.  Moreover the function returned only
    the *last* ``db`` instance created in the loop, which is not useful when
    several collections are built.

    This version:
    * Uses a single ``persist_directory`` for all collections.
    * After persisting each collection, calls the public ``db.get`` method with
      ``include=["embeddings"]`` to fetch the vector embeddings.
    * Returns a dictionary mapping each collection name to its list of
      embeddings so the caller can inspect them.
    """

    embeddings = OpenAIEmbeddings()
    persist_dir = "./data"

    # Store the embeddings for every collection we create
    all_embeddings = {}

    for collection_name in tqdm(["class1-sub2-chap3", "class2-sub3-chap4"]):
        documents = []
        doc_ids = []

        # Create three sample documents for the current collection
        for doc_index in range(3):
            cl, sub, chap = collection_name.split("-")
            content = f"This is {collection_name}-doc{doc_index}"
            doc = Document(
                page_content=content,
                metadata={
                    "chunk_num": doc_index,
                    "chapter": chap,
                    "class": cl,
                    "subject": sub,
                },
            )
            documents.append(doc)
            doc_ids.append(str(doc_index))

        # Build (or load) the Chroma collection and persist it
        db = Chroma.from_documents(
            collection_name=collection_name,
            documents=documents,
            ids=doc_ids,
            embedding=embeddings,
            persist_directory=persist_dir,
        )
        db.persist()

        # Retrieve the stored embeddings for this collection
        # ``include=["embeddings"]`` tells Chroma to return the vectors.
        retrieved = db.get(ids=doc_ids, include=["embeddings"])
        all_embeddings[collection_name] = retrieved.get("embeddings", [])

    # Example of adding a single new document to a separate collection
    initial_content = "This is an initial document content"
    document_id = "doc1"
    original_doc = Document(page_content=initial_content, metadata={"page": "0"})

    new_db = Chroma.from_documents(
        collection_name="test_collection",
        documents=[original_doc],
        ids=[document_id],
        embedding=embeddings,
        persist_directory=persist_dir,
    )
    new_db.persist()

    # Return all embeddings for inspection
    return all_embeddings


if __name__ == "__main__":
    result = run_agent()
    for coll, embeds in result.items():
        print(f"Collection: {coll}, number of embeddings: {len(embeds)}")
        # Printing the first embedding vector as an example (truncated)
        if embeds:
            print(f"  First embedding (first 5 values): {embeds[0][:5]}")
