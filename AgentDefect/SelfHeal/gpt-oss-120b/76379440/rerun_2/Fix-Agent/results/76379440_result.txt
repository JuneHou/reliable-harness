from tqdm import tqdm
from langchain.docstore.document import Document
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from dotenv import load_dotenv

load_dotenv()


def run_agent():
    """Create a few Chroma collections, persist them and then return the
    raw records (including the stored embeddings) for one of the collections.

    The original script attempted to call the private ``_collection.get``
    method on the last in‑memory ``Chroma`` object, which returned ``None``
    because the collection had not been re‑loaded from disk.  The fixed
    version loads the persisted collection via the public ``Chroma`` API and
    calls ``vectorstore.get`` (or ``vectorstore.get(ids=None)``) which returns a
    dictionary containing ``ids``, ``embeddings``, ``documents`` and
    ``metadatas``.
    """

    embeddings = OpenAIEmbeddings()
    persist_dir = "./data"

    # -----------------------------------------------------------------
    # 1️⃣  Create and persist two collections with dummy documents.
    # -----------------------------------------------------------------
    for db_collection_name in tqdm(["class1-sub2-chap3", "class2-sub3-chap4"]):
        documents = []
        doc_ids = []
        cl, sub, chap = db_collection_name.split("-")
        for doc_index in range(3):
            content = f"This is {db_collection_name}-doc{doc_index}"
            doc = Document(
                page_content=content,
                metadata={
                    "chunk_num": doc_index,
                    "chapter": chap,
                    "class": cl,
                    "subject": sub,
                },
            )
            documents.append(doc)
            doc_ids.append(str(doc_index))

        # ``from_documents`` both creates the collection *and* writes the
        # embeddings to the underlying Chroma store.
        db = Chroma.from_documents(
            collection_name=db_collection_name,
            documents=documents,
            ids=doc_ids,
            embedding=embeddings,
            persist_directory=persist_dir,
        )
        db.persist()  # ensure data are flushed to disk

    # -----------------------------------------------------------------
    # 2️⃣  Load a persisted collection and retrieve its raw data.
    # -----------------------------------------------------------------
    collection_to_inspect = "class1-sub2-chap3"

    # When we instantiate ``Chroma`` without providing ``documents`` we are
    # only opening an existing collection.  The ``embedding_function`` argument
    # is required by the constructor but is not used for a simple ``get``.
    vectorstore = Chroma(
        collection_name=collection_to_inspect,
        embedding_function=embeddings,
        persist_directory=persist_dir,
    )

    # ``ids=None`` asks Chroma to return *all* stored records.
    result = vectorstore.get(ids=None)

    # ``result`` is a dictionary with the keys:
    #   * ids          – list of document identifiers
    #   * embeddings   – list of embedding vectors (as lists of floats)
    #   * documents    – list of original page_content strings
    #   * metadatas    – list of metadata dictionaries
    return result


if __name__ == "__main__":
    data = run_agent()
    print("--- Retrieved collection data ---")
    print("IDs:", data["ids"])
    print("Number of embeddings:", len(data["embeddings"]))
    # Show the dimension of the first embedding (if any) to confirm we have
    # actual vectors rather than ``None``.
    if data["embeddings"]:
        print("Embedding dimension:", len(data["embeddings"][0]))
    print("Documents:", data["documents"])
    print("Metadatas:", data["metadatas"])
