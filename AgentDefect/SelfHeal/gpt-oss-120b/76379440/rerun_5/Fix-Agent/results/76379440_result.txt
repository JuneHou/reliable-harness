from tqdm import tqdm
from langchain.docstore.document import Document
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings
from dotenv import load_dotenv

load_dotenv()

def run_agent():
    """Create a few collections, persist them, and then demonstrate how to
    retrieve the stored embedding vectors from a collection.

    The function returns a dictionary containing the document ids and their
    corresponding embedding vectors for the ``test_collection``.
    """
    embeddings = OpenAIEmbeddings()

    # ---------------------------------------------------------------------
    # Create and persist example collections
    # ---------------------------------------------------------------------
    for collection_name in tqdm(["class1-sub2-chap3", "class2-sub3-chap4"]):
        documents = []
        doc_ids = []
        cl, sub, chap = collection_name.split("-")
        for i in range(3):
            content = f"This is {collection_name}-doc{i}"
            doc = Document(
                page_content=content,
                metadata={
                    "chunk_num": i,
                    "chapter": chap,
                    "class": cl,
                    "subject": sub,
                },
            )
            documents.append(doc)
            doc_ids.append(str(i))

        # Create the collection and persist it to disk
        db = Chroma.from_documents(
            collection_name=collection_name,
            documents=documents,
            ids=doc_ids,
            embedding=embeddings,
            persist_directory="./data",
        )
        db.persist()

    # ---------------------------------------------------------------------
    # Create a test collection that we will query for its embeddings
    # ---------------------------------------------------------------------
    initial_content = "This is an initial document content"
    document_id = "doc1"
    test_doc = Document(page_content=initial_content, metadata={"page": "0"})

    test_db = Chroma.from_documents(
        collection_name="test_collection",
        documents=[test_doc],
        ids=[document_id],
        embedding=embeddings,
        persist_directory="./data",
    )
    test_db.persist()

    # ---------------------------------------------------------------------
    # Retrieve the stored embedding vectors.  ``include=['embeddings']`` tells
    # Chroma to return the raw vector data alongside the ids.
    # ---------------------------------------------------------------------
    result = test_db.get(ids=[document_id], include=["embeddings"])
    return result

if __name__ == "__main__":
    outcome = run_agent()
    print(outcome)
