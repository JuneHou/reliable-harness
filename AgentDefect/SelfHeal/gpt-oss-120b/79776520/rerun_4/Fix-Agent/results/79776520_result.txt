from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    """Load environment variables, create the model, tools and the agent.

    Returns:
        tuple: ``(agent_executor, memory, config)`` where ``agent_executor`` is the
        LangGraph agent, ``memory`` is the ``MemorySaver`` checkpoint, and
        ``config`` contains the ``thread_id`` that ties together a conversation
        across multiple calls.
    """
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # Simple search tool – you can add more tools if you wish
    simple_search = TavilySearch(max_result=2)
    tools = [simple_search]

    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key,
    )

    # ``MemorySaver`` automatically stores the full graph state (including the
    # ``messages`` channel) after each step, so we do not need to manage history
    # manually.
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory,
    )

    # ``thread_id`` is the identifier for a single conversation.  Using the same
    # ID across calls lets the ``MemorySaver`` retrieve the previous state.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, config, user_input):
    """Send a single user message to the agent and return its response.

    The ``MemorySaver`` attached to the agent retains the previous messages, so
    we only need to pass the newest ``HumanMessage``.  The agent's internal state
    (including the message history) is automatically persisted and restored.
    """
    # Stream the execution; we only care about the final message produced by the
    # graph for this user turn.
    for step in agent_executor.stream(
        {"messages": [HumanMessage(content=user_input)]},
        config,
        stream_mode="values",
    ):
        final_message = step["messages"][-1]
    return final_message.content


def main():
    agent_executor, memory, config = initialize_agent()
    response1 = run_agent(agent_executor, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
