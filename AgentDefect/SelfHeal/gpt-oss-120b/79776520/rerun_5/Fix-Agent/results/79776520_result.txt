import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    """Initialise the React‑style agent and a memory check‑pointer.

    Returns
    -------
    tuple[AgentExecutor, MemorySaver, dict]
        The agent executor, the memory saver (used as a checkpointer), and a
        configuration dict containing a stable ``thread_id`` so that the same
        conversation history is reused across calls.
    """
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # Simple search tool – can be expanded with more tools later.
    simple_search = TavilySearch(max_result=2)
    tools = [simple_search]

    # Gemini 2.5 Flash model.
    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key,
    )

    # MemorySaver automatically persists the entire thread state (including the
    # chat history).  It is passed as the ``checkpointer`` for the agent.
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory,
    )

    # ``thread_id`` identifies the conversation.  Using a constant value keeps
    # the same memory across multiple ``run_agent`` calls.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, config, user_input: str) -> str:
    """Send a single user message to the agent and return the assistant reply.

    The ``MemorySaver`` attached to the executor automatically retrieves the
    prior conversation, so we only need to supply the new ``HumanMessage``.
    """
    # Only the fresh user message is supplied; the checkpointer merges it with
    # the stored history.
    input_data = {"messages": [HumanMessage(content=user_input)]}

    # ``stream`` yields intermediate states; the final assistant message is the
    # last ``messages`` entry.
    final_message = None
    for step in agent_executor.stream(input_data, config, stream_mode="values"):
        # ``step`` contains the full state after each node; the latest assistant
        # message is always the last item in the ``messages`` list.
        final_message = step["messages"][-1]

    if final_message is None:
        raise RuntimeError("Agent did not produce a response.")
    return final_message.content


def main():
    agent_executor, memory, config = initialize_agent()
    response1 = run_agent(agent_executor, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
