import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    """Load environment, create model, tools and the agent.
    Returns the agent executor, the memory saver (checkpointer) and a
    configuration dictionary that contains a deterministic ``thread_id``.
    """
    load_dotenv()

    google_api_key = os.getenv("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # Initialise the search tool
    simple_search = TavilySearch(max_result=2)
    tools = [simple_search]

    # Initialise the LLM
    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key,
    )

    # Use MemorySaver as the checkpointer – it will automatically store the
    # ``messages`` channel for each thread.
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory,
    )

    # ``thread_id`` identifies the conversation.  Using a constant value makes
    # it easy to reproduce the example, but any unique identifier works.
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, config, user_input: str) -> str:
    """Send a user message to the agent and return the assistant's reply.

    The ``MemorySaver`` attached to the agent automatically persists the
    conversation history, so we only need to provide the new ``HumanMessage``.
    """
    # Stream the execution – ``stream_mode="values"`` yields only the value
    # dicts (no checkpoint objects).  The final step contains the updated
    # ``messages`` channel.
    final_message = None
    for step in agent_executor.stream(
        {"messages": [HumanMessage(user_input)]},
        config,
        stream_mode="values",
    ):
        # Each ``step`` includes the current state of all channels.  Keep the
        # most recent assistant message.
        if "messages" in step:
            final_message = step["messages"][-1]

    if final_message is None:
        raise RuntimeError("Agent did not produce any output")
    return final_message.content


def main():
    agent_executor, memory, config = initialize_agent()
    response1 = run_agent(agent_executor, config, "Hi! I am Bob!")
    response2 = run_agent(agent_executor, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
