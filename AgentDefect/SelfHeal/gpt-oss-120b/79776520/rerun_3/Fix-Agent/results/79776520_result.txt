import os
from dotenv import load_dotenv

from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver
from langchain_tavily import TavilySearch
from langchain_core.messages import HumanMessage
from langchain_google_genai import ChatGoogleGenerativeAI


def initialize_agent():
    """Load environment variables, set up the model, tools and the agent.
    Returns the agent executor, a MemorySaver checkpoint and a config that
    contains a persistent ``thread_id`` so that the same conversation history
    is used across multiple calls.
    """
    load_dotenv()

    google_api_key = os.environ.get("GOOGLE_API_KEY")
    if not google_api_key:
        raise ValueError("GOOGLE_API_KEY not found in .env file")

    # Simple search tool – can be expanded with more tools if desired
    simple_search = TavilySearch(max_result=2)
    tools = [simple_search]

    # Gemini model – you can swap this for any other model supported by LangChain
    model = ChatGoogleGenerativeAI(
        model="gemini-2.5-flash",
        google_api_key=google_api_key,
    )

    # MemorySaver keeps the full state of the graph for a given ``thread_id``
    memory = MemorySaver()
    agent_executor = create_react_agent(
        model=model,
        tools=tools,
        checkpointer=memory,
    )

    # ``thread_id`` makes the checkpoint reusable across calls
    config = {"configurable": {"thread_id": "agent003"}}
    return agent_executor, memory, config


def run_agent(agent_executor, config, user_input: str) -> str:
    """Run a single turn of the agent.

    The ``MemorySaver`` attached to the agent automatically retrieves the
    previous messages for the supplied ``thread_id`` and stores the new state
    after the call finishes.  Therefore we only need to pass the fresh user
    message – the graph will prepend the historic messages internally.
    """
    # Build the input for the graph.  ``HumanMessage`` represents the user turn.
    inputs = {"messages": [HumanMessage(user_input)]}

    # ``invoke`` returns the final state after the graph has finished
    # processing the turn.  The ``messages`` channel contains the full
    # conversation, with the assistant's latest reply at the end.
    result = agent_executor.invoke(inputs, config)
    return result["messages"][-1].content


def main():
    agent_executor, memory, config = initialize_agent()
    # First turn – the agent learns the name "Bob"
    response1 = run_agent(agent_executor, config, "Hi! I am Bob!")
    # Second turn – the agent should now remember the name
    response2 = run_agent(agent_executor, config, "What's my name?")
    print(response1)
    print(response2)


if __name__ == "__main__":
    main()
