from llama_index.core import VectorStoreIndex, Document
from llama_index.vector_stores.postgres import PGVectorStore, Column
import os
from dotenv import load_dotenv

load_dotenv()


def create_vector_store(request):
    """Create a PGVectorStore with a custom column for ``external_id``.

    The default PGVectorStore stores all metadata in a JSONB column called ``metadata``.
    To have ``external_id`` as a first‑class column we:
    * Define a ``Column`` describing the extra column (name and PostgreSQL type).
    * Pass the column via ``metadata_columns``.
    * Disable the generic JSON column by setting ``metadata_json_column`` to ``None``.
    """
    host = "localhost"
    password = "password"
    port = 5432
    user = "postgres"

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name="vectorstore",
        embed_dim=1536,
        # Use the HNSW index configuration you had before
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
        # -----------------------------------------------------------------
        # Custom metadata handling
        # -----------------------------------------------------------------
        # ``metadata_json_column`` is set to ``None`` so the default JSONB column
        # is not created.  All metadata values that you want to store as separate
        # columns must be listed in ``metadata_columns``.
        metadata_json_column=None,
        metadata_columns=[
            Column(name="external_id", type="TEXT"),  # <-- new explicit column
        ],
    )
    return vector_store


def run_agent(request):
    vector_store = create_vector_store(request)

    # Documents now include ``external_id`` in their metadata. The value will be
    # written to the dedicated ``external_id`` column defined above.
    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )
    return index
