from llama_index.core import VectorStoreIndex, Document
from llama_index.vector_stores.postgres import PGVectorStore
import os
from dotenv import load_dotenv

load_dotenv()


def create_vector_store(request):
    """Create a PGVectorStore with a custom column for ``external_id``.

    The PGVectorStore stores embeddings in a PostgreSQL table. By default all
    metadata is kept in a JSONB column called ``metadata``.  ``metadata_schema``
    allows us to add extra columns that are created automatically when the
    table is first created.  Here we create a TEXT column called ``external_id``
    and the index will write the value from ``Document.metadata["external_id"]``
    into that column.
    """
    host = os.getenv("POSTGRES_HOST", "localhost")
    password = os.getenv("POSTGRES_PASSWORD", "password")
    port = int(os.getenv("POSTGRES_PORT", "5432"))
    user = os.getenv("POSTGRES_USER", "postgres")

    # Define the extra columns that should appear in the vector table.
    # The key is the metadata field name and the value is the PostgreSQL type.
    metadata_schema = {
        "external_id": "TEXT",  # creates a TEXT column named external_id
        # You can add more custom columns here, e.g. "source": "TEXT"
    }

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name="vectorstore",
        embed_dim=1536,
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
        # Pass the schema so PGVectorStore creates the extra column.
        metadata_schema=metadata_schema,
        # Ensure the table is created if it does not already exist.
        create_table_if_not_exists=True,
    )
    return vector_store


def run_agent(request):
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )
    return index
