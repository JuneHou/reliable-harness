import os
from dotenv import load_dotenv

from llama_index.core import Document, VectorStoreIndex
from llama_index.vector_stores.postgres import PGVectorStore

load_dotenv()


def create_vector_store(request):
    """Create a PGVectorStore with a custom column for ``external_id``.

    The default PGVectorStore stores all document metadata in a single JSONB
    column called ``metadata``.  By supplying a ``metadata_schema`` mapping we
    can ask the store to materialise selected keys as dedicated columns.  In
    this example we expose the ``external_id`` field as a ``TEXT`` column.
    """
    host = "localhost"
    password = "password"
    port = 5432
    user = "postgres"

    # Define extra columns – key: PostgreSQL column type
    extra_metadata_schema = {"external_id": "TEXT"}

    vector_store = PGVectorStore.from_params(
        database=request.tenant,
        host=host,
        password=password,
        port=port,
        user=user,
        table_name="vectorstore",
        embed_dim=1536,
        # Pass the custom schema so ``external_id`` becomes its own column
        metadata_schema=extra_metadata_schema,
        hnsw_kwargs={
            "hnsw_m": 16,
            "hnsw_ef_construction": 64,
            "hnsw_ef_search": 40,
            "hnsw_dist_method": "vector_cosine_ops",
        },
    )
    return vector_store


def run_agent(request):
    """Create an index using the custom PGVectorStore.

    The ``Document`` objects contain ``external_id`` inside their metadata.
    Because the store knows about this key (via ``metadata_schema``), the
    value is written to the dedicated ``external_id`` column instead of the
    generic JSONB ``metadata`` column.
    """
    vector_store = create_vector_store(request)

    documents = [
        Document(text="Sample document 1", metadata={"external_id": "ext_001"}),
        Document(text="Sample document 2", metadata={"external_id": "ext_002"}),
    ]

    index = VectorStoreIndex.from_documents(
        documents,
        vector_store=vector_store,
    )
    return index
