import os
from dotenv import load_dotenv

from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.prompts import PromptTemplate

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def build_chain() -> RetrievalQAWithSourcesChain:
    """Create a RetrievalQAWithSourcesChain that uses a custom (blank) prompt.

    The default prompt injected by LangChain includes explanatory text that the
    user wants to suppress. By providing our own ``PromptTemplate`` that only
    contains the placeholders we need – ``{question}`` and ``{context}`` – we can
    control exactly what is sent to the LLM.
    """
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    # Sample documents
    docs = [
        "This is a test document about France.",
        "Another document about Germany."
    ]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # Create a prompt that does *not* include any of the default wording.
    # It simply forwards the retrieved context and the user question.
    custom_prompt = PromptTemplate(
        input_variables=["context", "question"],
        template="{question}\n\nContext:\n{context}"
    )

    # ``combine_documents_chain_kwargs`` allows us to inject the custom prompt
    # into the underlying ``StuffDocumentsChain`` that ``RetrievalQAWithSourcesChain``
    # builds internally.
    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        combine_documents_chain_kwargs={"prompt": custom_prompt},
    )
    return chain


if __name__ == "__main__":
    qa_chain = build_chain()
    # Example query – the result will be formatted using our custom prompt.
    answer = qa_chain.run("Which countries are mentioned in the documents?")
    print(answer)
