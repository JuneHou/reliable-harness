import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings

warnings.filterwarnings("ignore")

# Load environment variables (including the OpenAI key)
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def run_agent(queries):
    """Run a RetrievalQA (map‑reduce) chain over the CSV files.

    The function now:
    * Creates the embeddings and vector store **once** per CSV file.
    * Uses a reasonable ``k`` (number of retrieved documents) – 10 works well for
      most small‑to‑medium CSVs and keeps token usage low.
    * Instantiates the ``RetrievalQA`` chain **once** per file instead of for every
      query, avoiding unnecessary re‑initialisation.
    * Returns a list of results in the order ``[(file, query, answer), ...]``.
    """
    results = []
    files = ["drivers.csv", "drivers_full.csv"]

    # Initialise the embeddings once – the same model is used for all files
    embeddings = OpenAIEmbeddings()

    for file in files:
        print("=" * 40)
        print(file)
        print("=" * 40)
        # Track token usage for each file separately
        with get_openai_callback() as cb:
            # Load the CSV – each row becomes a LangChain Document
            loader = CSVLoader(file_path=file, encoding="utf-8")
            docs = loader.load()

            # Build a Chroma vector store from the documents
            db = Chroma.from_documents(docs, embeddings)
            # Retrieve a modest number of most‑similar documents
            retriever = db.as_retriever(search_kwargs={"k": 10})

            # Create the RetrievalQA chain (map‑reduce) once per file
            qa_chain = RetrievalQA.from_chain_type(
                llm=OpenAI(temperature=0),
                chain_type="map_reduce",
                retriever=retriever,
                verbose=False,
            )

            for query in queries:
                print(f"Query: {query}")
                answer = qa_chain.run(query)
                results.append((file, query, answer))
                print(f"Answer: {answer}\n")

            # Show token/cost information for this file
            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German",
        "How many drivers are there?",
        "What is the driver number of the German driver named 'Max'?,",
        "Who is the oldest driver?",
    ]
    run_agent(queries)
