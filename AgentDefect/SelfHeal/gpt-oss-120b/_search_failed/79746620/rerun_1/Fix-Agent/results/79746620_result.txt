import os
from dotenv import load_dotenv

from llama_index.core import Settings, VectorStoreIndex
from llama_index.core.schema import TextNode
from llama_index.core.vector_stores import (
    MetadataFilter,
    MetadataFilters,
    FilterOperator,
)
from llama_index.embeddings.openai import OpenAIEmbedding

# ---------------------------------------------------------------------------
# Helper to normalise ``None`` values in metadata.
# LlamaIndex (and most vector‑store back‑ends) cannot filter on a literal
# ``None`` because the value is either omitted or converted to ``null`` in the
# underlying JSON representation, which the filter logic does not recognise.
# We replace ``None`` with a sentinel string ("__NULL__") that can be safely
# filtered on.  When the user wants to query for “null” metadata they should
# use the same sentinel.
# ---------------------------------------------------------------------------
NULL_SENTINEL = "__NULL__"

def normalise_metadata(md: dict) -> dict:
    """Replace ``None`` with a sentinel string that is filter‑friendly.

    Args:
        md: Original metadata dictionary.

    Returns:
        A new metadata dictionary where every ``None`` value is replaced by
        ``NULL_SENTINEL``.
    """
    return {k: (NULL_SENTINEL if v is None else v) for k, v in md.items()}


def run_agent(query: str):
    # Load environment variables (e.g. ``OPENAI_API_KEY``) from a .env file.
    load_dotenv()

    # ---------------------------------------------------------------------
    # Embed model – we keep the OpenAI model the original example used.  If an
    # API key is not available the script will raise an informative error.
    # ---------------------------------------------------------------------
    if not os.getenv("OPENAI_API_KEY"):
        raise RuntimeError(
            "OPENAI_API_KEY not found in environment. Please set it in a .env "
            "file or the environment before running the script."
        )
    Settings.embed_model = OpenAIEmbedding(model="text-embedding-3-small")

    # ---------------------------------------------------------------------
    # Create a node with ``None`` in its metadata.  The metadata is normalised
    # so that ``None`` becomes the sentinel string.
    # ---------------------------------------------------------------------
    raw_metadata = {"start_date": None}
    node_01 = TextNode(
        text="This document has None in the metadata",
        id_="node_01",
        metadata=normalise_metadata(raw_metadata),
    )

    # Build the index.
    doc_index = VectorStoreIndex([node_01])

    # Debug: show what is actually stored in the index.
    print("Index nodes (stored metadata):")
    for node in doc_index.docstore.docs.values():
        print(node.metadata)
    print()

    # ---------------------------------------------------------------------
    # Filter for the sentinel value that represents ``None``.
    # ---------------------------------------------------------------------
    filter_null_start_date = MetadataFilter(
        key="start_date",
        operator=FilterOperator.EQ,
        value=NULL_SENTINEL,
    )
    filters = MetadataFilters(filters=[filter_null_start_date])
    retriever = doc_index.as_retriever(filters=filters, similarity_top_k=1)
    nodes = retriever.retrieve(query)

    # Show results.
    print("Retrieved nodes:")
    for node in nodes:
        print(f"{node.node_id}: {node.metadata}")
    print()

    return nodes


if __name__ == "__main__":
