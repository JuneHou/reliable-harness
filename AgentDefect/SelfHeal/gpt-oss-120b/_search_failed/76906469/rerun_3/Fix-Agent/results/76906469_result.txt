from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType, load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Initialise the LLM (Chat model)
llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

# Load the tools that the agent can use
tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Conversation memory – stores the chat history under the key "chat_history"
memory = ConversationBufferMemory(memory_key="chat_history")

# NOTE:
# The `CHAT_ZERO_SHOT_REACT_DESCRIPTION` agent type does **not** include the
# `chat_history` variable in its prompt, so the supplied memory is ignored.
# To have the agent remember previous turns, we need to use the
# `CONVERSATIONAL_REACT_DESCRIPTION` agent type, which is designed to work
# with `ConversationBufferMemory`.
agent = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CONVERSATIONAL_REACT_DESCRIPTION,
    memory=memory,
    handle_parsing_errors=True,
    verbose=True,
)

def run_agent(prompt: str) -> str:
    """Run the agent with the provided prompt and return its answer."""
    return agent.run(prompt)

if __name__ == "__main__":
    # First query – the answer is fetched via the Wikipedia tool.
    result1 = run_agent("What is the capital of France?")
    print(result1)
    # Second query – the agent can now refer back to the previous turn.
    result2 = run_agent("What was the last question I asked?")
    print(result2)
