from langchain_community.chat_models import ChatOllama
from langchain.schema import SystemMessage, HumanMessage


def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Send a user prompt to Llama 3 via Ollama and return a single clean response.

    Args:
        user_prompt: The question or instruction from the user.
        system_prompt: A system‑level instruction that sets the assistant's behaviour.
    """
    # Initialise the chat model – it knows how to handle system/assistant messages.
    chat = ChatOllama(model="llama3")

    # Build the message list in the order expected by the model.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # Invoke the model – the result is a ``ChatMessage`` object.
    response = chat.invoke(messages)
    # ``response`` may be a ``BaseMessage``; return its textual content.
    return response.content
