from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()

def run_agent(prompt: str) -> str:
    """Run a simple conversational retrieval over a CSV of movies.

    The original implementation limited the retriever to ``k=1``, which meant only a
    single document (the most similar one) was ever returned.  For queries that should
    match multiple rows – e.g. *all action movies from 1999* – the chain therefore
    produced incomplete answers (only the first match, author "A").

    By increasing ``k`` (the number of documents fetched from the vector store) we
    allow the language model to see all relevant rows and produce a complete answer.
    ``k`` can be tuned – 10 is a reasonable default for small CSVs.
    """
    # Initialise the embedding model (OpenAI embeddings are used here)
    embeddings = OpenAIEmbeddings()

    # Load the CSV file; each row becomes a LangChain Document
    csv_loader = CSVLoader(file_path="movies.csv")
    documents = csv_loader.load()

    # Create a Chroma vector store from the documents
    vectorstore = Chroma.from_documents(documents, embeddings)

    # Retrieve more than one relevant document (k > 1). Adjust as needed.
    retriever = vectorstore.as_retriever(search_kwargs={"k": 10})

    # Build the ConversationalRetrievalChain
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # ``chat_history`` should be a list; an empty list works for a fresh query.
    result = chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return result["answer"]

if __name__ == "__main__":
    # Example query that should now return every matching movie (authors A and C, etc.)
    print(run_agent("Which movies from 1999 are action?"))
