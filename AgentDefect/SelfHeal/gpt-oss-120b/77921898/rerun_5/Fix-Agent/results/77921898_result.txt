import os
from dotenv import load_dotenv, find_dotenv
from glob import glob
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader

# Load environment variables (e.g., OPENAI_API_KEY)
_ = load_dotenv(find_dotenv())


def run_agent(
    user_input: str,
    document_paths: list[str] | None = None,
    persist_directory: str = "chroma_db",
) -> list:
    """Retrieve relevant documents for *user_input*.

    If ``document_paths`` are supplied they are loaded, split, and indexed in a
    Chroma collection that is persisted to ``persist_directory``.  When no paths
    are supplied the function simply loads the existing persisted collection.
    """

    # ---------------------------------------------------------------------
    # Initialise OpenAI embeddings – the API key is read from the environment.
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise RuntimeError("OPENAI_API_KEY not found in environment variables")
    os.environ["OPENAI_API_KEY"] = openai_api_key
    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    # ---------------------------------------------------------------------
    # Build or load the vector store.
    # ---------------------------------------------------------------------
    if document_paths:
        # Load and split the supplied documents.
        all_chunks = []
        for doc_path in document_paths:
            loader = TextLoader(doc_path)
            documents = loader.load()
            splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
            chunks = splitter.split_documents(documents)
            all_chunks.extend(chunks)

        # Create a new Chroma collection from the chunks and *persist* it.
        docsearch = Chroma.from_documents(
            documents=all_chunks,
            embedding=embeddings,
            persist_directory=persist_directory,
        )
        # Ensure the collection is written to disk so that subsequent runs can
        # load it without re‑indexing.
        docsearch.persist()
    else:
        # No new documents – simply load the persisted collection.
        docsearch = Chroma(
            persist_directory=persist_directory,
            embedding_function=embeddings,
        )

    # ---------------------------------------------------------------------
    # Retrieve the most relevant documents.
    # ---------------------------------------------------------------------
    retriever = docsearch.as_retriever(search_kwargs={"k": 5})
    docs = retriever.get_relevant_documents(user_input)
    print(f"Number of documents retrieved: {len(docs)}")
    return docs


if __name__ == "__main__":
    # Load all *.txt files inside the "docs" folder.
    document_paths = glob("docs/*.txt")

    docs = run_agent(query, document_paths)
