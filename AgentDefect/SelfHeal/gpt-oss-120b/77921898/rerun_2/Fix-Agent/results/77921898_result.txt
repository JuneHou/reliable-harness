import os
from dotenv import load_dotenv, find_dotenv
from glob import glob
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader

# Load environment variables (e.g., OPENAI_API_KEY)
_ = load_dotenv(find_dotenv())


def run_agent(user_input: str, document_paths=None, persist_directory: str = "chroma_db"):
    """Create (or load) a Chroma vector store from the supplied documents and
    return the most relevant chunks for *user_input*.

    If *document_paths* is provided the function will:
        1. Load each text file.
        2. Split the texts into manageable chunks.
        3. Embed the chunks with OpenAI embeddings.
        4. Persist the resulting Chroma collection to *persist_directory*.
    If *document_paths* is ``None`` the function will simply load an existing
    collection from *persist_directory*.
    """

    # Ensure the OpenAI key is available to the embeddings class
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment variables")
    os.environ["OPENAI_API_KEY"] = openai_api_key

    # Initialise the embedding model
    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    # ---------------------------------------------------------------------
    # Build or load the Chroma vector store
    # ---------------------------------------------------------------------
    if document_paths:
        # Load and chunk the source documents
        all_chunks = []
        for doc_path in document_paths:
            loader = TextLoader(doc_path)
            docs = loader.load()
            splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
            chunks = splitter.split_documents(docs)
            all_chunks.extend(chunks)

        # Create a new Chroma collection from the chunks and persist it
        docsearch = Chroma.from_documents(
            documents=all_chunks,
            embedding=embeddings,
            persist_directory=persist_directory,
        )
        # Write the index to disk so it can be re‑loaded later
        docsearch.persist()
    else:
        # Load an existing collection – note the use of keyword arguments so the
        # directory is interpreted correctly (the first positional argument is the
        # *collection_name*, not the directory).
        docsearch = Chroma(
            persist_directory=persist_directory,
            embedding_function=embeddings,
        )

    # ---------------------------------------------------------------------
    # Retrieve the most relevant documents for the query
    # ---------------------------------------------------------------------
    retriever = docsearch.as_retriever(search_kwargs={"k": 5})
    docs = retriever.get_relevant_documents(user_input)
    print(f"Number of documents retrieved: {len(docs)}")
    return docs


if __name__ == "__main__":
    # Example usage – adjust the glob pattern to match your actual docs folder
    document_paths = glob("docs/*.txt")
    query = "What is machine learning?"
    docs = run_agent(query, document_paths)

    print("\n=== User Query Result ===")
    print(f"Query: {query}")
    print(f"Number of relevant documents: {len(docs)}")

    for i, doc in enumerate(docs):
        print(f"\n--- Document {i + 1} ---")
        print(doc.page_content)
