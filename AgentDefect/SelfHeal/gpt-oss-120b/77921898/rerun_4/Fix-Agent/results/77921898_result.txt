import os
from dotenv import load_dotenv, find_dotenv
from glob import glob
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.document_loaders import TextLoader

# Load environment variables (including OPENAI_API_KEY)
_ = load_dotenv(find_dotenv())


def run_agent(
    user_input: str,
    document_paths: list | None = None,
    persist_directory: str = "chroma_db",
    collection_name: str = "default",
) -> list:
    """Create (or load) a Chroma vector store and retrieve relevant documents.

    Args:
        user_input: The query string.
        document_paths: Optional list of file paths that should be indexed.
        persist_directory: Directory where Chroma will store its data.
        collection_name: Name of the collection inside the Chroma DB.

    Returns:
        A list of `Document` objects most relevant to the query.
    """

    # ---------------------------------------------------------------------
    # Initialise OpenAI embeddings – the key is read from the environment.
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise ValueError("OPENAI_API_KEY not found in environment variables")

    os.environ["OPENAI_API_KEY"] = openai_api_key  # required by the wrapper
    embeddings = OpenAIEmbeddings(openai_api_key=openai_api_key)

    # --------------------------------------------------------
    # If we have new documents, ingest them into a fresh collection
    # --------------------------------------------------------
    if document_paths:
        # Load and split the documents
        all_chunks = []
        for doc_path in document_paths:
            loader = TextLoader(doc_path)
            documents = loader.load()
            splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
            chunks = splitter.split_documents(documents)
            all_chunks.extend(chunks)

        # Create (or overwrite) the Chroma collection with the new vectors
        docsearch = Chroma.from_documents(
            all_chunks,
            embedding=embeddings,
            persist_directory=persist_directory,
            collection_name=collection_name,
        )
        # Persist the newly created collection to disk so it can be re‑loaded later
        docsearch.persist()
    else:
        # Load an existing collection – this will raise if the directory is empty
        docsearch = Chroma(
            persist_directory=persist_directory,
            embedding_function=embeddings,
            collection_name=collection_name,
        )

    # ---------------------------------------------------------------------
    # Retrieve relevant documents using the retriever interface
    # ---------------------------------------------------------------------
    retriever = docsearch.as_retriever(search_kwargs={"k": 5})
    docs = retriever.get_relevant_documents(user_input)
    print(f"Number of documents retrieved: {len(docs)}")
    return docs


if __name__ == "__main__":
    # Resolve the txt files located in the ./docs directory
    document_paths = glob("docs/*.txt")
    query = "What is machine learning?"
    docs = run_agent(query, document_paths)

    print("\n=== User Query Result ===")
    print(f"Query: {query}")
    print(f"Number of relevant documents: {len(docs)}")
    for i, doc in enumerate(docs):
        print(f"\n--- Document {i+1} ---")
        print(doc.page_content)
