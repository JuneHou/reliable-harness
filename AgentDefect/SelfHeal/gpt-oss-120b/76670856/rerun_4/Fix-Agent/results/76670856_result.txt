import os
import sys
import json
from pathlib import Path
from dotenv import load_dotenv

from langchain.chat_models import ChatOpenAI
from langchain.chains import ConversationalRetrievalChain
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.indexes import VectorstoreIndexCreator
from langchain.document_loaders import JSONLoader

load_dotenv()

def run_agent(queries):
    """Run a Conversational Retrieval QA over a small JSON review dataset.

    The function creates (or re‑uses) a Chroma vector store from the JSON file
    ``data/review.json``.  It then answers each query using a ``ConversationalRetrievalChain``.
    The retriever is configured to return **all** documents (or a sufficiently large
    number) so that questions about the whole collection – e.g. "how many reviews are there?"
    or "what is the first review?" – can be answered correctly.
    """

    # ---------------------------------------------------------------------
    # Prepare the data directory and JSON payload if it does not already exist
    # ---------------------------------------------------------------------
    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)
    review_file = data_dir / "review.json"

    if not review_file.exists():
        payload = {
            "reviews": [
                {"text": "Great hotel, excellent service and comfortable rooms."},
                {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
                {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
                {"text": "Average hotel. The room was okay, but nothing special."},
                {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
                {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
                {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
                {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
                {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
                {"text": "Not a bad hotel, but there are better options available in the area."},
            ]
        }
        review_file.write_text(json.dumps(payload, indent=2))

    # ---------------------------------------------------------------------
    # Control persistence – useful when you want to reuse the same index for many queries
    # ---------------------------------------------------------------------
    PERSIST = False  # set to True to keep the vectorstore on disk between runs

    # ---------------------------------------------------------------------
    # Load (or re‑load) the vectorstore
    # ---------------------------------------------------------------------
    if PERSIST and os.path.isdir("persist"):
        # Re‑use an existing persisted Chroma collection
        vectorstore = Chroma(persist_directory="persist", embedding_function=OpenAIEmbeddings())
        index = VectorstoreIndexCreator().from_vectorstore(vectorstore)
    else:
        # Load the JSON, create a fresh Chroma collection, and optionally persist it
        loader = JSONLoader(file_path=str(review_file), jq_schema=".reviews[]", content_key="text")
        if PERSIST:
            index = VectorstoreIndexCreator(
                vectorstore_kwargs={"persist_directory": "persist"}
            ).from_loaders([loader])
        else:
            index = VectorstoreIndexCreator().from_loaders([loader])

    # ---------------------------------------------------------------------
    # IMPORTANT: configure the retriever to return *all* documents (or a very high k).
    # The default ``k=4`` would make count‑type questions return the size of the
    # retrieved subset instead of the full collection.
    # ---------------------------------------------------------------------
    retriever = index.vectorstore.as_retriever(search_kwargs={"k": 100})

    # ---------------------------------------------------------------------
    # Build the Conversational Retrieval QA chain
    # ---------------------------------------------------------------------
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model="gpt-3.5-turbo", temperature=0),
        retriever=retriever,
        return_source_documents=False,
    )

    # ---------------------------------------------------------------------
    # Run the queries, keeping chat history so follow‑up questions work
    # ---------------------------------------------------------------------
    chat_history = []
    results = []
    for query in queries:
        if query.lower() in {"quit", "q", "exit"}:
            sys.exit()
        response = chain({"question": query, "chat_history": chat_history})
        answer = response["answer"]
        print(answer)
        results.append(answer)
        chat_history.append((query, answer))

    return results
