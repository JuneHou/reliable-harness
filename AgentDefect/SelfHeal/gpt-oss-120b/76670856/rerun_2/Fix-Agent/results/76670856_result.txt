import os
import sys
import json
from pathlib import Path
from dotenv import load_dotenv

from langchain.chat_models import ChatOpenAI
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.indexes import VectorstoreIndexCreator
from langchain.document_loaders import JSONLoader
from langchain.chains import ConversationalRetrievalChain
from langchain.docstore.document import Document

load_dotenv()


def _prepare_documents(data_path: Path) -> list[Document]:
    """Load the JSON reviews and combine them into a single Document.

    Using ``JSONLoader`` with ``jq_schema=".reviews[]"`` creates a separate
    Document for each review. Retrieval then returns only the most similar few
    documents, which prevents the model from answering questions that require the
    whole collection (e.g., total count, first/last review). By merging all
    reviews into one Document we guarantee that every query sees the complete
    dataset.
    """
    raw = json.loads(data_path.read_text())
    reviews = raw.get("reviews", [])
    # Create a readable combined string, numbering each review.
    lines = [f"Review {i+1}: {rev.get('text', '')}" for i, rev in enumerate(reviews)]
    combined = "\n".join(lines)
    metadata = {"source": str(data_path), "review_count": len(reviews)}
    return [Document(page_content=combined, metadata=metadata)]


def run_agent(queries):
    # ---------------------------------------------------------------------
    # Prepare data directory and JSON file
    # ---------------------------------------------------------------------
    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)
    review_file = data_dir / "review.json"

    if not review_file.exists():
        payload = {
            "reviews": [
                {"text": "Great hotel, excellent service and comfortable rooms."},
                {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
                {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
                {"text": "Average hotel. The room was okay, but nothing special."},
                {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
                {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
                {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
                {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
                {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
                {"text": "Not a bad hotel, but there are better options available in the area."},
            ]
        }
        review_file.write_text(json.dumps(payload, indent=2))

    # ---------------------------------------------------------------------
    # Vector store creation / loading
    # ---------------------------------------------------------------------
    PERSIST = False  # Set True to reuse a persisted index
    vectorstore = None

    if PERSIST and os.path.exists("persist"):
        # Reuse an existing persisted Chroma index.
        vectorstore = Chroma(persist_directory="persist", embedding_function=OpenAIEmbeddings())
    else:
        # Load reviews as a single document so that every query sees the whole collection.
        docs = _prepare_documents(review_file)
        if PERSIST:
            creator = VectorstoreIndexCreator(vectorstore_kwargs={"persist_directory": "persist"})
        else:
            creator = VectorstoreIndexCreator()
        creator.from_documents(docs)
        vectorstore = creator.vectorstore

    # ---------------------------------------------------------------------
    # Conversational Retrieval Chain
    # ---------------------------------------------------------------------
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model="gpt-3.5-turbo"),
        retriever=vectorstore.as_retriever(),
        return_source_documents=False,
    )

    # ---------------------------------------------------------------------
    # Run the queries
    # ---------------------------------------------------------------------
    chat_history = []
    results = []
    for query in queries:
        if query.lower() in {"quit", "q", "exit"}:
            sys.exit()
        result = chain({"question": query, "chat_history": chat_history})
        answer = result["answer"]
        print(answer)
        results.append(answer)
        chat_history.append((query, answer))

    # Persist if required
    if PERSIST:
        vectorstore.persist()

    return results
