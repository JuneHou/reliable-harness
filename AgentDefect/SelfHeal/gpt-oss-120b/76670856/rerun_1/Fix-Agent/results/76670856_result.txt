import os
import sys
import json
from pathlib import Path

from dotenv import load_dotenv
from langchain.chat_models import ChatOpenAI
from langchain.chains import ConversationalRetrievalChain
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.docstore.document import Document
from langchain.indexes import VectorstoreIndexCreator

load_dotenv()


def _load_reviews(file_path: Path) -> list[Document]:
    """Load the JSON review file and return a list of LangChain Document objects.

    Each review becomes its own Document with the review text as ``page_content``.
    """
    payload = json.loads(file_path.read_text())
    return [Document(page_content=review["text"]) for review in payload.get("reviews", [])]


def run_agent(queries):
    # ---------------------------------------------------------------------
    # Prepare data directory and review JSON file (create it if it does not exist)
    # ---------------------------------------------------------------------
    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)
    review_file = data_dir / "review.json"

    if not review_file.exists():
        payload = {
            "reviews": [
                {"text": "Great hotel, excellent service and comfortable rooms."},
                {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
                {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
                {"text": "Average hotel. The room was okay, but nothing special."},
                {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
                {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
                {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
                {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
                {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
                {"text": "Not a bad hotel, but there are better options available in the area."},
            ]
        }
        review_file.write_text(json.dumps(payload, indent=2))

    # ---------------------------------------------------------------------
    # Index creation / loading – optionally persisted to disk
    # ---------------------------------------------------------------------
    PERSIST = False
    index = None

    if PERSIST and os.path.exists("persist"):
        # Re‑use an existing persisted vectorstore
        vectorstore = Chroma(persist_directory="persist", embedding_function=OpenAIEmbeddings())
        index = VectorstoreIndexCreator(vectorstore=vectorstore)
    else:
        # Load documents from the JSON file
        docs = _load_reviews(review_file)
        if PERSIST:
            index = VectorstoreIndexCreator(
                vectorstore_kwargs={"persist_directory": "persist"}
            ).from_documents(docs)
        else:
            index = VectorstoreIndexCreator().from_documents(docs)
        # If we are persisting, make sure the data is written to disk now
        if PERSIST:
            index.vectorstore.persist()

    # ---------------------------------------------------------------------
    # Build a ConversationalRetrievalChain that retrieves *all* relevant docs.
    # We increase the ``k`` parameter so the retriever returns enough context
    # for questions that require aggregation (e.g., counting reviews).
    # ---------------------------------------------------------------------
    retriever = index.vectorstore.as_retriever(search_kwargs={"k": 20})
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model="gpt-3.5-turbo"),
        retriever=retriever,
    )

    # ---------------------------------------------------------------------
    # Interactive / batch query loop
    # ---------------------------------------------------------------------
    chat_history = []
    results = []
    for query in queries:
        if query.lower() in {"quit", "q", "exit"}:
            sys.exit()
        result = chain({"question": query, "chat_history": chat_history})
        answer = result["answer"]
        print(answer)
        results.append(answer)
        chat_history.append((query, answer))
    return results
