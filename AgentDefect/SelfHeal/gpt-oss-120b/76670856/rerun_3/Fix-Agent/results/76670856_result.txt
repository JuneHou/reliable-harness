import os
import sys
import json
from pathlib import Path

from dotenv import load_dotenv
from langchain.chat_models import ChatOpenAI
from langchain.chains import ConversationalRetrievalChain
from langchain.document_loaders import JSONLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.indexes import VectorstoreIndexCreator
from langchain.schema import Document
from langchain.vectorstores import Chroma

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()


def _load_reviews_as_documents(json_path: str) -> list[Document]:
    """Load the ``reviews`` array from a JSON file and return a list of
    :class:`~langchain.schema.Document` objects where each document's ``page_content``
    is the review text.  This explicit loader avoids the subtle behaviour of
    ``JSONLoader`` where the entire JSON object is stored as metadata, which can
    lead to retrieval results that contain the wrong review or incorrect counts.
    """
    data = json.loads(Path(json_path).read_text())
    reviews = data.get("reviews", [])
    docs: list[Document] = []
    for idx, entry in enumerate(reviews):
        # ``entry`` is expected to be a dict like {"text": "..."}
        text = entry.get("text", "")
        # Store the original index as metadata – useful for debugging / tracing.
        docs.append(Document(page_content=text, metadata={"source": f"review_{idx}"}))
    return docs


def run_agent(queries: list[str]):
    """Run a conversational retrieval chain over a small set of hotel reviews.

    Parameters
    ----------
    queries: list[str]
        A list of user questions. The function returns the list of answers.
    """

    # ---------------------------------------------------------------------
    # Ensure the data file exists – create it the first time the script runs.
    # ---------------------------------------------------------------------
    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)
    review_file = data_dir / "review.json"

    if not review_file.exists():
        payload = {
            "reviews": [
                {"text": "Great hotel, excellent service and comfortable rooms."},
                {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
                {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
                {"text": "Average hotel. The room was okay, but nothing special."},
                {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
                {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
                {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
                {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
                {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
                {"text": "Not a bad hotel, but there are better options available in the area."},
            ]
        }
        review_file.write_text(json.dumps(payload, indent=2))

    # ---------------------------------------------------------------------
    # Index creation – we keep the ability to persist the vector store for
    # repeated runs, but the default behaviour is to build a fresh index each
    # time.  The key change is that we feed the index **documents** that contain
    # only the review text, avoiding the previous JSON‑loader confusion.
    # ---------------------------------------------------------------------
    PERSIST = False  # Change to True if you want to reuse the index on disk.

    if PERSIST and os.path.exists("persist"):
        vectorstore = Chroma(persist_directory="persist", embedding_function=OpenAIEmbeddings())
        index = VectorstoreIndexCreator(vectorstore=vectorstore)
    else:
        # Load the reviews as clean Document objects.
        docs = _load_reviews_as_documents(str(review_file))
        # ``VectorstoreIndexCreator`` can ingest a list of Documents directly.
        if PERSIST:
            index = VectorstoreIndexCreator(vectorstore_kwargs={"persist_directory": "persist"}).from_documents(docs)
        else:
            index = VectorstoreIndexCreator().from_documents(docs)

    # ---------------------------------------------------------------------
    # Build the conversational retrieval chain.  A modest ``k`` value keeps
    # the context focused and improves answer accuracy for this tiny dataset.
    # ---------------------------------------------------------------------
    retriever = index.vectorstore.as_retriever(search_kwargs={"k": 4})
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model="gpt-3.5-turbo"),
        retriever=retriever,
    )

    chat_history = []
    answers: list[str] = []
    for query in queries:
        if query.lower() in {"quit", "q", "exit"}:
            sys.exit()
        response = chain({"question": query, "chat_history": chat_history})
        answer = response["answer"]
        print(answer)
        answers.append(answer)
        chat_history.append((query, answer))

    return answers

if __name__ == "__main__":
    sample_queries = [
        "How many reviews are there?",
        "What is the first review?",
        "What is the last review?",
        "Summarize the overall sentiment of the reviews.",
    ]
    run_agent(sample_queries)
