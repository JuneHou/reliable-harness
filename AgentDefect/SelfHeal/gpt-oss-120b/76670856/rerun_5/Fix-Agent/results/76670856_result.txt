from dotenv import load_dotenv
from pathlib import Path
import json
import os
import sys

import openai
from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI
from langchain.document_loaders import JSONLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.indexes.vectorstore import VectorStoreIndexWrapper
from langchain.vectorstores import Chroma
from langchain.indexes import VectorstoreIndexCreator

load_dotenv()

def run_agent(queries):
    """Run a ConversationalRetrievalChain over a small JSON review dataset.

    The original implementation only retrieved a default number of documents
    (typically 4) from the vector store which caused inaccurate answers for
    questions that rely on the complete dataset (e.g., counting reviews or
    retrieving the first/last review).  The fix is to ensure the retriever
    returns *all* documents (or at least a number larger than the dataset size)
    by configuring the `search_kwargs` ``k`` parameter.  Additionally, the
    loader is left unchanged because the JSON format is already appropriate.
    """

    data_dir = Path("data")
    data_dir.mkdir(exist_ok=True)

    review_file = data_dir / "review.json"

    # Create a small review dataset if it does not already exist
    if not review_file.exists():
        payload = {
            "reviews": [
                {"text": "Great hotel, excellent service and comfortable rooms."},
                {"text": "I had a terrible experience at this hotel. The room was dirty and the staff was rude."},
                {"text": "Highly recommended! The hotel has a beautiful view and the staff is friendly."},
                {"text": "Average hotel. The room was okay, but nothing special."},
                {"text": "I absolutely loved my stay at this hotel. The amenities were top notch."},
                {"text": "Disappointing experience. The hotel was overpriced for the quality provided."},
                {"text": "The hotel exceeded my expectations. The room was spacious and clean."},
                {"text": "Avoid this hotel at all costs! The customer service was horrendous."},
                {"text": "Fantastic hotel with a great location. I would definitely stay here again."},
                {"text": "Not a bad hotel, but there are better options available in the area."},
            ]
        }
        review_file.write_text(json.dumps(payload, indent=2))

    # Set to True if you want to persist the vectorstore to disk between runs
    PERSIST = False

    # ---------------------------------------------------------------------
    # Vector store / index creation
    # ---------------------------------------------------------------------
    if PERSIST and os.path.exists("persist"):
        # Load an existing persisted vector store
        vectorstore = Chroma(persist_directory="persist", embedding_function=OpenAIEmbeddings())
        index = VectorStoreIndexWrapper(vectorstore=vectorstore)
    else:
        # Load JSON reviews – each ``.reviews[]`` entry becomes a separate Document
        loader = JSONLoader(str(review_file), jq_schema=".reviews[]", content_key="text")
        # Load the documents so we can specify a higher ``k`` later on
        docs = loader.load()

        # Create a vector store from the documents.  When ``PERSIST`` is True we also
        # store the index on disk for reuse.
        if PERSIST:
            vectorstore = Chroma.from_documents(
                docs,
                embedding=OpenAIEmbeddings(),
                persist_directory="persist",
            )
        else:
            vectorstore = Chroma.from_documents(docs, embedding=OpenAIEmbeddings())

        index = VectorStoreIndexWrapper(vectorstore=vectorstore)

    # ---------------------------------------------------------------------
    # Retriever configuration – return *all* docs (or at least more than the
    # total number of reviews) to avoid incomplete answers.
    # ---------------------------------------------------------------------
    # The dataset contains 10 reviews, so we request 20 to be safe.  For larger
    # datasets you could use ``len(docs)`` dynamically.
    retriever = index.vectorstore.as_retriever(search_kwargs={"k": 20})

    # ---------------------------------------------------------------------
    # Conversational Retrieval Chain
    # ---------------------------------------------------------------------
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(model="gpt-3.5-turbo"),
        retriever=retriever,
    )

    chat_history = []
    results = []
    for query in queries:
        if query.lower() in {"quit", "q", "exit"}:
            sys.exit()
        result = chain({"question": query, "chat_history": chat_history})
        print(result["answer"])
        results.append(result["answer"])
        chat_history.append((query, result["answer"]))
    return results
