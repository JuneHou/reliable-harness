import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

# Load environment variables from a .env file (e.g., OPENAI_API_KEY)
load_dotenv()


def run_agent(prompt: str) -> str:
    """Send a prompt to the OpenAI chat model and return the text response.

    Args:
        prompt: The user query to send to the model.

    Returns:
        The content of the AI's reply as a plain string.
    """
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise EnvironmentError("OPENAI_API_KEY not found in environment variables.")

    # Initialise the chat model (defaults to gpt-3.5-turbo). You can also specify
    # a different model via the `model` argument, e.g., model="gpt-4o".
    llm = ChatOpenAI(openai_api_key=api_key)

    # `invoke` returns a `BaseMessage` (AIMessage). Extract the `content` field.
    response_message = llm.invoke(prompt)
    return response_message.content


if __name__ == "__main__":
    user_prompt = "How can LangSmith help with testing?"
    answer = run_agent(user_prompt)
    print("LLM response:", answer)
