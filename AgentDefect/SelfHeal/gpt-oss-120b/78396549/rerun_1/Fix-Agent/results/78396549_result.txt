def run_agent(prompt: str) -> str:
    """Send a prompt to the OpenAI chat model and return the text response.

    The ``ChatOpenAI`` chain returns an ``AIMessage`` (or similar) object whose
    content is stored in the ``content`` attribute.  The original code printed
    the whole message object, which results in an unreadable or empty output.
    This function extracts the ``content`` attribute (fall‑back to ``str`` if
    necessary) so that callers receive a plain string.
    """
    # ``ChatOpenAI`` automatically reads the ``OPENAI_API_KEY`` environment
    # variable, so we don't need to pass it explicitly.
    llm = ChatOpenAI()

    response = llm.invoke(prompt)
    # ``response`` is typically an ``AIMessage``; extract the textual content.
    if hasattr(response, "content"):
        return response.content  # type: ignore[return-value]
    # Fallback – return the string representation.
    return str(response)
