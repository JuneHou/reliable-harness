import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.schema import HumanMessage

load_dotenv()


def run_agent(prompt: str) -> str:
    """Send a prompt to the OpenAI chat model and return the textual response.

    Args:
        prompt: The user message to send to the model.

    Returns:
        The content of the model's reply as a plain string.
    """
    api_key = os.getenv("OPENAI_API_KEY")
    # Initialise the chat model – the key can also be read automatically from the env.
    llm = ChatOpenAI(openai_api_key=api_key)

    # Chat models expect a list of messages; we wrap the user prompt in a HumanMessage.
    response = llm.invoke([HumanMessage(content=prompt)])

    # ``invoke`` returns an ``AIMessage`` (a subclass of ``BaseMessage``).  The actual
    # text is stored in the ``content`` attribute.
    return response.content


if __name__ == "__main__":
