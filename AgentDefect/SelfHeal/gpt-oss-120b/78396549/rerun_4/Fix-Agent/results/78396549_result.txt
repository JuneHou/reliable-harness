import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI

load_dotenv()

def run_agent(prompt: str) -> str:
    """Send *prompt* to the OpenAI chat model and return the text response.

    ``ChatOpenAI.invoke`` returns an ``AIMessage`` object; the actual generated
    text is stored in its ``content`` attribute.  This helper extracts that
    attribute so that callers receive a plain string, which matches the
    expectation of the quick‑start example.
    """
    api_key = os.getenv("OPENAI_API_KEY")
    llm = ChatOpenAI(openai_api_key=api_key)
    response = llm.invoke(prompt)
    # ``response`` is an ``AIMessage`` – pull out the textual content.
    return response.content if hasattr(response, "content") else str(response)

if __name__ == "__main__":

    answer = run_agent(user_prompt)
    print("LLM response:", answer)
