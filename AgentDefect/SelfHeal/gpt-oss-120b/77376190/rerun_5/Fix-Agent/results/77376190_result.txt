import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import SimpleDirectoryReader, GPTVectorStoreIndex, Settings
from llama_index.llms.openai import OpenAI
from llama_index.prompts import PromptTemplate

# Load environment variables
_ = load_dotenv(find_dotenv())

# Ensure OpenAI key is available to the SDK and the OpenAI python client
openai_api_key = os.getenv("OPENAI_API_KEY")
if not openai_api_key:
    raise EnvironmentError("OPENAI_API_KEY not found in environment variables")
os.environ["OPENAI_API_KEY"] = openai_api_key
openai.api_key = openai_api_key

# ---------------------------------------------------------------------------
# Helper: build (or load) a vector index from the supplied documents folder.
# ---------------------------------------------------------------------------
def build_vector_index(docs_path: str = "docs") -> GPTVectorStoreIndex:
    """Create a persistent vector index for the documents in *docs_path*.

    The function creates the index only once – if a persisted index already
    exists in the ``index_storage`` directory it will be loaded instead of
    rebuilt.  This saves time on subsequent calls.
    """
    # Configuration for the LLM used by the index
    Settings.llm = OpenAI(temperature=0.3, model="gpt-3.5-turbo", max_tokens=512)
    Settings.chunk_size = 600
    Settings.chunk_overlap = int(600 * 0.1)  # 60 tokens overlap

    storage_dir = "index_storage"
    # If a persisted index exists, load it; otherwise build a new one
    if os.path.isdir(storage_dir) and os.path.exists(os.path.join(storage_dir, "index.json")):
        from llama_index.core import load_index_from_storage
        return load_index_from_storage(persist_dir=storage_dir)

    # Load documents from the directory
    docs = SimpleDirectoryReader(docs_path).load_data()
    # Create the index
    index = GPTVectorStoreIndex.from_documents(docs)
    # Persist for future runs
    index.storage_context.persist(persist_dir=storage_dir)
    return index

# ---------------------------------------------------------------------------
# Core agent logic – decides how to answer the user's request.
# ---------------------------------------------------------------------------
def run_agent(user_input: str, docs_path: str = "docs") -> str:
    """Answer *user_input* either by querying the document index or, if the
    request is for a legal draft (e.g., a non‑disclosure agreement), by
    explicitly instructing the LLM to produce that draft.
    """
    # Build or load the vector index once per program execution
    vector_index = build_vector_index(docs_path)

    # -----------------------------------------------------------------------
    # Detect requests for a non‑disclosure agreement (NDA) or similar legal
    # drafts.  This is a simple keyword‑based check; more sophisticated NLU can
    # be added later.
    # -----------------------------------------------------------------------
    nda_keywords = [
        "non‑disclosure agreement",
        "nda",
        "draft a nondisclosure agreement",
        "create a non disclosure agreement",
        "write an nda",
    ]
    if any(kw.lower() in user_input.lower() for kw in nda_keywords):
        # Custom system prompt that tells the model to comply with the request
        system_prompt = (
            "You are a helpful AI assistant that always complies with the user's "
            "request to draft legal documents. Generate a concise, standard "
            "non‑disclosure agreement (NDA) without any disclaimer or refusal."
        )
        user_prompt = user_input
        # Build a full prompt for the LLM
        template = PromptTemplate(
            """{system_prompt}\n\nUser: {user_prompt}\n\nAssistant:"""
        )
        full_prompt = template.format(system_prompt=system_prompt, user_prompt=user_prompt)
        # Directly query the LLM – we bypass the document index because the user
        # explicitly wants a fresh NDA draft.
        llm = Settings.llm  # re‑use the same OpenAI instance defined in Settings
        response = llm.complete(full_prompt)
        return response.text.strip()

    # -----------------------------------------------------------------------
    # Default behaviour – answer by querying the vector store.
    # -----------------------------------------------------------------------
    query_engine = vector_index.as_query_engine()
    response = query_engine.query(user_input)
    return response.response

# ---------------------------------------------------------------------------
# Simple command‑line test harness – useful when the script is run directly.
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    test_query = "Draft a non‑disclosure agreement for a software project."
    answer = run_agent(test_query)
    print("--- Result ---")
    print(answer)
