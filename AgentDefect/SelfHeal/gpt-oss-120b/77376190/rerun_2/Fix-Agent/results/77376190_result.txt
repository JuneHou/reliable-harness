import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import SimpleDirectoryReader, GPTVectorStoreIndex, Settings
from llama_index.llms.openai import OpenAI

# Load environment variables from .env file
_ = load_dotenv(find_dotenv())

def _get_openai_key():
    """Retrieve the OpenAI API key from the environment and set it for both
    the `openai` package and LlamaIndex's Settings.
    """
    key = os.getenv("OPENAI_API_KEY")
    if not key:
        raise RuntimeError("OPENAI_API_KEY not found in environment variables")
    os.environ["OPENAI_API_KEY"] = key
    openai.api_key = key
    return key


def _create_vector_index(path: str) -> GPTVectorStoreIndex:
    """Create (or load) a vector index from the documents located in *path*.

    The function configures the LLM settings, reads the documents, builds the
    index and persists it to disk so that subsequent calls can reuse the
    persisted index.
    """
    # LLM configuration – these values work well for short Q&A over the docs
    max_input_tokens = 4096
    response_max_tokens = 512
    chunk_size = 600
    chunk_overlap_ratio = 0.1

    Settings.llm = OpenAI(
        temperature=0.3, model="gpt-3.5-turbo", max_tokens=response_max_tokens
    )
    Settings.chunk_size = chunk_size
    Settings.chunk_overlap = int(chunk_size * chunk_overlap_ratio)

    # Load the documents from the directory
    docs = SimpleDirectoryReader(path).load_data()

    # Build the vector index
    vector_index = GPTVectorStoreIndex.from_documents(docs)
    # Persist the index for future runs (the folder will contain the index files)
    vector_index.storage_context.persist(persist_dir="vector_index_storage")
    return vector_index


def _load_or_create_index(path: str) -> GPTVectorStoreIndex:
    """Load a persisted index if it exists, otherwise create a new one.
    """
    storage_dir = "vector_index_storage"
    if os.path.isdir(storage_dir):
        # Load the previously persisted index
        from llama_index.core import load_index_from_storage
        return load_index_from_storage(persist_dir=storage_dir)
    else:
        # No persisted index – create a fresh one
        return _create_vector_index(path)


def _generate_nda_draft(user_input: str) -> str:
    """Directly ask the OpenAI model to draft a Non‑Disclosure Agreement.

    We bypass the document index because the request is a *generation* task,
    not a *retrieval* task.  The prompt is crafted to encourage the model to
    produce a concise, neutral NDA template without triggering refusal logic.
    """
    # A clear system instruction that frames the model as a helpful legal
    # drafter.  Keeping the instruction neutral helps avoid policy‑related
    # refusals.
    system_prompt = (
        "You are an AI assistant that helps users draft standard legal "
        "documents.  Produce a short, generic Non‑Disclosure Agreement (NDA) "
        "that can be used as a starting point.  Do not include any personal "
        "information or jurisdiction‑specific clauses unless the user asks for "
        "them."
    )

    # The user prompt – we forward the original request so the model can tailor "
    # the draft if the user provides extra context.
    user_prompt = user_input

    # Call the OpenAI ChatCompletion API directly
    response = openai.ChatCompletion.create(
        model="gpt-3.5-turbo",
        temperature=0.2,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_prompt},
        ],
        max_tokens=1024,
    )
    return response["choices"][0]["message"]["content"].strip()


def run_agent(user_input: str, docs_path: str = "docs") -> str:
    """Main entry point.

    * If the user asks for an NDA draft, we generate it directly.
    * Otherwise we fall back to the vector‑store retrieval over the supplied
      documents.
    """
    _get_openai_key()

    # Simple keyword detection – more sophisticated NLU can be added later.
    if "non disclosure agreement" in user_input.lower() or "nda" in user_input.lower():
        return _generate_nda_draft(user_input)

    # For all other queries, use the document index.
    vector_index = _load_or_create_index(docs_path)
    query_engine = vector_index.as_query_engine()
    response = query_engine.query(user_input)
    return response.response


if __name__ == "__main__":
    # Example usage – replace with any query you wish to test.

    ]
    for q in test_queries:
        print("\n=== Query ===")
        print(q)
        ans = run_agent(q)
        print("\n=== Answer ===")
        print(ans)
