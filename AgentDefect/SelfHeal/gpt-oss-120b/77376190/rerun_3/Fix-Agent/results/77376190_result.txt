import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import SimpleDirectoryReader, GPTVectorStoreIndex, Settings
from llama_index.llms.openai import OpenAI

# Load environment variables
_ = load_dotenv(find_dotenv())

# Ensure SSL verification (optional – set to empty string to disable verification in some environments)
os.environ['REQUESTS_CA_BUNDLE'] = ''


def run_agent(user_input: str, docs_path: str = "docs") -> str:
    """Process a user query.

    If the query asks for a non‑disclosure agreement (NDA) draft, we generate one
    directly with the OpenAI LLM. Otherwise we fall back to a vector‑store lookup
    over the documents in *docs_path*.
    """

    # ------------------------------------------------------------
    # Initialise OpenAI credentials
    # ------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise RuntimeError("OPENAI_API_KEY not found in environment variables")
    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key

    # ------------------------------------------------------------
    # Helper: create (or load) a vector index from the document folder
    # ------------------------------------------------------------
    def create_vector_index(path: str) -> GPTVectorStoreIndex:
        # LLM settings for the index – a modest temperature keeps the answers
        # consistent for factual queries.
        Settings.llm = OpenAI(
            temperature=0.3, model="gpt-3.5-turbo", max_tokens=512
        )
        # Chunking parameters – fine‑tune as needed for your data.
        Settings.chunk_size = 600
        Settings.chunk_overlap = int(600 * 0.1)  # 10 % overlap

        # Load documents from the directory.
        docs = SimpleDirectoryReader(path).load_data()
        # Build the vector index.
        vector_index = GPTVectorStoreIndex.from_documents(docs)
        # Persist the index for faster subsequent loads.  The directory is used,
        # not a JSON file.
        vector_index.storage_context.persist(persist_dir="vector_index_storage")
        return vector_index

    # ------------------------------------------------------------
    # Helper: generate an NDA using a system prompt.
    # ------------------------------------------------------------
    def generate_nda(prompt: str) -> str:
        system_prompt = (
            "You are an AI assistant that drafts concise, legally‑sound non‑disclosure "
            "agreements (NDAs). Do NOT provide any disclaimer about legal advice; just "
            "output the NDA in plain English. Use standard sections such as Parties, "
            "Confidential Information, Obligations, Term, and Governing Law."
        )
        # Call OpenAI ChatCompletion directly.
        response = openai.ChatCompletion.create(
            model="gpt-3.5-turbo",
            temperature=0.0,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": prompt},
            ],
        )
        # The API returns a list of choices – we take the first one.
        return response["choices"][0]["message"]["content"].strip()

    # ------------------------------------------------------------
    # Branch based on the user's intention.
    # ------------------------------------------------------------
    lowered = user_input.lower()
    if "non‑disclosure" in lowered or "nda" in lowered:
        # The user explicitly wants an NDA draft.
        return generate_nda(user_input)

    # ------------------------------------------------------------
    # Otherwise answer the query using the vector index.
    # ------------------------------------------------------------
    # Build (or load) the index – for simplicity we rebuild each call; in a real
    # deployment you would cache the index after the first creation.
    vector_index = create_vector_index(docs_path)

    # Query engine that uses the index.
    query_engine = vector_index.as_query_engine()
    response = query_engine.query(user_input)
    # `response` is an object; the actual text is in `response.response`.
    return str(response.response)


if __name__ == "__main__":
    # Example usage – feel free to replace with any prompt you like.
    user_input = "Please draft a non‑disclosure agreement between Acme Corp and Beta LLC."
    result = run_agent(user_input)
    print("\n=== Agent Result ===")
    print(result)
