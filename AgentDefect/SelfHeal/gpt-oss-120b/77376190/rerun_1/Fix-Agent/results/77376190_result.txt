import os
from dotenv import load_dotenv, find_dotenv
import openai
import gradio as gr
from llama_index.core import SimpleDirectoryReader, GPTVectorStoreIndex, Settings
from llama_index.llms.openai import OpenAI

# Load environment variables
_ = load_dotenv(find_dotenv())

# Ensure OpenAI key is available
openai_api_key = os.getenv("OPENAI_API_KEY")
if not openai_api_key:
    raise EnvironmentError("OPENAI_API_KEY not found in environment variables.")
os.environ["OPENAI_API_KEY"] = openai_api_key
openai.api_key = openai_api_key

# ---------------------------------------------------------------------------
# Helper: create a vector index from documents (used for generic Q&A)
# ---------------------------------------------------------------------------
def create_vector_index(path: str) -> GPTVectorStoreIndex:
    """Create and persist a GPTVectorStoreIndex from the documents in *path*.

    The function configures the LLM settings, reads the files, builds the index,
    and persists it under a dedicated folder (not a JSON file – the original code
    attempted to persist to a file, which is unsupported).
    """
    # LLM configuration – keep temperature low for deterministic answers
    Settings.llm = OpenAI(
        temperature=0.3,
        model="gpt-3.5-turbo",
        max_tokens=512,
    )
    # Chunking parameters
    Settings.chunk_size = 600
    Settings.chunk_overlap = int(600 * 0.1)  # 10% overlap

    # Load documents from the directory
    docs = SimpleDirectoryReader(path).load_data()

    # Build the vector index
    vector_index = GPTVectorStoreIndex.from_documents(docs)

    # Persist the index – use a folder for storage_context
    persist_dir = os.path.join("persisted_index", os.path.basename(path))
    os.makedirs(persist_dir, exist_ok=True)
    vector_index.storage_context.persist(persist_dir=persist_dir)
    return vector_index

# ---------------------------------------------------------------------------
# Core agent logic – decides between document‑based Q&A and free‑form generation
# ---------------------------------------------------------------------------
def run_agent(user_input: str, docs_path: str = "docs") -> str:
    """Process *user_input* and return a response.

    If the request is clearly a request for a legal draft (e.g., an NDA), we
    bypass the vector index and ask the OpenAI model to generate the draft using
    a custom system prompt. Otherwise we perform a similarity‑search over the
    provided documents.
    """

    # Normalise the request for simple keyword detection
    lowered = user_input.lower()
    nda_keywords = [
        "non‑disclosure",
        "non disclosure",
        "nda",
        "confidentiality agreement",
    ]
    wants_nda = any(keyword in lowered for keyword in nda_keywords)

    if wants_nda:
        # -------------------------------------------------------------------
        # Direct LLM generation of a Non‑Disclosure Agreement draft
        # -------------------------------------------------------------------
        system_prompt = (
            "You are an AI assistant that drafts concise, legally sound "
            "non‑disclosure agreements. Produce a short NDA template that "
            "covers the parties, purpose, confidentiality obligations, "
            "duration, and governing law. Do not include any disclaimer or "
            "refusal."
        )
        messages = [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_input},
        ]
        response = openai.ChatCompletion.create(
            model="gpt-3.5-turbo",
            messages=messages,
            temperature=0.2,
            max_tokens=800,
        )
        return response.choices[0].message.content.strip()
    else:
        # -------------------------------------------------------------------
        # Answer using the document vector index
        # -------------------------------------------------------------------
        # Build (or load) the vector index for the supplied docs
        vector_index = create_vector_index(docs_path)
        # Query engine from the index
        query_engine = vector_index.as_query_engine()
        response = query_engine.query(user_input)
        return response.response

# ---------------------------------------------------------------------------
# Example usage (can be removed when integrated with a UI like Gradio)
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    # Example 1: generic question over the documents
    generic_query = "What is the main topic of the documents?"
    print("--- Generic Query Result ---")
    print(run_agent(generic_query))

    # Example 2: request for an NDA draft
    nda_query = "Please draft a non‑disclosure agreement for a software project."
    print("\n--- NDA Draft Result ---")
    print(run_agent(nda_query))
