def run_agent(user_input: str, docs_path: str = "docs") -> str:
    """Run a simple RAG chatbot.

    The function builds a vector index from the documents located in ``docs_path``
    and then queries the index with ``user_input``.  A custom ``system_prompt`` is
    provided to the LLM so that, when the user asks for a non‑disclosure agreement
    (NDA) draft, the model returns a template instead of a refusal.
    """

    # ---------------------------------------------------------------------
    # Initialise OpenAI credentials
    # ---------------------------------------------------------------------
    openai_api_key = os.getenv("OPENAI_API_KEY")
    if not openai_api_key:
        raise RuntimeError("OPENAI_API_KEY not found in environment variables")
    os.environ["OPENAI_API_KEY"] = openai_api_key
    openai.api_key = openai_api_key

    # ---------------------------------------------------------------------
    # Helper: build a persistent vector index from the supplied documents
    # ---------------------------------------------------------------------
    def create_vector_index(path: str) -> GPTVectorStoreIndex:
        # LLM configuration – we keep the temperature low for reproducible
        # drafting and use a system prompt that explicitly allows NDA generation.
        system_prompt = (
            "You are a helpful assistant that can draft standard legal documents. "
            "When asked to provide a non‑disclosure agreement (NDA), "
            "give a concise, well‑structured template."
        )
        Settings.llm = OpenAI(
            temperature=0.2,
            model="gpt-3.5-turbo",
            max_tokens=1024,
            system_prompt=system_prompt,
        )
        # Chunking settings – these work well for typical short‑to‑medium docs.
        Settings.chunk_size = 600
        Settings.chunk_overlap = int(600 * 0.1)  # 10 % overlap

        # Load documents from the directory
        docs = SimpleDirectoryReader(path).load_data()
        # Build the vector store index
        vector_index = GPTVectorStoreIndex.from_documents(docs)
        # Persist the index for later reuse (folder, not a file name)
        vector_index.storage_context.persist(persist_dir="vector_store")
        return vector_index

    # Build (or rebuild) the index – in a production setting you would check if the
    # persisted index already exists to avoid rebuilding on every call.
    vector_index = create_vector_index(docs_path)

    # ---------------------------------------------------------------------
    # Query the index
    # ---------------------------------------------------------------------
    query_engine = vector_index.as_query_engine()
    response = query_engine.query(user_input)
    # ``response`` is a ``Response`` object; we want the raw text.
    return str(response)


if __name__ == "__main__":
