import os
from fastapi import FastAPI
from dotenv import load_dotenv
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()


def run_agent(query: str) -> str:
    """Run a simple RAG chain that returns a sentence‑formatted answer.

    The original implementation produced fragmented answers because the prompt
    only contained a *human* message.  By adding an explicit system prompt that
    instructs the model to answer in full sentences, the output becomes well‑
    formed (e.g., like Llama 2).  The rest of the pipeline – loading the CSV,
    building a FAISS vector store, retrieving relevant chunks and invoking the
    chain – remains unchanged.
    """

    # Initialise the LLM
    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    # Load the data source
    products = pd.read_csv("./data/products.csv")

    # Build the vector store
    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings()
    )
    retriever = vectorstore.as_retriever()

    # ---- Prompt construction ------------------------------------------------
    # A system prompt gives the model a clear role and instructs it to output
    # complete sentences.  The human prompt supplies the retrieved context and
    # the user's question.
    system_prompt = (
        "You are a helpful assistant. Answer the question based on the provided "
        "context. Respond in clear, complete sentences."
    )
    human_prompt = (
        "Answer the question based on the following context:\n"
        "{context}\n\nQuestion: {question}"
    )
    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", system_prompt),
            ("human", human_prompt),
        ]
    )
    # --------------------------------------------------------------------------

    # Build the runnable chain
    chain = (
        {
            "context": retriever,
            "question": RunnablePassthrough(),
        }
        | prompt
        | model
        | StrOutputParser()
    )

    # Execute the chain
    result = chain.invoke(query)
    return result


if __name__ == "__main__":
    test_query = "What is the highest budget movie?"
    print(run_agent(test_query))
