import os
from fastapi import FastAPI
from dotenv import load_dotenv
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough, RunnableLambda

load_dotenv()


def run_agent(query: str) -> str:
    """Run the RAG chain and return a full‑sentence answer.

    The original implementation produced fragmentary replies because the
    prompt did not explicitly ask the model to respond in complete sentences.
    By adding a system instruction and clarifying the human prompt we guide
    the LLM to generate well‑formed paragraph‑style output similar to Llama 2.
    """

    # ---------------------------------------------------------------------
    # Model
    # ---------------------------------------------------------------------
    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    # ---------------------------------------------------------------------
    # Data & Vector Store
    # ---------------------------------------------------------------------
    products = pd.read_csv("./data/products.csv")
    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(), embedding=OpenAIEmbeddings()
    )
    retriever = vectorstore.as_retriever()

    # ---------------------------------------------------------------------
    # Prompt
    # ---------------------------------------------------------------------
    # A system message tells the assistant how to format its answer.
    # The human template explicitly asks for a full‑sentence response.
    ANSWER_TEMPLATE = """Answer the question based on the following context.
Provide a concise answer in complete sentences.

Context:
{context}

Question: {question}"""

    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", "You are a helpful AI assistant. Always answer in full sentences."),
            ("human", ANSWER_TEMPLATE),
        ]
    )

    # ---------------------------------------------------------------------
    # Chain – retrieve docs, flatten them to a string, then feed the prompt.
    # ---------------------------------------------------------------------
    def docs_to_string(docs):
        """Convert a list of Document objects to a single string."""
        return "\n\n".join([doc.page_content for doc in docs])

    chain = (
        {
            "context": retriever | RunnableLambda(docs_to_string),
            "question": RunnablePassthrough(),
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)
    return result


if __name__ == "__main__":
    # Example usage – the answer will now be a well‑formed sentence.
    query = "What is the highest budget movie?"
    print(run_agent(query))
