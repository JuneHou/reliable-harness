import os
from fastapi import FastAPI
from dotenv import load_dotenv
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

# Create the FastAPI app once (outside the function) – this is not required for the
# retrieval/answer logic but mirrors the original intent.
app = FastAPI(
    title="LangChain Server",
    version="1.0",
    description="Spin up a simple api server using LangChain's Runnable interfaces",
)


def run_agent(query: str) -> str:
    """Run a simple RAG pipeline and return a *sentence formatted* answer.

    The original implementation produced fragmented answers because the LLM was not
    given any instruction about the desired output style.  By adding a clear system
    prompt that asks for full‑sentence responses the model reliably returns the
    expected format (similar to the behaviour you see with Llama 2).
    """

    # ---------------------------------------------------------------------
    # LLM configuration – use the correct parameter name (`model_name`).
    # A low temperature encourages deterministic, well‑formed sentences.
    # ---------------------------------------------------------------------
    model = ChatOpenAI(
        temperature=0.1,
        model_name="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    # ---------------------------------------------------------------------
    # Load the product data and build a FAISS vector store for retrieval.
    # ---------------------------------------------------------------------
    products = pd.read_csv("./data/products.csv")
    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(), embedding=OpenAIEmbeddings()
    )
    retriever = vectorstore.as_retriever()

    # ---------------------------------------------------------------------
    # Prompt template – we now include a system message that explicitly tells the
    # model to answer in full sentences.  The human template supplies the context
    # and the user question.
    # ---------------------------------------------------------------------
    ANSWER_TEMPLATE = """Answer the question based on the following context:\n\n{context}\n\nQuestion: {question}"""

    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", "You are a helpful assistant. Answer the question using the provided context. Respond in clear, complete sentences."),
            ("human", ANSWER_TEMPLATE),
        ]
    )

    # ---------------------------------------------------------------------
    # Build the runnable chain: retrieve → format prompt → LLM → parse output.
    # ---------------------------------------------------------------------
    chain = (
        {
            "context": retriever,
            "question": RunnablePassthrough(),
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)
    return result


if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
