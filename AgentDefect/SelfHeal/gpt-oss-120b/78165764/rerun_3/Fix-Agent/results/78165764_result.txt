import os
from fastapi import FastAPI
from dotenv import load_dotenv
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()


def run_agent(query: str) -> str:
    """Run a simple RAG chain using ChatOpenAI.

    The prompt now includes a system message that explicitly asks the model
    to answer in complete, grammatically‑correct sentences. This prevents the
    model from returning fragmentary or list‑style outputs.
    """

    # Initialise the LLM – keep temperature low for deterministic answers
    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    # Load product data and build a FAISS vector store
    products = pd.read_csv("./data/products.csv")
    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(),
        embedding=OpenAIEmbeddings(),
    )
    retriever = vectorstore.as_retriever()

    # FastAPI app – defined here only for completeness; it is not used directly
    app = FastAPI(
        title="LangChain Server",
        version="1.0",
        description="Spin up a simple API server using LangChain's Runnable interfaces",
    )

    # Prompt template
    ANSWER_TEMPLATE = """Answer the question based on the following context:
    {context}

    Question: {question}
    """

    # System message that forces full‑sentence answers
    SYSTEM_MESSAGE = (
        "You are a helpful assistant. Use the supplied context to answer the question. "
        "Provide the answer in complete, well‑formed sentences (no fragments or bullet points)."
    )

    # Build a ChatPromptTemplate with a system and a human message
    prompt = ChatPromptTemplate.from_messages(
        [
            ("system", SYSTEM_MESSAGE),
            ("human", ANSWER_TEMPLATE),
        ]
    )

    # Assemble the chain: retrieve context → format prompt → LLM → string output
    chain = (
        {
            "context": retriever,
            "question": RunnablePassthrough(),
        }
        | prompt
        | model
        | StrOutputParser()
    )

    result = chain.invoke(query)
    return result


if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
