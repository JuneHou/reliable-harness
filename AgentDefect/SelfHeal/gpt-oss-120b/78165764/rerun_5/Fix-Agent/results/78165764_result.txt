import os
from fastapi import FastAPI
from dotenv import load_dotenv
import pandas as pd
from langchain.vectorstores import FAISS
from langchain.embeddings import OpenAIEmbeddings
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_core.runnables import RunnablePassthrough

load_dotenv()

# Initialise FastAPI once (outside the agent function) – this avoids recreating the app on every request.
app = FastAPI(
    title="LangChain Server",
    version="1.0",
    description="Spin up a simple API server using LangChain's Runnable interfaces",
)


def run_agent(query: str) -> str:
    """Run a simple RAG pipeline that returns a full‑sentence answer.

    The original implementation produced fragmentary answers because the prompt
    did not explicitly ask for complete sentences.  The updated prompt now
    instructs the model to answer using clear, grammatical sentences, matching
    the behaviour that the user expects (similar to Llama 2).
    """

    # ---------------------------------------------------------------------
    # LLM configuration
    # ---------------------------------------------------------------------
    model = ChatOpenAI(
        temperature=0.1,
        model="gpt-3.5-turbo",
        openai_api_key=os.getenv("OPENAI_API_KEY"),
    )

    # ---------------------------------------------------------------------
    # Load product data and construct the vector store
    # ---------------------------------------------------------------------
    products = pd.read_csv("./data/products.csv")
    vectorstore = FAISS.from_texts(
        products["text"].astype(str).tolist(), embedding=OpenAIEmbeddings()
    )
    retriever = vectorstore.as_retriever()

    # ---------------------------------------------------------------------
    # Prompt – explicitly request full‑sentence answers
    # ---------------------------------------------------------------------
    ANSWER_TEMPLATE = (
        "Answer the question using the context below. Provide a concise answer "
        "in complete, grammatically correct sentences.\n\n"
        "Context:\n{context}\n\n"
        "Question: {question}\n"
    )
    prompt = ChatPromptTemplate.from_template(ANSWER_TEMPLATE)

    # ---------------------------------------------------------------------
    # Build the LangChain runnable chain
    # ---------------------------------------------------------------------
    chain = (
        {"context": retriever, "question": RunnablePassthrough()}
        | prompt
        | model
        | StrOutputParser()
    )

    # Execute the chain and return the formatted answer
    result = chain.invoke(query)
    return result


if __name__ == "__main__":
    query = "What is the highest budget movie?"
    print(run_agent(query))
