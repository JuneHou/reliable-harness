import json
import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv

dotenv.load_dotenv()

# -----------------------------------------------------------------------------
# Tool definition
# -----------------------------------------------------------------------------
@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a *JSON* representation of the DataFrame.

    LangChain tools must return JSON‑serialisable data.  Returning a ``pandas``
    ``DataFrame`` directly causes a serialization error and, more importantly,
    the calling script never receives the actual data.  We therefore convert the
    DataFrame to JSON using the ``split`` orientation – this provides column
    names, index and data in a compact format that can be reconstructed with
    ``pd.read_json(..., orient='split')``.
    """
    df = (
        pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    )
    # Keep only object columns (as in the original code)
    df = df.select_dtypes(include=["object"])
    # Return a JSON string that can be parsed later
    return df.to_json(orient="split")

# -----------------------------------------------------------------------------
# Agent execution helper
# -----------------------------------------------------------------------------
def run_agent(user_prompt: str) -> pd.DataFrame:
    """Run the ReAct agent and return the DataFrame produced by the tool.

    The agent is asked to *load* the Excel file and the ``load_excel_response``
    tool returns a JSON string.  The agent will echo that string as its final
    ``output``.  We capture the ``output`` field, deserialize the JSON and rebuild
    the original ``DataFrame``.
    """
    llm = ChatOpenAI(model="gpt-4o-mini")
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # The agent expects a dict with an ``input`` key.
    response = agent_executor.invoke({"input": user_prompt})

    # ``response`` is typically a dict like {"output": "..."}
    output = response.get("output", "")

    # The tool returns a JSON string; the agent may wrap it in quotes or add
    # extra text.  We attempt to locate the first JSON object in the output.
    try:
        # Find the first '{' ... '}' substring that can be parsed as JSON.
        start = output.find('{')
        end = output.rfind('}') + 1
        json_str = output[start:end]
        df_json = json.loads(json_str)
        df = pd.read_json(json.dumps(df_json), orient="split")
        return df
    except Exception as exc:
        raise ValueError(
            f"Failed to extract DataFrame from agent output. Output was: {output}\nError: {exc}"
        )

# -----------------------------------------------------------------------------
# Example usage
# -----------------------------------------------------------------------------
if __name__ == "__main__":
    path = "excel_file.xlsx"
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."
    df = run_agent(user_prompt)
    print(df)
