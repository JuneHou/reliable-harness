import os
import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv

dotenv.load_dotenv()

# A simple in‑memory registry that the tool can use to expose a DataFrame to the outer script.
_DF_REGISTRY: dict[str, pd.DataFrame] = {}

@tool
def load_excel_response(filepath: str):
    """Load an Excel file and expose the resulting DataFrame as a variable named ``df``.

    The function reads the Excel file, performs the original transformations, stores the
    DataFrame in the global ``_DF_REGISTRY`` under the key ``"df"`` and returns a short textual
    confirmation.  Returning a plain string (instead of the DataFrame itself) satisfies the
    LangChain tool contract – tools must return JSON‑serialisable text – while still making the
    DataFrame available to the caller after the agent finishes.
    """
    # Load the workbook and keep only object‑typed columns (the original behaviour).
    df = pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    df = df.select_dtypes(include=["object"]).copy()

    # Store the DataFrame so the outer script can retrieve it after the agent run.
    _DF_REGISTRY["df"] = df

    # Return a human‑readable message; the agent can reference the variable name ``df`` in later steps.
    return "Excel file loaded successfully and assigned to variable 'df'."


def run_agent(user_prompt: str, file_name: str = "data.json") -> dict:
    """Execute a ReAct‑style agent with the supplied prompt.

    The function builds the required LLM, tools (including a Python REPL), and runs the
    agent.  The returned ``response`` dictionary is the standard LangChain output; the loaded
    DataFrame (if any) can be accessed via ``_DF_REGISTRY`` after this function returns.
    """
    llm = ChatOpenAI(model="gpt-4o-mini")
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    # List of tools the agent can use.
    tools_list = [load_excel_response, repl_tool, shell_tool]

    # Use the official ReAct prompt template.
    prompt = hub.pull("hwchase17/react")

    # Build the agent and executor.
    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # Execute the agent.
    response = agent_executor.invoke({"input": user_prompt})
    return response


if __name__ == "__main__":
    # Example usage – load an Excel file and expose the DataFrame as a variable named ``df``.
    path = "excel_file.xlsx"
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."
    result = run_agent(user_prompt)
    print("Agent response:")
    print(result)

    # Retrieve the DataFrame that the tool stored.
    df = _DF_REGISTRY.get("df")
    if df is not None:
        print("\nDataFrame retrieved from the tool:")
        print(df.head())
    else:
        print("\nNo DataFrame was loaded by the tool.")
