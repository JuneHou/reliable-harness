import pandas as pd
import json
import re
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv

dotenv.load_dotenv()

# ---------------------------------------------------------------------------
# TOOL DEFINITION
# ---------------------------------------------------------------------------
# The tool now returns a JSON string (records orientation) instead of a raw
# pandas.DataFrame.  Returning a plain string makes it possible for the REACT
# agent to include the data in its textual response, and it also allows the
# calling script to easily reconstruct the DataFrame after the agent finishes.
# ---------------------------------------------------------------------------
@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a JSON representation of the DataFrame.

    The JSON string can be deserialized with ``pd.read_json(..., orient='records')``
    to obtain the original DataFrame.
    """
    # Load the Excel file, skipping the first two rows and forcing a single column
    df = pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    # Keep only object (string) columns – this mirrors the original behaviour
    df = df.select_dtypes(include=["object"])
    # Convert to JSON (records orientation) – this is a plain string
    return df.to_json(orient="records")

# ---------------------------------------------------------------------------
# AGENT WRAPPER
# ---------------------------------------------------------------------------
def _extract_json_from_text(text: str) -> str | None:
    """Extract the first JSON object/array from a block of text.

    The REACT agent usually returns the tool output wrapped in markdown code
    fences (e.g. ```json ... ```).  This helper finds the JSON substring and
    returns it, or ``None`` if no JSON could be located.
    """
    # Look for a JSON code block first
    code_block_match = re.search(r"```json\n(?P<json>.*?)\n```", text, re.DOTALL)
    if code_block_match:
        return code_block_match.group("json")
    # Fall back to searching for the first `{` ... `}` or `[` ... `]` pattern
    json_match = re.search(r"([\[{].*?[\]}])", text, re.DOTALL)
    if json_match:
        return json_match.group(1)
    return None


def run_agent(user_prompt: str, file_name: str = "data.json") -> pd.DataFrame | dict:
    """Run the REACT agent and return the DataFrame produced by the tool.

    The function creates the REACT agent, executes it with the supplied prompt,
    extracts the JSON payload that the ``load_excel_response`` tool returned, and
    reconstructs a pandas.DataFrame from that JSON.  If the JSON cannot be
    extracted, the raw agent output is returned for debugging.
    """
    # Initialise LLM and tools
    llm = ChatOpenAI(model="gpt-4o-mini")
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]

    # Use the official REACT prompt from the Hub
    prompt = hub.pull("hwchase17/react")

    # Build the agent and executor
    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # Execute the agent
    response = agent_executor.invoke({"input": user_prompt})
    # ``response`` is a dict with an ``output`` key containing the textual reply
    output_text: str = response.get("output", "")

    # Try to pull the JSON representation of the DataFrame from the output
    json_str = _extract_json_from_text(output_text)
    if json_str:
        try:
            # Re‑create the DataFrame from the JSON string
            df = pd.read_json(json_str, orient="records")
            return df
        except Exception as e:
            # If reconstruction fails, fall back to returning the raw JSON and error
            return {"error": f"Failed to parse JSON: {e}", "json": json_str, "raw_output": output_text}
    # If no JSON was found, return the raw textual output for the caller to inspect
    return {"raw_output": output_text}

if __name__ == "__main__":
    path = "excel_file.xlsx"
