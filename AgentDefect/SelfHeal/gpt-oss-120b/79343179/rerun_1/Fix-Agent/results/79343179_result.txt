import pandas as pd
from langchain.agents import AgentExecutor, create_react_agent, Tool
from langchain.tools import tool
from langchain_community.tools import ShellTool
from langchain_openai import ChatOpenAI
from langchain_experimental.utilities import PythonREPL
from langchain import hub
import dotenv
import json

dotenv.load_dotenv()

# -----------------------------------------------------------------------------
# Tool definitions
# -----------------------------------------------------------------------------

@tool
def load_excel_response(filepath: str) -> str:
    """Load an Excel file and return a *JSON* representation of the DataFrame.

    The ReAct agent can only return text.  By converting the DataFrame to a JSON
    string (using the ``split`` orientation) we get a deterministic, serialisable
    representation that can be reconstructed with ``pd.read_json``.
    """
    df = (
        pd.read_excel(filepath, skiprows=2, names=["responses"]).reset_index(drop=True)
    )
    # Keep only object (string) columns – the original behaviour.
    df = df.select_dtypes(include=["object"])
    # Return a JSON string that can be parsed back into a DataFrame.
    return df.to_json(orient="split")

# -----------------------------------------------------------------------------
# Agent runner
# -----------------------------------------------------------------------------

def run_agent(user_prompt: str) -> pd.DataFrame | dict:
    """Execute a ReAct agent and return a ``pandas.DataFrame`` if possible.

    The function builds a ``ChatOpenAI`` LLM, a ``PythonREPL`` tool, a shell
    tool and the custom ``load_excel_response`` tool defined above.  After the
    agent finishes, we attempt to interpret the final output as a JSON‑encoded
    DataFrame.  If parsing fails, the raw response dictionary is returned so the
    caller can inspect what the agent produced.
    """
    llm = ChatOpenAI(model="gpt-4o-mini")
    python_repl = PythonREPL()
    shell_tool = ShellTool()

    repl_tool = Tool(
        name="python_repl",
        description="Run Python commands in a REPL environment.",
        func=python_repl.run,
    )

    tools_list = [load_excel_response, repl_tool, shell_tool]
    prompt = hub.pull("hwchase17/react")

    shell_agent = create_react_agent(llm=llm, tools=tools_list, prompt=prompt)
    agent_executor = AgentExecutor(agent=shell_agent, tools=tools_list, verbose=True)

    # The ReAct chain expects a dict with an "input" key.
    response = agent_executor.invoke({"input": user_prompt})

    # ``response`` is typically a dict containing an ``output`` key with the
    # textual answer from the LLM.
    output_text: str = response.get("output", "")

    # Try to reconstruct a DataFrame from the JSON string.
    try:
        df = pd.read_json(output_text, orient="split")
        return df
    except Exception:
        # If the output is not valid JSON (e.g., the agent refused to use the
        # tool or returned explanatory text) we simply return the raw response.
        return response

# -----------------------------------------------------------------------------
# Example usage
# -----------------------------------------------------------------------------
if __name__ == "__main__":
    path = "excel_file.xlsx"
    user_prompt = f"Load the Excel file located at {path} and assign the result to a variable named df."
    result = run_agent(user_prompt)
    if isinstance(result, pd.DataFrame):
        print("DataFrame successfully reconstructed:")
        print(result.head())
    else:
        print("Agent response (not a DataFrame):")
        print(result)
