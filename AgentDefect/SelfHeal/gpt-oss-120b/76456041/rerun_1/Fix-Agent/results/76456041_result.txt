import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings

warnings.filterwarnings('ignore')

# Load environment variables (including OPENAI_API_KEY)
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def run_agent(queries):
    """Run a map‑reduce RetrievalQA on each CSV file.

    The function now creates the vector store **once per file** and retrieves **all
    documents** for the map‑reduce chain. This guarantees that aggregation‑type
    questions (e.g., counting German drivers) see the complete dataset and also
    prevents the excessive token usage that occurred when the code rebuilt the
    index for every single query.
    """
    results = []
    files = ["drivers.csv", "drivers_full.csv"]

    for file in files:
        print("=" * 36)
        print(file)
        print("=" * 36)
        # Track token usage for the whole file
        with get_openai_callback() as cb:
            # Load the CSV – each row becomes a separate Document
            loader = CSVLoader(file_path=file, encoding="utf-8")
            docs = loader.load()

            # Create embeddings and the Chroma vector store
            embeddings = OpenAIEmbeddings()
            db = Chroma.from_documents(docs, embeddings)

            # Retrieve **all** documents for the map‑reduce chain.  Using the full
            # length of the collection guarantees that no relevant rows are
            # filtered out by similarity search, which is essential for correct
            # counting / aggregation answers.
            retriever = db.as_retriever(search_kwargs={"k": len(docs)})

            # Build the RetrievalQA chain **once** per file (not per query)
            qa_chain = RetrievalQA.from_chain_type(
                llm=OpenAI(temperature=0, batch_size=20),
                chain_type="map_reduce",
                retriever=retriever,
                verbose=False,
            )

            for query in queries:
                print(f"\nQuery: {query}")
                result = qa_chain.run(query)
                results.append({"file": file, "query": query, "answer": result})
                print(f"Answer: {result}\n")

            # Show token usage for this file
            print(cb)
    return results


if __name__ == "__main__":
    queries = [
        "Tell me the name of every driver who is German",
        "How many drivers are German?",
        "What is the driver number for the German driver named 'John Doe'?",
        "What is the oldest birthdate among all drivers?",
    ]
    res = run_agent(queries)
    # Optionally, print a structured view of the results
    for r in res:
        print(r)
