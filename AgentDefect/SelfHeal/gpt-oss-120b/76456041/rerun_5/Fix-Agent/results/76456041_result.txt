import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings

warnings.filterwarnings("ignore")

# Load environment variables
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def load_documents(file_paths):
    """Load CSV files and return a list of Documents.

    Args:
        file_paths (list[str]): Paths to CSV files.

    Returns:
        list[Document]: All documents from the provided CSV files.
    """
    docs = []
    for fp in file_paths:
        loader = CSVLoader(file_path=fp, encoding="utf-8")
        docs.extend(loader.load())
    return docs


def build_vectorstore(docs, persist_directory="langchain_chroma"):
    """Create a Chroma vector store from documents.

    The store is persisted to disk so that the expensive embedding step is
    performed only once per dataset.
    """
    embeddings = OpenAIEmbeddings()
    # If a persisted DB already exists we can load it; otherwise we create a new one.
    if os.path.isdir(persist_directory):
        db = Chroma(persist_directory=persist_directory, embedding_function=embeddings)
    else:
        db = Chroma.from_documents(docs, embeddings, persist_directory=persist_directory)
    return db


def run_queries(queries, retriever):
    """Execute a list of queries against a given retriever using a map‑reduce QA chain.

    The RetrievalQA chain is instantiated once and reused for all queries to avoid
    unnecessary API calls and token consumption.
    """
    # Using a moderate `k` keeps token usage reasonable while still retrieving
    # enough context for the map‑reduce chain.
    retriever.search_kwargs["k"] = 20

    # Initialise the RetrievalQA chain once.
    qa_chain = RetrievalQA.from_chain_type(
        llm=OpenAI(temperature=0, batch_size=1),
        chain_type="map_reduce",
        retriever=retriever,
        verbose=False,
    )

    results = []
    for query in queries:
        print("\n>>> Query:", query)
        result = qa_chain.run(query)
        print("Result:", result)
        results.append(result)
    return results


if __name__ == "__main__":
    # Define CSV files you want to index.  Both the small and the full file are
    # loaded once and combined into a single vector store.
    csv_files = ["drivers.csv", "drivers_full.csv"]
    documents = load_documents(csv_files)

    # Build (or load) the vector store – this step creates embeddings only once.
    vectorstore = build_vectorstore(documents, persist_directory="chroma_db")
    retriever = vectorstore.as_retriever(search_type="similarity")

    # Define the questions you want to ask.
    queries = [
        "Tell me the name of every driver who is German",
        "How many drivers are German?",
        "What is the birthdate of driver number 14?",
        "Who is the oldest driver?",
    ]

    # Run the queries inside the OpenAI callback to capture token usage & cost.
    with get_openai_callback() as cb:
        answers = run_queries(queries, retriever)
        print("\n--- Token usage & cost ---")
        print(cb)
