import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.text_splitters import CharacterTextSplitter

warnings.filterwarnings('ignore')

# Load environment variables (API key etc.)
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def run_agent(queries):
    """Run a RetrievalQA (map‑reduce) over a CSV file.

    The function builds a vector store *once* per file, keeps the retriever
    reasonably small (``k`` documents) and re‑uses the QA chain for every
    query.  This prevents the massive token usage that occurs when the retriever
    returns thousands of rows and the chain is instantiated inside the inner
    loop.
    """
    results = []
    files = ["drivers.csv", "drivers_full.csv"]

    for file in files:
        print("=====================================")
        print(file)
        print("=====================================")
        with get_openai_callback() as cb:
            # Load the CSV – each row becomes a Document with a ``page_content``
            # string like "col1, col2, ...".  Splitting the rows into smaller
            # chunks makes the vectors more useful and reduces the amount of
            # text that the LLM needs to process in the map step.
            loader = CSVLoader(file_path=file, encoding="utf-8")
            raw_docs = loader.load()

            # Optional: split each row into smaller pieces (max 500 chars).
            splitter = CharacterTextSplitter(chunk_size=500, chunk_overlap=0)
            docs = splitter.split_documents(raw_docs)

            # Create embeddings and vector store.
            embeddings = OpenAIEmbeddings()
            db = Chroma.from_documents(docs, embeddings)
            # Retrieve only the most relevant pieces – a large ``k`` forces the
            # map‑reduce chain to call the LLM on every document, exploding the
            # cost.
            retriever = db.as_retriever(search_kwargs={"k": 10})

            # Build the QA chain **once** for this file.
            qa = RetrievalQA.from_chain_type(
                llm=OpenAI(temperature=0),
                chain_type="map_reduce",
                retriever=retriever,
                verbose=False,
            )

            for query in queries:
                print(f"Query: {query}")
                result = qa.run(query)
                results.append(result)
                print(f"Result: {result}\n")

            # Show token usage / cost for this file.
            print(cb)
    return results


if __name__ == "__main__":
    # Example queries that require aggregation – the map‑reduce chain will
    # combine the per‑document answers correctly now that we limit ``k`` and
    # reuse the chain.
    queries = [
        "Tell me the name of every driver who is German.",
        "How many drivers are German?",
        "What is the oldest birthdate among all drivers?",
        "Give me the name of the driver with driver number 14."
    ]
    responses = run_agent(queries)
    for r in responses:
        print(r)

def _build_retriever(csv_path: str, k: int = 20) -> object:
    """Create a Chroma retriever for a CSV file.

    Parameters
    ----------
    csv_path: str
        Path to the CSV file.
    k: int, optional
        Number of relevant documents to fetch per query. Keeping this small
        dramatically reduces token usage for the map‑reduce chain.

    Returns
    -------
    object
        A LangChain retriever instance.
    """
    # Load CSV – each row becomes a Document (page_content is the row as a CSV string)
    loader = CSVLoader(file_path=csv_path, encoding="utf-8")
    docs = loader.load()

    # Create embeddings – batch size is set to a reasonable value to avoid hitting rate limits
    embeddings = OpenAIEmbeddings(chunk_size=1000)

    # Build the vector store.  The `persist_directory` ensures we don't recompute embeddings
    # when the same file is processed again in the same run.
    db = Chroma.from_documents(docs, embeddings, persist_directory=f"./chroma_{os.path.basename(csv_path)}")
    db.persist()

    # Return a retriever configured to return *k* most similar documents.
    return db.as_retriever(search_type="similarity", search_kwargs={"k": k})


def run_agent(queries, csv_files=None, k=20, chain_type="stuff"):
    """Execute a list of queries against one or more CSV files.

    The function builds a retriever for each CSV, runs the same RetrievalQA chain
    (with the chosen ``chain_type``) for every query, and returns the list of answers.
    By constructing the RetrievalQA object **once per CSV** (instead of once per query)
    we avoid unnecessary LLM calls and drastically lower the cost.
    """
    if csv_files is None:
        csv_files = ["drivers.csv", "drivers_full.csv"]

    all_results = []

    for csv_path in csv_files:
        print("=" * 40)
        print(csv_path)
        print("=" * 40)

        # Use a callback to track token usage per CSV file
        with get_openai_callback() as cb:
            retriever = _build_retriever(csv_path, k=k)

            # Build the RetrievalQA chain **once** for this CSV.  The same chain can be
            # reused for all queries because the underlying retriever is already set.
            qa = RetrievalQA.from_chain_type(
                llm=OpenAI(temperature=0, batch_size=20),
                chain_type=chain_type,  # "stuff" works well for small‑ish result sets
                retriever=retriever,
                verbose=False,
                return_source_documents=False,
            )

            for query in queries:
                print(f"Query: {query}")
                result = qa.run(query)
                all_results.append(result)
                print(f"Answer: {result}\n")

            # Print token usage for this CSV – useful for debugging cost issues
            print(cb)

    return all_results
