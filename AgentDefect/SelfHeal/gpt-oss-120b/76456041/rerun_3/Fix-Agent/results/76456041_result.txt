import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings

warnings.filterwarnings('ignore')

# Load environment variables (including OPENAI_API_KEY)
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")

# ---------------------------------------------------------------------------
# Helper: build a Chroma vector store for a given CSV file.
# We create the store **once** per file and reuse the retriever for all queries.
# ---------------------------------------------------------------------------

def build_retriever(csv_path: str, k: int = 20) -> "BaseRetriever":
    """Load a CSV, embed its rows and return a retriever.

    Args:
        csv_path: Path to the CSV file.
        k: Number of most‑similar documents to return per query. Keeping this
           relatively small dramatically reduces token consumption and cost.
    """
    loader = CSVLoader(file_path=csv_path, encoding="utf-8")
    docs = loader.load()
    embeddings = OpenAIEmbeddings()
    # Persist the collection on disk so we don't re‑embed on every run.
    # Using the CSV file name as the collection name guarantees a unique store.
    db = Chroma.from_documents(docs, embeddings, collection_name=os.path.basename(csv_path))
    return db.as_retriever(search_type="similarity", search_kwargs={"k": k})


def run_agent(queries, files=None, k=20):
    """Run RetrievalQA (map‑reduce) on each CSV file for the supplied queries.

    The function returns a list of results in the order they were produced.
    """
    if files is None:
        files = ["drivers.csv", "drivers_full.csv"]

    results = []
    # Build a retriever for each CSV only once – this avoids re‑creating the
    # embeddings and vector store on every query, cutting both latency and cost.
    retrievers = {file: build_retriever(file, k=k) for file in files}

    for file in files:
        print("=" * 40)
        print(file)
        print("=" * 40)
        retriever = retrievers[file]
        # Initialise the RetrievalQA chain **once per file** – the chain holds
        # references to the LLM and the retriever, so we don't need to rebuild it
        # for every individual query.
        qa_chain = RetrievalQA.from_chain_type(
            llm=OpenAI(temperature=0, batch_size=20),
            chain_type="map_reduce",
            retriever=retriever,
            verbose=False,
            # The map‑reduce chain works best when the LLM is asked to return a
            # concrete answer. The default combine prompt already does this, but
            # we keep the default for simplicity.
        )
        for query in queries:
            print(f"Query: {query}")
            with get_openai_callback() as cb:
                result = qa_chain.run(query)
                results.append(result)
                print(f"Result: {result}\n")
                # Show cost for the single query – helpful for debugging.
                print(cb)
    return results


if __name__ == "__main__":
    test_queries = [
        "Tell me the name of every driver who is German.",
        "How many drivers are German?",
        "What is the driver number of the oldest driver?",
        "What is the oldest birthdate in the data?",
        "Give me the name of driver number 14.",
    ]
    # Using a modest k (e.g., 20) keeps token usage low while still retrieving
    # enough context for the map‑reduce chain to answer correctly.
    run_agent(test_queries, k=20)
