import os
import warnings
from dotenv import load_dotenv, find_dotenv
from langchain.document_loaders import CSVLoader
from langchain.callbacks import get_openai_callback
from langchain.chains import RetrievalQA
from langchain.llms import OpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings.openai import OpenAIEmbeddings

warnings.filterwarnings('ignore')

# Load environment variables (including OPENAI_API_KEY)
load_dotenv(find_dotenv())
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


def build_retriever(csv_path: str, k: int = 10):
    """Create a Chroma retriever for a given CSV file.

    Args:
        csv_path: Path to the CSV file.
        k: Number of nearest neighbours to return for each query.

    Returns:
        A LangChain retriever instance.
    """
    loader = CSVLoader(file_path=csv_path, encoding="utf-8")
    docs = loader.load()
    embeddings = OpenAIEmbeddings()
    # Build the vector store (this is the expensive step – do it once per file)
    db = Chroma.from_documents(docs, embeddings)
    # Return a retriever that fetches the top‑k most similar documents
    return db.as_retriever(search_type="similarity", search_kwargs={"k": k})


def run_agent(queries, csv_files=None, k=10):
    """Run RetrievalQA (map‑reduce) on each CSV file for a list of queries.

    Args:
        queries: List of natural‑language questions.
        csv_files: List of CSV file names. If ``None`` the default demo files are used.
        k: Number of documents the retriever should return per query.

    Returns:
        A list of strings – the answers produced by the model.
    """
    if csv_files is None:
        csv_files = ["drivers.csv", "drivers_full.csv"]

    results = []
    # Create a single LLM instance – re‑using it saves token overhead.
    llm = OpenAI(temperature=0, model_name="gpt-3.5-turbo")

    for file in csv_files:
        print("=" * 40)
        print(f"Processing: {file}")
        print("=" * 40)
        # Build a retriever for the current CSV file.
        retriever = build_retriever(file, k=k)

        # Initialise the RetrievalQA chain once per file (instead of per query).
        qa_chain = RetrievalQA.from_chain_type(
            llm=llm,
            chain_type="map_reduce",
            retriever=retriever,
            verbose=False,
        )

        for query in queries:
            with get_openai_callback() as cb:
                print(f"Query: {query}")
                answer = qa_chain.run(query)
                results.append(answer)
                print(f"Answer: {answer}\n")
                # The callback prints token usage & cost – useful for debugging.
                print(cb)
    return results


if __name__ == "__main__":
    # Example queries that previously gave wrong answers.
    queries = [
        "Tell me the name of every driver who is German",
        "How many drivers are German?",
        "What is the driver number of the 14th driver?",
        "Which driver has the oldest birthdate?",
    ]
    run_agent(queries)
