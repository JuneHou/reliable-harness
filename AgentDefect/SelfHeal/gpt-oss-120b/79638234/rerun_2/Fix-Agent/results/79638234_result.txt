from langchain_ollama import ChatOllama
from langchain.schema import HumanMessage

def run_agent(input_text: str) -> str:
    """Generate a response using the Ollama chat model.

    Args:
        input_text: The user prompt.

    Returns:
        The model's reply as a plain string.
    """
    # Initialise the LLM once with non‑deterministic sampling parameters.
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.9,        # >0 enables randomness
        top_p=0.9,
        repeat_penalty=1.2,
        num_ctx=2048,
        # Ensure no fixed seed is used; Ollama will pick a random seed each call.
        seed=None,
        timeout=120,
    )

    # LangChain expects a Message object (or a list). Using HumanMessage makes the role explicit.
    message = HumanMessage(content=input_text)
    response = llm.invoke(message)
    return response.content

if __name__ == "__main__":
