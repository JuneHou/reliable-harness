from langchain_ollama import ChatOllama
from langchain.schema import HumanMessage

def run_agent(input_text: str) -> str:
    """Invoke the Ollama model with proper message handling.

    Args:
        input_text: The user's prompt.

    Returns:
        The model's generated response as a string.
    """
    # Initialise the Ollama chat model. Parameters such as temperature, top_p,
    # repeat_penalty, etc., are passed directly to the underlying Ollama API.
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,      # higher temperature → more randomness
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        timeout=120,
        # ``seed`` is omitted (or set to ``None``) so that each request gets a
        # fresh random seed, ensuring varied outputs for identical prompts.
    )

    # LangChain's chat models expect a list of ``BaseMessage`` objects.
    # ``HumanMessage`` represents the user's input.
    messages = [HumanMessage(content=input_text)]

    # ``invoke`` returns an ``AIMessage``; extract its ``content`` attribute.
    response = llm.invoke(messages)
    return response.content
