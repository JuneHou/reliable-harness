from langchain_ollama import ChatOllama
from langchain.schema import HumanMessage


def run_agent(input_text: str) -> str:
    """Send a prompt to an Ollama model and return the generated reply.

    The function constructs a proper ``HumanMessage`` list, which is the
    expected input type for ``ChatOllama.invoke``. Supplying a plain string can
    lead to deterministic behaviour because the underlying runnable treats the
    string as a raw prompt without role information. By providing the message
    object we ensure the model receives the prompt correctly and respects the
    stochastic parameters (temperature, top_p, repeat_penalty).
    """
    # Initialise the chat model with desired sampling parameters.
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        num_ctx=2048,
        repeat_penalty=1.2,
        timeout=120,
    )

    # Build the message list expected by ChatOllama.
    messages = [HumanMessage(content=input_text)]

    # Invoke the model and extract the content of the response.
    response = llm.invoke(messages)
    return response.content
