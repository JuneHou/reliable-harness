from langchain_ollama import ChatOllama
from langchain.schema import HumanMessage

def run_agent(input_text: str) -> str:
    """Generate a response from the Ollama model with stochastic sampling.

    Parameters
    ----------
    input_text: str
        The user prompt.

    Returns
    -------
    str
        The model's reply. Each call should produce a different answer when
        ``temperature`` > 0 because the request is sent as a proper chat message.
    """
    # Initialise the chat model with the desired sampling parameters.
    llm = ChatOllama(
        model="qwen2.5:14b",
        temperature=0.7,
        top_p=0.95,
        repeat_penalty=1.2,
        num_ctx=2048,
        timeout=120,
    )

    # LangChain's chat wrappers expect a list of ``BaseMessage`` objects.
    # Using ``HumanMessage`` ensures the temperature and other sampling args are
    # honoured by the Ollama server.
    messages = [HumanMessage(content=input_text)]
    response = llm.invoke(messages)
    # ``invoke`` on ``ChatOllama`` returns the assistant's reply as a plain string.
    return response

if __name__ == "__main__":
    print(run_agent("Please tell me a lighthearted joke"))
