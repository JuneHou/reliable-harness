import os
from typing import Any, List, Mapping, Optional
from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class Wrapper(LLM):
    """A thin wrapper around an existing LangChain LLM.

    The wrapper logs the prompt/response pairs to a file but *does not*
    trigger an additional LLM observation when used inside an agent.  It
    achieves this by calling the underlying LLM's private ``_call`` method
    directly (instead of the public ``__call__`` entry point which would
    generate its own observation).  This makes the agent treat the wrapped
    LLM exactly like the original LLM.
    """

    llm: OpenAI
    log: str

    def __init__(self, llm: OpenAI, log: str):
        super().__init__()
        self.llm = llm
        self.log = log

    @property
    def _llm_type(self) -> str:
        # Keep the type distinct so that LangChain does not try to cache
        # based on a known provider name.
        return "custom"

    def _call(self, prompt: str, stop: Optional[List[str]] = None) -> str:
        """Call the underlying LLM without generating a second observation.

        Parameters
        ----------
        prompt: str
            The prompt to send to the model.
        stop: Optional[List[str]]
            Stop tokens – passed straight through to the wrapped model.
        """
        # Directly invoke the inner LLM's private method to avoid the extra
        # LangChain callback/observation that ``self.llm(prompt)`` would add.
        reply = self.llm._call(prompt, stop=stop)
        # Log the interaction for debugging / auditing.
        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100 + "\n")
        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        """Return parameters that uniquely identify this wrapper.
        This is required by LangChain's caching and serialization logic.
        """
        return {"llm": self.llm}


def setup_components():
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    wllm = Wrapper(llm=llm, log="output.log")

    tools = [PythonAstREPLTool()]
    memory = ConversationBufferMemory(memory_key="chat_history")
    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt: str):
    try:
        wllm, tools, memory = setup_components()
    except Exception as e:
        return None

    try:
        agent = create_agent(wllm, tools, memory)
    except Exception as e:
        return None

    try:
        response = agent(prompt)
        return response
    except Exception as e:
        return None


if __name__ == "__main__":
    prompt = "What is 5-7?"
    response = run_agent(prompt)
    if response:
        print(response)
        # ``response`` is a dict for the default agents.
        print(type(response), response.get("output"))
    else:
        print("Agent failed to produce a response.")
