import os
from typing import Any, List, Mapping, Optional
from dotenv import load_dotenv
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class WrapperLLM(OpenAI):
    """A thin wrapper around the OpenAI LLM that logs prompts and responses.
    Inherits from ``OpenAI`` so all internal behaviours (including the
    ``_llm_type`` and identifying parameters) remain unchanged. This prevents
    the ``conversational-react-description`` agent from mis‑interpreting the
    wrapped LLM output and eliminates duplicate observations.
    """

    def __init__(self, *, log: str = "output.log", **openai_kwargs: Any) -> None:
        self.log = log
        super().__init__(**openai_kwargs)

    def _call(self, prompt: str, stop: Optional[List[str]] = None) -> str:
        # Call the parent OpenAI implementation to get the raw reply.
        reply = super()._call(prompt, stop=stop)
        # Log the interaction – using ``utf‑8`` and ensuring the file is closed.
        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100)
        # Return the reply unchanged so the agent sees exactly the same output
        # as it would from the original OpenAI LLM.
        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        # Extend the identifying parameters with the log path – everything else
        # is inherited from ``OpenAI`` which keeps the hashing stable.
        params = dict(super()._identifying_params)
        params["log"] = self.log
        return params


def setup_components():
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    # Wrap the original LLM – all behaviour stays the same, only logging changes.
    wllm = WrapperLLM(log="output.log", openai_api_key=os.getenv("OPENAI_API_KEY"), temperature=0)

    tools = [PythonAstREPLTool()]
    memory = ConversationBufferMemory(memory_key="chat_history")
    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt: str):
    try:
        wllm, tools, memory = setup_components()
    except Exception as e:
        raise RuntimeError(f"Component setup failed: {e}")

    try:
        agent = create_agent(wllm, tools, memory)
    except Exception as e:
        raise RuntimeError(f"Agent creation failed: {e}")

    # The agent returns a dictionary with keys like ``output`` and ``intermediate_steps``.
    response = agent(prompt)
    return response


if __name__ == "__main__":
    prompt = "What is 5-7?"
    response = run_agent(prompt)
    print(response)
    # ``response`` is a dict; ``output`` holds the final answer.
    print(type(response), response.get("output"))
