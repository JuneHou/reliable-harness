import os
from typing import Any, List, Mapping, Optional

from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class Wrapper(LLM):
    """A thin wrapper around an existing OpenAI LLM.

    The wrapper forwards the prompt to the underlying LLM, logs the
    request/response pair to a file and returns the raw text.  By inheriting
    from :class:`langchain.llms.base.LLM` and correctly implementing the
    required abstract properties/methods we ensure that the agent treats the
    wrapper exactly like a native LLM – no duplicate observations are
    produced.
    """

    llm: OpenAI
    log: str

    def __init__(self, llm: OpenAI, log: str):
        # ``BaseModel`` (the parent of ``LLM``) expects field values to be
        # passed via ``super().__init__`` so that they are stored in the model
        # state and can be used for hashing/identifying.
        super().__init__(llm=llm, log=log)

    @property
    def _llm_type(self) -> str:
        # ``_llm_type`` is used by LangChain for serialization; using a custom
        # value makes it clear that this is a user‑defined wrapper.
        return "custom_wrapper"

    def _call(self, prompt: str, stop: Optional[List[str]] = None) -> str:
        """Forward the prompt to the underlying OpenAI model.

        ``OpenAI.invoke`` is the public API that respects LangChain's callback
        system.  Using ``self.llm(prompt)`` would bypass those callbacks and can
        lead to the agent recording the same observation twice.
        """
        # Get the raw response from the wrapped LLM.
        reply = self.llm.invoke(prompt)

        # Log the interaction – this is purely for debugging and does not
        # interfere with LangChain's internal bookkeeping.
        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100)

        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        """Parameters that uniquely identify this LLM instance.

        Returning only primitive values (instead of the whole wrapped LLM
        object) prevents LangChain from thinking the LLM is a separate tool and
        eliminates duplicate observations.
        """
        return {
            "model_name": self.llm.model_name,
            "temperature": getattr(self.llm, "temperature", None),
        }

    @property
    def _default_params(self) -> Mapping[str, Any]:
        """Default parameters forwarded to the wrapped LLM.
        """
        return {}


def setup_components():
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    # Instantiate the wrapper with the underlying LLM and a log file.
    wllm = Wrapper(llm=llm, log="output.log")

    tools = [PythonAstREPLTool()]
    memory = ConversationBufferMemory(memory_key="chat_history")

    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt: str):
    try:
        wllm, tools, memory = setup_components()
    except Exception as e:
        raise RuntimeError("Failed to set up components") from e

    try:
        agent = create_agent(wllm, tools, memory)
    except Exception as e:
        raise RuntimeError("Failed to create agent") from e

    # ``agent`` is callable and returns a dictionary with an ``output`` key.
    response = agent(prompt)
    return response


if __name__ == "__main__":
    prompt = "What is 5-7?"
    response = run_agent(prompt)
    print(response)
    print(type(response), response.get("output"))
