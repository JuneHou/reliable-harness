import os
from typing import Any, List, Mapping, Optional

from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class WrapperLLM(LLM):
    """A thin wrapper around an existing LangChain LLM.

    The wrapper forwards the call to the underlying LLM **including** the
    ``stop`` argument (which the original implementation ignored).  It also
    logs the prompt/response pair to a file for debugging purposes.
    """

    # The underlying LLM we delegate to.
    llm: OpenAI
    # Path to the log file.
    log: str

    def __init__(self, llm: OpenAI, log: str, **kwargs: Any):
        super().__init__(**kwargs)
        self.llm = llm
        self.log = log

    @property
    def _llm_type(self) -> str:
        # Give the wrapper a distinct type name – this is used only for
        # identification and does not affect functionality.
        return "custom"

    def _call(self, prompt: str, stop: Optional[List[str]] = None) -> str:
        """Forward the prompt to the wrapped LLM.

        Parameters
        ----------
        prompt: str
            The text prompt to send to the model.
        stop: Optional[List[str]]
            Stop tokens – these must be passed through to the underlying model
            otherwise LangChain agents can mis‑interpret the returned string and
            generate duplicate observations.
        """
        # Forward the ``stop`` argument; ``self.llm`` is itself an LLM, so we can
        # call it directly with the same signature.
        reply = self.llm(prompt, stop=stop)

        # Log the interaction for later inspection.
        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100)
        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        # ``identifying_params`` is used by LangChain for caching and tracing.
        # We return a simple representation of the wrapped LLM.
        return {"llm": self.llm}


def setup_components():
    """Create the OpenAI LLM, wrap it, and prepare tools & memory."""
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    wllm = WrapperLLM(llm=llm, log="output.log")

    tools = [PythonAstREPLTool()]
    memory = ConversationBufferMemory(memory_key="chat_history")

    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    """Initialise a Conversational React Description agent with the wrapped LLM."""
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt: str):
    """Run the agent on a single user prompt and return the full response dict."""
    try:
        wllm, tools, memory = setup_components()
    except Exception as e:
        # Propagate the exception for easier debugging during development.
        raise e

    try:
        agent = create_agent(wllm, tools, memory)
    except Exception as e:
        raise e

    # ``agent`` is callable and returns a dict containing keys such as
    # ``output`` and ``intermediate_steps``.
    return agent(prompt)


if __name__ == "__main__":
    prompt = "What is 5-7?"
    response = run_agent(prompt)
    print(response)
    # The LangChain agent returns a dict; ``output`` holds the final answer.
    print(type(response), response.get("output"))
