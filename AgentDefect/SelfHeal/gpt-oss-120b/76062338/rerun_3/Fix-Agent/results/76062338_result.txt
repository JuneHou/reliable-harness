import os
from typing import Any, List, Mapping, Optional

from dotenv import load_dotenv
from langchain.llms.base import LLM
from langchain.llms.openai import OpenAI
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_experimental.tools import PythonAstREPLTool

load_dotenv()


class WrapperLLM(LLM):
    """A thin wrapper around an existing LangChain LLM.

    The wrapper forwards calls to the underlying LLM while optionally logging
    prompts/responses.  It mirrors the behaviour of the wrapped LLM so that the
    agent sees it as a regular OpenAI LLM – this prevents duplicate observations
    that arise when the agent treats a custom ``_llm_type`` as a new model.
    """

    llm: OpenAI
    log: str

    def __init__(self, llm: OpenAI, log: str, **kwargs: Any) -> None:
        super().__init__(**kwargs)
        self.llm = llm
        self.log = log

    @property
    def _llm_type(self) -> str:
        # Return the same type as the wrapped model so the agent handling logic
        # does not branch for a "custom" model.
        return "openai"

    def _call(self, prompt: str, stop: Optional[List[str]] = None) -> str:
        # Forward the call to the underlying LLM.
        reply = self.llm(prompt, stop=stop)
        # Simple file‑based logging (append mode).  Use a context manager for safety.
        with open(self.log, "a", encoding="utf-8") as f:
            f.write("\n{\n" + prompt + "\n}{\n" + reply + "\n}\n" + "." * 100 + "\n")
        return reply

    @property
    def _identifying_params(self) -> Mapping[str, Any]:
        # Include the parameters that uniquely identify the wrapped LLM.
        return {"model_name": self.llm.model_name, "temperature": self.llm.temperature}

    # Optional: expose token counting if the underlying LLM provides it.
    def get_num_tokens(self, text: str) -> int:  # pragma: no cover
        if hasattr(self.llm, "get_num_tokens"):
            return self.llm.get_num_tokens(text)  # type: ignore[attr-defined]
        return super().get_num_tokens(text)


def setup_components():
    llm = OpenAI(
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        temperature=0,
    )

    wllm = WrapperLLM(llm=llm, log="output.log")

    tools = [PythonAstREPLTool()]
    memory = ConversationBufferMemory(memory_key="chat_history")
    return wllm, tools, memory


def create_agent(wllm, tools, memory):
    agent = initialize_agent(
        tools,
        wllm,
        agent="conversational-react-description",
        verbose=True,
        memory=memory,
    )
    return agent


def run_agent(prompt: str):
    try:
        wllm, tools, memory = setup_components()
        agent = create_agent(wllm, tools, memory)
        response = agent(prompt)
        return response
    except Exception as e:
        # In a real application you would log the exception.
        return None


if __name__ == "__main__":
    prompt = "What is 5-7?"
    response = run_agent(prompt)
    if response:
        print(response)
        # ``response`` is a dict with an ``output`` key for the final answer.
        print(type(response), response.get("output"))
    else:
        print("Agent failed to produce a response.")
