import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import StructuredOutputParser, ResponseSchema

load_dotenv()


def get_output_parser():
    """Return a StructuredOutputParser that expects a list of dictionaries.

    Each dictionary in the list should contain two keys:
    - ``identifier``: the identifier string
    - ``text``: the associated text string
    """
    # The parser expects a *single* field whose value is a JSON array of objects.
    # We describe that field in the schema.
    result_schema = ResponseSchema(
        name="result",
        description=(
            "A JSON array (list) where each element is an object with the keys "
            "`identifier` (string) and `text` (string)."
        ),
    )
    return StructuredOutputParser.from_response_schemas([result_schema])


def predict_result(msg: str, PROMPT_TEMPLATE: str):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    # Pass both the message and the format instructions to the template.
    messages = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(messages)

    # ``response.content`` contains the raw LLM output. The parser will turn the
    # JSON string into a Python object (a dict with a ``result`` key).
    parsed = output_parser.parse(response.content)
    # Return just the list of dictionaries for convenience.
    return parsed["result"]
