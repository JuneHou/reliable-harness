import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import StructuredOutputParser, ResponseSchema

load_dotenv()


def get_output_parser() -> StructuredOutputParser:
    """Return a StructuredOutputParser that expects a **list** of dictionaries.

    Each dictionary in the list contains two keys:
    * ``identifier`` – the identifier extracted from the error message
    * ``text`` – the accompanying text for that identifier
    """
    # Define a single response schema whose description tells the model to emit a
    # JSON *array* of objects.  The name ``result`` is arbitrary – we will later
    # pull the list out of the parsed dictionary.
    result_schema = ResponseSchema(
        name="result",
        description=(
            "A JSON array where each element is an object with the keys "
            "'identifier' and 'text' representing a parsed pair from the error "
            "message."
        ),
    )
    return StructuredOutputParser.from_response_schemas([result_schema])


def predict_result(msg: str, PROMPT_TEMPLATE: str):
    """Parse *msg* using the LLM and return a list of ``{"identifier": ..., "text": ...}``.
    """
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    # ``msg`` and the generated ``format_instructions`` are injected into the template.
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    # ``parse`` returns a dict like {"result": [{...}, {...}]}
    parsed = output_parser.parse(response.content)
    # Return only the list for a cleaner API.
    return parsed.get("result", [])


if __name__ == "__main__":
    test_msg = (
        "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),"
        "AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"
    )
    PROMPT_TEMPLATE = """
    You are an android developer.
    Parse this error message and provide **a JSON array** of objects where each object
    contains the fields `identifier` and `text` for every occurrence found in the
    error message.
    --------
    Error message: {msg}
    --------
    {format_instructions}
    """
    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
