import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import StructuredOutputParser, ResponseSchema

load_dotenv()


def get_output_parser():
    """Return a StructuredOutputParser that expects a *list* of dictionaries.

    LangChain's ``StructuredOutputParser`` always returns a dictionary, but we can
    make the top‑level value a list by defining a single field (e.g. ``errors``) whose
    description tells the model to emit a JSON list of objects. The parser will then
    give us ``{"errors": [{...}, {...}]}`` which the caller can treat as the desired
    nested list.
    """
    # The field name that will hold the list of result objects.
    errors_schema = ResponseSchema(
        name="errors",
        description=(
            "A JSON list where each element is a dictionary with the keys "
            "'identifier' and 'text'. "
            "Example: [{\"identifier\": \"id1\", \"text\": \"t1\"}, "
            "{\"identifier\": \"id2\", \"text\": \"t2\"}]"
        ),
    )
    return StructuredOutputParser.from_response_schemas([errors_schema])


def predict_result(msg: str, PROMPT_TEMPLATE: str):
    """Send *msg* to the LLM using the provided prompt template and parse the
    response with the StructuredOutputParser defined above.
    """
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    # Build the prompt. ``ChatPromptTemplate`` automatically injects the variables
    # passed to ``format_messages``.
    prompt = ChatPromptTemplate.from_template(PROMPT_TEMPLATE)
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)

    response = model.invoke(message)

    # ``response.content`` is the raw LLM output – we parse it into a Python dict.
    parsed = output_parser.parse(response.content)
    # ``parsed`` is a dict like {"errors": [{...}, {...}]}; we return the list for
    # convenience.
    return parsed.get("errors", [])


if __name__ == "__main__":
    test_msg = (
        "ObjectNotFoundException AnyOf(AllOf(withId:identifier1, withText:text1),"
        "AllOf(withId:identifier2, withText:text1),AllOf(withId:identifier3, withText:text1))"

    element contains the fields `identifier` and `text` extracted from the message.
    ----------
    Error message: {msg}
    ----------
    {format_instructions}
    """

    result = predict_result(test_msg, PROMPT_TEMPLATE)
    print(result)
