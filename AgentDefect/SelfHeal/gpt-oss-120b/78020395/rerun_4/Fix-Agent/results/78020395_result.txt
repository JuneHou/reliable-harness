import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import StructuredOutputParser, ResponseSchema

load_dotenv()


def get_output_parser():
    """Return a StructuredOutputParser that expects a *list* of dictionaries.

    Each dictionary must contain two keys:
    - ``identifier`` – the missing identifier string
    - ``text`` – the missing text string

    The parser therefore describes a single response schema called ``results``
    whose value is a JSON array of the above objects.  LangChain's
    ``StructuredOutputParser`` will embed this description in the prompt and
    will convert the LLM's JSON response back into a Python list of dicts.
    """
    # The description must be explicit enough for the model to emit the
    # desired JSON structure.
    results_schema = ResponseSchema(
        name="results",
        description=(
            "A JSON array where each element is an object with two fields: "
            "'identifier' (string) – the identifier extracted from the error "
            "message, and 'text' (string) – the corresponding text."
        ),
    )

    output_parser = StructuredOutputParser.from_response_schemas([results_schema])
    return output_parser


def predict_result(msg: str, PROMPT_TEMPLATE: str):
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    # ``format_messages`` creates the final list of messages that will be sent
    # to the LLM.  We pass both the user supplied message and the parser's
    # formatting instructions.
    message = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(message)

    # ``response.content`` contains the raw LLM output.  The parser will turn
    # the JSON string into the desired Python structure (list of dicts).
    response_as_dict = output_parser.parse(response.content)
    return response_as_dict
