import os
from dotenv import load_dotenv
from langchain_openai import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.output_parsers import StructuredOutputParser, ResponseSchema

load_dotenv()


def get_output_parser() -> StructuredOutputParser:
    """Create a parser that expects a *list* of dictionaries.

    Each dictionary must contain two keys:
    - ``identifier`` – the identifier extracted from the error message
    - ``text`` – the corresponding text extracted from the error message

    The parser will therefore return a structure like::

        {"items": [{"identifier": "id1", "text": "txt1"}, ...]}
    """
    list_schema = ResponseSchema(
        name="items",
        description=(
            "A JSON array (list) where each element is a dictionary with two keys: "
            "'identifier' and 'text'. Each dictionary represents one AllOf block "
            "found in the error message."
        ),
    )
    return StructuredOutputParser.from_response_schemas([list_schema])


def predict_result(msg: str, PROMPT_TEMPLATE: str):
    """Send *msg* to the LLM and parse the response with the nested‑list parser.

    Returns the list of dictionaries directly for convenience.
    """
    model = ChatOpenAI(
        model="gpt-4o-mini",
        temperature=0,
        openai_api_key=os.getenv("OPENAI_API_KEY"),
        max_tokens=2000,
    )

    output_parser = get_output_parser()
    format_instructions = output_parser.get_format_instructions()

    # Build the prompt – we inject the user message and the JSON format instructions.
    prompt = ChatPromptTemplate.from_template(template=PROMPT_TEMPLATE)
    messages = prompt.format_messages(msg=msg, format_instructions=format_instructions)
    response = model.invoke(messages)

    # ``response.content`` contains the raw LLM text; the parser turns it into a Python dict.
    parsed = output_parser.parse(response.content)
    # ``parsed`` looks like {"items": [{...}, {...}]}
    return parsed.get("items", [])
