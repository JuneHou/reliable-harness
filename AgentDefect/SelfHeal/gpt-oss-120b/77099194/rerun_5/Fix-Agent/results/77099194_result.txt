from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()

def run_agent(prompt: str) -> str:
    """Run a conversational retrieval over the CSV data.

    The original bug stemmed from using ``k=1`` in the retriever, which limited the
    results to a single document. As a consequence, queries that should match
    multiple rows (e.g., *films from 1999* which have two different authors) only
    returned the top‑ranked document.  The fix is to increase ``k`` to a value that
    comfortably covers the expected number of matches.
    """
    # Initialise the embedding model
    embeddings = OpenAIEmbeddings()

    # Load the CSV into LangChain Document objects
    csv_loader = CSVLoader("movies.csv")
    csv_documents = csv_loader.load()

    # Build a vector store (Chroma) from the documents
    # ``persist_directory`` is optional but useful for debugging / reuse
    vector_store = Chroma.from_documents(csv_documents, embeddings)

    # Use a higher ``k`` so that multiple relevant documents are retrieved.
    # Adjust ``k`` as needed; 10 is a safe default for small CSVs.
    retriever = vector_store.as_retriever(search_kwargs={"k": 10})

    # Create the conversational retrieval chain
    chain = ConversationalRetrievalChain.from_llm(
        ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # Run the chain; an empty list is a valid initial chat history.
    result = chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return result["answer"]

if __name__ == "__main__":
    print(run_agent("which movies are action"))
