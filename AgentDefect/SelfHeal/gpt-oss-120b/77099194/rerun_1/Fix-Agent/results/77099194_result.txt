from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()

def run_agent(prompt: str, k: int = 5) -> str:
    """Run a simple Conversational Retrieval QA over a CSV file.

    Args:
        prompt: The user question.
        k: Number of top‑retrieved documents to feed into the LLM. The default of 5
           ensures that we get more than one matching row (e.g., both author "A"
           and author "C" for 1999 action movies) instead of the previous
           single‑result behaviour.
    """
    # Initialise the embedding model
    embeddings = OpenAIEmbeddings()

    # Load the CSV; each row becomes a Document with the row text as content
    # and the column names stored in ``metadata`` for possible future filtering.
    csvloader = CSVLoader("movies.csv")
    csv_documents = csvloader.load()

    # Build a Chroma vector store from the documents
    vectordb = Chroma.from_documents(csv_documents, embeddings)

    # Retrieve the *k* most similar documents for the given question.
    retriever = vectordb.as_retriever(search_kwargs={"k": k})

    # Set up a conversational retrieval chain. ``return_source_documents`` is
    # kept true so we can see which rows were used in the answer.
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # ``chat_history`` is empty for the first turn.
    result = chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return result["answer"]

if __name__ == "__main__":
    print(run_agent("which movies are action"))
