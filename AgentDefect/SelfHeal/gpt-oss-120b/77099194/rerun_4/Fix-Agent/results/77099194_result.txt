from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()

def run_agent(prompt: str) -> str:
    """Run a conversational retrieval QA against a CSV of movies.

    The original implementation limited the retriever to ``k=1`` which
    caused only the single most‑similar document to be returned. For queries
    that should match several rows – e.g. *all movies from 1999* – this meant
    the answer included only the first match (author "A") instead of the
    full set (authors "A" and "C").

    The fix is to increase the number of retrieved documents.  ``k=10`` is a
    reasonable default for small CSV files; it can be adjusted or even set to
    ``None`` (which returns all matches) depending on the dataset size.
    """
    # Initialise the embedding model (OpenAI embeddings are used here).
    embeddings = OpenAIEmbeddings()

    # Load the CSV – each row becomes a Document with the row content as
    # ``page_content`` and the column names stored in ``metadata``.
    csv_loader = CSVLoader(file_path="movies.csv")
    documents = csv_loader.load()

    # Build a Chroma vector store from the documents.
    vectorstore = Chroma.from_documents(documents, embeddings)

    # Retrieve *more* than one document for a given query.
    retriever = vectorstore.as_retriever(search_kwargs={"k": 10})

    # Set up the conversational retrieval chain.
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # ``chat_history`` should be a list of (human, AI) tuples; an empty list is fine.
    result = chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return result["answer"]

if __name__ == "__main__":
    print(run_agent("which movies are action"))
