from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()

def run_agent(prompt: str) -> str:
    """Run a simple conversational retrieval chain over a CSV file.

    The original implementation limited the retriever to *k=1* which meant only the
    single most‑similar document was passed to the LLM. For queries that should match
    multiple rows (e.g., *"which movies are action"* for movies released in 1999),
    this caused the answer to contain information from only one author.  The fix
    increases the number of retrieved documents (or lets the retriever use its
    default) so the model can see all relevant rows.
    """
    # Initialise embeddings – the OpenAI provider reads the API key from the .env file
    embeddings = OpenAIEmbeddings()

    # Load the CSV – each row becomes a Document with the row contents in page_content
    csv_loader = CSVLoader("movies.csv")
    csv_documents = csv_loader.load()

    # Build a Chroma vector store from the documents
    vectorstore = Chroma.from_documents(csv_documents, embeddings)

    # Use a larger `k` (or omit it) so multiple relevant documents are returned.
    # Here we ask for up to 10 matches; adjust as needed for the size of the dataset.
    retriever = vectorstore.as_retriever(search_kwargs={"k": 10})

    # Conversational retrieval chain – we keep temperature low for deterministic answers
    chain = ConversationalRetrievalChain.from_llm(
        ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # Execute the chain. ``chat_history`` is empty for a single‑turn query.
    result = chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return result["answer"]

if __name__ == "__main__":
    print(run_agent("which movies are action"))
