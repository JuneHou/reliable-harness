from dotenv import load_dotenv

from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.document_loaders.csv_loader import CSVLoader
from langchain.vectorstores import Chroma

from langchain.chains import ConversationalRetrievalChain
from langchain.chat_models import ChatOpenAI

load_dotenv()

def run_agent(prompt: str) -> str:
    """Run a simple conversational retrieval over a CSV of movies.

    The original implementation limited the retriever to ``k=1`` which meant
    only the single most‑relevant document was returned. For queries that should
    match multiple rows (e.g., "movies from 1999" which may have several
    entries) this caused incomplete answers – only the first match (author "A")
    was shown while other valid matches (author "C", etc.) were omitted.

    By increasing ``k`` we allow the retriever to fetch several candidate
    documents. The ConversationalRetrievalChain will then summarise the set,
    providing a more complete answer.
    """
    # Initialise the embedding model (uses OpenAI API key from env)
    embeddings = OpenAIEmbeddings()

    # Load CSV rows as LangChain Document objects
    csv_loader = CSVLoader(file_path="movies.csv")
    documents = csv_loader.load()

    # Create a vector store from the documents
    vectorstore = Chroma.from_documents(documents, embeddings)

    # Retrieve a larger set of relevant documents (adjust ``k`` as needed)
    retriever = vectorstore.as_retriever(search_kwargs={"k": 5})

    # Build the conversational retrieval chain
    chain = ConversationalRetrievalChain.from_llm(
        llm=ChatOpenAI(temperature=0.3),
        retriever=retriever,
        return_source_documents=True,
    )

    # ``chat_history`` expects a list of (question, answer) tuples; an empty list
    # works for a fresh conversation.
    result = chain({"question": prompt, "chat_history": []}, return_only_outputs=True)
    return result["answer"]

if __name__ == "__main__":
    print(run_agent("which movies are action"))
