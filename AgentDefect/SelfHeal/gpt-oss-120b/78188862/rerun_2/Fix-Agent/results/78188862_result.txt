import os
import uuid
from datetime import datetime
from dotenv import load_dotenv
import chromadb
from chromadb.utils import embedding_functions
from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import CharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()

config = {
    "OPENAI_API_KEY": os.getenv("OPENAI_API_KEY")
}

class configs:
    EMBEDDINGS_MODEL = "text-embedding-3-small"


class MockPDFFile:
    """Simple container mimicking an uploaded PDF file."""

    def __init__(self, filepath: str, filename: str):
        self.filepath = filepath
        self.filename = filename


def read_docs(pdf_file: MockPDFFile):
    """Load a PDF and split it into reasonably‑sized chunks.

    Returns a list of ``Document`` objects that LangChain/Chroma understand.
    """
    pdf_loader = PyPDFLoader(pdf_file.filepath)
    pdf_documents = pdf_loader.load()

    # Split each page into chunks of up to 1000 characters with 200 chars overlap
    text_splitter = CharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
    documents = text_splitter.split_documents(pdf_documents)
    return documents


def generate_and_store_embeddings(
    documents,
    pdf_file: MockPDFFile,
    user_id: str,
):
    """Create embeddings for *documents* and persist them in a Chroma collection.

    The function stores **per‑document metadata** (including the source PDF name)
    so that later calls can filter on ``source`` and retrieve only the IDs that
    belong to a particular file.
    """
    # ---------------------------------------------------------------------
    # Initialise a persistent Chroma client/collection
    # ---------------------------------------------------------------------
    client = chromadb.PersistentClient(path="./trained_db")
    collection = client.get_or_create_collection(
        name="PDF_Embeddings",
        embedding_function=embedding_functions.OpenAIEmbeddingFunction(
            api_key=config["OPENAI_API_KEY"], model_name=configs.EMBEDDINGS_MODEL
        ),
    )

    # ---------------------------------------------------------------------
    # Prepare IDs and per‑document metadata
    # ---------------------------------------------------------------------
    now = datetime.now().strftime("%d/%m/%Y %H:%M:%S")
    ids = [str(uuid.uuid4()) for _ in range(len(documents))]

    # Every chunk gets the same base metadata.  You could enrich it further
    # (e.g., add a chunk index) if desired.
    base_metadata = {
        "source": pdf_file.filename,
        "user": str(user_id),
        "created_at": now,
    }
    metadata_list = [base_metadata.copy() for _ in documents]

    # ---------------------------------------------------------------------
    # Create the Chroma vector store from the documents.
    # ---------------------------------------------------------------------
    try:
        vectordb = Chroma.from_documents(
            documents,
            embedding=OpenAIEmbeddings(
                openai_api_key=config["OPENAI_API_KEY"], model=configs.EMBEDDINGS_MODEL
            ),
            persist_directory="./trained_db",
            collection_name=collection.name,
            client=client,
            ids=ids,
            # ``metadata`` is a *list* (one dict per document).  ``collection_metadata``
            # would set metadata that applies to the whole collection, which is not
            # what we need for per‑file filtering.
            metadata=metadata_list,
        )
        # Ensure everything is flushed to disk.
        vectordb.persist()
    except Exception as err:  # pragma: no cover – defensive programming
        print(f"An error occurred while generating embeddings: {err=}, {type(err)=}")
        return {"answer": "An error occured while generating embeddings. Please check terminal for more details."}

    return vectordb


def run_agent(pdf_path: str, user_id: str):
    """High‑level entry point used by the ``__main__`` block.

    Returns a tuple ``(vectorstore, filename)`` or ``(None, filename)`` if the
    embedding step failed.
    """
    pdf_file = MockPDFFile(pdf_path, os.path.basename(pdf_path))

    documents = read_docs(pdf_file)
    print(f"    ✓ Loaded {len(documents)} document chunks")

    vectordb = generate_and_store_embeddings(documents, pdf_file, user_id)
    if isinstance(vectordb, dict):
        # An error dict was returned – signal failure to the caller.
        return None, pdf_file.filename

    return vectordb, pdf_file.filename


if __name__ == "__main__":
    # ---------------------------------------------------------------------
    # Simple demo: pick the first PDF in a ``docs`` directory and embed it.
    # ---------------------------------------------------------------------
    pdf_path = None
    user_id = "test_user_123"

    if os.path.isdir("docs"):
        for file in os.listdir("docs"):
            if file.lower().endswith(".pdf"):
                pdf_path = os.path.join("docs", file)
                break

    if not pdf_path:
        print("No PDF file found in docs folder")
        exit(1)

    vectordb, filename = run_agent(pdf_path, user_id)
    if vectordb is None:
        print("Agent failed to create embeddings")
        exit(1)

    # Retrieve only the entries that belong to the processed PDF.
    result = vectordb.get(where={"source": filename})
    print(f"Total chunks retrieved for '{filename}': {len(result.get('ids', []))}")
