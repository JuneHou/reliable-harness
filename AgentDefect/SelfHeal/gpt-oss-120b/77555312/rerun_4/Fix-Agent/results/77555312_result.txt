import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()


def run_agent(query: str):
    """Load a PDF, split it into non‑overlapping chunks, index the chunks with
    Chroma and retrieve *six* **unique** documents that are most relevant to the
    supplied query.
    
    The original implementation used a large overlap between chunks (200
    tokens). Overlapping chunks produce almost identical embeddings, causing the
    similarity search to return duplicate‑looking chunks. By removing the overlap
    and deduplicating the results we guarantee six distinct pieces of text.
    """
    # Load the PDF (each page is a separate Document by the loader)
    loader = UnstructuredFileLoader('paper2.pdf')
    docs = loader.load()

    # Split the documents into chunks. Setting `chunk_overlap=0` prevents
    # near‑duplicate chunks that would otherwise be returned by the retriever.
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=0,  # <-- No overlap to avoid duplicate content
        add_start_index=True,
    )
    all_splits = text_splitter.split_documents(docs)

    # Build the Chroma vector store.
    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
    )

    # Use Maximal Marginal Relevance (MMR) to increase diversity of the
    # retrieved chunks. This further reduces the chance of near‑duplicates.
    retriever = vectorstore.as_retriever(
        search_type="mmr",  # more diverse results than pure similarity
        search_kwargs={"k": 6, "lambda_mult": 0.5},
    )

    # Retrieve the candidate documents.
    retrieved_docs = retriever.get_relevant_documents(query)

    # Defensive deduplication: ensure we return exactly six *unique* documents.
    unique_docs = []
    seen_contents = set()
    for doc in retrieved_docs:
        # Use the text content as the uniqueness key. In practice you could also
        # combine metadata (e.g., source page) if that better captures uniqueness.
        if doc.page_content not in seen_contents:
            seen_contents.add(doc.page_content)
            unique_docs.append(doc)
        if len(unique_docs) >= 6:
            break

    # If the deduplication step removed some items (rare when using MMR), we
    # fall back to adding additional results until we have six.
    i = 0
    while len(unique_docs) < 6 and i < len(retrieved_docs):
        doc = retrieved_docs[i]
        if doc.page_content not in seen_contents:
            seen_contents.add(doc.page_content)
            unique_docs.append(doc)
        i += 1

    return unique_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    for idx, doc in enumerate(result, 1):
        print(f"--- Document {idx} ---")
        print(doc.page_content[:500])  # print a preview
        print("--- End ---\n")
