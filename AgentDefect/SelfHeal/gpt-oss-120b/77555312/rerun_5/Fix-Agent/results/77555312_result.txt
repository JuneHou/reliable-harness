import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()


def run_agent(query: str):
    """Load a PDF, split it into overlapping chunks, index the chunks with
    Chroma, and return six *unique* documents that are most relevant to the
    supplied query.
    
    The original implementation used a plain similarity search which can
    return many overlapping chunks that contain almost identical text,
    resulting in duplicate documents in the result set.  Two fixes are
    applied:

    1. **Use MMR (maximal marginal relevance) retrieval** – this search type
       favours diversity among the top‑k results, reducing the chance of
       returning near‑duplicate chunks.
    2. **Post‑process the retrieved list to enforce uniqueness** – we keep
       the first occurrence of each distinct ``page_content`` and stop once
       we have six unique documents.
    """
    # Load the PDF file.
    loader = UnstructuredFileLoader("paper2.pdf")
    docs = loader.load()

    # Split the document into overlapping chunks.
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=200,
        add_start_index=True,
    )
    all_splits = text_splitter.split_documents(docs)

    # Create a Chroma vector store from the chunks.
    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
    )

    # Retrieve using MMR to encourage diverse results.
    retriever = vectorstore.as_retriever(
        search_type="mmr",
        search_kwargs={
            "k": 6,            # Desired number of results (may be reduced after dedup).
            "fetch_k": 20,     # Number of candidates to consider for MMR.
            "lambda_mult": 0.5,  # Balance between relevance and diversity.
        },
    )

    # Get the initial set of documents.
    retrieved_docs = retriever.get_relevant_documents(query)

    # Ensure we return six *unique* documents based on their content.
    unique_docs = []
    seen_contents = set()
    for doc in retrieved_docs:
        if doc.page_content not in seen_contents:
            seen_contents.add(doc.page_content)
            unique_docs.append(doc)
        if len(unique_docs) == 6:
            break

    return unique_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    print(result)
