import os
from langchain.llms import OpenAI
import bs4  # noqa: F401  (kept for compatibility if needed elsewhere)
import langchain  # noqa: F401
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

# Load environment variables (e.g., OPENAI_API_KEY)
dotenv.load_dotenv()


def _unique_documents(docs):
    """Return a list of documents with unique page_content while preserving order.

    The similarity search may return overlapping chunks that look identical when
    the ``chunk_overlap`` parameter is used.  By de‑duplicating on the raw text we
    guarantee that the caller receives distinct passages.
    """
    seen = set()
    uniq = []
    for doc in docs:
        if doc.page_content not in seen:
            seen.add(doc.page_content)
            uniq.append(doc)
    return uniq


def run_agent(query: str):
    """Load *paper2.pdf*, split it into non‑overlapping chunks, index the chunks
    with Chroma, and return six *distinct* documents that are most similar to the
    supplied ``query``.
    """
    # ---------------------------------------------------------------------
    # 1️⃣ Load the PDF – UnstructuredFileLoader returns a single Document that
    #    contains the whole file's text.
    # ---------------------------------------------------------------------
    loader = UnstructuredFileLoader("paper2.pdf")
    docs = loader.load()

    # ---------------------------------------------------------------------
    # 2️⃣ Split the document.  Using ``chunk_overlap=0`` prevents the creation of
    #    overlapping windows that would otherwise cause near‑duplicate chunks.
    # ---------------------------------------------------------------------
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=0,  # <‑‑ changed from 200 to avoid duplicate content
        add_start_index=True,
    )
    all_splits = text_splitter.split_documents(docs)

    # ---------------------------------------------------------------------
    # 3️⃣ Build a **fresh** in‑memory Chroma collection.  By not providing a
    #    ``persist_directory`` we guarantee that a new collection is created on
    #    every function call, eliminating stale or duplicate entries that could
    #    have been written to disk in a previous run.
    # ---------------------------------------------------------------------
    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
        # `collection_name` is optional but helps with debugging.
        collection_name="paper2_collection",
    )

    # ---------------------------------------------------------------------
    # 4️⃣ Retrieve the top‑k similar chunks.
    # ---------------------------------------------------------------------
    retriever = vectorstore.as_retriever(
        search_type="similarity",
        search_kwargs={"k": 6},
    )
    retrieved_docs = retriever.get_relevant_documents(query)

    # ---------------------------------------------------------------------
    # 5️⃣ De‑duplicate any accidental repeats and ensure we return *exactly* six
    #    unique documents.  If fewer than six unique chunks exist we simply
    #    return whatever is available.
    # ---------------------------------------------------------------------
    unique_docs = _unique_documents(retrieved_docs)
    return unique_docs[:6]


if __name__ == "__main__":
    # Example usage – replace the query with whatever you need.
    query = "What is Toggle?"
    result = run_agent(query)
    for idx, doc in enumerate(result, start=1):
        print(f"--- Document {idx} ---")
        print(doc.page_content)
        print()
