import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()


def run_agent(query: str):
    """Load a PDF, split it into non‑overlapping chunks, embed the chunks with
    OpenAI, store them in a Chroma vector store and retrieve *six* distinct
    documents that are relevant to the supplied ``query``.

    The original implementation used a large ``chunk_overlap`` which caused
    many chunks to contain almost identical text. When the similarity retriever
    fetched the top‑k results, those overlapping chunks appeared as duplicates.
    By setting ``chunk_overlap=0`` and switching to a Maximal Marginal Relevance
    (MMR) retriever we guarantee that the returned documents are both relevant
    *and* diverse.
    """
    # Load the PDF – UnstructuredFileLoader returns a list with a single
    # Document that holds the whole file content.
    loader = UnstructuredFileLoader("paper2.pdf")
    docs = loader.load()

    # Split the document into chunks.  A chunk size of 1000 characters works
    # well for many LLM contexts; we remove overlap to avoid duplicate text
    # across consecutive chunks.
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=0,
        add_start_index=True,
    )
    all_splits = text_splitter.split_documents(docs)

    # Create (or load) a Chroma vector store from the chunks.
    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
    )

    # Use an MMR retriever to get *k* results that are both relevant and a
    # maximum margin away from each other, which eliminates near‑duplicate
    # chunks.
    retriever = vectorstore.as_retriever(
        search_type="mmr",
        search_kwargs={"k": 6, "lambda_mult": 0.5},
    )

    retrieved_docs = retriever.get_relevant_documents(query)

    # As an extra safety net, de‑duplicate any documents that might still share
    # identical content.  This step is cheap because we only have a handful of
    # results.
    unique_docs = []
    seen_contents = set()
    for doc in retrieved_docs:
        if doc.page_content not in seen_contents:
            seen_contents.add(doc.page_content)
            unique_docs.append(doc)
    # If deduplication reduced the count below six, we simply return whatever
    # unique docs we have – the caller can decide how to handle the shortfall.
    return unique_docs


if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    for i, doc in enumerate(result, 1):
        print(f"--- Document {i} ---")
        print(doc.page_content)
        print()
