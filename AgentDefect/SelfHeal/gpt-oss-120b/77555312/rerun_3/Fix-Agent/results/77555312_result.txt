import os
from langchain.llms import OpenAI
import bs4
import langchain
from langchain import hub
from langchain.document_loaders import UnstructuredFileLoader
from langchain.embeddings import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores import Chroma
import dotenv

dotenv.load_dotenv()

def _dedupe_documents(docs, desired_count):
    """Return up to ``desired_count`` unique documents.

    Uniqueness is determined by a combination of the source file name and the
    start index of the chunk (provided by ``add_start_index=True`` in the
    splitter). This works even when multiple chunks contain very similar text.
    """
    unique_docs = []
    seen = set()
    for doc in docs:
        # ``source`` is added automatically by the loader; ``start_index`` is
        # stored in metadata when ``add_start_index=True``.
        identifier = (
            doc.metadata.get("source"),
            doc.metadata.get("start_index"),
        )
        if identifier not in seen:
            seen.add(identifier)
            unique_docs.append(doc)
        if len(unique_docs) >= desired_count:
            break
    return unique_docs

def run_agent(query, top_k=6):
    # Load the PDF document.
    loader = UnstructuredFileLoader('paper2.pdf')
    docs = loader.load()

    # Split into overlapping chunks and keep the start index for later deduplication.
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=200,
        add_start_index=True,
    )
    all_splits = text_splitter.split_documents(docs)

    # Build the vector store.
    vectorstore = Chroma.from_documents(
        documents=all_splits,
        embedding=OpenAIEmbeddings(),
    )

    # Use Max‑Marginal Relevance (MMR) retrieval to encourage diversity.
    retriever = vectorstore.as_retriever(
        search_type="mmr",
        search_kwargs={"k": top_k * 2},  # fetch more than needed for deduplication
    )

    # Retrieve documents and then deduplicate.
    retrieved = retriever.get_relevant_documents(query)
    unique_docs = _dedupe_documents(retrieved, top_k)

    return unique_docs

if __name__ == "__main__":
    query = "What is Toggle?"
    result = run_agent(query)
    for i, doc in enumerate(result, 1):
        print(f"--- Document {i} (source: {doc.metadata.get('source')}, start: {doc.metadata.get('start_index')}) ---")
        print(doc.page_content[:500])
        print("\n")
