import os
from dotenv import load_dotenv

from langchain_openai import ChatOpenAI
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.vectorstores import FAISS
from langchain.prompts import PromptTemplate
from langchain.chains import LLMChain, RetrievalQAWithSourcesChain

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def run_agent(question: str) -> str:
    """Run a Retrieval QA chain with a custom (blank) prompt.

    Args:
        question: The user query to answer.

    Returns:
        The answer generated by the LLM without the default LangChain prompt template.
    """
    # Initialise LLM
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    # Example documents and metadata
    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    # Create vector store
    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # ---------------------------------------------------------------------
    # Custom prompt – blank template (or you can provide your own format).
    # The default RetrievalQA chain uses a fairly verbose prompt that shows
    # the retrieved context. By supplying our own PromptTemplate we completely
    # control what is sent to the LLM.
    # ---------------------------------------------------------------------
    custom_prompt = PromptTemplate(
        input_variables=["context", "question"],
        template="{question}"  # ignore the retrieved context in the prompt
    )
    combine_chain = LLMChain(llm=llm, prompt=custom_prompt)

    # Build the RetrievalQAWithSourcesChain using the custom combine chain.
    qa_chain = RetrievalQAWithSourcesChain(
        combine_documents_chain=combine_chain,
        retriever=db.as_retriever(),
        return_source_documents=True,
    )

    # Execute the chain
    result = qa_chain({"question": question})
    # result has keys: "answer" and "source_documents"
    answer = result.get("answer")
    return answer


if __name__ == "__main__":
    print(run_agent("Which document mentions France?"))
