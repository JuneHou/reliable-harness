from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings
from langchain.prompts import PromptTemplate
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def run_agent():
    """Create a RetrievalQA chain that uses a minimal, custom prompt.

    The default LangChain prompt (which includes instructions and source
    formatting) is replaced with a blank‑ish template so that only the
    question and retrieved context are sent to the LLM.
    """
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    # Sample documents and their metadata.
    docs = ["This is a test document about France.", "Another document about Germany."]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # Minimal prompt – just forward the question and the retrieved context.
    # Leaving out the default instructional text prevents LangChain from
    # showing its built‑in prompt template.
    custom_prompt = PromptTemplate(
        input_variables=["question", "context"],
        template="{question}\n\n{context}",
    )

    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        # Inject the custom prompt into the underlying combine‑documents chain.
        combine_documents_chain_kwargs={"prompt": custom_prompt},
    )

    return chain


if __name__ == "__main__":
    qa_chain = run_agent()
    # Example query to demonstrate the custom prompt in action.
    print(qa_chain.run("What countries are mentioned in the documents?"))
