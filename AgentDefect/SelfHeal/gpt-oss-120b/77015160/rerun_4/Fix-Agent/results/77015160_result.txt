def run_agent() -> RetrievalQAWithSourcesChain:
    """Create a RetrievalQAWithSourcesChain that uses a custom (blank) prompt.
    The function returns the ready‑to‑use chain.

    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    # Embeddings and vector store
    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # Leaving out any system‑message text prevents LangChain from adding its default template.

        llm=llm,
        chain_type="stuff",
        retriever=db.as_retriever(),
        combine_documents_chain_kwargs={"prompt": custom_prompt},
        return_source_documents=True,
    )
    return chain

import os
from dotenv import load_dotenv

from langchain_openai import ChatOpenAI
from langchain.chains import RetrievalQAWithSourcesChain
from langchain.vectorstores import FAISS
from langchain.embeddings.openai import OpenAIEmbeddings

# Load environment variables (including OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")


def run_agent(query: str):
    """Create a RetrievalQAWithSourcesChain and answer the given query.

    Args:
        query: The question you want to ask the retriever‑augmented LLM.

    Returns:
        A dictionary containing the answer and the source documents.
    """
    # Initialise the LLM
    llm = ChatOpenAI(openai_api_key=api_key, temperature=0)

    # Simple in‑memory documents for the demo
    docs = [
        "This is a test document about France.",
        "Another document about Germany."
    ]
    metadatas = [{"source": "doc1"}, {"source": "doc2"}]

    # Create embeddings and a FAISS vector store
    embeddings = OpenAIEmbeddings(openai_api_key=api_key)
    db = FAISS.from_texts(docs, embeddings, metadatas=metadatas)

    # Build the RetrievalQA chain (the "stuff" chain type is fine for small collections)
    chain = RetrievalQAWithSourcesChain.from_chain_type(
        llm,
        chain_type="stuff",
        retriever=db.as_retriever()
    )

    # Execute the chain with the supplied query
    result = chain({"question": query})
    return result


if __name__ == "__main__":
