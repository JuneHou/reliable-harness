def compute_similarity_pairs(csv_path: str,
                              duplicate_threshold: float = 0.9,
                              similar_threshold: float = 0.75):
    """Read a CSV of controls, compute embeddings for each control's Name + Description,
    and return lists of duplicate and similar pairs based on cosine similarity.

    Args:
        csv_path: Path to the CSV file. Expected columns are 'Name' and 'Description'.
        duplicate_threshold: Cosine similarity >= this value is considered a duplicate.
        similar_threshold: Cosine similarity >= this value (and < duplicate_threshold) is considered similar.

    Returns:
        dict with keys 'duplicates' and 'similar' containing lists of tuples
        (index_i, index_j, similarity). Indexes correspond to the row numbers in the CSV.
    """
    # Load CSV
    df = pd.read_csv(csv_path)
    if not {'Name', 'Description'}.issubset(df.columns):
        raise ValueError("CSV must contain 'Name' and 'Description' columns")

    # Combine fields into a single text for embedding
    df['text'] = df['Name'].astype(str) + ". " + df['Description'].astype(str)
    texts = df['text'].tolist()

    # Compute embeddings
    embedder = OpenAIEmbeddings()
    embeddings = embedder.embed_documents(texts)
    embeddings = np.array(embeddings, dtype=np.float32)

    # Normalize embeddings for cosine similarity
    norm = np.linalg.norm(embeddings, axis=1, keepdims=True)
    embeddings_norm = embeddings / np.where(norm == 0, 1, norm)

    # Cosine similarity matrix
    similarity_matrix = embeddings_norm @ embeddings_norm.T

    duplicates = []
    similar = []
    n = len(df)
    for i in range(n):
        for j in range(i + 1, n):
            sim = float(similarity_matrix[i, j])
            if sim >= duplicate_threshold:
                duplicates.append((i, j, sim))
            elif sim >= similar_threshold:
                similar.append((i, j, sim))

    return {
        "duplicates": duplicates,
        "similar": similar,
        "data_preview": df.head().to_dict(orient="records")  # optional helpful context
    }

import os
from dotenv import load_dotenv
import pandas as pd
import numpy as np
from typing import List, Dict, Tuple

from langchain.embeddings import OpenAIEmbeddings

load_dotenv()


def _embed_texts(texts: List[str], model: OpenAIEmbeddings) -> np.ndarray:
    """Embed a list of texts using the provided OpenAIEmbeddings instance.

    Returns a NumPy array of shape (len(texts), embedding_dim).
    """
    # OpenAIEmbeddings.embed_documents returns a list of lists (vectors)
    vectors = model.embed_documents(texts)
    return np.array(vectors, dtype=np.float32)


def _cosine_similarity_matrix(embeddings: np.ndarray) -> np.ndarray:
    """Compute the cosine similarity matrix for the given embeddings.

    The matrix is symmetric with 1 on the diagonal.
    """
    # Normalize each embedding
    norm = np.linalg.norm(embeddings, axis=1, keepdims=True)
    norm[norm == 0] = 1e-10  # avoid division by zero
    normalized = embeddings / norm
    # Cosine similarity = dot product of normalized vectors
    return np.dot(normalized, normalized.T)


def find_similar_and_duplicate_controls(
    csv_path: str = "control.csv",
    text_columns: Tuple[str, ...] = ("Name", "Description"),
    similarity_threshold: float = 0.85,
    duplicate_threshold: float = 0.99,
) -> List[Dict[str, object]]:
    """Return pairs of controls that are similar or duplicates.

    * **similarity_threshold** – minimum cosine similarity to consider two controls "similar".
    * **duplicate_threshold** – similarity above this value is flagged as a "duplicate" (a stricter

    The function reads ``csv_path`` with pandas, concatenates the supplied ``text_columns`` for each
    row, creates embeddings with ``OpenAIEmbeddings`` and then computes pair‑wise cosine similarity.
    It finally returns a list of dictionaries, each describing a pair:

    ```json
    {
        "control_a": "<Name of first control>",
        "control_b": "<Name of second control>",
        "similarity": 0.92,
        "type": "duplicate"  // or "similar"
    }
    ```
    """
    # Load CSV
    df = pd.read_csv(csv_path)
    if not set(text_columns).issubset(df.columns):
        missing = set(text_columns) - set(df.columns)
        raise ValueError(f"Columns {missing} not found in CSV.")

    # Build a single string per row that will be embedded
    df["_combined_text"] = df[list(text_columns)].astype(str).agg(" ".join, axis=1)

    # Compute embeddings
    embedder = OpenAIEmbeddings()
    embeddings = _embed_texts(texts, embedder)

    # Compute similarity matrix
    sim_matrix = _cosine_similarity_matrix(embeddings)

    results: List[Dict[str, object]] = []

    for i in range(n):

            results.append(
                {
                    "control_a": df.iloc[i]["Name"],
                    "control_b": df.iloc[j]["Name"],
                    "similarity": round(sim, 4),
                    "type": pair_type,
                }
            )
    return results
