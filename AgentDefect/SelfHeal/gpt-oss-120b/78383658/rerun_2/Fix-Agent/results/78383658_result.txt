def _cosine_similarity(a: np.ndarray, b: np.ndarray) -> float:
    """Return cosine similarity between two 1‑D vectors."""
    if np.linalg.norm(a) == 0 or np.linalg.norm(b) == 0:
        return 0.0
    return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))


def _pairwise_similarities(embeddings: List[np.ndarray]) -> List[Tuple[int, int, float]]:
    """Compute the upper‑triangular cosine similarities for a list of embeddings.

    Returns a list of tuples ``(i, j, similarity)`` where ``i < j``.
    """
    results: List[Tuple[int, int, float]] = []
    n = len(embeddings)
    for i in range(n):
        for j in range(i + 1, n):
            sim = _cosine_similarity(embeddings[i], embeddings[j])
            results.append((i, j, sim))
    return results

def _embed_documents(docs: list[Document], embedder: OpenAIEmbeddings) -> np.ndarray:
    """Return a matrix where each row is the embedding of a document.

    The OpenAIEmbeddings class provides an ``embed_documents`` method that
    accepts a list of strings.  We extract ``page_content`` from each Document.
    """
    texts = [doc.page_content for doc in docs]
    # ``embed_documents`` returns a list of lists (float vectors)
    embeddings = embedder.embed_documents(texts)
    return np.array(embeddings, dtype=np.float32)

def _cosine_similarity_matrix(emb_matrix: np.ndarray) -> np.ndarray:
    """Compute a symmetric cosine‑similarity matrix.

    The matrix is ``N x N`` where ``N`` is the number of documents.  Values are
    in the range ``[0, 1]`` because embeddings are non‑negative after the
    normalisation performed by OpenAI's model.
    """
    # Normalise each vector to unit length
    norms = np.linalg.norm(emb_matrix, axis=1, keepdims=True)
    # Guard against zero vectors – replace zero norm with 1 to avoid division by 0
    norms[norms == 0] = 1.0
    normalized = emb_matrix / norms
    # Cosine similarity is simply the dot product of normalised vectors
    return np.dot(normalized, normalized.T)

def _extract_pairs(
    similarity_mat: np.ndarray,
    docs: list[Document],
    similar_thr: float = SIMILARITY_THRESHOLD,
    duplicate_thr: float = DUPLICATE_THRESHOLD,

    Each key maps to a list of dictionaries describing the pair and the
    similarity score.  Pairs are ordered by decreasing similarity.

    similar_pairs = []
    duplicate_pairs = []

    # Iterate over the upper triangle of the matrix (i < j) to avoid duplicates
    for i, j in itertools.combinations(range(n), 2):
        score = float(similarity_mat[i, j])
        if score >= duplicate_thr:
            duplicate_pairs.append({
                "control_1": docs[i].metadata,
                "control_2": docs[j].metadata,
                "similarity": round(score, 4),

                "control_1": docs[i].metadata,
                "control_2": docs[j].metadata,
                "similarity": round(score, 4),
            })

    # Sort each list by similarity descending for readability
    similar_pairs.sort(key=lambda x: x["similarity"], reverse=True)
    duplicate_pairs.sort(key=lambda x: x["similarity"], reverse=True)
