def _embed_texts(texts: list[str]) -> np.ndarray:
    """Return a NumPy array of embeddings for a list of strings.

    The OpenAIEmbeddings class returns a list of lists (or a list of numpy
    arrays).  Converting to ``np.ndarray`` makes later similarity calculations
    easier.
    """

    # embedding vectors (list[float]).
    raw_embeds = embedder.embed_documents(texts)
    return np.array(raw_embeds)


def _find_similar_pairs(
    embeddings: np.ndarray,
    identifiers: list[str],
    similar_thresh: float = 0.80,
    duplicate_thresh: float = 0.95,

    * ``similar`` – pairs whose cosine similarity is >= ``similar_thresh``
    * ``duplicates`` – pairs whose similarity is >= ``duplicate_thresh``

    The function only returns each unordered pair once (i < j).
    """

    sim_matrix = cosine_similarity(embeddings)
    n = len(identifiers)
    similar_pairs: list[tuple[str, str, float]] = []
    duplicate_pairs: list[tuple[str, str, float]] = []

    for i in range(n):

                duplicate_pairs.append((identifiers[i], identifiers[j], round(score, 3)))
            elif score >= similar_thresh:

    return similar_pairs, duplicate_pairs
