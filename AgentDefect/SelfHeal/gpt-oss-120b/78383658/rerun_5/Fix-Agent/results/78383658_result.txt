def _cosine_similarity_matrix(embeds: np.ndarray) -> np.ndarray:
    """Return the cosine similarity matrix for a set of embeddings."""
    # Normalize vectors
    norm = np.linalg.norm(embeds, axis=1, keepdims=True)
    norm_embeds = embeds / np.where(norm == 0, 1, norm)
    # Cosine similarity = dot product of normalized vectors
    return norm_embeds @ norm_embeds.T

import pandas as pd

def _cosine_similarity(vec1: np.ndarray, vec2: np.ndarray) -> float:
    """Return cosine similarity between two 1‑D vectors."""
    if np.linalg.norm(vec1) == 0 or np.linalg.norm(vec2) == 0:
        return 0.0
    return float(np.dot(vec1, vec2) / (np.linalg.norm(vec1) * np.linalg.norm(vec2)))
