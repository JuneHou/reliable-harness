def _cosine_similarity_matrix(vectors: np.ndarray) -> np.ndarray:
    """Return a cosine similarity matrix for the given (n, d) array of vectors.
    The diagonal will be 1.0.
    """
    # Normalize vectors
    norms = np.linalg.norm(vectors, axis=1, keepdims=True)
    # Avoid division by zero
    norms[norms == 0] = 1e-10
    normed = vectors / norms
    # Cosine similarity = dot product of normalized vectors
    return normed @ normed.T

    Returns:

    embeddings = embedder.embed_documents(df[column].tolist())
    return np.array(embeddings)


def _cosine_similarity_matrix(embeddings: np.ndarray) -> np.ndarray:
    """Return the cosine similarity matrix for a set of embeddings.

    The matrix is symmetric with ones on the diagonal.
    """
    # Normalise each vector
    norms = np.linalg.norm(embeddings, axis=1, keepdims=True)
    normed = embeddings / np.where(norms == 0, 1, norms)
    # Cosine similarity = dot product of normalised vectors
    return np.dot(normed, normed.T)

def compute_similarity_matrix(embeddings: np.ndarray) -> np.ndarray:
    """Return cosine similarity matrix for given embeddings.
    Args:
        embeddings: 2D array of shape (n_items, dim)
    Returns:
        similarity matrix of shape (n_items, n_items) with values in [0, 1]
    """
    # Normalize embeddings
    norm = np.linalg.norm(embeddings, axis=1, keepdims=True)
    norm_embeddings = embeddings / np.where(norm == 0, 1, norm)
    # Cosine similarity is dot product of normalized vectors
    return np.dot(norm_embeddings, norm_embeddings.T)
