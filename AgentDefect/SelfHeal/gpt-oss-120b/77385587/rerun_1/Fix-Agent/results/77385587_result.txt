import os
from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import SQLiteStore
from langchain.schema import Document
import nltk

def run_agent(
    query,
    child_db_path="./chroma_db_child",
    parent_store_path="./parent_docstore.sqlite",
):
    """Retrieve relevant parent documents for *query*.

    The function uses a persistent ``Chroma`` vector store for the **child**
    chunks and a persistent ``SQLiteStore`` for the full parent documents.
    Because both stores are persisted to disk, the retriever can be re‑loaded
    in a later run without having to call ``add_documents`` again.
    """

    # Ensure the NLTK tokeniser is available
    nltk.download("punkt", quiet=True)

    # ------------------------------------------------------------------
    # Embedding model – cached locally to avoid redownloading on every run
    # ------------------------------------------------------------------
    hf_embed_path = "./hf_cache"
    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path,
    )

    # ------------------------------------------------------------------
    # Sample documents (the *parent* documents)
    # ------------------------------------------------------------------
    docs = [
        Document(
            page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."
        ),
        Document(
            page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."
        ),
        Document(
            page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context-aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."
        ),
    ]

    # ------------------------------------------------------------------
    # Child splitter – creates the chunks that are stored in the vector DB
    # ------------------------------------------------------------------
    child_splitter = NLTKTextSplitter(chunk_size=400)

    # ------------------------------------------------------------------
    # Persistent vector store for the *child* chunks
    # ------------------------------------------------------------------
    child_vectorstore = Chroma(
        collection_name="child_chunks",
        embedding_function=embedding_function,
        persist_directory=child_db_path,
    )

    # ------------------------------------------------------------------
    # Persistent docstore for the *parent* documents
    # ------------------------------------------------------------------
    docstore = SQLiteStore(parent_store_path)

    # ------------------------------------------------------------------
    # Build (or reload) the ParentDocumentRetriever
    # ------------------------------------------------------------------
    retriever = ParentDocumentRetriever(
        vectorstore=child_vectorstore,
        docstore=docstore,
        child_splitter=child_splitter,
        search_kwargs={"k": 5},
    )

    # --------------------------------------------------------------
    # Add documents only the first time – if the docstore already
    # contains them we skip re‑adding to avoid duplicate entries.
    # --------------------------------------------------------------
    # ``SQLiteStore`` stores documents keyed by their ``metadata['document_id']``.
    # If the user hasn't supplied a custom ID LangChain creates one for us.
    # We can simply try to read any of the ids; an empty result means the store is empty.
    existing = docstore.mget([doc.metadata.get("document_id", "") for doc in docs if doc.metadata.get("document_id")])
    if not existing:
        retriever.add_documents(docs)
        # Persist the newly created child vectors
        child_vectorstore.persist()
        # Commit the SQLite changes
        docstore.commit()

    # --------------------------------------------------------------
    # Retrieve the most relevant *parent* documents for the query.
    # --------------------------------------------------------------
    relevant_parents = retriever.get_relevant_documents(query)
    return relevant_parents

if __name__ == "__main__":
    query = "What is RAG?"
    results = run_agent(query)

        print(f"--- Result {i} ---")
