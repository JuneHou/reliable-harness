def run_agent(query: str, store_location: str = "./store_location"):
    """Create (or load) a persisted ParentDocumentRetriever and perform a search.

    The function persists **both** the child vectorstore (the chunks) *and* the

    nltk.download("punkt", quiet=True)

    # ---------------------------------------------------------------------
    # Embedding model – cached locally
    # ---------------------------------------------------------------------
    hf_embed_path = "./hf_cache"
    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path,
    )

    # ---------------------------------------------------------------------
    # Splitter for the *child* chunks (the pieces that go into the vector DB)
    # ---------------------------------------------------------------------
    child_splitter = NLTKTextSplitter(chunk_size=400)

    # ---------------------------------------------------------------------
    # Initialise (or load) the vector store that holds the child chunks.
    # ``persist_directory`` points to a folder on disk – on the first run the
    # collection will be created, on later runs the existing collection is
    # loaded automatically.
    # ---------------------------------------------------------------------
    vectorstore = Chroma(
        collection_name="child_chunks",
        embedding_function=embedding_function,
        persist_directory=os.path.join(store_location, "chroma_child"),
    )

    # Persistent store for the *parent* documents.  ``LocalFileStore`` writes a
    # JSON‑serialisable representation of the documents to ``store_location``.
    # ---------------------------------------------------------------------

    docstore = LocalFileStore(docstore_path)

        vectorstore=vectorstore,
        docstore=docstore,
        child_splitter=child_splitter,
    )

    # Only add the documents the first time – if the parent store already
    # contains entries we skip to avoid duplicate indexing.
    # ---------------------------------------------------------------------
    if not docstore.mget([doc.metadata.get("document_id") for doc in []]):

            ),

    return relevant_parents

def _initialize_retriever(
    embedding_function,
    vectorstore_dir: str = "./chroma_db_child",
    docstore_path: str = "./retriever_store.sqlite",
) -> ParentDocumentRetriever:
    """Create (or reload) a ParentDocumentRetriever that persists both the
    vectorstore (child chunks) *and* the docstore (full parent documents).

    The function will:
    1. Initialise a ``SQLiteStore`` which writes to ``docstore_path``.
    2. Initialise a ``Chroma`` vectorstore with ``persist_directory`` set to
       ``vectorstore_dir`` and ``client_settings`` that enable persistence.
    3. Attach the vectorstore and docstore to a ``ParentDocumentRetriever``.
    """

    # Persistent docstore – replaces the volatile InMemoryStore.
    docstore = SQLiteStore(sqlite_path=docstore_path)

    # Persistent vectorstore – ``chroma`` writes its data to ``vectorstore_dir``.
    vectorstore = Chroma(
        collection_name="child_chunks",
        embedding_function=embedding_function,
        persist_directory=vectorstore_dir,
    )

    # The child splitter decides how the original documents are chunked.
    child_splitter = NLTKTextSplitter(chunk_size=400)

    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=docstore,
        child_splitter=child_splitter,
        # ``search_type="mmr"`` mirrors the downstream MMR call used in the
        # original script.
        search_type="mmr",
    )
    return retriever


def _add_documents_if_missing(retriever: ParentDocumentRetriever, docs: list[Document]):
    """Add the supplied documents only the first time the store is empty.

    ``ParentDocumentRetriever`` does not expose a direct ``len`` check, but we can
    inspect the underlying docstore. ``SQLiteStore`` stores the full documents
    under the ``"document"`` collection; if that collection is empty we need to
    add the docs.

        retriever.add_documents(docs)
        # Persist the vectorstore (Chroma) – the docstore is already persisted
        # on each write because ``SQLiteStore`` writes immediately.
        retriever.vectorstore.persist()
