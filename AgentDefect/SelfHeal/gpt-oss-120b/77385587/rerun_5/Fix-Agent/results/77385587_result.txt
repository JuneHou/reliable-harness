import os
from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import FileStore
from langchain.schema import Document
import nltk


def run_agent(
    query: str,
    store_location: str = "./docstore",
    chroma_dir: str = "./chroma_db_child",
) -> list[Document]:
    """Retrieve relevant parent documents for *query* using a persisted
    ``ParentDocumentRetriever``.

    The function stores:
    * Full (parent) documents in a :class:`~langchain.storage.FileStore`
    * Child chunks in a :class:`~langchain.vectorstores.Chroma` collection

    On subsequent calls the same stores are loaded from disk, so the
    documents only need to be added once.
    """

    # Ensure NLTK tokenizer is available
    nltk.download("punkt", quiet=True)

    # ---------------------------------------------------------------------
    # 1️⃣ Initialise the embedding model (cached locally)
    # ---------------------------------------------------------------------
    hf_embed_path = "./hf_cache"
    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=hf_embed_path,
    )

    # ---------------------------------------------------------------------
    # 2️⃣ Persistent stores
    # ---------------------------------------------------------------------
    # Child chunks are stored in Chroma – it writes to ``chroma_dir`` on
    # ``persist()``.  The collection name must stay constant between runs.
    vectorstore = Chroma(
        collection_name="child_chunks",
        embedding_function=embedding_function,
        persist_directory=chroma_dir,
    )

    # Parent documents are stored in a simple file‑based store.  ``FileStore``
    # serialises the documents as JSON under ``store_location``.
    docstore = FileStore(store_location)

    # ---------------------------------------------------------------------
    # 3️⃣ Create the retriever – it will read from the persisted stores.
    # ---------------------------------------------------------------------
    child_splitter = NLTKTextSplitter(chunk_size=400)
    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=docstore,
        child_splitter=child_splitter,
    )

    # ---------------------------------------------------------------------
    # 4️⃣ Add documents only the first time.  If the docstore already
    #    contains entries we skip the costly embedding step.
    # ---------------------------------------------------------------------
    if not docstore.mget(["__initialized"]):
        # Sample parent documents
        docs = [
            Document(
                page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."
            ),
            Document(
                page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."
            ),
            Document(
                page_content="RAG most commonly refers to Retrieval‑Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up‑to‑date information (like documents or databases) to generate more accurate, context‑aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."
            ),
        ]
        retriever.add_documents(docs)
        # Persist the Chroma collection so the child chunks survive a restart.
        vectorstore.persist()
        # Mark the store as initialised – a tiny sentinel document.
        docstore.mset({"__initialized": Document(page_content="true")})

    # ---------------------------------------------------------------------
    # 5️⃣ Retrieve relevant parent documents for the query.
    # ---------------------------------------------------------------------
    # ``ParentDocumentRetriever`` returns the *parent* docs, while the
    # underlying vectorstore can be queried directly for child chunks if
    # desired.  Here we return the parent documents for simplicity.
    relevant_parents = retriever.get_relevant_documents(query)
    return relevant_parents
