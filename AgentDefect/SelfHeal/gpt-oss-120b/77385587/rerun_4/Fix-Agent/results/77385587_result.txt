import os
from dotenv import load_dotenv
from langchain.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import NLTKTextSplitter
from langchain.vectorstores import Chroma
from langchain.retrievers import ParentDocumentRetriever
from langchain.storage import SQLiteStore
from langchain.schema import Document
import nltk


def _init_nltk():
    """Download the NLTK tokenizer data once (safe to call repeatedly)."""
    nltk.download("punkt", quiet=True)


def _create_vectorstore(embedding_function, persist_dir):
    """Create (or load) a Chroma vector store that persists its data on disk.

    The `persist_directory` argument tells Chroma where to store the embeddings.
    If the directory already contains a collection, Chroma will load it; otherwise
    it will be created on the first call to `persist()`.
    """
    return Chroma(
        collection_name="child_chunks",
        embedding_function=embedding_function,
        persist_directory=persist_dir,
    )


def _create_docstore(db_path):
    """Create (or load) a SQLiteStore for persisting parent documents.

    SQLiteStore writes its data to a SQLite file on disk, so the mapping between
    child chunk IDs and their parent documents survives interpreter restarts.
    """
    return SQLiteStore(db_path=db_path)


def run_agent(
    query: str,
    *,
    store_location: str = "./store_location",
    embedding_cache: str = "./hf_cache",
    chroma_dir: str = "./chroma_db_child",
) -> list[Document]:
    """Retrieve the *parent* documents that are most relevant to ``query``.

    The function works in two phases:

    1. **Initialisation / first‑run** – if the persistence layers are empty, the
       example documents are added to the ``ParentDocumentRetriever``.  The child
       chunks are stored in a Chroma vector store and the parent‑to‑child mapping
       is stored in a SQLiteStore.
    2. **Retrieval** – on any subsequent call the already‑persisted stores are
       loaded and the retriever returns the parent documents directly.

    Args:
        query: The user query.
        store_location: Directory for the SQLite docstore (``store_location``).
        embedding_cache: Directory used by ``HuggingFaceEmbeddings`` to cache the
            transformer model.
        chroma_dir: Directory used by Chroma to persist the child‑chunk vectors.

    Returns:
        A list of ``Document`` objects representing the most relevant parent
        documents.
    """

    _init_nltk()

    # ---------------------------------------------------------------------
    # 1️⃣  Initialise the embedding model
    # ---------------------------------------------------------------------
    embedding_function = HuggingFaceEmbeddings(
        model_name="BAAI/bge-large-en-v1.5",
        cache_folder=embedding_cache,
    )

    # ---------------------------------------------------------------------
    # 2️⃣  Build (or load) the persistence layers
    # ---------------------------------------------------------------------
    # Child‑chunk vector store – persists embeddings on disk via Chroma.
    vectorstore = _create_vectorstore(embedding_function, persist_directory=chroma_dir)

    # Parent‑document store – persists the mapping from chunk IDs to their
    # original Document objects.  SQLiteStore writes to a file, so we give it a
    # concrete path inside ``store_location``.
    os.makedirs(store_location, exist_ok=True)
    docstore_path = os.path.join(store_location, "parent_docstore.sqlite")
    docstore = _create_docstore(docstore_path)

    # ---------------------------------------------------------------------
    # 3️⃣  Initialise the ParentDocumentRetriever
    # ---------------------------------------------------------------------
    child_splitter = NLTKTextSplitter(chunk_size=400)
    retriever = ParentDocumentRetriever(
        vectorstore=vectorstore,
        docstore=docstore,
        child_splitter=child_splitter,
    )

    # ---------------------------------------------------------------------
    # 4️⃣  If the stores are empty we need to add the example documents.
    # ---------------------------------------------------------------------
    # ``ParentDocumentRetriever`` does not expose a direct "is empty" flag, so we
    # fall back to checking the underlying vector store.  If the collection has
    # no embeddings we assume nothing has been indexed yet.
    if vectorstore._collection.count() == 0:  # type: ignore[attr-defined]
        docs = [
            Document(
                page_content="Artificial Intelligence (AI) is technology that enables computers and machines to simulate human intelligence."
            ),
            Document(
                page_content="Machine learning (ML) is a type of Artificial Intelligence (AI) that teaches computers to find patterns in large datasets and make decisions or predictions without being explicitly programmed for every scenario, learning and improving from experience, much like humans do."
            ),
            Document(
                page_content="RAG most commonly refers to Retrieval-Augmented Generation, an AI technique that improves large language models (LLMs) by letting them access and use external, up-to-date information (like documents or databases) to generate more accurate, context‑aware, and factual answers, preventing hallucinations and tailoring responses to specific knowledge bases without needing costly retraining."
            ),
        ]
        retriever.add_documents(docs)
        # Persist the child‑chunk vectors – Chroma only writes to disk when we call
        # ``persist`` (or when ``persist_directory`` is configured and the process
        # exits, but an explicit call is clearer).
        vectorstore.persist()
        # The SQLiteStore writes automatically on each operation, but we call
        # ``commit`` to guarantee the transaction is flushed.
        docstore.commit()

    # ---------------------------------------------------------------------
    # 5️⃣  Retrieve the *parent* documents for the supplied query.
    # ---------------------------------------------------------------------
    # ``ParentDocumentRetriever.get_relevant_documents`` returns the original
    # parent Document objects, not the individual child chunks.
    relevant_parents = retriever.get_relevant_documents(query)
    return relevant_parents


if __name__ == "__main__":
    user_query = "What is RAG?"
    results = run_agent(user_query)
    for i, doc in enumerate(results, 1):
        print(f"--- Document {i} ---")
        print(doc.page_content)
        print()
