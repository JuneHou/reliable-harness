def _load_pdf(path: str):
    """Load a PDF file and return a list of LangChain Document objects.

    Args:
        path: Path to the PDF file.
    """
    loader = UnstructuredPDFLoader(file_path=path)
    docs = loader.load()
    logging.info("PDF loaded – %d pages found", len(docs))
    if not docs:
        raise ValueError("No content could be extracted from the PDF")
    logging.info("First page preview: %s...", docs[0].page_content[:200].replace("\n", " "))
    return docs


def _split_documents(docs, chunk_size=2000, chunk_overlap=200):
    """Chunk the raw documents into manageable pieces.

    Smaller chunks improve retrieval quality and keep the LLM input size sane.
    """
    splitter = RecursiveCharacterTextSplitter(chunk_size=chunk_size, chunk_overlap=chunk_overlap)
    chunks = splitter.split_documents(docs)
    logging.info("Created %d text chunks (size=%d, overlap=%d)", len(chunks), chunk_size, chunk_overlap)
    return chunks


def _create_vectorstore(chunks, embed_model_name="nomic-embed-text", persist_dir="./chroma_db"):
    """Create a Chroma vector store from the supplied document chunks.

    The function returns a ``Chroma`` instance that can be used as a retriever.
    """
    # ``show_progress`` can cause the embedding call to block in non‑TTY environments.
    embeddings = OllamaEmbeddings(model=embed_model_name)
    logging.info("Ollama embedding model initialised: %s", embed_model_name)

    # Ensure the persistence directory exists – Chroma expects a writable folder.
    Path(persist_dir).mkdir(parents=True, exist_ok=True)

    vectorstore = Chroma.from_documents(
        documents=chunks,
        embedding=embeddings,
        persist_directory=persist_dir,
        collection_name="local-rag",
    )
    # Persist now so that subsequent runs can reload the collection quickly.
    vectorstore.persist()
    logging.info("Vector store created and persisted to %s", persist_dir)
    return vectorstore


def _setup_llm(model_name="llama3"):
    """Instantiate the Ollama chat model.

    ``temperature`` is set to 0 to make the output deterministic – handy for debugging.
    """
    llm = ChatOllama(model=model_name, temperature=0.0)
    logging.info("Chat model loaded: %s", model_name)
    return llm


def run_agent(question: str, pdf_path: str = "./paper1.pdf") -> str | None:
    """End‑to‑end RAG pipeline using local Ollama models.

    The function returns the final answer generated by the LLM or ``None`` if an
    exception occurs.
    """
    try:
        # -------------------------------------------------------------------
        # 1️⃣ Load and chunk the source document
        # -------------------------------------------------------------------
        raw_docs = _load_pdf(pdf_path)
        chunks = _split_documents(raw_docs)

        # -------------------------------------------------------------------
        # 2️⃣ Build the vector store (embedding step)
        # -------------------------------------------------------------------
        vector_db = _create_vectorstore(chunks)
        retriever = vector_db.as_retriever(search_kwargs={"k": 4})

        # -------------------------------------------------------------------
        # 3️⃣ Prepare the LLM and prompts
        # -------------------------------------------------------------------
        llm = _setup_llm()

        # Prompt that formats the retrieved context and the original question.
        rag_template = (
            "Answer the following question **using ONLY the provided context**.\n\n"
            "Context:\n{context}\n\n"
            "Question: {question}\n"
        )
        rag_prompt = ChatPromptTemplate.from_template(rag_template)

        # -------------------------------------------------------------------
        # 4️⃣ Build the retrieval‑augmented generation chain
        # -------------------------------------------------------------------
        # The retriever returns a list of Document objects; we join their text so
        # the prompt receives a single string.
        def format_docs(docs):
            return "\n---\n".join(doc.page_content for doc in docs)

        chain = (
            {"context": retriever | format_docs, "question": RunnablePassthrough()}
            | rag_prompt
            | llm
            | StrOutputParser()
        )

        logging.info("Running chain for question: %s", question)
        response = chain.invoke(question)
        logging.info("Chain completed successfully")
        return response

    except Exception as exc:
        logging.exception("An error occurred while running the RAG agent")
        return None
