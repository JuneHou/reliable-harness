def run_agent(question: str):
    """Run a simple RAG pipeline using local Ollama models.

    The function loads a PDF, splits it into manageable chunks, creates a

    try:

        # Load the PDF
        # ------------------------------------------------------------
        if not local_path:
            raise ValueError("No PDF file path provided")
        loader = UnstructuredPDFLoader(file_path=local_path)
        documents = loader.load()
        logging.info("PDF loaded successfully (%d pages)", len(documents))
        logging.info(
            "First page preview: %s...",
            documents[0].page_content[:200].replace("\n", " "),
        )

        # ------------------------------------------------------------
        # Split into chunks – a smaller chunk size keeps the Ollama
        # embedding model responsive and avoids time‑outs.
        # ------------------------------------------------------------
        splitter = RecursiveCharacterTextSplitter(chunk_size=1500, chunk_overlap=200)
        chunks = splitter.split_documents(documents)
        logging.info("Created %d text chunks", len(chunks))

        # ------------------------------------------------------------
        # Create the vector store using Ollama embeddings.
        # ------------------------------------------------------------
        embedding = OllamaEmbeddings(model="nomic-embed-text", show_progress=True)
        vector_db = Chroma.from_documents(
            documents=chunks,
            embedding=embedding,
            collection_name="local-rag",
            # Optional: persist to disk so that repeated runs are fast.
            persist_directory="./chroma_db",
        )
        logging.info("Vector store created successfully")

        # ------------------------------------------------------------
        # Initialise the LLM (Ollama) that will generate the multi‑queries

        # ------------------------------------------------------------

            input_variables=["question"],
            template=(
                "You are an AI assistant. Generate five different versions of the"
                " user question to improve retrieval. Return each version on a"
                " separate line.\nOriginal question: {question}"
            ),
        )

        # Multi‑Query retriever – it will call the LLM to get alternative
        # queries and then fetch documents for each.
        retriever = MultiQueryRetriever.from_llm(
            vector_db.as_retriever(),
            llm,
            prompt=query_prompt,
        )
        logging.info("Multi‑Query retriever initialised")

        # ------------------------------------------------------------
        # RAG prompt – combines retrieved context with the original question.
        # ------------------------------------------------------------
        rag_template = (
            "Answer the following question **only** using the provided context.\n"
            "If the context does not contain enough information, respond with"
            " \"I don't have enough information to answer.\"\n\n"
            "Context:\n{context}\n\nQuestion: {question}"
        )
        rag_prompt = ChatPromptTemplate.from_template(rag_template)

        # ------------------------------------------------------------
        # Build the chain.  The input to the chain must be a dict containing
        # the key ``question`` because the retriever expects that shape.
        # ------------------------------------------------------------
        chain = (
            {"context": retriever, "question": RunnablePassthrough()}
            | rag_prompt
            | llm
            | StrOutputParser()
        )
        logging.info("Chain built – invoking with question")

        # The chain expects a mapping, not a raw string.
        response = chain.invoke({"question": question})
        logging.info("Chain completed")
        logging.info("Response: %s", response)
        return response
