import logging
from langchain_community.document_loaders import UnstructuredPDFLoader
from langchain_community.embeddings import OllamaEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain.prompts import ChatPromptTemplate, PromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_community.chat_models import ChatOllama
from langchain_core.runnables import RunnablePassthrough
from langchain.retrievers.multi_query import MultiQueryRetriever

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s - %(levelname)s - %(message)s",
)


def run_agent(question: str):
    """Run a simple RAG pipeline using Ollama for both embeddings and LLM.

    The function loads a PDF, creates a Chroma vector store from the PDF chunks,
    builds a Multi‑Query retriever, and finally answers the user question.
    """
    local_path = "./paper1.pdf"
    try:
        # ------------------------------------------------------------------
        # Load PDF
        # ------------------------------------------------------------------
        if not local_path:
            raise ValueError("No PDF file provided")
        loader = UnstructuredPDFLoader(file_path=local_path)
        documents = loader.load()
        logging.info("PDF loaded – %s pages", len(documents))
        logging.info(
            "First page preview: %s...",
            documents[0].page_content[:200].replace("\n", " "),
        )

        # ------------------------------------------------------------------
        # Chunk the documents – keep the chunk size moderate for faster embed.
        # ------------------------------------------------------------------
        splitter = RecursiveCharacterTextSplitter(chunk_size=2000, chunk_overlap=200)
        chunks = splitter.split_documents(documents)
        logging.info("Created %s text chunks", len(chunks))

        # ------------------------------------------------------------------
        # Create the vector store.  ``show_progress=False`` avoids a blocking UI
        # when Ollama pulls the model for the first time.
        # ------------------------------------------------------------------
        embedding = OllamaEmbeddings(model="nomic-embed-text", show_progress=False)
        vector_store = Chroma.from_documents(
            documents=chunks,
            embedding=embedding,
            collection_name="local-rag",
            persist_directory="./chroma_db",  # persist so we don't rebuild each run
        )
        logging.info("Vector store created successfully")

        # ------------------------------------------------------------------
        # Initialise the LLM – set ``streaming=False`` so the call returns a
        # complete response rather than an endless generator.
        # ------------------------------------------------------------------
        llm = ChatOllama(model="llama3", temperature=0.0, streaming=False)
        logging.info("LLM initialised – model=%s", "llama3")

        # ------------------------------------------------------------------
        # Prompt to generate alternative queries for the Multi‑Query retriever.
        # ------------------------------------------------------------------
        query_prompt = PromptTemplate(
            input_variables=["question"],
            template=(
                "You are an AI assistant. Generate **five** distinct versions of the"
                " following user question. Return each version on a separate line.\n"
                "Original question: {question}"
            ),
        )

        # ------------------------------------------------------------------
        # Multi‑Query Retriever – it will use the LLM to create several queries
        # and then retrieve the most relevant chunks.
        # ------------------------------------------------------------------
        retriever = MultiQueryRetriever.from_llm(
            vector_store.as_retriever(), llm, prompt=query_prompt
        )
        logging.info("Multi‑Query retriever ready")

        # ------------------------------------------------------------------
        # RAG prompt – we only want the answer based on the retrieved context.
        # ------------------------------------------------------------------
        rag_template = (
            "Answer the question **using only** the supplied context.\n\n"
            "Context:\n{context}\n\nQuestion: {question}\n"
        )
        rag_prompt = ChatPromptTemplate.from_template(rag_template)

        # Build the chain: retrieve → prompt → LLM → parse string output
        chain = (
            {"context": retriever, "question": RunnablePassthrough()}
            | rag_prompt
            | llm
            | StrOutputParser()
        )
        logging.info("RAG chain assembled")

        # ------------------------------------------------------------------
        # Invoke the chain.  ``invoke`` expects a dict matching the keys used in
        # the first step of the chain.
        # ------------------------------------------------------------------
        response = chain.invoke({"question": question})
        logging.info("Chain invoked – got response")
        logging.info("Response: %s", response)
        return response

    except Exception as exc:  # pragma: no cover – defensive logging
        logging.exception("An error occurred while running the RAG pipeline")
        return None


if __name__ == "__main__":
    test_question = "What are the 5 pillars of global cooperation?"
    answer = run_agent(test_question)
    print("\nFinal Answer:\n", answer)
