import logging
from langchain_community.document_loaders import UnstructuredPDFLoader
from langchain_community.embeddings import OllamaEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain.prompts import ChatPromptTemplate, PromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_community.chat_models import ChatOllama
from langchain_core.runnables import RunnablePassthrough, RunnableLambda

logging.basicConfig(level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s")


def format_docs(docs):
    """Join a list of Document objects into a single string context."""
    return "\n".join([doc.page_content for doc in docs])


def run_agent(question: str):
    local_path = "./paper1.pdf"
    try:
        # Load PDF
        loader = UnstructuredPDFLoader(file_path=local_path)
        data = loader.load()
        logging.info("PDF loaded successfully")
        logging.info(f"First page preview: {data[0].page_content[:500]}...")

        # Split into chunks
        text_splitter = RecursiveCharacterTextSplitter(chunk_size=7500, chunk_overlap=100)
        chunks = text_splitter.split_documents(data)
        logging.info(f"Created {len(chunks)} text chunks")

        # Create embeddings and vector store
        embedding_model = OllamaEmbeddings(model="nomic-embed-text", show_progress=False)
        vector_db = Chroma.from_documents(
            documents=chunks,
            embedding=embedding_model,
            collection_name="local-rag",
        )
        logging.info("Vector store created")

        # Initialise LLM (chat model) for answering
        llm = ChatOllama(model="llama3")
        logging.info("Chat model loaded")

        # Simple retriever (no multi‑query to avoid hanging)
        retriever = vector_db.as_retriever(search_kwargs={"k": 4})
        logging.info("Retriever initialised")

        # RAG prompt
        rag_template = """Answer the question based ONLY on the following context:\n{context}\n\nQuestion: {question}"""
        rag_prompt = ChatPromptTemplate.from_template(rag_template)
        logging.info("RAG prompt created")

        # Build chain: retrieve docs → format → prompt → LLM → parse
        chain = (
            {"context": retriever | RunnableLambda(format_docs), "question": RunnablePassthrough()}
            | rag_prompt
            | llm
            | StrOutputParser()
        )
        logging.info("Chain constructed, invoking...")

        response = chain.invoke(question)
        logging.info("Chain invocation complete")
        logging.info(f"Response: {response}")
        return response

    except Exception as e:
        logging.error(f"An error occurred: {e}")
        return None


if __name__ == "__main__":
    q = "What are the 5 pillars of global cooperation?"
    ans = run_agent(q)
    print("\nFinal Output:", ans)
