import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent, AgentExecutor
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool

load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    """A simple calculator tool that evaluates a Python arithmetic expression."""

    name = "calculator"
    description = "Performs basic arithmetic. Input should be a valid Python arithmetic expression."

    def _run(self, query: str) -> str:
        # Evaluate the arithmetic expression and return the result as a string.
        # NOTE: ``eval`` is used here for demonstration purposes only – in a real
        # production setting you would want a safe parser/evaluator.
        return str(eval(query))

    async def _arun(self, query: str) -> str:  # pragma: no cover
        raise NotImplementedError()


def create_agent() -> AgentExecutor:
    """Create a LangChain Conversational React Agent with the Calculator tool.

    The agent is configured with a small ``max_iterations`` value because the
    desired behaviour is to stop immediately after the tool is invoked.  The
    ``early_stopping_method`` is set to ``"force_stop"`` which tells the executor
    to cease execution after the tool finishes rather than trying to generate a
    natural‑language response.
    """
    llm = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    return initialize_agent(
        tools=[CalculatorTool()],
        llm=llm,
        agent="chat-conversational-react-description",
        verbose=False,
        max_iterations=2,
        early_stopping_method="force_stop",  # stop after tool use
        memory=memory,
    )


def run_agent(prompt: str) -> str:
    """Run the agent and return **only** the raw output from the Calculator tool.

    The ``AgentExecutor`` normally returns the LLM‑generated final answer.  To
    obtain the raw tool result we request ``return_intermediate_steps=True`` and
    extract the latest tool result from the intermediate steps.
    """
    agent = create_agent()
    # ``agent`` is callable; it returns either a string or a tuple when
    # ``return_intermediate_steps`` is True.
    result, intermediate_steps = agent(prompt, return_intermediate_steps=True)
    if intermediate_steps:
        # The second element of each tuple is the tool's raw output.
        _, tool_output = intermediate_steps[-1]
        return tool_output
    # Fallback: if no tool was used, return the LLM's answer.
    return result
