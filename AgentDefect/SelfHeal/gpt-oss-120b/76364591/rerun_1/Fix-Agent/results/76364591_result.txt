import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool

load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    """A simple calculator tool that returns the raw result of an arithmetic expression.

    By setting ``return_direct = True`` the agent will immediately return the tool's
    output without any additional LLM‑generated wrapping or further reasoning steps.
    """

    name = "calculator"
    description = "Performs basic arithmetic"
    return_direct = True  # <-- ensures the raw tool output is returned directly

    def _run(self, query: str) -> str:
        # NOTE: ``eval`` is used here for simplicity; in production code you should
        # replace it with a safe arithmetic parser.
        return str(eval(query))

    async def _arun(self, query: str) -> str:  # pragma: no cover
        raise NotImplementedError()


def create_agent():
    llm_gpt4 = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    # ``return_direct`` on the tool makes the extra iterations unnecessary, but we keep
    # the defaults for safety.
    return initialize_agent(
        agent="chat-conversational-react-description",
        tools=[CalculatorTool()],
        llm=llm_gpt4,
        verbose=True,
        # The agent will stop as soon as the CalculatorTool returns a result because of
        # ``return_direct=True``.
        memory=memory,
    )


def run_agent(prompt: str) -> str:
    agent = create_agent()
    return agent.run(prompt)
