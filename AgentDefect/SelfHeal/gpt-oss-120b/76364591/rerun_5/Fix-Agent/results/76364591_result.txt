import os
from dotenv import load_dotenv
from langchain.agents import create_openai_functions_agent, AgentExecutor
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool

load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    """A simple calculator tool that evaluates a Python arithmetic expression.

    The tool returns the *raw* result of the evaluation as a string.  No additional
    formatting is added by the LLM – the agent will stop after this tool is called
    and return the tool's output directly.
    """

    name = "calculator"
    description = "Performs basic arithmetic. Input should be a valid Python arithmetic expression."

    def _run(self, query: str) -> str:
        # NOTE: ``eval`` is used here purely for demonstration purposes. In a real
        # application you should replace this with a safe parser/evaluator.
        return str(eval(query))

    async def _arun(self, query: str) -> str:
        raise NotImplementedError()


def create_agent() -> AgentExecutor:
    """Create an OpenAI Functions agent.

    Using the ``create_openai_functions_agent`` helper automatically configures the
    agent to employ function (tool) calling.  The agent will invoke the ``calculator``
    tool when it recognises an arithmetic request and will immediately return the
    tool's output without any extra LLM‑generated text.
    """
    llm = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(memory_key="chat_history", return_messages=True)
    tools = [CalculatorTool()]
    # ``create_openai_functions_agent`` sets ``early_stopping_method="force"`` under the hood,
    # which halts the agent as soon as a tool finishes execution.
    return create_openai_functions_agent(llm=llm, tools=tools, memory=memory)


def run_agent(prompt: str) -> str:
    """Run the agent on a user prompt and return the raw tool output.

    Because the agent uses OpenAI function calling, the returned string is exactly
    what the ``CalculatorTool`` returned.
    """
    agent = create_agent()
    return agent.run(prompt)
