import os
from dotenv import load_dotenv
from langchain.agents import initialize_agent
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI
from langchain.tools import BaseTool

load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")


class CalculatorTool(BaseTool):
    """Simple arithmetic calculator tool used by the agent."""

    name = "calculator"
    description = "Performs basic arithmetic"

    def _run(self, query: str) -> str:
        # NOTE: ``eval`` is used only for demonstration purposes; in production
        # you should replace it with a safe arithmetic parser.
        return str(eval(query))

    async def _arun(self, query: str) -> str:
        raise NotImplementedError()


def create_agent():
    """Create a conversational React‑style agent equipped with the calculator.

    The agent is configured to stop after the first tool execution.  The caller
    (``run_agent``) extracts the raw tool result and returns it directly, without
    any LLM‑generated wrapping.
    """
    llm_gpt4 = ChatOpenAI(model="gpt-4", temperature=0)
    memory = ConversationBufferMemory(
        memory_key="chat_history", return_messages=True
    )
    # ``max_iterations`` is set to 2 so the agent has one turn to decide on the
    # tool and a second turn to execute it.  ``early_stopping_method`` "generate"
    # ensures the agent stops after the tool run if the LLM does not request any
    # further actions.
    return initialize_agent(
        agent="chat-conversational-react-description",
        tools=[CalculatorTool()],
        llm=llm_gpt4,
        verbose=False,
        max_iterations=2,
        early_stopping_method="generate",
        memory=memory,
    )


def run_agent(prompt: str) -> str:
    """Run the agent and return **only** the raw calculator output.

    The ``AgentExecutor`` returned by ``initialize_agent`` can be called as a
    function with ``return_intermediate_steps=True``.  The intermediate steps
    contain tuples of ``(tool, tool_output)``.  Because we only have a single
    tool, the last entry in ``intermediate_steps`` holds the calculator result.
    If, for any reason, the agent does not invoke the tool, we fall back to the
    normal LLM response.
    """
    agent = create_agent()
    # The executor expects a dictionary with the key "input".
    result = agent({"input": prompt}, return_intermediate_steps=True)
    # ``result`` is a dict with keys "output" and "intermediate_steps".
    intermediate = result.get("intermediate_steps")
    if intermediate:
        # The tool output is the second element of the last tuple.
        _, tool_output = intermediate[-1]
        return tool_output.strip()
    # Fallback – return whatever the LLM produced.
    return result.get("output", "")
