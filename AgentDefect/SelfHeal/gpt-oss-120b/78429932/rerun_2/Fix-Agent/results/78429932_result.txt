from langchain_community.chat_models import ChatOllama
from langchain.schema import SystemMessage, HumanMessage

def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Generate a single, clean response using Ollama's Llama 3 model.

    Args:
        user_prompt: The query or instruction from the user.
        system_prompt: The system‑level instruction that sets the assistant's role.

    Returns:
        The assistant's reply as a plain string.
    """
    # Initialise the chat‑oriented Ollama model. ``ChatOllama`` knows how to format
    # the messages for Llama 3, so we don't have to prepend special tokens ourselves.
    llm = ChatOllama(model="llama3")

    # Construct the chat history. LangChain expects a list of Message objects.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # ``invoke`` returns a ``AIMessage``; we take its ``content`` attribute.
    response_message = llm.invoke(messages)
    return response_message.content
