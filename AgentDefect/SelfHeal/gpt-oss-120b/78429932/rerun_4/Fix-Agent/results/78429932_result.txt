from langchain_community.chat_models import ChatOllama
from langchain.schema import HumanMessage, SystemMessage


def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Generate a single clean assistant response using Ollama's Llama 3.

    Args:
        user_prompt: The question or instruction from the user.
        system_prompt: The system‑level instruction that sets the assistant's behavior.

    Returns:
        The assistant's reply as a plain string.
    """
    # Initialise the chat model – it handles the required token format internally.
    llm = ChatOllama(model="llama3")

    # Build the chat message list. LangChain will translate this into the correct
    # Llama 3 chat format (including <|begin_of_text|>, header tags, etc.).
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # Invoke the model and extract the textual content from the response object.
    response = llm.invoke(messages)
    return response.content if hasattr(response, "content") else str(response)
