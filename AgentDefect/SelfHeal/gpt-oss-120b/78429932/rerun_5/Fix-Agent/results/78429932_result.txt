from langchain_community.chat_models import OllamaChat
from langchain.schema import SystemMessage, HumanMessage

def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Generate a single clean response from an Ollama Llama 3 model.

    Args:
        user_prompt: The question or instruction from the user.
        system_prompt: The system instruction that sets the assistant's behavior.

    Returns:
        The assistant's response as a plain string.
    """
    # Initialise the Ollama chat model. "llama3" is the model name used by Ollama.
    chat = OllamaChat(model="llama3")

    # Build the list of messages in the format the Ollama chat endpoint expects.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # Invoke the model and extract the assistant's reply.
    response = chat.invoke(messages)
    # ``response`` is a ``ChatMessage``; its ``content`` attribute holds the text.
    return response.content

if __name__ == "__main__":

    )
    system_instruction = "You are a helpful assistant."
