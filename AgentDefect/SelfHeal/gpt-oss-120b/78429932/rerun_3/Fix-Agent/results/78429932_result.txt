from langchain_community.chat_models import ChatOllama
from langchain_core.messages import SystemMessage, HumanMessage

def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Generate a single, clean assistant response using Ollama + Llama 3.

    Args:
        user_prompt: The user's question or instruction.
        system_prompt: The system instruction that sets the assistant's behavior.
    Returns:
        The assistant's response as a plain string (no extra headers or duplication).
    """
    # Initialise the chat model – Ollama's Llama 3 works in chat mode.
    chat = ChatOllama(model="llama3")

    # Build the chat message list. LangChain will handle the token‑level prompting
    # and will strip the internal <|...|> markers, returning only the content.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # `invoke` returns a `AIMessage`; its `content` attribute holds the clean text.
    response = chat.invoke(messages)
    return response.content
