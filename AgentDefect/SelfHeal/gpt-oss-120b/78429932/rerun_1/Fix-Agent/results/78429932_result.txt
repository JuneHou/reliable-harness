from langchain_community.chat_models import ChatOllama
from langchain.schema import HumanMessage, SystemMessage

def run_agent(user_prompt: str, system_prompt: str) -> str:
    """Generate a single assistant response using Ollama's Llama 3 model.

    Args:
        user_prompt: The user's query.
        system_prompt: Instructions for the assistant.

    Returns:
        The assistant's response as a plain string.
    """
    # Initialise the chat model.  `ChatOllama` knows how to handle the
    # chat‑style message format required by Llama 3, so we don't need to craft
    # the special <|...|> tokens ourselves.
    chat = ChatOllama(model="llama3")

    # Build the message list in the order expected by the model.
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=user_prompt),
    ]

    # Invoke the model and extract the textual content from the response.
    response = chat.invoke(messages)
    return response.content
