import os
from dotenv import load_dotenv

from langchain.memory import ConversationBufferMemory
from langchain.agents import AgentType, initialize_agent, load_tools
from langchain_openai import ChatOpenAI

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Initialise the LLM (Chat model)
llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

# Load any tools you want the agent to have access to
tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# ------------------------------------------------------------
# MEMORY SET‑UP
# ------------------------------------------------------------
# For the CHAT_ZERO_SHOT_REACT_DESCRIPTION agent the expected key for
# the chat history is "chat_history".  The ConversationBufferMemory
# must be configured with that key *and* with the input key (the name of
# the variable that holds the user's question).  Explicitly passing the
# input_key avoids the rare case where the default "input" does not line
# up with the agent's prompt template.
memory = ConversationBufferMemory(
    memory_key="chat_history",
    input_key="input",
    return_messages=True,  # Store the full Message objects for better context
)

# ------------------------------------------------------------
# AGENT INITIALISATION
# ------------------------------------------------------------
# Pass the memory object to initialise_agent.  The chat‑react agent will
# automatically prepend the stored chat history to every new turn,
# allowing it to answer follow‑up questions like "what was the last
# question I asked?".
agent = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION,
    memory=memory,
    handle_parsing_errors=True,
    verbose=False,
)

def run_agent(prompt: str) -> str:
    """Run the agent with a single user prompt.

    The agent retains conversation memory between calls, so consecutive
    invocations can refer to earlier interactions.
    """
    return agent.run(prompt)

if __name__ == "__main__":
    # First question – the agent will look up the capital of France.
    result1 = run_agent("what is the capital of France?")
    print(result1)

    # Second question – thanks to the ConversationBufferMemory the agent
    # can recall the previous user input.
    result2 = run_agent("what is the last question I asked?")
    print(result2)
