from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Initialise the Chat model (GPT‑4 in this example)
llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

# Load the desired tools – the LLM‑math tool and Wikipedia lookup in this case
tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Set up a conversational memory buffer. The default memory_key for the
# conversational‑react agents is "chat_history", which matches the key we set
# here.
memory = ConversationBufferMemory(memory_key="chat_history")

# NOTE:
# The original code used AgentType.CHAT_ZERO_SHOT_REACT_DESCRIPTION.  That agent
# type does **not** read from or write to a memory object – it treats each call
# as an independent, stateless interaction.  To enable the agent to remember the
# previous user inputs we must use a *conversational* variant of the React
# agent.  For chat‑based LLMs the appropriate configuration is
# AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION.
agent_test = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    handle_parsing_errors=True,
    memory=memory,
    verbose=False,
)

def run_agent(prompt: str) -> str:
    """Run the agent with a user prompt and return the response string."""
    return agent_test.run(prompt)

if __name__ == "__main__":
    # First query – the answer is fetched via the Wikipedia tool.
    result1 = run_agent("what is the capital of France?")
    print(result1)

    # Second query – the agent now has access to the conversation buffer and can
    # correctly reference the prior question.
    result2 = run_agent("what is the last question I asked?")
    print(result2)
