import os
from dotenv import load_dotenv

from langchain.agents import AgentType, initialize_agent, load_tools
from langchain.memory import ConversationBufferMemory
from langchain_openai import ChatOpenAI

# Load environment variables (API key)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Initialize the LLM (Chat model)
llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

# Load tools that the agent can use
tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Set up conversation memory – the buffer will be stored under the key "chat_history"
memory = ConversationBufferMemory(memory_key="chat_history")

# NOTE:
# The `CHAT_ZERO_SHOT_REACT_DESCRIPTION` agent does **not** incorporate memory into its prompt.
# To have the agent remember previous turns we need to use an agent type that supports
# memory, such as `CONVERSATIONAL_REACT_DESCRIPTION`.
agent = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CONVERSATIONAL_REACT_DESCRIPTION,
    memory=memory,
    handle_parsing_errors=True,
    verbose=False,
)


def run_agent(prompt: str) -> str:
    """Run a single turn of the agent.

    The agent automatically adds the current `chat_history` from the memory to the
    prompt, so subsequent calls have access to earlier interactions.
    """
    return agent.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("what is the capital of France?")
    print(result1)

    # The second question should now be answered based on the remembered conversation.
    result2 = run_agent("what is the last question I asked?")
    print(result2)
