from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.agents import load_tools
from langchain.memory import ConversationBufferMemory
from dotenv import load_dotenv
import os

# Load environment variables (e.g., OPENAI_API_KEY)
load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Initialise the chat model (GPT‑4)
llm = ChatOpenAI(
    model_name="gpt-4",
    temperature=0,
    openai_api_key=api_key,
)

# Load the tools we want the agent to have access to
tools = load_tools(["llm-math", "wikipedia"], llm=llm)

# Conversation memory – this stores the full chat history and is injected into the prompt
memory = ConversationBufferMemory(memory_key="chat_history")

# NOTE:
# `CHAT_ZERO_SHOT_REACT_DESCRIPTION` does **not** make use of memory – it rewrites the
# prompt for each turn without persisting the prior interaction.  To obtain a reactive
# agent that can recall earlier messages you need the *conversational* variant:
# `CHAT_CONVERSATIONAL_REACT_DESCRIPTION`.
# This agent type adds a `{chat_history}` placeholder to the system prompt and combines
# it with the memory contents, enabling the second query ("what is the last question I asked?")
# to be answered correctly.

agent = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.CHAT_CONVERSATIONAL_REACT_DESCRIPTION,
    memory=memory,
    handle_parsing_errors=True,
    verbose=False,
)


def run_agent(prompt: str) -> str:
    """Run the agent with a user prompt and return the text response."""
    return agent.run(prompt)


if __name__ == "__main__":
    result1 = run_agent("What is the capital of France?")
    print("Result 1:", result1)

    # The second call now has access to the previous interaction via memory
    result2 = run_agent("What was the last question I asked?")
    print("Result 2:", result2)
