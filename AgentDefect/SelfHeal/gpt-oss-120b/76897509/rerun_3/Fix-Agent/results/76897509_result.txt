import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables and configure LLM cache
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# Pool of OpenAI API keys – you can extend this list if you have multiple keys
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]


def get_random_key(key_pool: list[str]) -> str:
    """Return a random, non‑empty API key from the pool."""
    return random.choice([k for k in key_pool if k])


def run_agent(query: str, pdf_paths: list[str]):
    """Run a ConversationalRetrievalChain over the supplied PDFs.

    The function builds a Chroma vector store from the PDFs, creates a
    ConversationalRetrievalChain, and then queries it while applying a
    *dynamic* metadata filter so that only documents whose ``source`` metadata
    matches the desired PDF are considered.
    """
    # ---------------------------------------------------------------------
    # 1️⃣ Load PDFs, split them and attach a ``source`` metadata field.
    # ---------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        docs = loader.load()
        splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        chunks = splitter.split_documents(docs)
        # Attach the source filename to each chunk – this is the metadata we will filter on.
        for chunk in chunks:
            chunk.metadata["source"] = pdf_path
        all_chunks.extend(chunks)

    # ---------------------------------------------------------------------
    # 2️⃣ Create the Chroma vector store.
    # ---------------------------------------------------------------------
    db = Chroma.from_documents(all_chunks, embeddings)

    # ---------------------------------------------------------------------
    # 3️⃣ Build the ConversationalRetrievalChain.
    #    We use the default VectorStoreRetriever – it forwards any ``filter``
    #    argument provided at query‑time to the underlying vector store.
    # ---------------------------------------------------------------------
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(
            openai_api_key=get_random_key(OPENAI_API_KEY_POOL),
            temperature=0,
            cache=True,
        ),
        retriever=db.as_retriever(),
        verbose=True,
        return_source_documents=True,
    )

    # ---------------------------------------------------------------------
    # 4️⃣ Define a *dynamic* metadata filter. Only chunks coming from
    #    ``paper2.pdf`` will be considered for this particular query.
    # ---------------------------------------------------------------------
    metadata_filter = {"source": "paper2.pdf"}

    # ---------------------------------------------------------------------
    # 5️⃣ Invoke the chain – note the key name ``metadata_filter``. LangChain
    #    expects this exact name to forward the filter to the retriever.
    # ---------------------------------------------------------------------
    result = qa_chain({
        "question": query,
        "chat_history": [],
        "metadata_filter": metadata_filter,
    })

    return result


if __name__ == "__main__":
    query = "What is Theia?"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    answer = run_agent(query, pdf_paths)
    print(answer["answer"])
