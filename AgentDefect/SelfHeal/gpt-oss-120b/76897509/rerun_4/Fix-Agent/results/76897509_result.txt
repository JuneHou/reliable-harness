import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables and set an in‑memory LLM cache for faster repeated calls
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# ---------------------------------------------------------------------------
# Helper – pick a random, non‑None OpenAI key from the pool (useful when you
# have several keys to rotate through).  In a real‑world scenario you would
# probably implement proper rate‑limit handling.
# ---------------------------------------------------------------------------
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]

def _get_random_key(key_pool: list[str]) -> str:
    """Return a random non‑None key from *key_pool*.

    LangChain expects a string, so we filter out any ``None`` entries that may
    appear when an environment variable is missing.
    """
    valid_keys = [k for k in key_pool if k]
    if not valid_keys:
        raise RuntimeError("No valid OpenAI API keys found in the pool.")
    return random.choice(valid_keys)


def run_agent(query: str, pdf_paths: list[str], source_filter: str | None = None) -> dict:
    """Run a conversational retrieval chain over a list of PDFs.

    Parameters
    ----------
    query: str
        The user question.
    pdf_paths: list[str]
        Paths to the PDF documents that will be loaded, split and indexed.
    source_filter: str | None, optional
        If provided, only chunks whose ``metadata["source"]`` value matches this
        string will be considered during retrieval.  This eliminates the need
        for creating separate chains for each source.
    """
    # ---------------------------------------------------------------------
    # 1. Load PDFs, split them and attach a ``source`` metadata field.
    # ---------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=_get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()
        splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        chunks = splitter.split_documents(documents)
        for chunk in chunks:
            # Store the originating file name so we can filter later.
            chunk.metadata["source"] = os.path.basename(pdf_path)
        all_chunks.extend(chunks)

    # ---------------------------------------------------------------------
    # 2. Create a Chroma vector store from the chunks.
    # ---------------------------------------------------------------------
    vectorstore = Chroma.from_documents(all_chunks, embeddings)

    # ---------------------------------------------------------------------
    # 3. Build a retriever that optionally applies a metadata filter.
    #    ``search_kwargs`` are passed straight through to the underlying
    #    ``self._vectorstore.similarity_search`` call.
    # ---------------------------------------------------------------------
    if source_filter is not None:
        retriever = vectorstore.as_retriever(search_kwargs={"filter": {"source": source_filter}})
    else:
        retriever = vectorstore.as_retriever()

    # ---------------------------------------------------------------------
    # 4. Initialise the ConversationalRetrievalChain.
    # ---------------------------------------------------------------------
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(
            openai_api_key=_get_random_key(OPENAI_API_KEY_POOL),
            cache=True,
            temperature=0,
        ),
        retriever=retriever,
        return_source_documents=True,
        verbose=True,
    )

    # ---------------------------------------------------------------------
    # 5. Execute the chain.  The chain expects a dict with ``question`` and
    #    ``chat_history`` – any extra keys (like a ``filter`` key) are ignored.
    # ---------------------------------------------------------------------
    response = qa_chain({"question": query, "chat_history": []})
    return response


if __name__ == "__main__":
    # Example usage – only allow answers that come from *paper2.pdf*.
    user_query = "What is Theia?"
    pdf_files = ["paper1.pdf", "paper2.pdf"]
    result = run_agent(user_query, pdf_files, source_filter="paper2.pdf")

    print("Answer:")
    print(result["answer"])
    print("\nSource documents used:")
    for doc in result.get("source_documents", []):
        print(f"- {doc.metadata.get('source')}: {doc.page_content[:200]}…")
