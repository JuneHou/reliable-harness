import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables and initialise LLM cache
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# A small pool of API keys – you can add more keys if you like
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]


def get_random_key(key_pool: list[str]) -> str:
    """Return a random non‑None API key from the supplied pool."""
    return random.choice([k for k in key_pool if k is not None])


def run_agent(query: str, pdf_paths: list[str]):
    """Run a ConversationalRetrievalChain that can be filtered on‑the‑fly.

    The function loads the PDFs, creates a Chroma vector store, and then
    builds a ConversationalRetrievalChain.  A metadata filter is applied *just
    before* the query is executed, allowing the same chain to be reused for
    different filters without needing to rebuild the chain.
    """

    # ---------------------------------------------------------------------
    # 1️⃣ Load PDFs and split them into chunks, attaching the source filename
    #    as metadata so we can filter on it later.
    # ---------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        docs = loader.load()
        splitter = RecursiveCharacterTextSplitter(chunk_size=1_000, chunk_overlap=200)
        chunks = splitter.split_documents(docs)
        for chunk in chunks:
            # Store the originating PDF name in metadata – this is the field we
            # will filter on (e.g. {'source': 'paper2.pdf'}).
            chunk.metadata["source"] = pdf_path
        all_chunks.extend(chunks)

    # ---------------------------------------------------------------------
    # 2️⃣ Create a Chroma vector store from the chunks.
    # ---------------------------------------------------------------------
    db = Chroma.from_documents(all_chunks, embeddings)

    # ---------------------------------------------------------------------
    # 3️⃣ Build a *single* ConversationalRetrievalChain.  The retriever is the
    #    default Chroma retriever; we will inject a filter into its
    #    `search_kwargs` at query time.
    # ---------------------------------------------------------------------
    retriever = db.as_retriever()  # No filter applied yet – it will be added later

    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(openai_api_key=get_random_key(OPENAI_API_KEY_POOL), cache=True, temperature=0),
        retriever=retriever,
        verbose=True,
        return_source_documents=True,
    )

    # ---------------------------------------------------------------------
    # 4️⃣ Define the dynamic filter and *inject* it into the retriever just
    #    before calling the chain.  This keeps the chain reusable for any
    #    filter you like.
    # ---------------------------------------------------------------------
    metadata_filter = {"source": "paper2.pdf"}  # Only use documents from paper2.pdf
    # Update the retriever's search_kwargs – Chroma respects the `filter` key.
    retriever.search_kwargs["filter"] = metadata_filter

    # ---------------------------------------------------------------------
    # 5️⃣ Execute the chain.
    # ---------------------------------------------------------------------
    result = qa_chain({"question": query, "chat_history": []})
    return result


if __name__ == "__main__":
    # Example usage – ask a question and restrict the answer to paper2.pdf.
    query = "What is Theia?"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    answer = run_agent(query, pdf_paths)
    print(answer["answer"]))
