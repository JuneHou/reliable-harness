import os
import random
from dotenv import load_dotenv, find_dotenv
from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.chains import ConversationalRetrievalChain
from langchain.llms import OpenAI
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.cache import InMemoryCache
from langchain.globals import set_llm_cache

# Load environment variables and set up an in‑memory LLM cache
_ = load_dotenv(find_dotenv())
set_llm_cache(InMemoryCache())

# A small pool of API keys – you can extend this as needed
OPENAI_API_KEY_POOL = [os.getenv("OPENAI_API_KEY")]


def get_random_key(key_pool: list[str]) -> str:
    """Return a random non‑null API key from the pool."""
    return random.choice([k for k in key_pool if k])


def run_agent(query: str, pdf_paths: list[str], metadata_filter: dict | None = None):
    """Run a conversational retrieval chain that respects an optional metadata filter.

    Parameters
    ----------
    query: str
        The user question.
    pdf_paths: list[str]
        Paths to the PDFs that will be loaded into the vector store.
    metadata_filter: dict | None, optional
        A Chroma metadata filter (e.g. {"source": "paper2.pdf"}). If ``None`` all
        documents are considered.
    """
    # ---------------------------------------------------------------------
    # 1. Load and split the PDFs, adding a ``source`` field to each chunk's
    #    metadata so we can filter later.
    # ---------------------------------------------------------------------
    all_chunks = []
    embeddings = OpenAIEmbeddings(openai_api_key=get_random_key(OPENAI_API_KEY_POOL))

    for pdf_path in pdf_paths:
        loader = PyPDFLoader(pdf_path)
        documents = loader.load()
        splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
        chunks = splitter.split_documents(documents)
        for chunk in chunks:
            # Ensure every chunk knows which file it came from
            chunk.metadata["source"] = pdf_path
        all_chunks.extend(chunks)

    # ---------------------------------------------------------------------
    # 2. Create the Chroma vector store.
    # ---------------------------------------------------------------------
    db = Chroma.from_documents(all_chunks, embeddings)

    # ---------------------------------------------------------------------
    # 3. Build a retriever.  We start with an empty ``search_kwargs`` dict and
    #    later inject the ``filter`` key dynamically – this avoids having to
    #    rebuild the entire chain for each different filter.
    # ---------------------------------------------------------------------
    retriever = db.as_retriever(search_kwargs={})

    # ---------------------------------------------------------------------
    # 4. Initialise the ConversationalRetrievalChain.
    # ---------------------------------------------------------------------
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=OpenAI(
            openai_api_key=get_random_key(OPENAI_API_KEY_POOL),
            cache=True,
            temperature=0,
        ),
        retriever=retriever,
        verbose=True,
        return_source_documents=True,
    )

    # ---------------------------------------------------------------------
    # 5. Apply the optional metadata filter *just before* the query is executed.
    #    Chroma respects the ``filter`` entry inside ``search_kwargs``.
    # ---------------------------------------------------------------------
    if metadata_filter:
        retriever.search_kwargs["filter"] = metadata_filter
    else:
        # Ensure no stale filter is left from a previous call.
        retriever.search_kwargs.pop("filter", None)

    # ---------------------------------------------------------------------
    # 6. Perform the query.
    # ---------------------------------------------------------------------
    result = qa_chain({"question": query, "chat_history": []})
    return result


if __name__ == "__main__":
    # Example usage – only documents from ``paper2.pdf`` will be consulted.
    query = "What is Theia?"
    pdf_paths = ["paper1.pdf", "paper2.pdf"]
    filter_by_source = {"source": "paper2.pdf"}
    result = run_agent(query, pdf_paths, metadata_filter=filter_by_source)
    print(result["answer"])
