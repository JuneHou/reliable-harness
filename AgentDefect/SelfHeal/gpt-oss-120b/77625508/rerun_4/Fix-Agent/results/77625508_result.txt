from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_verbose
from langchain.callbacks import ConsoleCallbackHandler
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Enable verbose mode globally – this makes LangChain components emit
# detailed logs when a callback manager is attached.
set_verbose(True)

def run_agent(topic: str) -> str:
    """Run a simple LCEL chain that tells a joke about *topic*.

    The `ConsoleCallbackHandler` prints each intermediate step (prompt
    construction, model request/response, and parsing) to stdout, giving
    the user visibility into the chain’s internals.
    """
    # Attach a console callback so that each step is printed.
    callbacks = [ConsoleCallbackHandler()]
    model = ChatOpenAI(openai_api_key=api_key, verbose=True, callbacks=callbacks)
    output_parser = StrOutputParser()
    prompt_template = "tell me a joke about {topic}"
    prompt = ChatPromptTemplate.from_template(prompt_template)

    # Build the LCEL pipeline.
    chain = prompt | model | output_parser

    # Execute the chain; the console callback will emit the verbose logs.
    return chain.invoke({"topic": topic})

if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
