from langchain.chat_models import ChatOpenAI
from langchain.prompts import ChatPromptTemplate
from langchain.schema.output_parser import StrOutputParser
from langchain.globals import set_verbose
from dotenv import load_dotenv
import os

load_dotenv()
api_key = os.getenv("OPENAI_API_KEY")
if not api_key:
    raise ValueError("OPENAI_API_KEY is not set in environment or .env file")

# Enable verbose mode globally – this makes LangChain emit log messages for
# each runnable that respects the flag.
set_verbose(True)

def run_agent(topic: str) -> str:
    # Pass ``verbose=True`` to the model so that the LLM call is logged.
    model = ChatOpenAI(openai_api_key=api_key, verbose=True)
    output_parser = StrOutputParser()
    prompt = ChatPromptTemplate.from_template("tell me a joke about {topic}")

    # Build the LCEL chain and explicitly enable verbose on the runnable.
    # ``with_config`` merges the provided config with the global one.
    chain = (prompt | model | output_parser).with_config({"verbose": True})

    # ``invoke`` will now emit logs for the prompt creation, the LLM request,
    # and the output parsing step.
    return chain.invoke({"topic": topic})

if __name__ == "__main__":
    result = run_agent("ice cream")
    print("Generated output:", result)
